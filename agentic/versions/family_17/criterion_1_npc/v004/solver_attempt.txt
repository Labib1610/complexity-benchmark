import sys

# Increase recursion depth just in case
sys.setrecursionlimit(3000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    M = 3 * q
    
    # Read triples
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triples.append((a, b, c))
        idx += 3
        
    # DLX arrays
    # Max nodes: 1 (root) + M (columns) + 3 * t (entries)
    max_nodes = 1 + M + 3 * t + 5
    
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    sz = [0] * (M + 1)
    
    # Initialize column headers
    for i in range(M + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        C[i] = i
    L[0] = M
    R[M] = 0
    
    node_cnt = M
    
    for r_idx, (a, b, c) in enumerate(triples):
        cols = [a, b, c]
        first = node_cnt + 1
        for col in cols:
            node_cnt += 1
            node = node_cnt
            C[node] = col
            row_id[node] = r_idx
            sz[col] += 1
            
            # Insert into column list
            U[node] = U[col]
            D[node] = col
            D[U[col]] = node
            U[col] = node
            
            # Insert into row list
            if col == cols[0]:
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first]
                R[node] = first
                R[L[first]] = node
                L[first] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with min size
        c = R[0]
        min_sz = sz[c]
        curr = R[c]
        while curr != 0:
            if sz[curr] < min_sz:
                min_sz = sz[curr]
                c = curr
                if min_sz == 0:
                    break
            curr = R[curr]
        
        if min_sz == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            ans.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            
            if search():
                return True
            
            ans.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]
            
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r_idx in ans:
            a, b, c = triples[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
