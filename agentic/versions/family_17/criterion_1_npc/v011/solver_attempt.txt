import sys

# Set recursion depth high enough for q <= 200
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n_artifacts = 3 * q
    
    triplets = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triplets.append((a, b, c))
        idx += 3

    # Map each artifact to the list of triplet indices that contain it
    artifact_to_triplets = [[] for _ in range(n_artifacts + 1)]
    for i, (a, b, c) in enumerate(triplets):
        artifact_to_triplets[a].append(i)
        artifact_to_triplets[b].append(i)
        artifact_to_triplets[c].append(i)

    # State tracking
    artifact_covered = [False] * (n_artifacts + 1)
    triplet_used = [False] * t
    solution = []
    
    # DLX / Backtracking with MRV (Minimum Remaining Values)
    def search(num_uncovered):
        if num_uncovered == 0:
            return True

        # Choose the uncovered artifact with the minimum number of valid triplets
        best_art = -1
        min_options = 10**9
        best_candidates = []

        for art in range(1, n_artifacts + 1):
            if not artifact_covered[art]:
                valid_triplets = []
                for trip_idx in artifact_to_triplets[art]:
                    a, b, c = triplets[trip_idx]
                    if not artifact_covered[a] and not artifact_covered[b] and not artifact_covered[c]:
                        valid_triplets.append(trip_idx)
                
                count = len(valid_triplets)
                if count == 0:
                    return False
                if count < min_options:
                    min_options = count
                    best_art = art
                    best_candidates = valid_triplets
                    if min_options == 1:
                        break

        for trip_idx in best_candidates:
            a, b, c = triplets[trip_idx]
            if artifact_covered[a] or artifact_covered[b] or artifact_covered[c]:
                continue
            
            # Choose this triplet
            artifact_covered[a] = True
            artifact_covered[b] = True
            artifact_covered[c] = True
            solution.append(trip_idx)

            if search(num_uncovered - 3):
                return True

            # Backtrack
            solution.pop()
            artifact_covered[a] = False
            artifact_covered[b] = False
            artifact_covered[c] = False

        return False

    if search(n_artifacts):
        print("YES")
        out = []
        for trip_idx in solution:
            a, b, c = triplets[trip_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
