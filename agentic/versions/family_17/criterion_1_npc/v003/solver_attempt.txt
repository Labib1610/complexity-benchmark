import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    triples = []
    idx = 2
    
    # Store triples
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triples.append((a, b, c))
        idx += 3
        
    # Element to triples mapping
    elem_to_triples = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triples):
        elem_to_triples[a].append(i)
        elem_to_triples[b].append(i)
        elem_to_triples[c].append(i)
        
    covered = [False] * (n + 1)
    uncovered = set(range(1, n + 1))
    solution = []
    
    def search():
        if not uncovered:
            return True
        
        # MRV: find uncovered element with minimum number of valid candidate triples
        best_elem = -1
        best_candidates = None
        min_candidates = 1000000
        
        for u in uncovered:
            cands = []
            for tid in elem_to_triples[u]:
                a, b, c = triples[tid]
                if not covered[a] and not covered[b] and not covered[c]:
                    cands.append(tid)
            
            cand_count = len(cands)
            if cand_count < min_candidates:
                min_candidates = cand_count
                best_candidates = cands
                best_elem = u
                if min_candidates <= 1:
                    break
        
        if min_candidates == 0 or best_candidates is None:
            return False
            
        for tid in best_candidates:
            a, b, c = triples[tid]
            # Choose this triple
            covered[a] = covered[b] = covered[c] = True
            uncovered.remove(a)
            uncovered.remove(b)
            uncovered.remove(c)
            solution.append(tid)
            
            if search():
                return True
                
            solution.pop()
            covered[a] = covered[b] = covered[c] = False
            uncovered.add(a)
            uncovered.add(b)
            uncovered.add(c)
            
        return False

    if search():
        print("YES")
        for tid in solution:
            a, b, c = triples[tid]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
