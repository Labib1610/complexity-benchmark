import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    idx = 2
    triples = []
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        if 1 <= a <= n and 1 <= b <= n and 1 <= c <= n:
            triples.append((a, b, c))
            
    # Quick check
    if len(triples) < q:
        print("NO")
        return

    # Element to triples map
    elem_to_triples = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triples):
        elem_to_triples[a].append(i)
        elem_to_triples[b].append(i)
        elem_to_triples[c].append(i)
        
    used_elem = [False] * (n + 1)
    used_triple = [False] * len(triples)
    solution = []
    
    # Prune impossible elements initially
    for i in range(1, n + 1):
        if not elem_to_triples[i]:
            print("NO")
            return

    # Backtracking with MRV heuristic
    def search(remaining_elements):
        if remaining_elements == 0:
            return True
            
        # Find element with minimum remaining valid triples
        best_elem = -1
        min_candidates = float('inf')
        best_cands = []
        
        for e in range(1, n + 1):
            if used_elem[e]:
                continue
            
            cands = []
            for tr_idx in elem_to_triples[e]:
                a, b, c = triples[tr_idx]
                if not used_elem[a] and not used_elem[b] and not used_elem[c]:
                    cands.append(tr_idx)
            
            cand_count = len(cands)
            if cand_count == 0:
                return False
            
            if cand_count < min_candidates:
                min_candidates = cand_count
                best_elem = e
                best_cands = cands
                if min_candidates == 1:
                    break
                    
        for tr_idx in best_cands:
            a, b, c = triples[tr_idx]
            if used_elem[a] or used_elem[b] or used_elem[c]:
                continue
                
            used_elem[a] = used_elem[b] = used_elem[c] = True
            solution.append(tr_idx)
            
            if search(remaining_elements - 3):
                return True
                
            solution.pop()
            used_elem[a] = used_elem[b] = used_elem[c] = False
            
        return False

    sys.setrecursionlimit(2000)
    if search(n):
        print("YES")
        for tr_idx in solution:
            a, b, c = triples[tr_idx]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
