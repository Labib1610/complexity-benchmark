```python
import sys

# The problem states the cabling options form a tree (n nodes, n-1 edges, 
# exactly one path between any two nodes).
# To connect a subset of nodes in a tree with minimum cost, we need to find 
# the Steiner Tree, which in a tree is simply the union of all paths 
# between the required nodes. This is equivalent to the sum of weights of 
# all edges that lie on at least one path between two required nodes.
# An edge lies on a path between two required nodes if and only if 
# removing that edge splits the tree into two components, each containing 
# at least one required node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many required nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_required = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in required:
        subtree_required[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        # If this subtree contains at least one required node and 
        # the rest of the tree contains at least one required node,
        # the edge to the parent must be included.
        if 0 < subtree_required[u] < t:
            total_cost += edge_to_parent_weight[u]
            
        subtree_required[parent[u]] += subtree_required[u]
        
    print(total_cost)

solve()
```
