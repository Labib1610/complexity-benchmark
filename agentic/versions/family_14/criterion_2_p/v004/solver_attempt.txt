```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, unique path between any two).
# To connect a subset of required nodes in a tree, the minimum cost is the sum of the weights
# of all edges that lie on the paths between any two required nodes.
# An edge is part of such a path if and only if the subtree formed by removing that edge
# contains at least one required node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set()
    idx = 3
    for _ in range(t):
        required.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    # count[u] will store the number of required nodes in the subtree of u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in required:
        count[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if count[u] > 0 and count[u] < t:
            total_cost += edge_weight[u]
        count[p] += count[u]
        
    print(total_cost)

solve()
```
