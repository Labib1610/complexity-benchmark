```python
import sys

# The problem describes a tree structure (n nodes, n-1 edges, unique path between any two nodes).
# To connect a subset of t priority facilities, we need to find the Steiner Tree in the tree.
# In a tree, the Steiner Tree is simply the union of all paths between the priority facilities.
# This is equivalent to the sum of weights of all edges that lie on at least one path 
# between two priority facilities.
# An edge (u, v) with weight w lies on a path between two priority facilities if and only if
# the subtree formed by removing (u, v) contains at least one priority facility on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = set()
    idx = 3
    for _ in range(t):
        priority_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # count[u] will store the number of priority nodes in the subtree rooted at u
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, edge_weight_to_parent
    order = []
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    
    visit_stack = [1]
    while visit_stack:
        u = visit_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                edge_to_parent[v] = w
                visit_stack.append(v)
                
    for u in reversed(order):
        if u in priority_nodes:
            count[u] += 1
        for v, w in adj[u]:
            if v != parent[u]:
                count[u] += count[v]
                # If the edge (u, v) separates the tree such that both sides have 
                # at least one priority node, it must be part of the Steiner Tree.
                if count[v] > 0 and count[v] < t:
                    total_cost += w
                    
    print(total_cost)

solve()
```
