```python
import sys

# The problem states the network is a single-path network (a tree).
# To connect a set of required nodes in a tree, we need to include all edges
# that lie on the unique path between any two required nodes.
# This is equivalent to the sum of weights of all edges that have at least one
# required node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set()
    idx = 3
    for _ in range(t):
        required.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many required nodes are in the subtree of each edge.
    # Since it's a tree, we can use DFS.
    # count[u] will store the number of required nodes in the subtree rooted at u.
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in required:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If an edge splits the required nodes into two non-empty sets,
            # it must be included.
            # The number of required nodes in the subtree is count[u].
            # The number of required nodes outside is t - count[u].
            if count[u] > 0 and count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
