```python
import sys

# The problem states the grid is a tree (n nodes, n-1 links, connected).
# To connect a subset of nodes in a tree, we need to select all edges that lie
# on the unique paths between any two nodes in the active set.
# This is equivalent to the Steiner Tree problem on a tree, which is simply
# the sum of weights of all edges that have at least one active node in each
# of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    active_nodes = set()
    idx = 3
    for _ in range(t):
        active_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_active_count[u] stores how many active nodes are in the subtree rooted at u
    subtree_active_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in active_nodes:
            subtree_active_count[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_active_count[p] += subtree_active_count[u]
        # If the edge (u, p) separates the tree into two parts, both containing
        # at least one active node, then this edge must be included.
        if subtree_active_count[u] > 0 and subtree_active_count[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

solve()
```
