```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes (spore-containing junctions)
# in a tree. This is equivalent to finding the weight of the Steiner Tree for the given
# set of terminals. In a tree, the Steiner Tree is simply the union of all paths between
# any two terminals. This is equivalent to the sum of weights of all edges that lie on
# at least one path between two terminals.
# An edge (u, v) with weight w lies on a path between two terminals if and only if
# the subtree formed by removing the edge contains at least one terminal on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    spores = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    # count[u] will store the number of spores in the subtree rooted at u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in spores:
        count[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        if count[u] > 0 and count[u] < t:
            total_cost += edge_weight[u]
        count[p] += count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
