```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    contaminated = set()
    idx = 3
    for _ in range(t):
        contaminated.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the sum of weights of all edges that lie on the path
    # between any two contaminated vats.
    # An edge is part of the minimal subtree connecting all contaminated vats
    # if and only if the subtree on one side of the edge contains at least
    # one contaminated vat and the subtree on the other side also contains
    # at least one contaminated vat.
    
    is_contaminated = [False] * (n + 1)
    for v in contaminated:
        is_contaminated[v] = True
        
    subtree_contaminated_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        subtree_contaminated_count[u] += (1 if is_contaminated[u] else 0)
        if parent[u] != 0:
            subtree_contaminated_count[parent[u]] += subtree_contaminated_count[u]
            # If this edge separates the tree into two parts, both having 
            # at least one contaminated vat, it must be included.
            count = subtree_contaminated_count[u]
            if count > 0 and count < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

solve()
```
