```python
import sys

# The problem states that the fragments and possible fusions form a tree (n nodes, n-1 edges).
# We need to find the minimum weight of a subtree that contains all t target nodes.
# This is equivalent to finding the sum of weights of all edges that lie on the paths
# between any two target nodes.
# An edge in a tree is part of the path between two target nodes if and only if
# the edge splits the tree into two components, each containing at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many target nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    target_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in targets:
            target_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_resin = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        # If the subtree at u contains some targets but not all,
        # the edge (u, p) must be included.
        if 0 < target_count[u] < t:
            total_resin += edge_to_parent_weight[u]
        target_count[p] += target_count[u]
        
    print(total_resin)

if __name__ == '__main__':
    solve()
```
