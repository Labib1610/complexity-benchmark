```python
import sys

# The problem asks for the minimum cost to connect all freight hubs.
# Since the graph is a tree, the set of edges that connect all freight hubs
# is exactly the set of edges that lie on the unique paths between any two freight hubs.
# This is equivalent to the union of all paths between all pairs of freight hubs.
# In a tree, this is the minimal subtree containing all freight hubs.
# An edge (u, v) with weight w is part of this subtree if and only if
# the subtree rooted at u (when the edge is removed) contains at least one freight hub
# AND the subtree rooted at v (when the edge is removed) contains at least one freight hub.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    hubs = set()
    idx = 3
    for _ in range(t):
        hubs.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use DFS to count hubs in subtrees
    # To avoid recursion depth issues, use iterative DFS
    hub_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in hubs:
        hub_count[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if hub_count[u] > 0 and hub_count[u] < t:
            total_cost += edge_to_parent_weight[u]
        hub_count[p] += hub_count[u]
        
    print(total_cost)

solve()
```
