import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    flagged = set()
    idx = 3
    for _ in range(t):
        flagged.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need the sum of weights of all edges that lie on the paths between any two flagged nodes.
    # An edge is part of the Steiner Tree if and only if the subtree formed by removing the edge
    # contains at least one flagged node on both sides.
    
    is_flagged = [False] * (n + 1)
    for f in flagged:
        is_flagged[f] = True
        
    subtree_flagged_count = [0] * (n + 1)
    total_latency = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_flagged[u]:
            subtree_flagged_count[u] += 1
        
        if parent[u] != 0:
            subtree_flagged_count[parent[u]] += subtree_flagged_count[u]
            # If this edge separates the tree into two parts, both containing at least one flagged node
            if 0 < subtree_flagged_count[u] < t:
                total_latency += edge_to_parent_weight[u]
                
    print(total_latency)

solve()
