```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# This is equivalent to finding the weight of the Steiner Tree in a tree graph,
# which is simply the sum of weights of all edges that lie on the paths between
# any two project offices.
# An edge is part of the Steiner Tree if and only if the subtree formed by
# removing that edge contains at least one project office on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    project_offices = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of project offices in the subtree rooted at u
    count = [0] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0, 0)] # node, parent, weight_to_parent, state
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    dfs_stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                weight_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in order:
        if u in project_offices:
            count[u] = 1
            
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        if count[u] > 0:
            # If this subtree contains at least one project office,
            # and the rest of the tree contains at least one (t - count[u] > 0),
            # then the edge to the parent must be included.
            if t - count[u] > 0:
                total_weight += weight_to_parent[u]
            count[p] += count[u]
            
    print(total_weight)

solve()
```
