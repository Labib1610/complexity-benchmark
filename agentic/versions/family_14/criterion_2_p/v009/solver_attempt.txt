```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# This is equivalent to finding the sum of weights of all edges that lie on the 
# unique path between any two required nodes.
# An edge is part of the minimal subtree connecting all required nodes if and 
# only if the edge splits the tree into two components, each containing at least 
# one required node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set()
    idx = 3
    for _ in range(t):
        required.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many required nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in required:
            count[u] += 1
        if parent[u] != -1:
            count[parent[u]] += count[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge between i and parent[i] separates the tree such that
        # both sides have at least one required node, it must be included.
        # count[i] is the number of required nodes in the subtree rooted at i.
        if count[i] > 0 and count[i] < t:
            total_cost += edge_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
