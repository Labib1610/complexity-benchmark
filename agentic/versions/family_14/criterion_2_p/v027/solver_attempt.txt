```python
import sys

# The problem describes a tree structure (n caverns, n-1 tunnels, unique path between any two).
# We need to find the minimum weight of a subtree that connects all t target nodes.
# This is equivalent to the sum of weights of all edges that lie on the paths between any two target nodes.
# An edge (u, v) with weight w is part of the required set if and only if the subtree 
# formed by removing (u, v) has target nodes in both components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set()
    idx = 3
    for _ in range(t):
        targets.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use iterative DFS to avoid recursion depth issues
    # count[u] stores the number of target nodes in the subtree rooted at u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            count[u] = 1
            
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        count[p] += count[u]
        # If the subtree at u contains some but not all targets, 
        # the edge to its parent must be included.
        if count[u] > 0 and count[u] < t:
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

solve()
```
