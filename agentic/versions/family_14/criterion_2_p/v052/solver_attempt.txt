```python
import sys

# The problem states that the network is a tree (n rooms, n-1 segments, connected).
# We need to find the minimum cost to connect a subset of t key rooms.
# This is equivalent to finding the weight of the Steiner Tree in a tree,
# which is simply the sum of weights of all edges that lie on the unique path
# between any two key rooms.
# An edge belongs to the Steiner Tree if and only if the subtree formed by
# removing that edge contains at least one key room on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    key_rooms = set()
    idx = 3
    for _ in range(t):
        key_rooms.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many key rooms are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_keys = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in key_rooms:
            subtree_keys[node] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        subtree_keys[p] += subtree_keys[u]
        
        # If this edge separates some key rooms from others, it must be included
        if 0 < subtree_keys[u] < t:
            total_cost += edge_to_parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
