import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_centers = set()
    idx = 3
    for _ in range(t):
        primary_centers.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need the sum of weights of all edges that lie on the path between any two primary centers.
    # This is equivalent to the sum of weights of all edges that have at least one primary center
    # in both components formed by removing that edge.
    
    is_primary = [False] * (n + 1)
    for p in primary_centers:
        is_primary[p] = True
        
    subtree_primary_count = [0] * (n + 1)
    total_energy = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0, 0)] # node, parent, weight_to_parent, state
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    dfs_stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                weight_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in reversed(order):
        count = 1 if is_primary[u] else 0
        for v, w in adj[u]:
            if v != parent[u]:
                count += subtree_primary_count[v]
        subtree_primary_count[u] = count
        
    for i in range(2, n + 1):
        # Edge between i and parent[i]
        # Number of primary centers in subtree of i
        c = subtree_primary_count[i]
        if c > 0 and c < t:
            total_energy += weight_to_parent[i]
            
    print(total_energy)

solve()
