```python
import sys

# The problem states that the network is a tree (n facilities, n-1 links, 
# unique path between any two). To connect a subset of priority facilities,
# we must include all edges that lie on the paths between any two priority 
# facilities. This is equivalent to finding the minimal subtree that spans 
# all priority facilities.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that are part of the 
    # minimal subtree connecting all priority nodes.
    # An edge (u, v) with weight w is in the minimal subtree if and only if
    # there is at least one priority node in the component containing u 
    # (when the edge is removed) and at least one priority node in the 
    # component containing v.
    
    # Root the tree at an arbitrary priority node (e.g., the first one)
    root = next(iter(priority_nodes))
    
    # count_priority[u] = number of priority nodes in the subtree rooted at u
    count_priority = [0] * (n + 1)
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    stack = [(root, -1)]
    order = []
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != p:
                parent[v] = u
                edge_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u in priority_nodes:
            count_priority[u] += 1
        if parent[u] != 0:
            count_priority[parent[u]] += count_priority[u]
            
    # An edge connecting u to its parent is part of the subtree if 
    # 0 < count_priority[u] < t
    for u in range(1, n + 1):
        if u != root and 0 < count_priority[u] < t:
            total_cost += edge_weight[u]
            
    print(total_cost)

solve()
```
