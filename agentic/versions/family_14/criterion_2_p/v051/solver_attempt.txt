import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need the sum of weights of the minimal subtree connecting all target nodes.
    # This is equivalent to the sum of weights of all edges that lie on at least 
    # one path between two target nodes.
    # An edge (u, v) with weight w is in the minimal subtree if and only if 
    # there is at least one target node in the subtree of u (when rooted at v)
    # and at least one target node in the subtree of v (when rooted at u).
    
    is_target = [False] * (n + 1)
    for node in targets:
        is_target[node] = True
        
    target_count_subtree = [0] * (n + 1)
    total_energy = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_target[u]:
            target_count_subtree[u] += 1
        if parent[u] != 0:
            target_count_subtree[parent[u]] += target_count_subtree[u]
            # If this edge separates the tree into two parts both containing targets
            if 0 < target_count_subtree[u] < t:
                total_energy += edge_to_parent_weight[u]
                
    print(total_energy)

solve()
