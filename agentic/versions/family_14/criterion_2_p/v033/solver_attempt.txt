```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, connected).
# To connect a subset of nodes in a tree, we must include all edges that lie
# on the unique path between any two required nodes.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the sum of weights of all edges that have at least one required node in
# each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        required[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Post-order traversal to count required nodes in subtrees
    subtree_required = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    for u in reversed(order):
        subtree_required[u] += (1 if required[u] else 0)
        if parent[u] != -1:
            p = parent[u]
            if subtree_required[u] > 0 and subtree_required[u] < t:
                total_cost += edge_to_parent_weight[u]
            subtree_required[p] += subtree_required[u]
            
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth for deep trees, though iterative approach is used
    sys.setrecursionlimit(300000)
    solve()
```
