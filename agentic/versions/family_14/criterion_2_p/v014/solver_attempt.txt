```python
import sys

# The problem asks for the minimum weight of a subtree that connects all command centers.
# In a tree, the unique path connecting all command centers is the union of all paths
# between any two command centers. This is equivalent to the minimal subtree
# containing all command centers.
# We can solve this by pruning the tree: any edge that is not part of a path
# between two command centers can be removed.
# An edge (u, v) with weight w is part of the minimal subtree if and only if
# the subtree rooted at u (when the edge is cut) contains at least one command center,
# and the subtree rooted at v contains at least one command center.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    command_centers = set()
    idx = 3
    for _ in range(t):
        command_centers.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # has_cc[u] will be true if the subtree rooted at u contains a command center
    has_cc = [False] * (n + 1)
    for node in command_centers:
        has_cc[node] = True
        
    total_weight = 0
    
    # Use post-order traversal to determine if an edge should be included
    # We can use an iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u == 1:
            continue
        if has_cc[u]:
            total_weight += edge_to_parent_weight[u]
            has_cc[parent[u]] = True
            
    print(total_weight)

if __name__ == '__main__':
    solve()
```
