```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, exactly one path between any two nodes).
# To connect a subset of t primary nodes, we need to activate all edges that lie on the paths 
# between any two primary nodes. This is equivalent to finding the Steiner Tree in a tree,
# which is simply the union of all paths between the primary nodes.
# In a tree, this is the minimal subtree containing all primary nodes.
# The total cost is the sum of weights of all edges that have at least one primary node 
# in one of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    primary_nodes = set()
    idx = 3
    for _ in range(t):
        primary_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count[u] will store the number of primary nodes in the subtree rooted at u
    count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in primary_nodes:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If the edge to parent separates at least one primary node from another,
            # it must be included.
            # This happens if count[u] > 0 and count[u] < t.
            if 0 < count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
