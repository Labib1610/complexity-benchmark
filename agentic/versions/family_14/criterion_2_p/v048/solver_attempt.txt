import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_relays = set()
    idx = 3
    for _ in range(t):
        priority_relays.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need the sum of weights of the minimal subtree containing all priority nodes.
    # This is equivalent to the sum of weights of all edges that lie on a path 
    # between any two priority nodes.
    # An edge (u, v) with weight w is in the minimal subtree if and only if 
    # there is at least one priority node in the subtree of u (when rooted at v)
    # and at least one priority node in the subtree of v (when rooted at u).
    
    has_priority = [False] * (n + 1)
    for p in priority_relays:
        has_priority[p] = True
        
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # subtree_has_priority[u] stores if the subtree rooted at u contains a priority node
    subtree_has_priority = [False] * (n + 1)
    for i in range(1, n + 1):
        if has_priority[i]:
            subtree_has_priority[i] = True
            
    # Process in reverse topological order
    edge_weight = {}
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            # Find weight of edge (p, u)
            w = 0
            for v, weight in adj[p]:
                if v == u:
                    w = weight
                    break
            
            if subtree_has_priority[u]:
                # Check if there's a priority node outside this subtree
                # Total priority nodes is t. If subtree_has_priority[u] is true,
                # and count of priority nodes in subtree is < t, then this edge is needed.
                # Actually, simpler: an edge is needed if the subtree contains 
                # at least one priority node AND the rest of the tree contains at least one.
                
                # Count priority nodes in subtree
                pass
    
    # Let's use a simpler approach:
    # count[u] = number of priority nodes in subtree of u
    count = [0] * (n + 1)
    for i in range(1, n + 1):
        if has_priority[i]:
            count[i] = 1
            
    for u in reversed(order):
        if parent[u] != -1:
            count[parent[u]] += count[u]
            
    ans = 0
    for u in range(2, n + 1):
        # Edge is between u and parent[u]
        # It is part of the minimal subtree if 0 < count[u] < t
        if 0 < count[u] < t:
            # Find weight
            p = parent[u]
            for v, w in adj[p]:
                if v == u:
                    ans += w
                    break
                    
    print(ans)

solve()
