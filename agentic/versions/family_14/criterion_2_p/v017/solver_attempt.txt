```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# This is equivalent to finding the weight of the Steiner Tree for the given subset.
# In a tree, the Steiner Tree is simply the union of all paths between the 
# high-priority nodes. This is equivalent to the sum of weights of all edges 
# that lie on at least one path between two high-priority nodes.
# An edge (u, v) with weight w lies on a path between two high-priority nodes 
# if and only if the subtree formed by removing the edge contains at least 
# one high-priority node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    high_priority = set()
    idx = 3
    for _ in range(t):
        high_priority.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many high-priority nodes are in the subtree of each node
    # when rooted at 1.
    subtree_hp = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in high_priority:
            subtree_hp[u] = 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        
        # If the subtree at u contains some but not all high-priority nodes,
        # the edge to its parent must be part of the Steiner Tree.
        if 0 < subtree_hp[u] < t:
            total_cost += edge_to_parent_weight[u]
            
        subtree_hp[parent[u]] += subtree_hp[u]
        
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
