```python
import sys

# The problem states that the pods and tubes form a tree (n nodes, n-1 edges, 
# exactly one path between any two nodes). We need to find the minimum cost 
# to connect all critical pods. This is equivalent to finding the sum of 
# weights of all edges that lie on the paths between any two critical pods.
# An edge is part of the Steiner Tree if and only if it separates the set of 
# critical pods into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_pods = set()
    idx = 3
    for _ in range(t):
        critical_pods.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use DFS to count how many critical nodes are in each subtree
    # An edge (u, v) where v is child of u is in the Steiner tree if
    # the number of critical nodes in v's subtree is > 0 and < t.
    
    subtree_critical_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, edge_weight_to_parent
    order = []
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    
    dfs_stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in reversed(order):
        if u in critical_pods:
            subtree_critical_count[u] += 1
        if parent[u] != 0:
            subtree_critical_count[parent[u]] += subtree_critical_count[u]
            
    for i in range(2, n + 1):
        count = subtree_critical_count[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent[i]
            
    print(total_cost)

solve()
```
