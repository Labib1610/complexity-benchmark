```python
import sys

# The problem states m = n - 1 and the graph is connected.
# This means the graph is already a tree.
# The requirements are:
# 1. No closed loops (already satisfied as it's a tree).
# 2. Number of trenches < n (already satisfied as m = n - 1).
# 3. All critical facilities must be connected.
# Since the graph is a tree, there is a unique path between any two nodes.
# To connect all critical facilities, we must include all edges that lie on the
# unique paths between any pair of critical facilities.
# This is equivalent to finding the smallest subtree that contains all critical nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Root the tree at the first critical node
    root = next(iter(critical_nodes))
    
    # We need to identify edges that are part of the minimal subtree.
    # An edge (u, v) is in the subtree if and only if the subtree rooted at the
    # child node contains at least one critical node and the rest of the tree
    # (outside that subtree) also contains at least one critical node.
    
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    has_critical = [False] * (n + 1)
    for node in critical_nodes:
        has_critical[node] = True
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        if has_critical[u]:
            total_cost += edge_to_parent_weight[u]
            has_critical[parent[u]] = True
            
    print(total_cost)

solve()
```
