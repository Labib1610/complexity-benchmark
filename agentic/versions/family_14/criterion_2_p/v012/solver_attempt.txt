```python
import sys

# The problem states that there are n facilities and m = n - 1 links,
# and it is possible to travel between any two facilities.
# This means the graph is a tree.
# We need to connect all priority centers. In a tree, there is a unique
# path between any two nodes. To connect a set of nodes, we must activate
# all edges that lie on the paths between any two priority centers.
# These edges are exactly the edges that belong to the minimal subtree
# containing all priority centers.
# An edge (u, v) with weight w is part of this subtree if and only if
# the subtree formed by removing (u, v) splits the priority centers into
# two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_centers = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_p[u] will store the number of priority centers in the subtree of u
    count_p = [0] * (n + 1)
    
    # DFS to calculate count_p
    # Use iterative DFS to avoid recursion depth issues
    stack = [(1, 0)]
    order = []
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    
    while stack:
        u, p = stack.pop()
        order.append(u)
        parent[u] = p
        for v, w in adj[u]:
            if v != p:
                edge_weight[v] = w
                stack.append((v, u))
                
    for u in reversed(order):
        if u in priority_centers:
            count_p[u] += 1
        if parent[u] != 0:
            count_p[parent[u]] += count_p[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If the edge between i and parent[i] separates the priority centers
        # such that both sides have at least one priority center,
        # then this edge must be included.
        c = count_p[i]
        if c > 0 and c < t:
            total_cost += edge_weight[i]
            
    print(total_cost)

solve()
```
