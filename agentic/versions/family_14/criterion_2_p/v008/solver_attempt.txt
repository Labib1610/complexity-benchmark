import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set()
    idx = 3
    for _ in range(t):
        required.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need the sum of weights of all edges that lie on the path between any two required nodes.
    # An edge is part of the Steiner Tree if and only if the subtree below it (when rooted)
    # contains at least one required node AND the rest of the tree contains at least one required node.
    
    is_required = [False] * (n + 1)
    for r in required:
        is_required[r] = True
        
    subtree_has_required = [False] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if is_required[u]:
            subtree_has_required[u] = True
        
        if subtree_has_required[u] and parent[u] != 0:
            total_cost += edge_to_parent_weight[u]
            subtree_has_required[parent[u]] = True
            
    print(total_cost)

solve()
