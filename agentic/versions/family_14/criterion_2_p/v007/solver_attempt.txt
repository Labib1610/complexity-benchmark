```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes in a tree.
# This is equivalent to finding the weight of the Steiner Tree in a tree graph.
# In a tree, the Steiner Tree is simply the union of all paths between the 
# required nodes. This is equivalent to the sum of weights of all edges that 
# have at least one required node in each of the two components formed by 
# removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set()
    idx = 3
    for _ in range(t):
        required.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many required nodes are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_required = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in required:
            subtree_required[u] += 1
        if parent[u] != 0:
            subtree_required[parent[u]] += subtree_required[u]
            
    total_cost = 0
    for i in range(2, n + 1):
        # If an edge splits the tree into two parts, and both parts contain
        # at least one required node, the edge must be part of the Steiner Tree.
        count = subtree_required[i]
        if count > 0 and count < t:
            total_cost += edge_to_parent_weight[i]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
