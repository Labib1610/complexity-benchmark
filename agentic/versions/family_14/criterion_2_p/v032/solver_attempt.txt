```python
import sys

# The problem states that there are n modules and m = n - 1 links,
# and if all links are activated, there is exactly one path between any two modules.
# This means the graph is a tree.
# We need to find the minimum cost to connect all t critical modules.
# In a tree, the set of edges that connect a subset of nodes is the union of all
# paths between every pair of critical nodes.
# This is equivalent to the edges in the minimal subtree containing all critical nodes.
# An edge is part of this subtree if and only if the subtree formed by removing
# that edge splits the critical nodes into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to count how many critical nodes are in the subtree of each node.
    # Since it's a tree, we can root it at an arbitrary node (e.g., 1).
    # For each edge (u, v) where v is a child of u, the edge is part of the
    # minimal subtree if the number of critical nodes in v's subtree is > 0
    # and < t.
    
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    critical_count = [0] * (n + 1)
    for node in critical_nodes:
        critical_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            if 0 < critical_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            critical_count[p] += critical_count[u]
            
    print(total_cost)

solve()
```
