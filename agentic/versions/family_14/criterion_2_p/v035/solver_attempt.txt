```python
import sys

# The problem states that the m available cabling options form a tree (n-1 edges, 
# connected). To connect a subset of t required sites, we must include all edges 
# that lie on the unique paths between any two required sites.
# This is equivalent to finding the Steiner Tree in a tree, which is simply 
# the union of all paths between the required nodes.
# In a tree, this is the minimal subtree containing all required nodes.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    required = set()
    idx = 3
    for _ in range(t):
        required.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find all edges that are part of a path between any two required nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # the subtree at u (when the edge is removed) contains at least one required node
    # AND the subtree at v contains at least one required node.
    
    required_count = [0] * (n + 1)
    for node in required:
        required_count[node] = 1
        
    total_required = t
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to count required nodes in subtrees
    edge_weights = {}
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            # Find weight of edge (u, p)
            w = 0
            for v, weight in adj[u]:
                if v == p:
                    w = weight
                    break
            
            if required_count[u] > 0 and required_count[u] < total_required:
                total_cost += w
            
            required_count[p] += required_count[u]
            
    print(total_cost)

solve()
```
