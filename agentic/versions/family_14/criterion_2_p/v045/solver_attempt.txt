```python
import sys

# The problem states the network is a tree (n depots, n-1 corridors).
# We need to find the minimum cost to connect all primary distribution centers.
# In a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must include all edges that lie on
# the paths between any two primary distribution centers.
# This is equivalent to the union of all paths between the primary centers.
# An edge is part of this union if and only if the subtree formed by removing
# the edge contains at least one primary center on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    if k <= 1:
        print(0)
        return
    
    primary_centers = set(map(int, input_data[3:3+k]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + k
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many primary centers are in the subtree of each node
    # We root the tree at an arbitrary node, say 1.
    subtree_primary_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in primary_centers:
            subtree_primary_count[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        
        p = parent[u]
        subtree_primary_count[p] += subtree_primary_count[u]
        
        # If the subtree at u contains some primary centers, but not all,
        # then the edge (u, parent[u]) must be included.
        count = subtree_primary_count[u]
        if count > 0 and count < k:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

solve()
```
