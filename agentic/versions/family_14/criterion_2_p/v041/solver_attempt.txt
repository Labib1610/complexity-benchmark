import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    mandatory = set()
    idx = 3
    for _ in range(t):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the smallest subtree containing all mandatory nodes.
    # This is equivalent to the sum of weights of all edges that lie on a path
    # between any two mandatory nodes.
    # An edge is part of the Steiner Tree if and only if the subtree formed by
    # removing that edge has mandatory nodes in both components.
    
    is_mandatory = [False] * (n + 1)
    for node in mandatory:
        is_mandatory[node] = True
        
    subtree_mandatory_count = [0] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process nodes in reverse topological order
    for u in reversed(order):
        if is_mandatory[u]:
            subtree_mandatory_count[u] += 1
        
        for v, w in adj[u]:
            if v == parent[u]:
                # This is the edge to the parent
                if subtree_mandatory_count[u] > 0 and subtree_mandatory_count[u] < t:
                    total_cost += w
            else:
                subtree_mandatory_count[u] += subtree_mandatory_count[v]
                
    print(total_cost)

solve()
