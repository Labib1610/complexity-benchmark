import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    required_sites = set()
    idx = 3
    for _ in range(t):
        required_sites.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to find the sum of weights of all edges that lie on the path
    # between any two required nodes. An edge is part of the solution if and
    # only if the subtree formed by removing that edge contains at least one
    # required node on both sides.
    
    is_required = [False] * (n + 1)
    for s in required_sites:
        is_required[s] = True
        
    total_cost = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_has_required = [False] * (n + 1)
    for i in range(1, n + 1):
        if is_required[i]:
            subtree_has_required[i] = True
            
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        if subtree_has_required[u]:
            total_cost += edge_to_parent_weight[u]
            subtree_has_required[p] = True
            
    print(total_cost)

solve()
