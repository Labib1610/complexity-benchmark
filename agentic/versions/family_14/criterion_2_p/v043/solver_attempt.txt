```python
import sys

# The problem states that the graph is a tree (n nodes, n-1 edges, connected).
# We need to find the minimum cost to connect a subset of t critical nodes.
# In a tree, the minimum subgraph connecting a set of nodes is the union of all
# simple paths between them. This is equivalent to the Steiner Tree problem,
# but since the graph is a tree, the Steiner Tree is unique: it consists of
# all edges that lie on the path between at least two critical nodes.
# An edge (u, v) with weight w is part of the Steiner Tree if and only if
# the subtree formed by removing the edge contains at least one critical node
# on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to count how many critical nodes are in each subtree.
    # Since n is up to 200,000, we use iterative DFS to avoid recursion depth issues.
    subtree_critical_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in critical_nodes:
            subtree_critical_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        count = subtree_critical_count[u]
        # If this subtree contains some but not all critical nodes,
        # the edge to the parent must be included.
        if 0 < count < t:
            total_cost += edge_weight_to_parent[u]
        
        subtree_critical_count[p] += count
        
    print(total_cost)

solve()
```
