```python
import sys

# The problem asks for the minimum weight of edges to connect all t reactive agents.
# Since the graph is a tree (n nodes, n-1 edges, connected), the set of edges
# required to connect a subset of nodes is the union of all paths between them.
# This is equivalent to the sum of weights of all edges that have at least one
# reactive agent in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    agents = set()
    idx = 3
    for _ in range(t):
        agents.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_agents[u] stores the number of reactive agents in the subtree rooted at u
    subtree_agents = [0] * (n + 1)
    total_energy = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u in agents:
            subtree_agents[u] += 1
        
        for v, w in adj[u]:
            if v == parent[u]:
                # This is the edge connecting u to its parent
                # If this edge separates some agents from others, it must be included
                count = subtree_agents[u]
                if count > 0 and count < t:
                    total_energy += w
            else:
                subtree_agents[u] += subtree_agents[v]
                
    print(total_energy)

solve()
```
