```python
import sys

# The problem asks for the minimum weight of a subtree that connects all t mural chambers.
# Since the graph is a tree (n nodes, n-1 edges, connected), the set of edges 
# required to connect a subset of nodes is exactly the set of edges on the 
# unique paths between all pairs of these nodes.
# This is equivalent to the sum of weights of all edges that have at least one 
# mural chamber in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    murals = set()
    idx = 3
    for _ in range(t):
        murals.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need to count how many mural chambers are in the subtree of each node.
    # Let count[u] be the number of mural chambers in the subtree rooted at u.
    # An edge (u, v) where v is a child of u is part of the Steiner tree if 
    # 0 < count[v] < t.
    
    count = [0] * (n + 1)
    total_weight = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in murals:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            
    for i in range(2, n + 1):
        # If the edge to parent separates the tree such that both sides have murals
        if count[i] > 0 and count[i] < t:
            total_weight += edge_to_parent_weight[i]
            
    print(total_weight)

solve()
```
