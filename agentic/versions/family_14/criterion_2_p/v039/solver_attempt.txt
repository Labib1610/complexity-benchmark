```python
import sys

# The problem states the graph is a tree (n nodes, n-1 edges, unique path).
# To connect all critical data centers, we must include all edges that lie on
# the unique path between any two critical data centers.
# An edge is part of the required set if and only if the subtree formed by
# removing that edge contains at least one critical data center on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        critical[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    # count_critical[u] is the number of critical nodes in the subtree rooted at u
    count_critical = [0] * (n + 1)
    for i in range(1, n + 1):
        if critical[i]:
            count_critical[i] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            count_critical[p] += count_critical[u]
            # If the subtree at u contains some but not all critical nodes,
            # the edge (u, p) must be included.
            if 0 < count_critical[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
