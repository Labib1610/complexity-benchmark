```python
import sys

# The problem asks for the sum of weights of all edges that lie on the unique 
# paths between any two reproductive nodes in a tree.
# An edge is part of the Steiner Tree connecting the set of reproductive nodes 
# if and only if the subtree formed by removing that edge contains at least 
# one reproductive node on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    reproductive_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many reproductive nodes are in the subtree of each node.
    # Since it's a tree, we can root it at node 1.
    subtree_reproductive_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in reproductive_nodes:
        subtree_reproductive_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains some but not all reproductive nodes,
            # the edge (u, p) must be included.
            count = subtree_reproductive_count[u]
            if count > 0 and count < t:
                total_weight += parent_edge_weight[u]
            subtree_reproductive_count[p] += count
            
    print(total_weight)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
