```python
import sys

# The problem asks for the sum of weights of all edges that lie on the paths 
# between any two telemetry servers. Since the graph is a tree, the set of 
# edges connecting a subset of nodes is the union of all paths between them.
# An edge is part of this union if and only if the subtree formed by removing 
# that edge contains at least one telemetry server on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    telemetry_nodes = set()
    idx = 3
    for _ in range(t):
        telemetry_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Count how many telemetry servers are in the subtree of each node
    # We use iterative DFS to avoid recursion depth issues
    subtree_telemetry_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for node in telemetry_nodes:
        subtree_telemetry_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        count = subtree_telemetry_count[u]
        if count > 0 and count < t:
            total_cost += edge_weight_to_parent[u]
        subtree_telemetry_count[p] += count
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
