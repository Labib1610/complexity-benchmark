```python
import sys

# The problem states that the network is a tree (n nodes, n-1 edges, connected).
# A configuration is valid if all critical modules are connected.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of critical nodes, we must include all edges that lie on the 
# unique paths between any pair of critical nodes.
# This is equivalent to the union of all paths between the critical nodes.
# This set of edges forms the minimal subtree containing all critical nodes (the Steiner Tree).
# Since the graph is a tree, the Steiner Tree is unique and consists of all edges
# that have at least one critical node in each of the two components formed by 
# removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set()
    idx = 3
    for _ in range(t):
        critical_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_critical_count[u] stores the number of critical nodes in the subtree rooted at u
    subtree_critical_count = [0] * (n + 1)
    
    # DFS to count critical nodes in subtrees
    # Using iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in order:
        if u in critical_nodes:
            subtree_critical_count[u] += 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        subtree_critical_count[p] += subtree_critical_count[u]
        
        # If the edge (u, p) separates the tree into two components,
        # and both components contain at least one critical node,
        # then this edge must be part of the Steiner Tree.
        count = subtree_critical_count[u]
        if count > 0 and count < t:
            total_cost += edge_weight_to_parent[u]
            
    print(total_cost)

solve()
```
