```python
import sys

# The problem asks for the minimum cost to connect all research centers in a tree.
# Since the graph is a tree, there is a unique path between any two nodes.
# To connect a set of nodes in a tree, we must include all edges that lie on the 
# unique paths between any pair of these nodes.
# This is equivalent to the sum of weights of all edges that have at least one 
# research center in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    research_centers = set()
    idx = 3
    for _ in range(t):
        research_centers.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use iterative DFS to avoid recursion depth issues
    # count[u] will store the number of research centers in the subtree of u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in research_centers:
        count[u] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if parent[u] != -1:
            p = parent[u]
            # If the subtree at u contains some but not all research centers,
            # the edge (u, p) must be included.
            if count[u] > 0 and count[u] < t:
                total_cost += edge_to_parent_weight[u]
            count[p] += count[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
