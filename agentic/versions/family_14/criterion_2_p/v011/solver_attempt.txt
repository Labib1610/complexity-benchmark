import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    vital_modules = set()
    idx = 3
    for _ in range(t):
        vital_modules.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need the sum of weights of all edges that lie on the path between any two vital modules.
    # An edge (u, v) with weight w is part of the Steiner tree if and only if
    # the subtree formed by removing the edge contains at least one vital module
    # on both sides.
    
    is_vital = [False] * (n + 1)
    for v in vital_modules:
        is_vital[v] = True
        
    subtree_vital_count = [0] * (n + 1)
    total_power = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = (u, w)
                stack.append(v)
                
    for u in reversed(order):
        subtree_vital_count[u] += (1 if is_vital[u] else 0)
        if parent[u] != 0:
            p, w = parent[u]
            subtree_vital_count[p] += subtree_vital_count[u]
            # If this edge separates vital nodes, it must be pressurized
            if subtree_vital_count[u] > 0 and subtree_vital_count[u] < t:
                total_power += w
                
    print(total_power)

solve()
