```python
import sys

# The problem asks for the minimum weight of a subtree that connects all t target nodes.
# This is equivalent to the Steiner Tree problem on a tree, which is simply the sum
# of the weights of all edges that lie on the unique path between any two target nodes.
# An edge is part of this minimal subtree if and only if it separates the tree into
# two components, both of which contain at least one target node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    targets = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use DFS to count how many target nodes are in the subtree rooted at each node.
    # Since it's a tree, we can use an iterative DFS to avoid recursion depth issues.
    target_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for node in targets:
        target_count[node] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        if target_count[u] > 0 and target_count[u] < t:
            total_weight += edge_weight[u]
        
        target_count[p] += target_count[u]
        
    print(total_weight)

if __name__ == '__main__':
    solve()
```
