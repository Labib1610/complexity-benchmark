```python
import sys

# The problem asks for the minimum weight of the minimal subtree that connects 
# all critical nodes. Since the graph is a tree (n nodes, n-1 edges, connected),
# the minimal subtree connecting a set of nodes is unique and consists of all 
# edges that lie on the simple path between any two critical nodes.
# An edge belongs to this subtree if and only if removing it splits the tree 
# into two components, each containing at least one critical node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    critical_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # count_crit[u] will store the number of critical nodes in the subtree rooted at u
    count_crit = [0] * (n + 1)
    total_weight = 0
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order (leaves to root)
    edge_weights = {}
    for u in reversed(order):
        if u in critical_nodes:
            count_crit[u] += 1
        
        for v, w in adj[u]:
            if v == parent[u]:
                # This is the edge (u, parent[u])
                if count_crit[u] > 0 and count_crit[u] < t:
                    total_weight += w
            else:
                count_crit[u] += count_crit[v]
                
    print(total_weight)

solve()
```
