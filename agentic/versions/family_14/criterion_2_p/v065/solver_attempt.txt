```python
import sys

# The problem states the architecture is a tree (n clusters, n-1 pipelines, no loops).
# We need to find the minimum cost to connect all t database clusters.
# This is equivalent to finding the sum of weights of all edges that lie on the 
# unique path between any two database clusters.
# An edge is part of the minimal subtree connecting all database clusters if and only if
# the subtree formed by removing that edge splits the database clusters into two non-empty sets.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    db_clusters = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many database clusters are in the subtree of each node.
    # Since n is up to 200,000, we use iterative DFS to avoid recursion depth issues.
    
    db_count = [0] * (n + 1)
    for node in db_clusters:
        db_count[node] = 1
        
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        # If the subtree at u contains some but not all database clusters,
        # the edge connecting u to its parent must be included.
        if 0 < db_count[u] < t:
            total_cost += edge_weight[u]
            
        # Propagate count to parent
        p = parent[u]
        db_count[p] += db_count[u]
        
    print(total_cost)

if __name__ == '__main__':
    solve()
```
