import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    flagged = set()
    idx = 3
    for _ in range(t):
        flagged.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need the sum of weights of all edges that lie on the path between any two flagged nodes.
    # An edge is part of the Steiner Tree if and only if the subtree below it (when rooted)
    # contains at least one flagged node AND the rest of the tree contains at least one flagged node.
    
    is_flagged = [False] * (n + 1)
    for f in flagged:
        is_flagged[f] = True
        
    subtree_has_flagged = [False] * (n + 1)
    total_flagged = t
    
    # Use iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    subtree_flagged_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if is_flagged[i]:
            subtree_flagged_count[i] = 1
            
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        
        p = parent[u]
        subtree_flagged_count[p] += subtree_flagged_count[u]
        
        # If this edge separates the tree into two parts both containing flagged nodes
        if subtree_flagged_count[u] > 0 and (total_flagged - subtree_flagged_count[u]) > 0:
            total_weight += edge_to_parent_weight[u]
            
    print(total_weight)

solve()
