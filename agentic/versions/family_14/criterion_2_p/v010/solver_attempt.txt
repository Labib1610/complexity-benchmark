```python
import sys

# The problem states m = n - 1 and the graph is connected, so it's a tree.
# We need to find the minimum weight of the subtree that connects all t priority buildings.
# In a tree, the minimal set of edges connecting a set of nodes S is the union of all paths
# between nodes in S. This is equivalent to the sum of weights of all edges that have at least
# one priority node in each of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority_nodes = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Use iterative DFS to avoid recursion depth issues
    # count[u] stores the number of priority nodes in the subtree rooted at u
    count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in priority_nodes:
        count[u] = 1
        
    total_weight = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if count[u] > 0 and count[u] < t:
            total_weight += edge_to_parent_weight[u]
        count[p] += count[u]
        
    print(total_weight)

solve()
```
