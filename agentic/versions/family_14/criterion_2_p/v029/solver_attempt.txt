import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    mandatory = set()
    idx = 3
    for _ in range(t):
        mandatory.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # We need the sum of weights of all edges that lie on the path between 
    # any two mandatory nodes. An edge is part of the Steiner tree if and only 
    # if the subtree below it (when rooted arbitrarily) contains at least one 
    # mandatory node and the rest of the tree contains at least one mandatory node.
    
    is_mandatory = [False] * (n + 1)
    for node in mandatory:
        is_mandatory[node] = True
        
    subtree_has_mandatory = [False] * (n + 1)
    total_cost = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_mandatory[u]:
            subtree_has_mandatory[u] = True
        
        if subtree_has_mandatory[u] and u != 1:
            total_cost += edge_to_parent_weight[u]
            
        if u != 1:
            if subtree_has_mandatory[u]:
                subtree_has_mandatory[parent[u]] = True
                
    print(total_cost)

solve()
