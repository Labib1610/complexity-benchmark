```python
import sys

# The problem states the graph is a tree (n habitats, n-1 tethers, connected).
# We need to find the minimum cost to connect all generator habitats.
# This is equivalent to finding the sum of weights of all edges that lie on the 
# unique path between any two generator habitats.
# An edge is part of the path between two generator habitats if and only if 
# removing that edge splits the tree into two components, each containing at least one generator.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    generators = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # Use DFS to count how many generators are in the subtree of each node
    # Since n can be up to 200,000, use iterative DFS to avoid recursion depth issues
    gen_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        if u in generators:
            gen_count[u] = 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains some generators but not all,
            # the edge (u, p) must be reinforced.
            if 0 < gen_count[u] < t:
                total_cost += edge_to_parent_weight[u]
            gen_count[p] += gen_count[u]
            
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth just in case, though iterative is used
    sys.setrecursionlimit(300000)
    solve()
```
