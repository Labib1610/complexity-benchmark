```python
import sys

# The problem asks for the minimum cost to connect a subset of nodes (keyholders)
# in a tree. This is equivalent to finding the weight of the Steiner Tree,
# which in a tree is simply the sum of weights of all edges that lie on the
# unique path between any two keyholders.
# An edge is part of the Steiner Tree if and only if the subtree formed by
# removing that edge contains at least one keyholder on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    keyholders = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Use DFS to count keyholders in each subtree
    # Since n is up to 200,000, use iterative DFS to avoid recursion depth issues
    key_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                stack.append(v)
                
    for node in order:
        if node in keyholders:
            key_count[node] = 1
            
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, -1, -1):
        u = order[i]
        if u == 1:
            continue
        p = parent[u]
        key_count[p] += key_count[u]
        # If this subtree contains some but not all keyholders, the edge must be used
        if 0 < key_count[u] < t:
            total_cost += edge_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
