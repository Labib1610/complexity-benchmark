```python
import sys

# The problem states the structure is a tree (n rooms, n-1 edges, unique path).
# We need to connect all fresco rooms with minimum total edge weight.
# This is equivalent to finding the Steiner Tree in a tree, which is simply
# the sum of weights of all edges that lie on the unique path between any two
# fresco rooms.
# An edge belongs to the Steiner Tree if and only if the subtree formed by
# removing that edge contains at least one fresco room on both sides.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    fresco_rooms = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Count how many fresco rooms are in the subtree of each node
    # when rooted at 1.
    fresco_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in fresco_rooms:
        fresco_count[u] = 1
        
    total_weight = 0
    # Process in reverse topological order (leaves to root)
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        fresco_count[p] += fresco_count[u]
        # If this edge separates some fresco rooms from others, it must be included
        if fresco_count[u] > 0 and fresco_count[u] < t:
            total_weight += parent_edge_weight[u]
            
    print(total_weight)

solve()
```
