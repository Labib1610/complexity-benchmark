import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    anomalies = set()
    idx = 3
    for _ in range(t):
        anomalies.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Find the root of the subtree containing all anomalies
    # We can pick any anomaly as the root
    root = next(iter(anomalies))
    
    # We need to find the smallest subtree containing all anomalies.
    # The total weight of this subtree is 2 * (sum of weights of edges in the subtree)
    # minus the weight of the longest path between two anomalies in the subtree.
    
    # First, identify all nodes that are part of the minimal subtree
    is_anomaly = [False] * (n + 1)
    for a in anomalies:
        is_anomaly[a] = True
        
    has_anomaly = [False] * (n + 1)
    
    # DFS to mark nodes that are on the path between any two anomalies
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if is_anomaly[u]:
            has_anomaly[u] = True
        if has_anomaly[u] and parent[u] != 0:
            has_anomaly[parent[u]] = True
            
    # Calculate total weight of edges in the minimal subtree
    total_weight = 0
    for i in range(1, n + 1):
        if has_anomaly[i] and parent[i] != 0:
            total_weight += edge_to_parent[i]
            
    # Find the longest path between two anomalies in the subtree
    # This is equivalent to the diameter of the minimal subtree
    def get_farthest(start_node):
        farthest_node = start_node
        max_dist = -1
        stack = [(start_node, 0)]
        visited = {start_node}
        while stack:
            u, dist = stack.pop()
            if is_anomaly[u]:
                if dist > max_dist:
                    max_dist = dist
                    farthest_node = u
            for v, w in adj[u]:
                if v not in visited and has_anomaly[v]:
                    visited.add(v)
                    stack.append((v, dist + w))
        return farthest_node, max_dist

    node_a, _ = get_farthest(root)
    node_b, diameter = get_farthest(node_a)
    
    print(2 * total_weight - diameter)

solve()
