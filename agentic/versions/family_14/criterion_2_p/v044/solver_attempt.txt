```python
import sys

# The problem states that the graph is a tree (n endpoints, m = n-1 channels, 
# and exactly one path between any two endpoints).
# To connect a subset of t essential endpoints in a tree, we must include 
# all edges that lie on the unique paths between any two essential endpoints.
# This is equivalent to finding the minimal subtree that contains all 
# essential endpoints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    essential = set()
    idx = 3
    for _ in range(t):
        essential.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the sum of weights of all edges that are part of the 
    # minimal subtree connecting all essential nodes.
    # An edge is in the minimal subtree if and only if both components 
    # formed by removing the edge contain at least one essential node.
    
    # Use DFS to count essential nodes in subtrees
    subtree_essential_count = [0] * (n + 1)
    parent = [0] * (n + 1)
    edge_to_parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent_weight[v] = w
                stack.append(v)
                
    for node in essential:
        subtree_essential_count[node] = 1
        
    total_cost = 0
    # Process in reverse topological order (bottom-up)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            subtree_essential_count[p] += subtree_essential_count[u]
            # If the subtree at u contains some but not all essential nodes,
            # the edge (u, p) must be included.
            if 0 < subtree_essential_count[u] < t:
                total_cost += edge_to_parent_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    # Increase recursion depth for deep trees
    sys.setrecursionlimit(300000)
    solve()
```
