```python
import sys

# The problem describes a tree structure (n caverns, n-1 shafts, connected).
# We need to find the minimum weight of the subtree that connects all t trapped caverns.
# In a tree, the set of edges forming the minimal connection between a set of nodes
# is the union of all paths between these nodes.
# This is equivalent to the sum of weights of all edges that have at least one
# trapped node in one of the two components formed by removing that edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    trapped = set()
    idx = 3
    for _ in range(t):
        trapped.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # subtree_trapped[u] stores the number of trapped nodes in the subtree rooted at u
    subtree_trapped = [0] * (n + 1)
    total_energy = 0
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    for u in reversed(order):
        if u in trapped:
            subtree_trapped[u] += 1
        if parent[u] != 0:
            subtree_trapped[parent[u]] += subtree_trapped[u]
            # If this edge separates some trapped nodes from others, it must be included
            if subtree_trapped[u] > 0 and subtree_trapped[u] < t:
                total_energy += edge_weight_to_parent[u]
                
    print(total_energy)

solve()
```
