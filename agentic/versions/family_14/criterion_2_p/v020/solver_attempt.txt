```python
import sys

# The problem states that the m = n-1 fusions form a tree structure.
# We need to find the minimum weight of the edges that connect all t incantation fragments.
# This is equivalent to finding the weight of the minimal subtree that contains all t nodes.
# In a tree, the minimal subtree containing a set of nodes is the union of all paths
# between these nodes. This is equivalent to the sum of weights of all edges that
# have at least one incantation fragment in each of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    incantation_fragments = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many incantation fragments are in each subtree.
    # Perform DFS to calculate this.
    count = [0] * (n + 1)
    total_resin = 0
    
    # Iterative DFS to avoid recursion depth issues
    stack = [(1, 0, 0)] # node, parent, weight_to_parent
    order = []
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    
    dfs_stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while dfs_stack:
        u = dfs_stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                weight_to_parent[v] = w
                dfs_stack.append(v)
                
    for u in reversed(order):
        if u in incantation_fragments:
            count[u] += 1
        if parent[u] != 0:
            count[parent[u]] += count[u]
            # If this edge separates the tree into two parts, both containing
            # at least one incantation fragment, then this edge must be included.
            if count[u] > 0 and count[u] < t:
                total_resin += weight_to_parent[u]
                
    print(total_resin)

solve()
```
