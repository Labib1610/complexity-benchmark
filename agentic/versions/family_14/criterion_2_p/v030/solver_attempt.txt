```python
import sys

# The problem states that the n-1 ducts form a tree (since they connect n nodes and are connected).
# We need to find the minimum weight of a subset of edges such that all priority zones are connected.
# In a tree, the unique path between any two nodes is fixed.
# To connect a set of nodes in a tree, we must select all edges that lie on the paths 
# between any pair of these nodes. This is equivalent to the union of all paths 
# between the priority nodes.
# This is equivalent to the edges in the minimal subtree containing all priority nodes.
# An edge belongs to this subtree if and only if the two components it creates 
# both contain at least one priority node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    priority_zones = set(map(int, input_data[3:3+t]))
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many priority nodes are in each subtree.
    # Root the tree at node 1.
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                stack.append(v)
                
    priority_count = [0] * (n + 1)
    for node in priority_zones:
        priority_count[node] = 1
        
    total_energy = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if p != 0:
            # If the subtree at u contains some but not all priority nodes,
            # the edge (u, p) must be included.
            if 0 < priority_count[u] < t:
                total_energy += edge_weight_to_parent[u]
            priority_count[p] += priority_count[u]
            
    print(total_energy)

solve()
```
