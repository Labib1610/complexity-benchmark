import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    essential = [False] * (n + 1)
    idx = 3
    for _ in range(t):
        essential[int(input_data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to find the smallest subtree containing all essential nodes.
    # This is equivalent to the sum of weights of all edges that lie on a path
    # between any two essential nodes.
    # An edge (u, v) with weight w is part of the minimal subtree if and only if
    # there is at least one essential node in the subtree of u (when rooted at v)
    # and at least one essential node in the subtree of v (when rooted at u).
    
    has_essential = [False] * (n + 1)
    
    # Iterative DFS to avoid recursion depth issues
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
                
    # Process in reverse topological order to propagate essential counts
    total_essential = t
    is_essential_subtree = [0] * (n + 1)
    for i in range(1, n + 1):
        if essential[i]:
            is_essential_subtree[i] = 1
            
    ans = 0
    # We need to store edge weights to parent
    edge_to_parent_w = [0] * (n + 1)
    for u in range(1, n + 1):
        for v, w in adj[u]:
            if v == parent[u]:
                edge_to_parent_w[u] = w
                
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        is_essential_subtree[p] += is_essential_subtree[u]
        if is_essential_subtree[u] > 0 and is_essential_subtree[u] < total_essential:
            ans += edge_to_parent_w[u]
            
    print(ans)

solve()
