```python
import sys

# The problem states that the network is a tree (n nodes, n-1 edges, connected).
# To connect a subset of nodes (core monoliths) in a tree, we must include all
# edges that lie on the unique paths between any two core monoliths.
# This is equivalent to finding the smallest subtree containing all core nodes.
# An edge is part of this subtree if and only if the two components formed by
# removing that edge both contain at least one core monolith.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_nodes = set()
    idx = 3
    for _ in range(t):
        core_nodes.add(int(input_data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
        idx += 3
        
    # subtree_core_count[u] will store the number of core nodes in the subtree
    # rooted at u (using 1 as root).
    subtree_core_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for node in core_nodes:
        subtree_core_count[node] = 1
        
    total_energy = 0
    # Process in reverse topological order (leaves to root)
    for i in range(n - 1, 0, -1):
        u = order[i]
        p = parent[u]
        if subtree_core_count[u] > 0 and subtree_core_count[u] < t:
            total_energy += parent_edge_weight[u]
        subtree_core_count[p] += subtree_core_count[u]
        
    print(total_energy)

if __name__ == '__main__':
    solve()
```
