```python
import sys

# The problem asks for the minimum cost to connect all target variables in a tree.
# Since the graph is a tree (n nodes, n-1 edges, connected), the unique path
# between any two nodes is the only way to enforce equality.
# To make all t target variables equal, we must select all edges that lie on
# the paths connecting the target variables.
# This is equivalent to finding the Steiner Tree for the set of target variables.
# In a tree, the Steiner Tree is the minimal subtree containing all target nodes.
# The edges in this subtree are exactly those edges that have at least one
# target node in each of the two components formed by removing the edge.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    targets = set(map(int, input_data[3:3+t]))
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # We need to count how many target nodes are in each subtree.
    # Root the tree at an arbitrary node, say 1.
    target_count = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    parent = [0] * (n + 1)
    order = []
    stack = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    for u in order:
        if u in targets:
            target_count[u] = 1
            
    total_cost = 0
    # Process in reverse topological order to propagate target counts
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        target_count[p] += target_count[u]
        # If this edge separates the tree into two parts, both containing
        # at least one target node, then this edge must be included.
        if target_count[u] > 0 and target_count[u] < t:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

solve()
```
