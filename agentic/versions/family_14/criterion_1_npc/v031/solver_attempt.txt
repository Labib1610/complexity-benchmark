```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    core_languages = [int(x) for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Dijkstra to find all-pairs shortest paths between core languages
    dist = [[float('inf')] * (n + 1) for _ in range(t)]
    for i in range(t):
        start_node = core_languages[i]
        dist[i][start_node] = 0
        pq = [(0, start_node)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    # Steiner Tree problem on a graph with n nodes and t terminals
    # DP state: dp[mask][v] is min cost to connect terminals in mask with v as root
    # mask is 2^t, v is 1..n
    dp = [[float('inf')] * (n + 1) for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][core_languages[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        for submask in range((mask - 1) & mask, 0, -1):
            submask &= mask
            for v in range(1, n + 1):
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v])
        
        # Relax edges
        pq = []
        for v in range(1, n + 1):
            if dp[mask][v] != float('inf'):
                heapq.heappush(pq, (dp[mask][v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << t) - 1][v] for v in range(1, n + 1))
    print(ans)

solve()
```
