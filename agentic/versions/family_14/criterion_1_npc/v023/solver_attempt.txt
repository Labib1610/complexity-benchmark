```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    K = int(input_data[2])
    
    anchors = [int(input_data[i + 3]) for i in range(K)]
    
    adj = [[] for _ in range(N + 1)]
    idx = 3 + K
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if K <= 1:
        print(0)
        return

    # Steiner Tree problem on a graph with N nodes and K terminals.
    # Since K is small (<= 10), we use the Dreyfus-Wagner algorithm.
    # dp[mask][i] is the min cost to connect the subset of anchors represented by mask,
    # with i being a node in the tree.
    
    inf = float('inf')
    dp = [[inf] * (N + 1) for _ in range(1 << K)]
    
    for i in range(K):
        dp[1 << i][anchors[i]] = 0
        
    for mask in range(1, 1 << K):
        # Update dp[mask][i] using submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, N + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Dijkstra to update dp[mask][i] for all i
        pq = []
        for i in range(1, N + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    ans = min(dp[(1 << K) - 1][i] for i in range(1, N + 1))
    print(ans)

if __name__ == '__main__':
    solve()
```
