```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_commodities = [int(x) for x in input_data[3:3+t]]
    
    if t <= 1:
        print(0)
        return
    
    adj = []
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj.append((u, v, w))
        idx += 3
        
    # Steiner Tree problem on a graph with n <= 100 and t <= 10.
    # Since t is very small, we use the Dreyfus-Wagner algorithm.
    # dp[mask][i] is the min cost to connect the subset of core commodities 
    # represented by 'mask' such that they are all connected to node 'i'.
    
    inf = float('inf')
    dp = [[inf] * (n + 1) for _ in range(1 << t)]
    
    # Base case: each core commodity is connected to itself with cost 0
    for i in range(t):
        dp[1 << i][core_commodities[i]] = 0
        
    # Precompute all-pairs shortest paths using Dijkstra
    dist = [[inf] * (n + 1) for _ in range(n + 1)]
    graph = [[] for _ in range(n + 1)]
    for u, v, w in adj:
        graph[u].append((v, w))
        graph[v].append((u, w))
    for i in range(1, n + 1):
        dist[i][i] = 0
        pq = [(0, i)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]: continue
            for v, w in graph[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    for mask in range(1, 1 << t):
        # Submask splitting
        submask = (mask - 1) & mask
        while submask > 0:
            for i in range(1, n + 1):
                dp[mask][i] = min(dp[mask][i], dp[submask][i] + dp[mask ^ submask][i])
            submask = (submask - 1) & mask
            
        # Relaxation using shortest paths
        pq = []
        for i in range(1, n + 1):
            if dp[mask][i] != inf:
                heapq.heappush(pq, (dp[mask][i], i))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]: continue
            for v in range(1, n + 1):
                if dp[mask][v] > d + dist[u][v]:
                    dp[mask][v] = d + dist[u][v]
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
