```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    core_sites = [int(x) - 1 for x in input_data[3:3+t]]
    
    adj = [[] for _ in range(n)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    # Dijkstra to find all-pairs shortest paths between core sites and all other nodes
    dist = [[float('inf')] * n for _ in range(t)]
    for i in range(t):
        d = dist[i]
        d[core_sites[i]] = 0
        pq = [(0, core_sites[i])]
        while pq:
            curr_d, u = heapq.heappop(pq)
            if curr_d > d[u]:
                continue
            for v, w in adj[u]:
                if d[v] > curr_d + w:
                    d[v] = curr_d + w
                    heapq.heappush(pq, (d[v], v))
                    
    # Steiner Tree problem for small t (t <= 10)
    # dp[mask][v] is min cost to connect subset of cores 'mask' with 'v' as a terminal
    dp = [[float('inf')] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][core_sites[i]] = 0
        
    for mask in range(1, 1 << t):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            for v in range(n):
                dp[mask][v] = min(dp[mask][v], dp[submask][v] + dp[mask ^ submask][v])
            submask = (submask - 1) & mask
            
        # Relax edges using Dijkstra
        pq = []
        for v in range(n):
            if dp[mask][v] != float('inf'):
                heapq.heappush(pq, (dp[mask][v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    print(min(dp[(1 << t) - 1]))

solve()
```
