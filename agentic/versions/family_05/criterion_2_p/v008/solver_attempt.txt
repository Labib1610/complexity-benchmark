import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Stage 1 guarantees:
    # Each specialist x is qualified on exactly one machine y.
    # Let's map specialist x to their unique machine y.
    spec_to_machine = {}
    
    # We need to select k triples (x, y, z) such that:
    # 1. Each x in 1..k is used once.
    # 2. Each y in 1..k is used once.
    # 3. Each z in 1..k is used once.
    
    # Since each x is tied to a unique y, the condition "each x used once"
    # and "each y used once" is satisfied if we pick exactly one triple
    # for each x.
    # Let the allowed triples for specialist x be (x, machine[x], z_i).
    # We need to pick one z_i for each x such that all chosen z_i are distinct.
    
    # This is a bipartite matching problem:
    # Left side: Specialists 1..k
    # Right side: Time slots 1..k
    # Edge (x, z) exists if (x, machine[x], z) is an allowed combination.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        spec_to_machine[x] = y
        adj[x].append(z)
        idx += 3
        
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            print(f"{x} {spec_to_machine[x]} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
