import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each extraction site (e) is exclusively compatible with one distillation column (d).
    # 2. No two extraction sites share the same distillation column.
    # This means there is a fixed mapping e -> d.
    # We need to select k blueprints (e, d, i) such that:
    # - Each e is used exactly once.
    # - Each d is used exactly once.
    # - Each i is used exactly once.
    
    # Since e -> d is fixed, we can represent the blueprints as a list of 
    # (e, d, i) triplets. For each e, we have a set of possible i's.
    # Let's group by e.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        e = int(input_data[idx])
        d = int(input_data[idx+1])
        i = int(input_data[idx+2])
        adj[e].append((d, i))
        idx += 3
        
    # Since each e maps to exactly one d, and we need to use each e once,
    # we must pick exactly one blueprint for each e.
    # Let the chosen blueprint for e be (e, d_e, i_e).
    # We need to ensure that all i_e are distinct.
    # This is a bipartite matching problem:
    # Left side: extraction sites (1..k)
    # Right side: injection wells (1..k)
    # Edge exists from e to i if there is a blueprint (e, d, i).
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    # Standard Hopcroft-Karp or DFS-based bipartite matching
    # Since k <= 1200, DFS is sufficient.
    
    graph = [[] for _ in range(k + 1)]
    for e in range(1, k + 1):
        for _, i in adj[e]:
            graph[e].append(i)
            
    def dfs(u, visited, match):
        for v in graph[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited, match):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the result
        # match[i] = e means injection well i is assigned to extraction site e
        # We need to find the d for each e.
        e_to_d = {}
        e_to_i = {}
        for e in range(1, k + 1):
            e_to_d[e] = adj[e][0][0]
        
        for i in range(1, k + 1):
            e = match[i]
            print(f"{e} {e_to_d[e]} {i}")
    else:
        print("NO")

solve()
