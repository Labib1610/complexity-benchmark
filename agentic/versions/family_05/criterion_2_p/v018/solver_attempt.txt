import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # foreman_to_vehicle maps foreman x to their unique vehicle y
    foreman_to_vehicle = {}
    # foreman_to_sectors maps foreman x to a list of sectors z
    foreman_to_sectors = {}
    # protocol_map stores (x, y, z) to keep track of original protocols
    # Since each (x, y) is unique, we can store (x, z) -> y or similar
    # Actually, we need to select one z for each x such that all z are distinct.
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        foreman_to_vehicle[x] = y
        if x not in foreman_to_sectors:
            foreman_to_sectors[x] = []
        foreman_to_sectors[x].append(z)
        
    # We need to select exactly one z for each x in {1..k}
    # such that all selected z are distinct.
    # This is a bipartite matching problem:
    # Left side: foremen {1..k}
    # Right side: sectors {1..k}
    # Edge exists if foreman x can work in sector z.
    
    adj = [[] for _ in range(k + 1)]
    for x in foreman_to_sectors:
        for z in foreman_to_sectors[x]:
            adj[x].append(z)
            
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            y = foreman_to_vehicle[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
