```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "For each specialist, every listed combination uses 
    # the same machine, and different specialists use different machines."
    # This means there is a fixed mapping from specialist x to machine y.
    # Let's find this mapping.
    
    spec_to_mach = {}
    # adj[x] will store a list of time slots z for specialist x
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in spec_to_mach:
            spec_to_mach[x] = y
        adj[x].append((y, z))
        
    # Check if all specialists are present
    if len(spec_to_mach) < k:
        print("NO")
        return
    
    # We need to select one z for each x such that all z are distinct (1..k)
    # This is a bipartite matching problem.
    # Left side: specialists 1..k
    # Right side: time slots 1..k
    # Edge exists if (x, y, z) is allowed.
    
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for _, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match[z] < 0 or dfs(match[z], visited):
                    match[z] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # match[z] = x, so we need to output x, spec_to_mach[x], z
        for z in range(1, k + 1):
            x = match[z]
            print(f"{x} {spec_to_mach[x]} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
