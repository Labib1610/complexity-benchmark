import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each starship s is paired with exactly one reactor module r.
    # We can map s -> r.
    s_to_r = {}
    triplets = []
    idx = 2
    for _ in range(t):
        s = int(input_data[idx])
        r = int(input_data[idx+1])
        g = int(input_data[idx+2])
        triplets.append((s, r, g))
        s_to_r[s] = r
        idx += 3
        
    # We need to select k triplets (s, r, g) such that:
    # 1. Each s in {1..k} is used once.
    # 2. Each r in {1..k} is used once.
    # 3. Each g in {1..k} is used once.
    # Since each s is already tied to a unique r, condition 1 and 2 are 
    # satisfied if we pick exactly one triplet for each s.
    # The problem reduces to: for each s, pick one g such that all g are distinct.
    
    # Group triplets by starship s
    adj = [[] for _ in range(k + 1)]
    for s, r, g in triplets:
        adj[s].append(g)
        
    # Bipartite matching: Starships (1..k) to Gantries (1..k)
    # Since each s must be used, we need a perfect matching.
    match_g = [-1] * (k + 1)
    match_s = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_g[v] < 0 or dfs(match_g[v], visited):
                    match_g[v] = u
                    match_s[u] = v
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the triplets
        # We need to output (s, r, g)
        # We have match_s[s] = g
        # We need r for each s
        for s in range(1, k + 1):
            g = match_s[s]
            r = s_to_r[s]
            print(f"{s} {r} {g}")
    else:
        print("NO")

solve()
