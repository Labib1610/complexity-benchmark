import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # pilot_to_starfighter mapping
    p_to_s = {}
    # adj[pilot] = list of missions
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        p = int(input_data[idx])
        s = int(input_data[idx+1])
        m = int(input_data[idx+2])
        idx += 3
        
        if p not in p_to_s:
            p_to_s[p] = s
        adj[p].append((s, m))
        
    # Since each pilot is paired with exactly one starfighter,
    # and we need to use each pilot and each starfighter exactly once,
    # we must use every pilot p from 1 to k.
    # The problem reduces to matching each pilot p to a mission m
    # such that all missions 1 to k are used exactly once.
    
    # match[mission] = pilot
    match = [0] * (k + 1)
    vis = [False] * (k + 1)
    
    # adj_m[pilot] = list of missions
    adj_m = [[] for _ in range(k + 1)]
    for p in range(1, k + 1):
        for s, m in adj[p]:
            adj_m[p].append(m)
            
    def dfs(u, visited, match):
        for v in adj_m[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] == 0 or dfs(match[v], visited, match):
                    match[v] = u
                    return True
        return False

    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if not dfs(i, visited, match):
            print("NO")
            return
            
    print("YES")
    # match[m] = p
    # We need to output p s m
    for m in range(1, k + 1):
        p = match[m]
        s = p_to_s[p]
        print(f"{p} {s} {m}")

solve()
