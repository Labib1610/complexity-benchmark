import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states that each captain is paired with exactly one cruiser.
    # Let's map each captain to their unique cruiser.
    cap_to_cruiser = {}
    # Store configurations as (captain, cruiser, sector)
    # Since each captain is tied to one cruiser, we can group by captain.
    # The problem asks to pick one configuration for each captain such that
    # all sectors are used exactly once.
    
    # cap_to_sectors[c] = list of sectors available for captain c
    cap_to_sectors = [[] for _ in range(k + 1)]
    # To reconstruct the answer, we need to know which sector was chosen for which captain
    # We can store (sector, cruiser) for each captain
    cap_to_options = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        c = int(input_data[idx])
        v = int(input_data[idx+1])
        s = int(input_data[idx+2])
        idx += 3
        cap_to_sectors[c].append(s)
        cap_to_options[c].append((s, v))
        
    # This is a bipartite matching problem.
    # Left side: captains (1 to k)
    # Right side: sectors (1 to k)
    # Edge exists if captain c can be assigned to sector s.
    
    adj = [[] for _ in range(k + 1)]
    for c in range(1, k + 1):
        for s in cap_to_sectors[c]:
            adj[c].append(s)
            
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # match[s] = c
        # We need to find the cruiser v for each captain c
        # We have cap_to_options[c] = [(s, v), ...]
        # We need to pick the (s, v) where s == match_inv[c]
        
        # Create inverse mapping: captain -> sector
        cap_to_sector = [0] * (k + 1)
        for s in range(1, k + 1):
            cap_to_sector[match[s]] = s
            
        for c in range(1, k + 1):
            s_chosen = cap_to_sector[c]
            for s, v in cap_to_options[c]:
                if s == s_chosen:
                    print(f"{c} {v} {s}")
                    break
    else:
        print("NO")

solve()
