import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each sensor cluster u is hardwired to exactly one relay station v.
    # 2. No two distinct sensor clusters share the same relay station.
    # This means there is a bijection between sensor clusters and relay stations.
    # Let's map each sensor cluster u to its unique relay station v.
    
    sensor_to_relay = {}
    # protocols_by_sensor[u] will store a list of (u, v, w)
    protocols_by_sensor = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        
        if u not in sensor_to_relay:
            sensor_to_relay[u] = v
        elif sensor_to_relay[u] != v:
            # This case shouldn't happen based on problem description
            print("NO")
            return
        
        protocols_by_sensor[u].append((u, v, w))
        
    # We need to pick exactly one protocol for each sensor cluster u in {1..k}.
    # Since each u has a unique v, picking one protocol for each u automatically
    # satisfies the condition that each relay station is used exactly once.
    # The constraint is that each control frequency w must also be used exactly once.
    # This is a bipartite matching problem:
    # Left side: sensor clusters {1..k}
    # Right side: control frequencies {1..k}
    # Edge exists between u and w if there is a protocol (u, v, w).
    
    adj = [[] for _ in range(k + 1)]
    for u in range(1, k + 1):
        if not protocols_by_sensor[u]:
            print("NO")
            return
        for _, _, w in protocols_by_sensor[u]:
            adj[u].append(w)
            
    # Hopcroft-Karp or simple DFS matching
    match_w = [0] * (k + 1)
    match_u = [0] * (k + 1)
    
    def dfs(u, visited):
        for w in adj[u]:
            if not visited[w]:
                visited[w] = True
                if match_w[w] == 0 or dfs(match_w[w], visited):
                    match_w[w] = u
                    match_u[u] = w
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        if dfs(i, [False] * (k + 1)):
            count += 1
            
    if count == k:
        print("YES")
        for u in range(1, k + 1):
            w = match_u[u]
            v = sensor_to_relay[u]
            print(f"{u} {v} {w}")
    else:
        print("NO")

solve()
