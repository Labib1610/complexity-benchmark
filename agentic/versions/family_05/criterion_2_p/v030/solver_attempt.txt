```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each catalyst x is tied to a unique solvent y.
    # 2. We need to pick k reactions such that each x, y, z is used exactly once.
    # Since each x is tied to a unique y, the condition "each x used once" 
    # and "each y used once" are equivalent.
    # Let's map each catalyst x to its unique solvent y.
    
    cat_to_sol = {}
    # adj[x] will store a list of (y, z) pairs for catalyst x
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x in cat_to_sol and cat_to_sol[x] != y:
            # This violates the problem statement "Each catalyst was permanently 
            # dissolved into a specific solvent".
            pass
        cat_to_sol[x] = y
        adj[x].append((y, z))
        
    # If we don't have k catalysts, we can't pick k reactions
    if len(cat_to_sol) < k:
        print("NO")
        return

    # We need to pick exactly one z for each x.
    # Since each x is tied to a unique y, we just need to pick one z for each x
    # such that all z are distinct.
    # This is a bipartite matching problem:
    # Left side: catalysts (1 to k)
    # Right side: temperatures (1 to k)
    # Edge exists if there is a reaction (x, y, z)
    
    match = [-1] * (k + 1)
    # Store which reaction (y, z) was used for catalyst x
    result = [None] * (k + 1)
    
    # Standard Hopcroft-Karp or DFS-based matching
    graph = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for y, z in adj[x]:
            graph[x].append((z, y))
            
    def dfs(u, visited):
        for z, y in graph[u]:
            if not visited[z]:
                visited[z] = True
                if match[z] < 0 or dfs(match[z], visited):
                    match[z] = u
                    result[u] = (y, z)
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for x in range(1, k + 1):
            y, z = result[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
```
