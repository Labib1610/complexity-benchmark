```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "each specialist is qualified on exactly one machine, 
    # and no two specialists share the same machine".
    # This means there is a fixed mapping from specialist x to machine y.
    # Let's find this mapping.
    spec_to_mach = {}
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in spec_to_mach:
            spec_to_mach[x] = y
        
        # We need to pick one z for each x.
        # Since x is uniquely tied to y, we just need to match each x to a unique z.
        # This is a bipartite matching problem between specialists (1..k) and time slots (1..k).
        adj[x].append(z)
        
    # Hopcroft-Karp or simple DFS-based matching
    match_z = [-1] * (k + 1)
    match_x = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_z[v] < 0 or dfs(match_z[v], visited):
                    match_z[v] = u
                    match_x[u] = v
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for x in range(1, k + 1):
            y = spec_to_mach[x]
            z = match_x[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
