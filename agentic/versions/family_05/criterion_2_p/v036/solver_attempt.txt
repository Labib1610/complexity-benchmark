import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each director d has exactly one film f.
    # Let's map director to their film.
    dir_to_film = {}
    # We need to select one configuration (d, f, s) for each d in 1..k.
    # Since each d has a unique f, we just need to pick one s for each d
    # such that all s are distinct.
    # This is a bipartite matching problem:
    # Left side: directors 1..k
    # Right side: rooms 1..k
    # Edge exists if (d, f, s) is an acceptable configuration.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        d = int(input_data[idx])
        f = int(input_data[idx+1])
        s = int(input_data[idx+2])
        adj[d].append((f, s))
        idx += 3
        
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    def dfs(u, visited, match_r):
        for f, s in adj[u]:
            if not visited[s]:
                visited[s] = True
                if match_r[s] < 0 or dfs(match_r[s], visited, match_r):
                    match_r[s] = u
                    return True
        return False

    match_r = [-1] * (k + 1)
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match_r):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the result
        # match_r[s] = d
        # We need to find the film f for each d
        d_to_f = {}
        # Re-read to get d_to_f
        idx = 2
        for _ in range(t):
            d = int(input_data[idx])
            f = int(input_data[idx+1])
            d_to_f[d] = f
            idx += 3
            
        for s in range(1, k + 1):
            d = match_r[s]
            print(f"{d} {d_to_f[d]} {s}")
    else:
        print("NO")

solve()
