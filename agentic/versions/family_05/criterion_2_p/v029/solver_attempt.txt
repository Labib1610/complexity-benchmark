```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "For each primary module, every listed combination 
    # uses the same secondary module, and different primary modules use 
    # different secondary modules."
    # This means there is a fixed mapping f: X -> Y.
    # We can represent the allowed combinations as a list of (x, z) for each x.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        idx += 3
        
    # We need to pick one (y, z) for each x such that all z are distinct.
    # This is a bipartite matching problem.
    # Left side: primary modules (1..k)
    # Right side: access tokens (1..k)
    # Edge exists if (x, y, z) is allowed.
    
    match = [-1] * (k + 1)
    # Store the y associated with the chosen z for each x
    y_for_z = [0] * (k + 1)
    
    def dfs(u, visited):
        for y, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match[z] < 0 or dfs(match[z], visited):
                    match[z] = u
                    y_for_z[z] = y
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            y = y_for_z[z]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
```
