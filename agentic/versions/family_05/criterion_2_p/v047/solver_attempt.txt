import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # pilot_to_mech[x] = y
    pilot_to_mech = {}
    # pilot_to_phases[x] = [z1, z2, ...]
    pilot_to_phases = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        pilot_to_mech[x] = y
        pilot_to_phases[x].append(z)
        
    # We need to pick exactly one phase z for each pilot x such that
    # the set of chosen phases is {1, 2, ..., k}.
    # This is a perfect matching problem in a bipartite graph where
    # one set is pilots {1..k} and the other is phases {1..k}.
    # An edge exists between pilot x and phase z if (x, y, z) is feasible.
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        if x not in pilot_to_mech:
            print("NO")
            return
        for z in pilot_to_phases[x]:
            adj[x].append(z)
            
    # Hopcroft-Karp or simple DFS based matching
    match_phase = [0] * (k + 1)
    match_pilot = [0] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_phase[v] == 0 or dfs(match_phase[v], visited):
                    match_phase[v] = u
                    match_pilot[u] = v
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for x in range(1, k + 1):
            z = match_pilot[x]
            y = pilot_to_mech[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
