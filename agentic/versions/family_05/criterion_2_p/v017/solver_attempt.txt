```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Since each operator x is associated with exactly one terminal y,
    # we can map each operator to its unique terminal.
    op_to_term = {}
    # We need to select one window z for each operator-terminal pair (x, y).
    # Let the pair be (x, y). We have a list of available windows z.
    # This is a bipartite matching problem:
    # Left side: operators (1 to k)
    # Right side: windows (1 to k)
    # Edge exists if (x, y, z) is in the input.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        op_to_term[x] = y
        adj[x].append((y, z))
        idx += 3
        
    # Hopcroft-Karp or simple DFS-based matching
    match_w = [-1] * (k + 1)
    match_o = [-1] * (k + 1)
    
    def dfs(u, visited):
        for y, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match_w[z] < 0 or dfs(match_w[z], visited):
                    match_w[z] = u
                    match_o[u] = z
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        if i in op_to_term:
            visited = [False] * (k + 1)
            if dfs(i, visited):
                count += 1
    
    if count == k:
        print("YES")
        for i in range(1, k + 1):
            print(f"{i} {op_to_term[i]} {match_o[i]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
