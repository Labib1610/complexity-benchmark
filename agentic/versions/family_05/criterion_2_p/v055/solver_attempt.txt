import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "for any given source x, all combinations involving x 
    # will feature the same waveplate y, and different sources will feature 
    # different waveplates."
    # This means each source x is uniquely mapped to a waveplate y.
    # Let's store the mapping source -> waveplate and the list of sensors for each source.
    
    source_to_waveplate = {}
    source_to_sensors = {}
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        source_to_waveplate[x] = y
        if x not in source_to_sensors:
            source_to_sensors[x] = []
        source_to_sensors[x].append(z)
        
    # We need to select one sensor for each source such that all sensors are distinct.
    # This is a bipartite matching problem.
    # Left side: sources (1 to k), Right side: sensors (1 to k).
    # Edge exists if source x can use sensor z.
    
    adj = [[] for _ in range(k + 1)]
    for x in source_to_sensors:
        for z in source_to_sensors[x]:
            adj[x].append(z)
            
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        if i in source_to_sensors:
            visited = [False] * (k + 1)
            if dfs(i, visited):
                count += 1
    
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            y = source_to_waveplate[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
