import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Stage 1: Each target c is paired with a unique guide sensor g.
    # We can map each target c to its unique guide sensor g.
    # The input guarantees that for a given c, g is always the same.
    target_to_guide = {}
    
    # We need to select k triples (c, g, s) such that:
    # 1. Each c in {1..k} is used exactly once.
    # 2. Each g in {1..k} is used exactly once.
    # 3. Each s in {1..k} is used exactly once.
    # Since each c is paired with a unique g, selecting a triple (c, g, s)
    # is equivalent to selecting a pair (c, s) because g is fixed by c.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        c = int(input_data[idx])
        g = int(input_data[idx+1])
        s = int(input_data[idx+2])
        idx += 3
        
        target_to_guide[c] = g
        # We have a bipartite matching problem between targets (c) and spectrometers (s).
        # Each edge (c, s) represents a configuration (c, target_to_guide[c], s).
        adj[c].append((s, g))
        
    # Hopcroft-Karp or simple DFS based matching
    match_s = [-1] * (k + 1)
    match_c = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v, g in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_s[v] < 0 or dfs(match_s[v], visited):
                    match_s[v] = u
                    match_c[u] = v
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for s in range(1, k + 1):
            c = match_s[s]
            g = target_to_guide[c]
            print(f"{c} {g} {s}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
