import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The rule implies that the allowed combinations form a set of disjoint 
    # "blocks". Within each block, there is a 1-to-1 mapping between 
    # specialists (X) and machines (Y).
    # Let's group by X and Y.
    
    x_to_y = {}
    y_to_x = {}
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        triples.append((x, y, z))
        idx += 3
        
        if x in x_to_y and x_to_y[x] != y:
            print("NO")
            return
        if y in y_to_x and y_to_x[y] != x:
            print("NO")
            return
        x_to_y[x] = y
        y_to_x[y] = x

    # Now we have a set of pairs (x, y) that are linked.
    # For each pair (x, y), we have a list of z's.
    # We need to pick exactly one z for each (x, y) such that all z are distinct.
    
    xy_to_z = {}
    for x, y, z in triples:
        if (x, y) not in xy_to_z:
            xy_to_z[(x, y)] = []
        xy_to_z[(x, y)].append(z)
        
    # We have k pairs (x, y). Let's identify them.
    # Since every x is mapped to a unique y, we can just use x as the key.
    # The pairs are (i, x_to_y[i]) for i in 1..k.
    
    pairs = []
    for i in range(1, k + 1):
        if i not in x_to_y:
            print("NO")
            return
        pairs.append((i, x_to_y[i]))
        
    # Now we have k pairs, each with a list of possible z's.
    # We need to pick one z for each pair such that all z are distinct.
    # This is a bipartite matching problem.
    # Left side: pairs (1..k), Right side: z (1..k).
    
    adj = [[] for _ in range(k)]
    for i in range(k):
        pair = pairs[i]
        adj[i] = xy_to_z[pair]
        
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(k):
        if dfs(i, [False] * (k + 1)):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x, y = pairs[match[z]]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
