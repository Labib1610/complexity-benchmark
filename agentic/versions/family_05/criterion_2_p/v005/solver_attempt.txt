import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states each specialist x is fixed to a specific machine y.
    # Let's map each specialist x to their machine y.
    spec_to_machine = {}
    # We need to select one (x, y, z) for each x in 1..k.
    # Since each x is fixed to one y, we can group allowed combinations by x.
    # Let adj[x] be the list of time slots z available for specialist x.
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        spec_to_machine[x] = y
        idx += 3
        
    # We need to match each x to a unique z.
    # This is a bipartite matching problem between specialists and time slots.
    # Since k <= 1200, Hopcroft-Karp or simple DFS-based matching is fine.
    
    match_z = [-1] * (k + 1)
    
    def dfs(u, visited):
        for y, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match_z[z] < 0 or dfs(match_z[z], visited):
                    match_z[z] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the result
        # match_z[z] = x
        results = []
        for z in range(1, k + 1):
            x = match_z[z]
            y = spec_to_machine[x]
            results.append(f"{x} {y} {z}")
        print("\n".join(results))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
