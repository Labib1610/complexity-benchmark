import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # pilot_to_aircraft[pilot] = aircraft
    # Since each pilot is certified for exactly one aircraft,
    # we first determine this mapping.
    pilot_to_aircraft = {}
    
    # adj[pilot] = list of flight paths
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in pilot_to_aircraft:
            pilot_to_aircraft[x] = y
        
        adj[x].append((y, z))
        
    # Check if all pilots have an aircraft assigned
    if len(pilot_to_aircraft) < k:
        print("NO")
        return
    
    # We need to match each pilot to a unique flight path z.
    # Since pilot x is fixed to aircraft y = pilot_to_aircraft[x],
    # we just need to find a perfect matching between pilots and flight paths.
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    # Build adjacency list for bipartite matching: pilot -> flight path
    graph = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for _, z in adj[x]:
            graph[x].append(z)
            
    def dfs(u, visited, match):
        for v in graph[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited, match):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match):
            count += 1
            
    if count == k:
        print("YES")
        # match[z] = pilot
        for z in range(1, k + 1):
            pilot = match[z]
            aircraft = pilot_to_aircraft[pilot]
            print(f"{pilot} {aircraft} {z}")
    else:
        print("NO")

solve()
