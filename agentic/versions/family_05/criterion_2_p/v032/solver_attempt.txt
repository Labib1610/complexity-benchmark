import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Any ground station x always uses the same encryption key y.
    # 2. No two ground stations use the same encryption key.
    # This means there is a fixed mapping from ground station to encryption key.
    # Let's store the links grouped by ground station.
    
    gs_to_key = {}
    gs_to_satellites = {}
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in gs_to_satellites:
            gs_to_satellites[x] = []
            gs_to_key[x] = y
        gs_to_satellites[x].append(z)
        
    # We need to select exactly one link for each ground station x from 1 to k.
    # This means we must have exactly k ground stations present in the input.
    if len(gs_to_satellites) < k:
        print("NO")
        return
    
    # We need to select one satellite z for each ground station x such that
    # all selected satellites are distinct (1 to k).
    # This is a bipartite matching problem.
    # Left side: ground stations 1..k
    # Right side: satellites 1..k
    # Edge exists if (x, y, z) is in the catalog.
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        if x not in gs_to_satellites:
            print("NO")
            return
        for z in gs_to_satellites[x]:
            adj[x].append(z)
            
    match = [0] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] == 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        results = []
        for z in range(1, k + 1):
            x = match[z]
            y = gs_to_key[x]
            results.append(f"{x} {y} {z}")
        print("\n".join(results))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
