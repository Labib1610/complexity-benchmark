import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each Data Source x is hardwired to a unique Encryption Module y.
    # We can group triads by x. Since each x has a unique y,
    # we can represent the viable triads as a list of (y, z) for each x.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        idx += 3
        
    # We need to pick exactly one triad for each x from 1 to k.
    # Let the chosen triad for x be (x, y_x, z_x).
    # Since each x maps to a unique y, the set of y_x will automatically be {1, ..., k}.
    # We only need to ensure that the set of z_x is also {1, ..., k}.
    # This is equivalent to finding a system of distinct representatives (SDR)
    # or a perfect matching in a bipartite graph where one side is x (1..k)
    # and the other side is z (1..k).
    
    # Build bipartite graph: x -> z
    # Since each x has multiple possible z's, we use Hopcroft-Karp or simple DFS matching.
    # Given k <= 1200, DFS matching with a visited array is sufficient.
    
    match_z = [-1] * (k + 1)
    match_x = [-1] * (k + 1)
    
    # To reconstruct the triad, we need to store which (y, z) was chosen for x
    chosen_triad = [None] * (k + 1)
    
    def dfs(u, visited):
        for y, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match_z[z] < 0 or dfs(match_z[z], visited):
                    match_z[z] = u
                    match_x[u] = z
                    chosen_triad[u] = (y, z)
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for i in range(1, k + 1):
            y, z = chosen_triad[i]
            print(f"{i} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
