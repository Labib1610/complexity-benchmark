import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states each protein x is bonded to exactly one RNA y.
    # We can map each x to its unique y.
    x_to_y = {}
    # We need to select k triples (x, y, z) such that each x, y, z is used once.
    # Since x and y are already paired, we just need to select one z for each x
    # such that all z are distinct.
    # Let's store viable z's for each x.
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        x_to_y[x] = y
        idx += 3
        
    # This is a bipartite matching problem:
    # Left side: proteins 1..k
    # Right side: cell cultures 1..k
    # Edge exists if (x, y, z) is a viable combination.
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    def dfs(u):
        for y, z in adj[u]:
            if not vis[z]:
                vis[z] = True
                if match[z] < 0 or dfs(match[z]):
                    match[z] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        vis = [False] * (k + 1)
        if dfs(i):
            count += 1
            
    if count == k:
        print("YES")
        results = []
        for z in range(1, k + 1):
            x = match[z]
            y = x_to_y[x]
            results.append(f"{x} {y} {z}")
        print("\n".join(results))
    else:
        print("NO")

solve()
