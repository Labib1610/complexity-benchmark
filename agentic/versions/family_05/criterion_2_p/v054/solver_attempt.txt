```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # core_to_mod[core] = modulator
    # core_mod_to_stabs[core, modulator] = list of stabilizers
    core_to_mod = {}
    core_mod_to_stabs = {}
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in core_to_mod:
            core_to_mod[x] = y
        
        pair = (x, y)
        if pair not in core_mod_to_stabs:
            core_mod_to_stabs[pair] = []
        core_mod_to_stabs[pair].append(z)
        
    # We need to pick exactly one stabilizer for each (core, modulator) pair
    # such that all stabilizers 1..k are used exactly once.
    # Since each core has exactly one modulator, we have k pairs (core_i, mod_i).
    # We need to match each pair to a unique stabilizer.
    
    if len(core_to_mod) < k:
        print("NO")
        return
    
    # Bipartite matching: pairs (core, mod) to stabilizers
    # adj[pair_index] = list of available stabilizers
    adj = []
    for i in range(1, k + 1):
        m = core_to_mod.get(i)
        if m is None:
            print("NO")
            return
        adj.append(core_mod_to_stabs.get((i, m), []))
        
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(k):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
        else:
            print("NO")
            return
            
    if count == k:
        print("YES")
        results = []
        for stab in range(1, k + 1):
            core = match[stab] + 1
            mod = core_to_mod[core]
            results.append(f"{core} {mod} {stab}")
        print("\n".join(results))
    else:
        print("NO")

solve()
```
