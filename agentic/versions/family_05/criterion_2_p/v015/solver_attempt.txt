import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each driver is assigned to exactly one truck.
    # We can map each driver to their unique truck.
    driver_to_truck = {}
    # adj[driver] = list of routes
    adj = [[] for _ in range(k + 1)]
    # To reconstruct the plan: (driver, truck, route)
    # We store (route, plan_index)
    plans = []
    
    idx = 2
    for i in range(t):
        d = int(input_data[idx])
        tr = int(input_data[idx+1])
        r = int(input_data[idx+2])
        idx += 3
        
        driver_to_truck[d] = tr
        adj[d].append((r, i))
        plans.append((d, tr, r))
        
    # We need to select one route for each driver such that all routes are distinct.
    # This is a bipartite matching problem:
    # Left side: Drivers (1 to k)
    # Right side: Routes (1 to k)
    # Edge exists if driver d can take route r.
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    # adj_list for matching
    graph = [[] for _ in range(k + 1)]
    for i in range(t):
        d, tr, r = plans[i]
        graph[d].append((r, i))
        
    def dfs(u, visited, match_r):
        for r, plan_idx in graph[u]:
            if not visited[r]:
                visited[r] = True
                if match_r[r] < 0 or dfs(match_r[r], visited, match_r):
                    match_r[r] = u
                    return True
        return False

    match_r = [-1] * (k + 1)
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match_r):
            count += 1
            
    if count == k:
        print("YES")
        # match_r[route] = driver
        # We need to output (driver, truck, route)
        # Reconstruct which plan was used
        results = []
        for r in range(1, k + 1):
            d = match_r[r]
            tr = driver_to_truck[d]
            results.append(f"{d} {tr} {r}")
        print("\n".join(results))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
