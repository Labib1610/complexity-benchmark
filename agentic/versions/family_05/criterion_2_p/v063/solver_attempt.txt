import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    links = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        links.append((x, y, z))
        idx += 3
        
    # The problem states:
    # 1. Each ground station x is hardwired to exactly one encryption key y.
    # 2. No two ground stations share the same encryption key.
    # This means the mapping x -> y is a bijection.
    # We need to select k links (x_i, y_i, z_i) such that:
    # - All x_i are distinct (1 to k)
    # - All y_i are distinct (1 to k)
    # - All z_i are distinct (1 to k)
    
    # Since x -> y is a bijection, selecting k links with distinct x 
    # automatically ensures distinct y.
    # We just need to select one link for each x in {1..k} such that 
    # the chosen satellites z are all distinct.
    
    # Group links by ground station x
    adj = [[] for _ in range(k + 1)]
    for x, y, z in links:
        adj[x].append(z)
        
    # We need to pick one z for each x such that all picked z are distinct.
    # This is a perfect matching problem in a bipartite graph:
    # Left side: ground stations {1..k}
    # Right side: satellites {1..k}
    # Edge exists if there is a link (x, y, z)
    
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the links
        # match[satellite] = ground_station
        # We need to find the y for each x. Since x -> y is fixed, 
        # we can store the mapping.
        x_to_y = {}
        for x, y, z in links:
            x_to_y[x] = y
            
        for sat in range(1, k + 1):
            gs = match[sat]
            print(f"{gs} {x_to_y[gs]} {sat}")
    else:
        print("NO")

solve()
