import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each control board x is locked to exactly one sensor array y.
    # 2. No two control boards share the same sensor array.
    # This means the triples are effectively (x, y_x, z) where y_x is a function of x.
    # Since we need to use every x, y, and z exactly once, and each x is tied to a unique y,
    # we just need to select one z for each x such that all selected z's are distinct.
    
    # Map each x to its unique y and a list of possible z's
    x_to_y = {}
    x_to_zs = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        x_to_y[x] = y
        x_to_zs[x].append(z)
        
    # Check if all x from 1 to k are present
    if len(x_to_y) < k:
        print("NO")
        return
    
    # We need to pick one z for each x such that all z are distinct.
    # This is a bipartite matching problem:
    # Left side: control boards (1 to k)
    # Right side: power modules (1 to k)
    # Edge exists if (x, y, z) is a certified triple.
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for z in x_to_zs[x]:
            adj[x].append(z)
            
    match = [0] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] == 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            print(f"{x} {x_to_y[x]} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
