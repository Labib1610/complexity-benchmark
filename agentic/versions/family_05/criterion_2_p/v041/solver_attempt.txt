```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Captains are 1 to k. Each captain is tied to exactly one jet.
    # We can map each captain to their unique jet.
    cap_to_jet = {}
    # We need to select one triplet (c, j, d) for each captain c.
    # Since each captain has a fixed jet, we just need to pick a destination d
    # for each captain such that all destinations are distinct.
    # This is a bipartite matching problem between captains and destinations.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        c = int(input_data[idx])
        j = int(input_data[idx+1])
        d = int(input_data[idx+2])
        idx += 3
        
        cap_to_jet[c] = j
        adj[c].append(d)
        
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        if i not in cap_to_jet:
            print("NO")
            return
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for d in range(1, k + 1):
            c = match[d]
            print(f"{c} {cap_to_jet[c]} {d}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
