import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "For each agent, every listed combination uses the same alias, 
    # and different agents use different aliases."
    # This means there is a fixed mapping from agent x to alias y.
    # Let's find this mapping.
    agent_to_alias = {}
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        agent_to_alias[x] = y
        # We need to pick one frequency z for each agent x.
        # The alias y is fixed for x.
        # So we just need to match agents to frequencies such that (x, y, z) is allowed.
        adj[x].append((y, z))
        
    # Now we have a bipartite matching problem between agents (1..k) and frequencies (1..k).
    # An edge exists between agent x and frequency z if (x, agent_to_alias[x], z) is allowed.
    
    match_f = [-1] * (k + 1)
    match_a = [-1] * (k + 1)
    
    def dfs(u, visited):
        for y, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match_f[z] < 0 or dfs(match_f[z], visited):
                    match_f[z] = u
                    match_a[u] = z
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for x in range(1, k + 1):
            y = agent_to_alias[x]
            z = match_a[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
