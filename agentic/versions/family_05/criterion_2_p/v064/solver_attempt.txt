import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each specialist x is assigned to exactly one machine y.
    # We can determine this mapping from the input.
    spec_to_machine = {}
    # Store available time slots for each specialist
    spec_to_times = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        spec_to_machine[x] = y
        spec_to_times[x].append((y, z))
        
    # Check if all specialists have a machine assigned
    if len(spec_to_machine) < k:
        print("NO")
        return
    
    # We need to select one (x, y, z) for each x such that all z are distinct.
    # This is a bipartite matching problem:
    # Left side: specialists (1 to k)
    # Right side: time slots (1 to k)
    # Edge exists if specialist x can work at time z.
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for y, z in spec_to_times[x]:
            adj[x].append(z)
            
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # match[time] = specialist
        for z in range(1, k + 1):
            x = match[z]
            y = spec_to_machine[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
