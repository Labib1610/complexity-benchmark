import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Stage 1 guarantees crew x is bound to airframe y.
    # We can map each crew x to their unique airframe y.
    crew_to_airframe = {}
    # We need to select one proposal for each crew x.
    # Since each crew x is bound to a unique airframe y,
    # selecting a proposal for each crew x automatically satisfies
    # the "every crew" and "every airframe" constraints.
    # We only need to ensure that the routes z are a permutation of 1..k.
    
    # proposals_by_crew[x] = list of (y, z)
    proposals_by_crew = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        proposals_by_crew[x].append((y, z))
        crew_to_airframe[x] = y
        idx += 3
        
    # This is a bipartite matching problem:
    # Left side: crews (1 to k)
    # Right side: routes (1 to k)
    # Edge exists if there is a proposal (x, y, z)
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for _, z in proposals_by_crew[x]:
            adj[x].append(z)
            
    match_route = [0] * (k + 1)
    match_crew = [0] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_route[v] == 0 or dfs(match_route[v], visited):
                    match_route[v] = u
                    match_crew[u] = v
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for x in range(1, k + 1):
            y = crew_to_airframe[x]
            z = match_crew[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
