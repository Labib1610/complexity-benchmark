import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # specialist_to_machine maps specialist x to their unique machine y
    specialist_to_machine = {}
    # adj[specialist] = list of (machine, time_slot, original_index)
    # Since each specialist has only one machine, we can simplify.
    # We need to select one (x, y, z) for each x in 1..k such that
    # all y are distinct (guaranteed by problem) and all z are distinct.
    
    # specialist_configs[x] = list of (y, z, original_input_line)
    specialist_configs = [[] for _ in range(k + 1)]
    
    idx = 2
    for i in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        specialist_configs[x].append((y, z))
        specialist_to_machine[x] = y
        idx += 3
        
    # We need to pick one z for each x such that all z are distinct.
    # This is a bipartite matching problem.
    # Left side: specialists 1..k
    # Right side: time slots 1..k
    # Edge exists if specialist x can work at time slot z.
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    # adj_list for matching
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        if x not in specialist_to_machine:
            print("NO")
            return
        for _, z in specialist_configs[x]:
            adj[x].append(z)
            
    def dfs(u, visited, match_r):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_r[v] < 0 or dfs(match_r[v], visited, match_r):
                    match_r[v] = u
                    return True
        return False

    match_r = [-1] * (k + 1)
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match_r):
            count += 1
            
    if count == k:
        print("YES")
        # match_r[z] = x
        # We need to output (x, machine, z)
        for z in range(1, k + 1):
            x = match_r[z]
            y = specialist_to_machine[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
