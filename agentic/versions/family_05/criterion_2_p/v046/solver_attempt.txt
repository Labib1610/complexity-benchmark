import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Since each pilot is assigned to exactly one aircraft,
    # we can map each pilot to their unique aircraft.
    pilot_to_aircraft = {}
    # We need to select one window for each pilot-aircraft pair.
    # This is equivalent to a bipartite matching problem:
    # Left side: Pilots (1 to k)
    # Right side: Windows (1 to k)
    # Edge exists if (pilot, aircraft, window) is in the input.
    
    adj = [[] for _ in range(k + 1)]
    edges = []
    
    idx = 2
    for _ in range(t):
        p = int(input_data[idx])
        a = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        
        pilot_to_aircraft[p] = a
        adj[p].append((w, p, a))
        
    # Hopcroft-Karp or simple DFS-based matching
    match = [0] * (k + 1)
    vis = [0] * (k + 1)
    
    def dfs(u, tag):
        for w, p, a in adj[u]:
            if vis[w] != tag:
                vis[w] = tag
                if match[w] == 0 or dfs(match[w], tag):
                    match[w] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        if dfs(i, i):
            count += 1
            
    if count == k:
        print("YES")
        results = []
        for w in range(1, k + 1):
            p = match[w]
            a = pilot_to_aircraft[p]
            results.append(f"{p} {a} {w}")
        print("\n".join(results))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
