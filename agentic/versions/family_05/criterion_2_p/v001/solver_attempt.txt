```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "For each pilot, every listed combination uses the same aircraft, 
    # and different pilots use different aircraft."
    # This means pilot x is fixed to a specific aircraft y.
    # Let's map pilot x to aircraft A[x].
    
    pilot_to_aircraft = {}
    # pilot_routes[pilot_x] = list of routes z
    pilot_routes = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        pilot_to_aircraft[x] = y
        pilot_routes[x].append(z)
        
    # We need to select one route z for each pilot x such that all routes 1..k are used.
    # This is a bipartite matching problem:
    # Left side: pilots 1..k
    # Right side: routes 1..k
    # Edge exists if (x, A[x], z) is an allowed combination.
    
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in pilot_routes[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for route in range(1, k + 1):
            pilot = match[route]
            print(f"{pilot} {pilot_to_aircraft[pilot]} {route}")
    else:
        print("NO")

solve()
```
