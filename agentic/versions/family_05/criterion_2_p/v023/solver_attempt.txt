import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # noble_to_chair[noble] = chair
    # Since each noble has exactly one chair, we can map noble -> chair
    # The input guarantees each noble has a fixed chair.
    noble_to_chair = {}
    
    # We need to select k triplets (x, y, z) such that:
    # 1. Each noble x is used exactly once (1 to k)
    # 2. Each chair y is used exactly once (1 to k)
    # 3. Each platter z is used exactly once (1 to k)
    # Since noble x is fixed to chair y, we only need to match noble x to platter z.
    
    # adj[noble] = list of platters
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        noble_to_chair[x] = y
        adj[x].append((y, z))
        
    # Bipartite matching: Noble -> Platter
    # We need to find a perfect matching in a bipartite graph where
    # left side is nobles, right side is platters.
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    def dfs(u, visited, match_r, adj_list):
        for v_chair, v_platter in adj_list[u]:
            if not visited[v_platter]:
                visited[v_platter] = True
                if match_r[v_platter] < 0 or dfs(match_r[v_platter], visited, match_r, adj_list):
                    match_r[v_platter] = u
                    return True
        return False

    match_r = [-1] * (k + 1)
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match_r, adj):
            count += 1
            
    if count == k:
        print("YES")
        # match_r[platter] = noble
        # We need to output (noble, chair, platter)
        for p in range(1, k + 1):
            noble = match_r[p]
            chair = noble_to_chair[noble]
            print(f"{noble} {chair} {p}")
    else:
        print("NO")

solve()
