import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each agent is permanently assigned to one vehicle.
    # 2. No vehicle is featured in entries for more than one agent.
    # This means for each agent x, there is exactly one vehicle y such that
    # all triplets (x, y, z) for that agent have the same y.
    
    # We need to select k triplets (x, y, z) such that:
    # - Each agent 1..k is used once.
    # - Each vehicle 1..k is used once.
    # - Each sector 1..k is used once.
    
    # Since each agent is tied to one vehicle, selecting one triplet for each
    # agent automatically satisfies the "each agent used once" and "each vehicle
    # used once" constraints (because each agent has a unique vehicle).
    # We just need to pick one sector z for each agent x such that all z are distinct.
    
    # Map agent to their fixed vehicle
    agent_to_vehicle = {}
    # Map agent to list of available sectors
    agent_to_sectors = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        agent_to_vehicle[x] = y
        agent_to_sectors[x].append(z)
        
    # We need to match agents 1..k to sectors 1..k
    # This is a bipartite matching problem.
    # Left side: agents 1..k
    # Right side: sectors 1..k
    # Edge exists if agent x can perform sector z
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    def dfs(u, visited, match_r, adj):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_r[v] < 0 or dfs(match_r[v], visited, match_r, adj):
                    match_r[v] = u
                    return True
        return False

    match_r = [-1] * (k + 1)
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match_r, agent_to_sectors):
            count += 1
            
    if count == k:
        print("YES")
        results = []
        for sector in range(1, k + 1):
            agent = match_r[sector]
            vehicle = agent_to_vehicle[agent]
            results.append(f"{agent} {vehicle} {sector}")
        print("\n".join(results))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
