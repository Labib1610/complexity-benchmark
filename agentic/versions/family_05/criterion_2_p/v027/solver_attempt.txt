import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each protein x is associated with exactly one receptor y.
    # 2. No two distinct proteins x1, x2 are associated with the same receptor y.
    # This means the mapping x -> y is a bijection.
    # We can represent this as a set of pairs (x, y).
    # A core set must contain exactly k triplets (x, y, z) such that:
    # - Every x in {1..k} appears exactly once.
    # - Every y in {1..k} appears exactly once.
    # - Every z in {1..k} appears exactly once.
    
    # Since x -> y is a bijection, if we pick a triplet (x, y, z),
    # we have already satisfied the requirement for x and y.
    # We just need to ensure that for each z in {1..k}, we pick exactly one
    # triplet (x, y, z) such that the set of chosen (x, y) pairs covers all {1..k}.
    
    # Group complexes by z:
    z_to_xy = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        z_to_xy[z].append((x, y))
        idx += 3
        
    # We need to select one (x, y) from each z_to_xy[z] such that
    # the set of selected x's is {1..k}.
    # Since x -> y is a bijection, selecting x is equivalent to selecting y.
    # This is a system of distinct representatives problem, but simpler:
    # We have k sets (one for each z), and we need to pick one element from each
    # such that all x's are distinct.
    
    # This is a bipartite matching problem:
    # Left side: z in {1..k}
    # Right side: x in {1..k}
    # Edge (z, x) exists if there is a complex (x, y, z).
    
    adj = [[] for _ in range(k + 1)]
    for z in range(1, k + 1):
        for x, y in z_to_xy[z]:
            adj[z].append((x, y))
            
    match_x = [0] * (k + 1)
    match_z = [0] * (k + 1)
    
    def dfs(u, visited):
        for x, y in adj[u]:
            if not visited[x]:
                visited[x] = True
                if match_x[x] == 0 or dfs(match_x[x], visited):
                    match_x[x] = u
                    match_z[u] = (x, y)
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        if dfs(i, [False] * (k + 1)):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x, y = match_z[z]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
