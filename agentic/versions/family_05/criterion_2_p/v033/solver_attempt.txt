import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # If (x, y, z1) and (x, y', z2) exist, then y = y'.
    # If (x1, y, z1) and (x2, y, z2) exist, then x1 = x2.
    # This means each specialist x is tied to exactly one machine y.
    # Let's group these pairs (x, y).
    
    pairs = {} # (x, y) -> list of z
    for i in range(t):
        x = int(input_data[2 + 3 * i])
        y = int(input_data[3 + 3 * i])
        z = int(input_data[4 + 3 * i])
        if (x, y) not in pairs:
            pairs[(x, y)] = []
        pairs[(x, y)].append(z)
        
    # Each specialist x must be used exactly once, and each machine y must be used exactly once.
    # Since each x is tied to a unique y, we have a set of "blocks" (x, y).
    # Let's identify these blocks.
    x_to_y = {}
    y_to_x = {}
    for (x, y) in pairs:
        x_to_y[x] = y
        y_to_x[y] = x
        
    if len(x_to_y) != k:
        print("NO")
        return
        
    # Now we have k blocks, each block i is (x_i, y_i) with a set of available z's.
    # We need to pick one z_i for each block such that all z_i are distinct (1..k).
    # This is a bipartite matching problem:
    # Left side: blocks (x_i, y_i), Right side: time slots z_j.
    # Edge exists if z_j is in pairs[(x_i, y_i)].
    
    adj = [[] for _ in range(k)]
    for i in range(k):
        # x is i+1
        x = i + 1
        y = x_to_y[x]
        for z in pairs[(x, y)]:
            adj[i].append(z - 1)
            
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * k
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(k):
        visited = [False] * k
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z_idx, x_idx in enumerate(match):
            x = x_idx + 1
            y = x_to_y[x]
            z = z_idx + 1
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
