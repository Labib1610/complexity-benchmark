import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Since each specialist x is tied to exactly one machine y,
    # we can map each specialist x to their unique machine y.
    # Let's store the allowed time slots z for each specialist x.
    
    spec_to_machine = {}
    spec_to_slots = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        spec_to_machine[x] = y
        spec_to_slots[x].append(z)
        
    # We need to select one z for each x such that all z are distinct.
    # This is a bipartite matching problem:
    # Left side: specialists 1..k
    # Right side: time slots 1..k
    # Edge exists if (x, y, z) is allowed.
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        adj[x] = spec_to_slots[x]
        
    def dfs(u, visited, match_r):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_r[v] < 0 or dfs(match_r[v], visited, match_r):
                    match_r[v] = u
                    return True
        return False

    match_r = [-1] * (k + 1)
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match_r):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match_r[z]
            y = spec_to_machine[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
