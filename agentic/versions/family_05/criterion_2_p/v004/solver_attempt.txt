import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each specialist x is assigned to exactly one machine y.
    # Let's map specialist x to machine m[x].
    # The input guarantees that for each x, all listed combinations use the same y.
    # We can store the available time slots z for each specialist x.
    
    spec_to_machine = {}
    spec_to_slots = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in spec_to_machine:
            spec_to_machine[x] = y
        spec_to_slots[x].append(z)
        
    # If any specialist has no machine assigned, we can't form k crews.
    if len(spec_to_machine) < k:
        print("NO")
        return
    
    # We need to pick one z for each x such that all z are distinct.
    # This is a bipartite matching problem:
    # Left side: specialists 1..k
    # Right side: time slots 1..k
    # Edge exists if (x, m[x], z) is an allowed combination.
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for z in spec_to_slots[x]:
            adj[x].append(z)
            
    match = [0] * (k + 1)
    vis = [0] * (k + 1)
    
    def dfs(u, tag):
        for v in adj[u]:
            if vis[v] != tag:
                vis[v] = tag
                if match[v] == 0 or dfs(match[v], tag):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        if dfs(i, i):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            print(f"{x} {spec_to_machine[x]} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
