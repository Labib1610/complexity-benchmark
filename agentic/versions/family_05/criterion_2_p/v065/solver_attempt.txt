import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each account 'a' is bound to exactly one VM 'v'.
    # We can map each account to its unique VM.
    acc_to_vm = {}
    # adj[a][s] stores the index of the configuration (a, v, s)
    # Since each account is bound to one VM, we only need to match
    # accounts to servers.
    adj = [[] for _ in range(k + 1)]
    configs = []
    
    idx = 2
    for i in range(t):
        a = int(input_data[idx])
        v = int(input_data[idx+1])
        s = int(input_data[idx+2])
        idx += 3
        acc_to_vm[a] = v
        adj[a].append((s, v, i))
        configs.append((a, v, s))

    # We need to select one configuration for each account such that
    # all servers are used exactly once.
    # This is a bipartite matching problem:
    # Left side: accounts 1..k
    # Right side: servers 1..k
    # Edge exists if there is a configuration (a, v, s)
    
    match_s = [-1] * (k + 1)
    match_a = [-1] * (k + 1)
    
    # Hopcroft-Karp or simple DFS based matching
    # Given k <= 1200, simple DFS with visited array is usually sufficient
    # if the graph is not adversarial.
    
    def dfs(u, visited):
        for s, v, i in adj[u]:
            if not visited[s]:
                visited[s] = True
                if match_s[s] < 0 or dfs(match_s[s], visited):
                    match_s[s] = u
                    match_a[u] = (s, v, i)
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for i in range(1, k + 1):
            s, v, idx_val = match_a[i]
            print(f"{i} {v} {s}")
    else:
        print("NO")

solve()
