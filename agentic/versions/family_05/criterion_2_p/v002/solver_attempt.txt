```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "each chef is assigned to exactly one kitchen, 
    # and no two chefs share the same kitchen".
    # This means for each chef x, there is exactly one kitchen y.
    # Let's map chef x to kitchen y.
    chef_to_kitchen = {}
    # We need to select one menu z for each chef x.
    # Since each kitchen is used exactly once, and each chef is used exactly once,
    # we just need to pick one (x, y, z) for each x such that all z are distinct.
    
    # chef_menus[x] will store a list of available menus for chef x
    chef_menus = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        chef_to_kitchen[x] = y
        chef_menus[x].append(z)
        idx += 3
        
    # Now we have a bipartite matching problem:
    # Left side: chefs 1..k
    # Right side: menus 1..k
    # Edge exists if menu z is allowed for chef x.
    
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in chef_menus[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            y = chef_to_kitchen[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
