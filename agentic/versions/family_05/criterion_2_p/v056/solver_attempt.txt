import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Since each payload x is compatible with exactly one rocket y,
    # we can map each payload to its unique rocket.
    # Let's store the viable configurations as (payload, rocket, band).
    # Since payload -> rocket is a bijection, we can treat (payload, rocket)
    # as a single entity. Let's call this entity 'pair'.
    # There are exactly k such pairs (x, y).
    
    payload_to_rocket = {}
    pair_to_bands = {}
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in payload_to_rocket:
            payload_to_rocket[x] = y
        
        pair = (x, y)
        if pair not in pair_to_bands:
            pair_to_bands[pair] = []
        pair_to_bands[pair].append(z)
        
    # We need to select one band z for each pair (x, y) such that
    # each band 1..k is used exactly once.
    # This is a bipartite matching problem:
    # Left side: k pairs (x, y)
    # Right side: k bands z
    # Edge exists if (x, y, z) is in the catalog.
    
    adj = [[] for _ in range(k)]
    pairs = sorted(payload_to_rocket.keys())
    pair_to_idx = {p: i for i, p in enumerate(pairs)}
    
    for p, bands in pair_to_bands.items():
        u = pair_to_idx[p]
        for z in bands:
            adj[u].append(z - 1)
            
    match = [-1] * k
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(k):
        visited = [False] * k
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for v in range(k):
            u = match[v]
            payload = pairs[u]
            rocket = payload_to_rocket[payload]
            band = v + 1
            print(f"{payload} {rocket} {band}")
    else:
        print("NO")

solve()
