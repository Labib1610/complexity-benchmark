import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # catalyst_to_reagent[c] = r
    # catalyst_to_solvents[c] = [list of solvents]
    # catalyst_to_indices[c] = [list of original indices]
    
    catalyst_to_reagent = {}
    catalyst_to_solvents = {}
    
    idx = 2
    for i in range(t):
        c = int(input_data[idx])
        r = int(input_data[idx+1])
        s = int(input_data[idx+2])
        idx += 3
        
        if c not in catalyst_to_reagent:
            catalyst_to_reagent[c] = r
            catalyst_to_solvents[c] = []
        
        catalyst_to_solvents[c].append((s, i))
        
    # We need to pick exactly one configuration for each catalyst 1..k.
    # Since each catalyst has a fixed reagent, picking one for each catalyst
    # automatically satisfies the "every catalyst and every reagent used once"
    # condition (given the problem constraints).
    # We just need to ensure every solvent 1..k is used exactly once.
    
    # This is a bipartite matching problem:
    # Left side: catalysts 1..k
    # Right side: solvents 1..k
    # Edge exists if catalyst c can use solvent s.
    
    adj = [[] for _ in range(k + 1)]
    for c in range(1, k + 1):
        if c in catalyst_to_solvents:
            for s, original_idx in catalyst_to_solvents[c]:
                adj[c].append((s, original_idx))
                
    match_s = [-1] * (k + 1)
    match_c = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v, original_idx in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_s[v] < 0 or dfs(match_s[v], visited):
                    match_s[v] = u
                    match_c[u] = (v, original_idx)
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        results = []
        for c in range(1, k + 1):
            s, _ = match_c[c]
            r = catalyst_to_reagent[c]
            results.append(f"{c} {r} {s}")
        print("\n".join(results))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
