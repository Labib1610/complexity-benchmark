import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # specialist_to_machine[s] = m
    # time_slots_for_specialist[s] = [list of z]
    specialist_to_machine = {}
    time_slots_for_specialist = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        s = int(input_data[idx])
        m = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        specialist_to_machine[s] = m
        time_slots_for_specialist[s].append(z)
        
    # We need to pick one z for each s such that all z are distinct.
    # This is a bipartite matching problem:
    # Left side: specialists 1..k
    # Right side: time slots 1..k
    # Edge exists if specialist s can work in time slot z.
    
    adj = [[] for _ in range(k + 1)]
    for s in range(1, k + 1):
        if s not in specialist_to_machine:
            print("NO")
            return
        for z in time_slots_for_specialist[s]:
            adj[s].append(z)
            
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            s = match[z]
            m = specialist_to_machine[s]
            print(f"{s} {m} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
