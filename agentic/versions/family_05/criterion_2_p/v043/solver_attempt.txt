```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each specialist x is associated with exactly one machine y.
    # 2. Each machine y is associated with exactly one specialist x.
    # This means we have k pairs of (specialist, machine).
    
    spec_to_mach = {}
    mach_to_spec = {}
    # groups[ (spec, mach) ] = list of time slots
    groups = {}
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x in spec_to_mach and spec_to_mach[x] != y:
            print("NO")
            return
        if y in mach_to_spec and mach_to_spec[y] != x:
            print("NO")
            return
        
        spec_to_mach[x] = y
        mach_to_spec[y] = x
        
        pair = (x, y)
        if pair not in groups:
            groups[pair] = []
        groups[pair].append(z)
        
    if len(spec_to_mach) < k:
        print("NO")
        return
    
    # We have k pairs (x, y). We need to pick one z for each pair such that
    # all z are distinct (1 to k).
    # This is a bipartite matching problem:
    # Left side: k pairs (x, y)
    # Right side: k time slots (z)
    # Edge exists if z is in groups[(x, y)]
    
    adj = [[] for _ in range(k)]
    pair_list = sorted(groups.keys())
    for i, pair in enumerate(pair_list):
        adj[i] = groups[pair]
        
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(k):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            pair = pair_list[match[z]]
            print(f"{pair[0]} {pair[1]} {z}")
    else:
        print("NO")

solve()
```
