import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each chassis x is fused to exactly one navigation unit y.
    # 2. No two different chassis share the same navigation unit.
    # This means the mapping from chassis to nav unit is a bijection.
    # Let's store the schematics as a list of lists:
    # chassis_to_nav[x] = y
    # nav_to_batteries[y] = [list of batteries z]
    
    chassis_to_nav = {}
    nav_to_batteries = {}
    schematics = []
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in chassis_to_nav:
            chassis_to_nav[x] = y
        if y not in nav_to_batteries:
            nav_to_batteries[y] = []
        nav_to_batteries[y].append((x, z))
        
    # Check if we have all k chassis and k nav units
    if len(chassis_to_nav) < k:
        print("NO")
        return
    
    # We need to pick one schematic for each chassis x.
    # Since each x maps to a unique y, we must pick one battery z for each y.
    # This is a bipartite matching problem between nav units (y) and batteries (z).
    
    adj = [[] for _ in range(k + 1)]
    for y, pairs in nav_to_batteries.items():
        for x, z in pairs:
            adj[y].append(z)
            
    # Standard Hopcroft-Karp or DFS-based matching
    match_z = [-1] * (k + 1)
    match_y = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_z[v] < 0 or dfs(match_z[v], visited):
                    match_z[v] = u
                    match_y[u] = v
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        if i in nav_to_batteries:
            visited = [False] * (k + 1)
            if dfs(i, visited):
                count += 1
    
    if count == k:
        print("YES")
        # Reconstruct the chosen schematics
        # We need to output (x, y, z)
        # We know x -> y, and we found y -> z
        # Let's map x to y
        x_to_y = chassis_to_nav
        # We have match_y[y] = z
        for x in range(1, k + 1):
            y = x_to_y[x]
            z = match_y[y]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
