import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "each operator is certified on exactly one assembly station, 
    # and no two operators are certified on the same station".
    # This means for each operator x, there is a unique station y.
    # Let's map operator x to its unique station y.
    op_to_station = {}
    
    # We need to select k triples (x, y, z) such that:
    # 1. Each x in 1..k is used once.
    # 2. Each y in 1..k is used once.
    # 3. Each z in 1..k is used once.
    # Since each x maps to a unique y, the condition "each x used once" 
    # and "each y used once" are satisfied if we pick exactly one triple for each x.
    # Let the triple for operator x be (x, station[x], shift[x]).
    # We need to pick shift[x] such that all shift[x] are distinct.
    
    # adj[x] will store a list of shifts z for operator x.
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        idx += 3
        
    # Hopcroft-Karp or simple DFS for bipartite matching
    # Left side: operators 1..k
    # Right side: shifts 1..k
    match = [-1] * (k + 1)
    
    def dfs(u, visited, current_adj):
        for v_y, v_z in current_adj[u]:
            if not visited[v_z]:
                visited[v_z] = True
                if match[v_z] < 0 or dfs(match[v_z], visited, current_adj):
                    match[v_z] = u
                    return True
        return False

    # To reconstruct the result, we need to know which y was chosen for each x.
    # The input guarantees each x has only one y.
    op_to_y_z = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for y, z in adj[x]:
            op_to_y_z[x].append((y, z))
            
    # Standard bipartite matching
    match_z_to_x = [-1] * (k + 1)
    
    def can_match():
        for i in range(1, k + 1):
            visited = [False] * (k + 1)
            if not dfs(i, visited, op_to_y_z):
                return False
        return True

    if can_match():
        print("YES")
        # match_z_to_x[z] = x
        # We need to output x, y, z
        # match[z] stores the operator x that was assigned to shift z
        results = []
        for z in range(1, k + 1):
            x = match[z]
            # Find the y associated with this x
            for y, shift in op_to_y_z[x]:
                if shift == z:
                    results.append(f"{x} {y} {z}")
                    break
        print("\n".join(results))
    else:
        print("NO")

solve()
