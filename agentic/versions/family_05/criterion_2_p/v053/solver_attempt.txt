```python
import sys

def solve():
    # Read k and t
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "For each container, every listed entry uses the same RFID tag,
    # and different containers use different RFID tags."
    # This means for each container u, there is exactly one RFID tag v associated with it.
    # Let's map container u to its unique RFID tag v.
    
    container_to_tag = {}
    # Group entries by container
    # entries_by_container[u] = list of (v, w)
    entries_by_container = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        entries_by_container[u].append((v, w))
        container_to_tag[u] = v
        idx += 3
        
    # We need to select exactly one entry (u, v, w) for each u in 1..k.
    # Since each u has a fixed v, we just need to pick one w for each u
    # such that all w are distinct (1..k).
    # This is a bipartite matching problem:
    # Left side: containers 1..k
    # Right side: vessels 1..k
    # Edge exists from u to w if there is an entry (u, v, w).
    
    adj = [[] for _ in range(k + 1)]
    for u in range(1, k + 1):
        for v, w in entries_by_container[u]:
            adj[u].append(w)
            
    # Hopcroft-Karp or simple DFS matching
    match_w = [-1] * (k + 1)
    match_u = [-1] * (k + 1)
    
    def dfs(u, visited):
        for w in adj[u]:
            if not visited[w]:
                visited[w] = True
                if match_w[w] < 0 or dfs(match_w[w], visited):
                    match_w[w] = u
                    match_u[u] = w
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for u in range(1, k + 1):
            w = match_u[u]
            v = container_to_tag[u]
            print(f"{u} {v} {w}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
