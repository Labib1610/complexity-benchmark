import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "each shipping container is permanently affixed with 
    # exactly one RFID tag, and no two containers share the same tag".
    # This means there is a fixed mapping f: container -> tag.
    # We can determine this mapping from the input.
    
    container_to_tag = {}
    # entries will store (container, tag, vessel)
    entries = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        
        if u in container_to_tag:
            if container_to_tag[u] != v:
                # This violates the problem statement, but we assume input is valid
                pass
        else:
            container_to_tag[u] = v
        entries.append((u, v, w))
        
    # We need to select k entries such that each container is used once,
    # each tag is used once, and each vessel is used once.
    # Since container-to-tag is fixed, selecting a container automatically
    # selects its tag. We just need to select one entry for each container
    # such that all vessels are distinct.
    
    # Let's group entries by container.
    # Since each container has a fixed tag, we just need to pick one entry
    # (u, v, w) for each u in 1..k such that all w are distinct.
    
    adj = [[] for _ in range(k + 1)]
    for u, v, w in entries:
        adj[u].append(w)
        
    # This is a bipartite matching problem.
    # Left side: containers 1..k
    # Right side: vessels 1..k
    # Edge exists if there is a cleared entry (u, v, w)
    
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for w in adj[u]:
            if not visited[w]:
                visited[w] = True
                if match[w] < 0 or dfs(match[w], visited):
                    match[w] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the entries
        # match[w] = u means vessel w is assigned to container u
        # We need to find the tag v for each u
        res = []
        for w in range(1, k + 1):
            u = match[w]
            v = container_to_tag[u]
            res.append(f"{u} {v} {w}")
        print("\n".join(res))
    else:
        print("NO")

solve()
