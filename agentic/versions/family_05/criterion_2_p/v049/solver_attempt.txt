```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each operator x is assigned to exactly one rig y.
    # We can map each operator to its rig.
    op_to_rig = {}
    # We need to select one record for each operator.
    # Since each operator is tied to one rig, selecting one record per operator
    # automatically selects one record per rig.
    # The constraint is that each zone must also be used exactly once.
    # This is a bipartite matching problem:
    # Left side: Operators (1 to k)
    # Right side: Zones (1 to k)
    # Edge exists if there is a record (x, y, z).
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((z, y))
        op_to_rig[x] = y
        idx += 3
        
    # Hopcroft-Karp or simple DFS matching
    match_z = [-1] * (k + 1)
    match_x = [-1] * (k + 1)
    
    def dfs(u, visited):
        for z, y in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match_z[z] < 0 or dfs(match_z[z], visited):
                    match_z[z] = u
                    match_x[u] = z
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for x in range(1, k + 1):
            z = match_x[x]
            y = op_to_rig[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
