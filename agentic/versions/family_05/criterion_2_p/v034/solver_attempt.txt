import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each catalyst has a fixed reagent.
    # Let cat_to_reagent[c] = r
    # We need to select one triad (c, r, s) for each c in 1..k
    # such that all r are distinct (guaranteed by problem statement)
    # and all s are distinct.
    
    # Since each catalyst c has a fixed reagent r, we can group triads by catalyst.
    # Let triads_by_cat[c] be a list of (r, s) pairs.
    # Since each catalyst has only one reagent, all triads for a catalyst c
    # will have the same r.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        c = int(input_data[idx])
        r = int(input_data[idx+1])
        s = int(input_data[idx+2])
        adj[c].append((r, s))
        idx += 3
        
    # We need to pick one (r, s) for each c such that all s are distinct.
    # This is a bipartite matching problem.
    # Left side: catalysts 1..k
    # Right side: solvents 1..k
    # Edge exists if there is a triad (c, r, s)
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    # Store which triad was used for the matching
    triad_map = {} # (c, s) -> r
    
    graph = [[] for _ in range(k + 1)]
    for c in range(1, k + 1):
        for r, s in adj[c]:
            graph[c].append(s)
            triad_map[(c, s)] = r
            
    def dfs(u):
        for v in graph[u]:
            if not vis[v]:
                vis[v] = True
                if match[v] < 0 or dfs(match[v]):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        vis = [False] * (k + 1)
        if dfs(i):
            count += 1
            
    if count == k:
        print("YES")
        for s in range(1, k + 1):
            c = match[s]
            r = triad_map[(c, s)]
            print(f"{c} {r} {s}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
