import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each emitter x is hardwired to exactly one filter y.
    # 2. No filter y is dedicated to more than one emitter.
    # This means there is a fixed mapping Emitter -> Filter.
    # Let's store the triplets grouped by the emitter (which implies the filter).
    
    emitter_to_triplets = {}
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        if x not in emitter_to_triplets:
            emitter_to_triplets[x] = []
        emitter_to_triplets[x].append((x, y, z))
        
    # We need to pick exactly one triplet for each emitter x in 1..k.
    # Since each emitter x has a fixed filter y, we just need to pick
    # one detector z for each emitter such that all chosen z are distinct.
    # This is a bipartite matching problem:
    # Left side: Emitters 1..k
    # Right side: Detectors 1..k
    # Edge exists from x to z if there is a triplet (x, y, z).
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        if x in emitter_to_triplets:
            for _, _, z in emitter_to_triplets[x]:
                adj[x].append(z)
                
    match = [0] * (k + 1)
    vis = [False] * (k + 1)
    
    def dfs(u):
        for v in adj[u]:
            if not vis[v]:
                vis[v] = True
                if match[v] == 0 or dfs(match[v]):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        vis = [False] * (k + 1)
        if dfs(i):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the triplets
        # match[z] = x
        results = []
        for z in range(1, k + 1):
            x = match[z]
            # Find the specific y for this x
            y = emitter_to_triplets[x][0][1]
            results.append(f"{x} {y} {z}")
        print("\n".join(results))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
