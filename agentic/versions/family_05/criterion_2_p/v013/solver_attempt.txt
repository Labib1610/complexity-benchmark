import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "all authorized configurations involving a particular 
    # specialist always feature the same machine, and no two specialists are 
    # associated with the same machine."
    # This means there is a fixed mapping from specialist x to machine y.
    # Let's store this mapping.
    
    spec_to_mach = {}
    # We need to select k configurations (x, y, z) such that:
    # 1. Each x in {1..k} is used exactly once.
    # 2. Each y in {1..k} is used exactly once.
    # 3. Each z in {1..k} is used exactly once.
    
    # Since each x maps to a unique y, if we pick one configuration for each x,
    # we automatically satisfy the condition that each x and each y is used exactly once.
    # The only constraint left is that each z must be used exactly once.
    
    # Let's group configurations by specialist x.
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        idx += 3
        
    # Since each x has a fixed y, all (y, z) pairs for a given x must have the same y.
    # Let's verify and store the mapping.
    fixed_y = {}
    for x in range(1, k + 1):
        if not adj[x]:
            print("NO")
            return
        y = adj[x][0][0]
        fixed_y[x] = y
        for item in adj[x]:
            if item[0] != y:
                # This shouldn't happen based on problem description
                print("NO")
                return
                
    # Now we need to pick one z for each x such that all z are distinct.
    # This is a bipartite matching problem:
    # Left side: specialists 1..k
    # Right side: time slots 1..k
    # Edge exists if (x, y, z) is in the database.
    
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    # Build adjacency for matching
    graph = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for _, z in adj[x]:
            graph[x].append(z)
            
    def dfs(u):
        for v in graph[u]:
            if not vis[v]:
                vis[v] = True
                if match[v] < 0 or dfs(match[v]):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        vis = [False] * (k + 1)
        if dfs(i):
            count += 1
            
    if count == k:
        print("YES")
        results = []
        for z in range(1, k + 1):
            x = match[z]
            y = fixed_y[x]
            results.append(f"{x} {y} {z}")
        print("\n".join(results))
    else:
        print("NO")

solve()
