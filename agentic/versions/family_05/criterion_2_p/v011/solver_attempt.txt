```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "each API endpoint is hardcoded to query exactly one 
    # database shard, and no two endpoints query the same shard."
    # This means for each e, there is exactly one d.
    # We can store the allowed configurations as a list of (e, d, i).
    # Since each e maps to a unique d, we can group by e.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        e = int(input_data[idx])
        d = int(input_data[idx+1])
        i = int(input_data[idx+2])
        adj[e].append((d, i))
        idx += 3
        
    # We need to select one (d, i) for each e such that all i are distinct.
    # This is a bipartite matching problem.
    # Left side: API endpoints (1..k)
    # Right side: IP addresses (1..k)
    # Edge exists between e and i if there is a configuration (e, d, i).
    
    match = [-1] * (k + 1)
    # Store which (d, i) was chosen for each e
    chosen_di = [None] * (k + 1)
    
    def dfs(u, visited):
        for d, i in adj[u]:
            if not visited[i]:
                visited[i] = True
                if match[i] < 0 or dfs(match[i], visited):
                    match[i] = u
                    chosen_di[u] = (d, i)
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for e in range(1, k + 1):
            d, i = chosen_di[e]
            print(f"{e} {d} {i}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
