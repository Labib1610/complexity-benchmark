import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each transmitter u is hardwired to exactly one module v.
    # 2. Different transmitters use different modules.
    # This means there is a fixed bijection f: {1..k} -> {1..k} such that
    # if (u, v, w) is a configuration, then v = f(u).
    
    # Let's find this mapping f.
    trans_to_mod = {}
    configs = []
    idx = 2
    for _ in range(t):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        configs.append((u, v, w))
        if u not in trans_to_mod:
            trans_to_mod[u] = v
            
    if len(trans_to_mod) < k:
        # If not all transmitters are present, we can't form k channels
        print("NO")
        return

    # Now we need to select one (u, v, w) for each u in 1..k
    # such that all w are distinct.
    # This is equivalent to finding a system of distinct representatives
    # or a perfect matching in a bipartite graph.
    # Left side: transmitters 1..k
    # Right side: frequencies 1..k
    # Edge (u, w) exists if there is a configuration (u, trans_to_mod[u], w)
    
    adj = [[] for _ in range(k + 1)]
    for u, v, w in configs:
        adj[u].append(w)
        
    # Hopcroft-Karp or simple DFS matching
    match = [0] * (k + 1)
    vis = [0] * (k + 1)
    
    def dfs(u, tag):
        for w in adj[u]:
            if vis[w] != tag:
                vis[w] = tag
                if match[w] == 0 or dfs(match[w], tag):
                    match[w] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        if dfs(i, i):
            count += 1
            
    if count == k:
        print("YES")
        # match[w] = u, so we need to invert this
        res = [None] * (k + 1)
        for w in range(1, k + 1):
            u = match[w]
            res[u] = (u, trans_to_mod[u], w)
        for i in range(1, k + 1):
            print(f"{res[i][0]} {res[i][1]} {res[i][2]}")
    else:
        print("NO")

solve()
