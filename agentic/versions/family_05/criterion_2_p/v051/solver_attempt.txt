import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Each scholar x is tied to exactly one archive y.
    # We can group proposals by scholar.
    # Since each scholar has a unique archive, we can map scholar -> archive.
    # The problem reduces to selecting one proposal for each scholar such that
    # all manuscripts are distinct.
    
    # scholar_proposals[x] = list of (y, z)
    scholar_proposals = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        scholar_proposals[x].append((y, z))
        idx += 3
        
    # Since each scholar has exactly one archive, we can simplify:
    # For each scholar, we have a set of manuscripts they can translate.
    # We need to pick one manuscript for each scholar such that all are distinct.
    # This is a bipartite matching problem:
    # Left side: scholars (1 to k)
    # Right side: manuscripts (1 to k)
    # Edge (x, z) exists if there is a proposal (x, y, z).
    
    adj = [[] for _ in range(k + 1)]
    # Store the archive y for each scholar to reconstruct the output
    scholar_archive = {}
    for x in range(1, k + 1):
        if not scholar_proposals[x]:
            print("NO")
            return
        scholar_archive[x] = scholar_proposals[x][0][0]
        for _, z in scholar_proposals[x]:
            adj[x].append(z)
            
    # Hopcroft-Karp or simple DFS matching
    match = [0] * (k + 1)
    vis = [False] * (k + 1)
    
    def dfs(u):
        for v in adj[u]:
            if not vis[v]:
                vis[v] = True
                if match[v] == 0 or dfs(match[v]):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        vis = [False] * (k + 1)
        if dfs(i):
            count += 1
            
    if count == k:
        print("YES")
        # match[manuscript] = scholar
        for z in range(1, k + 1):
            x = match[z]
            y = scholar_archive[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
