```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # complexes[protein_id] = (receptor_id, [list of enzymes])
    # Since protein-receptor mapping is fixed, we group by protein_id
    # The problem states: "any complex containing a given protein chain 
    # will always feature its uniquely associated receptor site."
    
    protein_to_receptor = {}
    protein_to_enzymes = {}
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in protein_to_receptor:
            protein_to_receptor[x] = y
            protein_to_enzymes[x] = []
        protein_to_enzymes[x].append(z)
        
    # We need to pick exactly one enzyme for each protein chain (1 to k)
    # such that all enzymes are distinct.
    # This is a bipartite matching problem:
    # Left side: protein chains (1 to k)
    # Right side: enzymes (1 to k)
    # Edge exists if protein x can use enzyme z.
    
    if len(protein_to_receptor) < k:
        print("NO")
        return
    
    # Check if all receptors are distinct (guaranteed by problem statement)
    # We need to select one enzyme for each protein such that all enzymes are distinct.
    # This is exactly a maximum bipartite matching problem.
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        if x in protein_to_enzymes:
            for z in protein_to_enzymes[x]:
                adj[x].append(z)
                
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False
    
    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            y = protein_to_receptor[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
```
