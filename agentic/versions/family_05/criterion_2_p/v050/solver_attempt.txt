import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Since each specialist x is tied to exactly one machinery y,
    # we can map each x to its unique y.
    # The problem states: "every feasible allocation involving a particular 
    # specialist will always list the exact same machinery".
    
    spec_to_mach = {}
    # adj[x] will store a list of windows z for specialist x
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        if x not in spec_to_mach:
            spec_to_mach[x] = y
        adj[x].append((y, z))
        
    # We need to pick one (y, z) for each x from 1 to k.
    # Since each x has a unique y, we just need to pick a unique z for each x.
    # This is a bipartite matching problem:
    # Left side: specialists 1..k
    # Right side: windows 1..k
    # Edge exists if there is an allocation (x, y, z)
    
    match = [-1] * (k + 1)
    
    # Hopcroft-Karp or simple DFS based matching
    # Since k <= 1200, simple DFS with visited array is sufficient
    
    def dfs(u, visited, current_match):
        for _, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if current_match[z] < 0 or dfs(current_match[z], visited, current_match):
                    current_match[z] = u
                    return True
        return False

    current_match = [-1] * (k + 1)
    for i in range(1, k + 1):
        if i not in spec_to_mach:
            print("NO")
            return
        visited = [False] * (k + 1)
        if not dfs(i, visited, current_match):
            print("NO")
            return
            
    print("YES")
    # current_match[z] = x
    # We need to output x y z
    for z in range(1, k + 1):
        x = current_match[z]
        y = spec_to_mach[x]
        print(f"{x} {y} {z}")

solve()
