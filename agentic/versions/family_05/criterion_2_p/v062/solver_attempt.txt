```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # p_to_a_m maps p -> (a, [list of m])
    p_to_a_m = {}
    idx = 2
    for _ in range(t):
        p = int(input_data[idx])
        a = int(input_data[idx+1])
        m = int(input_data[idx+2])
        idx += 3
        
        if p not in p_to_a_m:
            p_to_a_m[p] = (a, [])
        p_to_a_m[p][1].append(m)
        
    # The problem states:
    # 1. For a given p, all states have the same a.
    # 2. Different p have different a.
    # This means each p is associated with exactly one a.
    # To pick k states such that every p, a, m appears exactly once:
    # We must pick exactly one state for each p from 1 to k.
    # Since each p has a unique a, picking one state for each p 
    # automatically satisfies the "every a from 1 to k appears exactly once" 
    # condition IF AND ONLY IF we have exactly k distinct p values 
    # present in the input, and each p has at least one state.
    
    if len(p_to_a_m) < k:
        print("NO")
        return
    
    # We need to pick one m for each p such that all m are distinct.
    # This is a bipartite matching problem.
    # Left side: p in {1..k}, Right side: m in {1..k}
    # Edge exists if (p, a, m) is an observed state.
    
    adj = [[] for _ in range(k + 1)]
    for p, (a, m_list) in p_to_a_m.items():
        for m in m_list:
            adj[p].append(m)
            
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        if i in p_to_a_m:
            visited = [False] * (k + 1)
            if dfs(i, visited):
                count += 1
    
    if count == k:
        print("YES")
        # match[m] = p
        # We need to output p, a, m
        # Reconstruct a from p
        p_a_map = {p: data[0] for p, data in p_to_a_m.items()}
        for m in range(1, k + 1):
            p = match[m]
            print(f"{p} {p_a_map[p]} {m}")
    else:
        print("NO")

solve()
```
