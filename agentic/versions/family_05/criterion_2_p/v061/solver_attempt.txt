import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "any trial run that tested a particular reagent 
    # necessarily took place in the specific reaction vessel that held it."
    # This means there is a fixed mapping f: reagent -> vessel.
    # Let's find this mapping f.
    # If a reagent x is used in multiple trials, they must all have the same y.
    
    reagent_to_vessel = {}
    trials = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        trials.append((x, y, z))
        if x in reagent_to_vessel:
            if reagent_to_vessel[x] != y:
                # This case shouldn't happen based on problem description
                print("NO")
                return
        else:
            reagent_to_vessel[x] = y
            
    # If some reagents were never used, we can't form a protocol.
    if len(reagent_to_vessel) < k:
        print("NO")
        return
    
    # Now we need to select exactly one trial for each reagent x (which implies y = f(x))
    # such that each z is used exactly once.
    # This is equivalent to finding a perfect matching in a bipartite graph
    # where one set is {1..k} (reagents) and the other is {1..k} (pressures).
    # An edge exists between x and z if there is a trial (x, f(x), z).
    
    adj = [[] for _ in range(k + 1)]
    trial_map = {}
    for x, y, z in trials:
        adj[x].append(z)
        trial_map[(x, z)] = (x, y, z)
        
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for z in range(1, k + 1):
            x = match[z]
            y = reagent_to_vessel[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
