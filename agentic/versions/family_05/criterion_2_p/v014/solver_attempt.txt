```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each operator x is associated with exactly one rig y.
    # 2. No two operators are associated with the same rig.
    # This means there is a fixed mapping f: operator -> rig.
    # We can determine this mapping from the input.
    
    op_to_rig = {}
    # Store configurations as (operator, shift) -> rig
    # Actually, the input is (x, y, z) where x=op, y=rig, z=shift.
    # We need to select k triples (x, y, z) such that:
    # - Each x in {1..k} appears once.
    # - Each y in {1..k} appears once.
    # - Each z in {1..k} appears once.
    
    # Since x determines y, we can simplify the problem.
    # Let f(x) = y. We must use every x exactly once, which implies
    # we must use every y exactly once.
    # The problem reduces to: for each x, pick exactly one z such that
    # (x, f(x), z) is an approved configuration, and all chosen z are distinct.
    
    adj = [[] for _ in range(k + 1)]
    op_to_rig = {}
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        op_to_rig[x] = y
        adj[x].append(z)
        
    # Now we have a bipartite matching problem:
    # Left side: operators 1..k
    # Right side: shifts 1..k
    # Edge exists if (x, f(x), z) is approved.
    
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        if i not in op_to_rig:
            print("NO")
            return
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
        else:
            print("NO")
            return
            
    if count == k:
        print("YES")
        results = []
        for z in range(1, k + 1):
            x = match[z]
            y = op_to_rig[x]
            results.append(f"{x} {y} {z}")
        print("\n".join(results))
    else:
        print("NO")

solve()
```
