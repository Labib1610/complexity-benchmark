```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: "each satellite is permanently stationed in exactly one 
    # orbital slot, and no two satellites occupy the same slot."
    # This means for each satellite x, there is exactly one y such that (x, y, z) 
    # is a viable setup. We can pre-process this to map each satellite to its slot.
    
    sat_to_slot = {}
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        sat_to_slot[x] = y
        # We need to pick one setup for each satellite x.
        # The setup is (x, sat_to_slot[x], z).
        # We need to pick exactly one z for each x such that all z are distinct.
        # This is a bipartite matching problem:
        # Left side: satellites 1..k
        # Right side: frequencies 1..k
        # Edge exists if (x, sat_to_slot[x], z) is a viable setup.
        adj[x].append(z)
        
    # Hopcroft-Karp or simple DFS matching
    match = [-1] * (k + 1)
    vis = [False] * (k + 1)
    
    def dfs(u, visited, match):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited, match):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match):
            count += 1
            
    if count == k:
        print("YES")
        # match[freq] = sat
        for f in range(1, k + 1):
            s = match[f]
            print(f"{s} {sat_to_slot[s]} {f}")
    else:
        print("NO")

solve()
```
