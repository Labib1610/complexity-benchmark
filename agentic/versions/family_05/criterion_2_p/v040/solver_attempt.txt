import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # drone_to_station maps drone_id to station_id
    # drone_to_freqs maps drone_id to a list of available frequencies
    drone_to_station = {}
    drone_to_freqs = {}
    
    idx = 2
    for _ in range(t):
        d = int(input_data[idx])
        c = int(input_data[idx+1])
        f = int(input_data[idx+2])
        idx += 3
        
        if d not in drone_to_station:
            drone_to_station[d] = c
            drone_to_freqs[d] = []
        drone_to_freqs[d].append(f)
        
    if len(drone_to_station) < k:
        print("NO")
        return

    # We need to pick one frequency for each drone such that all frequencies are distinct.
    # This is a bipartite matching problem:
    # Left side: drones (1 to k)
    # Right side: frequencies (1 to k)
    # Edge exists if frequency f is available for drone d.
    
    adj = [[] for _ in range(k + 1)]
    for d in range(1, k + 1):
        for f in drone_to_freqs[d]:
            adj[d].append(f)
            
    match = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match[v] < 0 or dfs(match[v], visited):
                    match[v] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for f in range(1, k + 1):
            d = match[f]
            c = drone_to_station[d]
            print(f"{d} {c} {f}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
