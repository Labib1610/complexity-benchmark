```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states: each route is strictly paired with exactly one vehicle.
    # Let's map route x to its unique vehicle y.
    route_to_vehicle = {}
    # We need to match each route (and thus its vehicle) to a unique driver.
    # This is a bipartite matching problem between routes and drivers.
    # adj[route] = list of drivers
    adj = [[] for _ in range(k + 1)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        
        route_to_vehicle[x] = y
        adj[x].append((y, z))
        
    # Hopcroft-Karp or simple DFS matching
    match_driver_to_route = [-1] * (k + 1)
    
    def dfs(u, visited, current_match):
        for v_y, v_z in adj[u]:
            if not visited[v_z]:
                visited[v_z] = True
                if current_match[v_z] < 0 or dfs(current_match[v_z], visited, current_match):
                    current_match[v_z] = u
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited, match_driver_to_route):
            count += 1
            
    if count == k:
        print("YES")
        # match_driver_to_route[driver] = route
        # We need to output route, vehicle, driver
        for driver in range(1, k + 1):
            route = match_driver_to_route[driver]
            vehicle = route_to_vehicle[route]
            print(f"{route} {vehicle} {driver}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
