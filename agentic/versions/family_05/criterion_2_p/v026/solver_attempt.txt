import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Phase I: Sensor array s is locked to signal processor p.
    # We can map each s to its unique p.
    s_to_p = {}
    # We need to select k channels (s, p, a) such that:
    # 1. Each s is used once (1 to k)
    # 2. Each p is used once (1 to k)
    # 3. Each a is used once (1 to k)
    
    # Since s is locked to p, for each s, we only have a set of possible a's.
    # Let's store these as a bipartite matching problem between s and a.
    # The nodes are s in {1..k} and a in {1..k}.
    # An edge exists between s and a if there is a configuration (s, p, a).
    
    adj = [[] for _ in range(k + 1)]
    config_map = {} # (s, a) -> p
    
    idx = 2
    for _ in range(t):
        s = int(input_data[idx])
        p = int(input_data[idx+1])
        a = int(input_data[idx+2])
        idx += 3
        
        adj[s].append(a)
        config_map[(s, a)] = p
        s_to_p[s] = p
        
    # Hopcroft-Karp or simple DFS-based bipartite matching
    match_a = [-1] * (k + 1)
    match_s = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_a[v] < 0 or dfs(match_a[v], visited):
                    match_a[v] = u
                    match_s[u] = v
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for s in range(1, k + 1):
            a = match_s[s]
            p = s_to_p[s]
            print(f"{s} {p} {a}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
