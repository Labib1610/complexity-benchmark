import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # drone_to_station maps drone d to its unique station c
    drone_to_station = {}
    # station_to_drones maps station c to a list of (d, f)
    station_to_drones = {}
    
    idx = 2
    for _ in range(t):
        d = int(input_data[idx])
        c = int(input_data[idx+1])
        f = int(input_data[idx+2])
        idx += 3
        
        if d not in drone_to_station:
            drone_to_station[d] = c
        
        if c not in station_to_drones:
            station_to_drones[c] = []
        station_to_drones[c].append((d, f))
        
    # Check if we have all drones and stations
    if len(drone_to_station) < k or len(station_to_drones) < k:
        print("NO")
        return
    
    # We need to pick exactly one frequency for each station such that
    # all frequencies 1..k are used exactly once.
    # Since each station c is associated with exactly one drone d,
    # picking one (d, f) for each station c is equivalent to picking
    # one frequency f for each station c.
    
    # Bipartite matching: stations to frequencies
    # adj[station] = list of available frequencies
    adj = [[] for _ in range(k + 1)]
    for c in range(1, k + 1):
        for d, f in station_to_drones[c]:
            adj[c].append(f)
            
    match_f = [-1] * (k + 1)
    match_c = [-1] * (k + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                if match_f[v] < 0 or dfs(match_f[v], visited):
                    match_f[v] = u
                    match_c[u] = v
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        # Reconstruct the result
        # For each station c, we need the drone d and the frequency f = match_c[c]
        # We know drone_to_station is fixed, but we need to find which d was used.
        # Actually, the problem says each drone d is wired to a specific station c.
        # So for each station c, there is only one drone d.
        
        # Create a map from station to drone
        station_to_drone = {c: d for d, c in drone_to_station.items()}
        
        for c in range(1, k + 1):
            d = station_to_drone[c]
            f = match_c[c]
            print(f"{d} {c} {f}")
    else:
        print("NO")

solve()
