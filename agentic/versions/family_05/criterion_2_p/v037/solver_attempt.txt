```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # The problem states:
    # 1. Each specialist x is paired with exactly one machine y.
    # 2. No two specialists share the same machine.
    # This means there is a fixed mapping f: X -> Y.
    # We are given (x, y, z) triples. Since x determines y, 
    # we only care about the mapping from x to z.
    # We need to pick one z for each x such that all z are distinct.
    # This is a bipartite matching problem between X and Z.
    
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        idx += 3
        
    # match_z[z] stores the specialist x assigned to time slot z
    match_z = [0] * (k + 1)
    # match_x[x] stores the time slot z assigned to specialist x
    match_x = [0] * (k + 1)
    
    def dfs(u, visited):
        for y, z in adj[u]:
            if not visited[z]:
                visited[z] = True
                if match_z[z] == 0 or dfs(match_z[z], visited):
                    match_z[z] = u
                    match_x[u] = (y, z)
                    return True
        return False

    count = 0
    for i in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(i, visited):
            count += 1
            
    if count == k:
        print("YES")
        for i in range(1, k + 1):
            y, z = match_x[i]
            print(f"{i} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
