import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        l = int(input_data[idx]) - 1
        c = int(input_data[idx+1]) - 1
        r = int(input_data[idx+2]) - 1
        configs.append((l, c, r))
        idx += 3

    # We need to select k configurations such that:
    # Each l in 0..k-1 appears exactly once.
    # Each c in 0..k-1 appears exactly once.
    # Each r in 0..k-1 appears exactly once.
    # This is equivalent to finding a set of k triples (l, c, r) such that
    # they form a Latin Square representation (specifically, a set of k triples
    # that are mutually orthogonal in a sense, but here it's just a selection).
    # Since k is small (up to 22), we can use backtracking with pruning.
    
    # Group configs by l
    by_l = [[] for _ in range(k)]
    for i, (l, c, r) in enumerate(configs):
        by_l[l].append((c, r, i))
        
    used_c = [False] * k
    used_r = [False] * k
    result = []

    def backtrack(l_idx):
        if l_idx == k:
            return True
        
        for c, r, i in by_l[l_idx]:
            if not used_c[c] and not used_r[r]:
                used_c[c] = True
                used_r[r] = True
                result.append(configs[i])
                
                if backtrack(l_idx + 1):
                    return True
                
                result.pop()
                used_c[c] = False
                used_r[r] = False
        return False

    if backtrack(0):
        print("YES")
        for l, c, r in result:
            print(f"{l+1} {c+1} {r+1}")
    else:
        print("NO")

solve()
