import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # We need to select k triples such that each x, y, z is used exactly once.
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    # Group triples by x
    by_x = [[] for _ in range(k)]
    for x, y, z in triples:
        by_x[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        for y, z in by_x[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Optimization: sort by number of options to fail fast
    # But for k=22, simple backtracking might be slow.
    # Let's try to order by x with fewest options first.
    order = sorted(range(k), key=lambda i: len(by_x[i]))
    
    # Re-map to handle the order
    new_by_x = [by_x[i] for i in order]
    
    def backtrack_ordered(idx):
        if idx == k:
            return True
        
        x = order[idx]
        for y, z in by_x[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                if backtrack_ordered(idx + 1):
                    return True
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack_ordered(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
