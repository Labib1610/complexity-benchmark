import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    links = []
    idx = 2
    for _ in range(t):
        s = int(input_data[idx]) - 1
        c = int(input_data[idx+1]) - 1
        m = int(input_data[idx+2]) - 1
        links.append((s, c, m))
        idx += 3
        
    # We need to select k links such that each s, c, m appears exactly once.
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # Represent links as (s, c, m).
    # Use bitmasks for s, c, m.
    
    # Group links by s to simplify recursion
    links_by_s = [[] for _ in range(k)]
    for s, c, m in links:
        links_by_s[s].append((c, m))
        
    used_c = 0
    used_m = 0
    result = []
    
    def backtrack(s_idx):
        if s_idx == k:
            return True
        
        for c, m in links_by_s[s_idx]:
            if not (used_c & (1 << c)) and not (used_m & (1 << m)):
                used_c |= (1 << c)
                used_m |= (1 << m)
                result.append((s_idx + 1, c + 1, m + 1))
                
                if backtrack(s_idx + 1):
                    return True
                
                result.pop()
                used_c &= ~(1 << c)
                used_m &= ~(1 << m)
        return False

    if backtrack(0):
        print("YES")
        for link in result:
            print(f"{link[0]} {link[1]} {link[2]}")
    else:
        print("NO")

solve()
