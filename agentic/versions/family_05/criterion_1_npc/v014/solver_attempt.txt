import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Protocols are (s, m, t)
    # We need to select k triples such that each s, m, t in [1, k] appears exactly once.
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning or bitmask DP.
    
    protocols = []
    idx = 2
    for _ in range(t):
        s = int(input_data[idx]) - 1
        m = int(input_data[idx+1]) - 1
        time = int(input_data[idx+2]) - 1
        protocols.append((s, m, time))
        idx += 3
        
    # Group protocols by specialist
    by_s = [[] for _ in range(k)]
    for p in protocols:
        by_s[p[0]].append(p)
        
    used_m = [False] * k
    used_t = [False] * k
    result = []
    
    def backtrack(s_idx):
        if s_idx == k:
            return True
        
        for p in by_s[s_idx]:
            m, time = p[1], p[2]
            if not used_m[m] and not used_t[time]:
                used_m[m] = True
                used_t[time] = True
                result.append(p)
                
                if backtrack(s_idx + 1):
                    return True
                
                result.pop()
                used_m[m] = False
                used_t[time] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
