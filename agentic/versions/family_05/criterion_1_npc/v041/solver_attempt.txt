import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    trios = []
    idx = 2
    for _ in range(t):
        s = int(input_data[idx]) - 1
        c = int(input_data[idx+1]) - 1
        f = int(input_data[idx+2]) - 1
        trios.append((s, c, f))
        idx += 3

    # We need to select k trios such that all s, c, f are distinct.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group trios by starter
    by_s = [[] for _ in range(k)]
    for s, c, f in trios:
        by_s[s].append((c, f))
        
    used_c = [False] * k
    used_f = [False] * k
    result = []

    def backtrack(s_idx):
        if s_idx == k:
            return True
        
        for c, f in by_s[s_idx]:
            if not used_c[c] and not used_f[f]:
                used_c[c] = True
                used_f[f] = True
                result.append((s_idx + 1, c + 1, f + 1))
                
                if backtrack(s_idx + 1):
                    return True
                
                result.pop()
                used_c[c] = False
                used_f[f] = False
        return False

    if backtrack(0):
        print("YES")
        for trio in result:
            print(f"{trio[0]} {trio[1]} {trio[2]}")
    else:
        print("NO")

solve()
