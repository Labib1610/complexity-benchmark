import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    routes = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        routes.append((x, y, z))
        idx += 3

    # Group routes by source: routes_by_source[x] = list of (y, z)
    routes_by_source = [[] for _ in range(k)]
    for x, y, z in routes:
        routes_by_source[x].append((y, z))

    # We need to pick k routes such that each source, transformer, and destination
    # is used exactly once. This is equivalent to finding a set of k triples
    # (x, y, z) such that {x} = {0..k-1}, {y} = {0..k-1}, {z} = {0..k-1}.
    # Since we must pick exactly one route for each source, we can use recursion
    # with bitmasking for transformers and destinations.
    
    memo = {}

    def backtrack(source_idx, used_y, used_z):
        if source_idx == k:
            return []
        
        state = (source_idx, used_y, used_z)
        if state in memo:
            return memo[state]
        
        for y, z in routes_by_source[source_idx]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                res = backtrack(source_idx + 1, used_y | (1 << y), used_z | (1 << z))
                if res is not None:
                    memo[state] = [(source_idx + 1, y + 1, z + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
