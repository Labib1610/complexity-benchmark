import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        edges.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem. Since k <= 22, we can use backtracking
    # with pruning or bitmasking. Given the constraints and the nature of 
    # 3D matching, backtracking with state tracking is efficient enough.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        for i in range(start_idx, t):
            x, y, z = edges[i]
            if x <= k and y <= k and z <= k:
                if not used_x[x] and not used_y[y] and not used_z[z]:
                    used_x[x] = True
                    used_y[y] = True
                    used_z[z] = True
                    result.append((x, y, z))
                    
                    if backtrack(i + 1, count + 1):
                        return True
                    
                    result.pop()
                    used_x[x] = False
                    used_y[y] = False
                    used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
