import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # Group triples by specialist x
    by_x = [[] for _ in range(k)]
    for x, y, z in triples:
        by_x[x].append((y, z))

    # Meet-in-the-middle: split k into k1 = k // 2 and k2 = k - k1
    k1 = k // 2
    k2 = k - k1
    
    # Map state (mask_y, mask_z) to list of assignments
    # mask_y: bitmask of used machines, mask_z: bitmask of used time slots
    memo = {}

    def get_subsets(start_x, end_x):
        # Returns dict: (mask_y, mask_z) -> list of (x, y, z)
        results = {(0, 0): []}
        for x in range(start_x, end_x):
            new_results = {}
            for (my, mz), path in results.items():
                for y, z in by_x[x]:
                    if not (my & (1 << y)) and not (mz & (1 << z)):
                        new_mask = (my | (1 << y), mz | (1 << z))
                        if new_mask not in new_results:
                            new_results[new_mask] = path + [(x + 1, y + 1, z + 1)]
            results = new_results
        return results

    left_map = get_subsets(0, k1)
    right_map = get_subsets(k1, k)

    full_mask = (1 << k) - 1
    for (my1, mz1), path1 in left_map.items():
        target_my = full_mask ^ my1
        target_mz = full_mask ^ mz1
        if (target_my, target_mz) in right_map:
            print("YES")
            for crew in path1:
                print(f"{crew[0]} {crew[1]} {crew[2]}")
            for crew in right_map[(target_my, target_mz)]:
                print(f"{crew[0]} {crew[1]} {crew[2]}")
            return

    print("NO")

solve()
