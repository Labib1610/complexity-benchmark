import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        configs.append((x, y, z))
        idx += 3
        
    # We need to select k triples (x_i, y_i, z_i) such that
    # all x_i are distinct (1..k), all y_i are distinct (1..k),
    # and all z_i are distinct (1..k).
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    # Group configs by specialist x
    by_x = [[] for _ in range(k + 1)]
    for x, y, z in configs:
        by_x[x].append((y, z))
        
    # Sort by number of options to optimize backtracking
    order = sorted(range(1, k + 1), key=lambda i: len(by_x[i]))
    
    result = []
    
    def backtrack(idx):
        if idx == k:
            return True
        
        x = order[idx]
        for y, z in by_x[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x, y, z))
                
                if backtrack(idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
