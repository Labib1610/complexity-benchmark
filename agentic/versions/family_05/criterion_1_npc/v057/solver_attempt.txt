```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem (3D matching).
    # Since k <= 22, we use backtracking with pruning.
    # Represent constraints as bitmasks.
    # dish_mask: bit i set if dish i is used
    # key_mask: bit j set if key j is used
    # buffer_mask: bit l set if buffer l is used
    
    # Group configs by dish to simplify recursion
    by_dish = [[] for _ in range(k)]
    for c in configs:
        by_dish[c[0]].append(c)
        
    used_keys = 0
    used_buffers = 0
    result = []
    
    def backtrack(dish_idx):
        if dish_idx == k:
            return True
        
        for c in by_dish[dish_idx]:
            _, y, z = c
            if not (used_keys & (1 << y)) and not (used_buffers & (1 << z)):
                nonlocal used_keys, used_buffers
                used_keys |= (1 << y)
                used_buffers |= (1 << z)
                result.append(c)
                
                if backtrack(dish_idx + 1):
                    return True
                
                result.pop()
                used_keys &= ~(1 << y)
                used_buffers &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for c in result:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
