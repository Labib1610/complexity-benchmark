```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    linkages = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        linkages.append((x, y, z))
        idx += 3
        
    # We need to select k linkages such that each x, y, z is used exactly once.
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking with pruning.
    # State: (index of array to fill, mask of targets used, mask of bands used)
    # However, k=22 is too large for simple recursion.
    # But we only have k linkages to pick.
    
    # Group linkages by array_ID
    by_array = [[] for _ in range(k)]
    for x, y, z in linkages:
        by_array[x].append((y, z))
        
    # Sort arrays by number of available linkages to prune search space
    order = sorted(range(k), key=lambda i: len(by_array[i]))
    
    used_y = 0
    used_z = 0
    result = [None] * k
    
    def backtrack(idx):
        nonlocal used_y, used_z
        if idx == k:
            return True
        
        arr_id = order[idx]
        for y, z in by_array[arr_id]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result[arr_id] = (arr_id + 1, y + 1, z + 1)
                
                if backtrack(idx + 1):
                    return True
                
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
```
