import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        triplets.append((x - 1, y - 1, z - 1))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # State: (used_x_mask, used_y_mask, used_z_mask)
    # However, we can simplify: we need to pick k triplets such that
    # each x, y, z is used exactly once.
    # This is equivalent to finding a set of k triplets (x_i, y_i, z_i)
    # such that {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}.
    
    # Optimization: Sort triplets to pick x in order 0, 1, ..., k-1
    # This reduces the search space significantly.
    
    by_x = [[] for _ in range(k)]
    for tpl in triplets:
        by_x[tpl[0]].append(tpl)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for tpl in by_x[x_idx]:
            y, z = tpl[1], tpl[2]
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(tpl)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for tpl in result:
            print(f"{tpl[0]+1} {tpl[1]+1} {tpl[2]+1}")
    else:
        print("NO")

solve()
