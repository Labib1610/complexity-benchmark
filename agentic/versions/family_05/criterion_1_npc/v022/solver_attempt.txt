import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    nodes = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        nodes.append((x, y, z))
        idx += 3

    # Group nodes by x-coordinate
    by_x = [[] for _ in range(k)]
    for node in nodes:
        by_x[node[0]].append(node)

    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        for node in by_x[x]:
            _, y, z = node
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(node)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for node in result:
            print(f"{node[0] + 1} {node[1] + 1} {node[2] + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
