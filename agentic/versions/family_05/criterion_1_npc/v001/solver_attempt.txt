```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # Group setups by specialist (x)
    # adj[x] = list of (y, z)
    adj = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x].append((y, z))
        idx += 3
        
    # We need to pick one (y, z) for each x from 1 to k
    # such that all y's are distinct and all z's are distinct.
    # This is equivalent to finding a perfect matching in a 3D hypergraph,
    # or more simply, a backtracking search with bitmasks.
    
    used_y = 0
    used_z = 0
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for y, z in adj[x]:
            y_bit = 1 << (y - 1)
            z_bit = 1 << (z - 1)
            
            if not (used_y & y_bit) and not (used_z & z_bit):
                nonlocal used_y, used_z
                used_y |= y_bit
                used_z |= z_bit
                result.append((x, y, z))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y &= ~y_bit
                used_z &= ~z_bit
        return False

    if backtrack(1):
        print("YES")
        for row in result:
            print(f"{row[0]} {row[1]} {row[2]}")
    else:
        print("NO")

solve()
```
