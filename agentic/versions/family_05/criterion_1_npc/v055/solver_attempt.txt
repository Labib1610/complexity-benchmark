```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triples (x, y, z) such that all x are distinct,
    # all y are distinct, and all z are distinct.
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    # Group triples by surgeon (x)
    by_surgeon = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(triples):
        by_surgeon[x].append((y, z, i))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z, original_idx in by_surgeon[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(triples[original_idx])
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
