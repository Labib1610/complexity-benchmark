import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        d = int(input_data[idx])
        f = int(input_data[idx+1])
        m = int(input_data[idx+2])
        triples.append((d, f, m))
        idx += 3
        
    # We need to select k triples such that each d, f, m appears exactly once.
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # To optimize, we use bitmasks for the used components.
    
    # Group triples by d to make selection easier
    by_d = [[] for _ in range(k + 1)]
    for d, f, m in triples:
        by_d[d].append((f, m))
        
    used_f = 0
    used_m = 0
    result = []
    
    def backtrack(d):
        if d > k:
            return True
        
        for f, m in by_d[d]:
            f_bit = 1 << (f - 1)
            m_bit = 1 << (m - 1)
            
            if not (used_f & f_bit) and not (used_m & m_bit):
                nonlocal used_f, used_m
                used_f |= f_bit
                used_m |= m_bit
                result.append((d, f, m))
                
                if backtrack(d + 1):
                    return True
                
                result.pop()
                used_f &= ~f_bit
                used_m &= ~m_bit
        return False

    if backtrack(1):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
