import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        e = int(input_data[idx]) - 1
        s = int(input_data[idx+1]) - 1
        p = int(input_data[idx+2]) - 1
        triples.append((e, s, p))
        idx += 3

    # Group triples by enzyme to facilitate backtracking
    by_e = [[] for _ in range(k)]
    for e, s, p in triples:
        by_e[e].append((s, p))

    used_s = [False] * k
    used_p = [False] * k
    result = []

    def backtrack(e_idx):
        if e_idx == k:
            return True
        
        for s, p in by_e[e_idx]:
            if not used_s[s] and not used_p[p]:
                used_s[s] = True
                used_p[p] = True
                result.append((e_idx + 1, s + 1, p + 1))
                
                if backtrack(e_idx + 1):
                    return True
                
                result.pop()
                used_s[s] = False
                used_p[p] = False
        return False

    # Optimization: Sort by number of options to prune search tree
    # But for k=22, simple backtracking with bitmasks or pruning is needed.
    # Given the constraints and nature of 3D matching, this is NP-complete.
    # However, k=22 is small enough for efficient backtracking if implemented well.
    
    if backtrack(0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
