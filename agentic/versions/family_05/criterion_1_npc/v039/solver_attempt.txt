import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for x, y, z in triples:
        adj[x].append((y, z))
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x):
        if x == k:
            return True
        
        # Try to find a valid (y, z) for surgeon x
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort surgeons by number of options to optimize backtracking
    # (Heuristic: Most Constrained Variable)
    order = sorted(range(k), key=lambda i: len(adj[i]))
    
    # Re-map adj to the sorted order
    new_adj = [adj[i] for i in order]
    
    def backtrack_optimized(idx_in_order):
        if idx_in_order == k:
            return True
        
        x = order[idx_in_order]
        for y, z in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack_optimized(idx_in_order + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack_optimized(0):
        print("YES")
        for r in result:
            print(f"{r[0]} {r[1]} {r[2]}")
    else:
        print("NO")

solve()
