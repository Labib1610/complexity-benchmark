import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    triplets = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triplets.append((a - 1, b - 1, c - 1))
        idx += 3

    # The condition "sum of all 3k integers is 3*k*(k+1)/2"
    # is equivalent to saying that the set of all 'a's is {0..k-1},
    # the set of all 'b's is {0..k-1}, and the set of all 'c's is {0..k-1}.
    # This is exactly the definition of an orthogonal Latin square component
    # or a 3D matching problem in a k x k x k grid.
    
    # Since k <= 22, we can use backtracking with pruning.
    # We need to pick k triplets such that each coordinate is used exactly once.
    
    used_a = [False] * k
    used_b = [False] * k
    used_c = [False] * k
    selected = []

    def backtrack(idx):
        if len(selected) == k:
            return True
        
        # Optimization: pick the first available 'a' to reduce branching
        a_target = -1
        for i in range(k):
            if not used_a[i]:
                a_target = i
                break
        
        for i in range(idx, t):
            a, b, c = triplets[i]
            if a == a_target and not used_b[b] and not used_c[c]:
                used_a[a] = True
                used_b[b] = True
                used_c[c] = True
                selected.append((a + 1, b + 1, c + 1))
                
                if backtrack(i + 1):
                    return True
                
                selected.pop()
                used_a[a] = False
                used_b[b] = False
                used_c[c] = False
        return False

    # Pre-sort or group by 'a' to make the search faster
    triplets.sort()
    
    if backtrack(0):
        print("YES")
        for triplet in selected:
            print(f"{triplet[0]} {triplet[1]} {triplet[2]}")
    else:
        print("NO")

solve()
