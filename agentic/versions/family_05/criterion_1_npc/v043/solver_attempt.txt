import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    commands = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        commands.append((x, y, z))
        idx += 3

    # We need to select k commands such that all x are distinct, 
    # all y are distinct, and all z are distinct.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(cmd_idx, count):
        if count == k:
            return True
        if cmd_idx == t:
            return False
        
        # Pruning: if remaining commands are not enough
        if (t - cmd_idx) < (k - count):
            return False
            
        # Try including commands[cmd_idx]
        x, y, z = commands[cmd_idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append(commands[cmd_idx])
            if backtrack(cmd_idx + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Try skipping commands[cmd_idx]
        if backtrack(cmd_idx + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for cmd in result:
            print(f"{cmd[0]+1} {cmd[1]+1} {cmd[2]+1}")
    else:
        print("NO")

solve()
