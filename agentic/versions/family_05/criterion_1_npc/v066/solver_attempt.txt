import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    profiles = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        profiles.append((x, y, z))
        idx += 3
        
    # We need to select k triples (x_i, y_i, z_i) such that
    # {x_i} = {0..k-1}, {y_i} = {0..k-1}, {z_i} = {0..k-1}
    # This is equivalent to finding a perfect matching in a 3D hypergraph
    # or an exact cover problem. Since k <= 22, we can use backtracking with pruning.
    
    # Group profiles by specialist
    by_spec = [[] for _ in range(k)]
    for p in profiles:
        by_spec[p[0]].append(p)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(spec_idx):
        if spec_idx == k:
            return True
        
        for p in by_spec[spec_idx]:
            _, y, z = p
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(spec_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
