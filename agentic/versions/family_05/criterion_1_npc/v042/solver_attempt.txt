import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triples (t, k, l) such that each tower, key, and link
    # is used exactly once. This is equivalent to the 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    triples = []
    idx = 2
    for _ in range(t):
        p1 = int(input_data[idx]) - 1
        p2 = int(input_data[idx+1]) - 1
        p3 = int(input_data[idx+2]) - 1
        triples.append((p1, p2, p3))
        idx += 3
        
    # Group triples by tower to simplify recursion
    by_tower = [[] for _ in range(k)]
    for tpl in triples:
        by_tower[tpl[0]].append(tpl)
        
    used_key = [False] * k
    used_link = [False] * k
    result = []
    
    def backtrack(tower_idx):
        if tower_idx == k:
            return True
        
        for tpl in by_tower[tower_idx]:
            _, k_val, l_val = tpl
            if not used_key[k_val] and not used_link[l_val]:
                used_key[k_val] = True
                used_link[l_val] = True
                result.append(tpl)
                
                if backtrack(tower_idx + 1):
                    return True
                
                result.pop()
                used_key[k_val] = False
                used_link[l_val] = False
        return False

    if backtrack(0):
        print("YES")
        for tpl in result:
            print(f"{tpl[0]+1} {tpl[1]+1} {tpl[2]+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
