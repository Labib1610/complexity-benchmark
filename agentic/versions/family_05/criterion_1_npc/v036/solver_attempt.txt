import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3

    # Group configurations by frequency band
    by_band = [[] for _ in range(k)]
    for c in configs:
        by_band[c[0]].append(c)

    # We need to select exactly one configuration for each band x in 0..k-1
    # such that all y and z are distinct (i.e., each is a permutation of 0..k-1)
    
    # Use backtracking with bitmasks for positions and keys
    selected = [None] * k
    used_y = 0
    used_z = 0
    
    def backtrack(band_idx):
        nonlocal used_y, used_z
        if band_idx == k:
            return True
        
        for c in by_band[band_idx]:
            _, y, z = c
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                selected[band_idx] = c
                if backtrack(band_idx + 1):
                    return True
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for c in selected:
            print(f"{c[0]+1} {c[1]+1} {c[2]+1}")
    else:
        print("NO")

solve()
