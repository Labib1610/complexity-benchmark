import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        triples.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * (k + 1)
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    result = []
    
    def backtrack(idx_in_triples, count):
        if count == k:
            return True
        if idx_in_triples == t:
            return False
        
        # Heuristic: try to pick triples that help fill the remaining slots
        for i in range(idx_in_triples, t):
            x, y, z = triples[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = used_y[y] = used_z[z] = True
                result.append((x, y, z))
                
                if backtrack(i + 1, count + 1):
                    return True
                
                result.pop()
                used_x[x] = used_y[y] = used_z[z] = False
        return False

    # Sort triples to potentially find a solution faster
    triples.sort()
    
    if backtrack(0, 0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
