import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        p1 = int(input_data[idx]) - 1
        p2 = int(input_data[idx+1]) - 1
        p3 = int(input_data[idx+2]) - 1
        configs.append((p1, p2, p3))
        idx += 3

    # We need to select k configurations such that each p1, p2, p3 is used exactly once.
    # This is equivalent to finding a set of k triples (i, p2_i, p3_i) such that
    # {i | i in 0..k-1} = {0..k-1}, {p2_i} = {0..k-1}, {p3_i} = {0..k-1}.
    # Since each p1 must be used exactly once, we can group configs by p1.
    
    by_p1 = [[] for _ in range(k)]
    for c in configs:
        by_p1[c[0]].append((c[1], c[2]))
        
    # Backtracking with bitmasks
    # mask2: bitmask of used p2 values
    # mask3: bitmask of used p3 values
    
    memo = {}

    def backtrack(p1_idx, mask2, mask3):
        if p1_idx == k:
            return []
        
        state = (p1_idx, mask2, mask3)
        if state in memo:
            return memo[state]
        
        for p2, p3 in by_p1[p1_idx]:
            if not (mask2 & (1 << p2)) and not (mask3 & (1 << p3)):
                res = backtrack(p1_idx + 1, mask2 | (1 << p2), mask3 | (1 << p3))
                if res is not None:
                    memo[state] = [(p1_idx + 1, p2 + 1, p3 + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for row in result:
            print(f"{row[0]} {row[1]} {row[2]}")
    else:
        print("NO")

solve()
