import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    blueprints = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        blueprints.append((x, y, z))
        idx += 3

    # We need to select k triples (x_i, y_i, z_i) such that all x_i are distinct,
    # all y_i are distinct, and all z_i are distinct.
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Group blueprints by specialist x
    by_x = [[] for _ in range(k)]
    for b in blueprints:
        by_x[b[0]].append(b)
        
    used_y = [False] * k
    used_z = [False] * k
    result = []

    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for b in by_x[x_idx]:
            _, y, z = b
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(b)
                
                if backtrack(x_idx + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(0):
        print("YES")
        for b in result:
            print(f"{b[0]+1} {b[1]+1} {b[2]+1}")
    else:
        print("NO")

solve()
