```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triples (x, y, z) such that each x, y, z in 1..k appears exactly once.
    # This is equivalent to finding a perfect matching in a 3D hypergraph,
    # or more specifically, a Latin Square-like structure.
    # Since k <= 22, we can use backtracking with pruning.
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    # Sort triples to improve pruning
    triples.sort()
    
    def backtrack(start_idx, count):
        if count == k:
            return True
        
        # Optimization: if remaining triples are not enough to fill the rest
        if (t - start_idx) < (k - count):
            return False
            
        for i in range(start_idx, t):
            x, y, z = triples[i]
            if not used_x[x] and not used_y[y] and not used_z[z]:
                used_x[x] = used_y[y] = used_z[z] = True
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(i + 1, count + 1):
                    return True
                
                result.pop()
                used_x[x] = used_y[y] = used_z[z] = False
        return False

    if backtrack(0, 0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
```
