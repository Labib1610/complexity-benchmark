import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triads = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triads.append((x, y, z))
        idx += 3
        
    # Group triads by analyst
    by_analyst = [[] for _ in range(k)]
    for x, y, z in triads:
        by_analyst[x].append((y, z))
        
    # We need to pick one triad for each analyst (0 to k-1)
    # such that all y's are distinct and all z's are distinct.
    # This is equivalent to finding a perfect matching in a 3D hypergraph,
    # or more simply, a backtracking search with bitmasks.
    
    used_y = 0
    used_z = 0
    result = []
    
    def backtrack(analyst_idx):
        nonlocal used_y, used_z
        if analyst_idx == k:
            return True
        
        for y, z in by_analyst[analyst_idx]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append((analyst_idx + 1, y + 1, z + 1))
                
                if backtrack(analyst_idx + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    # Optimization: sort analysts by number of available choices
    # to prune the search tree faster.
    order = sorted(range(k), key=lambda i: len(by_analyst[i]))
    
    # Re-map to handle the sorted order
    new_by_analyst = [by_analyst[i] for i in order]
    
    def backtrack_optimized(idx_in_order):
        nonlocal used_y, used_z
        if idx_in_order == k:
            return True
        
        orig_analyst = order[idx_in_order]
        for y, z in new_by_analyst[idx_in_order]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append((orig_analyst + 1, y + 1, z + 1))
                
                if backtrack_optimized(idx_in_order + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack_optimized(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
