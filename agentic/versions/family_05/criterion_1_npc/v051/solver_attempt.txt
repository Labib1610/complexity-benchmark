import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    trios = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        trios.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning or bitmask DP.
    # Since we need to find one valid assignment, backtracking is efficient.
    
    # Group trios by linguist
    by_linguist = [[] for _ in range(k)]
    for x, y, z in trios:
        by_linguist[x].append((y, z))
        
    used_manuscript = [False] * k
    used_algorithm = [False] * k
    assignment = []
    
    def backtrack(linguist_idx):
        if linguist_idx == k:
            return True
        
        for m, a in by_linguist[linguist_idx]:
            if not used_manuscript[m] and not used_algorithm[a]:
                used_manuscript[m] = True
                used_algorithm[a] = True
                assignment.append((linguist_idx + 1, m + 1, a + 1))
                
                if backtrack(linguist_idx + 1):
                    return True
                
                assignment.pop()
                used_manuscript[m] = False
                used_algorithm[a] = False
        return False

    if backtrack(0):
        print("YES")
        for trio in assignment:
            print(f"{trio[0]} {trio[1]} {trio[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
