```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # Group triples by chassis (x)
    by_chassis = [[] for _ in range(k)]
    for x, y, z in triples:
        by_chassis[x].append((y, z))

    # memo[chassis_idx][mask_y][mask_z]
    # Since k <= 22, we use bitmasks.
    # However, 2^22 * 2^22 is too large.
    # But we only need to pick k triples such that each y and z is used once.
    # This is an Exact Cover problem.
    
    memo = {}

    def backtrack(c_idx, mask_y, mask_z):
        if c_idx == k:
            return []
        
        state = (c_idx, mask_y, mask_z)
        if state in memo:
            return memo[state]
        
        for y, z in by_chassis[c_idx]:
            if not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                res = backtrack(c_idx + 1, mask_y | (1 << y), mask_z | (1 << z))
                if res is not None:
                    memo[state] = [(c_idx + 1, y + 1, z + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
```
