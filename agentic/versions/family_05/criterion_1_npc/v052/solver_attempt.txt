import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    blueprints = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        blueprints.append((x, y, z))
        idx += 3
        
    # We need to select k blueprints such that all x, y, z are distinct.
    # This is equivalent to finding a perfect matching in a 3-uniform 3-partite hypergraph.
    # Since k <= 22, we can use backtracking with pruning.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    result = []
    
    def backtrack(bp_idx, count):
        if count == k:
            return True
        if bp_idx == t:
            return False
        
        # Pruning: if remaining blueprints are not enough
        if t - bp_idx < k - count:
            return False
            
        # Option 1: Include blueprints[bp_idx]
        x, y, z = blueprints[bp_idx]
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            result.append((x + 1, y + 1, z + 1))
            if backtrack(bp_idx + 1, count + 1):
                return True
            result.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Option 2: Skip blueprints[bp_idx]
        if backtrack(bp_idx + 1, count):
            return True
            
        return False

    # Sort blueprints to potentially hit a solution faster
    if backtrack(0, 0):
        print("YES")
        for bp in result:
            print(f"{bp[0]} {bp[1]} {bp[2]}")
    else:
        print("NO")

solve()
