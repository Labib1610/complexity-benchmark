import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    # Group triples by x
    by_x = [[] for _ in range(k)]
    for x, y, z in triples:
        by_x[x].append((y, z))
        
    # memo[x_mask][y_mask] = z_mask or None
    # Since k <= 22, we use recursion with memoization
    # To optimize, we only need to track x_mask (which is just the number of specialists used)
    # and the masks for y and z.
    memo = {}

    def backtrack(x, y_mask, z_mask):
        if x == k:
            return []
        
        state = (x, y_mask, z_mask)
        if state in memo:
            return memo[state]
        
        for y, z in by_x[x]:
            if not (y_mask & (1 << y)) and not (z_mask & (1 << z)):
                res = backtrack(x + 1, y_mask | (1 << y), z_mask | (1 << z))
                if res is not None:
                    memo[state] = [(x + 1, y + 1, z + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
