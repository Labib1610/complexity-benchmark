import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        m = int(input_data[idx]) - 1
        p = int(input_data[idx+1]) - 1
        s = int(input_data[idx+2]) - 1
        triplets.append((m, p, s))
        idx += 3

    # Group triplets by membrane type
    by_m = [[] for _ in range(k)]
    for m, p, s in triplets:
        by_m[m].append((p, s))

    # dp[mask_p][mask_s] stores the list of triplets used to reach this state
    # Since k <= 22, we use a recursive approach with memoization
    # State: (index of membrane, mask of pathways, mask of sequences)
    memo = {}

    def backtrack(m_idx, mask_p, mask_s):
        if m_idx == k:
            return []
        
        state = (m_idx, mask_p, mask_s)
        if state in memo:
            return memo[state]
        
        for p, s in by_m[m_idx]:
            if not (mask_p & (1 << p)) and not (mask_s & (1 << s)):
                res = backtrack(m_idx + 1, mask_p | (1 << p), mask_s | (1 << s))
                if res is not None:
                    memo[state] = [(m_idx + 1, p + 1, s + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for triplet in result:
            print(f"{triplet[0]} {triplet[1]} {triplet[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
