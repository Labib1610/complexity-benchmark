import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        e = int(input_data[idx])
        s = int(input_data[idx+1])
        p = int(input_data[idx+2])
        triples.append((e, s, p))
        idx += 3
        
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # Represent each triple as a bitmask or just use sets for tracking usage.
    
    used_e = [False] * (k + 1)
    used_s = [False] * (k + 1)
    used_p = [False] * (k + 1)
    result = []
    
    def backtrack(triple_idx, count):
        if count == k:
            return True
        if triple_idx == t:
            return False
        
        # Optimization: if remaining triples are not enough
        if t - triple_idx < k - count:
            return False
            
        # Try including triples[triple_idx]
        e, s, p = triples[triple_idx]
        if not used_e[e] and not used_s[s] and not used_p[p]:
            used_e[e] = used_s[s] = used_p[p] = True
            result.append(triples[triple_idx])
            if backtrack(triple_idx + 1, count + 1):
                return True
            result.pop()
            used_e[e] = used_s[s] = used_p[p] = False
            
        # Try excluding triples[triple_idx]
        if backtrack(triple_idx + 1, count):
            return True
            
        return False

    # For small k, simple backtracking is fine. 
    # Given k <= 22, we might need a more efficient approach if t is large.
    # However, the constraints and problem type suggest backtracking with pruning.
    
    if backtrack(0, 0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
