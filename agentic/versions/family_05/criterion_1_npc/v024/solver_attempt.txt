import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # We need to select k triples (d, f, m) such that each d, f, m appears exactly once.
    # This is equivalent to finding a perfect matching in a 3D matching problem.
    # Since k <= 22, we can use backtracking with pruning or bitmask DP.
    # Given the constraints and the nature of 3D matching, backtracking with 
    # bitmasks for used f and m for each d is efficient.
    
    triples = []
    idx = 2
    for _ in range(t):
        d = int(input_data[idx]) - 1
        f = int(input_data[idx+1]) - 1
        m = int(input_data[idx+2]) - 1
        triples.append((d, f, m))
        idx += 3
        
    # Group triples by demographic d
    by_d = [[] for _ in range(k)]
    for d, f, m in triples:
        by_d[d].append((f, m))
        
    used_f = 0
    used_m = [0] * k # used_m[f] stores bitmask of used facilities for formulation f
    # Actually, we need to ensure each f is used once and each m is used once.
    # Let's use bitmasks:
    # mask_f: bitmask of used formulations
    # mask_m: bitmask of used facilities
    
    res = []
    
    def backtrack(d, mask_f, mask_m):
        if d == k:
            return True
        
        for f, m in by_d[d]:
            if not (mask_f & (1 << f)) and not (mask_m & (1 << m)):
                res.append((d + 1, f + 1, m + 1))
                if backtrack(d + 1, mask_f | (1 << f), mask_m | (1 << m)):
                    return True
                res.pop()
        return False

    if backtrack(0, 0, 0):
        print("YES")
        for item in res:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
