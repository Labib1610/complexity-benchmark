import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3
        
    # We need to select k triples such that each x, y, z is used exactly once.
    # This is an exact cover problem. Since k <= 22, we can use backtracking with pruning.
    # Represent state by (mask_y, mask_z) for each specialist x.
    
    adj = [[] for _ in range(k)]
    for x, y, z in triples:
        adj[x].append((y, z))
        
    ans = [None] * k
    used_y = 0
    used_z = 0
    
    def backtrack(x):
        nonlocal used_y, used_z
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                ans[x] = (x + 1, y + 1, z + 1)
                
                if backtrack(x + 1):
                    return True
                
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    # Optimization: sort specialists by number of available options
    # But with k=22, simple backtracking might be slow. 
    # Let's use the standard Dancing Links approach or simple bitmask recursion.
    # Given the constraints and time limit, simple recursion with bitmasks is usually fine.
    
    if backtrack(0):
        print("YES")
        for row in ans:
            print(f"{row[0]} {row[1]} {row[2]}")
    else:
        print("NO")

solve()
