import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    plans = []
    idx = 2
    for _ in range(t):
        c = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        d = int(input_data[idx+2]) - 1
        plans.append((c, v, d))
        idx += 3
        
    # This is a 3D matching problem. Since k <= 22, we can use backtracking
    # with pruning or bitmasking. Given the constraints, a simple backtracking
    # with state tracking should work.
    
    used_c = [False] * k
    used_v = [False] * k
    used_d = [False] * k
    result = []
    
    def backtrack(idx_plan):
        if len(result) == k:
            return True
        
        for i in range(idx_plan, t):
            c, v, d = plans[i]
            if not used_c[c] and not used_v[v] and not used_d[d]:
                used_c[c] = used_v[v] = used_d[d] = True
                result.append(plans[i])
                
                if backtrack(i + 1):
                    return True
                
                result.pop()
                used_c[c] = used_v[v] = used_d[d] = False
        return False

    # Optimization: sort plans or use a more efficient search if needed.
    # With k=22, simple backtracking might be slow, but the constraints
    # on valid plans usually prune the search space significantly.
    if backtrack(0):
        print("YES")
        for c, v, d in result:
            print(f"{c + 1} {v + 1} {d + 1}")
    else:
        print("NO")

solve()
