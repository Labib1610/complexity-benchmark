import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    blocks = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        blocks.append((x, y, z))
        idx += 3

    # We need to select k blocks such that each shape, color, and material
    # appears exactly once. This is equivalent to finding a transversal in a 
    # 3D Latin Square or an exact cover problem.
    # Since k <= 22, we use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(blocks):
        adj[x].append((y, z, i))
        
    used_y = [False] * k
    used_z = [False] * k
    result = [None] * k
    
    def backtrack(x):
        if x == k:
            return True
        
        # Heuristic: pick the shape with the fewest available options
        # But for simplicity, just iterate through shapes
        for y, z, i in adj[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result[x] = (x + 1, y + 1, z + 1)
                if backtrack(x + 1):
                    return True
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort shapes by number of available blocks to prune search space
    adj.sort(key=len)
    
    # Re-map shapes to the sorted order
    new_adj = [[] for _ in range(k)]
    for i in range(k):
        new_adj[i] = adj[i]
    
    if backtrack(0):
        print("YES")
        for res in result:
            print(f"{res[0]} {res[1]} {res[2]}")
    else:
        print("NO")

solve()
