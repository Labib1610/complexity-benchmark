import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        configs.append((x, y, z))
        idx += 3

    # We need to select k triples (x, y, z) such that each x, y, z in 0..k-1
    # appears exactly once. This is an Exact Cover problem.
    # Since k <= 22, we can use backtracking with bitmasks.
    # To optimize, we can use the fact that we must pick exactly one triple for each x.
    # Let's group configs by x.
    by_x = [[] for _ in range(k)]
    for x, y, z in configs:
        by_x[x].append((y, z))
    
    used_y = 0
    used_z = 0
    result = []

    def backtrack(x):
        nonlocal used_y, used_z
        if x == k:
            return True
        
        for y, z in by_x[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
