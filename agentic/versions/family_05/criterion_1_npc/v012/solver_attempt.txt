import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triplets.append((x, y, z))
        idx += 3
        
    # We need to select k triplets such that each x, y, z appears exactly once.
    # This is an Exact Cover problem. Since k <= 22, we can use backtracking.
    # Represent the state by bitmasks for y and z.
    # x is implicitly handled by the order of recursion (x = 0, 1, ..., k-1).
    
    adj = [[] for _ in range(k)]
    for x, y, z in triplets:
        adj[x].append((y, z))
        
    used_y = 0
    used_z = 0
    result = []
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                nonlocal used_y, used_z
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for triplet in result:
            print(f"{triplet[0]} {triplet[1]} {triplet[2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
