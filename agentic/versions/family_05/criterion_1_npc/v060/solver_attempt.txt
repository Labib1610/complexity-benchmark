import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    proposals = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        proposals.append((x, y, z))
        idx += 3

    # This is an Exact Cover problem.
    # We need to select k triples (x, y, z) such that all x are distinct,
    # all y are distinct, and all z are distinct.
    # Since k <= 22, we can use backtracking with pruning.
    
    # Represent constraints as bitmasks
    # used_x, used_y, used_z
    
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(proposals):
        adj[x].append((y, z, i))
        
    selected = []
    used_y = 0
    used_z = 0
    
    def backtrack(x):
        if x == k:
            return True
        
        for y, z, i in adj[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                selected.append(proposals[i])
                
                if backtrack(x + 1):
                    return True
                
                selected.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for x, y, z in selected:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
