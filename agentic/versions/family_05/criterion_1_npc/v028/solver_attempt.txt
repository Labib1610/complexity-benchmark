```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triads = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triads.append((x, y, z))
        idx += 3
        
    # We need to select k triads such that each sculpture, pedestal, and sensor
    # is used exactly once. This is an Exact Cover problem.
    # Since k <= 22, we can use backtracking with bitmasks.
    # However, 22 is too large for a simple bitmask (2^22 is ~4M, but we have 3 sets).
    # Actually, we need to pick k triads.
    
    # Group triads by sculpture
    by_sculpture = [[] for _ in range(k)]
    for t_idx, (x, y, z) in enumerate(triads):
        by_sculpture[x].append((y, z, t_idx))
        
    used_pedestal = 0
    used_sensor = 0
    result = []
    
    def backtrack(sculpture_idx):
        if sculpture_idx == k:
            return True
        
        for y, z, t_idx in by_sculpture[sculpture_idx]:
            if not (used_pedestal & (1 << y)) and not (used_sensor & (1 << z)):
                # Try this triad
                nonlocal used_pedestal, used_sensor
                used_pedestal |= (1 << y)
                used_sensor |= (1 << z)
                result.append((sculpture_idx + 1, y + 1, z + 1))
                
                if backtrack(sculpture_idx + 1):
                    return True
                
                # Backtrack
                result.pop()
                used_pedestal &= ~(1 << y)
                used_sensor &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for triad in result:
            print(f"{triad[0]} {triad[1]} {triad[2]}")
    else:
        print("NO")

solve()
```
