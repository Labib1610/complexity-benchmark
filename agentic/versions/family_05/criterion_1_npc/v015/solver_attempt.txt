```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # Group triples by compute node
    by_node = [[] for _ in range(k)]
    for x, y, z in triples:
        by_node[x].append((y, z))

    # State: (node_idx, used_images_mask, used_shards_mask)
    # Since k <= 22, we use bitmasks.
    # However, 2^22 * 2^22 is too large.
    # But we only need to pick k triples such that each image and shard is used once.
    # This is an Exact Cover problem.
    
    memo = {}

    def backtrack(node_idx, img_mask, shard_mask):
        if node_idx == k:
            return []
        
        state = (node_idx, img_mask, shard_mask)
        if state in memo:
            return memo[state]
        
        for img, shard in by_node[node_idx]:
            if not (img_mask & (1 << img)) and not (shard_mask & (1 << shard)):
                res = backtrack(node_idx + 1, img_mask | (1 << img), shard_mask | (1 << shard))
                if res is not None:
                    memo[state] = [(node_idx + 1, img + 1, shard + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
```
