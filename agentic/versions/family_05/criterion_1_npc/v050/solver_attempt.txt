import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    specs = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        specs.append((x, y, z))
        idx += 3

    # Group specs by x: adj[x] = list of (y, z)
    adj = [[] for _ in range(k)]
    for x, y, z in specs:
        adj[x].append((y, z))

    # We need to pick one (y, z) for each x such that all y are distinct and all z are distinct.
    # This is equivalent to finding a set of k triples (x, y, z) such that x, y, z are all permutations.
    # Since k <= 22, we can use backtracking with bitmasks.
    
    used_y = 0
    used_z = 0
    result = []

    def backtrack(x):
        nonlocal used_y, used_z
        if x == k:
            return True
        
        for y, z in adj[x]:
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append((x + 1, y + 1, z + 1))
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for triple in result:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
