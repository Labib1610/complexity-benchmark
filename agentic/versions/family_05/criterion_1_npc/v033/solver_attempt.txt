import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # Group triples by specialist (x)
    by_x = [[] for _ in range(k)]
    for x, y, z in triples:
        by_x[x].append((y, z))

    # memo[x][mask_y][mask_z]
    # Since k <= 22, we use recursion with memoization.
    # However, 2^22 * 2^22 is too large.
    # But we only need to pick k triples, one for each x.
    # This is equivalent to finding a matching in a 3D hypergraph.
    # Given k <= 22, we can use backtracking with pruning.
    
    memo = {}

    def backtrack(x, mask_y, mask_z):
        if x == k:
            return []
        
        state = (x, mask_y, mask_z)
        if state in memo:
            return memo[state]
        
        for y, z in by_x[x]:
            if not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                res = backtrack(x + 1, mask_y | (1 << y), mask_z | (1 << z))
                if res is not None:
                    memo[state] = [(x + 1, y + 1, z + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for crew in result:
            print(f"{crew[0]} {crew[1]} {crew[2]}")
    else:
        print("NO")

solve()
