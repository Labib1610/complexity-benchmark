```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    procedures = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        procedures.append((x, y, z))
        idx += 3
        
    # This is an Exact Cover problem.
    # We need to select k triples (x, y, z) such that:
    # - Each x in {0..k-1} appears exactly once.
    # - Each y in {0..k-1} appears exactly once.
    # - Each z in {0..k-1} appears exactly once.
    
    # Since k <= 22, we can use backtracking with pruning.
    # Represent constraints as bitmasks.
    
    # Group procedures by isotope x
    by_x = [[] for _ in range(k)]
    for p in procedures:
        by_x[p[0]].append(p)
        
    used_y = 0
    used_z = 0
    result = []
    
    def backtrack(x_idx):
        if x_idx == k:
            return True
        
        for p in by_x[x_idx]:
            _, y, z = p
            if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                # Try this procedure
                nonlocal used_y, used_z
                used_y |= (1 << y)
                used_z |= (1 << z)
                result.append(p)
                
                if backtrack(x_idx + 1):
                    return True
                
                # Backtrack
                result.pop()
                used_y &= ~(1 << y)
                used_z &= ~(1 << z)
        return False

    if backtrack(0):
        print("YES")
        for p in result:
            print(f"{p[0]+1} {p[1]+1} {p[2]+1}")
    else:
        print("NO")

solve()
```
