```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    # templates[x][y] = list of z
    adj = [[[] for _ in range(k + 1)] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        adj[x][y].append(z)
        idx += 3

    # We need to select (x, y, z) for each x in 1..k such that
    # all y are distinct and all z are distinct.
    # This is equivalent to finding a Latin Square or a 3D matching.
    # Since k <= 22, we use backtracking with pruning.
    
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []

    def backtrack(x):
        if x > k:
            return True
        
        # Try all rooms y for surgeon x
        for y in range(1, k + 1):
            if not used_y[y]:
                for z in adj[x][y]:
                    if not used_z[z]:
                        used_y[y] = True
                        used_z[z] = True
                        result.append((x, y, z))
                        
                        if backtrack(x + 1):
                            return True
                        
                        result.pop()
                        used_z[z] = False
                        used_y[y] = False
        return False

    # Optimization: sort rooms by number of available z to fail faster
    if backtrack(1):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
```
