import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        triples.append((x, y, z))
        idx += 3

    # The condition sum(x+y+z) = 3k(k+1)/2 is equivalent to 
    # sum(x) + sum(y) + sum(z) = 3 * k(k-1)/2.
    # Since we must use each 0..k-1 exactly once, the sum of x, y, and z 
    # will always be 3 * k(k-1)/2.
    # This is a 3D matching problem. Since k <= 22, we use backtracking with pruning.
    
    used_x = [False] * k
    used_y = [False] * k
    used_z = [False] * k
    
    # Sort triples to improve pruning
    triples.sort()
    
    res = []
    
    def backtrack(idx, count):
        if count == k:
            return True
        if idx == t:
            return False
        
        # Pruning: if remaining triples are not enough to cover remaining slots
        if t - idx < k - count:
            return False
            
        x, y, z = triples[idx]
        
        # Option 1: Include this triple
        if not used_x[x] and not used_y[y] and not used_z[z]:
            used_x[x] = used_y[y] = used_z[z] = True
            res.append((x + 1, y + 1, z + 1))
            if backtrack(idx + 1, count + 1):
                return True
            res.pop()
            used_x[x] = used_y[y] = used_z[z] = False
            
        # Option 2: Skip this triple
        if backtrack(idx + 1, count):
            return True
            
        return False

    if backtrack(0, 0):
        print("YES")
        for triple in res:
            print(f"{triple[0]} {triple[1]} {triple[2]}")
    else:
        print("NO")

solve()
