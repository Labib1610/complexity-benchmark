```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    pockets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        pockets.append((x, y, z))
        idx += 3
        
    # This is a 3D matching problem, which is NP-hard.
    # However, k is small (up to 22).
    # We can model this as finding a set of k edges in a 3-partite hypergraph.
    # Since we need to pick k points such that all x, y, z are distinct,
    # we can use backtracking with pruning.
    
    adj = [[] for _ in range(k + 1)]
    for p in pockets:
        adj[p[0]].append(p)
        
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []
    
    def backtrack(x):
        if x > k:
            return True
        
        for p in adj[x]:
            _, y, z = p
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append(p)
                
                if backtrack(x + 1):
                    return True
                
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    # Sort pockets to potentially improve pruning
    for i in range(1, k + 1):
        adj[i].sort(key=lambda p: (used_y[p[1]], used_z[p[2]]))
        
    if backtrack(1):
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
```
