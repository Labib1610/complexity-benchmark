import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        triplets.append((x - 1, y - 1, z - 1))
        idx += 3

    # Exact Cover Problem:
    # Rows: triplets
    # Columns: 3*k constraints (k frequencies, k pathways, k thresholds)
    # We need to pick k rows such that each column is covered exactly once.
    
    # Using Dancing Links (DLX) approach or backtracking with pruning
    # Given k <= 22, this is a 3D matching problem.
    
    rows = triplets
    
    # State: (index_in_triplets, used_freq_mask, used_path_mask, used_thresh_mask, selected_indices)
    # Since k is small (up to 22), we can use bitmasks for columns.
    
    # Optimization: Sort triplets to prune search space
    # Backtracking with bitmasks
    
    selected = []
    
    def backtrack(start_idx, freq_mask, path_mask, thresh_mask, count):
        if count == k:
            return True
        
        # Heuristic: pick the first unused frequency and try all triplets that cover it
        # This reduces branching factor significantly
        f = 0
        while (freq_mask >> f) & 1:
            f += 1
            
        for i in range(start_idx, t):
            x, y, z = rows[i]
            if x == f:
                if not ((path_mask >> y) & 1) and not ((thresh_mask >> z) & 1):
                    selected.append(rows[i])
                    if backtrack(i + 1, freq_mask | (1 << x), path_mask | (1 << y), thresh_mask | (1 << z), count + 1):
                        return True
                    selected.pop()
        return False

    # The above simple backtracking might be slow. Let's use a more robust approach.
    # Since k is small, we can use the fact that we need to cover all k frequencies.
    
    def solve_dlx():
        # freq_mask, path_mask, thresh_mask
        # We need to pick k triplets.
        
        # Pre-filter: group by frequency
        by_freq = [[] for _ in range(k)]
        for i, (x, y, z) in enumerate(rows):
            by_freq[x].append((y, z, i))
            
        res = []
        def search(f, p_mask, t_mask):
            if f == k:
                return True
            
            for y, z, i in by_freq[f]:
                if not ((p_mask >> y) & 1) and not ((t_mask >> z) & 1):
                    res.append(rows[i])
                    if search(f + 1, p_mask | (1 << y), t_mask | (1 << z)):
                        return True
                    res.pop()
            return False

        if search(0, 0, 0):
            return res
        return None

    result = solve_dlx()
    if result:
        print("YES")
        for x, y, z in result:
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")

solve()
