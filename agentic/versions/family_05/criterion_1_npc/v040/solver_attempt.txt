```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        f = int(input_data[idx]) - 1
        c = int(input_data[idx+1]) - 1
        m = int(input_data[idx+2]) - 1
        triples.append((f, c, m))
        idx += 3
        
    # We need to select k triples such that each f, c, m appears exactly once.
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    adj = [[] for _ in range(k)]
    for i, (f, c, m) in enumerate(triples):
        adj[f].append(i)
        
    used_c = [False] * k
    used_m = [False] * k
    result = []
    
    def backtrack(f_idx):
        if f_idx == k:
            return True
        
        for triple_idx in adj[f_idx]:
            _, c, m = triples[triple_idx]
            if not used_c[c] and not used_m[m]:
                used_c[c] = True
                used_m[m] = True
                result.append(triples[triple_idx])
                
                if backtrack(f_idx + 1):
                    return True
                
                result.pop()
                used_c[c] = False
                used_m[m] = False
        return False

    if backtrack(0):
        print("YES")
        for f, c, m in result:
            print(f"{f + 1} {c + 1} {m + 1}")
    else:
        print("NO")

solve()
```
