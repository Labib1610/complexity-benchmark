import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    profiles = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        profiles.append((x, y, z))
        idx += 3

    # Group profiles by operator x
    by_x = [[] for _ in range(k)]
    for p in profiles:
        by_x[p[0]].append((p[1], p[2]))

    # State: (mask_y, mask_z)
    # Since k <= 22, we use bitmasks.
    # We need to pick one profile for each x from 0 to k-1.
    # This is equivalent to finding a set of k triples (x, y, z)
    # such that all x, y, z are distinct.
    
    memo = {}

    def backtrack(x, mask_y, mask_z):
        if x == k:
            return []
        
        state = (x, mask_y, mask_z)
        if state in memo:
            return memo[state]
        
        for y, z in by_x[x]:
            if not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                res = backtrack(x + 1, mask_y | (1 << y), mask_z | (1 << z))
                if res is not None:
                    memo[state] = [(x + 1, y + 1, z + 1)] + res
                    return memo[state]
        
        memo[state] = None
        return None

    result = backtrack(0, 0, 0)
    
    if result is not None:
        print("YES")
        for p in result:
            print(f"{p[0]} {p[1]} {p[2]}")
    else:
        print("NO")

solve()
