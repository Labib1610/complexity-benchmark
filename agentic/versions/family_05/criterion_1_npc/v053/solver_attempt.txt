import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    profiles_by_freq = [[] for _ in range(k + 1)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        profiles_by_freq[x].append((y, z))
        idx += 3
        
    # We need to select one profile for each frequency 1..k
    # such that all voltages 1..k and all phases 1..k are used exactly once.
    # This is equivalent to finding a system of distinct representatives or 
    # a matching in a specific hypergraph, but since k is small (up to 22),
    # we can use backtracking with pruning.
    
    used_v = [False] * (k + 1)
    used_p = [False] * (k + 1)
    result = []
    
    def backtrack(f):
        if f > k:
            return True
        
        for v, p in profiles_by_freq[f]:
            if not used_v[v] and not used_p[p]:
                used_v[v] = True
                used_p[p] = True
                result.append((f, v, p))
                
                if backtrack(f + 1):
                    return True
                
                result.pop()
                used_v[v] = False
                used_p[p] = False
        return False

    # Sort frequencies by number of available profiles to prune search space
    freqs = sorted(range(1, k + 1), key=lambda x: len(profiles_by_freq[x]))
    
    # Re-map to process in sorted order
    mapping = {f: i for i, f in enumerate(freqs, 1)}
    new_profiles = [[] for _ in range(k + 1)]
    for f in range(1, k + 1):
        new_profiles[mapping[f]] = profiles_by_freq[f]
    
    # Redefine backtrack to use new_profiles
    def backtrack_optimized(idx_f):
        if idx_f > k:
            return True
        for v, p in new_profiles[idx_f]:
            if not used_v[v] and not used_p[p]:
                used_v[v] = True
                used_p[p] = True
                result.append((freqs[idx_f-1], v, p))
                if backtrack_optimized(idx_f + 1):
                    return True
                result.pop()
                used_v[v] = False
                used_p[p] = False
        return False

    if backtrack_optimized(1):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
