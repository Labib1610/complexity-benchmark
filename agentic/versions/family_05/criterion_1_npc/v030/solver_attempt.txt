import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])
    
    candidates = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        candidates.append((x, y, z))
        idx += 3

    # This is an Exact Cover problem.
    # We need to pick k triples (x, y, z) such that all x are distinct,
    # all y are distinct, and all z are distinct.
    # Since 1 <= x, y, z <= k, this is equivalent to finding a permutation
    # of 1..k for y and a permutation of 1..k for z given x=1..k.
    # We can model this as finding a perfect matching in a 3-partite hypergraph,
    # or more simply, since x is fixed to 1..k, we need to pick k triples
    # (1, y1, z1), (2, y2, z2), ..., (k, yk, zk) such that
    # {y1, ..., yk} = {1, ..., k} and {z1, ..., z2} = {1, ..., k}.
    
    # Group candidates by x
    by_x = [[] for _ in range(k + 1)]
    for x, y, z in candidates:
        by_x[x].append((y, z))
        
    # Backtracking with pruning
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    result = []

    def backtrack(x):
        if x > k:
            return True
        for y, z in by_x[x]:
            if not used_y[y] and not used_z[z]:
                used_y[y] = True
                used_z[z] = True
                result.append((x, y, z))
                if backtrack(x + 1):
                    return True
                result.pop()
                used_y[y] = False
                used_z[z] = False
        return False

    if backtrack(1):
        print("YES")
        for item in result:
            print(f"{item[0]} {item[1]} {item[2]}")
    else:
        print("NO")

solve()
