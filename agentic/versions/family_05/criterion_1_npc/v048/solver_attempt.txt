import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    configs = []
    idx = 2
    for _ in range(t):
        c = int(input_data[idx]) - 1
        e = int(input_data[idx+1]) - 1
        g = int(input_data[idx+2]) - 1
        configs.append((c, e, g))
        idx += 3
        
    # This is a 3D matching problem, which is NP-complete.
    # Given k <= 22, we can use backtracking with pruning.
    
    used_c = [False] * k
    used_e = [False] * k
    used_g = [False] * k
    result = []
    
    # Group configs by chassis to speed up search
    by_c = [[] for _ in range(k)]
    for c, e, g in configs:
        by_c[c].append((e, g))
        
    def backtrack(c_idx):
        if c_idx == k:
            return True
        
        for e, g in by_c[c_idx]:
            if not used_e[e] and not used_g[g]:
                used_e[e] = True
                used_g[g] = True
                result.append((c_idx + 1, e + 1, g + 1))
                
                if backtrack(c_idx + 1):
                    return True
                
                result.pop()
                used_e[e] = False
                used_g[g] = False
        return False

    # Optimization: check if any component is impossible to satisfy
    # Count occurrences of each component
    count_c = [0] * k
    count_e = [0] * k
    count_g = [0] * k
    for c, e, g in configs:
        count_c[c] += 1
        count_e[e] += 1
        count_g[g] += 1
        
    if any(count_c[i] == 0 or count_e[i] == 0 or count_g[i] == 0 for i in range(k)):
        print("NO")
        return

    if backtrack(0):
        print("YES")
        for row in result:
            print(f"{row[0]} {row[1]} {row[2]}")
    else:
        print("NO")

solve()
