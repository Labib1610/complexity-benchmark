import sys
import heapq

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    terminals = [int(x) - 1 for x in data[3:3 + t]]
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n)]
    idx = 3 + t
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        w = int(data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    num_masks = 1 << t
    INF = float('inf')
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        dp[1 << i][terminals[i]] = 0
        
    for mask in range(1, num_masks):
        # Merge submasks
        submask = (mask - 1) & mask
        while submask > (mask ^ submask):
            other = mask ^ submask
            dp_sub = dp[submask]
            dp_oth = dp[other]
            dp_cur = dp[mask]
            for v in range(n):
                val = dp_sub[v] + dp_oth[v]
                if val < dp_cur[v]:
                    dp_cur[v] = val
            submask = (submask - 1) & mask
            
        # Dijkstra relaxation for the current mask
        dp_cur = dp[mask]
        pq = []
        for v in range(n):
            if dp_cur[v] < INF:
                pq.append((dp_cur[v], v))
        heapq.heapify(pq)
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp_cur[u]:
                continue
            for v, w in adj[u]:
                if dp_cur[v] > d + w:
                    dp_cur[v] = d + w
                    heapq.heappush(pq, (d + w, v))
                    
    ans = min(dp[num_masks - 1])
    print(ans)

if __name__ == '__main__':
    solve()
