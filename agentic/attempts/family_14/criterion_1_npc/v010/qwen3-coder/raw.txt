import sys
import heapq
from collections import defaultdict

def find_mst(n, edges, primary_set):
    # Prim's algorithm to find MST of the subgraph that connects all primary nodes
    # We can use a modified version that only cares about connecting primary nodes
    
    # First, let's build the graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If there's only one primary node, cost is 0
    if len(primary_set) <= 1:
        return 0
    
    # Use Dijkstra-like approach to find minimum cost to connect all primary nodes
    # This is essentially finding the minimum spanning tree that connects all primary nodes
    
    # We'll use a modified Prim's algorithm
    # Start with one primary node
    start = next(iter(primary_set))
    visited = set()
    min_heap = [(0, start)]
    total_cost = 0
    primary_connected = set()
    
    while min_heap and len(primary_connected) < len(primary_set):
        cost, node = heapq.heappop(min_heap)
        
        if node in visited:
            continue
            
        visited.add(node)
        if node in primary_set:
            primary_connected.add(node)
        total_cost += cost
        
        # Add all neighbors to the heap
        for neighbor, edge_cost in graph[node]:
            if neighbor not in visited:
                heapq.heappush(min_heap, (edge_cost, neighbor))
    
    return total_cost

def find_mst_optimal(n, edges, primary_nodes):
    if len(primary_nodes) <= 1:
        return 0
        
    # Create a graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Use Prim's algorithm to find MST that connects all primary nodes
    visited = set()
    primary_set = set(primary_nodes)
    min_heap = [(0, primary_nodes[0])]  # (cost, node)
    total_cost = 0
    primaries_connected = 0
    
    while min_heap and primaries_connected < len(primary_nodes):
        cost, node = heapq.heappop(min_heap)
        
        if node in visited:
            continue
            
        visited.add(node)
        total_cost += cost
        if node in primary_set:
            primaries_connected += 1
            
        # Add all neighbors to heap
        for neighbor, edge_cost in graph[node]:
            if neighbor not in visited:
                heapq.heappush(min_heap, (edge_cost, neighbor))
                
    return total_cost

# Read input
line1 = list(map(int, sys.stdin.readline().split()))
n, m, t = line1[0], line1[1], line1[2]

primary_nodes = list(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve using Prim's algorithm
result = find_mst_optimal(n, edges, primary_nodes)
print(result)
