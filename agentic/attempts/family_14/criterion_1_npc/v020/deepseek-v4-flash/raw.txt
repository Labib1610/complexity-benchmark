We need the minimum cost so that all primary languages are connected (mutually translatable). This is a classic *Steiner tree* problem, but since the number of primary languages \( t \le 10 \), we can use bitmask DP for Steiner tree on a graph with up to \( n = 100 \) nodes.

We'll compute shortest paths between all pairs first (Floyd–Warshall), then use DP with bitmasks.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    primary = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**15
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                nd = dist[i][k] + dist[k][j]
                if nd < dist[i][j]:
                    dist[i][j] = nd
    
    # Steiner tree DP
    # dp[mask][v] = min cost to connect terminals in mask, ending at node v
    dp = [[INF] * n for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][primary[i]] = 0
    
    for mask in range(1 << t):
        # combine submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub - 1) & mask
        
        # run Dijkstra-like relaxation using dist (or Bellman-Ford since n small)
        # Use Floyd–Warshall style DP: for each node pair, relax
        for u in range(n):
            if dp[mask][u] >= INF:
                continue
            for v in range(n):
                if dist[u][v] < INF:
                    nd = dp[mask][u] + dist[u][v]
                    if nd < dp[mask][v]:
                        dp[mask][v] = nd
    
    full_mask = (1 << t) - 1
    ans = min(dp[full_mask][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
