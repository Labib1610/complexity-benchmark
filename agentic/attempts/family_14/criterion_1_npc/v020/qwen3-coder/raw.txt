import sys
import heapq
from collections import defaultdict
import itertools

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    return dist

def main():
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    primary = list(map(int, data[idx:idx+t]))
    idx += t
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If there's only one primary language, no interpreters needed
    if t <= 1:
        print(0)
        return
    
    # Compute shortest paths from each primary language to all nodes
    distances = {}
    for p in primary:
        distances[p] = dijkstra(graph, p, n)
    
    # Build a complete graph of primary languages with edge weights as shortest path distances
    # Then find the minimum spanning tree of this graph
    
    # Generate all pairs of primary languages
    edges = []
    for i in range(t):
        for j in range(i+1, t):
            u = primary[i]
            v = primary[j]
            weight = distances[u][v]  # Shortest path distance between u and v
            edges.append((weight, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = {p: p for p in primary}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm to find MST
    total_cost = 0
    for weight, u, v in edges:
        if union(u, v):
            total_cost += weight
    
    print(total_cost)

if __name__ == "__main__":
    main()
