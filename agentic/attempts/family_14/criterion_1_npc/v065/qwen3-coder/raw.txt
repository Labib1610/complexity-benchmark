import sys
from itertools import combinations

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, nodes):
    parent = list(range(n))
    rank = [0] * n
    edges.sort(key=lambda x: x[2])
    cost = 0
    connected = 0
    needed = len(nodes) - 1
    
    if needed <= 0:
        return 0

    for u, v, w in edges:
        if union(parent, rank, u, v):
            cost += w
            connected += 1
            if connected == needed:
                break
    return cost

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical = list(map(int, sys.stdin.readline().split()))
    critical = [x - 1 for x in critical]  # Convert to 0-indexed
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1, w))  # Convert to 0-indexed
    
    # If there's 0 or 1 critical module, no connection needed
    if t <= 1:
        print(0)
        return
    
    # Try all 2^t subsets of critical nodes as terminals
    min_cost = float('inf')
    
    # We'll use a modified approach: create a complete graph on critical nodes
    # where edge weights are shortest path distances in the original graph
    # Then find MST of this complete graph
    
    # Floyd-Warshall to get all-pairs shortest paths
    INF = float('inf')
    dist = [[INF]*n for _ in range(n)]
    
    for i in range(n):
        dist[i][i] = 0
        
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # Create edges between all pairs of critical nodes with shortest path weights
    critical_edges = []
    for i in range(len(critical)):
        for j in range(i+1, len(critical)):
            u = critical[i]
            v = critical[j]
            if dist[u][v] < INF:
                critical_edges.append((u, v, dist[u][v]))
    
    # Run Kruskal's algorithm on this complete graph
    result = kruskal(n, critical_edges, critical)
    print(result)

solve()
