We can model this as a **Steiner tree** problem, but since the number of critical terminals is small (≤ 10), we can solve it using **bitmask DP** over subsets of terminals.

We’ll:
1. Run **Floyd–Warshall** to get shortest paths between all nodes.
2. Use **DP over subsets** of terminals:
   - `dp[mask][v]` = minimum cost to connect the terminals in `mask` with the tree rooted at node `v`.
   - Initialize with single-terminal masks.
   - Combine two disjoint masks at the same node.
   - Relax using shortest paths to other nodes.
3. Answer is `min over v` of `dp[full_mask][v]`.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]

    INF = 10**15
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    full_mask = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]

    # Initialize single terminal masks
    for i, term in enumerate(terminals):
        dp[1 << i][term] = 0

    # DP over subsets
    for mask in range(1, 1 << t):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    if dp[sub][v] < INF and dp[other][v] < INF:
                        val = dp[sub][v] + dp[other][v]
                        if val < dp[mask][v]:
                            dp[mask][v] = val
            sub = (sub - 1) & mask

        # Relax via shortest paths
        for v in range(n):
            if dp[mask][v] < INF:
                dv = dp[mask][v]
                for u in range(n):
                    if dist[v][u] < INF:
                        nd = dv + dist[v][u]
                        if nd < dp[mask][u]:
                            dp[mask][u] = nd

    ans = min(dp[full_mask])
    print(ans)

if __name__ == "__main__":
    solve()
```
