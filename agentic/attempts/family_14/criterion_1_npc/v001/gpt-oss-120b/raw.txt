```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    INF = 10**18
    # adjacency matrix for Floyd
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            ik = di[k]
            if ik == INF:
                continue
            for j in range(n):
                nd = ik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    if t <= 1:
        print(0)
        return
    full = 1 << t
    dp = [[INF] * n for _ in range(full)]
    # initialize singletons
    for idx, term in enumerate(terminals):
        mask = 1 << idx
        dterm = dist[term]
        row = dp[mask]
        for v in range(n):
            row[v] = dterm[v]
    # DP over subsets
    for mask in range(1, full):
        # skip singletons
        if mask & (mask - 1):
            # merge submasks
            sub = (mask - 1) & mask
            while sub:
                other = mask ^ sub
                if other:
                    row_mask = dp[mask]
                    row_sub = dp[sub]
                    row_other = dp[other]
                    for v in range(n):
                        val = row_sub[v] + row_other[v]
                        if val < row_mask[v]:
                            row_mask[v] = val
                sub = (sub - 1) & mask
        # propagate through graph: dp[mask][v] = min_i dp[mask][i] + dist[i][v]
        row = dp[mask]
        # compute min_i dp[mask][i] + dist[i][v] for each v
        new_row = [INF] * n
        for i in range(n):
            di = row[i]
            if di == INF:
                continue
            dist_i = dist[i]
            base = di
            for v in range(n):
                nd = base + dist_i[v]
                if nd < new_row[v]:
                    new_row[v] = nd
        dp[mask] = new_row
    ans = min(dp[full - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
