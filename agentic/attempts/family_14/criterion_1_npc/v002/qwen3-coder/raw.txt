import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, end_set):
    """Return the minimum distance from start to any node in end_set"""
    if start in end_set:
        return 0
    
    dist = {}
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if u in dist:
            continue
        dist[u] = d
        
        if u in end_set:
            return d
            
        for v, w in graph[u]:
            if v not in dist:
                heapq.heappush(pq, (d + w, v))
    
    return float('inf')

def main():
    # Read input
    n, m, t = map(int, sys.stdin.readline().split())
    delegated = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    # Build graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((u, v, w))
    
    # Compute minimum spanning tree for delegated languages
    # We use a modified Prim's algorithm
    
    # First, compute shortest paths between all pairs of delegated languages
    # to form a complete graph on delegated vertices
    delegated_set = set(delegated)
    cost_matrix = {}
    
    for i, lang1 in enumerate(delegated):
        cost_matrix[lang1] = {}
        for lang2 in delegated:
            if lang1 != lang2:
                min_cost = dijkstra(graph, lang1, {lang2})
                cost_matrix[lang1][lang2] = min_cost
    
    # Now find MST of this complete graph using Prim's algorithm
    included = {delegated[0]}
    remaining = set(delegated[1:])
    total_cost = 0
    edge_heap = []
    
    # Initialize heap with edges from first delegated language
    for lang in remaining:
        heapq.heappush(edge_heap, (cost_matrix[delegated[0]][lang], delegated[0], lang))
    
    while remaining and edge_heap:
        weight, u, v = heapq.heappop(edge_heap)
        
        # If both nodes are already included, skip
        if v in included:
            continue
            
        # Add the edge to MST
        included.add(v)
        remaining.discard(v)
        total_cost += weight
        
        # Add new edges to the heap
        for lang in remaining:
            heapq.heappush(edge_heap, (cost_matrix[v][lang], v, lang))
    
    print(total_cost)

if __name__ == "__main__":
    main()
