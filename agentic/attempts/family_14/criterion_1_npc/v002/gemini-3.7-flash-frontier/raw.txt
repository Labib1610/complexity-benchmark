```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    t = int(next(iterator))
    
    terminals = [int(next(iterator)) - 1 for _ in range(t)]
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        w = int(next(iterator))
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    if t <= 1:
        print(0)
        return

    num_masks = 1 << t
    INF = 10**18
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        dp[1 << i][terminals[i]] = 0

    for mask in range(1, num_masks):
        dp_m = dp[mask]
        
        # Combine submasks
        s = (mask - 1) & mask
        while s > (mask ^ s):
            dp_s = dp[s]
            dp_comp = dp[mask ^ s]
            for u in range(n):
                val = dp_s[u] + dp_comp[u]
                if val < dp_m[u]:
                    dp_m[u] = val
            s = (s - 1) & mask
            
        # Dijkstra
        pq = []
        for u in range(n):
            if dp_m[u] < INF:
                pq.append((dp_m[u], u))
        heapq.heapify(pq)
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp_m[u]:
                continue
            for v, w in adj[u]:
                if d + w < dp_m[v]:
                    dp_m[v] = d + w
                    heapq.heappush(pq, (d + w, v))
                    
    ans = min(dp[num_masks - 1])
    print(ans)

if __name__ == '__main__':
    solve()
```
