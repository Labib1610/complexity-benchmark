import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    terminals = set()
    for _ in range(t):
        terminals.add(int(next(it)))
    
    # Initialize adjacency matrix with infinity
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall to find all-pairs shortest paths
    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(1, n + 1):
                if di[j] > dik + dk[j]:
                    di[j] = dik + dk[j]
    
    terminals_list = list(terminals)
    num_terminals = len(terminals_list)
    
    if num_terminals <= 1:
        print(0)
        return
    
    # Steiner Tree DP
    # dp[mask][v] = min cost to connect terminals in mask, ending at node v
    dp = [[INF] * (n + 1) for _ in range(1 << num_terminals)]
    
    # Base case: single terminal
    for i, term in enumerate(terminals_list):
        dp[1 << i][term] = 0
    
    # Iterate over all masks
    for mask in range(1, 1 << num_terminals):
        # Skip single terminal masks (already initialized)
        if (mask & (mask - 1)) == 0:
            continue
        
        # Combine two submasks
        submask = (mask - 1) & mask
        while submask:
            other = mask ^ submask
            for v in range(1, n + 1):
                new_cost = dp[submask][v] + dp[other][v]
                if new_cost < dp[mask][v]:
                    dp[mask][v] = new_cost
            submask = (submask - 1) & mask
        
        # Propagate through Steiner vertices (Dijkstra-like)
        # Find minimum dp[mask][v] and relax neighbors
        # Since we have small n, we can use simple O(n^2) approach
        visited = [False] * (n + 1)
        for _ in range(n):
            min_dist = INF
            u = -1
            for v in range(1, n + 1):
                if not visited[v] and dp[mask][v] < min_dist:
                    min_dist = dp[mask][v]
                    u = v
            if u == -1:
                break
            visited[u] = True
            du = dist[u]
            dpm = dp[mask]
            dpmu = dpm[u]
            for v in range(1, n + 1):
                if dpm[v] > dpmu + du[v]:
                    dpm[v] = dpmu + du[v]
    
    full_mask = (1 << num_terminals) - 1
    result = min(dp[full_mask][v] for v in range(1, n + 1))
    print(int(result))

if __name__ == "__main__":
    solve()
