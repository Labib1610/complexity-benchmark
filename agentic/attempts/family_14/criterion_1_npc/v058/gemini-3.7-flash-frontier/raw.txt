```python
import sys


def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])
    t = int(data[2])

    idx = 3
    terminals = [int(data[idx + i]) - 1 for i in range(t)]
    idx += t

    if t <= 1:
        print(0)
        return

    INF = float("inf")
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        w = int(data[idx + 2])
        idx += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    for k in range(n):
        dist_k = dist[k]
        for i in range(n):
            dist_i = dist[i]
            dik = dist_i[k]
            if dik == INF:
                continue
            for j in range(n):
                if dist_k[j] != INF and dik + dist_k[j] < dist_i[j]:
                    dist_i[j] = dik + dist_k[j]

    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]

    for i in range(t):
        term = terminals[i]
        for u in range(n):
            dp[1 << i][u] = dist[term][u]

    for mask in range(1, num_masks):
        # Check if popcount >= 2
        if (mask & (mask - 1)) != 0:
            dp_mask = dp[mask]
            sub = (mask - 1) & mask
            while sub > (mask ^ sub):
                dp_sub = dp[sub]
                dp_diff = dp[mask ^ sub]
                for u in range(n):
                    val = dp_sub[u] + dp_diff[u]
                    if val < dp_mask[u]:
                        dp_mask[u] = val
                sub = (sub - 1) & mask

            # Relaxation over all shortest paths
            new_dp = [INF] * n
            for u in range(n):
                min_val = INF
                for v in range(n):
                    val = dp_mask[v] + dist[v][u]
                    if val < min_val:
                        min_val = val
                new_dp[u] = min_val
            dp[mask] = new_dp

    ans = min(dp[num_masks - 1])
    print(ans)


if __name__ == "__main__":
    main()
```
