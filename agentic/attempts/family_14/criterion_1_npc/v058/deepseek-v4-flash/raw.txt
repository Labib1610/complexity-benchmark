We need the minimum total impedance to connect all primary generators, possibly through other substations.  
This is essentially the **Steiner tree** problem, but with a small number of terminals (t ≤ 10), so we can solve it with **bitmask DP over subsets of terminals**.

We’ll:
1. Compute all-pairs shortest paths (Floyd–Warshall) since n ≤ 100.
2. Use DP over subsets of terminals:  
   `dp[mask][v]` = minimum cost to connect the terminals in `mask` with the tree rooted at node `v`.
3. Combine subsets and relax using shortest paths.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    # DP
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]
    
    # Initialize: single terminal
    for i, term in enumerate(terminals):
        dp[1 << i][term] = 0
    
    # For each mask, combine
    for mask in range(1, full_mask):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    best = INF
                    for a in range(n):
                        val = dp[sub][a] + dist[a][v]
                        if val < best:
                            best = val
                    for b in range(n):
                        val = dp[other][b] + dist[b][v]
                        if val < best:
                            best = val
                    if best < dp[mask][v]:
                        dp[mask][v] = best
            sub = (sub - 1) & mask
        
        # Relax within mask using shortest paths
        for v in range(n):
            if dp[mask][v] < INF:
                for u in range(n):
                    nd = dp[mask][v] + dist[v][u]
                    if nd < dp[mask][u]:
                        dp[mask][u] = nd
    
    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
