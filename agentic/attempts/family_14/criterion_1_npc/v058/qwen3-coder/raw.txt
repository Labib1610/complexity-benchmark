```python
import sys
import heapq
from collections import defaultdict

def find_mst_cost(n, edges, generators):
    if len(generators) <= 1:
        return 0
    
    # Create a graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Use Dijkstra's algorithm to find shortest paths from each generator to all other nodes
    # Then create a complete graph between generators with edge weights as shortest path distances
    gen_set = set(generators)
    
    # For each generator, compute shortest distances to all other nodes
    distances = {}
    for gen in generators:
        dist = [float('inf')] * (n + 1)
        dist[gen] = 0
        pq = [(0, gen)]
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
                
            for v, w in graph[u]:
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    heapq.heappush(pq, (dist[v], v))
        
        distances[gen] = dist
    
    # Create a complete graph with generators as nodes
    # Edge weight between two generators is the shortest path distance between them
    gen_edges = []
    for i in range(len(generators)):
        for j in range(i + 1, len(generators)):
            g1, g2 = generators[i], generators[j]
            weight = distances[g1][g2]
            gen_edges.append((weight, g1, g2))
    
    # Find MST of this complete graph using Kruskal's algorithm
    gen_edges.sort()
    
    parent = {}
    def find(x):
        if x not in parent:
            parent[x] = x
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in gen_edges:
        if union(u, v):
            total_cost += weight
            edges_used += 1
            if edges_used == len(generators) - 1:
                break
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

generators = list(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve and output
result = find_mst_cost(n, edges, generators)
print(result)
```
