```python
import sys
import heapq

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    if t == 1:
        print(0)
        return

    inf = 10**30
    total_masks = 1 << t
    dp = [[inf] * n for _ in range(total_masks)]

    for i, terminal in enumerate(terminals):
        dp[1 << i][terminal] = 0

    for mask in range(1, total_masks):
        current = dp[mask]

        if mask & (mask - 1):
            lowest = mask & -mask
            rest = mask ^ lowest
            subrest = (rest - 1) & rest

            while True:
                left = lowest | subrest
                right = mask ^ left
                dl = dp[left]
                dr = dp[right]

                for v in range(n):
                    candidate = dl[v] + dr[v]
                    if candidate < current[v]:
                        current[v] = candidate

                if subrest == 0:
                    break
                subrest = (subrest - 1) & rest

        heap = [(d, v) for v, d in enumerate(current) if d < inf]
        heapq.heapify(heap)

        while heap:
            distance, u = heapq.heappop(heap)
            if distance != current[u]:
                continue

            for v, weight in graph[u]:
                candidate = distance + weight
                if candidate < current[v]:
                    current[v] = candidate
                    heapq.heappush(heap, (candidate, v))

    print(min(dp[total_masks - 1]))

if __name__ == "__main__":
    main()
```
