import sys
import heapq

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    required = set()
    for _ in range(t):
        required.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    # Compute pairwise shortest distances between required nodes
    required_list = list(required)
    r_idx = {node: i for i, node in enumerate(required_list)}
    k = len(required_list)
    
    INF = 10**18
    dist_req = [[INF] * k for _ in range(k)]
    
    for i, start in enumerate(required_list):
        # Dijkstra from start
        dist = [INF] * (n + 1)
        dist[start] = 0
        heap = [(0, start)]
        while heap:
            d, u = heapq.heappop(heap)
            if d > dist[u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(heap, (nd, v))
        
        for j, node in enumerate(required_list):
            dist_req[i][j] = dist[node]
    
    # Steiner Tree DP on required nodes
    # dp[mask][v] = min cost to connect required nodes in mask, with v as endpoint
    # v is indexed by required_list (0 to k-1)
    
    full_mask = (1 << k) - 1
    dp = [[INF] * k for _ in range(1 << k)]
    
    # Base case: single node masks
    for i in range(k):
        dp[1 << i][i] = 0
    
    # Iterate over all masks
    for mask in range(1, 1 << k):
        # First, try to combine two submasks
        submask = (mask - 1) & mask
        while submask:
            other = mask ^ submask
            for i in range(k):
                if dp[submask][i] < INF and dp[other][i] < INF:
                    new_cost = dp[submask][i] + dp[other][i]
                    if new_cost < dp[mask][i]:
                        dp[mask][i] = new_cost
            submask = (submask - 1) & mask
        
        # Then, relax using edges between required nodes
        # Use Dijkstra-like relaxation
        # Priority queue: (cost, node_index)
        heap = []
        for i in range(k):
            if dp[mask][i] < INF:
                heapq.heappush(heap, (dp[mask][i], i))
        
        while heap:
            d, u = heapq.heappop(heap)
            if d > dp[mask][u]:
                continue
            for v in range(k):
                if v == u:
                    continue
                nd = d + dist_req[u][v]
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
                    heapq.heappush(heap, (nd, v))
    
    result = min(dp[full_mask])
    print(result)

if __name__ == "__main__":
    solve()
