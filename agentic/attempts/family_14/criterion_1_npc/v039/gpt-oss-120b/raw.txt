```python
import sys
import math

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]

    INF = 10**18
    # distance matrix
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w

    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            ik = di[k]
            if ik == INF:
                continue
            dj = dist
            for j in range(n):
                nd = ik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    if t <= 1:
        print(0)
        return

    full = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]

    # initialize singletons
    for idx, term in enumerate(terminals):
        mask = 1 << idx
        for v in range(n):
            dp[mask][v] = dist[term][v]

    # DP over subsets
    for mask in range(1, full + 1):
        # skip singletons (already initialized)
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            # combine only when both non-empty
            if other:
                dp_sub = dp[sub]
                dp_other = dp[other]
                dp_mask = dp[mask]
                for v in range(n):
                    val = dp_sub[v] + dp_other[v]
                    if val < dp_mask[v]:
                        dp_mask[v] = val
            sub = (sub - 1) & mask

        # closure: propagate through shortest paths
        dp_mask = dp[mask]
        # compute new values using min_u dp_mask[u] + dist[u][v]
        # simple O(n^2)
        new = [INF] * n
        for u in range(n):
            du = dp_mask[u]
            if du == INF:
                continue
            row = dist[u]
            for v in range(n):
                nd = du + row[v]
                if nd < new[v]:
                    new[v] = nd
        dp[mask] = new

    answer = min(dp[full])
    print(answer)


if __name__ == "__main__":
    solve()
```
