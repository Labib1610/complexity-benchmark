import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    t = int(next(iterator))
    
    focal = [int(next(iterator)) - 1 for _ in range(t)]
    
    if t <= 1:
        print(0)
        return

    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
        
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        w = int(next(iterator))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
            
    # Floyd-Warshall
    for k in range(n):
        dist_k = dist[k]
        for i in range(n):
            dist_i = dist[i]
            dik = dist_i[k]
            if dik == INF:
                continue
            for j in range(n):
                if dist_i[j] > dik + dist_k[j]:
                    dist_i[j] = dik + dist_k[j]

    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        term = focal[i]
        mask = 1 << i
        dp_mask = dp[mask]
        for v in range(n):
            dp_mask[v] = dist[term][v]

    for mask in range(1, num_masks):
        # Skip base cases already initialized
        if (mask & (mask - 1)) == 0:
            continue
            
        dp_mask = dp[mask]
        
        # We can fix the lowest set bit to avoid symmetry
        lowbit = mask & -mask
        submask = (mask - 1) & mask
        while submask > 0:
            if submask & lowbit:
                comp = mask ^ submask
                dp_sub = dp[submask]
                dp_comp = dp[comp]
                for v in range(n):
                    val = dp_sub[v] + dp_comp[v]
                    if val < dp_mask[v]:
                        dp_mask[v] = val
            submask = (submask - 1) & mask
            
        # Metric relaxation
        temp = dp_mask[:]
        for u in range(n):
            min_val = INF
            dist_u = dist[u]
            for v in range(n):
                val = temp[v] + dist_u[v]
                if val < min_val:
                    min_val = val
            dp_mask[u] = min_val

    ans = min(dp[(1 << t) - 1])
    print(ans)

if __name__ == '__main__':
    solve()
