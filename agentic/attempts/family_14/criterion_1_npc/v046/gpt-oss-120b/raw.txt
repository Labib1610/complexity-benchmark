```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            aik = di[k]
            if aik == INF:
                continue
            for j in range(n):
                b = aik + dk[j]
                if b < di[j]:
                    di[j] = b
    if t == 1:
        print(0)
        return
    full = 1 << t
    dp = [[INF] * n for _ in range(full)]
    # initialize singletons
    for i, term in enumerate(terminals):
        mask = 1 << i
        dterm = dist[term]
        row = dp[mask]
        for v in range(n):
            row[v] = dterm[v]
    # DP over subsets
    for mask in range(1, full):
        # skip singletons (already optimal)
        if mask & (mask - 1):
            # combine submasks
            sub = (mask - 1) & mask
            while sub:
                other = mask ^ sub
                if other:
                    dp_sub = dp[sub]
                    dp_other = dp[other]
                    dp_mask = dp[mask]
                    for v in range(n):
                        val = dp_sub[v] + dp_other[v]
                        if val < dp_mask[v]:
                            dp_mask[v] = val
                sub = (sub - 1) & mask
        # relaxation via shortest paths
        dp_mask = dp[mask]
        # min-plus multiplication dp_mask = dp_mask * dist
        # we can do for v: dp_mask[v] = min_u dp_mask[u] + dist[u][v]
        new = dp_mask[:]  # copy to avoid using updated values in same iteration
        for u in range(n):
            du = dp_mask[u]
            if du == INF:
                continue
            drow = dist[u]
            for v in range(n):
                nd = du + drow[v]
                if nd < new[v]:
                    new[v] = nd
        dp[mask] = new
    ans = min(dp[full - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
