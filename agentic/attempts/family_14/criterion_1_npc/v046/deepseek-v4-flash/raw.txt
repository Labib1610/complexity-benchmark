We can solve this by computing the **Steiner tree** on a small set of terminals (t ≤ 10).  
We’ll use the standard **DP over subsets** approach:

1. Compute all-pairs shortest paths (Floyd–Warshall) since n ≤ 100.
2. Let `dp[mask][v]` = minimum cost to connect the terminals in `mask` with the tree rooted at node `v`.
3. Initialize for each terminal `i` and node `v`: `dp[1<<i][v] = dist[terminal_i][v]`.
4. For each mask, combine two submasks and relax over all nodes.
5. Then run a shortest-path-like relaxation over all nodes for each mask (like Dijkstra or repeated relaxation).
6. Answer is `min_v dp[(1<<t)-1][v]`.

```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # DP over subsets
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]
    
    for i, term in enumerate(terminals):
        for v in range(n):
            dp[1 << i][v] = dist[term][v]
    
    for mask in range(1, full_mask):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    if dp[sub][v] + dp[other][v] < dp[mask][v]:
                        dp[mask][v] = dp[sub][v] + dp[other][v]
            sub = (sub - 1) & mask
        
        # Dijkstra-like relaxation for this mask
        pq = []
        for v in range(n):
            if dp[mask][v] < INF:
                heapq.heappush(pq, (dp[mask][v], v))
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(n):
                if dist[u][v] < INF:
                    nd = d + dist[u][v]
                    if nd < dp[mask][v]:
                        dp[mask][v] = nd
                        heapq.heappush(pq, (nd, v))
    
    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
