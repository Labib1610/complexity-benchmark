import sys
import heapq

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    if t == 1:
        print(0)
        return

    size = 1 << t
    inf = 10**30
    dp = [[inf] * n for _ in range(size)]

    for i, terminal in enumerate(terminals):
        dp[1 << i][terminal] = 0

    for mask in range(1, size):
        current = dp[mask]

        if mask & (mask - 1):
            fixed_bit = mask & -mask
            sub = (mask - 1) & mask

            while sub:
                if sub & fixed_bit:
                    other = mask ^ sub
                    left = dp[sub]
                    right = dp[other]
                    for v in range(n):
                        value = left[v] + right[v]
                        if value < current[v]:
                            current[v] = value
                sub = (sub - 1) & mask

        if mask == size - 1:
            print(min(current))
            return

        heap = [(distance, v) for v, distance in enumerate(current)
                if distance < inf]
        heapq.heapify(heap)

        while heap:
            distance, u = heapq.heappop(heap)
            if distance != current[u]:
                continue

            for v, weight in graph[u]:
                new_distance = distance + weight
                if new_distance < current[v]:
                    current[v] = new_distance
                    heapq.heappush(heap, (new_distance, v))

if __name__ == "__main__":
    main()
