We can solve this by treating it as a **Steiner tree** problem, but since the number of required sites is small (≤ 10), we can use **bitmask DP over required sites** after computing all-pairs shortest paths.

```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    required = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # DP over subsets of required sites
    # dp[mask][v] = min cost to connect required sites in mask, ending at v
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]
    
    for i in range(t):
        dp[1 << i][required[i]] = 0
    
    for mask in range(full_mask):
        # Combine subsets
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for u in range(n):
                    if dp[sub][u] == INF:
                        continue
                    for v in range(n):
                        if dp[other][v] == INF:
                            continue
                        if dp[mask][u] > dp[sub][u] + dp[other][v] + dist[u][v]:
                            dp[mask][u] = dp[sub][u] + dp[other][v] + dist[u][v]
            sub = (sub - 1) & mask
        
        # Relax via shortest paths
        for u in range(n):
            if dp[mask][u] == INF:
                continue
            for v in range(n):
                if dist[u][v] == INF:
                    continue
                if dp[mask][v] > dp[mask][u] + dist[u][v]:
                    dp[mask][v] = dp[mask][u] + dist[u][v]
    
    ans = min(dp[full_mask - 1])
    print(ans)

if __name__ == "__main__":
    solve()
```
