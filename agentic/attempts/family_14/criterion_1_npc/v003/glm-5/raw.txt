```python
import sys
import heapq

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    required = set()
    for _ in range(t):
        required.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Prim's algorithm for Steiner tree (variant)
    # Start with one required node, grow until all required nodes are connected
    
    if t <= 1:
        print(0)
        return
    
    # Pick first required node
    start = next(iter(required))
    
    in_tree = [False] * (n + 1)
    # dist[v] = min cost to connect v to current tree
    dist = [float('inf')] * (n + 1)
    
    # Initialize: start node is in tree with cost 0
    in_tree[start] = True
    total_cost = 0
    required_in_tree = 1
    
    # Initialize distances from start
    for v, w in adj[start]:
        if w < dist[v]:
            dist[v] = w
    
    # Use a heap to efficiently get minimum edge
    # Push all neighbors of start
    heap = []
    for v, w in adj[start]:
        heapq.heappush(heap, (w, v))
    
    while required_in_tree < t:
        # Find minimum cost edge to add a new node
        while heap:
            w, v = heapq.heappop(heap)
            if in_tree[v]:
                continue
            # Check if this is still the best edge to v
            if w > dist[v]:
                continue
            # Add v to tree
            in_tree[v] = True
            total_cost += w
            if v in required:
                required_in_tree += 1
            # Update distances
            for to, weight in adj[v]:
                if not in_tree[to] and weight < dist[to]:
                    dist[to] = weight
                    heapq.heappush(heap, (weight, to))
            break
    
    print(total_cost)

solve()
```
