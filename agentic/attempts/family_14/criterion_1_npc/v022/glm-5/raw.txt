```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    terminals = [int(input_data[3 + i]) - 1 for i in range(t)]
    
    adj = [[] for _ in range(n)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    # Dijkstra from each terminal
    dist = [[float('inf')] * n for _ in range(t)]
    for i in range(t):
        s = terminals[i]
        dist[i][s] = 0
        pq = [(0, s)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[i][u]:
                continue
            for v, w in adj[u]:
                if dist[i][v] > d + w:
                    dist[i][v] = d + w
                    heapq.heappush(pq, (dist[i][v], v))
                    
    # Steiner Tree DP
    # dp[mask][v] = min cost to connect terminals in 'mask' using a tree rooted at v
    dp = [[float('inf')] * n for _ in range(1 << t)]
    for i in range(t):
        dp[1 << i][terminals[i]] = 0
        
    for mask in range(1, 1 << t):
        # Subset combination
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            for v in range(n):
                dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub - 1) & mask
            
        # Dijkstra to relax dp[mask] using graph edges
        pq = []
        for v in range(n):
            if dp[mask][v] < float('inf'):
                heapq.heappush(pq, (dp[mask][v], v))
                
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v, w in adj[u]:
                if dp[mask][v] > d + w:
                    dp[mask][v] = d + w
                    heapq.heappush(pq, (dp[mask][v], v))
                    
    full_mask = (1 << t) - 1
    ans = min(dp[full_mask])
    print(ans)

if __name__ == '__main__':
    solve()
```
