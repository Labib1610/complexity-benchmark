import sys
import heapq
from collections import defaultdict
import itertools

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    
    return dist

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    generators = list(map(int, data[idx:idx+t]))
    idx += t
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    # Compute all-pairs shortest paths among generators
    gen_dist = {}
    for g in generators:
        dist = dijkstra(graph, g, n)
        gen_dist[g] = dist
    
    # Create a complete graph on generators with shortest path weights
    gen_graph = []
    for i in range(t):
        for j in range(i+1, t):
            u = generators[i]
            v = generators[j]
            w = gen_dist[u][v]
            gen_graph.append((w, u, v))
    
    # Now we need to find the MST of this complete graph
    # We'll use Kruskal's algorithm
    
    # Union-Find data structure
    parent = {g: g for g in generators}
    rank = {g: 0 for g in generators}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # Sort edges by weight
    gen_graph.sort()
    
    total_weight = 0
    edges_used = 0
    
    for w, u, v in gen_graph:
        if union(u, v):
            total_weight += w
            edges_used += 1
            if edges_used == t - 1:
                break
    
    print(total_weight)

if __name__ == "__main__":
    main()
