 We can model this as a minimum Steiner tree problem, but since the number of key commodities is small (≤ 10), we can solve it using dynamic programming over subsets of terminals.

We’ll:
1. Run Floyd–Warshall to get shortest paths between all nodes.
2. Use DP where `dp[mask][v]` = minimum cost to connect the terminals in `mask` with `v` as the root.
3. Combine subsets to build the Steiner tree.

Here’s the implementation:

```python
import sys

def main():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))
    terminals = [x - 1 for x in terminals]  # 0-index

    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            for j in range(n):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    # DP over subsets of terminals
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]

    # Initialize: single terminal
    for i in range(t):
        for v in range(n):
            dp[1 << i][v] = dist[terminals[i]][v]

    # Combine subsets
    for mask in range(1, full_mask):
        # Try splitting mask into two non-empty subsets
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if other < sub:  # avoid duplicate work
                sub = (sub - 1) & mask
                continue
            for v in range(n):
                if dp[sub][v] + dp[other][v] < dp[mask][v]:
                    dp[mask][v] = dp[sub][v] + dp[other][v]
            sub = (sub - 1) & mask

        # Relax via shortest paths
        # Run Dijkstra-like relaxation using dist matrix
        for v in range(n):
            for u in range(n):
                if dp[mask][v] + dist[v][u] < dp[mask][u]:
                    dp[mask][u] = dp[mask][v] + dist[v][u]

    ans = min(dp[full_mask - 1][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    main()
```
