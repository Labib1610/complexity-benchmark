import sys
import heapq
from collections import defaultdict
import itertools

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    key_commodities = list(map(int, sys.stdin.readline().split()))
    
    # If there's only one key commodity, no contracts needed
    if t <= 1:
        print(0)
        return
    
    # Build graph
    graph = defaultdict(list)
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths from each key commodity
    distances = {}
    for commodity in key_commodities:
        distances[commodity] = dijkstra(graph, commodity, n)
    
    # Build complete graph of key commodities with shortest path weights
    # Then find minimum spanning tree
    edges = []
    for i in range(t):
        for j in range(i + 1, t):
            u = key_commodities[i]
            v = key_commodities[j]
            weight = distances[u][v]
            if weight < float('inf'):
                edges.append((weight, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = {commodity: commodity for commodity in key_commodities}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm to find MST
    total_cost = 0
    for weight, u, v in edges:
        if union(u, v):
            total_cost += weight
    
    print(total_cost)

solve()
