import sys
import heapq

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    full = (1 << t) - 1
    INF = 10**30
    dp = [[INF] * n for _ in range(full + 1)]

    for i, terminal in enumerate(terminals):
        dp[1 << i][terminal] = 0

    for mask in range(1, full + 1):
        current = dp[mask]

        first_bit = mask & -mask
        submask = (mask - 1) & mask
        while submask:
            if submask & first_bit:
                other = mask ^ submask
                left = dp[submask]
                right = dp[other]
                for v in range(n):
                    cost = left[v] + right[v]
                    if cost < current[v]:
                        current[v] = cost
            submask = (submask - 1) & mask

        if mask == full:
            print(min(current))
            return

        heap = [(current[v], v) for v in range(n) if current[v] < INF]
        heapq.heapify(heap)

        while heap:
            dist, u = heapq.heappop(heap)
            if dist != current[u]:
                continue
            for v, weight in graph[u]:
                new_dist = dist + weight
                if new_dist < current[v]:
                    current[v] = new_dist
                    heapq.heappush(heap, (new_dist, v))

if __name__ == "__main__":
    main()
