We can solve this by computing shortest paths between all pairs of required sites, then finding the minimum Steiner tree over those required sites using dynamic programming over subsets.

```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    required = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall to get all-pairs shortest paths
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    # If only one required site, cost is 0
    if t == 1:
        print(0)
        return
    
    # DP over subsets of required sites
    # dp[mask][i] = min cost to connect required sites in mask, ending at node i
    # i is an index in the required list
    dp = [[INF] * t for _ in range(1 << t)]
    
    for i in range(t):
        dp[1 << i][i] = 0
    
    for mask in range(1 << t):
        # Split mask into two non-empty subsets
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid duplicate work
                for i in range(t):
                    if dp[sub][i] == INF:
                        continue
                    for j in range(t):
                        if dp[other][j] == INF:
                            continue
                        cost = dp[sub][i] + dp[other][j] + dist[required[i]][required[j]]
                        if cost < dp[mask][i]:
                            dp[mask][i] = cost
                        if cost < dp[mask][j]:
                            dp[mask][j] = cost
            sub = (sub - 1) & mask
        
        # Extend by moving to another required node
        for i in range(t):
            if dp[mask][i] == INF:
                continue
            for j in range(t):
                if mask & (1 << j):
                    continue
                nmask = mask | (1 << j)
                cost = dp[mask][i] + dist[required[i]][required[j]]
                if cost < dp[nmask][j]:
                    dp[nmask][j] = cost
    
    full_mask = (1 << t) - 1
    ans = min(dp[full_mask])
    print(ans)

if __name__ == "__main__":
    solve()
```
