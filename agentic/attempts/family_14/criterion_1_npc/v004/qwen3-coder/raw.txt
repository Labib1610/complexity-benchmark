import sys
import heapq
from collections import defaultdict
import itertools

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def main():
    input_lines = []
    for line in sys.stdin:
        input_lines.append(line.strip())
    
    line_idx = 0
    
    n, m, t = map(int, input_lines[line_idx].split())
    line_idx += 1
    
    required = list(map(int, input_lines[line_idx].split()))
    line_idx += 1
    
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, input_lines[line_idx].split())
        line_idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    # Compute shortest paths between all required nodes
    required_dist = {}
    
    for r in required:
        dist = dijkstra(graph, r, n)
        required_dist[r] = {r2: dist[r2] for r2 in required}
    
    # Create a complete graph of required nodes with shortest path weights
    # Then find MST of this complete graph
    # We use Prim's algorithm on the complete graph
    
    # Since t <= 10, we can afford to compute all pairwise distances
    req_set = set(required)
    
    # Build edges of complete graph on required vertices
    edges = []
    for i in range(len(required)):
        for j in range(i+1, len(required)):
            u, v = required[i], required[j]
            w = required_dist[u][v]
            edges.append((w, u, v))
    
    # Now find MST using Kruskal's algorithm with Union-Find
    class UnionFind:
        def __init__(self, elements):
            self.parent = {x: x for x in elements}
            self.rank = {x: 0 for x in elements}
        
        def find(self, x):
            if self.parent[x] != x:
                self.parent[x] = self.find(self.parent[x])
            return self.parent[x]
        
        def union(self, x, y):
            px, py = self.find(x), self.find(y)
            if px == py:
                return False
            if self.rank[px] < self.rank[py]:
                px, py = py, px
            self.parent[py] = px
            if self.rank[px] == self.rank[py]:
                self.rank[px] += 1
            return True
    
    edges.sort()
    uf = UnionFind(required)
    total_cost = 0
    edges_used = 0
    
    for w, u, v in edges:
        if uf.union(u, v):
            total_cost += w
            edges_used += 1
            if edges_used == len(required) - 1:
                break
                
    print(total_cost)

if __name__ == "__main__":
    main()
