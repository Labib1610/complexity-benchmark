import sys
import heapq
from collections import defaultdict

def find_mst(n, edges, vertices):
    if len(vertices) <= 1:
        return 0
    
    # Convert vertices to a set for quick lookup
    vertex_set = set(vertices)
    
    # Prim's algorithm
    # Start from one of the core vertices
    start = vertices[0]
    
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Priority queue: (weight, vertex)
    pq = [(0, start)]
    visited = set()
    total_cost = 0
    
    while pq and len(visited) < len(vertex_set):
        weight, u = heapq.heappop(pq)
        
        if u in visited:
            continue
            
        visited.add(u)
        total_cost += weight
        
        for v, w in adj[u]:
            if v not in visited or (v in vertex_set and v not in visited):
                heapq.heappush(pq, (w, v))
    
    return total_cost

        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

        px, py = find(x), find(y)
        if px == py:
            return False

            px, py = py, px
