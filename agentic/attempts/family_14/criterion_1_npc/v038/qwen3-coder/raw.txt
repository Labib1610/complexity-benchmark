```python
import sys
import heapq
from collections import defaultdict

def find_mst_with_pseudo_node(n, edges, core_modules):
    # Create a graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If there's only one core module, no bridges needed
    if len(core_modules) <= 1:
        return 0
    
    # Create a pseudo node connected to all core modules with weight 0
    pseudo_node = 0
    for core in core_modules:
        graph[pseudo_node].append((core, 0))
        graph[core].append((pseudo_node, 0))
    
    # Prim's algorithm to find MST
    visited = set()
    min_heap = [(0, pseudo_node)]  # (weight, node)
    total_weight = 0
    
    while min_heap and len(visited) <= n:
        weight, node = heapq.heappop(min_heap)
        
        if node in visited:
            continue
            
        visited.add(node)
        total_weight += weight
        
        for neighbor, edge_weight in graph[node]:
            if neighbor not in visited:
                heapq.heappush(min_heap, (edge_weight, neighbor))
    
    return total_weight

# Read input
line1 = list(map(int, sys.stdin.readline().split()))
n, m, t = line1[0], line1[1], line1[2]

core_modules = list(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve and print result
result = find_mst_with_pseudo_node(n, edges, core_modules)
print(result)
```
