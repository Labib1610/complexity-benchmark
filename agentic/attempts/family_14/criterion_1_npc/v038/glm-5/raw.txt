import sys
import heapq

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    core_modules = set()
    for _ in range(t):
        core_modules.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If only one core module, no bridges needed
    if t <= 1:
        print(0)
        return
    
    # Compute all-pairs shortest paths between core modules using Dijkstra
    INF = float('inf')
    core_list = list(core_modules)
    core_index = {node: idx for idx, node in enumerate(core_list)}
    num_core = len(core_list)
    
    # dist_core[i][j] = shortest path from core_list[i] to core_list[j]
    dist_core = [[INF] * num_core for _ in range(num_core)]
    
    for i, start in enumerate(core_list):
        dist = [INF] * (n + 1)
        dist[start] = 0
        pq = [(0, start)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))
        
        for j, node in enumerate(core_list):
            dist_core[i][j] = dist[node]
    
    # Prim's algorithm to find MST on the complete graph of core modules
    # where edge weight is the shortest path distance
    in_mst = [False] * num_core
    min_edge = [INF] * num_core
    min_edge[0] = 0
    total_weight = 0
    
    for _ in range(num_core):
        # Find minimum weight edge from non-MST vertices
        u = -1
        for v in range(num_core):
            if not in_mst[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
        
        in_mst[u] = True
        total_weight += min_edge[u]
        
        # Update minimum edges
        for v in range(num_core):
            if not in_mst[v] and dist_core[u][v] < min_edge[v]:
                min_edge[v] = dist_core[u][v]
    
    print(total_weight)

if __name__ == "__main__":
    solve()
