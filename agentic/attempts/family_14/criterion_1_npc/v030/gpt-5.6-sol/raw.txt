import sys
import heapq

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    if t == 1:
        print(0)
        return

    inf = 10**30
    size = 1 << t
    dp = [[inf] * n for _ in range(size)]

    for i, terminal in enumerate(terminals):
        dp[1 << i][terminal] = 0

    for mask in range(1, size):
        dist = dp[mask]

        if mask & (mask - 1):
            fixed_bit = mask & -mask
            submask = (mask - 1) & mask

            while submask:
                if submask & fixed_bit:
                    other = mask ^ submask
                    left = dp[submask]
                    right = dp[other]
                    for v in range(n):
                        candidate = left[v] + right[v]
                        if candidate < dist[v]:
                            dist[v] = candidate
                submask = (submask - 1) & mask

        heap = [(d, v) for v, d in enumerate(dist) if d < inf]
        heapq.heapify(heap)

        while heap:
            current, u = heapq.heappop(heap)
            if current != dist[u]:
                continue

            for v, weight in graph[u]:
                candidate = current + weight
                if candidate < dist[v]:
                    dist[v] = candidate
                    heapq.heappush(heap, (candidate, v))

    print(min(dp[size - 1]))

if __name__ == "__main__":
    main()
