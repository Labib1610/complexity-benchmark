We need to find the minimum total weight of a set of edges such that all **t** mandated nodes are connected.  
This is a **Steiner tree** problem with a small number of terminals (t ≤ 10). We can solve it using **Bitmask DP + Floyd–Warshall**.

Here’s the Python solution:

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))
    terminals = [x - 1 for x in terminals]  # zero-index

    # Initialize distance matrix
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
        graph[u].append((v, w))
        graph[v].append((u, w))

    # Floyd-Warshall for all-pairs shortest paths
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]

    # DP over subsets of terminals
    # dp[mask][v] = minimum cost to connect terminals in mask, ending at node v
    full_mask = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]

    # Initialize: each terminal alone cost 0 at that terminal
    for i in range(t):
        dp[1 << i][terminals[i]] = 0

    # Iterate over all masks
    for mask in range(1 << t):
        # Combine two submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if other < sub:
                for v in range(n):
                    if dp[sub][v] < INF and dp[other][v] < INF:
                        dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub - 1) & mask

        # Relax via shortest paths
        # Use a priority queue (Dijkstra-like) over current dp[mask]
        pq = []
        for v in range(n):
            if dp[mask][v] < INF:
                heapq.heappush(pq, (dp[mask][v], v))
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            for v in range(n):
                if dist[u][v] < INF and dp[mask][v] > d + dist[u][v]:
                    dp[mask][v] = d + dist[u][v]
                    heapq.heappush(pq, (dp[mask][v], v))

    ans = min(dp[full_mask][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
