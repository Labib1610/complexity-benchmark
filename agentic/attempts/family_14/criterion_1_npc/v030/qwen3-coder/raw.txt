```python
import sys
import heapq
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    mandated = list(map(int, data[idx:idx+t]))
    idx += t
    mandated = [x - 1 for x in mandated]  # Convert to 0-based indexing
    
    # If there's only one mandated ledger, no contracts needed
    if t <= 1:
        print(0)
        return
    
    # Build graph
    graph = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        w = int(data[idx+2])
        idx += 3
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((u, v, w))
    
    # For each mandated node, run Dijkstra to find shortest distance to all other nodes
    dist = []
    for start in mandated:
        d = [float('inf')] * n
        d[start] = 0
        pq = [(0, start)]
        while pq:
            cur_dist, u = heapq.heappop(pq)
            if cur_dist > d[u]:
                continue
            for v, w in graph[u]:
                if d[u] + w < d[v]:
                    d[v] = d[u] + w
                    heapq.heappush(pq, (d[v], v))
        dist.append(d)
    
    # Create a complete graph among mandated nodes with edge weights as shortest path
    # Then find MST of this complete graph
    # We'll use Prim's algorithm on the complete graph induced by mandated nodes
    
    # Compute all pairwise distances among mandated nodes
    k = len(mandated)
    # Create mapping from original index to index in mandated list
    mand_index = {mandated[i]: i for i in range(k)}
    
    # To build MST, we can use the fact that we want to connect all mandated nodes
    # with minimal cost. The optimal way is to compute the MST of the complete graph
    # where edge weights are the shortest path distances between mandated nodes.
    
    # But since k<=10, we can afford to compute all pairwise shortest paths
    # and then compute MST of the complete graph on mandated nodes.
    
    # Build adjacency matrix for mandated nodes based on shortest paths
    adj = [[float('inf')]*k for _ in range(k)]
    for i in range(k):
        for j in range(k):
            if i != j:
                adj[i][j] = dist[i][mandated[j]]
    
    # Now compute MST using Prim's algorithm on this complete graph
    visited = [False] * k
    min_edge = [float('inf')] * k
    min_edge[0] = 0
    total_cost = 0
    
    for _ in range(k):
        # Find the minimum edge to add to MST
        u = -1
        for v in range(k):
            if not visited[v] and (u == -1 or min_edge[v] < min_edge[u]):
                u = v
        
        visited[u] = True
        total_cost += min_edge[u]
        
        # Update min_edge values for adjacent nodes
        for v in range(k):
            if not visited[v] and adj[u][v] < min_edge[v]:
                min_edge[v] = adj[u][v]
    
    print(total_cost)

if __name__ == "__main__":
    main()
```
