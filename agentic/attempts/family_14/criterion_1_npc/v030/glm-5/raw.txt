import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    mandated = [int(next(it)) for _ in range(t)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((w, u, v))
    
    if t <= 1:
        print(0)
        return
    
    mandated_set = set(mandated)
    # We need to connect all mandated nodes in a single component.
    # Non-mandated nodes can be used as Steiner points.
    # Since t <= 10, we can use Dreyfus-Wagner DP for Steiner tree.
    # State: dp[mask][v] = min cost to connect subset of mandated nodes (mask) with v in the tree.
    # Final answer: min over v of dp[(1<<t)-1][v].
    
    INF = float('inf')
    # Map mandated node to index
    idx = {node: i for i, node in enumerate(mandated)}
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    full = (1 << t) - 1
    # dp[mask][v]
    dp = [[INF] * (n + 1) for _ in range(1 << t)]
    
    # Initialize: each mandated node alone
    for i, node in enumerate(mandated):
        dp[1 << i][node] = 0
    
    # Dijkstra for each mask to compute shortest paths
    import heapq
    
    for mask in range(1, 1 << t):
        # First, combine submasks
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            if sub < other:
                break
            # sub and other are complementary within mask
            # Actually we want sub and other = mask ^ sub, both non-empty
            # We iterate sub over proper non-empty subsets
            # But to avoid double counting, we can just do all and let min handle it
            for v in range(1, n + 1):
                if dp[sub][v] < INF and dp[other][v] < INF:
                    val = dp[sub][v] + dp[other][v]
                    if val < dp[mask][v]:
                        dp[mask][v] = val
            sub = (sub - 1) & mask
        
        # Now run Dijkstra from all nodes with finite dp[mask][v]
        # to propagate via edges
        dist = dp[mask]
        # Multi-source Dijkstra
        pq = []
        for v in range(1, n + 1):
            if dist[v] < INF:
                heapq.heappush(pq, (dist[v], v))
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))
    
    ans = min(dp[full][v] for v in range(1, n + 1))
    print(ans)

solve()
