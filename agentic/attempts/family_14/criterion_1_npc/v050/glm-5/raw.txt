import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    core = []
    for _ in range(t):
        core.append(int(data[idx]) - 1)
        idx += 1
    
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t <= 1:
        print(0)
        return
    
    # DP over subsets of core dialects
    full = (1 << t) - 1
    dp = [INF] * (1 << t)
    for i in range(t):
        dp[1 << i] = 0
    
    for mask in range(1, 1 << t):
        if dp[mask] == INF:
            continue
        # Try extending by adding one new core dialect
        for i in range(t):
            if mask & (1 << i):
                continue
            best = INF
            for j in range(t):
                if mask & (1 << j):
                    d = dist[core[j]][core[i]]
                    if d < best:
                        best = d
            if best < INF:
                new_mask = mask | (1 << i)
                if dp[mask] + best < dp[new_mask]:
                    dp[new_mask] = dp[mask] + best
    
    print(dp[full])

solve()
