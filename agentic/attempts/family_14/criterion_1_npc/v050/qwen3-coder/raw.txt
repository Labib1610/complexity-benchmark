import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, must_connect):
    # Create a mapping from relevant nodes to indices
    nodes = list(set(must_connect))
    for u, v, w in edges:
        nodes.extend([u, v])
    nodes = list(set(nodes))
    node_index = {node: i for i, node in enumerate(nodes)}
    
    # If there's only one or zero core dialects, no cost is needed
    if len(must_connect) <= 1:
        return 0
    
    # Try all possible ways to connect the core dialects
    # We'll use a modified Kruskal's algorithm approach
    
    # First, let's build a graph with all edges
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Since t <= 10, we can try all possible spanning trees of the core dialects
    # But a better approach is to find the minimum Steiner tree
    
    # For small t, we can use the following approach:
    # 1. Compute shortest paths between all pairs of core dialects
    # 2. Try all possible spanning trees on the core dialects using these distances
    
    # Floyd-Warshall to get all-pairs shortest paths
    INF = float('inf')
    dist = defaultdict(lambda: defaultdict(lambda: INF))
    
    # Initialize distances
    for u in nodes:
        dist[u][u] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
    
    # Floyd-Warshall
    for k in nodes:
        for i in nodes:
            for j in nodes:
                if dist[i][j] > dist[i][k] + dist[k][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # Now we have shortest paths between all nodes
    # We need to find minimum spanning tree on core dialects using these distances
    core = list(must_connect)
    t = len(core)
    
    if t <= 1:
        return 0
    
    # Generate all possible spanning trees? No, that's too much.
    # Use dynamic programming approach for Steiner tree would be overkill.
    # Let's use a simpler approach since t <= 10.
    
    # Actually, let's just run Kruskal on the complete graph of core dialects
    # where edge weights are the shortest path distances
    
    # Create edges between all pairs of core dialects with their shortest path distances
    core_edges = []
    for i in range(t):
        for j in range(i+1, t):
            u, v = core[i], core[j]
            if dist[u][v] < INF:
                core_edges.append((dist[u][v], u, v))
    
    # Sort edges by weight
    core_edges.sort()
    
    # Kruskal's algorithm to find MST of core dialects
    parent = {node: node for node in nodes}
    rank = {node: 0 for node in nodes}
    
    total_cost = 0
    edges_used = 0
    
    for w, u, v in core_edges:
        if find(parent, u) != find(parent, v):
            union(parent, rank, u, v)
            total_cost += w
            edges_used += 1
            if edges_used == t - 1:
                break
    
    return total_cost

def main():
    import sys

    idx = 0
    n = int(data[idx])

    idx += 1
    
    must_connect = []
    for _ in range(t):
        must_connect.append(int(data[idx]))
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((u, v, w))
    
    result = kruskal(n, edges, must_connect)
    print(result)
