import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    if t == 1:
        print(0)
        return

    INF = 10**30
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    # All-pairs shortest paths.
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                candidate = dik + dk[j]
                if candidate < di[j]:
                    di[j] = candidate

    total_masks = 1 << t
    dp = [None] * total_masks

    for i, terminal in enumerate(terminals):
        dp[1 << i] = dist[terminal][:]

    for mask in range(1, total_masks):
        if dp[mask] is not None:
            continue

        merged = [INF] * n
        lowest_bit = mask & -mask
        rest = mask ^ lowest_bit

        subset_rest = (rest - 1) & rest
        while True:
            left_mask = subset_rest | lowest_bit
            right_mask = rest ^ subset_rest
            left = dp[left_mask]
            right = dp[right_mask]

            for v in range(n):
                value = left[v] + right[v]
                if value < merged[v]:
                    merged[v] = value

            if subset_rest == 0:
                break
            subset_rest = (subset_rest - 1) & rest

        current = [INF] * n
        for u in range(n):
            base = merged[u]
            if base == INF:
                continue
            du = dist[u]
            for v in range(n):
                value = base + du[v]
                if value < current[v]:
                    current[v] = value

        dp[mask] = current

    print(min(dp[total_masks - 1]))

if __name__ == "__main__":
    main()
