import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) for _ in range(t)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    INF = 10**18
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    for u, v, w in edges:
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    for k in range(1, n + 1):
        dk = dist[k]
        for i in range(1, n + 1):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(1, n + 1):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    K = t
    # dp[mask][i] = min cost to connect terminals in mask, with Steiner tree rooted at node i
    dp = [[INF] * (n + 1) for _ in range(1 << K)]
    for i in range(K):
        dp[1 << i][terminals[i]] = 0
    
    for mask in range(1, 1 << K):
        # Combine submasks
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            if sub < other:
                for v in range(1, n + 1):
                    if dp[sub][v] < INF and dp[other][v] < INF:
                        val = dp[sub][v] + dp[other][v]
                        if val < dp[mask][v]:
                            dp[mask][v] = val
            sub = (sub - 1) & mask
        # Extend via shortest paths
        best = [INF] * (n + 1)
        for v in range(1, n + 1):
            if dp[mask][v] < INF:
                for u in range(1, n + 1):
                    nd = dp[mask][v] + dist[v][u]
                    if nd < best[u]:
                        best[u] = nd
        for v in range(1, n + 1):
            if best[v] < dp[mask][v]:
                dp[mask][v] = best[v]
    
    full = (1 << K) - 1
    ans = min(dp[full][v] for v in range(1, n + 1))
    print(ans)

solve()
