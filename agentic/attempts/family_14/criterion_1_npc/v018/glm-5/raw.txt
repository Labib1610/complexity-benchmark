import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    critical = []
    for _ in range(t):
        critical.append(int(data[idx]) - 1)
        idx += 1
    
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t <= 1:
        print(0)
        return
    
    full = 1 << t
    dp = [[INF] * n for _ in range(full)]
    for i in range(t):
        c = critical[i]
        dp[1 << i][c] = 0
    
    for mask in range(1, full):
        dpm = dp[mask]
        sub = (mask - 1) & mask
        while sub > 0:
            comp = mask ^ sub
            dps = dp[sub]
            dpc = dp[comp]
            for v in range(n):
                if dps[v] < INF:
                    dv = dist[v]
                    for u in range(n):
                        if dpc[u] < INF:
                            nd = dps[v] + dv[u]
                            if nd < dpm[v]:
                                dpm[v] = nd
            sub = (sub - 1) & mask
        
        best = [INF] * n
        for v in range(n):
            if dpm[v] < INF:
                dv = dist[v]
                for u in range(n):
                    nd = dpm[v] + dv[u]
                    if nd < best[u]:
                        best[u] = nd
        
        for u in range(n):
            if best[u] < dpm[u]:
                dpm[u] = best[u]
    
    ans = min(dp[full - 1])
    print(ans)

solve()
