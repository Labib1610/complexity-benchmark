```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return
    
    idx = 3
    terminals = [int(x) - 1 for x in data[idx:idx + t]]
    idx += t
    
    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
        
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        w = int(data[idx + 2])
        idx += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
            
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                if dik + dk[j] < di[j]:
                    di[j] = dik + dk[j]
                    
    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        u = terminals[i]
        mask = 1 << i
        for v in range(n):
            dp[mask][v] = dist[u][v]
            
    for mask in range(1, num_masks):
        if (mask & (mask - 1)) == 0:
            continue
        
        dp_mask = dp[mask]
        first_bit = mask & -mask
        sub = (mask - 1) & mask
        while sub >= first_bit:
            if sub & first_bit:
                comp = mask ^ sub
                dp_sub = dp[sub]
                dp_comp = dp[comp]
                for u in range(n):
                    val = dp_sub[u] + dp_comp[u]
                    if val < dp_mask[u]:
                        dp_mask[u] = val
            sub = (sub - 1) & mask
            
        temp = list(dp_mask)
        for v in range(n):
            min_val = dp_mask[v]
            for u in range(n):
                cost = temp[u] + dist[u][v]
                if cost < min_val:
                    min_val = cost
            dp_mask[v] = min_val
            
    ans = min(dp[(1 << t) - 1])
    print(ans)

if __name__ == '__main__':
    main()
```
