import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def minimum_spanning_tree(n, edges):
    parent = list(range(n))
    rank = [0] * n
    edges.sort()
    cost = 0
    for w, u, v in edges:
        if union(parent, rank, u, v):
            cost += w
    return cost

def dijkstra(graph, src, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[src] = 0
    pq = [(0, src)]
    visited = set()
    
    while pq:
        d, u = heapq.heappop(pq)
        if u in visited:
            continue
        visited.add(u)
        
        for v, w in graph[u]:
            if v in nodes and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append((w, u, v))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Create a complete graph on critical nodes with shortest path weights
    critical_edges = []
    critical_indices = [c - 1 for c in critical]
    
    for i in range(t):
        distances = dijkstra(graph, critical_indices[i], set(critical_indices))
        for j in range(i + 1, t):
            critical_edges.append((distances[critical_indices[j]], critical_indices[i], critical_indices[j]))
    
    # Find MST of the complete graph on critical nodes
    parent = list(range(n))
    rank = [0] * n
    critical_edges.sort()
    cost = 0
    
    for w, u, v in critical_edges:
        if union(parent, rank, u, v):
            cost += w
            
    print(cost)

solve()
