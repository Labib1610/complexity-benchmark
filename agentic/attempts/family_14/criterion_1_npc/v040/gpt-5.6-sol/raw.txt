import sys
import heapq

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    if t == 1:
        print(0)
        return

    inf = 10**30
    total_masks = 1 << t
    dp = [[inf] * n for _ in range(total_masks)]

    for i, terminal in enumerate(terminals):
        dp[1 << i][terminal] = 0

    for mask in range(1, total_masks):
        cur = dp[mask]

        lowest_bit = mask & -mask
        sub = (mask - 1) & mask
        while sub:
            if sub & lowest_bit:
                other = mask ^ sub
                left = dp[sub]
                right = dp[other]
                for v in range(n):
                    candidate = left[v] + right[v]
                    if candidate < cur[v]:
                        cur[v] = candidate
            sub = (sub - 1) & mask

        heap = [(distance, v) for v, distance in enumerate(cur) if distance < inf]
        heapq.heapify(heap)

        while heap:
            distance, u = heapq.heappop(heap)
            if distance != cur[u]:
                continue
            for v, weight in graph[u]:
                new_distance = distance + weight
                if new_distance < cur[v]:
                    cur[v] = new_distance
                    heapq.heappush(heap, (new_distance, v))

    print(min(dp[-1]))

if __name__ == "__main__":
    main()
