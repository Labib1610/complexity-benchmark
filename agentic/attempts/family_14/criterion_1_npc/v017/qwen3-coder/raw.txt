```python
import sys
import heapq
from collections import defaultdict

def find_mst_cost(n, edges, primary_centers):
    if len(primary_centers) <= 1:
        return 0
    
    # Create a graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Use Dijkstra's algorithm to find shortest paths from each primary center
    # to all other nodes
    distances = {}
    for center in primary_centers:
        dist = [float('inf')] * (n + 1)
        dist[center] = 0
        pq = [(0, center)]
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
                
            for v, w in graph[u]:
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    heapq.heappush(pq, (dist[v], v))
        
        distances[center] = dist
    
    # Create a complete graph of primary centers with edge weights as shortest paths
    primary_graph = []
    primary_list = list(primary_centers)
    
    for i in range(len(primary_list)):
        for j in range(i + 1, len(primary_list)):
            u = primary_list[i]
            v = primary_list[j]
            weight = distances[u][v]
            primary_graph.append((weight, u, v))
    
    # Kruskal's algorithm to find MST of primary centers graph
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Sort edges by weight
    primary_graph.sort()
    
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in primary_graph:
        if union(u, v):
            total_cost += weight
            edges_used += 1
            if edges_used == len(primary_centers) - 1:
                break
    
    return total_cost

# Read input
line = input().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

primary_centers = set(map(int, input().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, input().split())
    edges.append((u, v, w))

# Solve and output
result = find_mst_cost(n, edges, primary_centers)
print(result)
```
