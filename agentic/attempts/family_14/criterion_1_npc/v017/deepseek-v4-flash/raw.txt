We can solve this by treating it as a **Steiner tree** problem, but since \( t \le 10 \), we can use a **bitmask DP over subsets of terminals** combined with shortest paths.

Here’s the plan:

1. Run **Floyd-Warshall** to get all-pairs shortest paths.
2. Use **DP over subsets**:
   - `dp[mask][v]` = minimum cost to connect the terminals in `mask` and end at node `v`.
   - Initialize for each terminal `i`: `dp[1<<i][terminal_i] = 0`.
   - Combine subsets: for each mask, try splitting into two submasks and merging at some node.
   - Also relax using shortest paths: `dp[mask][v] = min(dp[mask][v], dp[mask][u] + dist[u][v])`.
3. The answer is `min(dp[(1<<t)-1][v])` over all `v`.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        for i in range(n):
            if dist[i][k] == INF:
                continue
            for j in range(n):
                if dist[k][j] == INF:
                    continue
                nd = dist[i][k] + dist[k][j]
                if nd < dist[i][j]:
                    dist[i][j] = nd
    
    full_mask = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]
    
    for i, term in enumerate(terminals):
        dp[1 << i][term] = 0
    
    for mask in range(1 << t):
        # Combine subsets
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    best = INF
                    for u in range(n):
                        if dp[sub][u] < INF and dp[other][u] < INF:
                            cand = dp[sub][u] + dp[other][u]
                            if cand < best:
                                best = cand
                    if best < dp[mask][v]:
                        dp[mask][v] = best
            sub = (sub - 1) & mask
        
        # Relax via shortest paths
        for _ in range(n):
            updated = False
            for u in range(n):
                if dp[mask][u] == INF:
                    continue
                for v in range(n):
                    if dist[u][v] == INF:
                        continue
                    nd = dp[mask][u] + dist[u][v]
                    if nd < dp[mask][v]:
                        dp[mask][v] = nd
                        updated = True
            if not updated:
                break
    
    ans = min(dp[full_mask])
    print(ans)

if __name__ == "__main__":
    solve()
```
