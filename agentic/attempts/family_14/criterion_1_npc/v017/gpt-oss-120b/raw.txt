```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]

    INF = 10**15
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = dist[v][u] = w

    # Floyd‑Warshall for all‑pairs shortest paths
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            ik = di[k]
            if ik == INF:
                continue
            for j in range(n):
                nd = ik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    full = (1 << t) - 1
    dp = [[INF] * n for _ in range(1 << t)]

    # base masks: single terminal
    for idx, term in enumerate(terminals):
        mask = 1 << idx
        for v in range(n):
            dp[mask][v] = dist[term][v]

    # DP over subsets
    for mask in range(1, full + 1):
        # combine two proper submasks
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            # avoid duplicate work by processing only sub < other
            if sub < other:
                dp_sub = dp[sub]
                dp_other = dp[other]
                dp_mask = dp[mask]
                for v in range(n):
                    val = dp_sub[v] + dp_other[v]
                    if val < dp_mask[v]:
                        dp_mask[v] = val
            sub = (sub - 1) & mask

        # propagate through shortest paths
        dp_mask = dp[mask]
        for k in range(n):
            dk = dist[k]
            base = dp_mask[k]
            if base == INF:
                continue
            for v in range(n):
                nd = base + dk[v]
                if nd < dp_mask[v]:
                    dp_mask[v] = nd

    answer = min(dp[full])
    print(answer)

if __name__ == "__main__":
    solve()
```
