import sys
from itertools import combinations

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, must_connect):
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    cost = 0
    sorted_edges = sorted(edges, key=lambda x: x[2])
    
    # First connect all required nodes
    for u, v, w in sorted_edges:
        if find(parent, u) != find(parent, v):
            if union(parent, rank, u, v):
                cost += w
    
    # Check if all must_connect nodes are connected
    root = find(parent, must_connect[0])
    for node in must_connect[1:]:
        if find(parent, node) != root:
            return float('inf')  # Not all connected
            
    return cost

def solve_mst_approach():
    n, m, t = map(int, sys.stdin.readline().split())

    parent = list(range(n + 1))

        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    total_cost = 0
    sorted_edges = sorted(edges, key=lambda x: x[2])
    
    # Count how many separate groups of key companies we have
    key_groups = {}
    for key in key_companies:
        key_groups[key] = key
    
    groups_count = len(key_companies)
    
    for u, v, w in sorted_edges:

            total_cost += w
            # Check if this union connects two key company groups
            root_u = find(u)
            root_v = find(v)
            
            # Update all keys that were in root_v to point to root_u
            updated_keys = []

                    updated_keys.append(key)

            distinct_groups = len(set(find(key) for key in key_companies))
