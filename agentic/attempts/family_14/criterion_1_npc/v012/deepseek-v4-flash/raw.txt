We can model this as finding the minimum spanning tree (MST) over the critical nodes, but allowing non-critical nodes as Steiner points. Since t ≤ 10, we can use a **Steiner tree** DP approach.

Here’s the Python solution:

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    critical = list(map(int, input().split()))
    critical_set = set(critical)
    
    # Build adjacency list
    graph = [[] for _ in range(n+1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # If only one critical node, no links needed
    if t == 1:
        print(0)
        return
    
    # Map critical nodes to indices 0..t-1
    idx = {node: i for i, node in enumerate(critical)}
    
    # All-pairs shortest paths from each critical node
    INF = 10**18
    dist = [[INF]*(n+1) for _ in range(t)]
    for i, src in enumerate(critical):
        d = [INF]*(n+1)
        d[src] = 0
        pq = [(0, src)]
        while pq:
            du, u = heapq.heappop(pq)
            if du > d[u]:
                continue
            for v, w in graph[u]:
                if d[v] > du + w:
                    d[v] = du + w
                    heapq.heappush(pq, (d[v], v))
        dist[i] = d
    
    # DP over subsets of critical nodes
    # dp[mask][v] = min cost to connect critical nodes in mask, ending at node v
    dp = [[INF]*(n+1) for _ in range(1<<t)]
    for i in range(t):
        dp[1<<i][critical[i]] = 0
    
    for mask in range(1<<t):
        # Combine two subsets
        sub = (mask-1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(1, n+1):
                    dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
            sub = (sub-1) & mask
        
        # Relax via shortest paths
        # Use Dijkstra-like relaxation for this mask
        pq = []
        for v in range(1, n+1):
            if dp[mask][v] < INF:
                heapq.heappush(pq, (dp[mask][v], v))
        while pq:
            du, u = heapq.heappop(pq)
            if du > dp[mask][u]:
                continue
            for v, w in graph[u]:
                if dp[mask][v] > du + w:
                    dp[mask][v] = du + w
                    heapq.heappush(pq, (dp[mask][v], v))
    
    full_mask = (1<<t) - 1
    ans = min(dp[full_mask][v] for v in range(1, n+1))
    print(ans)

if __name__ == "__main__":
    solve()
```
