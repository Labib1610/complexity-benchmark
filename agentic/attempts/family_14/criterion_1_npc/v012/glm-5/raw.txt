```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical = [int(x) for x in input_data[3:3+t]]
    
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dist[i][i] = 0
        
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
            
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            if dist[i][k] == INF:
                continue
            for j in range(1, n + 1):
                if dist[k][j] == INF:
                    continue
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
                    
    if t <= 1:
        print(0)
        return
        
    K = t
    dp = [[INF] * K for _ in range(1 << K)]
    
    for i in range(K):
        dp[1 << i][i] = 0
        
    for mask in range(1, 1 << K):
        for i in range(K):
            if not (mask & (1 << i)):
                continue
            for j in range(K):
                if mask & (1 << j):
                    continue
                nxt_mask = mask | (1 << j)
                if dp[mask][i] + dist[critical[i]][critical[j]] < dp[nxt_mask][j]:
                    dp[nxt_mask][j] = dp[mask][i] + dist[critical[i]][critical[j]]
                    
    full_mask = (1 << K) - 1
    ans = min(dp[full_mask])
    print(ans)

if __name__ == '__main__':
    solve()
```
