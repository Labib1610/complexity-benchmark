import sys
import heapq
from collections import defaultdict

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    critical = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((w, u, v))
    
    edges.sort()
    
    # We need to connect all critical modules with minimum cost
    # This is a Steiner tree problem, which is NP-hard in general
    # But since t <= 10, we can use dynamic programming with bitmasks
    
    # First, compute all-pairs shortest paths
    INF = float('inf')
    dist = [[INF] * (n + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        dist[i][i] = 0
    
    for w, u, v in edges:
        dist[u][v] = min(dist[u][v], w)
        dist[v][u] = min(dist[v][u], w)
    
    # Floyd-Warshall
    for k in range(1, n + 1):
        for i in range(1, n + 1):
            for j in range(1, n + 1):
                if dist[i][k] + dist[k][j] < dist[i][j]:
                    dist[i][j] = dist[i][k] + dist[k][j]
    
    # Create a complete graph on critical nodes with shortest path weights
    k = len(critical)
    crit_id = {v: i for i, v in enumerate(critical)}
    
    # For small t, we can try all possible spanning trees
    # But actually, we want minimum Steiner tree
    # Let's use the DP approach with bitmasks
    
    # dp[mask][i] = minimum cost to connect nodes in mask, rooted at critical[i]
    # But this is still complex. Let's think differently.
    
    # Since t <= 10, let's build a graph with only critical nodes and compute MST
    # Weight between two critical nodes is their shortest path distance
    
    # Build complete graph on critical nodes
    crit_edges = []
    for i in range(k):
        for j in range(i + 1, k):
            u, v = critical[i], critical[j]
            if dist[u][v] < INF:
                crit_edges.append((dist[u][v], i, j))
    
    crit_edges.sort()
    
    # Kruskal's algorithm for MST
    parent = list(range(k))
    rank = [0] * k
    total_cost = 0
    edges_used = 0
    
    for w, u, v in crit_edges:
        if union(parent, rank, u, v):
            total_cost += w
            edges_used += 1
            if edges_used == k - 1:
                break
    
    print(total_cost)

solve()
