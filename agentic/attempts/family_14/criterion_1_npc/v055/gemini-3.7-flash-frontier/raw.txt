```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])
    t = int(data[2])

    if t <= 1:
        print(0)
        return

    idx = 3
    terminals = [int(x) - 1 for x in data[idx:idx + t]]
    idx += t

    INF = float('inf')
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        w = int(data[idx + 2])
        idx += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                if dik + dk[j] < di[j]:
                    di[j] = dik + dk[j]

    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]

    for i in range(t):
        term = terminals[i]
        mask = 1 << i
        for u in range(n):
            dp[mask][u] = dist[term][u]

    for mask in range(1, num_masks):
        if (mask & (mask - 1)) == 0:
            continue

        dp_mask = dp[mask]
        sub = (mask - 1) & mask
        while sub > (mask ^ sub):
            dp_sub = dp[sub]
            dp_comp = dp[mask ^ sub]
            for u in range(n):
                cost = dp_sub[u] + dp_comp[u]
                if cost < dp_mask[u]:
                    dp_mask[u] = cost
            sub = (sub - 1) & mask

        new_dp = [INF] * n
        for v in range(n):
            cost_v = dp_mask[v]
            if cost_v == INF:
                continue
            dv = dist[v]
            for u in range(n):
                c = cost_v + dv[u]
                if c < new_dp[u]:
                    new_dp[u] = c
        dp[mask] = new_dp

    ans = min(dp[num_masks - 1])
    print(ans)

if __name__ == '__main__':
    main()
```
