import sys
from itertools import combinations
from collections import defaultdict
import heapq

def dijkstra(graph, start, n):
    dist = [float('inf')] * (n + 1)
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
                
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    core_commodities = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
        
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        
    # Compute shortest paths between all pairs of core commodities
    # We only need distances between core commodities
    core_distances = {}
    
    for core in core_commodities:
        dist = dijkstra(graph, core, n)
        core_distances[core] = dist
        
    # Build a complete graph of core commodities with edge weights as shortest path distances
    # Then find MST of this complete graph
    edges = []
    
    for i in range(t):
        for j in range(i + 1, t):
            u, v = core_commodities[i], core_commodities[j]
            w = core_distances[u][v]
            if w < float('inf'):
                edges.append((w, u, v))
    
    # Kruskal's algorithm for MST
    parent = list(range(n + 1))
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    edges.sort()
    total_cost = 0
    
    for w, u, v in edges:
        if union(u, v):
            total_cost += w
            
    print(total_cost)

solve()
