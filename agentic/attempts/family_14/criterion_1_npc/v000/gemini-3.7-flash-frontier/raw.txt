import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))

    terminals = [int(next(it)) - 1 for _ in range(t)]

    INF = float("inf")
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                if di[j] > dik + dk[j]:
                    di[j] = dik + dk[j]

    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]

    for i in range(t):
        term = terminals[i]
        for v in range(n):
            dp[1 << i][v] = dist[term][v]

    for mask in range(1, num_masks):
        # Submask combination
        sub = (mask - 1) & mask
        while sub > (mask ^ sub):
            other = mask ^ sub
            dp_mask = dp[mask]
            dp_sub = dp[sub]
            dp_other = dp[other]
            for v in range(n):
                cost = dp_sub[v] + dp_other[v]
                if cost < dp_mask[v]:
                    dp_mask[v] = cost
            sub = (sub - 1) & mask

        # Distance propagation
        dp_mask = dp[mask]
        new_dp = [INF] * n
        for u in range(n):
            val_u = dp_mask[u]
            if val_u == INF:
                continue
            dist_u = dist[u]
            for v in range(n):
                if val_u + dist_u[v] < new_dp[v]:
                    new_dp[v] = val_u + dist_u[v]
        dp[mask] = new_dp

    ans = min(dp[num_masks - 1])
    print(ans)


if __name__ == "__main__":
    solve()
