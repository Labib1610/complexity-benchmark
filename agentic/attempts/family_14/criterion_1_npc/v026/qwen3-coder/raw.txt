import sys
import heapq
from collections import defaultdict

def find_mst_with_pseudo_node(n, m, t, critical_systems, edges):
    # If there's only one critical system, no links needed
    if t <= 1:
        return 0
    
    # Create a mapping for critical systems to indices for easier handling
    critical_set = set(critical_systems)
    
    # We'll use a modified version of Prim's algorithm
    # Add a pseudo node that connects to all critical nodes with cost 0
    # Then find MST of this augmented graph
    
    # But there's a simpler approach:
    # We want to connect all critical nodes with minimum cost
    # This is essentially finding the minimum spanning tree of the subgraph 
    # induced by critical nodes and some helper (non-critical) nodes
    
    # Let's use a different approach: 
    # Since t is small (<= 10), we can consider all pairwise distances between critical nodes
    # and then find MST of the complete graph on critical nodes with these distances
    
    # First, let's build an adjacency list representation of the graph
    adj = [[] for _ in range(n + 1)]
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Function to compute shortest paths from a source using Dijkstra
    def dijkstra(src):
        dist = [float('inf')] * (n + 1)
        dist[src] = 0
        pq = [(0, src)]
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
                
            for v, w in adj[u]:
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    heapq.heappush(pq, (dist[v], v))
                    
        return dist
    
    # Compute all-pairs shortest paths among critical nodes
    # Since t <= 10, we can run Dijkstra from each critical node
    critical_distances = {}
    for c in critical_systems:
        dist = dijkstra(c)
        critical_distances[c] = {cc: dist[cc] for cc in critical_systems}
    
    # Now we have a complete graph on critical nodes with weights as shortest path distances
    # Find MST of this graph using Prim's algorithm
    
    if t <= 1:
        return 0
        
    # Prim's algorithm on the complete graph of critical nodes
    in_mst = set()
    total_cost = 0
    
    # Start with the first critical node
    start_node = critical_systems[0]
    in_mst.add(start_node)
    
    # Priority queue for edges to add to MST
    edge_pq = []
    
    # Add all edges from start_node to other critical nodes
    for c in critical_systems[1:]:
        heapq.heappush(edge_pq, (critical_distances[start_node][c], c))
    
    while len(in_mst) < t:
        # Find the minimum weight edge that connects a node in MST to a node not in MST
        while edge_pq:
            weight, node = heapq.heappop(edge_pq)
            if node not in in_mst:
                # Add this node to MST
                in_mst.add(node)
                total_cost += weight
                break
        
        # Add edges from newly added node to all nodes not yet in MST
        for c in critical_systems:
            if c not in in_mst:
                heapq.heappush(edge_pq, (critical_distances[node][c], c))
                
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

if t == 0:
    print(0)
else:
    critical_systems = list(map(int, sys.stdin.readline().split()))
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
        
    result = find_mst_with_pseudo_node(n, m, t, critical_systems, edges)
    print(result)
