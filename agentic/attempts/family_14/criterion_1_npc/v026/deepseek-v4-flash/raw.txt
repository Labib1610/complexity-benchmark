We can solve this by computing the **Minimum Spanning Tree (MST)** over all nodes, but only requiring that all critical nodes are connected.  
This is equivalent to finding the MST of the whole graph and then taking the maximum edge on the path between critical nodes — but simpler: we can run Kruskal’s algorithm and stop once all critical nodes are connected.

```python
import sys

def find(parent, x):
    while parent[x] != x:
        parent[x] = parent[parent[x]]
        x = parent[x]
    return x

def union(parent, rank, a, b):
    ra = find(parent, a)
    rb = find(parent, b)
    if ra == rb:
        return False
    if rank[ra] < rank[rb]:
        parent[ra] = rb
    elif rank[ra] > rank[rb]:
        parent[rb] = ra
    else:
        parent[rb] = ra
        rank[ra] += 1
    return True

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    critical = [int(next(it)) for _ in range(t)]
    critical_set = set(critical)
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((w, u, v))
    
    edges.sort()
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Track how many critical nodes are in each component
    comp_crit_count = {}
    for i in range(1, n + 1):
        comp_crit_count[i] = 1 if i in critical_set else 0
    
    total_cost = 0
    connected_crit = 0
    
    # Initially, each critical node is its own component
    for c in critical:
        connected_crit += 1
    
    if t == 1:
        print(0)
        return
    
    for w, u, v in edges:
        ru = find(parent, u)
        rv = find(parent, v)
        if ru != rv:
            # Merge
            if rank[ru] < rank[rv]:
                ru, rv = rv, ru
            parent[rv] = ru
            if rank[ru] == rank[rv]:
                rank[ru] += 1
            # Merge critical counts
            comp_crit_count[ru] += comp_crit_count[rv]
            total_cost += w
            if comp_crit_count[ru] == t:
                print(total_cost)
                return

if __name__ == "__main__":
    main()
```
