import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    critical = []
    for _ in range(t):
        critical.append(int(data[idx]) - 1)
        idx += 1
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
    
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    if t <= 1:
        print(0)
        return
    
    # Steiner tree DP over critical terminals
    K = t
    full_mask = (1 << K) - 1
    dp = [[INF] * n for _ in range(1 << K)]
    
    for i in range(K):
        node = critical[i]
        dp[1 << i][node] = 0
    
    for mask in range(1, 1 << K):
        # Merge submasks
        sub = (mask - 1) & mask
        while sub > 0:
            other = mask ^ sub
            for v in range(n):
                val = dp[sub][v] + dp[other][v]
                if val < dp[mask][v]:
                    dp[mask][v] = val
            sub = (sub - 1) & mask
        
        # Relax via shortest paths (Dijkstra-like since non-negative)
        # We can do repeated relaxation; since all weights non-negative,
        # a simple approach: use priority queue
        import heapq
        pq = []
        for v in range(n):
            if dp[mask][v] < INF:
                heapq.heappush(pq, (dp[mask][v], v))
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp[mask][u]:
                continue
            du = dist[u]
            for v in range(n):
                nd = d + du[v]
                if nd < dp[mask][v]:
                    dp[mask][v] = nd
                    heapq.heappush(pq, (nd, v))
    
    ans = min(dp[full_mask])
    print(ans)

solve()
