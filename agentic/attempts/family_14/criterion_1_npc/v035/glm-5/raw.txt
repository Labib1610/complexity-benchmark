```python
import sys
import heapq

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    core_ids = set()
    for _ in range(t):
        core_ids.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    INF = 10**18
    dist = [INF] * (n + 1)
    pq = []
    
    for u in core_ids:
        dist[u] = 0
        heapq.heappush(pq, (0, u))
    
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist[u]:
            continue
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))
    
    # Build subgraph of edges that are on shortest paths
    sub_adj = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        if dist[u] == INF:
            continue
        for v, w in adj[u]:
            if dist[v] == dist[u] + w:
                sub_adj[u].append((v, w))
    
    # Prim's MST on core nodes using Dijkstra-like expansion
    in_mst = [False] * (n + 1)
    min_edge = [INF] * (n + 1)
    pq2 = []
    
    start = next(iter(core_ids))
    min_edge[start] = 0
    heapq.heappush(pq2, (0, start))
    
    mst_cost = 0
    nodes_in_mst = 0
    
    while pq2 and nodes_in_mst < t:
        d, u = heapq.heappop(pq2)
        if in_mst[u]:
            continue
        in_mst[u] = True
        mst_cost += d
        if u in core_ids:
            nodes_in_mst += 1
        
        for v, w in sub_adj[u]:
            if not in_mst[v] and w < min_edge[v]:
                min_edge[v] = w
                heapq.heappush(pq2, (w, v))
    
    print(mst_cost)

if __name__ == "__main__":
    solve()
```
