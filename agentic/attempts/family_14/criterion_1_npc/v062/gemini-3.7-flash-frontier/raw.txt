```python
import sys
import heapq

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    idx = 3
    focal = [int(data[idx + i]) - 1 for i in range(t)]
    idx += t
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return

    INF = float('inf')
    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        dp[1 << i][focal[i]] = 0

    for mask in range(1, num_masks):
        # Combine submasks
        lb = mask & -mask
        sub = (mask - 1) & mask
        while sub > 0:
            if sub & lb:
                sub2 = mask ^ sub
                dp_mask = dp[mask]
                dp_sub = dp[sub]
                dp_sub2 = dp[sub2]
                for v in range(n):
                    val = dp_sub[v] + dp_sub2[v]
                    if val < dp_mask[v]:
                        dp_mask[v] = val
            sub = (sub - 1) & mask
        
        # Dijkstra for the current mask
        dp_mask = dp[mask]
        pq = [(dp_mask[v], v) for v in range(n) if dp_mask[v] < INF]
        heapq.heapify(pq)
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp_mask[u]:
                continue
            for v, w in adj[u]:
                if d + w < dp_mask[v]:
                    dp_mask[v] = d + w
                    heapq.heappush(pq, (d + w, v))
                    
    ans = min(dp[num_masks - 1])
    print(ans)

if __name__ == '__main__':
    solve()
```
