import sys
import heapq
from collections import defaultdict

def dijkstra(graph, start, nodes):
    dist = {node: float('inf') for node in nodes}
    dist[start] = 0
    pq = [(0, start)]
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
            
        if u in graph:
            for v, w in graph[u]:
                if dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w
                    heapq.heappush(pq, (dist[v], v))
                    
    return dist

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    mandatory = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
        
    graph = defaultdict(list)
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))
        
    # Create a complete graph over mandatory nodes with shortest path weights
    mandatory_graph = []
    nodes = set()
    for u in mandatory:
        nodes.add(u)
        for v in mandatory:
            nodes.add(v)
            
    # For each pair of mandatory nodes, compute shortest path
    distances = {}
    for start in mandatory:
        distances[start] = dijkstra(graph, start, range(1, n+1))
        
    # Build edges between mandatory nodes using shortest paths
    edges = []
    for i in range(t):
        for j in range(i+1, t):
            u, v = mandatory[i], mandatory[j]
            weight = distances[u][v]
            if weight < float('inf'):
                edges.append((weight, u, v))
                
    # Kruskal's algorithm for MST
    edges.sort()
    
    parent = {node: node for node in mandatory}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
        
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
        
    total_cost = 0
    edges_used = 0
    
    for weight, u, v in edges:
        if union(u, v):
            total_cost += weight
            edges_used += 1
            if edges_used == t - 1:
                break
                
    print(total_cost)

solve()
