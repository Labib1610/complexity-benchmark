import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    if t == 1:
        print(0)
        return

    # All-pairs shortest paths.
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd

    full = (1 << t) - 1
    dp = [None] * (full + 1)

    for i, terminal in enumerate(terminals):
        dp[1 << i] = dist[terminal][:]

    for mask in range(1, full + 1):
        if mask & (mask - 1) == 0:
            continue

        cur = [INF] * n

        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:
                left = dp[sub]
                right = dp[other]
                for v in range(n):
                    value = left[v] + right[v]
                    if value < cur[v]:
                        cur[v] = value
            sub = (sub - 1) & mask

        if mask == full:
            print(min(cur))
            return

        closed = [INF] * n
        for v in range(n):
            dv = dist[v]
            best = INF
            for u in range(n):
                value = cur[u] + dv[u]
                if value < best:
                    best = value
            closed[v] = best

        dp[mask] = closed

solve()
