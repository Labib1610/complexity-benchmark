```python
import sys
import heapq

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    idx = 3
    terminals = [int(data[idx + i]) - 1 for i in range(t)]
    idx += t
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        w = int(data[idx + 2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    if t <= 1:
        print(0)
        return

    INF = float('inf')
    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]
    
    for mask in range(1, num_masks):
        # Check if mask is a power of 2
        if (mask & (mask - 1)) == 0:
            terminal_idx = (mask.bit_length() - 1)
            dp[mask][terminals[terminal_idx]] = 0
        else:
            # Iterate through submasks containing the lowest set bit of mask
            low_bit = mask & -mask
            sub = (mask - 1) & mask
            while sub > 0:
                if sub & low_bit:
                    other = mask ^ sub
                    dp_sub = dp[sub]
                    dp_other = dp[other]
                    dp_mask = dp[mask]
                    for u in range(n):
                        cost = dp_sub[u] + dp_other[u]
                        if cost < dp_mask[u]:
                            dp_mask[u] = cost
                sub = (sub - 1) & mask

        # Dijkstra step to propagate dp[mask] values across the graph
        dp_mask = dp[mask]
        pq = [(dp_mask[u], u) for u in range(n) if dp_mask[u] < INF]
        heapq.heapify(pq)
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp_mask[u]:
                continue
            for v, w in adj[u]:
                if d + w < dp_mask[v]:
                    dp_mask[v] = d + w
                    heapq.heappush(pq, (d + w, v))

    full_mask = num_masks - 1
    ans = min(dp[full_mask])
    print(ans)

if __name__ == '__main__':
    main()
```
