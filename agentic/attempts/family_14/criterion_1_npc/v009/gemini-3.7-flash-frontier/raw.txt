import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    ptr = 3
    req = [int(data[ptr + i]) - 1 for i in range(t)]
    ptr += t
    
    INF = 10**18
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
        
    for _ in range(m):
        u = int(data[ptr]) - 1
        v = int(data[ptr + 1]) - 1
        w = int(data[ptr + 2])
        ptr += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
            
    for k in range(n):
        dist_k = dist[k]
        for i in range(n):
            dist_i = dist[i]
            d_ik = dist_i[k]
            if d_ik == INF:
                continue
            for j in range(n):
                if dist_i[j] > d_ik + dist_k[j]:
                    dist_i[j] = d_ik + dist_k[j]

    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        mask = 1 << i
        r = req[i]
        dp[mask] = list(dist[r])
        
    for mask in range(1, num_masks):
        if (mask & (mask - 1)) == 0:
            continue
        
        dp_mask = dp[mask]
        sub = (mask - 1) & mask
        while sub > (mask ^ sub):
            dp_sub = dp[sub]
            dp_comp = dp[mask ^ sub]
            for u in range(n):
                val = dp_sub[u] + dp_comp[u]
                if val < dp_mask[u]:
                    dp_mask[u] = val
            sub = (sub - 1) & mask
            
        # Relaxation via shortest paths
        new_dp = [INF] * n
        for v in range(n):
            min_val = INF
            for u in range(n):
                val = dp_mask[u] + dist[u][v]
                if val < min_val:
                    min_val = val
            new_dp[v] = min_val
        dp[mask] = new_dp

    ans = min(dp[num_masks - 1])
    print(ans)

if __name__ == '__main__':
    solve()
