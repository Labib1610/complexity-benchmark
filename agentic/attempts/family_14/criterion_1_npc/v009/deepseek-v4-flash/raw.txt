We can approach this problem with a **bitmask dynamic programming over Steiner tree** formulation.  

Since \( t \leq 10\), we can use the Dreyfus-Wagner algorithm for the Steiner tree on a small set of terminals, but we must first precompute shortest paths between all pairs of nodes.

```python
import sys
import heapq

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    required = list(map(int, input().split()))
    required = [x - 1 for x in required]  # 0-index

    INF = 10**18

    # Distance matrix
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
        adj[u].append((v, w))
        adj[v].append((u, w))

    # Run Dijkstra from each node to have all-pairs shortest paths
    for src in range(n):
        d = [INF] * n
        d[src] = 0
        pq = [(0, src)]
        while pq:
            du, u = heapq.heappop(pq)
            if du != d[u]:
                continue
            for v, w in adj[u]:
                nd = du + w
                if nd < d[v]:
                    d[v] = nd
                    heapq.heappush(pq, (nd, v))
        dist[src] = d

    # DP
    full_mask = 1 << t
    dp = [[INF] * n for _ in range(full_mask)]

    # Initialize with single terminal nodes
    for i in range(t):
        dp[1 << i][required[i]] = 0

    # Steiner DP
    for mask in range(1, full_mask):
        # Split submask part
        sub = (mask - 1) & mask
        while sub:
            other = mask ^ sub
            if sub < other:  # avoid double work
                for v in range(n):
                    val = dp[sub][v] + dp[other][v]
                    if val < dp[mask][v]:
                        dp[mask][v] = val
            sub = (sub - 1) & mask

        # Dijkstra / relaxation step for the current mask
        # Use heap to update distances
        pq = []
        for v in range(n):
            if dp[mask][v] < INF:
                heapq.heappush(pq, (dp[mask][v], v))
        while pq:
            du, u = heapq.heappop(pq)
            if du != dp[mask][u]:
                continue
            for v in range(n):
                if du + dist[u][v] < dp[mask][v]:
                    dp[mask][v] = du + dist[u][v]
                    heapq.heappush(pq, (dp[mask][v], v))

    ans = min(dp[full_mask - 1][v] for v in range(n))
    print(ans)

if __name__ == "__main__":
    solve()
```
