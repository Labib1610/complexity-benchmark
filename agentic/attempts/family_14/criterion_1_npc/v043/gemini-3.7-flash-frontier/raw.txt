```python
import sys
import heapq

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    ptr = 3
    anchors = [int(data[ptr + i]) - 1 for i in range(t)]
    ptr += t
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(data[ptr]) - 1
        v = int(data[ptr + 1]) - 1
        w = int(data[ptr + 2])
        ptr += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    INF = float('inf')
    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        dp[1 << i][anchors[i]] = 0
        
    for mask in range(1, num_masks):
        # Combine submasks
        submask = (mask - 1) & mask
        while submask > 0:
            other = mask ^ submask
            if submask <= other:
                dp_sub = dp[submask]
                dp_other = dp[other]
                dp_mask = dp[mask]
                for u in range(n):
                    val = dp_sub[u] + dp_other[u]
                    if val < dp_mask[u]:
                        dp_mask[u] = val
            submask = (submask - 1) & mask
            
        # Dijkstra relaxation for the current mask
        pq = []
        dp_mask = dp[mask]
        for u in range(n):
            if dp_mask[u] < INF:
                heapq.heappush(pq, (dp_mask[u], u))
                
        while pq:
            d, u = heapq.heappop(pq)
            if d > dp_mask[u]:
                continue
            for v, w in adj[u]:
                if d + w < dp_mask[v]:
                    dp_mask[v] = d + w
                    heapq.heappush(pq, (d + w, v))
                    
    ans = min(dp[num_masks - 1])
    print(ans)

if __name__ == '__main__':
    solve()
```
