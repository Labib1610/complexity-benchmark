import sys
from itertools import combinations

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, must_connect):
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    cost = 0
    edges.sort(key=lambda x: x[2])
    
    # First, connect all must_connect nodes
    for u, v, w in edges:
        if find(parent, u) != find(parent, v):
            if any(u in group or v in group for group in must_connect):
                union(parent, rank, u, v)
                cost += w
    
    # Then add remaining edges for connectivity
    for u, v, w in edges:
        if find(parent, u) != find(parent, v):
            union(parent, rank, u, v)
            cost += w
            
    return cost

def connected_components(parent, anchors):
    components = {}
    for anchor in anchors:
        root = find(parent, anchor)
        if root not in components:
            components[root] = []
        components[root].append(anchor)
    return list(components.values())

def solve():
    n, m, t = map(int, sys.stdin.readline().split())
    anchors = list(map(int, sys.stdin.readline().split()))
    
    if t <= 1:
        print(0)
        return
        
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
    
    # Try all possible ways to connect anchors
    min_cost = float('inf')
    
    # We use a greedy approach with Union-Find
    # Generate all possible states where anchors are connected
    # But since t<=10, we can try a different approach
    
    # For small t, we can check connectivity after each union operation
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    edges.sort(key=lambda x: x[2])
    
    cost = 0
    for u, v, w in edges:
        if find(parent, u) != find(parent, v):
            union(parent, rank, u, v)
            cost += w
            # Check if all anchors are connected
            anchor_roots = set(find(parent, anchor) for anchor in anchors)
            if len(anchor_roots) == 1:
                min_cost = min(min_cost, cost)
                break
                
    if min_cost == float('inf'):
        # Fallback - compute MST that connects all anchors
        parent = list(range(n + 1))
        rank = [0] * (n + 1)
        cost = 0
        anchors_set = set(anchors)
        
        # Connect all components containing anchors
        edges.sort(key=lambda x: x[2])
        for u, v, w in edges:
            if find(parent, u) != find(parent, v):
                union(parent, rank, u, v)
                cost += w
                # Check if all anchors are in same component
                roots = set()
                for anchor in anchors:
                    roots.add(find(parent, anchor))
                if len(roots) == 1:
                    min_cost = cost
                    break
    
    print(min_cost if min_cost != float('inf') else 0)

solve()
