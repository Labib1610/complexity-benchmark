import sys


def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])
    t = int(data[2])

    ptr = 3
    terminals = [int(x) - 1 for x in data[ptr : ptr + t]]
    ptr += t

    INF = float("inf")
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0

    for _ in range(m):
        u = int(data[ptr]) - 1
        v = int(data[ptr + 1]) - 1
        w = int(data[ptr + 2])
        ptr += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w

    for k in range(n):
        dist_k = dist[k]
        for i in range(n):
            dist_i = dist[i]
            d_ik = dist_i[k]
            if d_ik == INF:
                continue
            for j in range(n):
                if d_ik + dist_k[j] < dist_i[j]:
                    dist_i[j] = d_ik + dist_k[j]

    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]

    for i in range(t):
        dp[1 << i][terminals[i]] = 0

    for mask in range(1, num_masks):
        dp_mask = dp[mask]
        sub = (mask - 1) & mask
        while sub > (mask ^ sub):
            other = mask ^ sub
            dp_sub = dp[sub]
            dp_other = dp[other]
            for u in range(n):
                val = dp_sub[u] + dp_other[u]
                if val < dp_mask[u]:
                    dp_mask[u] = val
            sub = (sub - 1) & mask

        new_dp = list(dp_mask)
        for u in range(n):
            du = dp_mask[u]
            if du < INF:
                dist_u = dist[u]
                for v in range(n):
                    if du + dist_u[v] < new_dp[v]:
                        new_dp[v] = du + dist_u[v]
        dp[mask] = new_dp

    ans = min(dp[num_masks - 1])
    print(ans)


if __name__ == "__main__":
    solve()
