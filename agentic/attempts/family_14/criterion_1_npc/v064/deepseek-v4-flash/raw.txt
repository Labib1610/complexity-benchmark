We can solve this by treating it as a **Steiner tree** problem, but since `t ≤ 10`, we can use **bitmask DP over subsets of terminals**.

We'll:
1. Run **Floyd-Warshall** to get shortest paths between all nodes.
2. Use DP where `dp[mask][v]` = minimum cost to connect the terminals in `mask` with the tree rooted at node `v`.
3. Combine subsets using the standard Steiner tree DP.

```python
import sys
input = sys.stdin.readline

INF = 10**18

n, m, t = map(int, input().split())
terminals = list(map(int, input().split()))
terminal_set = set(terminals)

dist = [[INF] * n for _ in range(n)]
for i in range(n):
    dist[i][i] = 0

for _ in range(m):
    u, v, w = map(int, input().split())
    u -= 1
    v -= 1
    if w < dist[u][v]:
        dist[u][v] = w
        dist[v][u] = w

# Floyd-Warshall
for k in range(n):
    for i in range(n):
        for j in range(n):
            if dist[i][k] + dist[k][j] < dist[i][j]:
                dist[i][j] = dist[i][k] + dist[k][j]

# Map terminals to indices 0..t-1
term_idx = {}
for idx, node in enumerate(terminals):
    term_idx[node - 1] = idx

full_mask = (1 << t) - 1

# dp[mask][v] = min cost to connect terminals in mask, with v as root
dp = [[INF] * n for _ in range(1 << t)]

# Initialize: single terminal
for v in range(n):
    for term in range(t):
        dp[1 << term][v] = dist[v][terminals[term] - 1]

# Combine subsets
for mask in range(1, 1 << t):
    # Split mask into two non-empty subsets
    sub = (mask - 1) & mask
    while sub:
        other = mask ^ sub
        if sub < other:  # avoid double work
            for v in range(n):
                dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[other][v])
        sub = (sub - 1) & mask

    # Relax via shortest paths
    for _ in range(n):
        updated = False
        for u in range(n):
            for v in range(n):
                if dp[mask][u] + dist[u][v] < dp[mask][v]:
                    dp[mask][v] = dp[mask][u] + dist[u][v]
                    updated = True
        if not updated:
            break

# Answer: min over all nodes for full mask
ans = min(dp[full_mask][v] for v in range(n))
print(ans)
```
