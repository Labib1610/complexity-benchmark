import sys
import heapq

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    regulated = [int(next(it)) for _ in range(t)]
    regulated_set = set(regulated)
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If t <= 1, no edges needed
    if t <= 1:
        print(0)
        return
    
    # Compute all-pairs shortest path using Dijkstra from each regulated node
    dist = [[float('inf')] * (n + 1) for _ in range(t)]
    for i, start in enumerate(regulated):
        d = dist[i]
        d[start] = 0
        pq = [(0, start)]
        while pq:
            du, u = heapq.heappop(pq)
            if du > d[u]:
                continue
            for v, w in adj[u]:
                dv = du + w
                if dv < d[v]:
                    d[v] = dv
                    heapq.heappush(pq, (dv, v))
    
    # Build complete graph on regulated nodes with edge weights as shortest path distances
    # Then find MST of this graph
    # Use Prim's algorithm
    INF = float('inf')
    min_edge = [INF] * t
    used = [False] * t
    min_edge[0] = 0
    mst_cost = 0
    
    for _ in range(t):
        # Find minimum edge to non-MST vertex
        v = -1
        for j in range(t):
            if not used[j] and (v == -1 or min_edge[j] < min_edge[v]):
                v = j
        used[v] = True
        mst_cost += min_edge[v]
        # Update min_edge for all non-MST vertices
        for j in range(t):
            if not used[j]:
                # Edge weight between regulated[v] and regulated[j]
                w = dist[v][regulated[j]]
                if w < min_edge[j]:
                    min_edge[j] = w
    
    print(mst_cost)

if __name__ == "__main__":
    solve()
