import sys
import heapq

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    primary = set()
    for _ in range(t):
        primary.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    # Prim's algorithm for Minimum Spanning Tree on the Steiner graph
    # State: (cost, node, primary_mask)
    INF = float('inf')
    # Map primary anchors to bit positions
    primary_list = list(primary)
    primary_id = {node: i for i, node in enumerate(primary_list)}
    
    # dist[node][mask] = min cost to connect 'node' with set of primaries 'mask'
    # We use a dict for dist to save space, or a 2D array
    dist = [[INF] * (1 << t) for _ in range(n + 1)]
    pq = []
    
    # Initialize: each primary anchor is a trivial tree (cost 0, mask with only itself)
    for i, node in enumerate(primary_list):
        mask = 1 << i
        dist[node][mask] = 0
        heapq.heappush(pq, (0, node, mask))
    
    while pq:
        d, u, mask_u = heapq.heappop(pq)
        
        if d > dist[u][mask_u]:
            continue
        
        # Goal check: if we have all primaries connected
        if mask_u == (1 << t) - 1:
            print(d)
            return
        
        # Transition 1: Extend the tree by adding an edge (u, v)
        for v, w in adj[u]:
            if d + w < dist[v][mask_u]:
                dist[v][mask_u] = d + w
                heapq.heappush(pq, (dist[v][mask_u], v, mask_u))
        
        # Transition 2: Merge two trees at the same node
        # We iterate over all possible masks. To optimize, we can iterate over existing masks at u.
        # However, with small t, iterating all masks is acceptable.
        # We try to merge current (u, mask_u) with any existing (u, mask_v)
        # Actually, standard Dijkstra/MST property: when we pop (u, mask_u), it's optimal.
        # We can try to combine it with other masks at u.
        # But simpler: when we pop (u, mask_u), we try to update all mask_v at u.
        # Wait, the standard DP transition is: dist[u][mask_u | mask_v] <= dist[u][mask_u] + dist[u][mask_v]
        # This transition happens when we are at node u.
        
        # Let's iterate all possible masks to merge
        # To avoid O(2^t * 2^t) per node, we can do it more cleverly,
        # but with t <= 10, 1024*1024 is too big.
        # However, we only merge with masks that are already reached at u.
        # We can keep track of reached masks at each node, or just iterate.
        # Given constraints, let's iterate all masks.
        # Actually, a better way: when we pop (u, mask_u), we iterate over all masks 'm' such that dist[u][m] is not INF.
        # But we don't have that list readily available without scanning.
        # Let's just scan all masks for simplicity, given t <= 10, 1024 iterations is fine per pop.
        # BUT number of pops can be large.
        # Optimization: The merge transition is dist[u][mask_u | mask_v] = min(dist[u][mask_u | mask_v], dist[u][mask_u] + dist[u][mask_v])
        # We can process this by iterating 'mask_v' from 0 to (1<<t)-1.
        
        for mask_v in range(1 << t):
            if dist[u][mask_v] == INF:
                continue
            new_mask = mask_u | mask_v
            if dist[u][mask_u] + dist[u][mask_v] < dist[u][new_mask]:
                # Note: dist[u][mask_u] is 'd' which is minimal.
                # dist[u][mask_v] might not be minimal if we haven't popped it yet?
                # Actually, for the merge transition, we need both to be optimal.
                # But since we are popping (u, mask_u), d is optimal.
                # dist[u][mask_v] might be an estimate.
                # However, the relaxation dist[u][new] <= dist[u][mask_u] + dist[u][mask_v] is valid.
                # If dist[u][mask_v] is not optimal, the resulting dist[u][new] might not be optimal,
                # but it's a valid upper bound. When dist[u][mask_v] becomes optimal later, we will relax again.
                # Wait, the correct logic is:
                # if d + dist[u][mask_v] < dist[u][new_mask]:
                #    update and push.
                cost_merge = d + dist[u][mask_v]
                if cost_merge < dist[u][new_mask]:
                    dist[u][new_mask] = cost_merge
                    heapq.heappush(pq, (cost_merge, u, new_mask))
    
    # Should not reach here given problem guarantees
    print(-1)

if __name__ == '__main__':
    solve()
