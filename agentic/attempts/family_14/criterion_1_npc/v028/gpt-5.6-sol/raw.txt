import sys
import heapq

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminals = [next(it) - 1 for _ in range(t)]

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    if t == 1:
        print(0)
        return

    size = 1 << t
    inf = 10**30
    dp = [[inf] * n for _ in range(size)]

    for i, terminal in enumerate(terminals):
        dp[1 << i][terminal] = 0

    for mask in range(1, size):
        current = dp[mask]

        if mask & (mask - 1):
            anchor = mask & -mask
            rest = mask ^ anchor
            subset_rest = (rest - 1) & rest

            while True:
                left = anchor | subset_rest
                right = rest ^ subset_rest
                dl = dp[left]
                dr = dp[right]

                for v in range(n):
                    candidate = dl[v] + dr[v]
                    if candidate < current[v]:
                        current[v] = candidate

                if subset_rest == 0:
                    break
                subset_rest = (subset_rest - 1) & rest

        heap = [(dist, v) for v, dist in enumerate(current) if dist < inf]
        heapq.heapify(heap)

        while heap:
            dist, u = heapq.heappop(heap)
            if dist != current[u]:
                continue

            for v, weight in graph[u]:
                new_dist = dist + weight
                if new_dist < current[v]:
                    current[v] = new_dist
                    heapq.heappush(heap, (new_dist, v))

    print(min(dp[size - 1]))

if __name__ == "__main__":
    main()
