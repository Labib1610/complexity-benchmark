import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    if t <= 1:
        print(0)
        return
    
    idx = 3
    focal = [int(input_data[idx + i]) - 1 for i in range(t)]
    idx += t
    
    INF = 10**15
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
        
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        w = int(input_data[idx + 2])
        idx += 3
        if w < dist[u][v]:
            dist[u][v] = w
            dist[v][u] = w
            
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            di = dist[i]
            dik = di[k]
            if dik == INF:
                continue
            for j in range(n):
                if di[j] > dik + dk[j]:
                    di[j] = dik + dk[j]
                    
    num_masks = 1 << t
    dp = [[INF] * n for _ in range(num_masks)]
    
    for i in range(t):
        term = focal[i]
        mask = 1 << i
        for v in range(n):
            dp[mask][v] = dist[term][v]
            
    for mask in range(1, num_masks):
        if (mask & (mask - 1)) == 0:
            continue
        
        # Submask iteration
        first_bit = mask & -mask
        sub = (mask - 1) & mask
        while sub:
            if sub & first_bit:
                comp = mask ^ sub
                dp_sub = dp[sub]
                dp_comp = dp[comp]
                dp_mask = dp[mask]
                for v in range(n):
                    val = dp_sub[v] + dp_comp[v]
                    if val < dp_mask[v]:
                        dp_mask[v] = val
            sub = (sub - 1) & mask
            
        # Metric closure update
        dp_mask = dp[mask]
        new_dp_mask = list(dp_mask)
        for u in range(n):
            du = dist[u]
            val = min(dp_mask[v] + du[v] for v in range(n))
            if val < new_dp_mask[u]:
                new_dp_mask[u] = val
        dp[mask] = new_dp_mask

    full_mask = num_masks - 1
    ans = min(dp[full_mask][v] for v in range(n))
    print(ans)

if __name__ == '__main__':
    solve()
