```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    db_clusters = set()
    for _ in range(t):
        db_clusters.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # BFS to find parent and depth for each node
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    depth = [0] * (n + 1)
    
    root = next(iter(db_clusters))
    queue = deque([root])
    parent[root] = 0
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                depth[v] = depth[u] + 1
                queue.append(v)
    
    # max_edge_to_root[u] = max edge weight on path from u to root
    max_edge_to_root = [0] * (n + 1)
    
    # Process nodes in order of decreasing depth
    nodes_by_depth = [[] for _ in range(n)]
    for i in range(1, n + 1):
        nodes_by_depth[depth[i]].append(i)
    
    total_cost = 0
    
    for d in range(n - 1, 0, -1):
        for u in nodes_by_depth[d]:
            if u in db_clusters:
                # This is a database cluster, must connect to parent
                total_cost += max_edge_to_root[u]
            else:
                # Not a database cluster, can potentially skip
                # Pass up the max edge to parent
                max_to_parent = max(max_edge_to_root[u], parent_weight[u])
                p = parent[u]
                if max_to_parent > max_edge_to_root[p]:
                    max_edge_to_root[p] = max_to_parent
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
