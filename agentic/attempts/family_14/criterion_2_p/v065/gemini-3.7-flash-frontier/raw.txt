```python
import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    idx = 3
    terminals = [int(input_data[idx + i]) for i in range(t)]
    idx += t
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    
    # BFS to establish parent-child relationships
    order.append(1)
    visited = [False] * (n + 1)
    visited[1] = True
    
    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                order.append(v)
                
    count = [0] * (n + 1)
    for term in terminals:
        count[term] = 1
        
    total_cost = 0
    for v in reversed(order):
        if v == 1:
            continue
        p = parent[v]
        count[p] += count[v]
        if 0 < count[v] < t:
            total_cost += edge_weight[v]
            
    print(total_cost)

if __name__ == '__main__':
    main()
```
