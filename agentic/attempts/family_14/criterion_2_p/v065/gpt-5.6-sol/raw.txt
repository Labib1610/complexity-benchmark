```python
import sys
from collections import deque

def main():
    data = map(int, sys.stdin.buffer.read().split())
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    terminal = bytearray(n)
    for _ in range(t):
        terminal[next(it) - 1] = 1

    adjacency = [[] for _ in range(n)]
    total_cost = 0

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        adjacency[u].append((v, w))
        adjacency[v].append((u, w))
        total_cost += w

    degree = [len(neighbors) for neighbors in adjacency]
    queue = deque(
        node for node in range(n)
        if degree[node] == 1 and not terminal[node]
    )

    while queue:
        node = queue.popleft()
        if degree[node] != 1 or terminal[node]:
            continue

        degree[node] = 0
        for neighbor, cost in adjacency[node]:
            if degree[neighbor] > 0:
                total_cost -= cost
                degree[neighbor] -= 1
                if degree[neighbor] == 1 and not terminal[neighbor]:
                    queue.append(neighbor)
                break

    print(total_cost)

if __name__ == "__main__":
    main()
```
