import sys
input = sys.stdin.read

class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True

def main():
    data = input().split()
    idx = 0
    
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    critical = set()
    for _ in range(t):
        critical.add(int(data[idx]) - 1)
        idx += 1
    
    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
    
    # Sort edges by weight
    edges.sort()
    
    # Create a reverse mapping for critical nodes
    critical_list = list(critical)
    
    # Use Union-Find to track connectivity among critical nodes
    uf = UnionFind(n)
    total_cost = 0
    connected_critical = set()
    
    # Initially, each critical node is its own component
    for node in critical:
        connected_critical.add(uf.find(node))
    
    for w, u, v in edges:
        # Check if u and v are already connected
        if uf.find(u) != uf.find(v):
            # Union the components
            uf.union(u, v)
            
            # Update the set of connected critical components
            new_connected = set()
            for node in critical:
                new_connected.add(uf.find(node))
            
            # If we've reduced the number of components, add the edge cost
            if len(new_connected) < len(connected_critical):
                total_cost += w
                
            connected_critical = new_connected
            
            # If all critical nodes are connected, we're done
            if len(connected_critical) == 1:
                break
    
    print(total_cost)

if __name__ == "__main__":
    main()
