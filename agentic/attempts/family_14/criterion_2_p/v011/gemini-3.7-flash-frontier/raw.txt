```python
import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])

    ptr = 3
    vital_nodes = [int(input_data[ptr + i]) for i in range(t)]
    ptr += t

    if t <= 1:
        print(0)
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        w = int(input_data[ptr + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        ptr += 3

    is_vital = [0] * (n + 1)
    for v in vital_nodes:
        is_vital[v] = 1

    root = vital_nodes[0]
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    
    queue = [root]
    parent[root] = -1

    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_edge_weight[v] = w
                queue.append(v)

    vital_count = list(is_vital)
    total_power = 0

    for u in reversed(order):
        if u == root:
            continue
        if vital_count[u] > 0:
            total_power += parent_edge_weight[u]
            vital_count[parent[u]] += vital_count[u]

    print(total_power)

if __name__ == '__main__':
    main()
```
