We can solve this by pruning the tree: remove all branches that don’t contain any vital module, then find the minimum spanning tree (which is just the sum of all remaining edges) on the pruned tree.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    vital = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # Find which nodes are needed (contain vital modules in their subtree)
    needed = [False] * (n + 1)
    
    def dfs(u, parent):
        if u in vital:
            needed[u] = True
        for v, w in adj[u]:
            if v != parent:
                if dfs(v, u):
                    needed[u] = True
        return needed[u]
    
    dfs(1, -1)
    
    total = 0
    for u, v, w in edges:
        if needed[u] and needed[v]:
            total += w
    
    print(total)

if __name__ == "__main__":
    solve()
```
