```python
import sys

def main():
    input = sys.stdin.buffer.readline

    n, m, t = map(int, input().split())

    vital = []
    while len(vital) < t:
        vital.extend(map(int, input().split()))

    is_vital = bytearray(n + 1)
    for node in vital:
        is_vital[node] = 1

    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1

    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in graph[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            parent_weight[v] = w
            stack.append(v)

    subtree_vitals = list(is_vital)
    answer = 0

    for u in reversed(order):
        if u == 1:
            continue
        count = subtree_vitals[u]
        if 0 < count < t:
            answer += parent_weight[u]
        subtree_vitals[parent[u]] += count

    print(answer)

if __name__ == "__main__":
    main()
```
