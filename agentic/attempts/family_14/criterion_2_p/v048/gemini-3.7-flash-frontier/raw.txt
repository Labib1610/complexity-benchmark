import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return
    
    idx = 3
    priority_nodes = set()
    for _ in range(t):
        priority_nodes.add(int(data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    
    # BFS to establish tree structure rooted at 1
    stack = [1]
    parent[1] = -1
    
    # To get a valid topological / post-order traversal, BFS/DFS order reversed works.
    bfs_order = []
    head = 0
    bfs_order.append(1)
    
    while head < len(bfs_order):
        u = bfs_order[head]
        head += 1
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                edge_weight[v] = w
                bfs_order.append(v)
                
    cnt = [0] * (n + 1)
    for u in priority_nodes:
        cnt[u] = 1
        
    total_latency = 0
    for u in reversed(bfs_order):
        if parent[u] != -1:
            if 0 < cnt[u] < t:
                total_latency += edge_weight[u]
            cnt[parent[u]] += cnt[u]
            
    print(total_latency)

if __name__ == '__main__':
    main()
