```python
import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    t = int(next(iterator))
    
    if t <= 1:
        print(0)
        return
    
    is_priority = [0] * (n + 1)
    for _ in range(t):
        p = int(next(iterator))
        is_priority[p] = 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # BFS to establish parent and order
    order = []
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    queue = [1]
    visited[1] = True
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                queue.append(v)
                
    count = list(is_priority)
    total_cost = 0
    
    for u in reversed(order):
        if 0 < count[u] < t:
            total_cost += edge_weight[u]
        p = parent[u]
        if p != 0:
            count[p] += count[u]
            
    print(total_cost)

if __name__ == '__main__':
    main()
```
