import sys
from collections import defaultdict, deque

def find_min_cost_to_connect_critical_modules():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    if t <= 1:
        return 0
    
    critical = set()
    for _ in range(t):
        critical.add(int(data[idx]))
        idx += 1
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Since the topology is a tree (m = n-1 and all nodes connected when all edges on),
    # we can find the minimum spanning tree that connects all critical nodes.
    # This is essentially the Steiner tree problem on a tree, which can be solved optimally:
    # 1. Build the subtree that contains all critical nodes and the paths between them.
    # 2. The optimal solution is the sum of weights of edges in this subtree.
    
    # Strategy:
    # 1. Find the subtree that spans all critical nodes.
    # 2. Since it's a tree, we can do this by:
    #    - Starting from one critical node
    #    - Doing BFS/DFS to find paths to all other critical nodes
    #    - The union of all these paths gives us the minimal subtree connecting all critical nodes.
    
    # However, there's a simpler way for a tree:
    # The minimal subtree connecting a set of nodes in a tree can be found by:
    # 1. Taking all paths between pairs of critical nodes
    # 2. The union of these paths is the minimal subtree
    # But that's expensive.
    
    # Better approach:
    # In a tree, the minimal subtree connecting a set of nodes S is the same as:
    # 1. Find the subtree induced by S: take all nodes that lie on some path between two nodes in S
    # 2. This is equivalent to finding the "closure" of S under the tree topology
    # 3. A simpler way: 
    #    - Root the tree at any node
    #    - Find the lowest common ancestor (LCA) concepts, but maybe overkill
    
    # Even simpler:
    # Since it's a tree, we can:
    # 1. Pick any critical node as root
    # 2. Do BFS from that node, and mark all nodes visited
    # 3. Then do BFS from all other critical nodes, but only traverse edges that help connect to other critical nodes
    # Actually, let's think differently.
    
    # Key insight: In a tree, to connect k critical nodes, we need exactly (k-1) edges in the optimal solution.
    # But we don't know which edges. However, since the whole structure is a tree, we can:
    # 1. Find the minimal subtree that connects all critical nodes.
    # 2. The cost is the sum of edge weights in that subtree.
    
    # Efficient way:
    # 1. Root the tree at a critical node (if any exists)
    # 2. For each node, compute if its subtree contains any critical node
    # 3. An edge should be included in the solution if it connects two components that both contain critical nodes
    
    # Let's implement this.
    
    # First, build the tree structure
    adj = defaultdict(list)
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If there's only one critical node, cost is 0
    if len(critical) <= 1:
        return 0
    
    # Function to find edges that must be included
    # We'll do a DFS and for each edge, check if it separates critical nodes
    
    # Let's root the tree at the first critical node
    root = next(iter(critical))
    
    # For each node, we'll store:
    # - whether its subtree contains any critical node
    # - parent and edge weight to parent
    contains_critical = [False] * (n + 1)
    parent = [-1] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    
    # First DFS to set up parent relationships and find subtree critical containment
    def dfs_setup(node, par, edge_w):
        parent[node] = par
        parent_edge_weight[node] = edge_w
        contains_critical[node] = node in critical
        for neighbor, weight in adj[node]:
            if neighbor == par:
                continue
            dfs_setup(neighbor, node, weight)
            contains_critical[node] |= contains_critical[neighbor]
    
    dfs_setup(root, -1, 0)
    
    # Now, collect all edges that must be in the solution
    # An edge {u, v} where u is parent of v, is needed if:
    # - v's subtree contains at least one critical node
    # - The rest of the tree (including u's other subtrees and u itself) contains at least one critical node
    
    total_cost = 0
    for node in range(1, n + 1):
        if parent[node] != -1 and contains_critical[node]:
            # Check if the parent side also has critical nodes
            # This is true if the total number of critical nodes > number in node's subtree
            # But we need to be more careful
            # The edge from parent[node] to node is needed if:
            # - node's subtree has at least one critical
            # - the rest of the tree has at least one critical
            
            # Since we're in a tree, if node's subtree has some critical nodes,
            # and there are critical nodes in total, then the edge is needed
            # iff not all critical nodes are in node's subtree
            # i.e., if count_in_subtree < total_critical_count
            
            # Let's count total critical nodes
            total_critical = len(critical)
            count_in_subtree = 0
            
            # To count critical nodes in subtree, we can do another DFS
            # But we already have contains_critical which is a boolean
            # We need the actual count. Let's modify our approach.
            
    # Let's re-think. We need the exact count of critical nodes in each subtree.
    
    critical_count = [0] * (n + 1)
    
    def dfs_count(node, par):
        count = 1 if node in critical else 0
        for neighbor, _ in adj[node]:
            if neighbor == par:
                continue
            count += dfs_count(neighbor, node)
        critical_count[node] = count
        return count
    
    total_critical_nodes = dfs_count(root, -1)
    
    # Now, for each edge, check if it's needed
    # An edge between parent[u] and u is needed if:
    # - critical_count[u] > 0 (subtree has critical nodes)
    # - critical_count[u] < total_critical_nodes (not all critical nodes are in subtree)
    
    total_cost = 0
    for node in range(1, n + 1):
        if parent[node] != -1:  # Not the root
            if 0 < critical_count[node] < total_critical_nodes:
                total_cost += parent_edge_weight[node]
                
    return total_cost

print(find_min_cost_to_connect_critical_modules())
