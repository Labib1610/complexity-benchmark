```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    priority = set(map(int, sys.stdin.readline().split()))
    
    # Build the tree
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # If there's only one priority node, cost is 0
    if t <= 1:
        print(0)
        return
    
    # Find the minimal subtree that connects all priority nodes
    # We can do this by finding the Steiner tree approximation on a tree
    # Which is equivalent to: take all edges in the subtree spanned by the priority nodes
    
    # First, we find the minimal subtree that contains all priority nodes
    # by marking all nodes and edges on paths between priority nodes
    
    # To do this efficiently, we can find the "closure" of the priority nodes:
    # the minimal subtree connecting them
    
    # A good approach: find the union of paths between all pairs would be too slow
    # Instead, we can do a single DFS to find the minimal subtree
    
    # Mark all priority nodes
    is_priority = [False] * (n + 1)
    for x in priority:
        is_priority[x] = True
    
    # We'll do a DFS to find the minimal subtree that includes all priority nodes
    # The idea is to traverse the tree and keep track of whether a subtree contains
    # any priority node. An edge should be included if both its subtrees contain
    # at least one priority node.
    
    # Let's root the tree at node 1
    parent = [-1] * (n + 1)
    parent_edge = [-1] * (n + 1)
    children = defaultdict(list)
    
    # First, build the tree rooted at 1
    visited = [False] * (n + 1)
    queue = deque([1])
    visited[1] = True
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge[v] = w
                children[u].append((v, w))
                queue.append(v)
    
    # Now, do a post-order traversal to count priority nodes in each subtree
    # and decide which edges to include
    
    # count[v] = number of priority nodes in subtree rooted at v
    count = [0] * (n + 1)
    # Mark which nodes are visited during post-order
    visited_post = [False] * (n + 1)
    # Stack for post-order traversal
    stack = [1]
    order = []
    
    while stack:
        u = stack.pop()
        if not visited_post[u]:
            visited_post[u] = True
            order.append(u)
            for v, w in children[u]:
                if not visited_post[v]:
                    stack.append(v)
    
    # Process in reverse order (post-order)
    total_cost = 0
    for u in reversed(order):
        if is_priority[u]:
            count[u] += 1
        for v, w in children[u]:
            count[u] += count[v]
            # If both u's subtree (after including v) and v's subtree have priority nodes,
            # then this edge must be included
            if count[v] > 0 and (count[u] - count[v]) > 0:
                total_cost += w
    
    print(total_cost)

if __name__ == "__main__":
    main()
```
