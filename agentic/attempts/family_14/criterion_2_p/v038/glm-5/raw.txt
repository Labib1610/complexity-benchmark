```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return
    
    priority_set = set()
    for _ in range(t):
        try:
            priority_set.add(int(next(iterator)))
        except StopIteration:
            return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            w = int(next(iterator))
            adj[u].append((v, w))
            adj[v].append((u, w))
        except StopIteration:
            break
    
    # Prim's algorithm for Minimum Spanning Tree on the Steiner Tree problem
    # We need to connect all priority facilities. The optimal solution is a tree
    # that is a subgraph of the original tree connecting all priority nodes.
    # This is equivalent to finding the MST of the subgraph induced by priority nodes
    # and Steiner nodes (intermediate nodes).
    # For a tree, the Steiner tree can be found by running Prim's starting from
    # any priority node, and adding edges until all priority nodes are connected.
    
    in_mst = [False] * (n + 1)
    min_edge = [float('inf')] * (n + 1)
    
    # Start with an arbitrary priority node
    start_node = next(iter(priority_set))
    min_edge[start_node] = 0
    
    # Priority queue: (weight, node)
    pq = [(0, start_node)]
    
    total_cost = 0
    connected_priority = 0
    
    while pq and connected_priority < t:
        w, u = heapq.heappop(pq)
        
        if in_mst[u]:
            continue
        
        in_mst[u] = True
        total_cost += w
        
        if u in priority_set:
            connected_priority += 1
        
        for v, weight in adj[u]:
            if not in_mst[v] and weight < min_edge[v]:
                min_edge[v] = weight
                heapq.heappush(pq, (weight, v))
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
