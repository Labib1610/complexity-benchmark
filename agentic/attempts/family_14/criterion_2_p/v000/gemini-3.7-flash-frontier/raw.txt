```python
import sys

# Increase recursion depth just in case, though iterative is preferred
sys.setrecursionlimit(300000)

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return
    
    idx = 3
    required = set()
    for _ in range(t):
        required.add(int(data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # Iterative BFS/DFS to find tree structure rooted at 1
    root = 1
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    # BFS using queue
    from collections import deque
    queue = deque([root])
    while queue:
        u = queue.popleft()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                queue.append(v)
                
    # Count required vertices in subtrees bottom-up
    req_count = [0] * (n + 1)
    for u in required:
        req_count[u] = 1
        
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if 0 < req_count[u] < t:
            total_cost += edge_weight[u]
        p = parent[u]
        req_count[p] += req_count[u]
        
    print(total_cost)

if __name__ == '__main__':
    main()
```
