```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    research_centers = list(map(int, data[idx:idx+t]))
    idx += t
    
    # Build the tree
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-based indexing
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        w = int(data[idx])
        idx += 1
        graph[u].append((v, w))
        graph[v].append((u, w))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    research_set = set(x - 1 for x in research_centers)  # Convert to 0-based
    
    # We need to find the minimal subtree that connects all research centers.
    # This is the Steiner tree problem on a tree, which can be solved by:
    # 1. Root the tree at any node
    # 2. For each node, compute whether the subtree contains any research center
    # 3. An edge is needed in the solution if it lies on the path between any two research centers
    
    # Let's use a different approach:
    # Since the graph is a tree, we can:
    # 1. Root the tree at node 0
    # 2. Compute for each node, the nearest research center in its subtree
    # 3. But actually, since it's a tree, the minimal cost to connect all research centers
    #    is the sum of weights of edges in the minimal subtree containing all research centers.
    
    # Efficient approach:
    # The minimal subtree connecting all terminals (research centers) in a tree
    # is called the Steiner tree. For a tree, we can compute it efficiently:
    # 1. Root the tree at an arbitrary node
    # 2. Do a post-order traversal
    # 3. For each node, count how many research centers are in its subtree
    # 4. If a node has k research centers in its subtree, then the edge to its parent
    #    contributes to the final cost only if 0 < k < t
    
    # Let's root the tree at node 0
    parent = [-1] * n
    parent_weight = [0] * n  # weight of edge to parent
    children = [[] for _ in range(n)]
    
    # BFS to build rooted tree
    visited = [False] * n
    queue = deque()
    queue.append(0)
    visited[0] = True
    
    while queue:
        u = queue.popleft()
        for v, w in graph[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_weight[v] = w
                children[u].append(v)
                queue.append(v)
    
    # Now do post-order traversal to count research centers in subtrees
    count = [0] * n  # count of research centers in subtree rooted at node i
    
    def dfs(u):
        if u in research_set:
            count[u] = 1
        else:
            count[u] = 0
        for v in children[u]:
            dfs(v)
            count[u] += count[v]
    
    dfs(0)
    
    # Now sum up the weights of edges where 0 < count < t
    total_cost = 0
    for u in range(1, n):  # Skip root, it has no parent
        c = count[u]
        if 0 < c < t:
            total_cost += parent_weight[u]
    
    print(total_cost)

if __name__ == "__main__":
    main()
```
