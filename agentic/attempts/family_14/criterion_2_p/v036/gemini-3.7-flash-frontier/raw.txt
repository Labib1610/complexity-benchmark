```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return

    idx = 3
    is_target = [False] * (n + 1)
    for _ in range(t):
        is_target[int(data[idx])] = True
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # BFS to establish parent relationships and topological order
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    
    root = 1
    queue = [root]
    parent[root] = -1
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                edge_weight[v] = w
                queue.append(v)
                
    count = [1 if is_target[i] else 0 for i in range(n + 1)]
    total_cost = 0
    
    for u in reversed(order):
        if parent[u] != -1:
            if 0 < count[u] < t:
                total_cost += edge_weight[u]
            count[parent[u]] += count[u]
            
    print(total_cost)

if __name__ == '__main__':
    main()
```
