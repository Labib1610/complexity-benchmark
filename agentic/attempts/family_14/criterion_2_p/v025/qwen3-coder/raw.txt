import sys
from collections import defaultdict, deque

def find_parent(parent, x):
    if parent[x] != x:
        parent[x] = find_parent(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find_parent(parent, x), find_parent(parent, y)
    if px == py:
        return
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1

def kruskal(n, edges, terminals):
    if len(terminals) <= 1:
        return 0
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    # Initialize DSU
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Connect all terminals initially as they're in the same component
    root = terminals[0]
    terminal_set = set(terminals)
    
    cost = 0
    components = n
    
    for u, v, w in edges:
        pu = find_parent(parent, u)
        pv = find_parent(parent, v)
        
        if pu != pv:
            # If either u or v is a terminal, or they connect to different components
            union(parent, rank, u, v)
            cost += w
            components -= 1
            
            # If we've connected all nodes, we're done
            if components == 1:
                break
    
    # Now we need to find the minimum spanning tree that connects all terminals
    # This is essentially the Steiner tree problem, but we can use a simpler approach
    # since we're dealing with a tree structure
    
    # Let's build the actual tree and then find the minimal subtree connecting terminals
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Find edges in the minimal subtree connecting all terminals
    return steiner_tree(adj, n, terminals)

def steiner_tree(adj, n, terminals):
    if len(terminals) <= 1:
        return 0
    
    # Build a complete graph where edge weights are shortest paths between terminals
    # Then find MST of this complete graph
    
    # Floyd-Warshall would be too slow, so we do BFS for each terminal
    dist = defaultdict(lambda: defaultdict(lambda: float('inf')))
    
    terminal_set = set(terminals)
    
    # For each terminal, compute distances to all other nodes
    for t in terminals:
        queue = deque([(t, 0)])
        visited = set()
        dist[t][t] = 0
        
        while queue:
            node, d = queue.popleft()
            if node in visited:
                continue
            visited.add(node)
            dist[t][node] = d
            
            for neighbor, weight in adj[node]:
                if neighbor not in visited:
                    queue.append((neighbor, d + weight))
    
    # Create edges between terminals based on shortest paths
    terminal_edges = []
    for i in range(len(terminals)):
        for j in range(i + 1, len(terminals)):
            u, v = terminals[i], terminals[j]
            # Distance is shortest path in original graph
            terminal_edges.append((u, v, dist[u][v]))
    
    # Now run Kruskal's algorithm on this complete graph of terminals
    terminal_edges.sort(key=lambda x: x[2])
    
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    total_cost = 0
    edges_used = 0
    
    for u, v, w in terminal_edges:
        pu = find_parent(parent, u)
        pv = find_parent(parent, v)
        
        if pu != pv:
            union(parent, rank, u, v)
            total_cost += w
            edges_used += 1
            
            if edges_used == len(terminals) - 1:
                break
    
    return total_cost

def steiner_tree_on_tree(n, edges, terminals):
    if len(terminals) <= 1:
        return 0
    
    # Since the input graph is a tree, we can find the minimal subtree
    # that connects all terminals by:
    # 1. Finding all paths between terminal pairs
    # 2. Taking the union of all these paths
    # 3. Summing the weights of edges in this union
    
    # But there's a more efficient way:
    # The minimal Steiner tree on a tree is the subgraph induced by 
    # the terminals and all nodes on paths between terminal pairs
    
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # BFS to find distance and parent for each node from a root
    def bfs(start):
        dist = [-1] * (n + 1)
        parent = [-1] * (n + 1)
        dist[start] = 0
        queue = deque([start])
        
        while queue:
            u = queue.popleft()
            for v, w in adj[u]:
                if dist[v] == -1:
                    dist[v] = dist[u] + w
                    parent[v] = u
                    queue.append(v)
        
        return dist, parent
    
    # Find path between two nodes
    def get_path(u, v):
        # BFS from u to find parent relationships
        visited = [False] * (n + 1)
        parent = [-1] * (n + 1)
        queue = deque([u])
        visited[u] = True
        
        while queue:
            node = queue.popleft()
            if node == v:
                break
            for neighbor, _ in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    parent[neighbor] = node
                    queue.append(neighbor)
        
        # Reconstruct path
        path = []
        current = v
        while current != -1:
            path.append(current)
            current = parent[current]
        
        return path[::-1]  # Reverse to get path from u to v
    
    # Find all edges in paths between terminal pairs
    used_edges = set()
    
    # For a tree, we can just take any terminal as root and find paths
    # from root to all other terminals. The union of these paths gives
    # us the minimal Steiner tree.
    
    root = terminals[0]
    terminal_set = set(terminals)
    
    # BFS from root to find paths to all nodes
    visited = [False] * (n + 1)
    parent_edge = [-1] * (n + 1)  # Store the edge that leads to each node
    queue = deque([root])
    visited[root] = True
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent_edge[v] = (u, v, w)  # Store the edge
                queue.append(v)
    
    # Now for each terminal, trace back to root and mark edges as used
    total_cost = 0
    edge_in_tree = set()
    
    for terminal in terminals[1:]:  # Skip the root
        # Trace path from terminal back to root
        current = terminal
        while current != root:
            edge = parent_edge[current]
            # Normalize edge representation (smaller id first)
            u, v, w = edge
            if u > v:
                u, v = v, u
            edge_key = (u, v)
            
            if edge_key not in edge_in_tree:
                edge_in_tree.add(edge_key)
                total_cost += w
                
            # Move to parent
            if parent_edge[current][0] == current:
                current = parent_edge[current][1]
            else:
                current = parent_edge[current][0]
    
    return total_cost
