import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    is_core = [False] * (n + 1)
    for _ in range(t):
        u = int(next(it))
        is_core[u] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    setrecursionlimit(n + 20)
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    depth = [0] * (n + 1)
    order = [1]
    parent[1] = 1
    
    for u in order:
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                depth[v] = depth[u] + 1
                order.append(v)
    
    core_count = [0] * (n + 1)
    for u in range(1, n + 1):
        if is_core[u]:
            core_count[u] = 1
    
    for u in reversed(order):
        if u != 1:
            core_count[parent[u]] += core_count[u]
    
    total_energy = 0
    for u in range(2, n + 1):
        if 0 < core_count[u] < t:
            total_energy += parent_weight[u]
    
    sys.stdout.write(str(total_energy))

if __name__ == '__main__':
    solve()
