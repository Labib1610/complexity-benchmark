```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    ptr = 3
    priority = [int(x) for x in data[ptr:ptr+t]]
    ptr += t
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[ptr])
        v = int(data[ptr+1])
        w = int(data[ptr+2])
        ptr += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    cnt = [0] * (n + 1)
    for p in priority:
        cnt[p] = 1
        
    root = priority[0]
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    
    order = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_weight[v] = w
                order.append(v)
                
    ans = 0
    for u in reversed(order):
        if u == root:
            continue
        if cnt[u] > 0:
            ans += parent_weight[u]
            cnt[parent[u]] += cnt[u]
            
    print(ans)

if __name__ == '__main__':
    main()
```
