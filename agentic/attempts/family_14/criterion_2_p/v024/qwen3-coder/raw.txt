```python
import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, t = map(int, sys.stdin.readline().split())
    hubs = set(map(int, sys.stdin.readline().split()))
    
    # Build adjacency list
    adj = [[] for _ in range(n+1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((w, u, v))
    
    # If there's only one hub or no hubs, cost is 0
    if t <= 1:
        print(0)
        return
    
    # Find the spanning tree that connects all hubs with minimal cost
    # This is essentially finding the Minimum Spanning Tree of the subgraph induced by hubs
    # But since we have a tree, we need to find the minimal subtree that connects all hubs
    
    # Approach: 
    # 1. Mark all nodes that are hubs
    # 2. Run a modified Prim's or Kruskal's to connect all hubs
    # Actually, since it's a tree, we can do this:
    # - We want to find the minimal connected subtree that includes all hubs
    # - This is the Steiner tree problem on a tree, which can be solved efficiently
    
    # Efficient approach for trees:
    # 1. Root the tree at any node
    # 2. For each node, compute whether its subtree contains any hub
    # 3. An edge is needed in the solution if it's on the path between two hubs
    # 4. In a tree, this is equivalent to: include an edge if both its subtrees contain at least one hub
    
    # Let's root the tree at node 1
    parent = [-1] * (n+1)
    children = [[] for _ in range(n+1)]
    visited = [False] * (n+1)
    
    # BFS to build rooted tree
    queue = [1]
    visited[1] = True
    while queue:
        u = queue.pop(0)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append((v, w))
                queue.append(v)
    
    # Count hubs in each subtree
    hub_count = [0] * (n+1)
    hub_in_subtree = [False] * (n+1)  # Whether subtree contains any hub
    
    # Post-order traversal to count hubs
    def dfs(u):
        if u in hubs:
            hub_count[u] = 1
            hub_in_subtree[u] = True
        else:
            hub_count[u] = 0
            hub_in_subtree[u] = False
            
        for v, w in children[u]:
            dfs(v)
            hub_count[u] += hub_count[v]
            if hub_in_subtree[v]:
                hub_in_subtree[u] = True
                
        if u in hubs:
            hub_in_subtree[u] = True
    
    dfs(1)
    
    # Now calculate the minimum cost
    total_cost = 0
    # For each edge in the rooted tree, check if it should be included
    # An edge from parent[v] to v is included if both v's subtree and the rest contain hubs
    def calculate_cost(u):
        nonlocal total_cost
        for v, w in children[u]:
            # Edge from u to v with weight w
            # Include this edge if:
            # 1. v's subtree has at least one hub (hub_count[v] > 0) AND
            # 2. the rest of the tree (including u's other subtrees and u's ancestors) has at least one hub
            #    This is true if either:
            #    a) u's parent subtree has a hub, or
            #    b) another sibling subtree of v has a hub
            if hub_count[v] > 0:  # v's subtree has hubs
                # Check if rest of tree has hubs
                # Total hubs outside v's subtree = t - hub_count[v]
                if t - hub_count[v] > 0:  # Rest of tree has hubs
                    total_cost += w
            calculate_cost(v)
    
    calculate_cost(1)
    print(total_cost)

if __name__ == "__main__":
    main()
```
