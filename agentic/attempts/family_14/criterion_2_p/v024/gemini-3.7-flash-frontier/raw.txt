```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    idx = 3
    hubs = [int(x) for x in data[idx:idx+t]]
    idx += t
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    is_hub = [False] * (n + 1)
    for h in hubs:
        is_hub[h] = True
        
    root = hubs[0]
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    
    # BFS to establish tree structure
    queue = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                queue.append(v)
                
    cnt = [0] * (n + 1)
    for h in hubs:
        cnt[h] = 1
        
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if cnt[u] > 0:
            total_cost += edge_weight[u]
            cnt[parent[u]] += cnt[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
