We can solve this by pruning the tree: remove all subtrees that contain no priority sites, then sum the weights of the remaining edges.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    priority = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to mark which nodes are needed
    needed = [False] * (n + 1)
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    
    stack = [(1, 0, 0)]  # node, parent, weight from parent
    order = []
    while stack:
        node, par, w = stack.pop()
        parent[node] = par
        parent_edge_weight[node] = w
        order.append(node)
        for nei, weight in adj[node]:
            if nei != par:
                stack.append((nei, node, weight))
    
    # Mark priority nodes
    for p in priority:
        needed[p] = True
    
    # Post-order: propagate needed status upward
    total_cost = 0
    for node in reversed(order):
        if needed[node]:
            # If this node is needed and not root, add edge from parent
            if parent[node] != 0:
                total_cost += parent_edge_weight[node]
                needed[parent[node]] = True
        else:
            # If not needed, but parent is needed, we don't add anything
            pass
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
