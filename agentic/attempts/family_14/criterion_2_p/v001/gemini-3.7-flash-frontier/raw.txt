```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return
    
    ptr = 3
    is_priority = [False] * (n + 1)
    for _ in range(t):
        is_priority[int(data[ptr])] = True
        ptr += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[ptr])
        v = int(data[ptr + 1])
        w = int(data[ptr + 2])
        ptr += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    order.append(1)
    
    visited = [False] * (n + 1)
    visited[1] = True
    
    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                order.append(v)
                
    count = [0] * (n + 1)
    for i in range(1, n + 1):
        if is_priority[i]:
            count[i] = 1
            
    total_cost = 0
    for u in reversed(order):
        if u != 1:
            p = parent[u]
            c = count[u]
            count[p] += c
            if 0 < c < t:
                total_cost += parent_edge_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    main()
```
