```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    priority = [0] * (n + 1)
    idx = 3
    for _ in range(t):
        p = int(input_data[idx])
        priority[p] = 1
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    parent = [0] * (n + 1)
    weight_from_parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    
    # Iterative DFS to avoid recursion limit and establish parent-child relationships
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                weight_from_parent[v] = w
                stack.append(v)
                
    # Process nodes in reverse order of DFS (bottom-up)
    max_edge = [0] * (n + 1)
    total_cost = 0
    
    for u in reversed(order):
        if u == 1:
            continue
        p = parent[u]
        if priority[u] >= 2:
            priority[p] += 1
            total_cost += max_edge[u]
        elif priority[u] == 1:
            priority[p] += 1
            max_edge[p] = max(max_edge[p], weight_from_parent[u])
        else:
            max_edge[p] = max(max_edge[p], max_edge[u])
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
