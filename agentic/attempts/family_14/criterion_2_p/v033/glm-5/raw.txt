import sys
sys.setrecursionlimit(400000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    required = [False] * (n + 1)
    for _ in range(t):
        required[int(next(it))] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    children = [[] for _ in range(n + 1)]
    
    # Build tree structure with root at 1
    stack = [1]
    parent[1] = 0
    while stack:
        u = stack.pop()
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                children[u].append(v)
                stack.append(v)
    
    # Post-order traversal to compute max edge weight to required descendant
    max_edge = [0] * (n + 1)
    has_req_below = [False] * (n + 1)
    
    stack = [(1, False)]
    while stack:
        u, visited = stack.pop()
        if not visited:
            stack.append((u, True))
            for v in children[u]:
                stack.append((v, False))
        else:
            if required[u]:
                has_req_below[u] = True
            for v in children[u]:
                if has_req_below[v]:
                    has_req_below[u] = True
                    if max_edge[v] > max_edge[u]:
                        max_edge[u] = max_edge[v]
            # Check if we need to include edge to parent
            if has_req_below[u] and required[parent[u]]:
                if parent_weight[u] > max_edge[u]:
                    max_edge[u] = parent_weight[u]
    
    total_cost = 0
    for u in range(2, n + 1):
        if has_req_below[u] and required[parent[u]]:
            total_cost += max_edge[u]
    
    print(total_cost)

solve()
