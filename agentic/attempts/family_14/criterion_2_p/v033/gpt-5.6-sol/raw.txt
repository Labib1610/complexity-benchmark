```python
import sys
from array import array
from collections import deque

it = iter(map(int, sys.stdin.buffer.read().split()))
n = next(it)
m = next(it)
t = next(it)

required = bytearray(n)
for _ in range(t):
    required[next(it) - 1] = 1

head = array('i', [-1]) * n
degree = array('i', [0]) * n
to = array('i')
weight = array('i')
nxt = array('i')

total_cost = 0

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    w = next(it)

    to.append(v)
    weight.append(w)
    nxt.append(head[u])
    head[u] = len(to) - 1

    to.append(u)
    weight.append(w)
    nxt.append(head[v])
    head[v] = len(to) - 1

    degree[u] += 1
    degree[v] += 1
    total_cost += w

removed = bytearray(n)
queue = deque(
    i for i in range(n)
    if degree[i] <= 1 and not required[i]
)

while queue:
    u = queue.popleft()
    if removed[u]:
        continue

    edge = head[u]
    while edge != -1 and removed[to[edge]]:
        edge = nxt[edge]

    removed[u] = 1

    if edge != -1:
        v = to[edge]
        total_cost -= weight[edge]
        degree[v] -= 1
        if degree[v] == 1 and not required[v]:
            queue.append(v)

print(total_cost)
```
