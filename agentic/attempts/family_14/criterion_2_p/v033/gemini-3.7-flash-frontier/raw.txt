```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])

    req = [int(x) for x in input_data[3 : 3 + t]]

    idx = 3 + t
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))

    is_req = [0] * (n + 1)
    for r in req:
        is_req[r] = 1

    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    parent[1] = -1

    queue = [1]
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                weight_to_parent[v] = w
                queue.append(v)

    ans = 0
    cnt = is_req[:]

    for v in reversed(queue):
        if v == 1:
            continue
        p = parent[v]
        cnt[p] += cnt[v]
        if 0 < cnt[v] < t:
            ans += weight_to_parent[v]

    print(ans)


if __name__ == "__main__":
    solve()
```
