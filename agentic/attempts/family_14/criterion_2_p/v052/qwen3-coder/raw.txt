import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal_with_terminals(n, edges, terminals):
    if len(terminals) <= 1:
        return 0
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    # Initialize DSU
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    # Track which components have terminals
    component_terminals = defaultdict(set)
    for t in terminals:
        component_terminals[t].add(t)
    
    total_cost = 0
    
    for u, v, w in edges:
        pu, pv = find(parent, u), find(parent, v)
        if pu == pv:
            continue
            
        # Union the components
        if rank[pu] < rank[pv]:
            pu, pv = pv, pu
        parent[pv] = pu
        if rank[pu] == rank[pv]:
            rank[pu] += 1
            
        # Merge terminal sets
        component_terminals[pu] |= component_terminals[pv]
        component_terminals.pop(pv, None)
        
        total_cost += w
        
        # If all terminals are connected, we're done
        if len(component_terminals[pu]) == len(terminals):
            break
    
    return total_cost

# Read input
line = sys.stdin.readline().split()
n, m, t = int(line[0]), int(line[1]), int(line[2])

terminals = set(map(int, sys.stdin.readline().split()))

edges = []
for _ in range(m):
    u, v, w = map(int, sys.stdin.readline().split())
    edges.append((u, v, w))

# Solve
result = kruskal_with_terminals(n, edges, terminals)
print(result)
