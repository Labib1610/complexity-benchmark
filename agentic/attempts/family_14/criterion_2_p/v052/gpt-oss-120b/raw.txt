```python
import sys
sys.setrecursionlimit(1_000_000)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))   # = n-1
    t = int(next(it))
    terminals = [int(next(it)) - 1 for _ in range(t)]
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))

    is_term = [0] * n
    for v in terminals:
        is_term[v] = 1

    total_term = t
    ans = 0

    parent = [-1] * n
    stack = [0]
    order = []
    parent[0] = -2  # mark visited
    while stack:
        u = stack.pop()
        order.append(u)
        for v, _ in adj[u]:
            if parent[v] == -1:
                parent[v] = u
                stack.append(v)

    sub_cnt = [0] * n
    for u in reversed(order):
        cnt = is_term[u]
        for v, w in adj[u]:
            if parent[v] == u:          # v is child
                cnt_child = sub_cnt[v]
                if cnt_child and total_term - cnt_child:
                    ans += w
                cnt += cnt_child
        sub_cnt[u] = cnt

    print(ans)

if __name__ == "__main__":
    solve()
```
