```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    is_key = [0] * n
    for _ in range(t):
        is_key[next(it) - 1] = 1

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    parent = [-1] * n
    parent_weight = [0] * n
    order = [0]
    parent[0] = 0

    for u in order:
        for v, w in graph[u]:
            if parent[v] == -1:
                parent[v] = u
                parent_weight[v] = w
                order.append(v)

    count = is_key[:]
    answer = 0

    for u in reversed(order[1:]):
        if 0 < count[u] < t:
            answer += parent_weight[u]
        count[parent[u]] += count[u]

    print(answer)

if __name__ == "__main__":
    main()
```
