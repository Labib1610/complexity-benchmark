import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    flagged = set(map(int, sys.stdin.readline().split()))
    
    # Since m = n-1 and it's a tree (connected, no cycles)
    adj = [[] for _ in range(n+1)]
    edges = []
    
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((w, u, v))
    
    # If there's only one flagged node, we don't need any edges
    if t == 1:
        print(0)
        return
    
    # We need to find the minimum spanning tree that connects all flagged nodes.
    # This is essentially the Steiner tree problem on a tree, which can be solved optimally.
    # The optimal solution is: 
    # 1. Find the subtree that contains all flagged nodes (i.e., the minimal connected subgraph containing all flagged nodes).
    # 2. The cost is the sum of weights of all edges in that subtree.
    
    # In a tree, the minimal connected subgraph containing all flagged nodes is obtained by:
    # - Marking all flagged nodes as "special"
    # - For each non-special node, if removing it disconnects two special nodes, then we must include it (but since it's a tree,
    #   we just need to include all edges in the paths between special nodes)
    # Actually, in a tree, the minimal way to connect a subset of nodes is to take the union of all paths between them.
    # But there's a simpler way:
    # The answer is the total weight of the subtree induced by the flagged nodes.
    # Algorithm:
    # 1. Root the tree at any node.
    # 2. Compute for each node whether its subtree contains any flagged node.
    # 3. An edge {u,v} is included if both subtrees (after removing the edge) contain at least one flagged node.
    # Equivalently, an edge is included if it's on the path between any two flagged nodes.
    # There's a linear-time algorithm for this.
    
    # Let's use the standard approach for trees:
    # Build the "Steiner tree" which is the minimal subtree connecting all terminals (flagged nodes).
    # In a tree, this is equivalent to finding the minimal connected subgraph containing all terminals.
    # The cost is the sum of edge weights in this subtree.
    
    # Efficient approach:
    # 1. Root the tree at node 1.
    # 2. Do a DFS to compute for each node whether its subtree contains any flagged node.
    # 3. Count an edge if both sides have at least one flagged node.
    
    # Let's implement this.
    
    # Build adjacency list with edge indices
    adj_list = [[] for _ in range(n+1)]
    edge_weights = [0] * (m + 1)
    edge_index = 0
    edge_map = {}
    
    for u, v, w in edges:
        edge_index += 1
        adj_list[u].append((v, w, edge_index))
        adj_list[v].append((u, w, edge_index))
        edge_weights[edge_index] = w
        edge_map[(u,v)] = edge_index
        edge_map[(v,u)] = edge_index
        
    visited = [False] * (n + 1)
    has_flagged = [False] * (n + 1)
    result = 0
    
    def dfs(v, parent_edge_index):
        nonlocal result
        visited[v] = True
        count = 1 if v in flagged else 0
        
        for neighbor, weight, edge_idx in adj_list[v]:
            if not visited[neighbor]:
                child_count = dfs(neighbor, edge_idx)
                count += child_count
                # If the child subtree has at least one flagged node and the current subtree will have at least one flagged node,
                # then this edge should be included in the Steiner tree
                if child_count > 0:
                    result += weight
                    
        has_flagged[v] = (count > 0)
        return count
    
    # Start DFS from node 1
    dfs(1, -1)
    
    print(result)

if __name__ == "__main__":
    main()
