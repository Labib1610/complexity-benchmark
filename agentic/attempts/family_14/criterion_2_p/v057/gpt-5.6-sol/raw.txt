```python
import sys

def main():
    input_data = sys.stdin.buffer
    n, m, t = map(int, input_data.readline().split())
    flagged = list(map(int, input_data.readline().split()))

    if t == 1:
        print(0)
        return

    marked = bytearray(n + 1)
    for node in flagged:
        marked[node] = 1

    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input_data.readline().split())
        graph[u].append((v, w))
        graph[v].append((u, w))

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    parent[1] = -1
    order = [1]

    for u in order:
        for v, w in graph[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                order.append(v)

    terminal_count = [0] * (n + 1)
    for node in flagged:
        terminal_count[node] = 1

    answer = 0
    for u in reversed(order[1:]):
        count = terminal_count[u]
        if 0 < count < t:
            answer += parent_weight[u]
        terminal_count[parent[u]] += count

    print(answer)

if __name__ == "__main__":
    main()
```
