```python
import sys

data = sys.stdin.buffer.read().split()
it = iter(data)

n = int(next(it))
m = int(next(it))
t = int(next(it))

terminals = [int(next(it)) for _ in range(t)]

if t == 1:
    print(0)
    sys.exit()

adj = [[] for _ in range(n + 1)]
for _ in range(m):
    u = int(next(it))
    v = int(next(it))
    w = int(next(it))
    adj[u].append((v, w))
    adj[v].append((u, w))

terminal_count = [0] * (n + 1)
for node in terminals:
    terminal_count[node] = 1

parent = [0] * (n + 1)
parent_weight = [0] * (n + 1)
order = [1]

for u in order:
    for v, w in adj[u]:
        if v != parent[u]:
            parent[v] = u
            parent_weight[v] = w
            order.append(v)

answer = 0

for u in reversed(order[1:]):
    count = terminal_count[u]
    if 0 < count < t:
        answer += parent_weight[u]
    terminal_count[parent[u]] += count

print(answer)
```
