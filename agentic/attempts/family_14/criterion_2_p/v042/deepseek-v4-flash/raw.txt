We can solve this by finding the minimal subtree that connects all the primary life-support centers.  
Since the graph is a tree, the minimal connected subgraph containing a set of nodes is the union of all paths between those nodes.  
We can compute this by:

1. Mark the primary nodes.
2. Do a DFS from any primary node, and count edges that are needed to connect all primary nodes.
3. The total energy is the sum of weights of those edges.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    primary = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t == 1:
        print(0)
        return
    
    # DFS to find edges that are part of the minimal subtree
    visited = [False] * (n + 1)
    total_cost = 0
    
    def dfs(u, parent):
        nonlocal total_cost
        visited[u] = True
        has_primary = u in primary
        for v, w in adj[u]:
            if v == parent:
                continue
            child_has_primary = dfs(v, u)
            if child_has_primary:
                total_cost += w
            has_primary = has_primary or child_has_primary
        return has_primary
    
    # Start from any primary node
    start = next(iter(primary))
    dfs(start, -1)
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
