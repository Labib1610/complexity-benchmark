```python
import sys

def main():
    input = sys.stdin.buffer.readline
    n, m, t = map(int, input().split())

    leased = bytearray(n)
    remaining = t
    while remaining:
        for rack in map(int, input().split()):
            leased[rack - 1] = 1
            remaining -= 1

    if t == 1:
        print(0)
        return

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append((v, w))
        graph[v].append((u, w))

    parent = [-2] * n
    parent_weight = [0] * n
    parent[0] = -1
    order = [0]

    for u in order:
        for v, w in graph[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                order.append(v)

    terminal_count = list(leased)
    answer = 0

    for i in range(n - 1, 0, -1):
        u = order[i]
        count = terminal_count[u]
        if 0 < count < t:
            answer += parent_weight[u]
        terminal_count[parent[u]] += count

    print(answer)

if __name__ == "__main__":
    main()
```
