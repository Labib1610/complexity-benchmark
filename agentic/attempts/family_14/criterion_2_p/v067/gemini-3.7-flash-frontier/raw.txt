```python
import sys

# Increase recursion depth just in case, though iterative is safer
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    ptr = 3
    primaries = [int(x) for x in data[ptr:ptr+t]]
    ptr += t
    
    if t <= 1:
        print(0)
        return
    
    is_primary = [0] * (n + 1)
    for p in primaries:
        is_primary[p] = 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[ptr])
        v = int(data[ptr+1])
        w = int(data[ptr+2])
        ptr += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # BFS to establish parent pointers and post-order traversal
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    
    root = primaries[0]
    queue = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                queue.append(v)
                
    # Bottom-up DP to count primary vertices in each subtree
    primary_count = [0] * (n + 1)
    for u in range(1, n + 1):
        primary_count[u] = is_primary[u]
        
    total_cost = 0
    for u in reversed(order):
        p = parent[u]
        if p != 0:
            primary_count[p] += primary_count[u]
            if 0 < primary_count[u] < t:
                total_cost += parent_edge_weight[u]
                
    print(total_cost)

if __name__ == '__main__':
    solve()
```
