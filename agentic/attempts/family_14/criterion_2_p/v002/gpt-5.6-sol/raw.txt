```python
import sys

def integers():
    data = sys.stdin.buffer.read()
    num = 0
    in_num = False
    for b in data:
        if 48 <= b <= 57:
            num = num * 10 + b - 48
            in_num = True
        elif in_num:
            yield num
            num = 0
            in_num = False
    if in_num:
        yield num

it = iter(integers())
n = next(it)
m = next(it)
t = next(it)

critical = [False] * (n + 1)
for _ in range(t):
    critical[next(it)] = True

graph = [[] for _ in range(n + 1)]
for _ in range(m):
    u = next(it)
    v = next(it)
    w = next(it)
    graph[u].append((v, w))
    graph[v].append((u, w))

parent = [0] * (n + 1)
parent_weight = [0] * (n + 1)
order = [1]
parent[1] = -1

for u in order:
    for v, w in graph[u]:
        if v != parent[u]:
            parent[v] = u
            parent_weight[v] = w
            order.append(v)

count = [0] * (n + 1)
for v in range(1, n + 1):
    if critical[v]:
        count[v] = 1

answer = 0
for v in reversed(order[1:]):
    if 0 < count[v] < t:
        answer += parent_weight[v]
    count[parent[v]] += count[v]

print(answer)
```
