```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return

    idx = 3
    targets = [int(data[idx + i]) for i in range(t)]
    idx += t
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        w = int(data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    root = targets[0]
    
    is_target = [False] * (n + 1)
    for tgt in targets:
        is_target[tgt] = True
        
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    order = []
    
    # BFS to establish parent relationships and a topological order
    queue = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                weight_to_parent[v] = w
                queue.append(v)
                
    has_target = [is_target[i] for i in range(n + 1)]
    total_energy = 0
    
    for u in reversed(order):
        if u == root:
            continue
        if has_target[u]:
            total_energy += weight_to_parent[u]
            has_target[parent[u]] = True
            
    print(total_energy)

if __name__ == '__main__':
    main()
```
