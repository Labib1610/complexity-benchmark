```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return
    
    ptr = 3
    is_mural = [0] * (n + 1)
    for _ in range(t):
        u = int(data[ptr])
        is_mural[u] = 1
        ptr += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[ptr])
        v = int(data[ptr + 1])
        w = int(data[ptr + 2])
        ptr += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    order = []
    
    # BFS to establish tree structure
    queue = [1]
    parent[1] = -1
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                weight_to_parent[v] = w
                queue.append(v)
                
    cnt = is_mural[:]
    ans = 0
    
    # Bottom-up DP
    for u in reversed(order):
        if parent[u] != -1:
            p = parent[u]
            cnt[p] += cnt[u]
            if 0 < cnt[u] < t:
                ans += weight_to_parent[u]
                
    print(ans)

if __name__ == '__main__':
    solve()
```
