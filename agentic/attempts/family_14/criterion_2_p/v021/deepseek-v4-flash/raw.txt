We can solve this by treating the tunnels as a tree and finding the minimal subtree that connects all mural chambers.  
The minimal cost is the sum of all edges in that minimal subtree.  

A standard approach:  
- Mark mural nodes.  
- Do a DFS from any mural node, and for each edge, if the subtree below it contains at least one mural node, then that edge must be included.  
- Sum the weights of such edges.  

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    murals = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # pick any mural as root
    root = next(iter(murals))
    
    parent = [0] * (n + 1)
    parent_w = [0] * (n + 1)
    order = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    # BFS/DFS to build parent info
    stack = [root]
    while stack:
        u = stack.pop()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_w[v] = w
                stack.append(v)
                order.append(v)
    
    has_mural = [False] * (n + 1)
    for x in murals:
        has_mural[x] = True
    
    total = 0
    # process in reverse order (post-order)
    for u in reversed(order):
        if u == root:
            continue
        if has_mural[u]:
            total += parent_w[u]
            # propagate up
            p = parent[u]
            has_mural[p] = True
    
    print(total)

if __name__ == "__main__":
    solve()
```
