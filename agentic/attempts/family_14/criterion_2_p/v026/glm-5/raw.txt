import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    flagged = [False] * (n + 1)
    for _ in range(t):
        x = int(data[idx]); idx += 1
        flagged[x] = True
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    if t <= 1:
        print(0)
        return
    
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == 0 and v != parent[1]:
                parent[v] = u
                parent_weight[v] = w
                stack.append(v)
    
    need = [False] * (n + 1)
    for i in range(1, n + 1):
        if flagged[i]:
            need[i] = True
    
    total = 0
    for u in reversed(order):
        if u == 1:
            continue
        if need[u]:
            total += parent_weight[u]
            need[parent[u]] = True
    
    print(total)

solve()
