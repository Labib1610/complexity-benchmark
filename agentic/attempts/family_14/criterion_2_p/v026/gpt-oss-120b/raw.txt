import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)
    n = next(it)
    m = next(it)          # equals n-1
    t = next(it)

    special = [0] * (n + 1)
    for _ in range(t):
        special[next(it)] = 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        w = next(it)
        adj[u].append((v, w))
        adj[v].append((u, w))

    subtree_cnt = [0] * (n + 1)
    ans = 0

    # iterative DFS: (node, parent, weight_to_parent, state)
    # state 0 = first time, state 1 = after processing children
    stack = [(1, 0, 0, 0)]
    while stack:
        node, parent, w_par, state = stack.pop()
        if state == 0:
            stack.append((node, parent, w_par, 1))
            for nb, wt in adj[node]:
                if nb != parent:
                    stack.append((nb, node, wt, 0))
        else:
            cnt = special[node]
            for nb, _ in adj[node]:
                if nb != parent:
                    cnt += subtree_cnt[nb]
            subtree_cnt[node] = cnt
            if parent != 0:
                if cnt > 0 and t - cnt > 0:
                    ans += w_par

    sys.stdout.write(str(ans))

if __name__ == "__main__":
    solve()
