```python
import sys

data = list(map(int, sys.stdin.buffer.read().split()))
idx = 0

n = data[idx]
m = data[idx + 1]
t = data[idx + 2]
idx += 3

flagged = bytearray(n + 1)
for _ in range(t):
    flagged[data[idx]] = 1
    idx += 1

graph = [[] for _ in range(n + 1)]
for _ in range(m):
    u = data[idx]
    v = data[idx + 1]
    w = data[idx + 2]
    idx += 3
    graph[u].append((v, w))
    graph[v].append((u, w))

del data

parent = [-1] * (n + 1)
parent_weight = [0] * (n + 1)
parent[1] = 0
order = [1]

for u in order:
    for v, w in graph[u]:
        if v != parent[u]:
            parent[v] = u
            parent_weight[v] = w
            order.append(v)

count = [0] * (n + 1)
answer = 0

for u in reversed(order):
    count[u] += flagged[u]
    if parent[u] != 0:
        if 0 < count[u] < t:
            answer += parent_weight[u]
        count[parent[u]] += count[u]

print(answer)
```
