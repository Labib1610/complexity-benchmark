```python
import sys
from collections import deque

data = list(map(int, sys.stdin.buffer.read().split()))
n, m, t = data[0], data[1], data[2]

if t <= 1:
    print(0)
    sys.exit()

required = bytearray(n + 1)
idx = 3
for i in range(t):
    required[data[idx + i]] = 1
idx += t

adj = [[] for _ in range(n + 1)]
degree = [0] * (n + 1)
answer = 0

for _ in range(m):
    u, v, w = data[idx], data[idx + 1], data[idx + 2]
    idx += 3
    adj[u].append((v, w))
    adj[v].append((u, w))
    degree[u] += 1
    degree[v] += 1
    answer += w

active = bytearray(b'\x01') * (n + 1)
active[0] = 0
queue = deque(
    v for v in range(1, n + 1)
    if not required[v] and degree[v] <= 1
)

while queue:
    v = queue.popleft()
    if not active[v]:
        continue

    active[v] = 0
    for neighbor, weight in adj[v]:
        if active[neighbor]:
            answer -= weight
            degree[neighbor] -= 1
            if not required[neighbor] and degree[neighbor] == 1:
                queue.append(neighbor)

print(answer)
```
