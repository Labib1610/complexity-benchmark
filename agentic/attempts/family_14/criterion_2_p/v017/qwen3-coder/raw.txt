import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    if t <= 1:
        print(0)
        return
    
    high_priority = set()
    for _ in range(t):
        high_priority.add(int(data[idx]))
        idx += 1
    
    # Build the tree
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # Find the spanning tree that connects all high priority nodes
    # We need to find the minimal subtree (Steiner tree) that connects all high priority nodes
    # Since the graph is a tree, the Steiner tree is the union of paths between all pairs
    # But there's a simpler way: find the minimal connected subtree containing all high priority nodes
    
    # Mark which nodes are in the subtree
    # We can do this by starting from one high priority node and doing a DFS/BFS
    # to find all nodes that are on paths between high priority nodes
    
    # A simpler and correct approach:
    # The minimal cost is the sum of edge weights in the subtree that connects all high priority nodes
    # Since the original graph is a tree, we can:
    # 1. Mark all high priority nodes
    # 2. Do a DFS from any high priority node to find which edges are needed
    # Actually, the correct way is to find the minimal subtree that connects all high priority nodes
    # This is the Steiner tree problem on a tree, which can be solved by:
    # - Rooting the tree at any node
    # - Finding the minimal connected subtree that contains all high priority nodes
    
    # Let's use the following approach:
    # 1. Root the tree at an arbitrary node (e.g., node 1)
    # 2. For each node, compute if its subtree contains any high priority node
    # 3. An edge should be included if it connects two subtrees that both contain high priority nodes,
    #    or if one subtree contains a high priority node and the edge leads to a node whose subtree
    #    also contains a high priority node
    
    # Actually, let's use the standard algorithm for Steiner tree on a tree:
    # 1. Root the tree at any node
    # 2. For each node, compute whether its subtree contains any terminal (high priority node)
    # 3. Include an edge in the solution if both its endpoints' subtrees contain terminals
    
    # But there's an even simpler way for trees:
    # The Steiner tree is the union of paths between all pairs of high priority nodes
    # But we can compute it more efficiently:
    # Root the tree, then for each edge, include it if removing it would separate the tree into
    # two components, each containing at least one high priority node
    
    # Let's implement this:
    
    # Root the tree at node 1
    root = 1
    parent = [-1] * (n+1)
    parent_weight = [-1] * (n+1)  # weight of edge to parent
    children = defaultdict(list)
    
    # BFS to build rooted tree
    visited = [False] * (n+1)
    queue = deque([root])
    visited[root] = True
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_weight[v] = w
                children[u].append((v, w))
                queue.append(v)
    
    # For each node, compute if its subtree contains any high priority node
    contains_high = [False] * (n+1)
    
    # Post-order traversal to compute contains_high
    def dfs(u):
        if u in high_priority:
            contains_high[u] = True
        for v, w in children[u]:
            dfs(v)
            if contains_high[v]:
                contains_high[u] = True
    
    dfs(root)
    
    # Now compute the total cost
    total_cost = 0
    # For each node (except root), check if the edge to its parent should be included
    for u in range(1, n+1):
        if u != root and contains_high[u]:
            total_cost += parent_weight[u]
    
    print(total_cost)

if __name__ == "__main__":
    main()
