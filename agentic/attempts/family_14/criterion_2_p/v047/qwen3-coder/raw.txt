import sys
from collections import defaultdict
import heapq

def find(parent, x):
    if parent[x] != x:
        parent[x] = find(parent, parent[x])
    return parent[x]

def union(parent, rank, x, y):
    px, py = find(parent, x), find(parent, y)
    if px == py:
        return False
    if rank[px] < rank[py]:
        px, py = py, px
    parent[py] = px
    if rank[px] == rank[py]:
        rank[px] += 1
    return True

def kruskal(n, edges, critical):
    if len(critical) <= 1:
        return 0
    
    # Sort edges by weight
    edges.sort(key=lambda x: x[2])
    
    # Initialize Union-Find
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    total_cost = 0
    edges_used = 0
    critical_set = set(critical)
    critical_count = len(critical)
    
    # First, add edges to connect all critical facilities
    for u, v, w in edges:
        if find(parent, u) != find(parent, v):
            if union(parent, rank, u, v):
                total_cost += w
                edges_used += 1
                # Check if all critical facilities are connected
                roots = set(find(parent, c) for c in critical)
                if len(roots) == 1:
                    break
    
    return total_cost
