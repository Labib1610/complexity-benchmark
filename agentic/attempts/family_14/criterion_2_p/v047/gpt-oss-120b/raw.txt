```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); t = int(next(it))
    terminals = set(int(next(it)) for _ in range(t))
    if t == 1:
        print(0)
        return

    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w, len(edges)))
        adj[v].append((u, w, len(edges)))
        edges.append((u, v, w))

    parent = [0] * (n + 1)
    parent_edge = [-1] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, idx in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            parent_edge[v] = idx
            stack.append(v)

    term_cnt = [0] * (n + 1)
    for v in terminals:
        term_cnt[v] = 1

    total = 0
    for u in reversed(order):
        if parent[u] == -1:
            continue
        cnt = term_cnt[u]
        if 0 < cnt < t:
            total += edges[parent_edge[u]][2]
        term_cnt[parent[u]] += cnt

    print(total)

if __name__ == "__main__":
    solve()
```
