```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    idx = 3
    targets = [int(x) for x in data[idx:idx+t]]
    idx += t
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    root = targets[0]
    is_target = [0] * (n + 1)
    for x in targets:
        is_target[x] = 1
        
    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    
    queue = [root]
    parent[root] = -1
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                edge_weight[v] = w
                queue.append(v)
                
    count = [0] * (n + 1)
    for i in range(1, n + 1):
        count[i] = is_target[i]
        
    total_cost = 0
    for u in reversed(order):
        if u == root:
            continue
        if count[u] > 0:
            total_cost += edge_weight[u]
            count[parent[u]] += count[u]
            
    print(total_cost)

if __name__ == '__main__':
    main()
```
