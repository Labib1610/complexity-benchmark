```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))  # = n-1
    t = int(next(it))
    required = [0] * (n + 1)
    for _ in range(t):
        required[int(next(it))] = 1
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    parent = [0] * (n + 1)
    pw = [0] * (n + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            pw[v] = w
            stack.append(v)

    sub_req = [0] * (n + 1)
    for u in reversed(order):
        cnt = required[u]
        for v, w in adj[u]:
            if parent[v] == u:
                cnt += sub_req[v]
        sub_req[u] = cnt

    total_req = t
    ans = 0
    for v in range(2, n + 1):
        cnt = sub_req[v]
        if cnt > 0 and total_req - cnt > 0:
            ans += pw[v]
    print(ans)

if __name__ == "__main__":
    solve()
```
