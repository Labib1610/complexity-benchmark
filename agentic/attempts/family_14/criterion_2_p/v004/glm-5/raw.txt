```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1

    required = [False] * (n + 1)
    for _ in range(t):
        v = int(data[idx]); idx += 1
        required[v] = True

    edges = []
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))

    if t <= 1:
        print(0)
        return

    LOG = max(1, (n).bit_length())
    up = [[0] * (n + 1) for _ in range(LOG)]
    max_edge = [[0] * (n + 1) for _ in range(LOG)]
    depth = [0] * (n + 1)

    stack = [(1, 0, 0, 0)]
    visited = [False] * (n + 1)
    visited[1] = True

    while stack:
        node, parent, w, d = stack.pop()
        up[0][node] = parent
        max_edge[0][node] = w
        depth[node] = d
        for nxt, weight in adj[node]:
            if not visited[nxt]:
                visited[nxt] = True
                stack.append((nxt, node, weight, d + 1))

    for k in range(1, LOG):
        up_k = up[k]
        up_prev = up[k - 1]
        max_k = max_edge[k]
        max_prev = max_edge[k - 1]
        for v in range(1, n + 1):
            mid = up_prev[v]
            up_k[v] = up_prev[mid]
            max_k[v] = max_prev[v] if max_prev[v] > max_prev[mid] else max_prev[mid]

    def get_max_edge(u, v):
        if u == v:
            return 0
        res = 0
        if depth[u] < depth[v]:
            u, v = v, u
        diff = depth[u] - depth[v]
        k = 0
        while diff:
            if diff & 1:
                if max_edge[k][u] > res:
                    res = max_edge[k][u]
                u = up[k][u]
            diff >>= 1
            k += 1
        if u == v:
            return res
        for k in range(LOG - 1, -1, -1):
            if up[k][u] != up[k][v]:
                if max_edge[k][u] > res:
                    res = max_edge[k][u]
                if max_edge[k][v] > res:
                    res = max_edge[k][v]
                u = up[k][u]
                v = up[k][v]
        if max_edge[0][u] > res:
            res = max_edge[0][u]
        if max_edge[0][v] > res:
            res = max_edge[0][v]
        return res

    req_nodes = [i for i in range(1, n + 1) if required[i]]
    req_nodes.sort(key=lambda x: depth[x])

    parent_uf = list(range(n + 1))

    def find(x):
        root = x
        while parent_uf[root] != root:
            root = parent_uf[root]
        while parent_uf[x] != root:
            parent_uf[x], x = root, parent_uf[x]
        return root

    def union(x, y):
        rx = find(x)
        ry = find(y)
        if rx != ry:
            parent_uf[rx] = ry
            return True
        return False

    total = 0
    for node in req_nodes:
        root = find(node)
        if root != node:
            total += get_max_edge(node, root)
            union(node, root)

    print(total)

solve()
```
