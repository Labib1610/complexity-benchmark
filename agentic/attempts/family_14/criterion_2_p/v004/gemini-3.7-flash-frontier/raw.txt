```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    idx = 3
    if t <= 1:
        print(0)
        return
    
    req_count = [0] * (n + 1)
    for _ in range(t):
        req_count[int(input_data[idx])] = 1
        idx += 1
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # BFS to build tree and ordering
    parent = [0] * (n + 1)
    edge_to_parent = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    order = []
    queue = [1]
    visited[1] = True
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_to_parent[v] = w
                queue.append(v)
    
    total_cost = 0
    for u in reversed(order):
        p = parent[u]
        if p != 0:
            if 0 < req_count[u] < t:
                total_cost += edge_to_parent[u]
            req_count[p] += req_count[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
