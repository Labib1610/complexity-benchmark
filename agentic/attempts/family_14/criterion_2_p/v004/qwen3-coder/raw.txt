```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    required = list(map(int, sys.stdin.readline().split()))
    
    # If only one required site, cost is 0
    if t <= 1:
        print(0)
        return
    
    required_set = set(required)
    
    # Build the tree
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # We need to find the minimal subtree that connects all required nodes.
    # This is the Steiner tree problem on a tree, which can be solved efficiently.
    
    # Approach:
    # 1. Root the tree at any node (say node 1)
    # 2. Compute for each node the "important" descendants (required nodes in subtree)
    # 3. An edge is needed in the solution if and only if removing it separates 
    #    the tree into two components, each containing at least one required node.
    
    # Let's use a two-pass DFS approach:
    # First DFS: compute for each node, which required nodes are in its subtree
    # Second DFS: count edges where both sides have at least one required node
    
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    children = [[] for _ in range(n + 1)]
    
    # First, build a rooted tree (DFS from node 1)
    visited = [False] * (n + 1)
    stack = [1]
    visited[1] = True
    while stack:
        u = stack.pop()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                children[u].append(v)
                stack.append(v)
    
    # Now, for each node, compute the set of required nodes in its subtree
    # We'll represent this as a count per node (since we only care about presence)
    req_in_subtree = [0] * (n + 1)
    
    # Post-order traversal to compute req_in_subtree
    def dfs1(u):
        if u in required_set:
            req_in_subtree[u] = 1
        for v in children[u]:
            dfs1(v)
            req_in_subtree[u] += req_in_subtree[v]
    
    dfs1(1)
    
    # Now, count how many required nodes there are in total
    total_required = len(required_set)
    
    # Total cost is sum of edge weights where:
    #   req_in_subtree[v] > 0 and (total_required - req_in_subtree[v]) > 0
    # That means: v's subtree has some required nodes, and the rest of the tree has some too.
    total_cost = 0
    for u in range(2, n + 1):  # Skip root
        v = u
        w = parent_edge_weight[v]
        count_in_subtree = req_in_subtree[v]
        count_outside = total_required - count_in_subtree
        if count_in_subtree > 0 and count_outside > 0:
            total_cost += w
    
    print(total_cost)

if __name__ == "__main__":
    main()
```
