```python
import sys

def main():
    input = sys.stdin.buffer.readline
    n, m, t = map(int, input().split())

    keyholders = []
    while len(keyholders) < t:
        keyholders.extend(map(int, input().split()))

    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))

    marked = bytearray(n + 1)
    for u in keyholders:
        marked[u] = 1

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = [1]
    parent[1] = -1

    for u in order:
        for v, w in graph[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                order.append(v)

    count = [0] * (n + 1)
    for u in keyholders:
        count[u] = 1

    answer = 0
    for i in range(n - 1, 0, -1):
        u = order[i]
        if 0 < count[u] < t:
            answer += parent_weight[u]
        count[parent[u]] += count[u]

    print(answer)

if __name__ == "__main__":
    main()
```
