```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))
    
    keyholders = set()
    for _ in range(t):
        keyholders.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Find the Steiner tree for keyholders
    # Since the graph is a tree, we can use a greedy approach
    
    # Start with keyholders as initial components
    # Use modified Prim's algorithm
    
    in_tree = [False] * (n + 1)
    min_edge = [float('inf')] * (n + 1)
    
    # Priority queue: (weight, node)
    pq = []
    
    # Initialize with all keyholders
    for k in keyholders:
        min_edge[k] = 0
        heapq.heappush(pq, (0, k))
    
    total_cost = 0
    nodes_in_tree = 0
    
    while pq and nodes_in_tree < n:
        w, u = heapq.heappop(pq)
        if in_tree[u]:
            continue
        in_tree[u] = True
        total_cost += w
        nodes_in_tree += 1
        
        for v, weight in adj[u]:
            if not in_tree[v] and weight < min_edge[v]:
                min_edge[v] = weight
                heapq.heappush(pq, (weight, v))
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
