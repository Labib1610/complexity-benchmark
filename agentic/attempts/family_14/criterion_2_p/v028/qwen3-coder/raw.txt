```python
import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    if t <= 1:
        print(0)
        return
    
    spores = set()
    for _ in range(t):
        spore = int(data[idx])
        idx += 1
        spores.add(spore)
    
    # Build graph
    adj = [[] for _ in range(n+1)]
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((w, u, v))
    
    # Since the structure is a tree (m = n-1 and connected), we can use Union-Find or DFS
    # We'll find the minimal subtree connecting all spore nodes (Steiner tree on a tree)
    # For a tree, the minimal Steiner tree can be found by:
    # 1. Root the tree at any node
    # 2. Perform DFS to compute for each node whether it needs to be included
    # 3. A node is needed if it contains a spore or has more than one subtree containing spores
    
    # Let's root the tree at node 1
    parent = [-1] * (n+1)
    children = [[] for _ in range(n+1)]
    visited = [False] * (n+1)
    
    # BFS to build rooted tree
    queue = [1]
    visited[1] = True
    while queue:
        u = queue.pop()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append((v, w))
                queue.append(v)
    
    # Now perform post-order DFS to determine which edges to include
    # For each node, we track:
    # - whether the subtree contains any spore
    # - total cost to connect all spores in this subtree
    
    needed = [False] * (n+1)  # Whether this node is needed in the Steiner tree
    cost = [0] * (n+1)        # Cost to connect spores within this subtree
    
    def dfs(u):
        has_spore = (u in spores)
        child_needed = 0
        total_cost = 0
        
        for v, w in children[u]:
            dfs(v)
            if needed[v]:  # If child subtree needs connection
                child_needed += 1
                total_cost += cost[v] + w  # Include edge cost
        
        # If current node has spore or more than one child subtree needs connection
        if has_spore or child_needed > 1:
            needed[u] = True
            cost[u] = total_cost
        elif child_needed == 1:
            # Only one child needs connection, pass up the cost
            needed[u] = True
            cost[u] = total_cost
        else:
            # No spores below and not a spore itself
            needed[u] = False
            cost[u] = 0
    
    dfs(1)
    print(cost[1])

if __name__ == "__main__":
    main()
```
