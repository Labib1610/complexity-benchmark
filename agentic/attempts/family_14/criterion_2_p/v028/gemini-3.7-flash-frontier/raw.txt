```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])
    t = int(data[2])

    idx = 3
    spores = [int(data[idx + i]) for i in range(t)]
    idx += t

    if t <= 1:
        print(0)
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        w = int(data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3

    root = spores[0]
    is_spore = [False] * (n + 1)
    for s in spores:
        is_spore[s] = True

    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    
    # BFS to determine tree structure and post-order processing
    queue = [root]
    visited = [False] * (n + 1)
    visited[root] = True

    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                queue.append(v)

    spore_count = [1 if is_spore[i] else 0 for i in range(n + 1)]
    total_cost = 0

    for u in reversed(order):
        if u == root:
            continue
        if spore_count[u] > 0:
            total_cost += parent_edge_weight[u]
            spore_count[parent[u]] += spore_count[u]

    print(total_cost)

if __name__ == '__main__':
    main()
```
