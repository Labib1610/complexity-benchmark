```python
import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    ptr = 3
    is_terminal = [0] * (n + 1)
    for _ in range(t):
        term = int(input_data[ptr])
        is_terminal[term] = 1
        ptr += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        w = int(input_data[ptr + 2])
        ptr += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    
    queue = [1]
    visited = [False] * (n + 1)
    visited[1] = True
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_weight[v] = w
                queue.append(v)
                
    term_count = list(is_terminal)
    total_cost = 0
    
    for u in reversed(order):
        p = parent[u]
        if p != 0:
            if 0 < term_count[u] < t:
                total_cost += parent_weight[u]
            term_count[p] += term_count[u]
            
    print(total_cost)

if __name__ == '__main__':
    main()
```
