```python
import sys

def main():
    input = sys.stdin.buffer.readline
    n, m, t = map(int, input().split())
    terminals = list(map(int, input().split()))

    marked = [False] * (n + 1)
    for node in terminals:
        marked[node] = True

    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v, w = map(int, input().split())
        graph[u].append((v, w))
        graph[v].append((u, w))

    if t <= 1:
        print(0)
        return

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    parent[1] = -1
    order = []
    stack = [1]

    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in graph[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            parent_weight[v] = w
            stack.append(v)

    count = [0] * (n + 1)
    answer = 0

    for u in reversed(order):
        if marked[u]:
            count[u] += 1
        if parent[u] != -1:
            if 0 < count[u] < t:
                answer += parent_weight[u]
            count[parent[u]] += count[u]

    print(answer)

if __name__ == "__main__":
    main()
```
