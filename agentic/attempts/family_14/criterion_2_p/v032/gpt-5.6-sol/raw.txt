```python
import sys

data = sys.stdin.buffer.read()
idx = 0
length = len(data)

def next_int():
    global idx
    while idx < length and data[idx] <= 32:
        idx += 1
    value = 0
    while idx < length and data[idx] > 32:
        value = value * 10 + data[idx] - 48
        idx += 1
    return value

n = next_int()
m = next_int()
t = next_int()

critical = [0] * (n + 1)
for _ in range(t):
    critical[next_int()] = 1

graph = [[] for _ in range(n + 1)]
for _ in range(m):
    u = next_int()
    v = next_int()
    w = next_int()
    graph[u].append((v, w))
    graph[v].append((u, w))

parent = [0] * (n + 1)
parent_weight = [0] * (n + 1)
order = [1]
parent[1] = -1

for u in order:
    for v, w in graph[u]:
        if v != parent[u]:
            parent[v] = u
            parent_weight[v] = w
            order.append(v)

count = critical
answer = 0

for v in reversed(order[1:]):
    if 0 < count[v] < t:
        answer += parent_weight[v]
    count[parent[v]] += count[v]

print(answer)
```
