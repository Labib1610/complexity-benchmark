```python
import sys
from collections import deque

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
t = next(it)

essential = bytearray(n)
for _ in range(t):
    essential[next(it) - 1] = 1

graph = [[] for _ in range(n)]
degree = [0] * n
total = 0

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    w = next(it)
    graph[u].append((v, w))
    graph[v].append((u, w))
    degree[u] += 1
    degree[v] += 1
    total += w

queue = deque(
    i for i in range(n)
    if degree[i] == 1 and not essential[i]
)
removed = bytearray(n)

while queue:
    u = queue.popleft()
    if removed[u] or degree[u] != 1 or essential[u]:
        continue

    for v, w in graph[u]:
        if not removed[v]:
            removed[u] = 1
            degree[u] = 0
            degree[v] -= 1
            total -= w

            if degree[v] == 1 and not essential[v]:
                queue.append(v)
            break

print(total)
```
