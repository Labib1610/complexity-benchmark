```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, t = map(int, sys.stdin.readline().split())
    essential = set(map(int, sys.stdin.readline().split()))
    
    # Since m = n-1 and there's exactly one path between any two nodes,
    # the graph is a tree.
    
    adj = defaultdict(list)
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # If there's only one essential node, no edges are needed.
    if t <= 1:
        print(0)
        return
    
    # We need to find the minimal subtree that connects all essential nodes.
    # This is known as the Steiner tree problem on a tree, which can be solved efficiently.
    
    # Approach:
    # 1. Find all nodes that are essential or lie on paths between essential nodes.
    # 2. This can be done by finding the "closure" of essential nodes under the tree structure.
    # 3. A simpler way: find the union of paths between all pairs of essential nodes.
    # But since it's a tree, we can do better:
    # - Root the tree at some node.
    # - Find the lowest common ancestors (LCA) of all essential nodes.
    # - The minimal subtree includes all nodes that are on paths between essential nodes.
    
    # However, there's a simpler greedy approach for trees:
    # - Root the tree at an arbitrary node.
    # - For each node, compute whether it's necessary (i.e., lies on a path between two essential nodes).
    # - A node is necessary if it has >= 2 subtrees containing essential nodes, or
    #   it is essential and has >= 1 subtree containing essential nodes.
    
    # Let's root the tree at node 1.
    root = 1
    
    # Build parent/children relationships via BFS
    parent = [-1] * (n + 1)
    children = defaultdict(list)
    edge_weight = {}
    
    visited = [False] * (n + 1)
    queue = deque([root])
    visited[root] = True
    
    while queue:
        u = queue.popleft()
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append(v)
                edge_weight[(u, v)] = w
                edge_weight[(v, u)] = w
                queue.append(v)
    
    # Now compute for each node how many essential descendants it has (in its subtree).
    # We'll use post-order traversal.
    essential_in_subtree = [0] * (n + 1)
    
    def dfs(u):
        count = 1 if u in essential else 0
        for v in children[u]:
            count += dfs(v)
        essential_in_subtree[u] = count
        return count
    
    dfs(root)
    
    # Now determine which edges to include.
    # An edge {u, v} where u is the parent of v is included if:
    # - v's subtree contains at least one essential node, and
    # - not all essential nodes are in v's subtree (i.e., there are essential nodes outside v's subtree).
    
    total_power = 0
    total_essential = len(essential)
    
    for u in range(1, n + 1):
        for v in children[u]:
            # Edge from u to v
            # Include this edge if:
            # 1. v's subtree has at least one essential node
            # 2. v's subtree does NOT contain ALL essential nodes
            if essential_in_subtree[v] > 0 and essential_in_subtree[v] < total_essential:
                total_power += edge_weight[(u, v)]
    
    print(total_power)

if __name__ == "__main__":
    main()
```
