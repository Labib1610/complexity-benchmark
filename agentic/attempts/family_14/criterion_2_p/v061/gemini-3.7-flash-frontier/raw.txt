```python
import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])

    if t <= 1:
        print(0)
        return

    ptr = 3
    essential = [False] * (n + 1)
    root = int(input_data[ptr])
    for _ in range(t):
        essential[int(input_data[ptr])] = True
        ptr += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        w = int(input_data[ptr + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        ptr += 3

    parent = [0] * (n + 1)
    edge_weight = [0] * (n + 1)
    order = []
    order.append(root)
    visited = [False] * (n + 1)
    visited[root] = True

    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight[v] = w
                order.append(v)

    has_essential = [0] * (n + 1)
    for u in range(1, n + 1):
        if essential[u]:
            has_essential[u] = 1

    ans = 0
    for u in reversed(order):
        if u == root:
            continue
        if has_essential[u] > 0:
            ans += edge_weight[u]
            has_essential[parent[u]] += has_essential[u]

    print(ans)

if __name__ == '__main__':
    main()
```
