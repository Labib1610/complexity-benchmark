```python
import sys
from collections import deque

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
t = next(it)

critical = bytearray(n)
for _ in range(t):
    critical[next(it) - 1] = 1

graph = [[] for _ in range(n)]
total_cost = 0

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    w = next(it)
    graph[u].append((v, w))
    graph[v].append((u, w))
    total_cost += w

degree = [len(neighbors) for neighbors in graph]
removed = bytearray(n)
queue = deque(
    i for i in range(n)
    if not critical[i] and degree[i] <= 1
)

while queue:
    node = queue.popleft()
    if removed[node]:
        continue

    removed[node] = 1
    for neighbor, weight in graph[node]:
        if not removed[neighbor]:
            total_cost -= weight
            degree[neighbor] -= 1
            if not critical[neighbor] and degree[neighbor] == 1:
                queue.append(neighbor)
            break

print(total_cost)
```
