```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))  # = n-1
    t = int(next(it))
    specials = [0] * (n + 1)
    for _ in range(t):
        specials[int(next(it))] = 1

    adj = [[] for _ in range(n + 1)]
    edges = []
    for idx in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        edges.append((u, v, w))
        adj[u].append((v, w, idx))
        adj[v].append((u, w, idx))

    if t <= 1:
        print(0)
        return

    parent = [0] * (n + 1)
    parent_edge = [-1] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, eid in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            parent_edge[v] = (w, eid)
            stack.append(v)

    sub_cnt = [0] * (n + 1)
    for u in reversed(order):
        cnt = specials[u]
        for v, w, eid in adj[u]:
            if parent[v] == u:  # child
                cnt += sub_cnt[v]
        sub_cnt[u] = cnt

    total = 0
    for v in range(2, n + 1):
        cnt = sub_cnt[v]
        if 0 < cnt < t:
            total += parent_edge[v][0]

    print(total)

if __name__ == "__main__":
    solve()
```
