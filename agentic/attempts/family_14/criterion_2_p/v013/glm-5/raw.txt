```python
import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    critical = [False] * (n + 1)
    for _ in range(t):
        c = int(data[idx]); idx += 1
        critical[c] = True
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    if t <= 1:
        print(0)
        return
    
    # BFS to find parent relationships and order
    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    visited = [False] * (n + 1)
    root = 1
    visited[root] = True
    queue = [root]
    head = 0
    while head < len(queue):
        u = queue[head]; head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_weight[v] = w
                queue.append(v)
    
    # cnt[u] = number of critical facilities in subtree of u
    cnt = [0] * (n + 1)
    for u in order:
        if critical[u]:
            cnt[u] = 1
    
    for u in reversed(order):
        p = parent[u]
        if p != 0:
            cnt[p] += cnt[u]
    
    total = 0
    for u in range(2, n + 1):
        if cnt[u] > 0 and cnt[u] < t:
            total += parent_weight[u]
    
    print(total)

solve()
```
