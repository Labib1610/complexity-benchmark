```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])

    if t <= 1:
        print(0)
        return

    idx = 3
    active = [False] * (n + 1)
    for _ in range(t):
        node = int(input_data[idx])
        active[node] = True
        idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3

    # BFS from root = 1 to establish tree structure and topological order
    parent = [0] * (n + 1)
    edge_weight_to_parent = [0] * (n + 1)
    order = []
    order.append(1)
    visited = [False] * (n + 1)
    visited[1] = True

    head = 0
    while head < len(order):
        u = order[head]
        head += 1
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_weight_to_parent[v] = w
                order.append(v)

    # Post-order traversal to compute subtree active node counts
    count = [0] * (n + 1)
    for u in order:
        if active[u]:
            count[u] = 1

    total_cost = 0
    for u in reversed(order):
        if count[u] > 0 and count[u] < t:
            total_cost += edge_weight_to_parent[u]
        if parent[u] != 0:
            count[parent[u]] += count[u]

    print(total_cost)

if __name__ == '__main__':
    solve()
```
