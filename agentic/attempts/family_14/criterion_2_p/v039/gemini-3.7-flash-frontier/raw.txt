```python
import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    t = int(next(it))

    if t <= 1:
        print(0)
        return

    is_critical = [False] * (n + 1)
    for _ in range(t):
        c = int(next(it))
        is_critical[c] = True

    head = [-1] * (n + 1)
    to = [0] * (2 * m)
    weight = [0] * (2 * m)
    nxt = [0] * (2 * m)
    edge_idx = 0

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))

        to[edge_idx] = v
        weight[edge_idx] = w
        nxt[edge_idx] = head[u]
        head[u] = edge_idx
        edge_idx += 1

        to[edge_idx] = u
        weight[edge_idx] = w
        nxt[edge_idx] = head[v]
        head[v] = edge_idx
        edge_idx += 1

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    order = []
    order.append(1)
    parent[1] = -1

    queue_idx = 0
    while queue_idx < len(order):
        u = order[queue_idx]
        queue_idx += 1

        e = head[u]
        while e != -1:
            v = to[e]
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = weight[e]
                order.append(v)
            e = nxt[e]

    crit_count = [0] * (n + 1)
    for i in range(1, n + 1):
        if is_critical[i]:
            crit_count[i] = 1

    ans = 0
    for u in reversed(order):
        if parent[u] != -1:
            crit_count[parent[u]] += crit_count[u]
            if 0 < crit_count[u] < t:
                ans += parent_weight[u]

    print(ans)

if __name__ == '__main__':
    main()
```
