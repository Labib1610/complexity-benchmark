We can solve this by pruning the tree: remove all subtrees that contain no critical data centers, then sum the weights of the remaining edges.  

Here’s the code:  

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    critical = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    if t == 1:
        print(0)
        return
    
    parent = [0] * (n + 1)
    parent_w = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    # DFS to build parent relationships
    stack = [1]
    visited[1] = True
    order = []
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_w[v] = w
                stack.append(v)
    
    has_critical = [False] * (n + 1)
    for c in critical:
        has_critical[c] = True
    
    # Post-order processing
    total_cost = 0
    for u in reversed(order):
        if u == 1:
            continue
        if has_critical[u]:
            total_cost += parent_w[u]
            has_critical[parent[u]] = True
        else:
            # If this subtree has no critical, skip its edge
            pass
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
