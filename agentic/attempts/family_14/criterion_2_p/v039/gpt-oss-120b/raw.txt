```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))  # = n-1
    t = int(next(it))
    terminals = set()
    for _ in range(t):
        terminals.add(int(next(it)) - 1)  # zero‑based
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w, len(edges)))
        adj[v].append((u, w, len(edges)))
        edges.append((u, v, w))

    parent = [-1] * n
    parent_edge = [-1] * n
    order = []
    stack = [0]
    parent[0] = 0
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w, idx in adj[u]:
            if parent[v] == -1:
                parent[v] = u
                parent_edge[v] = idx
                stack.append(v)

    cnt = [0] * n
    for i in range(n):
        if i in terminals:
            cnt[i] = 1

    total = 0
    for u in reversed(order):
        if u == 0:
            continue
        p = parent[u]
        cnt[p] += cnt[u]
        # edge (p,u) is needed if both sides contain terminals
        if 0 < cnt[u] < t:
            total += edges[parent_edge[u]][2]

    print(total)

if __name__ == "__main__":
    solve()
```
