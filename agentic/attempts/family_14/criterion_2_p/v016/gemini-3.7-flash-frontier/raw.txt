```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return

    idx = 3
    critical = set()
    for _ in range(t):
        critical.add(int(data[idx]))
        idx += 1
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        w = int(data[idx + 2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    # BFS to establish tree structure from root 1
    root = 1
    order = []
    parent = [0] * (n + 1)
    edge_w = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    visited[root] = True
    queue = [root]
    
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                edge_w[v] = w
                queue.append(v)
                
    # Bottom-up DP to count critical nodes
    cnt = [0] * (n + 1)
    for u in critical:
        cnt[u] = 1
        
    total_cost = 0
    for u in reversed(order):
        if u != root:
            p = parent[u]
            cnt[p] += cnt[u]
            if 0 < cnt[u] < t:
                total_cost += edge_w[u]
                
    print(total_cost)

if __name__ == '__main__':
    main()
```
