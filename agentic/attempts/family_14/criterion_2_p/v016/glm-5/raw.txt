```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    critical = set()
    for _ in range(t):
        try:
            critical.add(int(next(iterator)))
        except StopIteration:
            break
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            w = int(next(iterator))
            adj[u].append((v, w))
            adj[v].append((u, w))
        except StopIteration:
            break

    # Special case: if t <= 1, no connections needed
    if t <= 1:
        print(0)
        return

    # We perform a modified Prim's algorithm.
    # The goal is to find the Minimum Spanning Tree of the subgraph induced by critical nodes,
    # where the weight between two critical nodes is the shortest path in the original tree.
    # Since the original graph is a tree, the shortest path is unique.
    # We can adapt Prim's: start with one critical node, and expand the "active" region.
    # The cost to add a new critical node is the minimum path weight to it from the current active set.
    # We maintain a priority queue of edges on the boundary of the active set.
    
    # Start with an arbitrary critical node
    start_node = next(iter(critical))
    
    in_mst = [False] * (n + 1)
    min_edge = [float('inf')] * (n + 1)
    
    # Priority queue stores (weight, neighbor)
    pq = []
    
    # Initialize with start node
    # We can treat the start node as having cost 0 to "enter" the MST
    heapq.heappush(pq, (0, start_node))
    
    total_cost = 0
    critical_count = 0
    
    while pq:
        w, u = heapq.heappop(pq)
        
        if in_mst[u]:
            continue
        
        in_mst[u] = True
        
        # If u is critical, we must pay the cost to connect it
        if u in critical:
            total_cost += w
            critical_count += 1
            if critical_count == t:
                break
        
        # Explore neighbors
        for v, weight in adj[u]:
            if not in_mst[v]:
                # The cost to reach v through u
                # In standard Prim's, we'd check if weight < min_edge[v]
                # Here, the cost to reach v is cost_to_reach_u + edge_weight
                # But wait, the logic for w is subtle.
                # w is the cost accumulated so far to reach u from the MST.
                # If u is critical, w is the path cost. If u is non-critical, w is the path cost.
                # The path cost propagates.
                # So the cost to reach v is w + weight.
                # However, we want the minimum cost to reach v.
                # Standard Prim's relaxation: if edge_weight < min_edge[v], update.
                # Here: if path_cost_to_v < min_edge[v], update.
                # path_cost_to_v = w + weight.
                
                new_dist = w + weight
                if new_dist < min_edge[v]:
                    min_edge[v] = new_dist
                    heapq.heappush(pq, (new_dist, v))
    
    print(total_cost)

if __name__ == '__main__':
    solve()
```
