```python
import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    critical = set()
    for _ in range(t):
        critical.add(int(data[idx]))
        idx += 1
    
    # If there's only one critical facility, cost is 0
    if t <= 1:
        print(0)
        return
    
    edges = []
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((w, u, v))
    
    # Sort edges by weight for Kruskal's algorithm
    edges.sort()
    
    # Union-Find data structure
    parent = list(range(n + 1))
    rank = [0] * (n + 1)
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        if rank[px] < rank[py]:
            px, py = py, px
        parent[py] = px
        if rank[px] == rank[py]:
            rank[px] += 1
        return True
    
    # We need to connect all critical nodes with minimum cost
    # This is essentially finding the Minimum Spanning Tree of the subgraph induced by critical nodes
    # But since we're given a tree (m = n-1), we can't build a new MST directly
    # Instead, we use the fact that we want to connect all critical nodes in the existing tree
    # with minimal edge weight sum. This is equivalent to finding the Steiner tree for the critical nodes
    # in the given tree. However, computing the exact Steiner tree is complex.
    # But since the graph is a tree, we can do the following:
    # 1. Build a virtual graph where nodes are critical facilities
    # 2. For each pair of critical nodes, compute their distance in the original tree
    # 3. Find the MST of this complete graph on critical nodes
    
    # First, build adjacency list of the original tree
    adj = [[] for _ in range(n + 1)]
    for w, u, v in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Function to compute distances from a node using BFS/DFS
    def bfs_distances(start):
        dist = [-1] * (n + 1)
        dist[start] = 0
        stack = [start]
        while stack:
            u = stack.pop()
            for v, w in adj[u]:
                if dist[v] == -1:
                    dist[v] = dist[u] + w
                    stack.append(v)
        return dist
    
    # Precompute distances from each critical node
    critical_list = list(critical)
    distances = {}
    for node in critical_list:
        distances[node] = bfs_distances(node)
    
    # Create a complete graph on critical nodes with edge weights as distances in the tree
    virtual_edges = []
    for i in range(len(critical_list)):
        for j in range(i+1, len(critical_list)):
            u = critical_list[i]
            v = critical_list[j]
            w = distances[u][v]
            virtual_edges.append((w, u, v))
    
    # Now run Kruskal's algorithm on the virtual edges to get the MST
    virtual_edges.sort()
    total_cost = 0
    components = len(critical_list)
    
    for w, u, v in virtual_edges:
        if find(u) != find(v):
            union(u, v)
            total_cost += w
            components -= 1
            if components == 1:
                break
                
    print(total_cost)

if __name__ == "__main__":
    main()
```
