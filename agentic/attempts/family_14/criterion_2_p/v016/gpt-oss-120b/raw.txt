```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))  # m = n-1
    t = int(next(it))
    terminals = set()
    for _ in range(t):
        terminals.add(int(next(it)) - 1)
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    if t <= 1:
        print(0)
        return

    parent = [-1] * n
    pw = [0] * n          # weight to parent
    order = []
    stack = [0]
    parent[0] = -2  # mark visited
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if parent[v] == -1:
                parent[v] = u
                pw[v] = w
                stack.append(v)

    term_cnt = [0] * n
    for v in terminals:
        term_cnt[v] = 1

    total = 0
    # process in reverse order (post-order)
    for u in reversed(order):
        if parent[u] >= 0:
            cnt = term_cnt[u]
            if 0 < cnt < t:
                total += pw[u]
            term_cnt[parent[u]] += cnt

    print(total)

if __name__ == "__main__":
    solve()
```
