```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    it = iter(data)

    n = next(it)
    m = next(it)
    t = next(it)

    required = bytearray(n + 1)
    for _ in range(t):
        required[next(it)] = 1

    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        w = next(it)
        graph[u].append((v, w))
        graph[v].append((u, w))

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    parent[1] = -1
    order = [1]

    for u in order:
        for v, w in graph[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                order.append(v)

    count = list(required)
    answer = 0

    for v in reversed(order[1:]):
        if 0 < count[v] < t:
            answer += parent_weight[v]
        count[parent[v]] += count[v]

    print(answer)

if __name__ == "__main__":
    main()
```
