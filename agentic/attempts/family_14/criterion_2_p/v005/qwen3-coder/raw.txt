import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    
    required = list(map(int, data[idx:idx+t]))
    idx += t
    
    # If only one required site, cost is 0
    if t <= 1:
        print(0)
        return
    
    required_set = set(required)
    
    # Build the tree
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # Since it's a tree, we can use the fact that we need to find the minimal subtree
    # that connects all required nodes. This is the Steiner tree on a tree.
    # For a tree, the Steiner tree is obtained by taking the union of paths
    # between all pairs of required nodes, then removing leaves that are not required.
    
    # However, there's a simpler way: since it's a tree, the minimal subtree connecting
    # all required nodes is the union of all paths between required nodes.
    # But even simpler: we can do a traversal to find which edges are needed.
    
    # Better approach: Root the tree at any node, compute for each node
    # whether its subtree contains any required node. Then an edge is needed
    # if it connects two subtrees both containing required nodes.
    
    # Let's root the tree at node 1
    root = 1
    # Build adjacency list for tree traversal
    tree = defaultdict(list)
    for u, v, w in edges:
        tree[u].append((v, w))
        tree[v].append((u, w))
    
    # First, let's make sure the graph is connected and is a tree
    # We'll perform BFS to build parent relationships and depths
    parent = {root: None}
    depth = {root: 0}
    weight_to_parent = {}
    queue = deque([root])
    
    while queue:
        u = queue.popleft()
        for v, w in tree[u]:
            if v not in parent:
                parent[v] = u
                weight_to_parent[v] = w
                depth[v] = depth[u] + 1
                queue.append(v)
    
    # Now, for each node, we want to know if its subtree contains a required node
    # We'll do a post-order traversal (DFS) to compute this
    visited = set()
    has_required = {node: False for node in parent}
    
    # Get all nodes
    all_nodes = list(parent.keys())
    
    # Sort nodes by depth descending for post-order processing
    all_nodes.sort(key=lambda x: -depth[x])
    
    # Mark required nodes
    for r in required:
        if r in has_required:
            has_required[r] = True
    
    # Propagate required status upwards
    for node in all_nodes:
        if node != root:
            p = parent[node]
            if has_required[node]:
                has_required[p] = True
    
    # Now count the cost: include edge (u, parent[u]) if both u and its subtree
    # have required nodes (but since it's a tree, if u has required in subtree,
    # and we're going up, we just need to check if subtree of u has >=2 required nodes
    # intersecting this edge. Actually, simpler:
    # An edge (u,v) where u is parent of v is included if v's subtree contains
    # at least one required node and the rest of the tree contains at least one required node.
    # But in a tree, it's easier:
    # Edge (p, u) is needed if u's subtree contains at least one required node
    # and there's at least one required node outside u's subtree.
    # Total required nodes = t
    # Nodes in u's subtree that are required = count_in_subtree[u]
    # So edge to parent is needed if 0 < count_in_subtree[u] < t
    
    # Let's compute count_in_subtree for each node via DFS
    # We'll use post-order again
    count_in_subtree = {node: 0 for node in parent}
    
    # Initialize with required nodes
    for r in required:
        count_in_subtree[r] = 1
        
    # Propagate counts upward
    for node in all_nodes:
        if node != root:
            p = parent[node]
            count_in_subtree[p] += count_in_subtree[node]
            
    # Calculate total cost
    total_cost = 0
    for node in parent:
        if node != root:
            p = parent[node]
            # Edge from 'node' to 'p' (parent)
            # It's needed if 0 < count_in_subtree[node] < t
            if 0 < count_in_subtree[node] < t:
                total_cost += weight_to_parent[node]
                
    print(total_cost)

if __name__ == "__main__":
    main()
