We can solve this by finding the minimal subtree that connects all the required offices.  
Since the graph is a tree, the minimal connected subgraph containing all required nodes is the union of paths between them, which is essentially the **Steiner tree** in a tree — and that’s just the minimal subtree spanning all required nodes.

We can do this efficiently:

1. Mark the required nodes.
2. Do a DFS to count how many required nodes are in each subtree.
3. Any edge whose both sides contain at least one required node must be included in the minimal subtree.
4. Sum the weights of such edges.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    required = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to count required nodes in each subtree
    visited = [False] * (n + 1)
    count = [0] * (n + 1)
    
    def dfs(u):
        visited[u] = True
        cnt = 1 if u in required else 0
        for v, w in adj[u]:
            if not visited[v]:
                cnt += dfs(v)
        count[u] = cnt
        return cnt
    
    # Start DFS from any required node (or 1 if none)
    start = next(iter(required)) if required else 1
    dfs(start)
    
    total = 0
    for u, v, w in edges:
        # Edge is needed if both sides have at least one required node
        if count[u] > 0 and count[v] > 0:
            # But careful: count is from root perspective
            # If one side has all required nodes, the other side has 0
            # So we check if both sides have >0
            # Since it's a tree, one of count[u] or count[v] is the subtree size
            # The other is t - that size
            if count[u] != 0 and count[u] != t:
                total += w
            elif count[v] != 0 and count[v] != t:
                total += w
    
    print(total)

if __name__ == "__main__":
    solve()
```
