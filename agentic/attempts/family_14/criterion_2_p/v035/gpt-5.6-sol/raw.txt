```python
import sys
from array import array
from collections import deque

data = sys.stdin.buffer.read()

def integers():
    value = 0
    reading = False
    for c in data:
        if 48 <= c <= 57:
            value = value * 10 + c - 48
            reading = True
        elif reading:
            yield value
            value = 0
            reading = False
    if reading:
        yield value

it = iter(integers())
n = next(it)
m = next(it)
t = next(it)

required = bytearray(n)
for _ in range(t):
    required[next(it) - 1] = 1

head = array('i', [-1]) * n
degree = array('i', [0]) * n
to = array('i')
nxt = array('i')
weight = array('i')

total = 0

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    w = next(it)

    to.append(v)
    weight.append(w)
    nxt.append(head[u])
    head[u] = len(to) - 1

    to.append(u)
    weight.append(w)
    nxt.append(head[v])
    head[v] = len(to) - 1

    degree[u] += 1
    degree[v] += 1
    total += w

alive = bytearray([1]) * n
queue = deque(
    i for i in range(n)
    if not required[i] and degree[i] <= 1
)

while queue:
    u = queue.popleft()
    if not alive[u] or required[u]:
        continue

    alive[u] = 0
    edge = head[u]

    while edge != -1:
        v = to[edge]
        if alive[v]:
            total -= weight[edge]
            degree[v] -= 1
            if not required[v] and degree[v] <= 1:
                queue.append(v)
        edge = nxt[edge]

print(total)
```
