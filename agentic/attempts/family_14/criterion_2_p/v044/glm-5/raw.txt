```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    essential = [int(data[idx + i]) for i in range(t)]
    idx += t
    
    if t <= 1:
        print(0)
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        w = int(data[idx]); idx += 1
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    LOG = 20
    parent = [[0] * (n + 1) for _ in range(LOG)]
    max_edge = [[0] * (n + 1) for _ in range(LOG)]
    depth = [0] * (n + 1)
    
    stack = [(1, 0, 0, 0)]
    visited = [False] * (n + 1)
    
    while stack:
        u, p, w, d = stack.pop()
        if visited[u]:
            continue
        visited[u] = True
        parent[0][u] = p
        max_edge[0][u] = w
        depth[u] = d
        for v, wv in adj[u]:
            if not visited[v]:
                stack.append((v, u, wv, d + 1))
    
    for k in range(1, LOG):
        for v in range(1, n + 1):
            p = parent[k - 1][v]
            if p != 0:
                parent[k][v] = parent[k - 1][p]
                max_edge[k][v] = max(max_edge[k - 1][v], max_edge[k - 1][p])
            else:
                parent[k][v] = 0
                max_edge[k][v] = max_edge[k - 1][v]
    
    def get_max_edge(u, v):
        if depth[u] < depth[v]:
            u, v = v, u
        diff = depth[u] - depth[v]
        res = 0
        k = 0
        while diff:
            if diff & 1:
                res = max(res, max_edge[k][u])
                u = parent[k][u]
            diff >>= 1
            k += 1
        
        if u == v:
            return res
        
        for k in range(LOG - 1, -1, -1):
            if parent[k][u] != parent[k][v]:
                res = max(res, max_edge[k][u], max_edge[k][v])
                u = parent[k][u]
                v = parent[k][v]
        
        res = max(res, max_edge[0][u], max_edge[0][v])
        return res
    
    total = 0
    for i in range(t - 1):
        total += get_max_edge(essential[i], essential[i + 1])
    
    print(total)

solve()
```
