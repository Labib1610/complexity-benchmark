```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))  # = n-1
    t = int(next(it))
    essential = [0] * (n + 1)
    for _ in range(t):
        essential[int(next(it))] = 1
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))

    if t <= 1:
        print(0)
        return

    parent = [0] * (n + 1)
    pw = [0] * (n + 1)          # weight to parent
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v == parent[u]:
                continue
            parent[v] = u
            pw[v] = w
            stack.append(v)

    cnt = [0] * (n + 1)
    for i in range(1, n + 1):
        cnt[i] = essential[i]

    total_needed = 0
    total_ess = t
    for u in reversed(order):
        if parent[u] != -1:
            cnt[parent[u]] += cnt[u]
            # edge u - parent[u] is needed if both sides have essential nodes
            if cnt[u] > 0 and total_ess - cnt[u] > 0:
                total_needed += pw[u]

    print(total_needed)

if __name__ == "__main__":
    solve()
```
