We can solve this by treating the tree and only considering the minimal subtree that connects all critical nodes.  
The idea is:

1. Mark all critical nodes.
2. Do a DFS to count how many critical nodes are in each subtree.
3. An edge is needed if both sides of the edge contain at least one critical node.
4. Sum the weights of all such edges.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m, t = map(int, input().split())
    critical = set(map(int, input().split()))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        adj[u].append((v, w))
        adj[v].append((u, w))
        edges.append((u, v, w))
    
    # DFS to count critical nodes in subtree
    parent = [0] * (n + 1)
    weight_to_parent = [0] * (n + 1)
    order = []
    stack = [1]
    parent[1] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                weight_to_parent[v] = w
                stack.append(v)
    
    sub_critical = [0] * (n + 1)
    for node in critical:
        sub_critical[node] = 1
    
    total_cost = 0
    for u in reversed(order):
        if u != 1:
            if sub_critical[u] > 0 and (t - sub_critical[u]) > 0:
                total_cost += weight_to_parent[u]
            if parent[u] != -1:
                sub_critical[parent[u]] += sub_critical[u]
    
    print(total_cost)

if __name__ == "__main__":
    solve()
```
