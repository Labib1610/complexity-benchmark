```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    t = int(input_data[2])
    
    critical_nodes = set(int(x) for x in input_data[3:3+t])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3 + t
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        adj[u].append((v, w))
        adj[v].append((u, w))
        idx += 3
        
    if t <= 1:
        print(0)
        return

    parent = [0] * (n + 1)
    parent_weight = [0] * (n + 1)
    depth = [0] * (n + 1)
    order = []
    
    # BFS to setup tree structure and get traversal order
    q = deque([1])
    parent[1] = -1
    while q:
        u = q.popleft()
        order.append(u)
        for v, w in adj[u]:
            if v != parent[u]:
                parent[v] = u
                parent_weight[v] = w
                depth[v] = depth[u] + 1
                q.append(v)
                
    cnt = [0] * (n + 1)
    for u in order:
        if u in critical_nodes:
            cnt[u] = 1
            
    # Process in reverse order of traversal (bottom-up)
    for i in range(len(order) - 1, 0, -1):
        u = order[i]
        p = parent[u]
        cnt[p] += cnt[u]
        
    total_cost = 0
    for u in range(2, n + 1):
        if 0 < cnt[u] < t:
            total_cost += parent_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    solve()
```
