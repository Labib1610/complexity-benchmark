```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    t = int(data[2])
    
    if t <= 1:
        print(0)
        return
    
    idx = 3
    critical_pods = [int(x) for x in data[idx:idx+t]]
    idx += t
    
    is_critical = [False] * (n + 1)
    for pod in critical_pods:
        is_critical[pod] = True
        
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        adj[u].append((v, w))
        adj[v].append((u, w))
        
    root = critical_pods[0]
    
    # Non-recursive DFS to compute critical counts in subtrees
    parent = [0] * (n + 1)
    parent_edge_weight = [0] * (n + 1)
    order = []
    stack = [root]
    visited = [False] * (n + 1)
    visited[root] = True
    
    while stack:
        u = stack.pop()
        order.append(u)
        for v, w in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge_weight[v] = w
                stack.append(v)
                
    crit_count = [0] * (n + 1)
    for u in order:
        if is_critical[u]:
            crit_count[u] = 1
            
    total_cost = 0
    # Process in reverse topological (post-order) order
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        crit_count[p] += crit_count[u]
        if crit_count[u] > 0:
            total_cost += parent_edge_weight[u]
            
    print(total_cost)

if __name__ == '__main__':
    main()
```
