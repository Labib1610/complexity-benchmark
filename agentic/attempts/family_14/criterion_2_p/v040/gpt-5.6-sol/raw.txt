```python
import sys
from array import array

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
terminal_count = next(it)

is_terminal = bytearray(n + 1)
for _ in range(terminal_count):
    is_terminal[next(it)] = 1

head = array('i', [-1]) * (n + 1)
to = array('i')
weight = array('i')
nxt = array('i')

for _ in range(m):
    u = next(it)
    v = next(it)
    w = next(it)

    to.append(v)
    weight.append(w)
    nxt.append(head[u])
    head[u] = len(to) - 1

    to.append(u)
    weight.append(w)
    nxt.append(head[v])
    head[v] = len(to) - 1

if terminal_count <= 1:
    print(0)
    sys.exit()

parent = array('i', [0]) * (n + 1)
parent_weight = array('i', [0]) * (n + 1)
order = array('i')

root = 1
parent[root] = -1
stack = [root]

while stack:
    u = stack.pop()
    order.append(u)
    edge = head[u]

    while edge != -1:
        v = to[edge]
        if v != parent[u]:
            parent[v] = u
            parent_weight[v] = weight[edge]
            stack.append(v)
        edge = nxt[edge]

subtree_terminals = array('i', [0]) * (n + 1)
for u in range(1, n + 1):
    subtree_terminals[u] = is_terminal[u]

answer = 0

for i in range(len(order) - 1, 0, -1):
    u = order[i]
    count = subtree_terminals[u]

    if 0 < count < terminal_count:
        answer += parent_weight[u]

    subtree_terminals[parent[u]] += count

print(answer)
```
