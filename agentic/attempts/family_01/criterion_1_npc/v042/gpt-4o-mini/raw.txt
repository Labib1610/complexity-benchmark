def museum_vault_assignment():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    incompatible_pairs = [tuple(map(int, line.split())) for line in data[1:m + 1]]
    
    graph = defaultdict(list)
    for u, v in incompatible_pairs:
        graph[u - 1].append(v - 1)
        graph[v - 1].append(u - 1)
    
    color = [-1] * n
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0  # Start coloring with 0
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == -1:  # Not colored yet
                    color[neighbor] = (color[node] + 1) % 3
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:  # Conflict in coloring
                    return False
        return True
    
    for i in range(n):
        if color[i] == -1:  # Not visited
            if not bfs(i):
                print("NO")
                return
    
    # Adjust colors to be 1-indexed and lexicographically smallest
    result = [c + 1 for c in color]
    print("YES")
    print(" ".join(map(str, result)))

museum_vault_assignment()
