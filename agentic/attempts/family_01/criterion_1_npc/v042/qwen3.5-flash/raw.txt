import sys

# Increase recursion depth

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    # Precompute neighbors < u and > u
    # This helps in the DFS to avoid checking indices repeatedly
    adj_lt = [[] for _ in range(n + 1)]
    adj_gt = [[] for _ in range(n + 1)]
    
    for u in range(1, n + 1):
        for v in adj[u]:
            if v < u:
                adj_lt[u].append(v)
            else:
                adj_gt[u].append(v)

    # colors[i] stores the vault assigned to artifact i (0 if unassigned)
    # 1-based indexing for artifacts 1..n
    colors = [0] * (n + 1)
    
    # available[i] is a bitmask of available colors for artifact i
    # 1 -> bit 0 (val 1), 2 -> bit 1 (val 2), 3 -> bit 2 (val 4)
    # Initially all 1, 2, 3 available -> mask 7
    available = [7] * (n + 1)
    
    # DFS function
    def dfs(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            # 1. Check compatibility with already assigned neighbors (v < u)
            # We use adj_lt[u] which contains only neighbors < u
            conflict = False
            for v in adj_lt[u]:
                if colors[v] == c:
                    conflict = True
                    break
            if conflict:
                continue
            
            # 2. Check consistency with future neighbors (v > u) - Forward Checking
            # We use adj_gt[u] which contains neighbors > u
            future_conflict = False
            for v in adj_gt[u]:
                # Check if color c is available for v
                if (available[v] & (1 << (c - 1))):
                    # If it is available, check if it's the ONLY available color
                    if (available[v] & ~(1 << (c - 1))) == 0:
                        future_conflict = True
                        break
            
            if future_conflict:
                continue
            
            # If valid, proceed

            for v in adj_gt[u]:
                available[v] &= ~(1 << (c - 1))

            if dfs(u + 1):
                return True
            
            # Backtrack
            colors[u] = 0
            for v in adj_gt[u]:
                available[v] |= (1 << (c - 1))
        
        return False
