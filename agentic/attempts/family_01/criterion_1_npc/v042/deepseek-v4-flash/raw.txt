 We need to check if the graph is 3-colorable, and if so, output a valid 3-coloring.  
Since \( n \leq 40 \), we can use backtracking with pruning.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n  # 0 = uncolored, 1,2,3 = rings

    # Sort vertices by degree (largest first) for better pruning
    order = sorted(range(n), key=lambda x: -len(adj[x]))
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # Build adjacency in new order
    adj_ordered = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            adj_ordered[pos[u]].append(pos[v])

    def backtrack(idx):
        if idx == n:
            return True
        used = [False] * 4
        for nb in adj_ordered[idx]:
            if color[nb] != 0:
                used[color[nb]] = True
        for c in range(1, 4):
            if not used[c]:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False

    if backtrack(0):
        res = [0] * n
        for i in range(n):
            res[order[i]] = color[i]
        print("YES")
        print(" ".join(map(str, res)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
