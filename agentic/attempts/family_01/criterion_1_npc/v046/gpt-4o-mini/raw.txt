def solve():
    import sys
    from collections import defaultdict

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    conflicts = defaultdict(set)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        conflicts[u].add(v)
        conflicts[v].add(u)
    
    # To hold the shift assignments
    shifts = [0] * (n + 1)  # Using 1-based indexing
    
    def can_assign(member):
        if member > n:
            return True
        if shifts[member] != 0:
            return can_assign(member + 1)
        
        for shift in range(1, 4):  # Shift options are 1, 2, 3
            valid = True
            for neighbor in conflicts[member]:
                if shifts[neighbor] == shift:
                    valid = False
                    break
            if valid:
                shifts[member] = shift
                if can_assign(member + 1):
                    return True
                shifts[member] = 0
        
        return False

    if can_assign(1):
        print("YES")
        print(" ".join(map(str, shifts[1:])))
    else:
        print("NO")
