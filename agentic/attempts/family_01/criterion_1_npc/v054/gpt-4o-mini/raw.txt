def solve():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    
    # Create adjacency list for the graph
    graph = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)
    
    # This will hold the band assigned to each stream
    bands = [0] * (n + 1)  # 1-indexed
    
    def bfs(start):
        queue = deque([start])
        bands[start] = 1  # Start with band 1
        
        while queue:
            u = queue.popleft()
            current_band = bands[u]
            for v in graph[u]:
                if bands[v] == 0:  # Not assigned yet
                    # Assign a band that is different from u's band
                    for band in range(1, 4):
                        if band != current_band:
                            bands[v] = band
                            break
                    queue.append(v)
                elif bands[v] == current_band:  # Conflict detected
                    return False
        return True
    
    # Check for each component in the graph
    for i in range(1, n + 1):
        if bands[i] == 0:  # Not assigned yet
            if not bfs(i):
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, bands[1:n + 1])))
