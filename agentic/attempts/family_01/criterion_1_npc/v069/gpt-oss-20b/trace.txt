def dfs(idx):
    if idx==n:
        return True
    for c in [1,2,3]:
        conflict = False
        for j in adj[idx]:
            if colors[j]==c:
                conflict=True
                break
        if conflict: continue
        colors[idx]=c
        if dfs(idx+1):
            return True
        colors[idx]=0
    return False
