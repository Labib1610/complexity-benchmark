 We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use backtracking with pruning or a meet-in-the-middle approach.  
A simple backtracking that tries assigning portfolios 1, 2, 3 to each vertex while checking constraints will work for \( n=40 \) if we prune well.

Here’s the Python solution:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    color = [-1] * n
    # Sort vertices by degree descending for better pruning
    order = list(range(n))
    order.sort(key=lambda x: bin(adj[x]).count('1'), reverse=True)

    def backtrack(idx):
        if idx == n:
            return True
        v = order[idx]
        for c in range(3):
            # Check if any neighbor already has color c
            ok = True
            for u in range(n):
                if (adj[v] >> u) & 1 and color[u] == c:
                    ok = False
                    break
            if ok:
                color[v] = c
                if backtrack(idx + 1):
                    return True
                color[v] = -1
        return False

    if backtrack(0):
        print("YES")
        res = [0] * n
        for i in range(n):
            res[i] = color[i] + 1
        print(*res)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
