```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n  # bitmask adjacency
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [-1] * n  # 0,1,2

    # For fast conflict check, maintain for each color a bitmask of vertices colored with it
    color_mask = [0, 0, 0]

    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        forbidden = 0
        # compute which colors are forbidden by neighbors already colored
        for c in range(3):
            if adj[v] & color_mask[c]:
                forbidden |= 1 << c
        for c in range(3):
            if not (forbidden >> c) & 1:
                colors[v] = c
                color_mask[c] |= 1 << v
                if dfs(idx + 1):
                    return True
                color_mask[c] ^= 1 << v
                colors[v] = -1
        return False

    possible = dfs(0)
    out = []
    if not possible:
        out.append("NO")
    else:
        out.append("YES")
        out.append(' '.join(str(c + 1) for c in colors))
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
