import sys

def solve():

    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    for _ in range(m):

        u -= 1
        v -= 1

        adj[v] |= (1 << u)

    # If we pick s_i = k, we must ensure that there exists a valid 3-coloring 
    # for the remaining nodes {i+1, ..., n-1} given the choices already made.

        Checks if the remaining nodes [idx, n-1] can be colored with 3 colors
        given the constraints and existing assignments.

        # To optimize, we can use a simple backtracking with pruning or 
        # a more robust 3-coloring check. Since n is small (40), 
        # we need a reasonably efficient check.
        
        # Heuristic: Try to color the most constrained nodes first.
        # However, for the 'existence' check, we just need any valid coloring.
        
        # Since we are calling this frequently, we use a recursive check with a limit
        # or a specific 3-coloring algorithm.
        # Given n=40, 3-coloring is NP-complete, but the graph might be sparse or 
        # the search space limited by existing assignments.
        
        memo = {}

        def backtrack(u, colors):
            state = (u, tuple(colors))
            if state in memo: return memo[state]
            
            if u == n:
                return True
            
            # Try colors 1, 2, 3
            for c in (1, 2, 3):
                # Check if color c is valid for node u
                valid = True
                # Only check neighbors that are already colored
                # Since we process in order, neighbors < u are already colored
                # But neighbors > u are not.
                # Wait, we need to check all neighbors of u that have been assigned.
                for v in range(n):

                        valid = False
                        break
                
                if valid:
                    colors[u] = c

                        return True
                    colors[u] = 0
            
            memo[state] = False
            return False

        # The above backtrack is too slow. Let's use a more efficient approach.
        # We only need to check if a valid coloring exists for the remaining.
        return False

    # Because n=40 is small but 3-coloring is hard, 
    # we use backtracking with a simple pruning and a limit.
    
    final_assignment = [0] * n
    
    def is_possible(node_idx):
        # Check if nodes [node_idx, n-1] can be colored
        # given assignments for [0, node_idx-1]
        temp_colors = list(final_assignment)
        
        def check(u):
            if u == n:
                return True
            
            # Try colors 1, 2, 3
            for c in (1, 2, 3):
                # Check conflicts with all neighbors
                conflict = False
                # Check neighbors < u (already assigned)
                # and neighbors >= u (will be assigned)
                # Actually, just check if any neighbor already has color c
                # We only care about neighbors that are already assigned.
                # Since we are filling from u to n-1, we check neighbors < u

                # Efficient conflict check using bitmasks
                # This is tricky because colors are not bits.

                for v in range(n):

                        conflict = True
                        break

                    temp_colors[u] = c
                    if check(u + 1):

                    temp_colors[u] = 0
            return False

        return check(node_idx)

    # Given the constraints and the "lexicographically smallest" requirement,
    # we must try colors 1, 2, 3 for each node in order.
    
    def solve_recursive(u):
        if u == n:
            return True
        
        for c in (1, 2, 3):

            conflict = False
            for v in range(u):

                final_assignment[u] = c

                # To avoid TLE, we can't do a full 3-coloring check every time.

                    if backtrack_fast(curr + 1):

                    local_colors[curr] = 0
            return False
