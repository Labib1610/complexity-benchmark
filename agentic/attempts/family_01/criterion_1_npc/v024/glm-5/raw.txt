import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                if not dfs(v, 3 - c):
                    return False
            elif color[v] == c:
                return False
        return True
    
    for i in range(n):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
    
    # Group vertices by color
    g1 = [i for i in range(n) if color[i] == 1]
    g2 = [i for i in range(n) if color[i] == 2]
    
    # Check if g1 is an independent set
    for u in g1:
        for v in adj[u]:
            if color[v] == 1:
                print("NO")
                return
    
    # Check if g2 is an independent set
    for u in g2:
        for v in adj[u]:
            if color[v] == 2:
                print("NO")
                return
    
    # Assign colors: g1 -> 1, g2 -> 2, remaining -> 3
    ans = [0] * n
    for u in g1:
        ans[u] = 1
    for u in g2:
        ans[u] = 2
    for u in range(n):
        if ans[u] == 0:
            ans[u] = 3
    
    print("YES")
    print(" ".join(map(str, ans)))

solve()
