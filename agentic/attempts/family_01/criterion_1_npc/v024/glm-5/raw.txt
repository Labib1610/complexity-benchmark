import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Precompute connected components
    comp_id = [-1] * n
    comps = []
    for i in range(n):
        if comp_id[i] == -1:
            comp = []
            stack = [i]
            comp_id[i] = len(comps)
            while stack:
                u = stack.pop()
                comp.append(u)
                bits = adj[u]
                while bits:
                    b = bits & -bits
                    v = b.bit_length() - 1
                    if comp_id[v] == -1:
                        comp_id[v] = len(comps)
                        stack.append(v)
                    bits ^= b
            comps.append(comp)

    # For each component, find all valid 3-colorings
    comp_colorings = []
    for comp in comps:
        k = len(comp)
        if k > 3:
            print("NO")
            return
        colorings = []
        # Enumerate all 3^k colorings
        for mask in range(3**k):
            valid = True
            for i in range(k):
                ci = (mask // (3**i)) % 3
                bits = adj[comp[i]]
                for j in range(i+1, k):
                    if bits & (1 << comp[j]):
                        cj = (mask / (3**j)) % 3
                        if ci == cj:
                            valid = False
                            break
                if not valid:
                    break
            if valid:
                coloring = [0] * n
                for i in range(k):
                    coloring[comp[i]] = (mask // (3**i)) % 3
                colorings.append(coloring)
        if not colorings:
            print("NO")
            return
        comp_colorings.append(colorings)

    # Greedy assignment with backtracking
    assignment = [-1] * n
    num_comps = len(comps)

    def backtrack(comp_idx):
        if comp_idx == num_comps:
            return True
        for coloring in comp_colorings[comp_idx]:
            ok = True
            for v in comps[comp_idx]:
                if coloring[v] < assignment[v]:
                    ok = False
                    break
                if coloring[v] > assignment[v]:
                    break
            if not ok:
                continue
            saved = []
            for v in comps[comp_idx]:
                if assignment[v] == -1:
                    assignment[v] = coloring[v]
                    saved.append(v)
            if backtrack(comp_idx + 1):
                return True
            for v in saved:
                assignment[v] = -1
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(c + 1) for c in assignment))
    else:
        print("NO")

solve()
