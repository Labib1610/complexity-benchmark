We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use a backtracking or meet-in-the-middle approach.  
A simple backtracking with pruning works well here.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    color = [0] * n
    # Precompute possible colors for each node
    possible = [0] * n
    for i in range(n):
        possible[i] = 0b111  # all 3 colors possible initially
    
    # Backtracking
    def backtrack(idx):
        if idx == n:
            return True
        # Try colors 1,2,3
        for c in range(1, 4):
            ok = True
            for j in range(n):
                if (adj[idx] >> j) & 1:
                    if color[j] == c:
                        ok = False
                        break
            if ok:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(str(c) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
