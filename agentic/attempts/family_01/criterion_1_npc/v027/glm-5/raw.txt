```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Greedy assignment for lexicographically smallest valid sequence
    bands = [0] * n
    possible = True
    
    for i in range(n):
        # Try bands in order 1, 2, 3
        for b in [1, 2, 3]:
            valid = True
            for neighbor in adj[i]:
                if neighbor < i and bands[neighbor] == b:
                    valid = False
                    break
            if valid:
                bands[i] = b
                break
        if bands[i] == 0:
            possible = False
            break
    
    if possible:
        print("YES")
        print(' '.join(map(str, bands)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
