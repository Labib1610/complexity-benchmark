import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    color = [0] * (n + 1)

    def dfs(u):
        for c in (1, 2, 3):
            if adj[u] & color[u] == 0:
                color[u] |= c
                ok = True
                v = u + 1
                while v <= n and ok:
                    if adj[u] & (1 << v):
                        if color[v] & c:
                            ok = False
                        else:
                            color[v] |= c
                    v += 1
                if ok:
                    if u == n:
                        return True
                    if dfs(u + 1):
                        return True
                v = u + 1
                while v <= n:
                    if adj[u] & (1 << v):
                        color[v] &= ~c
                    v += 1
                color[u] &= ~c
        return False

    if dfs(1):
        out = ["YES"]
        out.append(" ".join(str(bin(color[i]).count("1")) for i in range(1, n + 1)))
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

solve()
