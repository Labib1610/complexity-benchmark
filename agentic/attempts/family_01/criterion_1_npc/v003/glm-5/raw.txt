import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Precompute connected components
    visited = [False] * n
    components = []
    
    for i in range(n):
        if not visited[i]:
            comp_nodes = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp_nodes.append(u)
                for v in range(n):
                    if (adj[u] >> v) & 1 and not visited[v]:
                        visited[v] = True
                        stack.append(v)
            components.append(comp_nodes)
    
    # Sort components by minimum vertex index to ensure lexicographic order
    components.sort(key=lambda x: min(x))
    
    # Map: original vertex -> index in component
    node_to_comp_idx = {}
    for c_idx, comp in enumerate(components):
        for v_idx, v in enumerate(comp):
            node_to_comp_idx[v] = (c_idx, v_idx)
    
    # Precompute adjacency masks for each component
    comp_adj_masks = []
    for comp in components:
        size = len(comp)
        masks = [0] * size
        for i, u in enumerate(comp):
            mask = 0
            for j, v in enumerate(comp):
                if (adj[u] >> v) & 1:
                    mask |= 1 << j
            masks[i] = mask
        comp_adj_masks.append(masks)
    
    # DP for each component
    # dp[c] = set of valid colorings (bitmask) for component c
    dp = []
    
    for c_idx in range(len(components)):
        size = len(components[c_idx])
        masks = comp_adj_masks[c_idx]
        
        # Check if component is bipartite
        color = [-1] * size
        is_bipartite = True
        for start in range(size):
            if color[start] == -1:
                color[start] = 0
                stack = [start]
                while stack:
                    u = stack.pop()
                    for v in range(size):
                        if (masks[u] >> v) & 1:
                            if color[v] == -1:
                                color[v] = 1 - color[u]
                                stack.append(v)
                            elif color[v] == color[u]:
                                is_bipartite = False
                                break
                    if not is_bipartite:
                        break
            if not is_bipartite:
                break
        
        valid = set()
        
        if is_bipartite:
            # Generate all valid 3-colorings from bipartition
            # Partitions A and B
            part0 = [i for i in range(size) if color[i] == 0]
            part1 = [i for i in range(size) if color[i] == 1]
            
            # Colors for part0 and part1
            for c0 in [1, 2, 3]:
                for c1 in [1, 2, 3]:
                    if c0 == c1:
                        continue
                    mask = 0
                    for i in part0:
                        mask |= (c0 << (2 * i))
                    for i in part1:
                        mask |= (c1 << (2 * i))
                    valid.add(mask)
        else:
            # Not bipartite: try all 3^size colorings
            # Pruning: check validity incrementally
            # We use recursive generation with pruning
            def generate_colorings(pos, current_mask):
                if pos == size:
                    valid.add(current_mask)
                    return
                # Determine which colors are forbidden
                forbidden = 0
                for prev in range(pos):
                    if (masks[pos] >> prev) & 1:
                        c = (current_mask >> (2 * prev)) & 3
                        forbidden |= c
                for c in [1, 2, 3]:
                    if c != forbidden:
                        generate_colorings(pos + 1, current_mask | (c << (2 * pos)))
            
            generate_colorings(0, 0)
        
        if not valid:
            print("NO")
            return
        
        dp.append(valid)
    
    # Merge components
    # We maintain a dictionary: forbidden_mask -> best_coloring
    # forbidden_mask: bitmask of forbidden colors for each vertex (2 bits per vertex)
    # Initially, all colors allowed (forbidden_mask = 0)
    
    # To save memory, we process components one by one
    # and keep only the current state
    
    # forbidden_mask: for each vertex v, 2 bits indicating forbidden colors
    # We use a dictionary: forbidden_mask -> best_coloring
    
    # However, forbidden_mask can be large (2n bits)
    # Instead, we track the coloring directly and check validity
    
    # We use a DP where state is the partial coloring
    # and we extend it with each component
    
    # To ensure lexicographic order, we process components in order
    # and for each component, we try colorings in lexicographic order
    
    # But we need to check compatibility with previous components
    
    # Let's use a different approach:
    # We maintain a dictionary: partial_coloring -> True
    # and extend it with each component
    
    # To save space, we only keep the best partial coloring for each "signature"
    # But the signature is the partial coloring itself
    
    # Since n <= 40, we can represent coloring as a 64-bit integer (2 bits per vertex)
    
    # We process components in order
    # For each component, we try all valid colorings
    # and check compatibility with the current best partial coloring
    
    # We maintain the current best partial coloring (as a bitmask)
    # Initially, it's empty (we haven't assigned any vertex)
    
    # But we need to consider all possible partial colorings
    # because a coloring that is lexicographically smaller now might become invalid later
    
    # So we need to keep all valid partial colorings
    
    # However, the number of valid colorings can be exponential
    # But with n <= 40 and 3 colors, it's manageable
    
    # We use a set to store all valid partial colorings
    # and update it for each component
    
    # To ensure we get the lexicographically smallest, we sort colorings at the end
    
    # But we can prune: for each partial coloring, we only need to keep the lexicographically smallest one
    # that has the same "forbidden pattern" for future vertices
    
    # The forbidden pattern for future vertices is determined by the colors of their neighbors in previous components
    
    # So we can use a dictionary: forbidden_mask_for_remaining -> best_partial_coloring
    
    # forbidden_mask_for_remaining: for each remaining vertex, the forbidden colors
    
    # Let's implement this
    
    # We process components in order
    # For each component, we compute the forbidden colors for its vertices
    # based on the current partial coloring
    
    # We maintain a dictionary: state -> best_coloring
    # state: forbidden colors for all unassigned vertices (including those in future components)
    
    # Initially, state is all zeros (no forbidden colors)
    # best_coloring is 0 (no vertices assigned)
    
    # For each component, we update the dictionary
    
    # To represent state efficiently, we use a bitmask: 2 bits per vertex
    # bit 2v and 2v+1: forbidden colors for vertex v
    # We use a single 64-bit integer
    
    # But we only need to track forbidden colors for unassigned vertices
    # After assigning a component, we update the forbidden colors for remaining vertices
    
    # Let's implement this
    
    # We'll use a dictionary: state -> best_coloring
    # state: forbidden colors for all vertices (initially 0)
    # best_coloring: the coloring of assigned vertices (initially 0)
    
    # For each component, we generate new states
    
    # However, this might be too slow
    
    # Alternative: since components are independent, we can just concatenate their colorings
    # But we need to ensure no conflicts between components
    
    # Wait, there are no edges between components, so no conflicts!
    # So we can independently choose the lexicographically smallest coloring for each component
    
    # But we need the overall sequence to be lexicographically smallest
    # Since components are processed in order of their minimum vertex,
    # we just need the lexicographically smallest coloring for each component
    
    # Let's verify: if we have components C1, C2, ... with min vertices v1 < v2 < ...
    # Then the coloring of C1 affects vertices v1, ... which are the smallest
    # So we should choose the lexicographically smallest coloring for C1
    # Then for C2, we choose the lexicographically smallest coloring, etc.
    
    # This works because there are no edges between components!
    
    # So we just need to find the lexicographically smallest valid coloring for each component
    
    # Let's implement this
    
    result = [0] * n
    
    for c_idx in range(len(components)):
        comp = components[c_idx]
        size = len(comp)
        valid_colorings = dp[c_idx]
        
        # Find the lexicographically smallest coloring
        # We need to compare colorings based on the original vertex indices
        
        # We sort the vertices in the component by their original index
        sorted_vertices = sorted(comp)
        
        # We try all valid colorings and find the lexicographically smallest
        best_coloring = None
        best_mask = None
        
        for mask in valid_colorings:
            # Extract colors for vertices in sorted order
            current = []
            for v in sorted_vertices:
                # Find position of v in component
                pos = comp.index(v)
                c = (mask >> (2 * pos)) & 3
                current.append(c)
            
            if best_coloring is None or current < best_coloring:
                best_coloring = current
                best_mask = mask
        
        # Assign the best coloring to result
        for pos, v in enumerate(comp):
            c = (best_mask >> (2 * pos)) & 3
            result[v] = c
    
    print("YES")
    print(' '.join(map(str, result)))

solve()
