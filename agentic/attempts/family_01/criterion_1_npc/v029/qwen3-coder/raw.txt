def solve():
    import sys
    import itertools
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try all possible colorings with 3 colors
    for coloring in itertools.product([1, 2, 3], repeat=n):
        valid = True
        # Check if this coloring is valid
        for u, v in edges:
            if coloring[u-1] == coloring[v-1]:
                valid = False
                break
        
        if valid:
            print("YES")
            print(" ".join(map(str, coloring)))
            return
    
    print("NO")

solve()
