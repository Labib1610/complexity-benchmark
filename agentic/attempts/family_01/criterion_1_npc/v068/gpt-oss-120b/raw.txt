```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    # order vertices by descending degree for better pruning
    order = sorted(range(n), key=lambda x: -len(adj[x]))
    # map from position in order to vertex id and reverse
    pos2v = order
    v2pos = [0] * n
    for p, v in enumerate(pos2v):
        v2pos[v] = p

    colors = [0] * n  # 0 = uncolored, else 1..3

    # adjacency in terms of positions for quick check
    adj_pos = [set(v2pos[nb] for nb in adj[v]) for v in range(n)]

    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = pos2v[idx]
        # try colors 1..3
        for c in (1, 2, 3):
            conflict = False
            for nb_pos in adj_pos[v]:
                if nb_pos < idx and colors[pos2v[nb_pos]] == c:
                    conflict = True
                    break
            if not conflict:
                colors[v] = c
                if dfs(idx + 1):
                    return True
                colors[v] = 0
        return False

    if dfs(0):
        out = ["YES", " ".join(str(colors[i]) for i in range(n))]
    else:
        out = ["NO"]
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
