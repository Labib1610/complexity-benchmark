import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # We use backtracking with constraint propagation.
    # colors[i] = 0 (unassigned), 1, 2, or 3.
    colors = [0] * n
    
    # To find the lexicographically smallest solution, we try colors 1, 2, 3 in order.
    # We maintain a list of available colors for each node.
    # available[i] is a bitmask of allowed colors.
    # Bit 0: color 1, Bit 1: color 2, Bit 2: color 3.
    # Initially all 3 colors are available (mask 0b111 = 7).
    available = [7] * n
    
    # We use a stack to manage assignments for backtracking.
    # The stack stores tuples: (index, changed_list)
    # where changed_list is a list of (node, old_mask) that were modified by the assignment.
    stack = []
    
    # Find the node with the smallest index that is currently unassigned.
    # Returns -1 if all nodes are assigned.
    def get_unassigned_node():
        for i in range(n):
            if colors[i] == 0:
                return i
        return -1
    
    # Forward checking function.
    # When we assign 'color' to 'node', we propagate the constraint to neighbors.
    # If a neighbor's domain becomes empty, we return False (failure).
    # Otherwise, we return True.
    def propagate(node, color, changed):
        bit_to_remove = 1 << (color - 1)
        for neighbor in adj[node]:
            if colors[neighbor] == 0:
                if available[neighbor] & bit_to_remove:
                    # This neighbor can no longer use 'color'.
                    # Save old state for backtracking.
                    changed.append((neighbor, available[neighbor]))
                    available[neighbor] &= ~bit_to_remove
                    if available[neighbor] == 0:
                        return False
        return True
    
    while True:
        u = get_unassigned_node()
        
        if u == -1:
            # All nodes are assigned. Solution found.
            print("YES")
            print(*(colors))
            return
        
        # Try to assign a color to node u.
        # We try colors 1, 2, 3 in order to ensure lexicographically smallest result.
        found_assignment = False
        for c in range(1, 4):
            bit = 1 << (c - 1)
            if available[u] & bit:
                # This color is available. Try it.
                changed = []
                
                # Propagate constraint to neighbors.
                if propagate(u, c, changed):
                    # Propagation successful.
                    colors[u] = c
                    stack.append((u, changed))
                    found_assignment = True
                    break
                else:
                    # Propagation failed, this color leads to a dead end.
                    # Restore the state of neighbors.
                    for neighbor, old_mask in changed:
                        available[neighbor] = old_mask
        
        if not found_assignment:
            # Backtrack.
            # We need to undo the last assignment on the stack.
            # If the stack is empty, there is no solution.
            if not stack:
                print("NO")
                return
            
            # Undo the last assignment.
            last_u, changed = stack.pop()
            
            # Restore available masks of neighbors.
            for neighbor, old_mask in changed:
                available[neighbor] = old_mask
            
            # The color we tried for last_u led to a dead end.
            # We must forbid this color for last_u in the current branch.
            last_color = colors[last_u]
            colors[last_u] = 0 # Unassign.
            
            bit_to_remove = 1 << (last_color - 1)
            available[last_u] &= ~bit_to_remove
            
            # If last_u still has available colors, we push it back to the stack
            # with an empty changed list, so the loop will try other colors for it.
            # Otherwise, we continue backtracking.
            if available[last_u] != 0:
                stack.append((last_u, []))
            else:
                # If available[last_u] is 0, we must continue backtracking.
                # We need to signal the outer loop to backtrack again.
                # We can achieve this by ensuring the next iteration of the while loop
                # fails to find an assignment for the node popped from the stack.
                # Since we just popped and found available[last_u] == 0,
                # we need to push a "failed" state or just let the loop continue.
                # Actually, the logic is:
                # We popped (last_u, changed).
                # We unassigned last_u.
                # We removed the color we tried.
                # If available[last_u] is 0, it means we have no more colors to try for last_u.
                # So we must backtrack further.
                # The while loop will continue, and get_unassigned_node will return last_u again.
                # Then the for loop will find no available colors, found_assignment will be False.
                # Then we come back here to pop from stack again.
                # This works correctly.
                pass

if __name__ == '__main__':
    solve()
