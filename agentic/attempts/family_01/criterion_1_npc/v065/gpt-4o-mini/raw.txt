def solve():
    import sys
    from collections import defaultdict
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    graph = defaultdict(set)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].add(v)
        graph[v].add(u)
    
    storage = [0] * (n + 1)
    
    def can_assign(idx):
        if idx > n:
            return True
        
        for option in range(1, 4):
            valid = True
            for neighbor in graph[idx]:
                if storage[neighbor] == option:
                    valid = False
                    break
            
            if valid:
                storage[idx] = option
                if can_assign(idx + 1):
                    return True
                storage[idx] = 0
        
        return False
    
    if can_assign(1):
        print("YES")
        print(' '.join(map(str, storage[1:])))
    else:
        print("NO")
