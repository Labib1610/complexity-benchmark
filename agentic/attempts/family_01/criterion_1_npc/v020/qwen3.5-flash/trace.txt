import sys

# Increase recursion depth just in case
sys.setrecursionlimit(5000)

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n  # 0 means uncolored, 1, 2, 3 assigned
    forbidden = [0] * n # bitmask of forbidden colors (bits 0, 1, 2)
    
    # We need lexicographically smallest.
    # So we iterate 0..n-1. For each, try colors 1, 2, 3.
    
    def backtrack(idx):
        if idx == n:
            return True
        
        # Try colors 1, 2, 3 (mapped to 0, 1, 2 for bitmask)
        for c in range(3):
            # Check if color c is allowed for current vertex idx
            # Forbidden means (forbidden[idx] >> c) & 1 is True
            if (forbidden[idx] >> c) & 1:
                continue
            
            # Assign color c+1 to idx
            colors[idx] = c + 1
            
            # Update neighbors
            changed = []
            conflict = False
            
            # We only need to update forbidden for uncolored neighbors
            # because colored neighbors are already satisfied (checked before)
            # and we don't care about their forbidden status anymore.
            for v in adj[idx]:
                if colors[v] == 0:
                    old_val = forbidden[v]
                    forbidden[v] |= (1 << c)
                    changed.append((v, old_val))
                    # Pruning: if any uncolored neighbor has all 3 colors forbidden
                    if forbidden[v] == 7:
                        conflict = True
                        break
            
            if conflict:
                # Revert changes
                for v, old_val in changed:
                    forbidden[v] = old_val
                colors[idx] = 0
                continue
            
            if backtrack(idx + 1):
                return True
            
            # Backtrack
            colors[idx] = 0
            for v, old_val in changed:
                forbidden[v] = old_val
        
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")
