```python
import sys

def solve():
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # assignment stores the shift (1, 2, or 3) for each staff member
    assignment = [0] * n

    def is_safe(node, shift):
        for neighbor in adj[node]:
            if assignment[neighbor] == shift:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        
        # Try shifts 1, 2, 3 in order to find the lexicographically smallest sequence
        for shift in (1, 2, 3):
            if is_safe(node, shift):
                assignment[node] = shift
                # We use a simple backtracking search. 
                # Since n is small (up to 40), this is often feasible 
                # but 3-coloring is NP-complete.
                # However, the constraints and the requirement for 
                # lexicographically smallest suggest a search.
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    # Optimization: 3-coloring is hard, but with N=40, pruning or 
    # a more efficient approach might be needed. 
    # However, the lexicographical requirement forces us to try 
    # shift 1, then 2, then 3 for the first person, then second, etc.
    
    # To improve performance, we can check if the remaining graph is 3-colorable,
    # but that's also NP-complete. Let's use the basic backtracking first.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
