def solve():
    import sys
    from collections import defaultdict

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    incompatibilities = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        incompatibilities[u].append(v)
        incompatibilities[v].append(u)

    shifts = [0] * (n + 1)  # We use 1-based indexing

    def is_valid(shift, staff):
        for incompatible in incompatibilities[staff]:
            if shifts[incompatible] == shift:
                return False
        return True

    def assign_shifts(staff):
        if staff > n:
            return True  # All staff have been assigned
        
        for shift in range(1, 4):  # Shift can be 1, 2, or 3
            if is_valid(shift, staff):
                shifts[staff] = shift
                if assign_shifts(staff + 1):
                    return True
                shifts[staff] = 0  # Backtrack
        
        return False

    if assign_shifts(1):
        print("YES")
        print(" ".join(map(str, shifts[1:])))
    else:
        print("NO")
