import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * n  # 0 = uncolored, 1..3 = colors

    # order vertices by degree descending for initial heuristic
    order = sorted(range(n), key=lambda x: -len(adj[x]))

    # map from position in order to vertex
    pos2v = order
    # inverse mapping for quick lookup of position
    v2pos = [0] * n
    for i, v in enumerate(pos2v):
        v2pos[v] = i

    # precompute neighbor positions for fast checking
    neighbor_pos = [ [v2pos[nb] for nb in adj[v]] for v in range(n) ]

    found = False

    def dfs(idx: int):
        nonlocal found
        if found:
            return
        if idx == n:
            found = True
            return
        v = pos2v[idx]
        # collect colors used by already colored neighbors
        used = 0
        for nb in adj[v]:
            c = colors[nb]
            if c:
                used |= 1 << c
        for col in (1, 2, 3):
            if not (used & (1 << col)):
                colors[v] = col
                dfs(idx + 1)
                if found:
                    return
                colors[v] = 0

    dfs(0)

    if not found:
        print("NO")
    else:
        print("YES")
        print(' '.join(str(c) for c in colors))

if __name__ == "__main__":
    solve()
