import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    col = [0] * n          # 0 or bit 1,2,4 representing color
    forbid = [0] * n       # bitmask of forbidden colors (bits 0..2)
    colored_mask = 0

    def propagate(v, cbit):
        nonlocal colored_mask
        assigned = [v]
        changes = []
        stack = [v]
        col[v] = cbit
        while stack:
            x = stack.pop()
            colored_mask |= 1 << x
            cb = col[x]
            nb = adj[x] & ~colored_mask
            while nb:
                lsb = nb & -nb
                u = lsb.bit_length() - 1
                nb ^= lsb
                old = forbid[u]
                new = old | cb
                if new == old:
                    continue
                if new == 7:
                    # conflict: undo everything from this propagation
                    for u2, old2 in changes:
                        forbid[u2] = old2
                    for v2 in assigned:
                        col[v2] = 0
                        colored_mask &= ~(1 << v2)
                    return None
                forbid[u] = new
                changes.append((u, old))
                avail = 7 ^ new
                if (avail & (avail - 1)) == 0 and col[u] == 0:
                    col[u] = avail
                    assigned.append(u)
                    stack.append(u)
        return assigned, changes

    def dfs(i):
        if i == n:
            return True
        if col[i]:
            return dfs(i + 1)
        avail = (~forbid[i]) & 7
        if avail == 0:
            return False
        for bit in (1, 2, 4):
            if avail & bit:
                res = propagate(i, bit)
                if res is not None:
                    assigned, changes = res
                    if dfs(i + 1):
                        return True
                    for u, old in changes:
                        forbid[u] = old
                    for v2 in assigned:
                        col[v2] = 0
                        colored_mask &= ~(1 << v2)
        return False

    if dfs(0):
        out = ['YES', ' '.join(str(c.bit_length()) for c in col)]
        sys.stdout.write('\n'.join(out))
    else:
        sys.stdout.write('NO')

if __name__ == '__main__':
    solve()
