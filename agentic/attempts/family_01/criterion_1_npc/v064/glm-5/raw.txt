import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    def dfs(u):
        for v in adj[u]:
            if color[v] == 0:
                color[v] = 3 - color[u]
                if not dfs(v):
                    return False
            elif color[v] == color[u]:
                return False
        return True
    
    # Check bipartiteness and assign colors 1 or 2
    for i in range(n):
        if color[i] == 0:
            color[i] = 1
            if not dfs(i):
                print("NO")
                return
    
    # Check if any node needs all 3 colors (degree >= 3 with neighbors of both colors)
    for u in range(n):
        has_1 = False
        has_2 = False
        for v in adj[u]:
            if color[v] == 1:
                has_1 = True
            else:
                has_2 = True
        if has_1 and has_2:
            color[u] = 3
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
