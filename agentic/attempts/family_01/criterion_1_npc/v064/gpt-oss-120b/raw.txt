import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    colors = [0] * n          # 0 = uncolored, 1..3 = color
    allowed = [0b111] * n     # bits 0,1,2 correspond to colors 1,2,3

    # order of vertices for tie-breaking (higher degree first)
    degree = [bin(adj[i]).count('1') for i in range(n)]
    order = sorted(range(n), key=lambda x: -degree[x])

    sys.setrecursionlimit(10000)

    def select_vertex():
        # MRV heuristic: choose uncolored vertex with fewest allowed colors
        best = -1
        best_cnt = 4
        best_deg = -1
        for v in order:
            if colors[v] == 0:
                cnt = allowed[v].bit_count()
                if cnt < best_cnt or (cnt == best_cnt and degree[v] > best_deg):
                    best = v
                    best_cnt = cnt
                    best_deg = degree[v]
        return best

    def dfs(colored_cnt: int) -> bool:
        if colored_cnt == n:
            return True
        v = select_vertex()
        if v == -1:
            return False
        avail = allowed[v]
        # try each possible color
        for col_bit in (1, 2, 4):  # bits for colors 1,2,3
            if not (avail & col_bit):
                continue
            c = (col_bit.bit_length())  # 1->1,2->2,4->3
            # assign
            colors[v] = c
            changed = []  # list of (u, old_allowed)
            ok = True
            neigh_mask = adj[v]
            u = 0
            while neigh_mask:
                lsb = neigh_mask & -neigh_mask
                u = (lsb.bit_length() - 1)
                if colors[u] == 0:
                    old = allowed[u]
                    new = old & ~col_bit
                    if new != old:
                        allowed[u] = new
                        changed.append((u, old))
                        if new == 0:
                            ok = False
                            break
                neigh_mask ^= lsb
            if ok and dfs(colored_cnt + 1):
                return True
            # revert
            for u, old in reversed(changed):
                allowed[u] = old
            colors[v] = 0
        return False

    if dfs(0):
        sys.stdout.write("YES\n")
        sys.stdout.write(' '.join(map(str, colors)) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
