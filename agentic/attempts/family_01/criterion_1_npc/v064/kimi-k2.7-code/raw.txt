import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    visited = [False] * n
    ans = [0] * n
    FULL = (1 << 1) | (1 << 2) | (1 << 3)  # bits for colors 1,2,3
    col = [0] * n
    dom = [0] * n
    sys.setrecursionlimit(10000)

    for i in range(n):
        if visited[i]:
            continue
        comp = []
        q = deque([i])
        visited[i] = True
        while q:
            v = q.popleft()
            comp.append(v)
            for u in adj[v]:
                if not visited[u]:
                    visited[u] = True
                    q.append(u)

        comp.sort()
        for v in comp:
            col[v] = 0
            dom[v] = FULL

        stack = []

        def assign(v, c):
            stack.append((v, col[v], dom[v]))
            col[v] = c
            dom[v] = 1 << c
            bitc = 1 << c
            for u in adj[v]:
                if col[u] == 0:
                    old = dom[u]
                    new = old & ~bitc
                    if new == 0:
                        return False
                    if new != old:
                        stack.append((u, 0, old))
                        dom[u] = new
                        if new & (new - 1) == 0:
                            cu = (new & -new).bit_length() - 1
                            if not assign(u, cu):
                                return False
            return True

        def dfs(pos):
            k = len(comp)
            while pos < k and col[comp[pos]] != 0:
                pos += 1
            if pos == k:
                return True
            v = comp[pos]
            mask = dom[v]
            for c in (1, 2, 3):
                bit = 1 << c
                if mask & bit:
                    snap = len(stack)
                    if assign(v, c):
                        if dfs(pos + 1):
                            return True
                    while len(stack) > snap:
                        u, oldc, oldd = stack.pop()
                        col[u] = oldc
                        dom[u] = oldd
            return False

        if not dfs(0):
            print("NO")
            return

        for v in comp:
            ans[v] = col[v]

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
