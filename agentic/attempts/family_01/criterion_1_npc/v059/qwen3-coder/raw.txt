import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try all possible colorings with 3 colors
    # We want lexicographically smallest, so we try colors in order 1, 2, 3
    coloring = [0] * (n + 1)  # 1-indexed
    
    def is_valid(node, color):
        for neighbor in adj[node]:
            if coloring[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for color in [1, 2, 3]:
            if is_valid(node, color):
                coloring[node] = color
                if backtrack(node + 1):
                    return True
                coloring[node] = 0  # backtrack
                
        return False
    
    # Check if graph is 2-colorable first (bipartite)
    def is_bipartite():
        color = [0] * (n + 1)
        for start in range(1, n + 1):
            if color[start] == 0:
                queue = [start]
                color[start] = 1
                while queue:
                    node = queue.pop()
                    for neighbor in adj[node]:
                        if color[neighbor] == 0:
                            color[neighbor] = 3 - color[node]
                            queue.append(neighbor)
                        elif color[neighbor] == color[node]:
                            return False
        return True
    
    # Special case: if m=0, all stations can use band 1
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Try to find a valid 3-coloring
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, coloring[1:])))
    else:
        print("NO")

solve()
