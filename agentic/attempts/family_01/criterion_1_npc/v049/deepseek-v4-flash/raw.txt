We can model this as a graph coloring problem with 3 colors.  
We can solve it with backtracking since \( n \leq 40 \), but we need to be efficient.

Here’s the Python solution:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n

    # Try to color with backtracking, using MRV heuristic
    def backtrack(idx):
        if idx == n:
            return True
        # Choose uncolored vertex with most colored neighbors (MRV)
        best = -1
        best_deg = -1
        for i in range(n):
            if color[i] == -1:
                deg = sum(1 for nb in adj[i] if color[nb] != -1)
                if deg > best_deg:
                    best_deg = deg
                    best = i
        v = best
        # Try colors 1,2,3
        for c in range(3):
            ok = True
            for nb in adj[v]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[v] = c
                if backtrack(idx + 1):
                    return True
                color[v] = -1
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(c + 1) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
