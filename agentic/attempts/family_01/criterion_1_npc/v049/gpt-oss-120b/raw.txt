import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [0] * n  # 0 = uncolored, else 1..3

    # For each vertex keep a mask of forbidden colors from already colored neighbors
    forbidden = [0] * n  # bitmask of colors (bit 0 for color1, etc)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # compute available colors from forbidden mask
        forb = forbidden[v]
        for col in range(1, 4):
            if forb & (1 << (col - 1)):
                continue
            # assign
            colors[v] = col
            # update forbidden for uncolored neighbors
            changed = []
            nb_mask = adj[v]
            while nb_mask:
                u = (nb_mask & -nb_mask).bit_length() - 1
                nb_mask &= nb_mask - 1
                if colors[u] == 0:
                    if not (forbidden[u] & (1 << (col - 1))):
                        forbidden[u] |= 1 << (col - 1)
                        changed.append(u)
            if dfs(idx + 1):
                return True
            # backtrack
            for u in changed:
                forbidden[u] &= ~(1 << (col - 1))
            colors[v] = 0
        return False

    possible = dfs(0)
    out = []
    if possible:
        out.append("YES")
        out.append(" ".join(str(c) for c in colors))
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
