import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        edges.append((u, v))
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Check if graph is 3-colorable using backtracking
    color = [0] * n
    
    def backtrack(node):
        if node == n:
            return True
        for c in range(1, 4):
            ok = True
            for nb in adj[node]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        return False
    
    if not backtrack(0):
        print("NO")
        return
    
    # Check water constraint: each greenhouse has at most 45 species
    counts = [0, 0, 0, 0]
    for c in color:
        counts[c] += 1
    for i in range(1, 4):
        if counts[i] > 45:
            print("NO")
            return
    
    # Check filtration budget: at most 800 toxic pairs in different greenhouses
    cross = 0
    for u, v in edges:
        if color[u] != color[v]:
            cross += 1
    if cross > 800:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(c) for c in color))

solve()
