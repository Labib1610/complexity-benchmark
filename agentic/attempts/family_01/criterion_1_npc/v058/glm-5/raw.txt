import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    assignment = [0] * n
    possible = True
    
    for i in range(n):
        if assignment[i] == 0:
            # BFS traversal for connected component
            queue = [i]
            assignment[i] = 1
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                
                # Collect neighbors' tiers
                neighbor_tiers = set()
                for v in adj[u]:
                    if assignment[v] != 0:
                        neighbor_tiers.add(assignment[v])
                
                valid_tiers = []
                # Try to maintain current assignment if valid, else find another
                for t in [assignment[u], 1, 2, 3]:
                    if t in neighbor_tiers:
                        continue
                    valid = True
                    for v in adj[u]:
                        if assignment[v] != 0 and abs(t - assignment[v]) < 1:
                            valid = False
                            break
                    if valid:
                        valid_tiers.append(t)
                
                if not valid_tiers:
                    possible = False
                    break
                
                assignment[u] = valid_tiers[0]
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 1
                        queue.append(v)
            
            if not possible:
                break
    
    if possible:
        # Verify all edges
        for u, v in edges:
            if abs(assignment[u] - assignment[v]) < 1:
                possible = False
                break
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
