import sys

# Increase recursion depth to handle deep recursion trees
sys.setrecursionlimit(5000)

def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    # Find connected components
    visited = [False] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            # Sort component to ensure we process vertices in increasing index order
            comp.sort()
            components.append(comp)

    colors = [0] * (n + 1)
    available = [0] * (n + 1) # bitmasks: 1, 2, 3 -> bits 0, 1, 2
    
    # Helper for bipartite check
    def is_bipartite(comp):
        colors_map = {}
        stack = [comp[0]]
        colors_map[comp[0]] = 1
        
        while stack:
            u = stack.pop()
            c = colors_map[u]
            next_c = 3 - c 
            
            for v in adj[u]:
                if v not in colors_map:
                    colors_map[v] = next_c
                    stack.append(v)
                else:
                    if colors_map[v] == c:
                        return False
        return True

    comp_allowed = []
    for comp in components:
        if is_bipartite(comp):
            comp_allowed.append([1, 2])
        else:
            comp_allowed.append([1, 2, 3])
            
    def dfs(comp, allowed_colors, idx):
        if idx == len(comp):
            return True
        
        u = comp[idx]
        
        for c in allowed_colors:
            # Check conflict with assigned neighbors
            conflict = False
            for v in adj[u]:
                if colors[v] == c:
                    conflict = True
                    break
            
            if conflict:
                continue
            
            # Forward check
            modified = []
            possible = True
            
            for v in adj[u]:
                if colors[v] == 0:
                    mask = 1 << (c - 1)
                    # Only update if the color was available
                    if available[v] & mask:
                        available[v] &= ~mask
                        modified.append(v)

                    if available[v] == 0:
                        possible = False
                        break

                for v in modified:
                    available[v] |= (1 << (c - 1))
