We can model this as a graph coloring problem with 3 colors (shifts).  
Since \( n \le 40 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n  # 0 = unassigned, 1,2,3 = shifts
    
    def is_safe(node, c):
        for nb in adj[node]:
            if color[nb] == c:
                return False
        return True
    
    def backtrack(idx):
        if idx == n:
            return True
        for c in range(1, 4):
            if is_safe(idx, c):
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
