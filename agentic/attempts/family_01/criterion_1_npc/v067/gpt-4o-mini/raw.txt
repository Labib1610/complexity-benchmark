def is_valid_schedule(n, m, incompatibilities):
    from collections import defaultdict, deque

    graph = defaultdict(list)
    for u, v in incompatibilities:
        graph[u].append(v)
        graph[v].append(u)

    # Result array to hold the shift assignments
    shifts = [0] * (n + 1)
    
    # Function to perform BFS and assign shifts
    def bfs(start):
        queue = deque([start])
        shifts[start] = 1  # Start assigning shift 1
        
        while queue:
            node = queue.popleft()
            current_shift = shifts[node]
            next_shift = 1 if current_shift != 1 else 2  # Start with shift 1, then 2, then 3
            
            for neighbor in graph[node]:
                if shifts[neighbor] == 0:  # Not assigned yet
                    shifts[neighbor] = next_shift
                    queue.append(neighbor)
                elif shifts[neighbor] == current_shift:  # Conflict detected
                    return False
                # Rotate shifts for the next neighbor
                next_shift = 3 if next_shift == 2 else 2 if next_shift == 1 else 1
        return True

    # Try to assign shifts for all components
    for i in range(1, n + 1):
        if shifts[i] == 0:  # Not yet assigned
            if not bfs(i):
                print("NO")
                return

    # Convert shifts to lexicographically smallest if necessary
    result = [str(shifts[i]) for i in range(1, n + 1)]
    print("YES")
    print(" ".join(result))

# Reading input
import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
incompatibilities = [tuple(map(int, line.split())) for line in data[1:m+1]]

is_valid_schedule(n, m, incompatibilities)
