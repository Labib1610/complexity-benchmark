```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    order = sorted(range(n), key=lambda x: -len(adj[x]))
    color = [-1] * n
    found = False

    def dfs(pos: int):
        nonlocal found
        if found:
            return
        if pos == n:
            found = True
            return
        v = order[pos]
        # try colors 0,1,2
        used = 0
        for nb in adj[v]:
            c = color[nb]
            if c != -1:
                used |= 1 << c
        for c in range(3):
            if not (used >> c) & 1:
                color[v] = c
                dfs(pos + 1)
                if found:
                    return
                color[v] = -1

    dfs(0)

    if not found:
        print("NO")
    else:
        print("YES")
        # output in original order, convert to 1..3
        print(' '.join(str(color[i] + 1) for i in range(n)))

if __name__ == "__main__":
    solve()
```
