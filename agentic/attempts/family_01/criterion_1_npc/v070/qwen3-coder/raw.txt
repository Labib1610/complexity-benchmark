import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to 3-color the graph
    color = [-1] * n
    
    def dfs(node, c):
        color[node] = c
        for neighbor in adj[node]:
            if color[neighbor] == -1:
                if not dfs(neighbor, 1 - c if c != 1 else 2):
                    return False
            elif color[neighbor] == c:
                return False
        return True
    
    # Check if the graph is 3-colorable
    for i in range(n):
        if color[i] == -1:
            if not dfs(i, 0):
                print("NO")
                return
    
    print("YES")
    result = []
    for c in color:
        if c == 0:
            result.append("1")
        elif c == 1:
            result.append("2")
        else:
            result.append("3")
    print(" ".join(result))

if __name__ == "__main__":
    main()
