import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    assignment = [0] * n
    
    # Check if assigning color c to node u is valid
    def is_safe(u, c):
        for v in adj[u]:
            if assignment[v] == c:
                return False
        return True
    
    # Backtracking with pruning
    def backtrack(u):
        if u == n:
            return True
        
        for c in range(1, 4):
            if is_safe(u, c):
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
