```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [0] * n
    # neighbor list in order indices for quick check
    neigh_order = [[] for _ in range(n)]
    for v in range(n):
        mask = adj[v]
        while mask:
            u = (mask & -mask).bit_length() - 1
            neigh_order[v].append(pos_in_order[u])
            mask &= mask - 1

    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # collect colors used by already colored neighbors
        used = 0
        for nb_pos in neigh_order[v]:
            if nb_pos < idx:  # neighbor already colored
                used |= 1 << (colors[order[nb_pos]] - 1)
        for c in range(1, 4):
            if not (used & (1 << (c - 1))):
                colors[v] = c
                if dfs(idx + 1):
                    return True
                colors[v] = 0
        return False

    possible = dfs(0)
    out = []
    if possible:
        out.append("YES")
        out.append(" ".join(str(colors[i]) for i in range(n)))
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
