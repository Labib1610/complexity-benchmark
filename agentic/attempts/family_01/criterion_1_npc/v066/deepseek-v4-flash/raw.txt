Here’s the Python 3 solution for the problem:  

```python
n, m = map(int, input().split())
adj = [[False] * n for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u][v] = adj[v][u] = True

color = [0] * n

def backtrack(idx):
    if idx == n:
        return True
    for c in range(1, 4):
        ok = True
        for j in range(idx):
            if adj[idx][j] and color[j] == c:
                ok = False
                break
        if ok:
            color[idx] = c
            if backtrack(idx + 1):
                return True
            color[idx] = 0
    return False

if backtrack(0):
    print("YES")
    print(" ".join(map(str, color)))
else:
    print("NO")
```
