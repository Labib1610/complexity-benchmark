Constructing `out_adj` is O(M).
Backtracking is the main part.
I think it's optimal for Python.

```python
# Final check of the code structure
import sys

# Increase recursion depth
sys.setrecursionlimit(5000)

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    out_adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        if u < v:
            out_adj[u].append(v)
        else:
            out_adj[v].append(u)
            
    color_count = [[0] * 4 for _ in range(n + 1)]
    forbidden_count = [0] * (n + 1)
    
    assignment = [0] * (n + 1)
    
    result = None
    
    def backtrack(u):
        nonlocal result
        
        if u > n:
            result = assignment[1:]
            return True
        
        for c in range(1, 4):
            if color_count[u][c] == 0:
                assignment[u] = c
                
                blocked = False
                
                for v in out_adj[u]:
                    if color_count[v][c] == 0:
                        forbidden_count[v] += 1
                    color_count[v][c] += 1
                    if forbidden_count[v] == 3:
                        blocked = True
                
                if not blocked:
                    if backtrack(u + 1):
                        return True
                
                for v in out_adj[u]:
                    color_count[v][c] -= 1
                    if color_count[v][c] == 0:
                        forbidden_count[v] -= 1
        
        return False

    if backtrack(1):
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
