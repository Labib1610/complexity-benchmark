    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:

    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))

        if idx > n:

        if valid_colors[idx] == 0:
            return False

        # to ensure lexicographically smallest solution
