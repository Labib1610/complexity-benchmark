import sys

def solve():
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # shifts[i] stores the shift assigned to employee i (1 to n)
    # 0 means unassigned
    shifts = [0] * (n + 1)
    
    def is_valid(u, color):
        """Check if assigning 'color' to employee 'u' is valid."""
        for v in adj[u]:
            if shifts[v] == color:
                return False
        return True

    def backtrack(u):
        """
        Try to assign shift to employee u and subsequent employees.
        Since we want the lexicographically smallest sequence, 
        we try shifts 1, 2, 3 in order.
        """
        if u > n:
            return True
        
        # Try shifts 1, 2, 3 in order for lexicographical smallest result
        for color in (1, 2, 3):
            if is_valid(u, color):
                shifts[u] = color
                if backtrack(u + 1):
                    return True
                shifts[u] = 0
        
        return False

    # For n=40, simple backtracking might be slow in worst case (3^40),
    # but for the 3-coloring problem, it often works if the graph is sparse
    # or highly constrained. However, we can optimize by checking if the 
    # remaining graph is still 3-colorable. 
    # Given n=40, we will use backtracking with a simple heuristic or just raw.
    
    if backtrack(1):
        print("YES")
        print(*(shifts[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
