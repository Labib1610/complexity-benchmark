```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [-1] * n
    found = False

    sys.setrecursionlimit(10000)

    def dfs(idx: int):
        nonlocal found
        if found:
            return
        if idx == n:
            found = True
            return
        v = order[idx]
        # compute forbidden colors from already colored neighbors
        forbid = 0
        neigh = adj[v]
        while neigh:
            u = (neigh & -neigh).bit_length() - 1
            neigh &= neigh - 1
            c = colors[u]
            if c != -1:
                forbid |= 1 << c
        for c in range(3):
            if not (forbid >> c) & 1:
                colors[v] = c
                dfs(idx + 1)
                if found:
                    return
                colors[v] = -1

    dfs(0)

    if not found:
        print("NO")
    else:
        print("YES")
        # output in original order, convert to 1..3
        print(' '.join(str(colors[i] + 1) for i in range(n)))

if __name__ == "__main__":
    solve()
```
