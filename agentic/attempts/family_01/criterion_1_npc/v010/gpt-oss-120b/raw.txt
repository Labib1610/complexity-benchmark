```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [-1] * n
    found = False

    def dfs(idx: int) -> bool:
        nonlocal found
        if idx == n:
            return True
        v = order[idx]
        # try colors 0,1,2
        forbidden = 0
        neigh = adj[v]
        # collect colors used by already colored neighbors
        mask = neigh
        while mask:
            u = (mask & -mask).bit_length() - 1
            mask &= mask - 1
            c = colors[u]
            if c != -1:
                forbidden |= 1 << c
        for c in range(3):
            if not (forbidden >> c) & 1:
                colors[v] = c
                if dfs(idx + 1):
                    return True
                colors[v] = -1
        return False

    if dfs(0):
        print("YES")
        # output in original order with 1-based colors
        print(' '.join(str(colors[i] + 1) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
