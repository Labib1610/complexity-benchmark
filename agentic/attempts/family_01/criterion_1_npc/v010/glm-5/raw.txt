def solve_comp(comp_mask, nbr, deg, assigned, PC3, n):
    avail = [7]*n
    FM = [comp_mask]
    amask = [0, 0, 0]
    trail = []
    
    def propagate(v, c):
        stack = [(v, c)]
        push = stack.append
        pop = stack.pop
        while stack:
            x, cx = pop()
            if nbr[x] & amask[cx]:
                return False
            bit = 1 << cx
            aff = nbr[x] & FM[0]
            while aff:
                lb = aff & -aff; aff ^= lb
                u = lb.bit_length() - 1
                au = avail[u]
                na = au & ~bit
                if na != au:
                    if na == 0:
                        return False
                    trail.append((0, u, au))
                    avail[u] = na
                    if not (na & (na - 1)):
                        cu = na.bit_length() - 1
                        trail.append((1, u, cu))
                        assigned[u] = cu
                        amask[cu] |= lb
                        FM[0] ^= lb
                        push((u, cu))
        return True

    def undo_to(mark):
        t = trail
        while len(t) > mark:
            op = t.pop()
            if op[0] == 0:
                avail[op[1]] = op[2]
            else:
                u = op[1]; cu = op[2]
                assigned[u] = -1
                amask[cu] ^= 1 << u
                FM[0] |= 1 << u

    def search(used):
        fm = FM[0]
        if fm == 0:
            return True
        best_v = -1; best_pc = 9; best_d = -1
        while fm:
            b = fm & -fm; fm ^= b
            v = b.bit_length() - 1
            a = avail[v]
            pc = PC3[a]
            if pc < best_pc or (pc == best_pc and deg[v] > best_d):
                best_pc = pc; best_d = deg[v]; best_v = v

        A = avail[v]
        mark = len(trail)
        cand = A & used
        # try colors already used
        cb = 1

            if cand & cb:

                trail.append((0, v, A))

                return True
            undo_to(mark)
        return False

    return search(0)
