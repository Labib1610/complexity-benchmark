import sys

def solve():
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # We need the lexicographically smallest assignment.
    # This means we try assigning channel 1, then 2, then 3 for technician 1,
    # then move to technician 2, and so on.
    
    assignment = [0] * (n + 1)

    def is_safe(node, color):
        for neighbor in adj[node]:
            if assignment[neighbor] == color:
                return False
        return True

    def backtrack(node):
        if node > n:
            return True
        
        # Try channels 1, 2, 3 in order for lexicographical smallest result
        for color in (1, 2, 3):
            if is_safe(node, color):
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        
        return False

    # Optimization: The 3-coloring problem is NP-complete. 
    # However, n=40 is small enough for backtracking with some pruning 
    # or simple search if the graph isn't specifically designed to be a worst-case.
    # To handle potentially harder cases, we can use a slightly more 
    # optimized approach, but standard backtracking is the first attempt.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
