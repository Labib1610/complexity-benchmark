def solve():
    import sys
    from collections import defaultdict, deque

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    graph = defaultdict(list)

    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)

    colors = [0] * (n + 1)

    def bfs(start):
        queue = deque([start])
        colors[start] = 1

        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if colors[neighbor] == 0:  # Not colored
                    colors[neighbor] = 3 - colors[node]  # Alternate colors 1 and 2
                    queue.append(neighbor)
                elif colors[neighbor] == colors[node]:  # Conflict
                    return False
        return True

    for technician in range(1, n + 1):
        if colors[technician] == 0:  # Not visited
            if not bfs(technician):
                print("NO")
                return

    # Now we need to assign channels
    assignment = []
    for technician in range(1, n + 1):
        if colors[technician] == 1:
            assignment.append(1)
        elif colors[technician] == 2:
            assignment.append(2)
        else:
            assignment.append(3)

    print("YES")
    print(" ".join(map(str, assignment)))
