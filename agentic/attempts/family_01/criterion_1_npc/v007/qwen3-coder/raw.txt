def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    conflicts = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        conflicts.append((u, v))
        index += 2
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in conflicts:
        adj[u].append(v)
        adj[v].append(u)
    
    # Check if graph is tripartite (3-colorable)
    # We'll use backtracking to find the lexicographically smallest valid assignment
    color = [0] * n  # 0 means unassigned, 1,2,3 are the shifts
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
            # Also check the product constraint: c * color[neighbor] >= 2
            if color[neighbor] != 0 and c * color[neighbor] < 2:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
            
        for c in [1, 2, 3]:
            if is_valid(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        return False
    
    # Special case: if m=0, all workers can be assigned to shift 1
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

solve()
