import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    neigh = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    full_mask = (1 << n) - 1

    parent = list(range(n))
    size = [1] * n
    parity = [0] * n
    stack = []

    def find(x):
        p = 0
        while parent[x] != x:
            p ^= parity[x]
            x = parent[x]
        return x, p

    def union(a, b, w):
        ra, pa = find(a)
        rb, pb = find(b)
        if ra == rb:
            return (pa ^ pb) == w
        if size[ra] < size[rb]:
            ra, rb = rb, ra
            pa, pb = pb, pa
        parent[rb] = ra
        parity[rb] = pa ^ pb ^ w
        stack.append((rb, ra, size[ra]))
        size[ra] += size[rb]
        return True

    def rollback(to):
        while len(stack) > to:
            rb, ra, sz = stack.pop()
            parent[rb] = rb
            parity[rb] = 0
            size[ra] = sz

    S = 0

    sys.setrecursionlimit(10000)
    def dfs(i, inc_mask, exc_mask):
        nonlocal S
        if i == n:
            S = inc_mask
            return True
        # try to put vertex i into shift 1 (include in S)
        if (adj[i] & inc_mask) == 0:
            if dfs(i + 1, inc_mask | (1 << i), exc_mask):
                return True
        # otherwise put it into the bipartite complement
        start = len(stack)
        ok = True
        for u in neigh[i]:
            if (exc_mask >> u) & 1:
                if not union(i, u, 1):
                    ok = False
                    break
        if ok:
            if dfs(i + 1, inc_mask, exc_mask | (1 << i)):
                return True
        rollback(start)
        return False

    if not dfs(0, 0, 0):
        print("NO")
        return

    colors = [1] * n
    comp_mask = full_mask ^ S
    visited = 0
    side = [0] * n

    while comp_mask:
        v = (comp_mask & -comp_mask).bit_length() - 1
        q = deque([v])
        comp_vertices = [v]
        visited |= 1 << v
        side[v] = 0
        min_v = v
        while q:
            x = q.popleft()
            nb = adj[x] & comp_mask & ~visited
            while nb:
                lsb = nb & -nb
                y = lsb.bit_length() - 1
                visited |= 1 << y
                side[y] = side[x] ^ 1
                if y < min_v:
                    min_v = y
                comp_vertices.append(y)
                q.append(y)
                nb ^= lsb
        for x in comp_vertices:
            colors[x] = 2 if side[x] == side[min_v] else 3
        comp_mask &= ~visited

    out = ['YES', ' '.join(map(str, colors))]
    sys.stdout.write('\n'.join(out))

if __name__ == '__main__':
    solve()
