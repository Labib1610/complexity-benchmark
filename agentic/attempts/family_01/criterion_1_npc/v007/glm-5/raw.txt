import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1; v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Find connected components
    comp_id = [-1] * n
    comps = []
    for i in range(n):
        if comp_id[i] == -1:
            comp = []
            stack = [i]
            comp_id[i] = len(comps)
            while stack:
                v = stack.pop()
                comp.append(v)
                w = adj[v]
                while w:
                    b = w & (-w)
                    u = b.bit_length() - 1
                    if comp_id[u] == -1:
                        comp_id[u] = len(comps)
                        stack.append(u)
                    w ^= b
            comps.append(comp)

    # For each component, find all valid colorings
    all_colorings = []
    for comp in comps:
        size = len(comp)
        if size == 1:
            all_colorings.append([[1], [2], [3]])
            continue
        # Order vertices by degree descending for better pruning
        comp_sorted = sorted(comp, key=lambda x: -bin(adj[x]).count('1'))
        idx_map = {v: i for i, v in enumerate(comp_sorted)}
        local_adj = [0] * size
        for v in comp:
            vi = idx_map[v]
            w = adj[v]
            while w:
                b = w & (-w)
                u = b.bit_length() - 1
                ui = idx_map[u]
                local_adj[vi] |= 1 << ui
                w ^= b

        colorings = []
        colors = [0] * size

        def backtrack(pos):
            if pos == size:
                colorings.append(colors[:])
                return
            used = 0
            w = local_adj[pos]
            while w:
                b = w & (-w)
                u = b.bit_length() - 1
                if u < pos:
                    used |= 1 << colors[u]
                w ^= b
            for c in range(1, 4):
                if not (used & (1 << c)):
                    colors[pos] = c
                    backtrack(pos + 1)
                    if len(colorings) >= 2:
                        return
            colors[pos] = 0

        backtrack(0)
        if not colorings:
            print("NO")
            return
        all_colorings.append(colorings)

    # Combine colorings from all components
    result = [0] * n
    for ci, comp in enumerate(comps):
        coloring = all_colorings[ci][0]
        for i, v in enumerate(comp):
            result[v] = coloring[i]

    print("YES")
    print(' '.join(map(str, result)))

solve()
