We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use a backtracking approach with pruning.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n

    # Sort vertices by degree (largest first) for better pruning
    order = sorted(range(n), key=lambda x: -len(adj[x]))

    def backtrack(idx):
        if idx == n:
            return True
        v = order[idx]
        used = [False] * 4
        for u in adj[v]:
            if color[u] != 0:
                used[color[u]] = True
        for c in range(1, 4):
            if not used[c]:
                color[v] = c
                if backtrack(idx + 1):
                    return True
                color[v] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(color[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
