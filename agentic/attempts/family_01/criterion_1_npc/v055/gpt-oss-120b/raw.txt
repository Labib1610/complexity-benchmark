```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    order = sorted(range(n), key=lambda x: -adj[x].bit_count())  # vertices by degree desc
    color = [-1] * n
    found = False

    sys.setrecursionlimit(10000)

    def dfs(pos: int):
        nonlocal found
        if found:
            return
        if pos == n:
            found = True
            return
        v = order[pos]
        # try colors 0,1,2
        neighbor_mask = adj[v]
        for c in range(3):
            conflict = False
            # check neighbors already colored
            nb = neighbor_mask
            while nb:
                u = (nb & -nb).bit_length() - 1
                if color[u] == c:
                    conflict = True
                    break
                nb &= nb - 1
            if not conflict:
                color[v] = c
                dfs(pos + 1)
                if found:
                    return
                color[v] = -1

    dfs(0)

    if not found:
        print("NO")
    else:
        print("YES")
        # output in original order 1..n with colors 1..3
        print(' '.join(str(color[i] + 1) for i in range(n)))

if __name__ == "__main__":
    solve()
```
