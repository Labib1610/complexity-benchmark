We need to assign each machinist to one of 3 periods such that no two conflicting machinists share a period, and the sequence is lexicographically smallest.

This is essentially a **3-coloring** problem on a graph with up to 40 nodes.  
We can solve it with **backtracking** while enforcing lexicographic order.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n  # 0 = unassigned, 1,2,3 = periods
    
    def is_safe(node, c):
        for nb in adj[node]:
            if color[nb] == c:
                return False
        return True
    
    def backtrack(idx):
        if idx == n:
            return True
        for c in (1, 2, 3):
            if is_safe(idx, c):
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
