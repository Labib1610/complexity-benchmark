import sys
from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        # No conflicts, assign everyone to block 1
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list for conflicts (graph representation)
    adj = defaultdict(set)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-indexed
        v = int(data[index+1]) - 1
        adj[u].add(v)
        adj[v].add(u)
        index += 2
    
    # Try to 3-color the graph using backtracking
    color = [-1] * n  # -1 means unassigned
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
        
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = -1  # Backtrack
                
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

solve()
