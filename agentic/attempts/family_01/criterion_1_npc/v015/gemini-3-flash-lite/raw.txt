import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def can_assign(u, color):
        for v in range(1, u):
            if (adj[u] & (1 << v)) and assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for color in [1, 2, 3]:
            if can_assign(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
