import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # possible colors bitmask: bit 0->color1,1->color2,2->color3
    ALL_COL = 0b111
    poss = [ALL_COL] * n
    color = [-1] * n
    unassigned = set(range(n))

    # order vertices by degree initially
    degree = [bin(adj[i]).count('1') for i in range(n)]
    order = sorted(range(n), key=lambda x: -degree[x])

    # helper to pick next vertex: MRV heuristic
    def select_vertex():
        # choose from unassigned with smallest possible colors count
        best = None
        best_cnt = 4
        for v in unassigned:
            cnt = (poss[v] & ALL_COL).bit_count()
            if cnt < best_cnt:
                best_cnt = cnt
                best = v
                if cnt == 1:
                    break
        return best

    def dfs() -> bool:
        if not unassigned:
            return True
        v = select_vertex()
        unassigned.remove(v)
        saved_poss = poss[:]
        # try each possible color
        mask = poss[v]
        for c in range(3):
            if not (mask >> c) & 1:
                continue
            # assign color c
            color[v] = c
            # forward checking: remove c from neighbors' possibilities
            conflict = False
            changed = []
            nb = adj[v]
            while nb:
                u = (nb & -nb).bit_length() - 1
                nb &= nb - 1
                if color[u] == -1:
                    if poss[u] >> c & 1:
                        poss[u] &= ~(1 << c)
                        changed.append(u)
                        if poss[u] == 0:
                            conflict = True
                            break
                elif color[u] == c:
                    conflict = True
                    break
            if not conflict and dfs():
                return True
            # revert changes
            for u in changed:
                poss[u] = saved_poss[u]
            color[v] = -1
        # backtrack vertex
        poss[:] = saved_poss
        unassigned.add(v)
        return False

    # quick check: if graph is empty, just assign 1s
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return

    if dfs():
        print("YES")
        # convert 0/1/2 to 1/2/3
        print(" ".join(str(c + 1) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
