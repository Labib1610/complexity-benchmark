import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neigh = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        neigh[u].append(v)
        neigh[v].append(u)

    FULL = 14          # bits for colors 1,2,3 are set
    sys.setrecursionlimit(1000000)

    def propagate(color, forbid, stack):
        while stack:
            v = stack.pop()
            if color[v]:
                continue
            dom = FULL & ~forbid[v]
            if dom == 0:
                return False
            if dom & (dom - 1):
                continue
            c = (dom & -dom).bit_length() - 1
            color[v] = c
            bit = 1 << c
            for w in neigh[v]:
                if color[w] == 0:
                    old = forbid[w]
                    new = old | bit
                    if new != old:
                        forbid[w] = new
                        dom2 = FULL & ~new
                        if dom2 == 0:
                            return False
                        if dom2 & (dom2 - 1) == 0:
                            stack.append(w)
        return True

    def dfs(color, forbid):
        v = -1
        best = 4
        for i in range(n):
            if color[i] == 0:
                dom = FULL & ~forbid[i]
                sz = dom.bit_count()
                if sz < best:
                    best = sz
                    v = i
        if v == -1:
            return True
        dom = FULL & ~forbid[v]
        d = dom
        while d:
            lsb = d & -d
            c = lsb.bit_length() - 1
            color2 = color[:]
            forbid2 = forbid[:]
            ok = True
            for w in neigh[v]:
                if color2[w] == c:
                    ok = False
                    break
            if ok:
                color2[v] = c
                stack = []
                bit = 1 << c
                for w in neigh[v]:
                    if color2[w] == 0:
                        old = forbid2[w]
                        new = old | bit
                        if new != old:
                            forbid2[w] = new
                            dom2 = FULL & ~new
                            if dom2 == 0:
                                ok = False
                                break
                            if dom2 & (dom2 - 1) == 0:
                                stack.append(w)
                if ok and propagate(color2, forbid2, stack):
                    if dfs(color2, forbid2):
                        return True
            d ^= lsb
        return False

    def colorable(prefix):
        color = prefix[:]
        forbid = [0] * n
        stack = []
        for v in range(n):
            c = color[v]
            if c:
                bit = 1 << c
                for w in neigh[v]:
                    if color[w] == c:
                        return False
                    if color[w] == 0:
                        forbid[w] |= bit
        for v in range(n):
            if color[v] == 0:
                dom = FULL & ~forbid[v]
                if dom == 0:
                    return False
                if dom & (dom - 1) == 0:
                    stack.append(v)
        if not propagate(color, forbid, stack):
            return False
        return dfs(color, forbid)

    ans = [0] * n
    for i in range(n):
        placed = False
        for c in (1, 2, 3):
            bad = False
            for w in neigh[i]:
                if w < i and ans[w] == c:
                    bad = True
                    break
            if bad:
                continue
            tmp = ans[:]
            tmp[i] = c
            if colorable(tmp):
                ans[i] = c
                placed = True
                break
        if not placed:
            sys.stdout.write("NO\n")
            return

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
