```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * n
    # order vertices by degree descending for initial heuristic
    order = sorted(range(n), key=lambda x: -len(adj[x]))

    # precompute neighbor list for fast check
    neighbor = adj

    # helper to select next vertex: uncolored with most colored neighbors (DSATUR)
    def select_vertex():
        best = -1
        best_sat = -1
        best_deg = -1
        for v in range(n):
            if colors[v] != 0:
                continue
            used = {colors[u] for u in neighbor[v] if colors[u] != 0}
            sat = len(used)
            deg = len(neighbor[v])
            if sat > best_sat or (sat == best_sat and deg > best_deg):
                best = v
                best_sat = sat
                best_deg = deg
        return best

    def dfs(colored_cnt: int) -> bool:
        if colored_cnt == n:
            return True
        v = select_vertex()
        used = {colors[u] for u in neighbor[v] if colors[u] != 0}
        for c in (1, 2, 3):
            if c in used:
                continue
            colors[v] = c
            if dfs(colored_cnt + 1):
                return True
            colors[v] = 0
        return False

    possible = dfs(0)
    out = sys.stdout
    if not possible:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(" ".join(map(str, colors)))

if __name__ == "__main__":
    solve()
```
