```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    colors = [-1] * n
    order = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)

    # Precompute neighbor lists for fast check
    neigh = [ [j for j in range(n) if (adj[i] >> j) & 1] for i in range(n) ]

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        # select uncolored vertex with most colored neighbors (DSATUR heuristic)
        best = -1
        best_sat = -1
        best_deg = -1
        for v in order:
            if colors[v] == -1:
                sat = 0
                used = 0
                for nb in neigh[v]:
                    c = colors[nb]
                    if c != -1:
                        if not (used >> c) & 1:
                            used |= 1 << c
                            sat += 1
                deg = len(neigh[v])
                if sat > best_sat or (sat == best_sat and deg > best_deg):
                    best = v
                    best_sat = sat
                    best_deg = deg
        v = best
        # try colors 0,1,2
        for col in range(3):
            conflict = False
            for nb in neigh[v]:
                if colors[nb] == col:
                    conflict = True
                    break
            if not conflict:
                colors[v] = col
                if dfs(idx + 1):
                    return True
                colors[v] = -1
        return False

    possible = dfs(0)
    out = sys.stdout
    if not possible:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(' '.join(str(c + 1) for c in colors))

if __name__ == "__main__":
    solve()
```
