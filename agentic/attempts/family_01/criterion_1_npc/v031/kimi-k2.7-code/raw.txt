import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_sets = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj_sets[u].add(v)
        adj_sets[v].add(u)
    adj = [list(s) for s in adj_sets]

    colors_in_mask = [[] for _ in range(8)]
    for mask in range(8):
        for c in range(3):
            if (mask >> c) & 1:
                colors_in_mask[mask].append(c)
    singleton_color = [-1] * 8
    for c in range(3):
        singleton_color[1 << c] = c

    color = [-1] * n
    avail = [7] * n
    ans = [0] * n

    sys.setrecursionlimit(10000)

    def propagate(col, lst, stack):
        changed = []
        while stack:
            v, c = stack.pop()
            if col[v] != -1:
                if col[v] != c:
                    return None
                continue
            if not ((lst[v] >> c) & 1):
                return None
            changed.append((v, lst[v]))
            col[v] = c
            lst[v] = 1 << c
            for u in adj[v]:
                if col[u] == -1:
                    if (lst[u] >> c) & 1:
                        changed.append((u, lst[u]))
                        lst[u] &= ~(1 << c)
                        if lst[u] == 0:
                            return None
                        sc = singleton_color[lst[u]]
                        if sc != -1:
                            stack.append((u, sc))
        return changed

    def solve_component(comp, col, lst):
        while True:
            forced = -1
            min_v = -1
            for v in comp:
                if col[v] == -1:
                    if min_v == -1 or v < min_v:
                        min_v = v
                    m = lst[v]
                    if m & (m - 1) == 0:
                        forced = v
                        break
            if forced != -1:
                c = singleton_color[lst[forced]]
                if propagate(col, lst, [(forced, c)]) is None:
                    return False
                continue
            if min_v == -1:
                return True
            m = lst[min_v]
            for c in colors_in_mask[m]:
                changed = propagate(col, lst, [(min_v, c)])
                if changed is not None:
                    if solve_component(comp, col, lst):
                        return True
                    for v, old in reversed(changed):
                        col[v] = -1
                        lst[v] = old
            return False

    def solve_tail(col, lst):
        tcol = col[:]
        tlst = lst[:]
        uncolored = [v for v in range(n) if tcol[v] == -1]
        if not uncolored:
            ans[:] = tcol[:]
            return True

        visited = [False] * n
        components = []
        for v in uncolored:
            if not visited[v]:
                comp = []
                stack = [v]
                visited[v] = True
                while stack:
                    x = stack.pop()
                    comp.append(x)
                    for y in adj[x]:
                        if tcol[y] == -1 and not visited[y]:
                            visited[y] = True
                            stack.append(y)
                components.append(comp)

        components.sort(key=lambda c: min(c))
        for comp in components:
            if not solve_component(comp, tcol, tlst):
                return False
        ans[:] = tcol[:]
        return True

    def dfs():
        i = 0
        while i < n and color[i] != -1:
            i += 1
        if i == n:
            ans[:] = color[:]
            return True

        tail = True
        for v in range(i, n):
            if color[v] == -1 and avail[v].bit_count() == 3:
                tail = False
                break
        if tail:
            return solve_tail(color, avail)

        mask = avail[i]
        for c in colors_in_mask[mask]:
            changed = propagate(color, avail, [(i, c)])
            if changed is not None:
                if dfs():
                    return True
                for v, old in reversed(changed):
                    color[v] = -1
                    avail[v] = old
        return False

    if dfs():
        sys.stdout.write("YES\n")
        sys.stdout.write(' '.join(str(x + 1) for x in ans))
        sys.stdout.write('\n')
    else:
        sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
