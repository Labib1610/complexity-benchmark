import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    color = [0] * n
    visited = [False] * n
    components = []

    for start in range(n):
        if visited[start]:
            continue
        comp = []
        stack = [start]
        visited[start] = True
        color[start] = 1
        while stack:
            u = stack.pop()
            comp.append(u)
            for v in range(n):
                if adj[u] & (1 << v):
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
        components.append(comp)

    # Each component is bipartite with parts color 1 and color 2.
    # We need to choose for each component which part goes to shift 1 vs shift 2,
    # and the other part goes to shift 3. This is a subset-sum over component sizes.
    # Since n <= 40, we can split components into two halves and meet in the middle.

    half = len(components) // 2
    left = components[:half]
    right = components[half:]

    left_maps = []
    for comp in left:
        c1 = sum(1 for u in comp if color[u] == 1)
        c2 = len(comp) - c1
        left_maps.append((c1, c2))

    right_maps = []
    for comp in right:
        c1 = sum(1 for u in comp if color[u] == 1)
        c2 = len(comp) - c1
        right_maps.append((c1, c2))

    target = n // 3

    # Enumerate left subsets
    left_results = {}
    L = len(left_maps)
    for mask in range(1 << L):
        s = 0
        for i in range(L):
            if mask & (1 << i):
                s += left_maps[i][0]
            else:
                s += left_maps[i][1]
        left_results.setdefault(s, mask)

    # Enumerate right subsets and look for complement
    R = len(right_maps)
    found = None
    for mask in range(1 << R):
        s = 0
        for i in range(R):
            if mask & (1 << i):
                s += right_maps[i][0]
            else:
                s += right_maps[i][1]
        need = target - s
        if need in left_results:
            found = (left_results[need], mask)
            break

    if found is None:
        print("NO")
        return

    left_mask, right_mask = found

    assignment = [0] * n
    for i, comp in enumerate(left):
        use_first = bool(left_mask & (1 << i))
        for u in comp:
            if color[u] == 1:
                assignment[u] = 1 if use_first else 3
            else:
                assignment[u] = 3 if use_first else 1
    for i, comp in enumerate(right):
        use_first = bool(right_mask & (1 << i))
        for u in comp:
            if color[u] == 1:
                assignment[u] = 1 if use_first else 3
            else:
                assignment[u] = 3 if use_first else 1

    print("YES")
    print(' '.join(str(x) for x in assignment))

solve()
