import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    color = [0] * (n + 1)
    
    def dfs(u):
        for v in range(1, n + 1):
            if adj[u] & (1 << v):
                if color[v] == color[u]:
                    return False
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    if not dfs(v):
                        return False
        return True
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            if not dfs(i):
                print("NO")
                return
    
    # Now we have a valid 2-coloring using colors 1 and 2.
    # We want lexicographically smallest valid assignment using colors {1,2,3}.
    # Greedily assign each worker the smallest color in {1,2,3} that doesn't conflict with already assigned workers.
    final = [0] * (n + 1)
    for i in range(1, n + 1):
        used = 0
        for v in range(1, n + 1):
            if adj[i] & (1 << v):
                used |= 1 << final[v]
        c = 1
        while c <= 3 and (used & (1 << c)):
            c += 1
        if c > 3:
            print("NO")
            return
        final[i] = c
    
    print("YES")
    print(' '.join(str(final[i]) for i in range(1, n + 1)))

solve()
