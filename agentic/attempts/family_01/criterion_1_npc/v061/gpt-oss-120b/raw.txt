import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [-1] * n  # -1 uncolored, 0,1,2 for colors 1,2,3

    # order vertices by degree descending for initial heuristic
    order = sorted(range(n), key=lambda x: -len(adj[x]))

    # map from position in order to vertex
    pos_of = {v: i for i, v in enumerate(order)}

    # precompute neighbor positions for quick check
    neigh_pos = [sorted([pos_of[v] for v in adj[vv]], key=lambda x: x) for vv in range(n)]

    def select_vertex():
        # choose uncolored vertex with minimum remaining colors (MRV)
        best = -1
        best_cnt = 4
        for v in range(n):
            if colors[v] != -1:
                continue
            used = {colors[u] for u in adj[v] if colors[u] != -1}
            cnt = 3 - len(used)
            if cnt < best_cnt:
                best_cnt = cnt
                best = v
                if cnt == 1:
                    break
        return best

    def dfs(colored_cnt: int) -> bool:
        if colored_cnt == n:
            return True
        v = select_vertex()
        used = {colors[u] for u in adj[v] if colors[u] != -1}
        for c in range(3):
            if c in used:
                continue
            colors[v] = c
            if dfs(colored_cnt + 1):
                return True
            colors[v] = -1
        return False

    possible = dfs(0)
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(str(c + 1) for c in colors))

if __name__ == "__main__":
    solve()
