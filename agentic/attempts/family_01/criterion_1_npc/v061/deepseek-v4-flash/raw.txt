We can solve this by trying to assign phases greedily from module 1 to n, checking feasibility with backtracking and constraint propagation. Since n ≤ 40, we can use a recursive search with pruning.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    # Check if graph is 3-colorable and find lexicographically smallest coloring
    colors = [0] * n  # 0 = uncolored, 1,2,3 = phases

    # Precompute adjacency bitmasks for speed
    adj_mask = [0] * n
    for i in range(n):
        mask = 0
        for j in adj[i]:
            mask |= (1 << j)
        adj_mask[i] = mask

    # Backtracking with pruning
    def backtrack(idx):
        if idx == n:
            return True
        # Try colors 1,2,3 in order for lexicographic minimality
        for c in (1, 2, 3):
            ok = True
            for nb in adj[idx]:
                if colors[nb] == c:
                    ok = False
                    break
            if ok:
                colors[idx] = c
                if backtrack(idx + 1):
                    return True
                colors[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
