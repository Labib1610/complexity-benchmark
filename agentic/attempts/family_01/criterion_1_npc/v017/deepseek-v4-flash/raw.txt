We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use backtracking with pruning.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n

    def backtrack(node):
        if node == n:
            return True
        used = [False] * 4
        for nei in adj[node]:
            if color[nei] != 0:
                used[color[nei]] = True
        for c in range(1, 4):
            if not used[c]:
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
