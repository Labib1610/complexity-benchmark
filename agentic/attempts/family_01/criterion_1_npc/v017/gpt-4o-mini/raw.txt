def can_assign_shifts(n, m, conflicts):
    from collections import defaultdict, deque

    graph = defaultdict(list)
    for u, v in conflicts:
        graph[u].append(v)
        graph[v].append(u)

    # Shift assignments for each employee, 0 means unassigned
    shifts = [0] * (n + 1)

    # Try to color the graph with 3 colors
    def bfs(start):
        queue = deque([start])
        shifts[start] = 1  # Start coloring with 1
        while queue:
            current = queue.popleft()
            current_shift = shifts[current]
            next_shifts = {1, 2, 3} - {current_shift}
            for neighbor in graph[current]:
                if shifts[neighbor] == 0:  # Unassigned
                    shifts[neighbor] = next_shifts.pop()  # Assign a valid shift
                    queue.append(neighbor)
                elif shifts[neighbor] in {current_shift}:  # Conflict detected
                    return False
        return True

    for i in range(1, n + 1):
        if shifts[i] == 0:  # Unvisited
            if not bfs(i):
                return "NO"
    
    return "YES", shifts[1:]

# Reading input
import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
conflicts = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = can_assign_shifts(n, m, conflicts)
if result == "NO":
    print("NO")
else:
    print("YES")
    print(" ".join(map(str, result[1])))
