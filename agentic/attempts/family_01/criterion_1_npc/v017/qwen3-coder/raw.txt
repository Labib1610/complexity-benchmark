def solve():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for the graph
    adj = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Color array: -1 means unvisited, 0,1,2 are colors
    color = [-1] * n
    visited = [False] * n
    
    # Check each component
    for start in range(n):
        if visited[start]:
            continue
            
        # BFS to color the component
        queue = deque()
        queue.append(start)
        color[start] = 0
        visited[start] = True
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = (current_color + 1) % 3
                    queue.append(neighbor)
                else:
                    # Check for conflict
                    if color[neighbor] == current_color:
                        print("NO")
                        return
    
    # If we reach here, a valid 3-coloring exists
    print("YES")
    result = [str(c + 1) for c in color]  # Convert back to 1-based colors
    print(" ".join(result))

solve()
