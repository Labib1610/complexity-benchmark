```python
import sys
from collections import defaultdict

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, edges

def solve(n, m, edges):
    # Build adjacency list
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Color nodes with 3 colors using BFS/DFS
    colors = [0] * (n + 1)  # 0 means uncolored, 1, 2, 3 are colors
    
    def dfs(node, color):
        colors[node] = color
        for neighbor in adj[node]:
            if colors[neighbor] == 0:
                # Try to color neighbor with a different color
                next_color = (color % 3) + 1
                if not dfs(neighbor, next_color):
                    return False
            elif colors[neighbor] == color:
                return False
        return True
    
    # Check each connected component
    for i in range(1, n + 1):
        if colors[i] == 0:
            if not dfs(i, 1):
                return "NO"
    
    return "YES\n" + " ".join(map(str, colors[1:]))

def main():
    n, m, edges = read_input()
    if m == 0:
        # No constraints, any assignment works
        result = "YES\n" + " ".join(["1"] * n)
    else:
        # Need to check bipartiteness with 3 colors
        result = solve(n, m, edges)
    print(result)

if __name__ == "__main__":
    main()
```
