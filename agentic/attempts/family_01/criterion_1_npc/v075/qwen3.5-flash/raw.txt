    # cnt[u][c] stores the number of colored neighbors of u that have color c
    # Indices 1, 2, 3 are used for colors
    cnt = [[0] * 4 for _ in range(n + 1)]
    
    # We will use a list for updates to efficiently backtrack the cnt array
    # However, since we are inside a recursive function, we can just maintain the state
    # and revert it. Using a list to store changes is cleaner.
    
    def backtrack(idx):
        if idx > n:
            return True
        
        # Determine allowed colors for current node idx
        # A color c is allowed if cnt[idx][c] == 0
        # This means no colored neighbor has color c.
        allowed = [c for c in range(1, 4) if cnt[idx][c] == 0]
        
        if not allowed:
            return False
        
        # To ensure lexicographically smallest sequence, we try allowed colors in increasing order.
        # allowed is naturally sorted because we iterate 1..3.
        
        for c in allowed:
            # Assign color c to idx
            color[idx] = c
            
            # Update cnt for neighbors and check for doomed nodes
            # We store changes to revert them later
            updates = [] 
            doomed = False
            
            # Only need to check uncolored neighbors for doomed status
            # Colored neighbors are already consistent (checked by allowed selection)
            # But wait, we need to check if coloring idx with c makes any neighbor doomed.
            # Neighbors v are doomed if they have neighbors of all colors 1, 2, 3.
            # We update cnt[v][c] += 1.
            
            for v in adj[idx]:
                if color[v] == 0: # v is uncolored
                    cnt[v][c] += 1
                    updates.append((v, c))
                    # Check if v is now doomed (all 3 colors forbidden)
                    # Forbidden means cnt[v][k] > 0
                    if cnt[v][1] > 0 and cnt[v][2] > 0 and cnt[v][3] > 0:
                        doomed = True
                        # We can break early, but we need to ensure updates list is handled correctly
                        # If we break, updates contains partial list.
                        # We will iterate updates to revert.
                        break
            
            if not doomed:
                if backtrack(idx + 1):
                    return True
            
            # Backtrack: revert cnt changes
            for v, c_val in updates:
                cnt[v][c_val] -= 1
            
            # Revert color
            color[idx] = 0
            
        return False

    if backtrack(1):
        print("YES")
        print(*(color[1:]))
    else:
        print("NO")
