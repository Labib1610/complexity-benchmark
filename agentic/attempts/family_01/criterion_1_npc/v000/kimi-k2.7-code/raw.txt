 ```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            sys.stdout.write("NO\n")
            return
        adj[u].add(v)
        adj[v].add(u)
    adj = [list(s) for s in adj]

    color = [-1] * n          # -1 uncolored, otherwise 0/1/2
    avail = [7] * n           # bitmask of allowed colors {1,2,3}
    changes = []              # stack of (v, c, [neighbors whose avail changed])
    sol = None

    def assign(v, c, queue):
        bit = 1 << c
        color[v] = c
        changed = []
        for nb in adj[v]:
            if color[nb] == -1 and (avail[nb] & bit):
                avail[nb] &= ~bit
                changed.append(nb)
                if avail[nb] == 0:
                    changes.append((v, c, changed))
                    return False
                # exactly one color left -> force it later
                if avail[nb] & (avail[nb] - 1) == 0:
                    queue.append(nb)
        changes.append((v, c, changed))
        return True

    def unassign():
        v, c, changed = changes.pop()
        bit = 1 << c
        color[v] = -1
        for nb in changed:
            avail[nb] |= bit

    def propagate(queue):
        while queue:
            v = queue.pop()
            if color[v] != -1:
                continue
            mask = avail[v]
            if mask == 0:
                return False
            if mask & (mask - 1):
                continue
            c = mask.bit_length() - 1
            if not assign(v, c, queue):
                return False
        return True

    sys.setrecursionlimit(10000)

    def dfs(i):
        nonlocal sol
        if sol is not None:
            return
        while i < n and color[i] != -1:
            i += 1
        if i == n:
            sol = color.copy()
            return
        if avail[i] == 0:
            return
        for c in (0, 1, 2):
            if avail[i] & (1 << c):
                queue = []
                start = len(changes)
                if assign(i, c, queue) and propagate(queue):
                    dfs(i + 1)
                while len(changes) > start:
                    unassign()
                if sol is not None:
                    return

    dfs(0)
    if sol is None:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n")
        sys.stdout.write(' '.join(str(c + 1) for c in sol) + "\n")

if __name__ == "__main__":
    solve()
```
