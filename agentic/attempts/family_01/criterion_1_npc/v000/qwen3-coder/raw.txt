def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(set)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Try all 3-colorings
    # For each connected component, we'll try to 3-color it
    
    color = [-1] * n  # -1 means uncolored
    
    def dfs(node, c):
        # Try to color node with color c
        # Check if any neighbor has the same color
        for nei in adj[node]:
            if color[nei] == c:
                return False
        
        color[node] = c
        
        # Now try to color neighbors with other colors
        for nei in adj[node]:
            if color[nei] == -1:
                # Try the other two colors
                found = False
                for nc in [(c+1)%3, (c+2)%3]:
                    if dfs(nei, nc):
                        found = True
                        break
                if not found:
                    color[node] = -1
                    return False
            else:
                # Already colored, check compatibility
                if color[nei] == c:
                    color[node] = -1
                    return False
                    
        return True
    
    # Check if graph is 3-colorable
    visited = [False] * n
    
    for i in range(n):
        if color[i] == -1:
            # Try coloring this component starting with color 0
            success = False
            for c in range(3):
                old_color = color[:]
                if dfs(i, c):
                    success = True
                    break
                else:
                    color = old_color[:]
            
            if not success:
                print("NO")
                return
    
    # Verify the solution
    for u, v in edges:
        if color[u] == color[v]:
            print("NO")
            return
    
    print("YES")
    print(" ".join(str(c + 1) for c in color))

solve()
