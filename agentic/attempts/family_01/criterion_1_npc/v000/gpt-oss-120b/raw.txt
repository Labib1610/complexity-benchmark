```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # possible colors bitmask for each vertex, 1<<c where c=0,1,2
    poss = [0b111] * n
    color = [-1] * n
    uncolored = set(range(n))

    # precompute degree
    degree = [bin(adj[i]).count('1') for i in range(n)]

    # helper to pick next vertex: MRV then max degree
    def select_vertex():
        best = None
        best_cnt = 4
        best_deg = -1
        for v in uncolored:
            cnt = (poss[v] & 0b111).bit_count()
            if cnt < best_cnt or (cnt == best_cnt and degree[v] > best_deg):
                best = v
                best_cnt = cnt
                best_deg = degree[v]
        return best

    sys.setrecursionlimit(10000)

    def dfs() -> bool:
        if not uncolored:
            return True
        v = select_vertex()
        mask = poss[v] & 0b111
        # try colors in order 0,1,2
        for c in range(3):
            if not (mask >> c) & 1:
                continue
            # assign
            color[v] = c
            uncolored.remove(v)
            # record changes to revert
            changed = []
            conflict = False
            nb = adj[v]
            while nb:
                u = (nb & -nb).bit_length() - 1
                nb &= nb - 1
                if color[u] == -1:
                    if poss[u] & (1 << c):
                        poss[u] &= ~(1 << c)
                        changed.append(u)
                        if poss[u] == 0:
                            conflict = True
                            break
            if not conflict and dfs():
                return True
            # revert
            for u in changed:
                poss[u] |= (1 << c)
            color[v] = -1
            uncolored.add(v)
        return False

    if dfs():
        print("YES")
        print(' '.join(str(c+1) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
