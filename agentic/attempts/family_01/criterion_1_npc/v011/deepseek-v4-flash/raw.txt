 Here's a possible solution using a backtracking approach with pruning, since \( n \leq 40 \) makes brute force over \( 3^n \) infeasible. We can use a recursive assignment with constraint checking.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].add(v)
        adj[v].add(u)
    
    color = [0] * n  # 0 = unassigned, 1,2,3 = modules
    
    # Pre-check: if any vertex has degree >= 3 and n > 1, still possible, but we'll just backtrack.
    
    # Sort vertices by degree descending for better pruning
    order = list(range(n))
    order.sort(key=lambda x: -len(adj[x]))
    
    def backtrack(idx):
        if idx == n:
            return True
        v = order[idx]
        used = set()
        for u in adj[v]:
            if color[u] != 0:
                used.add(color[u])
        for c in (1, 2, 3):
            if c not in used:
                color[v] = c
                if backtrack(idx + 1):
                    return True
                color[v] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(str(color[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
