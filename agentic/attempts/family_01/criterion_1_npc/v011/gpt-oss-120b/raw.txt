```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [0] * n  # 0 = uncolored, else 1..3
    # for each vertex keep mask of colors used by its colored neighbours
    neighbor_used = [0] * n  # bits 0,1,2 for colors 1,2,3

    def dfs(k: int) -> bool:
        if k == n:
            return True
        v = order[k]
        # compute forbidden colors from already colored neighbours
        forbid = neighbor_used[v]
        for col in range(1, 4):
            if forbid & (1 << (col - 1)):
                continue
            # assign
            colors[v] = col
            # update neighbor_used for uncolored neighbours
            changed = []
            mask = adj[v]
            while mask:
                u = (mask & -mask).bit_length() - 1
                mask &= mask - 1
                if colors[u] == 0:
                    neighbor_used[u] |= 1 << (col - 1)
                    changed.append(u)
            if dfs(k + 1):
                return True
            # backtrack
            for u in changed:
                neighbor_used[u] &= ~(1 << (col - 1))
            colors[v] = 0
        return False

    possible = dfs(0)
    out = sys.stdout
    if not possible:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(" ".join(str(c) for c in colors))

if __name__ == "__main__":
    solve()
```
