```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    neigh = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    order = sorted(range(n), key=lambda x: -len(neigh[x]))
    pos_in_order = [0]*n
    for i, v in enumerate(order):
        pos_in_order[v] = i

    colors = [-1]*n
    # adjacency check using order positions for early pruning
    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # try colors
        for c in range(3):
            ok = True
            for nb in neigh[v]:
                if colors[nb] == c:
                    ok = False
                    break
            if not ok:
                continue
            colors[v] = c
            if dfs(idx+1):
                return True
            colors[v] = -1
        return False

    if dfs(0):
        print("YES")
        # output in original 1..n order, shift numbers 1..3
        print(' '.join(str(colors[i]+1) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
