```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # domains: 3-bit mask, bit 0 -> color 1, bit1 -> color2, bit2 -> color3
    domains = [0b111] * n
    assignment = [0] * n  # 0 means unassigned, else 1..3

    # order vertices by degree for heuristic
    order = sorted(range(n), key=lambda x: -bin(adj[x]).count("1"))

    # map from position in order to vertex
    pos2v = order
    v2pos = [0] * n
    for i, v in enumerate(order):
        v2pos[v] = i

    # Precompute neighbor list
    neigh = [ [j for j in range(n) if (adj[i] >> j) & 1] for i in range(n) ]

    sys.setrecursionlimit(10000)

    def select_unassigned():
        # MRV heuristic: choose unassigned vertex with smallest domain size
        best = -1
        best_sz = 4
        for v in range(n):
            if assignment[v] == 0:
                sz = bin(domains[v]).count("1")
                if sz < best_sz:
                    best_sz = sz
                    best = v
                    if sz == 1:
                        break
        return best

    def dfs(assigned_cnt: int) -> bool:
        if assigned_cnt == n:
            return True
        v = select_unassigned()
        if v == -1:
            return False
        mask = domains[v]
        # try each possible color
        for c in range(3):
            if mask & (1 << c):
                # assign color c+1 to v
                assignment[v] = c + 1
                # store changes to revert
                changed = []
                ok = True
                for nb in neigh[v]:
                    if assignment[nb] == 0:
                        if domains[nb] & (1 << c):
                            new_mask = domains[nb] & ~(1 << c)
                            if new_mask == 0:
                                ok = False
                                # no need to continue propagation
                                # but still record change for consistency
                            changed.append((nb, domains[nb]))
                            domains[nb] = new_mask
                if ok and dfs(assigned_cnt + 1):
                    return True
                # revert
                for nb, old in reversed(changed):
                    domains[nb] = old
                assignment[v] = 0
        return False

    possible = dfs(0)
    out = []
    if not possible:
        out.append("NO")
    else:
        out.append("YES")
        out.append(" ".join(str(assignment[i]) for i in range(n)))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
