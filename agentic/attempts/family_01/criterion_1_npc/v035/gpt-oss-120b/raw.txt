```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"), reverse=True)
    pos_in_order = [0] * n
    for i, v in enumerate(order):
        pos_in_order[v] = i

    colors = [0] * n  # 0 = uncolored, else 1..3

    # precompute neighbor list in order indices for fast check
    neigh_order = [[] for _ in range(n)]
    for v in range(n):
        mask = adj[v]
        while mask:
            u = (mask & -mask).bit_length() - 1
            neigh_order[v].append(pos_in_order[u])
            mask &= mask - 1

    # backtracking
    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # try colors 1..3
        for c in (1, 2, 3):
            ok = True
            for nb_idx in neigh_order[v]:
                if nb_idx < idx:  # neighbor already colored
                    nb = order[nb_idx]
                    if colors[nb] == c:
                        ok = False
                        break
            if ok:
                colors[v] = c
                if dfs(idx + 1):
                    return True
                colors[v] = 0
        return False

    if dfs(0):
        out = ["YES", " ".join(str(colors[i]) for i in range(n))]
    else:
        out = ["NO"]
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
