import sys

def solve() -> None:
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            print("NO")
            return
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    color = [-1] * n          # -1 uncolored, else 0/1/2
    allow = [7] * n           # bits 0,1,2 allowed

    sys.setrecursionlimit(10000)

    def bit_index(x: int) -> int:
        return x.bit_length() - 1

    def assign_and_propagate(v: int, c: int, changes: list) -> bool:
        stack = [v]
        changes.append((v, color[v], allow[v]))
        color[v] = c
        allow[v] = 1 << c
        bit = 1 << c
        while stack:
            v = stack.pop()
            c = color[v]
            bit = 1 << c
            nb = adj[v]
            while nb:
                lsb = nb & -nb
                u = bit_index(lsb)
                nb ^= lsb
                if u == v:
                    return False
                if color[u] != -1:
                    if color[u] == c:
                        return False
                    continue
                if allow[u] & bit:
                    changes.append((u, color[u], allow[u]))
                    allow[u] &= ~bit
                    if allow[u] == 0:
                        return False
                    if allow[u] & (allow[u] - 1) == 0:
                        cu = bit_index(allow[u])
                        changes.append((u, color[u], allow[u]))
                        color[u] = cu
                        allow[u] = 1 << cu
                        stack.append(u)
        return True

    def dfs(pos: int) -> bool:
        n_local = n
        col = color
        while pos < n_local and col[pos] != -1:
            pos += 1
        if pos == n_local:
            return True
        mask = allow[pos]
        for c in range(3):
            if mask & (1 << c):
                changes = []
                if assign_and_propagate(pos, c, changes):
                    if dfs(pos):
                        return True
                for v, old_color, old_allow in reversed(changes):
                    color[v] = old_color
                    allow[v] = old_allow
        return False

    if dfs(0):
        out = ['YES', ' '.join(str(c + 1) for c in color)]
        sys.stdout.write('\n'.join(out))
    else:
        sys.stdout.write('NO')

if __name__ == '__main__':
    solve()
