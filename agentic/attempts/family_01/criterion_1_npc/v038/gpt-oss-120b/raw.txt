import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * n  # 0 = uncolored, else 1..3

    # order vertices by degree descending for initial heuristic
    order = sorted(range(n), key=lambda x: -len(adj[x]))

    # Precompute neighbor list for fast check
    neigh = adj

    # DSATUR style: at each step pick vertex with max saturation (different colors among colored neighbors),
    # break ties by degree.
    def select_vertex():
        best = -1
        best_sat = -1
        best_deg = -1
        for v in range(n):
            if colors[v] != 0:
                continue
            sat_set = set()
            for nb in neigh[v]:
                c = colors[nb]
                if c:
                    sat_set.add(c)
            sat = len(sat_set)
            deg = len(neigh[v])
            if sat > best_sat or (sat == best_sat and deg > best_deg):
                best = v
                best_sat = sat
                best_deg = deg
        return best

    def dfs(colored_cnt: int) -> bool:
        if colored_cnt == n:
            return True
        v = select_vertex()
        # try colors 1..3
        for col in (1, 2, 3):
            conflict = False
            for nb in neigh[v]:
                if colors[nb] == col:
                    conflict = True
                    break
            if not conflict:
                colors[v] = col
                if dfs(colored_cnt + 1):
                    return True
                colors[v] = 0
        return False

    possible = dfs(0)
    out = sys.stdout
    if not possible:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(" ".join(str(c) for c in colors))

if __name__ == "__main__":
    solve()
