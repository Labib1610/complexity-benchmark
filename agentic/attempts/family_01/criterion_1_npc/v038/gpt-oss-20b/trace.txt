def solve_component(vertices):
    nonlocal ans, avail
    L = len(vertices)
    def dfs(idx):
        if idx==L:
            return True
        v = vertices[idx]
        for c in (1,2,3):
            bit = 1 << (c-1)
            if not (avail[v] & bit):
                continue
            conflict=False
            changes=[]
            for u in adj[v]:
                if ans[u]!=0:
                    if ans[u]==c:
                        conflict=True
                        break
                else:
                    if avail[u] & bit:
                        avail[u] ^= bit
                        changes.append(u)
            if conflict:
                for u in changes:
                    avail[u] ^= bit
                continue
            ans[v]=c
            if dfs(idx+1):
                return True
            ans[v]=0
            for u in changes:
                avail[u] ^= bit
        return False
    return dfs(0)
