import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Sort neighbors to ensure deterministic processing order
    for i in range(n):
        adj[i].sort()
    
    # Best solution found (lexicographically smallest)
    best = None
    
    def dfs(u, colors):
        nonlocal best
        if best is not None:
            return
        
        if u == n:
            best = colors[:]
            return
        
        # Try colors 1, 2, 3 in order
        for c in [1, 2, 3]:
            valid = True
            for v in adj[u]:
                if v < u and colors[v] == c:
                    valid = False
                    break
            if valid:
                colors[u] = c
                dfs(u + 1, colors)
                colors[u] = 0
                if best is not None:
                    return
    
    colors = [0] * n
    dfs(0, colors)
    
    if best is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, best)))

solve()
