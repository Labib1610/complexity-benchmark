```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u][v] = True
        adj[v][u] = True
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        forbidden = set()
        for j in range(1, i):
            if adj[i][j]:
                forbidden.add(colors[j])
        
        c = 1
        while c in forbidden:
            c += 1
            
        if c > 3:
            print("NO")
            return
            
        colors[i] = c
        
    print("YES")
    print(" ".join(map(str, colors[1:])))

if __name__ == '__main__':
    solve()
```
