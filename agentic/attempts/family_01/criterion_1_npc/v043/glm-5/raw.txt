import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # Sort adjacency lists to ensure we check neighbors in order
    for i in range(n):
        adj[i].sort()

    colors = [0] * n
    
    # Backtracking with pruning
    # We assign colors to vertices 0 to n-1
    # To find the lexicographically smallest valid coloring, we try colors 1, 2, 3 in order
    
    def backtrack(index):
        if index == n:
            return True
        
        u = index
        # Determine which colors are forbidden by already colored neighbors
        forbidden = [False] * 4
        for v in adj[u]:
            if v < u: # Neighbors with smaller index are already colored
                forbidden[colors[v]] = True
        
        for c in range(1, 4):
            if not forbidden[c]:
                colors[u] = c
                if backtrack(index + 1):
                    return True
                colors[u] = 0 # Backtrack
        
        return False

    if backtrack(0):
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, colors)) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
