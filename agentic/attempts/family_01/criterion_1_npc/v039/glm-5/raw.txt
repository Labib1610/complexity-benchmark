import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for c in range(1, 4):
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        
        return False

    if backtrack(0):
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, colors)) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
