```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n  # bitmask of neighbors (up to 40 bits)
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # possible colors per vertex as 3-bit mask (bits 0,1,2)
    ALL_COL = 0b111
    poss = [ALL_COL] * n
    color = [-1] * n

    # order vertices by degree for tie-breaking
    degree = [bin(adj[i]).count('1') for i in range(n)]
    order = sorted(range(n), key=lambda x: -degree[x])

    # helper to pick next vertex: MRV, then degree order
    def select_vertex():
        min_cnt = 4
        cand = -1
        for v in order:
            if color[v] == -1:
                cnt = (poss[v] & 0b111).bit_count()
                if cnt < min_cnt:
                    min_cnt = cnt
                    cand = v
                    if cnt == 1:
                        break
        return cand

    def dfs(assigned: int) -> bool:
        if assigned == n:
            return True
        v = select_vertex()
        if v == -1:
            return False
        mask = poss[v]
        # try each possible color
        for c in range(3):
            if mask & (1 << c):
                # assign
                color[v] = c
                saved = []
                ok = True
                # remove this color from neighbors' possibilities
                nb = adj[v]
                while nb:
                    u = (nb & -nb).bit_length() - 1
                    nb &= nb - 1
                    if color[u] == -1:
                        if poss[u] & (1 << c):
                            saved.append((u, poss[u]))
                            poss[u] &= ~(1 << c)
                            if poss[u] == 0:
                                ok = False
                                break
                if ok and dfs(assigned + 1):
                    return True
                # rollback
                for u, old in saved:
                    poss[u] = old
                color[v] = -1
        return False

    if dfs(0):
        print("YES")
        # convert 0/1/2 to 1/2/3
        print(' '.join(str(c + 1) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
