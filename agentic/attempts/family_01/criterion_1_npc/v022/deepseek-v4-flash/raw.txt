 We can model this as a graph coloring problem with 3 colors (tiers), where edges require different colors.  
Since \( n \le 40 \), we can use backtracking with constraint propagation.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    colors = [0] * n  # 0 = unassigned, 1..3 = tier
    
    # Check if assigning color c to node v is valid
    def is_safe(v, c):
        for nb in adj[v]:
            if colors[nb] == c:
                return False
        return True
    
    # Backtracking with MRV heuristic
    def backtrack():
        # Find unassigned vertex with most constraints (degree)
        unassigned = [i for i in range(n) if colors[i] == 0]
        if not unassigned:
            return True
        # MRV: pick vertex with most colored neighbors
        v = max(unassigned, key=lambda x: sum(1 for nb in adj[x] if colors[nb] != 0))
        for c in range(1, 4):
            if is_safe(v, c):
                colors[v] = c
                if backtrack():
                    return True
                colors[v] = 0
        return False
    
    if backtrack():
        print("YES")
        print(" ".join(str(colors[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
