We can model this as a bipartite graph problem.  
Each emitter is a node, and each interference channel is an edge that requires the two endpoints to have different states (1 or 2).  
This is exactly a 2-coloring problem. If the graph is not bipartite, it’s impossible.  
If it is bipartite, we want the lexicographically smallest assignment, so we try to assign state 1 whenever possible, starting from node 1.

We can solve it with BFS/DFS and assign colors greedily.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 0  # 0 means state 1, 1 means state 2
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = color[u] ^ 1
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

possible = True
for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            possible = False
            break

if not possible:
    print("NO")
else:
    print("YES")
    # Convert 0 -> 1, 1 -> 2
    print(' '.join(str(c + 1) for c in color))
```
