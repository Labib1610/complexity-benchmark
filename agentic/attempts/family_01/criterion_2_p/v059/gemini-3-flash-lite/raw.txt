import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # We want the lexicographically smallest configuration.
    # This means for each connected component, we try to assign 1 to the
    # smallest node index first, then propagate.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start DFS/BFS for this component.
            # To get lexicographically smallest, we must pick color 1 for 
            # the smallest node in the component.
            stack = [i]
            color[i] = 1
            component = []
            
            # Use stack to traverse component
            idx_comp = 0
            comp_nodes = [i]
            visited_indices = {i}
            
            # BFS/DFS to find all nodes in component
            q = [i]
            while q:
                u = q.pop()
                for v in adj[u]:
                    if v not in visited_indices:
                        visited_indices.add(v)
                        q.append(v)
                        comp_nodes.append(v)
            
            # Now verify Bipartite and assign colors greedily
            # To be lexicographically smallest, we must ensure the first node
            # encountered in the component gets 1.
            q = [i]
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
