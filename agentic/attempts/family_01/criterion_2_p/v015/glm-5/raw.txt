```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
    
    # color array: 0=unassigned, 1=period 1, 2=period 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for this component
            queue = deque([i])
            color[i] = 1
            
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        # Conflict found
                        sys.stdout.write("NO")
                        return
    
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color[1:])))

if __name__ == "__main__":
    solve()
```
