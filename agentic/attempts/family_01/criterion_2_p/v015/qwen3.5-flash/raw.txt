            color[i] = 1

                        return

# Increase recursion depth just in case, though we use iterative BFS.
sys.setrecursionlimit(300000)

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list
    adj = [[] for _ in range(n + 1)]
    
    # Parse edges
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break
            
    # Color array: 0 = uncolored, 1 = sector 1, 2 = sector 2
    color = [0] * (n + 1)
    
    # Process each component
    # To achieve lexicographically smallest sequence (s_1, s_2, ..., s_n),
    # we iterate i from 1 to n. If i is uncolored, it's the first node of its component.
    # We assign it sector 1, which is the smallest possible value.
    # This determines the rest of the component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                c = color[u]
                next_c = 3 - c
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        queue.append(v)
                    elif color[v] != next_c:
                        # Not bipartite
                        print("NO")
                        return

    # If valid
    print("YES")
    # Efficiently print the result
    sys.stdout.write(" ".join(map(str, color[1:])) + "\n")

if __name__ == '__main__':
    solve()

    if not input_data:

        m = int(next(iterator))

        return

    adj = [[] for _ in range(n + 1)]

    for _ in range(m):

            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break
            
    # Color array: 0 = uncolored, 1 = sector 1, 2 = sector 2
    color = [0] * (n + 1)
    
    # Iterate through all nodes to handle disconnected components
    # We want the lexicographically smallest sequence s_1, ..., s_n.
    # This means we want s_1=1 if possible, then s_2=1 if possible, etc.
    # Since components are independent, for each component, we can choose the coloring.
    # When we encounter the first uncolored node 'i' (which is the smallest index in its component),
    # we must assign it 1 to minimize the sequence at index 'i'.
    # This choice fixes the coloring for the entire component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the component containing i
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                next_color = 3 - current_color # 3-1=2, 3-2=1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_color
                        queue.append(v)
                    elif color[v] != next_color:
                        # Conflict detected: graph is not bipartite
                        print("NO")
                        return

    # If we successfully colored the graph without conflicts
    print("YES")
    # Output the assignments for 1 to n
    # Using sys.stdout.write for efficiency with large output
    sys.stdout.write(" ".join(map(str, color[1:])) + "\n")

if __name__ == '__main__':
