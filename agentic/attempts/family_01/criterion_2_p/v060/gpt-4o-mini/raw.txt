def main():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().splitlines()
    
    N, M = map(int, data[0].split())
    graph = defaultdict(list)
    
    for i in range(1, M + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)

    color = [0] * (N + 1)  # 0 means uncolored, 1 and 2 will be the two colors
    
    def bfs(start):
        queue = deque([start])
        color[start] = 1  # Start coloring with 1
        
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == 0:  # Not colored yet
                    color[neighbor] = 3 - color[node]  # Alternate color
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:  # Conflict
                    return False
        return True

    for i in range(1, N + 1):
        if color[i] == 0:  # Not colored yet
            if not bfs(i):
                print("NO")
                return

    # Output the result
    print("YES")
    print(' '.join(map(str, color[1:])))
    
if __name__ == "__main__":
    main()
