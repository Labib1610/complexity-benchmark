import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)  # 0 unvisited, 1 or 2

    for start in range(1, n + 1):
        if color[start]:
            continue
        color[start] = 1
        q = deque([start])
        ok = True
        while q and ok:
            u = q.popleft()
            cu = color[u]
            nv = 3 - cu
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = nv
                    q.append(v)
                elif color[v] == cu:
                    ok = False
                    break
        if not ok:
            sys.stdout.write("NO")
            return

    out = ['YES']
    out.append(' '.join(map(str, color[1:])))
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
