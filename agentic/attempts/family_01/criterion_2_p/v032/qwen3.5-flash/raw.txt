import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
    
    colors = [0] * (n + 1)
    
    # We want to assign colors such that s_i != s_j for edges (i, j)
    # To get lexicographically smallest sequence, we want s_1 to be 1 if possible.
    # If s_1 is part of a component, we set s_1 = 1 and propagate.
    # If s_1 is not part of any component (isolated), s_1 = 1.
    # Generally, for each connected component, we process it starting from the smallest index node in it.
    # We set that node's color to 1. This ensures the lexicographically smallest choice for that component.
    # Since components are independent, this greedy strategy works.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # We force colors[i] = 1 to minimize lexicographically
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = colors[u]
                next_color = 3 - current_color # 1 -> 2, 2 -> 1
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_color
                        queue.append(v)
                    else:
                        if colors[v] != next_color:
                            # Conflict detected
                            print("NO")
                            return

    print("YES")
    print(*(colors[1:]))
