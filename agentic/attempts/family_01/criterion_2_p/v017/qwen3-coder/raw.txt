```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 = unvisited, 1 = block 1, 2 = block 2
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            next_color = 2 if current_color == 1 else 1
            
            for neighbor in adj[node]:
                if color[neighbor] == 0:
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    return False
        return True
    
    # Process each connected component
    for i in range(1, n + 1):
        if color[i] == 0:
            if not bfs(i):
                print("NO")
                return
    
    print("YES")
    # For lexicographically smallest, we want person 1 to be in block 1 if possible
    # If person 1 was colored 2, we need to flip all colors in its component
    # But since we start coloring with 1, and use BFS, it's already optimal
    # Just output the colors (1 and 2)
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append('1')  # Isolated node, assign to block 1
        else:
            result.append(str(color[i]))
    
    print(' '.join(result))

solve()
```
