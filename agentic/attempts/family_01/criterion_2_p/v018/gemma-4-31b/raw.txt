import sys

# Increase recursion depth for deep DFS in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # color[i] will be 0 (unvisited), 1 (Block 1), or 2 (Block 2)
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, for each connected component,
    # we want the smallest indices to have the smallest possible block number.
    # However, once the color of one node in a component is fixed, the rest are determined.
    # For a component, there are only two possible colorings (swap 1s and 2s).
    # To minimize lexicographically, we check the first node (smallest index) in the component.
    # Assigning it '1' will always be lexicographically smaller than assigning it '2',
    # provided a valid 2-coloring exists.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            stack = [(i, 1)]
            color[i] = 1
            component_nodes = []
            
            # Using iterative DFS to avoid recursion limit and manage component
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component_nodes.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # In a bipartite component, we have two choices:
            # Choice A: node i is 1, its neighbors are 2, etc.
            # Choice B: node i is 2, its neighbors are 1, etc.
            # Since we want the lexicographically smallest sequence, we want
            # the smallest index in the component to be 1. 
            # Since we processed nodes in increasing order (i from 1 to n),
            # 'i' is the smallest index in this component.
            # Setting color[i] = 1 is always optimal for the first available node.
            # The logic above already sets color[i] = 1.

    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
