```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # We want the lexicographically smallest configuration.
    # Greedy strategy: iterate fields from 1 to n.
    # For each field u, if not assigned, assign it to 1.
    # Then, all unassigned neighbors v must be assigned to 2.
    # This propagation is done using a queue.
    # If we encounter a conflict (neighbor already assigned same value), it's impossible.
    
    color = [0] * n
    possible = True
    
    for i in range(n):
        if color[i] == 0:
            # Start BFS for this component
            queue = deque([i])
            color[i] = 1
            
            while queue:
                u = queue.popleft()
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
            
    if not possible:
        print("NO")
    else:
        print("YES")
        print(*(color))

if __name__ == '__main__':
    solve()
```
