```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    import threading
    
    def main():
        import sys
        sys.setrecursionlimit(1 << 25)
        
        n, m = map(int, sys.stdin.readline().split())
        
        if m == 0:
            print("YES")
            print(" ".join(["1"] * n))
            return
        
        adj = defaultdict(list)
        edges = []
        
        for _ in range(m):
            u, v = map(int, sys.stdin.readline().split())
            adj[u].append(v)
            adj[v].append(u)
            edges.append((u, v))
        
        color = [0] * (n + 1)
        visited = [False] * (n + 1)
        
        def bfs(start):
            queue = deque()
            queue.append(start)
            color[start] = 1
            visited[start] = True
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 3 - current_color  # Switches between 1 and 2
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        return False
            return True
        
        for i in range(1, n + 1):
            if not visited[i]:
                if not bfs(i):
                    print("NO")
                    return
        
        print("YES")
        result = []
        for i in range(1, n + 1):
            if color[i] == 0:
                result.append("1")
            else:
                result.append(str(color[i]))
        print(" ".join(result))
    
    threading.Thread(target=main).start()

solve()
```
