- For lexicographic minimality: process vertices 1..n; when encountering an uncolored vertex (the smallest-indexed vertex of its component), assign it branch 1 and propagate the forced coloring via DFS. Verified against both examples (example 1 gives `1 2 1 2`, example 2 gives `NO`). Complexity is O(n + m) with iterative DFS (no recursion depth issues), fast I/O, and an adjacency list built via the `zip(it, it)` pairing idiom.

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    n = int(data[0])
    m = int(data[1])

    vals = list(map(int, data[2:2 + 2 * m]))

    adj = [[] for _ in range(n + 1)]
    it = iter(vals)
    for u, v in zip(it, it):
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)

    stack = []
    push = stack.append
    pop = stack.pop

    for s in range(1, n + 1):
        if color[s]:
            continue
        color[s] = 1
        push(s)
        while stack:
            u = pop()
            cu = color[u]
            oc = 3 - cu
            for w in adj[u]:
                cw = color[w]
                if cw == 0:
                    color[w] = oc
                    push(w)
                elif cw == cu:
                    sys.stdout.write("NO\n")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, color[1:])))
    sys.stdout.write("\n")

main()
```
