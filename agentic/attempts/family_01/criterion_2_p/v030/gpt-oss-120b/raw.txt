```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)          # 0 = unvisited, 1 or -1
    for start in range(1, n + 1):
        if color[start]:
            continue
        # start is the smallest index in its component
        q = deque([start])
        color[start] = 1
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if not color[v]:
                    color[v] = -cu
                    q.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    # construct answer: 1 -> branch 1, -1 -> branch 2
    ans = ['1' if color[i] == 1 else '2' for i in range(1, n + 1)]
    print("YES")
    print(' '.join(ans))

if __name__ == "__main__":
    solve()
```
