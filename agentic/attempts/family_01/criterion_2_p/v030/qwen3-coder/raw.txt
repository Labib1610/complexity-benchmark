import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 = unvisited, 1 = branch 1, 2 = branch 2
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS
    for start in range(1, n + 1):
        if color[start] == 0:
            queue = deque([start])
            color[start] = 1
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = 3 - color[node]  # Flip color (1->2, 2->1)
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If we reach here, it's bipartite
    # Now we need to find the lexicographically smallest assignment
    # where employee 1 is in branch 1 if possible
    
    # Try assigning employee 1 to branch 1 first
    result = color[1:]
    
    # If employee 1 is already assigned to branch 2, we need to flip all assignments
    if color[1] == 2:
        for i in range(1, n + 1):
            result[i-1] = 3 - color[i]
    else:
        for i in range(1, n + 1):
            result[i-1] = color[i]
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
