```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    graph = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        graph[u].append(v)
        graph[v].append(u)
        index += 2
    
    color = [-1] * n
    
    for start in range(n):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0
            while queue:
                node = queue.popleft()
                current_color = color[node]
                for neighbor in graph[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - current_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    assignment = []
    for i in range(n):
        if color[i] == -1:
            assignment.append("1")
        else:
            assignment.append(str(color[i] + 1))
    
    print("YES")
    print(" ".join(assignment))

if __name__ == "__main__":
    main()
```
