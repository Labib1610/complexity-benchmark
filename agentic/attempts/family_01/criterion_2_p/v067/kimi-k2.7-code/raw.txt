 ```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    dq = deque()

    for start in range(1, n + 1):
        if color[start]:
            continue
        color[start] = 1
        dq.append(start)
        while dq:
            u = dq.popleft()
            cu = color[u]
            opp = 3 - cu
            for w in adj[u]:
                if color[w] == 0:
                    color[w] = opp
                    dq.append(w)
                elif color[w] != opp:
                    print("NO")
                    return

    sys.stdout.write("YES\n" + " ".join(map(str, color[1:])))

if __name__ == "__main__":
    solve()
```
