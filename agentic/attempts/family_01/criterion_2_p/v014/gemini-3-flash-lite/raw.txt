import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We want the lexicographically smallest sequence (s1, s2, ..., sn)
    # s_i can be 1 or 2. We prefer 1 over 2.
    # This is a standard 2-coloring problem.
    # For each connected component, we can pick the assignment of the first
    # node (smallest index) to be 1, then propagate.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, always try assigning 1 first.
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            # Use stack for DFS to avoid recursion depth issues
            comp_stack = [i]
            while comp_stack:
                u = comp_stack.pop()
                component.append(u)
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        comp_stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # The component is 2-colorable. 
            # We already assigned based on the first node i being 1.
            # This is locally optimal for the lexicographical requirement.

if __name__ == '__main__':
