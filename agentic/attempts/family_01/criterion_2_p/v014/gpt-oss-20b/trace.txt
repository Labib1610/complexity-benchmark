import sys
def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data: return
    it = iter(data)
    n = next(it); m = next(it)
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = next(it); v = next(it)
        adj[u].append(v); adj[v].append(u)
    color = [ -1]*(n+1)
    from collections import deque
    for start in range(1,n+1):
        if color[start] != -1: continue
        # BFS
        color[start] = 0
        dq = deque([start])
        while dq:
            u = dq.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1-cu
                    dq.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO")
                    return
    # output
    out = ['YES\n']
    out.append(' '.join(str(c+1) for c in color[1:]))
    sys.stdout.write(''.join(out))

if __name__ == "__main__":
    main()
