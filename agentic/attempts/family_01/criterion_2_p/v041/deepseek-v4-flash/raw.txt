We can model this as a bipartite graph problem.  
If the graph is not bipartite, it’s impossible.  
If it is bipartite, we can assign colors 1 and 2, but we need the lexicographically smallest assignment.

To get the lexicographically smallest assignment, we can try to assign 1 to as many early nodes as possible, using BFS/DFS with backtracking if needed.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 0
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = color[u] ^ 1
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

# Check bipartiteness
for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            print("NO")
            sys.exit()

# Now we have a valid 2-coloring, but we need lexicographically smallest.
# We can try to flip colors in each connected component to make early nodes 1.
visited = [False] * n

for i in range(n):
    if not visited[i]:
        # BFS to collect component
        comp = []
        q = deque([i])
        visited[i] = True
        while q:
            u = q.popleft()
            comp.append(u)
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    q.append(v)
        # Try to set color[i] = 0 (window 1) if possible
        # We can flip all colors in this component if needed
        # Check if flipping makes i color 0
        if color[i] != 0:
            for u in comp:
                color[u] ^= 1

print("YES")
print(' '.join(str(c + 1) for c in color))
```
