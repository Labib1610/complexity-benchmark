import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        adj[u].append((v, len(edges) - 1))
        adj[v].append((u, len(edges) - 1))
        index += 2
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
        
    # For each edge, the sum of loads must be 3
    # So if one node is 1, the other must be 2, and vice versa.
    # This is equivalent to a 2-coloring (bipartite check) of the graph formed by the edges.
    
    color = [0] * (n + 1)  # 0: unvisited, 1: color 1, 2: color 2
    
    # We'll do BFS on each component
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if not visited[start]:
            queue = deque()
            queue.append(start)
            color[start] = 1
            visited[start] = True
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                opposite_color = 3 - current_color  # flips 1->2, 2->1
                
                for v, idx in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = opposite_color
                        queue.append(v)
                    else:
                        # Check consistency
                        if color[v] != opposite_color:
                            print("NO")
                            return
    
    # If we reach here, a valid coloring exists
    # We need the lexicographically smallest one
    # That means we prefer color 1 over color 2 for nodes that can be chosen
    
    # Actually, since we're doing BFS from lowest numbered node first,
    # and always assigning color 1 to the starting node, this should give us
    # the lex smallest valid assignment.
    
    # But let's think more carefully about "lexicographically smallest"
    # It means we want the first node to be 1 if possible.
    # In our algorithm, the first node we visit in a component gets color 1.
    # To get the lex smallest, we should ensure that in each component,
    # we start BFS from the smallest numbered node, and we process components
    # in order of their smallest node.
    
    # Let's re-implement with proper handling for lex order
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if not visited[start]:
            # Start BFS
            queue = deque()
            queue.append(start)
            color[start] = 1
            visited[start] = True
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                opposite_color = 3 - current_color
                
                # Sort neighbors to maintain consistent processing order
                adj[u].sort()
                for v, idx in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = opposite_color
                        queue.append(v)
                    else:
                        # Consistency check
                        if color[v] != opposite_color:
                            print("NO")
                            return
    
    # Construct result
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            # Isolated node, can choose 1 to minimize lex order
            result.append("1")
        else:
            result.append(str(color[i]))
            
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
