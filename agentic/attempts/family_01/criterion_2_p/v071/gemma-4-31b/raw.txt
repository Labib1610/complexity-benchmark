import sys

# Increase recursion depth for deep DFS in connected components
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # color[i] = 0: unvisited, 1: Vault 1, 2: Vault 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence s_1, s_2, ..., s_n,
    # we process components in order of their smallest node index.
    # For each component, we want to assign the smallest possible value (1) 
    # to the smallest index in that component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To ensure lexicographically smallest, the first node of the component
            # (which is i) should be assigned 1 if possible.
            stack = [(i, 1)]
            color[i] = 1
            component_nodes = []
            
            # Iterative DFS to avoid recursion limits and find all nodes in component
            q = [i]
            visited_in_comp = {i}
            idx = 0
            while idx < len(q):
                u = q[idx]
                idx += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The component is bipartite. 
            # We assigned color[i] = 1. This is the best we can do for the 
            # smallest index in the component to minimize the sequence.
            # Since the component is connected, the coloring is forced once 
            # the color of the first node is chosen.
            # Any other choice (color[i] = 2) would result in a lexicographically 
            # larger sequence because the first differing element s_i would be 2 instead of 1.

    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
