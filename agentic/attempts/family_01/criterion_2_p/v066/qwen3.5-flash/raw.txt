def solve():

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))

    except StopIteration:

    # Iterate through all nodes to handle disconnected components
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start a traversal for the component containing i
            # We assign color 1 to i to ensure lexicographically smallest sequence
            colors[i] = 1
            stack = [i]
            
            while stack:
                u = stack.pop()
                current_color = colors[u]
                next_color = 3 - current_color  # If 1 -> 2, if 2 -> 1

                        colors[v] = next_color
                        stack.append(v)
                    elif colors[v] != next_color:
                        # Conflict detected: adjacent nodes have same color
                        print("NO")
                        return
    
    # If no conflicts found
    print("YES")
    print(' '.join(map(str, colors[1:])))
