import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # 0/1 provisional colors
    ans = [0] * (n + 1)             # final shifts 1/2

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == cu:
                    print("NO")
                    return
        # decide orientation for lexicographically smallest assignment
        comp.sort()                     # vertices in increasing order
        # compare two possible assignments
        better = 0  # 0 -> keep as is (0->1,1->2), 1 -> flip
        for v in comp:
            a = 1 if color[v] == 0 else 2
            b = 2 if color[v] == 0 else 1
            if a < b:
                better = 0
                break
            if b < a:
                better = 1
                break
        # apply chosen orientation
        if better == 0:
            for v in comp:
                ans[v] = 1 if color[v] == 0 else 2
        else:
            for v in comp:
                ans[v] = 2 if color[v] == 0 else 1

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
