import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # color[i] = 0: unvisited, 1: Mode 1, 2: Mode 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we want to assign 
    # Mode 1 to component i as early as possible.
    # For each connected component, we try assigning Mode 1 to the smallest 
    # index in that component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color this connected component
            # We force the smallest index in the component to be 1
            stack = [i]
            color[i] = 1
            
            # Use iterative DFS to avoid recursion limit and for speed
            visited_in_comp = [i]
            idx = 0
            while idx < len(visited_in_comp):
                u = visited_in_comp[idx]
                idx += 1
                
                curr_color = color[u]
                next_color = 2 if curr_color == 1 else 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_color
                        visited_in_comp.append(v)
                    elif color[v] == curr_color:
                        print("NO")
                        return
    
    # The logic above assigns Mode 1 to the first node of every component.
    # Since we process nodes from 1 to n, and for each component we 
    # greedily pick the smallest available index to be Mode 1,
    # this naturally leads to the lexicographically smallest sequence.
    
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
