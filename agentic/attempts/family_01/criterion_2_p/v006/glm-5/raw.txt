import sys
from sys import setrecursionlimit

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    stack = []

    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        color[start] = 1
        stack.append(start)
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - cu
                    stack.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return

    # Greedy lexicographically smallest assignment
    assign = [0] * (n + 1)
    for u in range(1, n + 1):
        if assign[u] != 0:
            continue
        assign[u] = 1
        stack = [u]
        while stack:
            x = stack.pop()
            cx = assign[x]
            for y in adj[x]:
                if assign[y] == 0:
                    assign[y] = 3 - cx
                    stack.append(y)

    out = ["YES"]
    out.append(" ".join(str(assign[i]) for i in range(1, n + 1)))
    sys.stdout.write("\n".join(out) + "\n")

if __name__ == "__main__":
    solve()
