import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n          # 0 or 1 provisional
    ans = [0] * n             # final 1/2 values
    from collections import deque

    for start in range(n):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while q:
            v = q.popleft()
            cv = color[v]
            for to in g[v]:
                if color[to] == -1:
                    color[to] = cv ^ 1
                    q.append(to)
                    comp.append(to)
                elif color[to] == cv:
                    print("NO")
                    return
        # decide orientation for lexicographically smallest
        comp.sort()                     # increasing indices
        seq1 = [1 if color[v] == 0 else 2 for v in comp]
        seq2 = [2 if color[v] == 0 else 1 for v in comp]
        if seq2 < seq1:                 # flip needed
            for v in comp:
                color[v] ^= 1
        # write final answer
        for v in comp:
            ans[v] = 1 if color[v] == 0 else 2

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
