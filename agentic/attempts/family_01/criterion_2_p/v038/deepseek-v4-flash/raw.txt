We can model this as a graph bipartiteness check.  
If the graph is not bipartite, output `NO`.  
Otherwise, we need the lexicographically smallest assignment, which means we try to assign `1` to as many early modules as possible, but we must respect the bipartite coloring.

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(1 << 20)
input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
possible = True

# First, try to color with BFS/DFS, but for lexicographically smallest,
# we process nodes in order and assign 1 if possible.
for i in range(n):
    if color[i] == -1:
        color[i] = 1
        stack = [i]
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 3 - color[u]
                    stack.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    # Now we have a valid 2-coloring, but we want lexicographically smallest.
    # We can flip entire components if it makes earlier nodes 1.
    # We'll do BFS/DFS again, but for each unvisited node, try color 1 first.
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            color[i] = 1
            stack = [i]
            comp = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 3 - color[u]
                        stack.append(v)
                        comp.append(v)
            # Check if flipping this component gives lexicographically smaller
            # We compare: if first node in comp is 1 already, keep; else flip
            # But since we process in order, we only flip if it makes earlier index 1
            # Actually we already assigned 1 to the first unvisited node, so it's fine.
    print("YES")
    print(" ".join(str(c) for c in color))
```
