 ```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n
    possible = True

    for i in range(n):
        if color[i] != 0:
            continue
        color[i] = 1
        dq = deque([i])
        while dq:
            u = dq.popleft()
            cu = color[u]
            for w in adj[u]:
                if color[w] == 0:
                    color[w] = 3 - cu
                    dq.append(w)
                elif color[w] == cu:
                    possible = False
                    dq.clear()
                    break
        if not possible:
            break

    if not possible:
        sys.stdout.write("NO")
    else:
        sys.stdout.write("YES\n")
        sys.stdout.write(' '.join(str(color[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
