import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    graph = defaultdict(list)
    index = 2
    edges = []
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        index += 2
        graph[u].append(v)
        graph[v].append(u)
        edges.append((u, v))
    
    # Check if bipartite and find lexicographically smallest valid configuration
    color = [-1] * n
    queue = deque()
    
    for start in range(n):
        if color[start] == -1:
            color[start] = 0
            queue.append(start)
            while queue:
                node = queue.popleft()
                for neighbor in graph[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # Convert colors to deployment modes: 0 -> 1, 1 -> 2
    # To get lexicographically smallest, we want satellite 1 to have mode 1 if possible
    result = []
    for i in range(n):
        if color[i] == 0:
            result.append("1")
        else:
            result.append("2")
    
    # Verify all constraints are satisfied
    for u, v in edges:
        if (color[u] + 1) + (color[v] + 1) != 3:
            print("NO")
            return
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
