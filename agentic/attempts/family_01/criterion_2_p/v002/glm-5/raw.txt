import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    stack = []
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        color[start] = 1
        stack.append(start)
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - cu
                    stack.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return
    
    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(str(color[i]) for i in range(1, n + 1)) + '\n')

solve()
