```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 means unvisited, 1 and 2 are the two colors
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
            
        queue = deque()
        queue.append(start)
        color[start] = 1  # Start with color 1
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            next_color = 3 - current_color  # flips 1->2 and 2->1
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = next_color
                    queue.append(v)
                elif color[v] == current_color:
                    print("NO")
                    return
    
    # If we reach here, it's bipartite
    # The coloring gives us a valid assignment
    # We need the lexicographically smallest one
    # That means we prefer assigning 1 to nodes if possible
    
    # To get lex smallest, we can just make sure node 1 gets color 1
    # But since we already did BFS starting from 1, and set color[1]=1,
    # and the rest followed accordingly, this should be fine.
    # Actually, our BFS might not start from node 1 if it's isolated.
    # Let's re-do the coloring to ensure lex smallest.
    
    # A better approach: 
    # Run BFS from each component, always assigning the first node color 1
    
    color = [0] * (n + 1)
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
            
        queue = deque()
        queue.append(start)
        color[start] = 1  # Always assign 1 to the starting node of a component
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            next_color = 3 - current_color
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = next_color
                    queue.append(v)
                elif color[v] == current_color:
                    print("NO")
                    return
                    
    print("YES")
    print(" ".join(str(color[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
```
