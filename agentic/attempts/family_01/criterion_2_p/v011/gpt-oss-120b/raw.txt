import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)   # 0 or 1
    ans = [0] * (n + 1)

    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        # start new component, assign i color 0 to make it bus 1
        dq = deque([i])
        color[i] = 0
        comp_nodes = [i]
        ok = True
        while dq:
            u = dq.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    dq.append(v)
                    comp_nodes.append(v)
                elif color[v] == cu:
                    print("NO")
                    return
        # assign buses: color 0 -> 1, color 1 -> 2
        for node in comp_nodes:
            ans[node] = 1 if color[node] == 0 else 2

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
