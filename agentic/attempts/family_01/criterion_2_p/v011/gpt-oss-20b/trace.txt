def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    data = sys.stdin.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    color = [-1]*(n+1)
    result = [0]*(n+1)
    from collections import deque
    for i in range(1,n+1):
        if color[i]==-1:
            dq=deque([i]); color[i]=0
            comp_nodes=[i]
            is_bip=True
            while dq:
                u=dq.popleft()
                cu=color[u]
                for v in adj[u]:
                    if color[v]==-1:
                        color[v]=cu^1
                        comp_nodes.append(v)
                        dq.append(v)
                    elif color[v]==cu:
                        print('NO')
                        return
            # after component
            min_node = min(comp_nodes)
            min_color = color[min_node]
            if min_color==0:
                for node in comp_nodes:
                    result[node]=color[node]+1
            else:
                for node in comp_nodes:
                    result[node]=2 - color[node]
    # after all
    out = ['YES', ' '.join(str(result[i]) for i in range(1,n+1))]
    sys.stdout.write('\n'.join(out))
