import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    comp = [-1] * n
    comp_min = []
    q = deque()

    for i in range(n):
        if color[i] != -1:
            continue
        cid = len(comp_min)
        color[i] = 0
        comp[i] = cid
        q.append(i)
        mn = i
        while q:
            u = q.popleft()
            if u < mn:
                mn = u
            cu = color[u]
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = cu ^ 1
                    comp[w] = cid
                    q.append(w)
                elif color[w] == cu:
                    sys.stdout.write("NO\n")
                    return
        comp_min.append(mn)

    out = []
    for i in range(n):
        mn = comp_min[comp[i]]
        if color[mn] == 0:
            out.append(str(color[i] + 1))
        else:
            out.append(str(2 - color[i]))
    sys.stdout.write("YES\n" + " ".join(out))

if __name__ == "__main__":
    solve()
