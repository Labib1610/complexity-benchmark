def spectrum_allocation(n, m, edges):
    from collections import defaultdict, deque
    
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    colors = [0] * (n + 1)  # 0 means uncolored, 1 and 2 are the two channels
    
    def bfs(start):
        queue = deque([start])
        colors[start] = 1  # Start with color 1
        while queue:
            node = queue.popleft()
            current_color = colors[node]
            next_color = 2 if current_color == 1 else 1
            for neighbor in graph[node]:
                if colors[neighbor] == 0:  # Uncolored
                    colors[neighbor] = next_color
                    queue.append(neighbor)
                elif colors[neighbor] == current_color:  # Conflict found
                    return False
        return True

    for i in range(1, n + 1):
        if colors[i] == 0:  # Not visited
            if not bfs(i):
                print("NO")
                return
    
    # Create the lexicographically smallest result
    result = [0] * n
    for i in range(1, n + 1):
        result[i - 1] = colors[i]

    print("YES")
    print(" ".join(map(str, result)))

import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]
spectrum_allocation(n, m, edges)
