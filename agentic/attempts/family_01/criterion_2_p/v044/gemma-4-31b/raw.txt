import sys

# Increase recursion depth for deep DFS in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Read n and m from standard input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will be 0 if unvisited, 1 or 2 if assigned a channel
    color = [0] * (n + 1)
    
    # To find the lexicographically smallest sequence, we process components
    # in order of the smallest available node index.
    # For each component, we have two choices for the starting node's color (1 or 2).
    # To make the sequence lexicographically smallest, we want the smallest indices
    # to have the smallest possible colors.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # This is a new connected component.
            # We want the smallest index in this component to be 1 if possible.
            # Since the graph is bipartite, once the first node's color is fixed,
            # the rest are determined.
            
            component = []
            stack = [i]
            color[i] = 1 # Try assigning 1 to the smallest index in the component
            
            # BFS or iterative DFS to avoid recursion limits and find all nodes in component
            visited_in_comp = []
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                visited_in_comp.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We need to check if swapping 1 <-> 2 would result in a lexicographically smaller sequence.
            # The sequence is determined by the values at indices in increasing order.
            # Let's find the smallest index in this component. It's already 'i'.
            # We assigned color[i] = 1. Since 1 is the smallest possible value for the 
            # first element of the component encountered in the sequence (s_1, ..., s_n),
            # assigning color[i] = 1 is always optimal for the lexicographical requirement.
            
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
