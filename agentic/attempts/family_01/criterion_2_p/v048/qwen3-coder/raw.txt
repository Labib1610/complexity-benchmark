```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    graph = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        graph[u].append(v)
        graph[v].append(u)
        index += 2
    
    color = [-1] * n
    visited = [False] * n
    
    for start in range(n):
        if visited[start]:
            continue
        queue = deque()
        queue.append(start)
        visited[start] = True
        color[start] = 0
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            next_color = 1 - current_color
            
            for neighbor in graph[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] != next_color:
                    print("NO")
                    return
    
    # If we reach here, the graph is bipartite
    result = []
    for i in range(n):
        if color[i] == -1:
            result.append('1')
        else:
            result.append(str(color[i] + 1))
    
    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    main()
```
