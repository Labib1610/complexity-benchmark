```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        g[u].append(v)
        g[v].append(u)

    color = [ -1 ] * (n + 1)          # 0/1 provisional colors
    answer = [0] * (n + 1)            # final phases 1/2

    from collections import deque

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS to collect component and provisional colors
        comp_nodes = []
        dq = deque([start])
        color[start] = 0
        ok = True
        while dq:
            v = dq.popleft()
            comp_nodes.append(v)
            for to in g[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    dq.append(to)
                elif color[to] == color[v]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            print("NO")
            return

        # decide orientation: we want the smallest index in this component (which is start)
        # to have phase 1.
        # If provisional color of start is 0, map 0->1,1->2; else map 0->2,1->1.
        if color[start] == 0:
            map01 = (1, 2)
        else:
            map01 = (2, 1)

        for v in comp_nodes:
            answer[v] = map01[color[v]]

    print("YES")
    print(' '.join(str(answer[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
