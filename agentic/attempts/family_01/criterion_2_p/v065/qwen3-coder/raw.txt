```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)  # 0 means unvisited, 1 and 2 are the two colors/facilities
    visited = [False] * (n + 1)
    
    # Check if bipartite using BFS for all components
    for start in range(1, n + 1):
        if visited[start]:
            continue
            
        queue = deque()
        queue.append(start)
        visited[start] = True
        color[start] = 1
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            next_color = 2 if current_color == 1 else 1
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] != next_color:
                    print("NO")
                    return
    
    # If we reach here, it's bipartite. Now produce the lex smallest coloring.
    # Since we're assigning 1 and 2, and we want lexicographically smallest,
    # we should try to assign 1 to the component containing node 1 if possible.
    # But our BFS already ensures a valid 2-coloring. We just need to make sure
    # for each component, the smaller color comes first in lex order.
    
    # Actually, since we're building the assignment node by node and we always
    # start with color 1 for the first node of each component, this already gives
    # us a valid coloring. But to get the lex smallest, if a component can be flipped,
    # we choose the version that makes the first node in the component (in terms of
    # numbering) have the smaller value.
    
    # Let's re-do the coloring to ensure lexicographically smallest result.
    # For that, we can do a single BFS/DFS starting from node 1, then node 2, etc.
    
    final_color = [0] * (n + 1)
    visited_final = [False] * (n + 1)
    
    for start_node in range(1, n + 1):
        if visited_final[start_node]:
            continue
            
        queue = deque()
        queue.append(start_node)
        visited_final[start_node] = True
        final_color[start_node] = 1  # Always start with 1 for the first node in component
        
        while queue:
            node = queue.popleft()
            current_col = final_color[node]
            next_col = 2 if current_col == 1 else 1
            
            for neighbor in adj[node]:
                if not visited_final[neighbor]:
                    visited_final[neighbor] = True
                    final_color[neighbor] = next_col
                    queue.append(neighbor)
                elif final_color[neighbor] != next_col:
                    print("NO")
                    return
    
    print("YES")
    print(" ".join(str(final_color[i]) for i in range(1, n + 1)))

solve()
```
