We can model this as a bipartite graph problem: each edge requires its two endpoints to have different tiers (1 and 2).  
So the graph must be bipartite. If it is, we can assign tiers by coloring.

To get the lexicographically smallest assignment, we try to assign tier 1 to as many early nodes as possible.  
We can do a BFS/DFS that tries to color node 1 as 1, then propagates, and if a conflict occurs, we try the opposite coloring for the first component.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start, first_color):
    q = deque([start])
    color[start] = first_color
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 1 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

# Try coloring node 0 as 1 first
ok = True
for i in range(n):
    if color[i] == -1:
        if not bfs(i, 1):
            ok = False
            break

if not ok:
    # Try coloring node 0 as 2 instead
    color = [-1] * n
    ok = True
    for i in range(n):
        if color[i] == -1:
            if not bfs(i, 0):
                ok = False
                break

if not ok:
    print("NO")
else:
    print("YES")
    print(" ".join(str(c + 1) for c in color))
```
