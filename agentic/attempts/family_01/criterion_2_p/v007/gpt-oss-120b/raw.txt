```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # 0/1 within component
    answer = [0] * (n + 1)          # final shift (1 or 2)

    from collections import deque

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp_nodes = [start]
        min_node = start
        bip = True
        while q:
            u = q.popleft()
            if u < min_node:
                min_node = u
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                elif color[v] == cu:
                    bip = False
        if not bip:
            print("NO")
            return
        # decide whether to flip so that min_node gets shift 1
        flip = (color[min_node] == 1)   # if its current color is 1, we need to flip
        for v in comp_nodes:
            c = color[v] ^ (1 if flip else 0)
            answer[v] = 1 if c == 0 else 2

    # all components processed
    print("YES")
    sys.stdout.write(' '.join(str(answer[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
