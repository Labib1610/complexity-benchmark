```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    parity = [ -1 ] * (n + 1)   # -1 unvisited, else 0/1
    crew = [0] * (n + 1)

    for start in range(1, n + 1):
        if parity[start] != -1:
            continue
        # BFS to get component and parity
        q = deque([start])
        parity[start] = 0
        comp = [start]
        ok = True
        while q and ok:
            u = q.popleft()
            pu = parity[u]
            for v in adj[u]:
                if parity[v] == -1:
                    parity[v] = pu ^ 1
                    q.append(v)
                    comp.append(v)
                elif parity[v] == pu:
                    ok = False
                    break
        if not ok:
            print("NO")
            return
        # decide mapping to get lexicographically smallest:
        # smallest index in component is 'start' (we iterate in order)
        # we want crew[start] = 1, so parity[start] maps to 1
        for v in comp:
            crew[v] = 1 if parity[v] == parity[start] else 2

    print("YES")
    print(' '.join(str(crew[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
