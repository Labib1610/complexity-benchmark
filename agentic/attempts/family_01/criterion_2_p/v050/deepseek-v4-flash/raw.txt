 We can solve this using a graph bipartite check with a greedy coloring approach.  
We’ll assign crew 1 or 2 while doing BFS/DFS, and whenever we have a choice, we pick the smaller crew number to ensure lexicographical minimality.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n + 1)]

for _ in range(m):
    u, v = map(int, input().split())
    adj[u].append(v)
    adj[v].append(u)

color = [0] * (n + 1)  # 0 = uncolored, 1 or 2 = crew

def bfs(start):
    q = deque([start])
    color[start] = 1
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == 0:
                # assign opposite color, but choose smallest possible
                color[v] = 3 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(1, n + 1):
    if color[i] == 0:
        # Try to assign 1 if possible, else 2
        color[i] = 1
        if not bfs(i):
            # If conflict, try starting with 2
            color = [0] * (n + 1)
            color[i] = 2
            if not bfs(i):
                print("NO")
                sys.exit()

print("YES")
print(" ".join(str(color[i]) for i in range(1, n + 1)))
```
