import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a component is not colored, we try to start with color 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            
            # Use a list to store nodes in the current component to sort/process
            # However, standard 2-coloring is fixed once the first node is chosen.
            # To ensure lexicographical smallest, we must be careful.
            # Actually, for each connected component, there are only 2 colorings (swap 1 and 2).
            # We want the one that makes the sequence (s1, s2, ..., sn) smallest.
            # This is equivalent to saying for the first node 'i' in the component,
            # we want s[i] = 1.
            
            component = []
            q = [i]
            color[i] = 1
            component.append(i)
            
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The assignment is already fixed by color[i] = 1.
            # Since we iterate 1..n, the first node 'i' of any component
            # will always be assigned 1. This is the lexicographically smallest
            # choice for that component.
            
    print("YES")
    print(*(color[1:]))

solve()
