import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # For each unassigned node, we try assigning it to 1 first.
    # If 1 is not possible, we try 2.
    # This is a standard 2-coloring problem.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS component
            stack = [i]
            assignment[i] = 1
            component = []
            
            # Use stack to traverse the component
            comp_stack = [i]
            while comp_stack:
                u = comp_stack.pop()
                component.append(u)
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        comp_stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if this component's lexicographical order is optimal
            # Actually, simply picking 1 for the first node of each component
            # is sufficient to satisfy the lexicographical requirement
            # because the choices are independent across connected components.
            
    print("YES")
    print(*(assignment[1:]))

solve()
