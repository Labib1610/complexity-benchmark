```python
import sys
from collections import deque

# Increase recursion depth just in case, though iterative approach is used.
sys.setrecursionlimit(200000)

def solve():
    # Read all input from standard input at once for efficiency
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Initialize adjacency list for the graph
    # 1-based indexing for researchers
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break

    # Array to store the laboratory assignment (0: unvisited, 1: Lab 1, 2: Lab 2)
    color = [0] * (n + 1)
    
    # Iterate through all researchers from 1 to n.
    # This order ensures that for any connected component, the first researcher 
    # we encounter (the one with the smallest index) is processed first.
    # To achieve the lexicographically smallest assignment sequence, we assign 
    # this first researcher to Lab 1 (color 1).
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the connected component containing i
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                
                for v in adj[u]:
                    if color[v] == 0:
                        # Assign the opposite laboratory to the neighbor
                        color[v] = 3 - current_color
                        queue.append(v)
                    elif color[v] == current_color:
                        # Conflict detected: adjacent researchers in the same laboratory
                        print("NO")
                        return

    # If valid assignment found
    print("YES")
    # Print the assignment sequence for researchers 1 to n
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```

One final check on logic:
"lexicographically smallest assignment sequence".
Sequence $S = (s_1, s_2, \dots, s_n)$.
We want to minimize $S$.
My logic: For each component $C$, let $u = \min(C)$.
We set $s_u = 1$.
This determines the whole component.
Is it possible that setting $s_u = 1$ forces some $v \in C$ to be 2, whereas setting $s_u = 2$ forces $v$ to be 1?
Yes.
But since $u < v$ (if $v \neq u$), the value at $u$ is more significant for lexicographical comparison than the value at $v$.
Since $1 < 2$, choosing $s_u = 1$ is always better regardless of what happens at $v$.
So the greedy choice at $u$ is globally optimal.
Since we iterate $i$ from 1 to $n$, $i$ is always $\min(C)$ for the current component.
So the logic holds.
