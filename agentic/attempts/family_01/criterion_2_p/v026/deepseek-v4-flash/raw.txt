We can model this as a graph bipartiteness check.  
If the graph is not bipartite, output `NO`.  
Otherwise, we need the lexicographically smallest assignment, which we can get by a BFS/DFS that assigns `1` to the first unvisited node and propagates alternating colors, but with a twist: we must try to assign `1` as early as possible.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 1
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 3 - color[u]
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            print("NO")
            sys.exit()

print("YES")
print(" ".join(str(color[i]) for i in range(n)))
```
