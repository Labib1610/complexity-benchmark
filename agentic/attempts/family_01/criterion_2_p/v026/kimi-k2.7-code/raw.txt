 ```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    ans = [0] * n

    for start in range(n):
        if color[start] != -1:
            continue

        dq = deque([start])
        color[start] = 0
        comp = []
        minv = start
        ok = True

        while dq:
            v = dq.popleft()
            comp.append(v)
            if v < minv:
                minv = v
            cv = color[v]
            for w in adj[v]:
                if color[w] == -1:
                    color[w] = cv ^ 1
                    dq.append(w)
                elif color[w] == cv:
                    ok = False

        if not ok:
            sys.stdout.write("NO")
            return

        if color[minv] == 0:
            map0, map1 = 1, 2
        else:
            map0, map1 = 2, 1

        for v in comp:
            ans[v] = map0 if color[v] == 0 else map1

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
