import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS to check bipartiteness
        queue = [start]
        color[start] = 1
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            cu = color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - cu
                    queue.append(v)
                elif color[v] == cu:
                    print("NO")
                    return
    
    # Now build the 2-SAT implication graph
    # Variable x_i: worker i is in shift 1
    # x_i = True means shift 1, x_i = False means shift 2
    # For each edge (u, v): x_u != x_v
    #   x_u => ~x_v, ~x_v => x_u
    #   x_v => ~x_u, ~x_u => x_v
    # Node 2*i-1 = x_i, 2*i = ~x_i
    
    N = 2 * n
    g = [[] for _ in range(N + 1)]
    
    for u in range(1, n + 1):
        for v in adj[u]:
            if v > u:
                # x_u => ~x_v
                g[2 * u - 1].append(2 * v)
                # ~x_v => x_u
                g[2 * v].append(2 * u - 1)
                # x_v => ~x_u
                g[2 * v - 1].append(2 * u)
                # ~x_u => x_v
                g[2 * u].append(2 * v - 1)
    
    # Kosaraju SCC
    visited = [False] * (N + 1)
    order = []
    
    sys.setrecursionlimit(1 << 25)
    
    # Iterative DFS for finishing order
    for i in range(1, N + 1):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                node, idx = stack[-1]
                if idx == 0:
                    visited[node] = True
                if idx < len(g[node]):
                    child = g[node][idx]
                    stack[-1] = (node, idx + 1)
                    if not visited[child]:
                        stack.append((child, 0))
                else:
                    stack.pop()
                    order.append(node)
    
    # Reverse graph
    gr = [[] for _ in range(N + 1)]
    for u in range(1, N + 1):
        for v in g[u]:
            gr[v].append(u)
    
    comp = [0] * (N + 1)
    cid = 0
    visited = [False] * (N + 1)
    
    for node in reversed(order):
        if not visited[node]:
            cid += 1
            stack = [node]
            visited[node] = True
            while stack:
                u = stack.pop()
                comp[u] = cid
                for v in gr[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
    
    # Check satisfiability
    for i in range(1, n + 1):
        if comp[2 * i - 1] == comp[2 * i]:
            print("NO")
            return
    
    # Build condensation DAG
    outdegree = [0] * (cid + 1)
    for u in range(1, N + 1):
        for v in g[u]:
            if comp[u] != comp[v]:
                outdegree[comp[u]] += 1
    
    from collections import deque
    q = deque()
    for c in range(1, cid + 1):
        if outdegree[c] == 0:
            q.append(c)
    
    topo = []
    while q:
        c = q.popleft()
        topo.append(c)
        for u in range(1, N + 1):
            if comp[u] == c:
                for v in gr[u]:
                    if comp[v] != c:
                        outdegree[comp[v]] -= 1
                        if outdegree[comp[v]] == 0:
                            q.append(comp[v])
    
    assign = [False] * (cid + 1)
    for c in topo:
        if not assign[c]:
            assign[c] = True
            # Find the complement component
            for u in range(1, N + 1):
                if comp[u] == c:
                    if u % 2 == 1:
                        comp_c = comp[u + 1]
                    else:
                        comp_c = comp[u - 1]
                    assign[comp_c] = False
                    break
    
    result = []
    for i in range(1, n + 1):
        if assign[comp[2 * i - 1]]:
            result.append('1')
        else:
            result.append('2')
    
    print("YES")
    print(' '.join(result))

solve()
