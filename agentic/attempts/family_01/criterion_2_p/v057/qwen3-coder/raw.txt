```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        # No constraints, assign all to mode 1 (lexicographically smallest)
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    graph = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
        index += 2
    
    # Color coding: 0 - unvisited, 1 - color 1, 2 - color 2
    color = [0] * (n + 1)
    
    # Check if bipartite (2-colorable) and find coloring
    for start in range(1, n + 1):
        if color[start] == 0:
            queue = deque()
            queue.append(start)
            color[start] = 1  # Assign color 1 to starting node
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in graph[node]:
                    if color[neighbor] == 0:
                        # Assign opposite color
                        color[neighbor] = 3 - current_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        # Conflict found, not bipartite
                        print("NO")
                        return
    
    # If we reach here, it's bipartite. Now we need lexicographically smallest assignment.
    # We want to minimize p_1, then p_2, etc.
    # In our coloring, nodes with color 1 can be assigned mode 1 or 2, and color 2 can be assigned mode 2 or 1 respectively.
    # But since we want lexicographically smallest, we should assign:
    # color 1 -> mode 1
    # color 2 -> mode 2
    # This ensures minimal lexicographical order.
    
    result = []
    for i in range(1, n + 1):
        if color[i] == 1:
            result.append('1')
        else:
            result.append('2')
            
    print("YES")
    print(" ".join(result))

solve()
```
