import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    ans = [0] * n

    for start in range(n):
        if color[start] != -1:
            continue
        q = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while q:
            v = q.popleft()
            for w in adj[v]:
                if color[w] == -1:
                    color[w] = color[v] ^ 1
                    q.append(w)
                    comp.append(w)
                elif color[w] == color[v]:
                    ok = False
        if not ok:
            sys.stdout.write("NO")
            return

        min_v = min(comp)
        if color[min_v] == 0:
            val0, val1 = 1, 2
        else:
            val0, val1 = 2, 1
        for v in comp:
            ans[v] = val0 if color[v] == 0 else val1

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
