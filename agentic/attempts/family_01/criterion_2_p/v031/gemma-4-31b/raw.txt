import sys

# Increase recursion depth for deep DFS in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will be 0 (unvisited), 1, or 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, for each connected component,
    # we want to assign the smallest possible value to the smallest index in that component.
    # Since the graph is bipartite, if we fix the color of one node, the rest of the 
    # component's colors are forced.
    # For a component, we have two choices: 
    # 1. Root node gets color 1 (then others are determined)
    # 2. Root node gets color 2 (then others are determined)
    # To minimize lexicographically, we want the smallest index in the component to be 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to find all nodes in this component
            component = []
            stack = [i]
            color[i] = 1 # Temporarily assume root is 1
            
            # Using a stack for iterative DFS to avoid recursion limits and find component
            visited_in_comp = {i}
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We have two options for the component: 
            # Option A: Use the current colors (root i is 1).
            # Option B: Flip all colors (root i becomes 2).
            # To make the sequence lexicographically smallest, we look at the 
            # smallest index in the component. We already started with the smallest index i.
            # Assigning color[i] = 1 is always better than color[i] = 2 for lexicographical order.
            # Thus, the current assignment (where color[i]=1) is already the best.
            
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
