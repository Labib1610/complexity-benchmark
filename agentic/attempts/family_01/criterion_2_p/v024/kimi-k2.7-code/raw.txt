```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [[] for _ in range(n + 1)]
    p = 2
    for _ in range(m):
        u = int(data[p])
        v = int(data[p + 1])
        p += 2
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    q = deque()

    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        color[i] = 0
        q.append(i)
        while q:
            u = q.popleft()
            cu = color[u]
            for w in adj[u]:
                cw = color[w]
                if cw == -1:
                    color[w] = cu ^ 1
                    q.append(w)
                elif cw == cu:
                    sys.stdout.write("NO")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(str(color[i] + 1) for i in range(1, n + 1)))

solve()
```
