import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # We want lexicographically smallest assignment.
    # We iterate i from 1 to n. If not assigned, try assigning 1.
    # If 1 leads to contradiction, backtrack or fail.
    # This is a standard 2-coloring problem.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for the connected component
            stack = [i]
            assignment[i] = 1
            component = []
            
            # Using a stack for iterative DFS to avoid recursion limit issues
            q = [i]
            while q:
                u = q.pop()
                component.append(u)
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # For this component, we have two possible colorings:
            # 1. Start with assignment[i] = 1
            # 2. Start with assignment[i] = 2
            # Since we want the lexicographically smallest, we check if 
            # the current component's assignment is already the best.
            # Actually, because we process nodes 1 to n, if we encounter an 
            # unassigned node, we must pick the smallest available channel (1).
            # This greedy choice is globally optimal for lexicographical order.

solve()
