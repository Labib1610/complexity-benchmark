def solve():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    graph = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)
    
    color = [0] * (n + 1)  # 0 means uncolored, 1 and 2 are the two colors
    
    def bfs(start):
        queue = deque([start])
        color[start] = 1  # Start coloring with 1
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == 0:  # If not colored
                    color[neighbor] = 2 if color[node] == 1 else 1
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:  # Conflict detected
                    return False
        return True
    
    for i in range(1, n + 1):
        if color[i] == 0:  # Not colored yet
            if not bfs(i):
                print("NO")
                return
    
    print("YES")
    print(" ".join(map(str, color[1:])))
