```python
import sys

# Increase recursion depth just in case, though we will use iterative BFS/DFS
sys.setrecursionlimit(300000)

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    # 0: uncolored, 1: shift 1, 2: shift 2
    
    # We need to iterate 1 to n to ensure lexicographical smallest
    # But actually we just need to iterate and process components.
    # The order of processing components is implicitly determined by the loop 1..n.
    
    # However, we can just run a loop 1..n.
    # If node i is uncolored, it's the smallest index in its component.
    # We must set colors[i] = 1.
    
    # Let's use a stack for DFS or queue for BFS. BFS is fine.
    
    from collections import deque
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start a new component
            # We want colors[i] = 1
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = colors[u]
                next_color = 3 - current_color

                    if colors[v] == 0:
                        colors[v] = next_color
                        queue.append(v)
                    elif colors[v] == current_color:
                        # Conflict
                        print("NO")
                        return

    # If we reach here, a valid assignment was found
    print("YES")
    # Output the sequence s_1 ... s_n
    print(*(colors[1:]))
```

```python
import sys
from collections import deque

# Increase recursion depth just in case, though iterative approach is used.
sys.setrecursionlimit(200005)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    # Nodes are 1-indexed, so size is n + 1
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store the shift assignment for each technician
    # 0: unassigned, 1: shift 1, 2: shift 2
    colors = [0] * (n + 1)
    
    # We iterate through each node from 1 to n.
    # If a node is unassigned, it belongs to a new connected component.
    # We process the component. To ensure the lexicographically smallest sequence,
    # we must assign the smallest index in the component to shift 1.
    # Since we iterate i from 1 to n, the first unassigned node i we encounter
    # is guaranteed to be the smallest index in its component.
    # Thus, we set colors[i] = 1.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS for this component
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_c = colors[u]
                next_c = 3 - current_c # 1 -> 2, 2 -> 1
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        queue.append(v)
                    elif colors[v] == current_c:
                        # Conflict detected: adjacent nodes have the same color
                        print("NO")
                        return

    # If we successfully assign shifts to all technicians without conflicts
    print("YES")
    # Print the assignment sequence s_1, s_2, ..., s_n
    # Using sys.stdout.write for potentially faster output, though print is likely fine
    # colors[1:] contains the assignments for 1 to n
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
