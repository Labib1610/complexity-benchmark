    e = list(map(int, data[2:2 + 2 * m]))
    for i in range(0, 2 * m, 2):
        u = e[i]
        v = e[i + 1]
        adj[u].append(v)
        adj[v].append(u)
    color = [0] * (n + 1)
    for s in range(1, n + 1):
        if color[s]:
            continue
        color[s] = 1
        stack = [s]
        while stack:
            u = stack.pop()
            cu = color[u]
            nv = 3 - cu
            for v in adj[u]:
                cv = color[v]
                if cv == 0:
                    color[v] = nv
                    stack.append(v)
                elif cv == cu:
                    sys.stdout.write("NO\n")
                    return
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color[1:])))
    sys.stdout.write("\n")

main()
