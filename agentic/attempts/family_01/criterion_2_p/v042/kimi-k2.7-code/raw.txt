```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n
    for i in range(n):
        if color[i]:
            continue
        stack = [i]
        color[i] = 1
        while stack:
            v = stack.pop()
            cv = color[v]
            for w in adj[v]:
                if color[w] == 0:
                    color[w] = 3 - cv
                    stack.append(w)
                elif color[w] == cv:
                    sys.stdout.write("NO")
                    return

    out = ["YES", " ".join(str(c) for c in color)]
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
