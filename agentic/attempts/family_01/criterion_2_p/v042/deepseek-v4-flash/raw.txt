We can model this as a bipartite graph coloring problem.  
If the graph is bipartite, we can assign vaults 1 and 2 to the two colors.  
To get the lexicographically smallest assignment, we try to assign vault 1 to as many early nodes as possible.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    
    def bfs(start):
        from collections import deque
        q = deque([start])
        color[start] = 0
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    return False
        return True
    
    for i in range(n):
        if color[i] == -1:
            if not bfs(i):
                print("NO")
                return
    
    # Now we have a valid bipartition, but we want lexicographically smallest.
    # We can try to flip colors in each connected component if it makes earlier nodes 1.
    visited = [False] * n
    for i in range(n):
        if not visited[i]:
            # collect component
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            # Check if flipping helps lexicographically
            # Count how many nodes in comp have color 0 vs 1
            # We want smallest index to be 1 if possible
            # If first node in comp (by index) has color 1, we flip
            first = min(comp)
            if color[first] == 1:
                for u in comp:
                    color[u] ^= 1
    
    print("YES")
    print(' '.join(str(c + 1) for c in color))

if __name__ == "__main__":
    solve()
```
