```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, sort its neighbors for consistent ordering
    for i in range(1, n + 1):
        adj[i] = sorted(adj[i])
    
    # We want to find a triangle (x,y,z) with x<y<z and min lex order
    # So we iterate through all edges and look for common neighbors
    
    min_x, min_y, min_z = n + 1, n + 1, n + 1
    
    # Iterate over all edges (u,v) with u < v
    for u in range(1, n + 1):
        neighbors_u = adj[u]
        # For each pair of neighbors of u
        for i in range(len(neighbors_u)):
            v = neighbors_u[i]
            if v <= u:
                continue
            for j in range(i + 1, len(neighbors_u)):
                w = neighbors_u[j]
                if w <= u:
                    continue
                # Now check if v and w are connected
                if w in adj[v]:
                    # Found triangle (u, v, w) with u < v < w
                    if u < min_x or (u == min_x and v < min_y) or (u == min_x and v == min_y and w < min_z):
                        min_x, min_y, min_z = u, v, w
    
    print(min_x, min_y, min_z)

if __name__ == "__main__":
    main()
```
