 ```python
import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [0] * n
    for i in range(m):
        u = data[2 + 2 * i] - 1
        v = data[3 + 2 * i] - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    allbits = (1 << n) - 1
    suffix = [0] * n
    for i in range(n):
        suffix[i] = allbits ^ ((1 << (i + 1)) - 1)

    for a in range(n):
        candidates = adj[a] & suffix[a]
        temp = candidates
        while temp:
            lsb = temp & -temp
            b = lsb.bit_length() - 1
            common = candidates & adj[b] & suffix[b]
            if common:
                c = (common & -common).bit_length() - 1
                sys.stdout.write(f"{a + 1} {b + 1} {c + 1}")
                return
            temp ^= lsb

if __name__ == "__main__":
    solve()
```
