import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    conflicts = set()
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        conflicts.add((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Get all nodes and sort them
    nodes = sorted(adj.keys())
    
    # For each node i, check pairs among its neighbors
    for i in nodes:
        neighbors = sorted(adj[i])
        len_neigh = len(neighbors)
        for x in range(len_neigh):
            for y in range(x+1, len_neigh):
                u = neighbors[x]
                v = neighbors[y]
                if u >= v:
                    u, v = v, u
                if (u, v) in conflicts:
                    # Found a triangle: i, u, v
                    triple = sorted([i, u, v])
                    print(triple[0], triple[1], triple[2])
                    return

if __name__ == "__main__":
    main()
