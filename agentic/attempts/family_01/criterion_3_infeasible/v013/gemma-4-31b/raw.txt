import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    # Adjacency list using sorted lists for efficient iteration
    adj_list = [[] for _ in range(n + 1)]
    
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        ptr += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        adj_list[u].append(v)
        adj_list[v].append(u)

    # Sort the adjacency lists to ensure we find the lexicographically smallest triple
    for i in range(1, n + 1):
        adj_list[i].sort()

    # We are looking for the lexicographically smallest (a, b, c) with a < b < c
    # Iterating a from 1 to n, then b from adj_list[a] (where b > a), 
    # then checking if there exists c > b such that c is connected to both a and b.
    for a in range(1, n + 1):
        # Candidates for b are neighbors of a that are greater than a
        neighbors_a = [v for v in adj_list[a] if v > a]
        
        for b in neighbors_a:
            # We need to find the smallest c > b such that c is a neighbor of both a and b.
            # The neighbors of a are already sorted. We can iterate through them.
            # Since we want the smallest c, and neighbors_a is sorted,
            # we just need to check which v in neighbors_a is > b and also connected to b.
            
            # To optimize, we only iterate through neighbors of a that are greater than b.
            # Since neighbors_a is sorted, we can use binary search or just a slice.
            # However, since we need the absolute smallest c, and we are iterating b in 
            # increasing order, we must check all possible c for each b.
            
            # Optimization: If the size of adj[a] is much larger than adj[b], 
            # it's faster to iterate over adj[b].
            if len(adj[a]) < len(adj[b]):
                # Iterate through neighbors of a that are > b
                # Since neighbors_a is sorted, we can find the starting point
                import bisect
                idx = bisect.bisect_right(neighbors_a, b)
                for i in range(idx, len(neighbors_a)):
                    c = neighbors_a[i]
                    if c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Iterate through neighbors of b that are > b
                # We need c > b and c in adj[a]
                # To keep it lexicographical, we need the smallest such c.
                # Neighbors of b are sorted in adj_list[b].
                for c in adj_list[b]:
                    if c > b:
                        if c in adj[a]:
                            print(f"{a} {b} {c}")
                            return

if __name__ == "__main__":
    solve()
