import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for consistent ordering
    for key in adj:
        adj[key] = sorted(adj[key])
    
    # Create a list of all nodes
    nodes = list(range(1, n+1))
    
    # We want the lexicographically smallest triangle
    # So we iterate through all combinations (i,j,k) with i<j<k
    # For efficiency, we iterate through all edges and find a third node
    
    edge_set = set(edges)
    found = False
    result = None
    
    # Try all edges (u,v) and see if there's a node w such that (u,w) and (v,w) are also edges
    # To ensure lexicographical order, we make sure u < v < w
    for u in nodes:
        if found:
            break
        for v in adj[u]:
            if v <= u:
                continue
            # Now look for w > v such that (u,w) and (v,w) exist
            # Iterate through neighbors of v that are > v
            for w in adj[v]:
                if w <= v:
                    continue
                # Check if (u,w) exists
                if w in adj[u]:
                    result = (u, v, w)
                    found = True
                    break
        if found:
            break
    
    if result:
        print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
