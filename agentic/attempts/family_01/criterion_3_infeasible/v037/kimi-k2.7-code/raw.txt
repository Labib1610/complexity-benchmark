 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_bits = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj_bits[u] |= 1 << v
        adj_bits[v] |= 1 << u
        adj[u].append(v)
        adj[v].append(u)
    for i in range(1, n + 1):
        adj[i].sort()
    for a in range(1, n + 1):
        ab = adj_bits[a]
        for b in adj[a]:
            if b <= a:
                continue
            common = ab & adj_bits[b]
            t = common >> (b + 1)
            if t:
                c = b + (t & -t).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
