import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        ptr += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We should iterate a from 1 to n.
    # For a fixed a, we look for pairs (b, c) such that a < b < c and both are neighbors of a.
    # Among those, we want the smallest b, then smallest c.
    
    # To optimize, for each 'a', we only consider neighbors of 'a' that are > a.
    # Let these neighbors be sorted: b1, b2, ..., bk.
    # We check pairs (bi, bj) where i < j.
    
    # Since n is up to 2000, a simple O(n^3) or O(n * m) might be too slow.
    # However, the number of edges is 2*10^5.
    # For a fixed 'a', we only check edges (b, c) where both b and c are neighbors of 'a'.
    
    # Bitsets are very efficient for this in Python using large integers.
    bit_adj = [0] * (n + 1)
    for u in range(1, n + 1):
        for v in adj[u]:
            bit_adj[u] |= (1 << v)

    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # We want the smallest b from neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now we need the smallest c from neighbors_a that is > b and connected to b
            # We can intersect the bitset of a's neighbors and b's neighbors
            # and find the smallest bit set at position > b.
            
            # Mask for bits > b
            mask = ~((1 << (b + 1)) - 1)
            common = bit_adj[a] & bit_adj[b] & mask
            
            if common:
                # Find the lowest set bit
                # common & -common gives the lowest set bit
                c = (common & -common).bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
