def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    except StopIteration:
        return

    # adj_mask[u] has bit (v-1) set if u and v are connected
    adj_mask = [0] * (n + 1)

    adj_list = [[] for _ in range(n + 1)]

        v = int(next(iterator))

        adj_mask[u] |= (1 << (v - 1))
        adj_mask[v] |= (1 << (u - 1))
        
        # Update lists
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # Sort adjacency lists to process neighbors in increasing order
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate to find the lexicographically smallest triple (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # a goes from 1 to n
    # For a fixed a, b iterates through neighbors of a in increasing order
    # We need b > a.
    # For fixed a, b, we want smallest c > b such that (a, c) and (b, c) are edges.
    
    for a in range(1, n + 1):
        # Optimization: if a is large enough that no b > a and c > b exists, we could break,
        # but n is small enough (2000) that we don't need complex pruning.
        # We just iterate.
        
        # Get neighbors of a
        neighbors = adj_list[a]
        
        for b in neighbors:
            if b <= a:
                continue
            
            # Edge (a, b) exists. Check for common neighbors c > b.
            # Common neighbors mask:
            common = adj_mask[a] & adj_mask[b]
            
            # We need bits in 'common' that are at positions >= b.
            # Bit position k corresponds to vertex k+1.
            # We want vertex c > b => c-1 >= b.
            # Shift right by b to bring bit b to position 0.
            shifted = common >> b
            
            if shifted > 0:
                # Found at least one common neighbor c > b.
                # We need the smallest such c, which corresponds to the lowest set bit in 'shifted'.
                # (shifted & -shifted) isolates the lowest set bit.

                # For a power of 2, bit_length gives 1 + log2(value).
                # If value is 2^k, bit_length is k+1.
                # So index k = bit_length - 1.
                k = (shifted & -shifted).bit_length() - 1
