import sys
from bisect import bisect_right

# Increase recursion depth just in case, though not needed here
sys.setrecursionlimit(2000)

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency masks
    # adj_mask[u] stores the bitset of neighbors of u.
    # The v-th bit is 1 if there is an edge between u and v.
    # We use 1-based indexing for guilds, so size n + 1.
    adj_mask = [0] * (n + 1)
    
    # Adjacency lists to iterate neighbors in sorted order
    # Using sets to handle potential duplicate edges and then convert to sorted lists
    adj = [set() for _ in range(n + 1)]
    
    # Precompute powers of 2 for bit manipulation
    # 1 << i corresponds to the i-th bit being set.
    pow2 = [1 << i for i in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
        except StopIteration:
            break
        
        # Add to adjacency sets
        adj[u].add(v)
        adj[v].add(u)
        
        # Update adjacency masks
        adj_mask[u] |= pow2[v]
        adj_mask[v] |= pow2[u]
        
    # Sort adjacency lists to allow lexicographical traversal
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate through all possible guilds 'a'
    for a in range(1, n + 1):
        # We need to find a neighbor 'b' of 'a' such that b > a
        # Since adj[a] is sorted, we can skip elements <= a
        # Find the first index where element > a
        idx = bisect_right(adj[a], a)
        
        # Iterate over neighbors b > a
        for i in range(idx, len(adj[a])):
            b = adj[a][i]
            
            # We need to find a neighbor 'c' of both 'a' and 'b' such that c > b
            # common neighbors are represented by the intersection of their masks
            common = adj_mask[a] & adj_mask[b]
            
            # We are only interested in bits set at positions > b
            # Shifting right by b + 1 moves the bit at position b+1 to position 0
            # and discards bits at positions <= b.
            shifted = common >> (b + 1)
            
            if shifted:
                # If shifted is non-zero, it means there exists at least one c > b
                # that is a common neighbor.
                # We want the lexicographically smallest c, which corresponds to
                # the lowest set bit in 'shifted'.
                # (shifted & -shifted) isolates the lowest set bit.
                # bit_length() gives the number of bits required to represent the integer,
                # which is index + 1 for a power of 2.
                # So index = bit_length() - 1.
                # The actual index c is (index of bit in shifted) + b + 1.
                
                lowest_bit = shifted & -shifted
                c = lowest_bit.bit_length() - 1 + b + 1
                
                # Found the lexicographically smallest triple (a, b, c)
                print(f"{a} {b} {c}")
                return

        adj_mask[v] |= pow2[u]
        
    # Sort adjacency lists to ensure we iterate neighbors in increasing order (lexicographical requirement)
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate to find the lexicographically smallest triangle (a, b, c) with a < b < c
    # We iterate 'a' from 1 to n to find the smallest 'a'.
    for a in range(1, n + 1):
        # We only consider neighbors b > a.
        # Since adj[a] is sorted, we can use binary search to find the start index.
        # bisect_right returns the insertion point after all elements <= a.
        idx = bisect_right(adj[a], a)
        
        # Iterate over neighbors b > a
        for i in range(idx, len(adj[a])):
            b = adj[a][i]
            
            # Find common neighbors c of a and b.
            # The common neighbors are indicated by the bits set in (adj_mask[a] & adj_mask[b]).
            common = adj_mask[a] & adj_mask[b]
            
            # We need c > b.
            # We can shift the common mask right by (b + 1) bits.
            # This shifts the bit corresponding to b+1 to position 0, and discards bits <= b.
            # If 'common' has bits set at positions > b, 'shifted' will be non-zero.
            shifted = common >> (b + 1)
            
            if shifted:
                # If shifted is non-zero, there exists at least one c > b.
                # To find the smallest such c (lexicographically smallest triple),
                # we need the lowest set bit in 'shifted'.
                # (shifted & -shifted) isolates the lowest set bit.
                # .bit_length() returns the number of bits required to represent the integer.
                # For a power of 2 (2^k), bit_length() is k + 1.
                # So the index k is bit_length() - 1.
                # The actual guild index c is k + (b + 1) because we shifted right by b + 1.
                
                lowest_bit = shifted & -shifted
                c = lowest_bit.bit_length() - 1 + b + 1
                
                print(f"{a} {b} {c}")
                return
