```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Try to find triangle containing edge (u, v)
    for u, v in edges:
        if u > v:
            u, v = v, u
        # Find common neighbor w such that w > v (to ensure u < v < w)
        common = adj[u] & adj[v]
        for w in common:
            if w > v:
                print(u, v, w)
                return
    
    # If not found, do full search (lexicographically smallest)
    for a in range(1, n - 1):
        for b in adj[a]:
            if b <= a:
                continue
            for c in adj[a]:
                if c <= b:
                    continue
                if b in adj[c]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
```
