```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    incompatible = [set() for _ in range(n + 1)]
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        edges.append((u, v))
        incompatible[u].add(v)
        incompatible[v].add(u)
    
    # For each node, we'll store its neighbors for fast lookup
    adj = [set() for _ in range(n + 1)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # We need to find a triangle where all three edges are present (mutually incompatible)
    # That means we're looking for a triangle in the graph
    
    # To get the lexicographically smallest, we iterate through all triangles
    found = False
    result = None
    
    # Try all combinations of three nodes i < j < k
    # But since n can be up to 2000, we can't check all combinations
    # Instead, let's look for triangles more efficiently
    
    # For each edge (u,v), we look for a third node w such that (u,w) and (v,w) are also edges
    # Since we want the lexicographically smallest, we process edges in order
    
    # Let's sort the edges to help with ordering
    edges.sort()
    
    # Create a mapping from node to sorted list of neighbors
    neighbors = [None] * (n + 1)
    for i in range(1, n + 1):
        neighbors[i] = sorted(list(adj[i]))
    
    min_triangle = None
    
    # For each node u, check pairs of its neighbors (v, w) to see if they're connected
    for u in range(1, n + 1):
        neigh_list = neighbors[u]
        len_neigh = len(neigh_list)
        for i in range(len_neigh):
            v = neigh_list[i]
            if v <= u:
                continue
            for j in range(i + 1, len_neigh):
                w = neigh_list[j]
                if w <= v:
                    continue
                # Check if v and w are connected
                if w in adj[v]:
                    triangle = tuple(sorted([u, v, w]))
                    if min_triangle is None or triangle < min_triangle:
                        min_triangle = triangle
                    found = True
                    break
            if found and min_triangle and min_triangle[0] < u:
                break
        if found and min_triangle and min_triangle[0] < u:
            break
    
    if min_triangle:
        print(min_triangle[0], min_triangle[1], min_triangle[2])
    else:
        # Fallback - should not happen according to problem statement
        print(1, 2, 3)

if __name__ == "__main__":
    main()
```
