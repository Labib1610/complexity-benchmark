The problem asks us to find the lexicographically smallest triangle (a set of three nodes $a, b, c$ such that $a < b < c$ and edges exist between all pairs) in a given undirected graph with $n$ nodes and $m$ edges. We are guaranteed that at least one triangle exists. The constraints are $n \le 2000$ and $m \le 2 \cdot 10^5$.

Let the triangle be $(a, b, c)$ with $a < b < c$. We want to minimize $a$, then $b$, then $c$.
This suggests we can iterate through possible values of $a$, then $b$, then $c$ and check if they form a triangle. However, a naive iteration would be $O(n^3)$, which is too slow for $n=2000$ (approx $8 \cdot 10^9$ operations). We need something faster, likely close to $O(m \sqrt{m})$ or $O(n^2)$ or $O(m)$. Given $m$ is up to $2 \cdot 10^5$, $O(n^2)$ is $4 \cdot 10^6$, which is well within the 2-second time limit.

import sys
from bisect import bisect_right

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    # Fast I/O
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency matrix for O(1) lookups
    adj_matrix = [[False] * (n + 1) for _ in range(n + 1)]
    
    # Adjacency lists
    adj_list = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj_matrix[u][v] = True
        adj_matrix[v][u] = True
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # Sort adjacency lists
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate to find the lexicographically smallest triad (a, b, c) with a < b < c
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        len_a = len(neighbors_a)
        
        # Enumerate to get index i for optimization
        for i, b in enumerate(neighbors_a):
            if b <= a:
                continue
            
            # We have a < b. We need c > b.
            # We want smallest c in N(a) & N(b) with c > b.
            
            # Optimization: iterate over the smaller neighbor set.
            if len_a < len(adj_list[b]):
                # Iterate N(a) starting from i + 1
                # neighbors_a[i+1:] contains all c in N(a) such that c > b
                for idx_c in range(i + 1, len_a):
                    c = neighbors_a[idx_c]
                    if adj_matrix[b][c]:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Iterate N(b) starting from first element > b
                neighbors_b = adj_list[b]
                # Find start index in neighbors_b where elements > b
                start_idx = bisect_right(neighbors_b, b)
                for idx_c in range(start_idx, len(neighbors_b)):
                    c = neighbors_b[idx_c]
                    if adj_matrix[a][c]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
