 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(1, n + 1):
        mask = adj[a] & ~((1 << (a + 1)) - 1)
        while mask:
            lsb = mask & -mask
            b = lsb.bit_length() - 1
            common = adj[a] & adj[b]
            cand = common & ~((1 << (b + 1)) - 1)
            if cand:
                c = (cand & -cand).bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}")
                return
            mask ^= lsb

if __name__ == "__main__":
    solve()
```
