def find_interfering_triplet(n, m, edges):
    from collections import defaultdict

    # Create a graph represented as an adjacency list
    graph = defaultdict(set)

    for u, v in edges:
        graph[u].add(v)
        graph[v].add(u)

    # Look for triplets
    for a in range(1, n + 1):
        neighbors_a = list(graph[a])
        len_neighbors = len(neighbors_a)
        for i in range(len_neighbors):
            for j in range(i + 1, len_neighbors):
                b = neighbors_a[i]
                c = neighbors_a[j]
                if b in graph[c]:  # Check if b and c are also connected
                    return a, b, c

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = find_interfering_triplet(n, m, edges)
print(*result)
