import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Create a list of nodes for consistent ordering
    nodes = sorted(adj.keys())
    
    # For each node, get its neighbors and sort them
    neighbors = {}
    for u in nodes:
        neighbors[u] = sorted(adj[u])
    
    # We want to find a triangle (triplet where each pair is connected)
    # That means we need three nodes a, b, c such that:
    # a-b, b-c, and a-c are all edges
    
    # Strategy:
    # Iterate through all edges (u,v)
    # For each such edge, look for a common neighbor w of u and v
    # That would form a triangle u-v-w
    
    # To ensure lexicographical order and efficiency:
    # We'll process edges in a way that ensures we check triangles in lex order
    
    # Let's reindex the edges by sorting them
    sorted_edges = []
    for u, v in edges:
        if u > v:
            u, v = v, u
        sorted_edges.append((u, v))
    sorted_edges.sort()
    
    # Now for each edge (a,b) with a < b, we look for a node c such that:
    # c is connected to both a and b
    # and a < b < c
    
    best_triplet = None
    
    # For optimization, let's go through each edge and find common neighbors
    for a, b in sorted_edges:
        if a >= b:
            continue
        # Get neighbors of a and b
        neigh_a = adj[a]
        neigh_b = adj[b]
        
        # Find common neighbors that are greater than b
        # Convert to sets for intersection
        common = neigh_a & neigh_b
        
        # Filter those greater than b
        candidates = [c for c in common if c > b]
        
        if not candidates:
            continue
            
        # Sort candidates to get the smallest one
        c = min(candidates)
        
        # This gives us a triplet (a, b, c) with a < b < c
        triplet = (a, b, c)
        
        if best_triplet is None or triplet < best_triplet:
            best_triplet = triplet
    
    if best_triplet:
        print(best_triplet[0], best_triplet[1], best_triplet[2])

if __name__ == "__main__":
    main()
