def find_incompatible_triplet(n, m, incompatibilities):
    from collections import defaultdict

    # Create an adjacency set for incompatibilities
    incompatible = defaultdict(set)
    for u, v in incompatibilities:
        incompatible[u].add(v)
        incompatible[v].add(u)

    # Check each pair of workers for a common incompatible third worker
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in incompatible[a]:
                # Find a third worker incompatible with both a and b
                for c in range(b + 1, n + 1):
                    if c in incompatible[a] and c in incompatible[b]:
                        return a, b, c

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
incompatibilities = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = find_incompatible_triplet(n, m, incompatibilities)
print(result[0], result[1], result[2])
