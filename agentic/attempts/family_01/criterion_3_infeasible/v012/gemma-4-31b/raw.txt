import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency list and adjacency matrix for fast lookup
    # Since n <= 2000, a bitset or a list of sets/booleans can work.
    # A list of sets is flexible, but for n=2000, a bitset-like approach 
    # (using Python's large integers) can be very efficient for intersections.
    
    adj_bits = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj_list[u].append(v)
        adj_bits[u] |= (1 << v)
        adj_bits[v] |= (1 << u)

    # Sort adjacency lists to ensure we check in lexicographical order
    for i in range(1, n + 1):
        adj_list[i].sort()

    # To find the lexicographically smallest (a, b, c):
    # We iterate a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    # To optimize, we only check b and c that are neighbors of a.
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        # We only care about neighbors > a
        filtered_neighbors = [v for v in neighbors_a if v > a]
        
        # Bitmask of neighbors of a that are > a
        mask_a = 0
        for v in filtered_neighbors:
            mask_a |= (1 << v)
            
        # Now we need to find b, c in filtered_neighbors such that b < c and (b, c) is an edge.
        # Iterate through filtered_neighbors as b
        for i in range(len(filtered_neighbors)):
            b = filtered_neighbors[i]
            # We need c > b and c is a neighbor of both a and b.
            # The candidates for c are (neighbors of b) AND (neighbors of a) AND (nodes > b).
            # mask_a already contains neighbors of a > a.
            # We need neighbors of b that are also in mask_a and are > b.
            
            # To find the smallest c > b:
            # common = adj_bits[b] & mask_a
            # We need the smallest bit set in 'common' that is > b.
            common = adj_bits[b] & mask_a
            if common:
                # Find the smallest bit set in common that is > b.
                # We can mask out bits <= b.
                # A fast way to clear bits 0 to b is common & ~((1 << (b + 1)) - 1)
                valid_c_mask = common & ~((1 << (b + 1)) - 1)
                if valid_c_mask:
                    # Find the lowest set bit (LSB)
                    # LSB is (valid_c_mask & -valid_c_mask). 
                    # To get the index: .bit_length() - 1 of (LSB) or using bin().count()
                    # But in Python 3.10+, int.bit_count() exists, but not for index.
                    # The simplest way to get the index of the lowest set bit:
                    c = (valid_c_mask & -valid_c_mask).bit_length() - 1
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
