```python
import sys

def solve():
    # Read n and m from the first line of input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list to store the graph
    adj = [[] for _ in range(n + 1)]
    # Use a bitset-like approach or a set for fast lookup
    # Given n=2000, an adjacency matrix (bitset) is feasible (2000*2000 bits ~ 0.5 MB)
    # In Python, we can use a list of integers as bitsets.
    adj_bitset = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        
        # Store in adjacency list for iterating over neighbors
        # To ensure lexicographical order, we will sort these later
        adj[u].append(v)
        adj[v].append(u)
        
        # Store in bitset for O(1) (or O(n/64)) intersection
        adj_bitset[u] |= (1 << v)
        adj_bitset[v] |= (1 << u)

    # Sort adjacency lists to iterate in increasing order
    for i in range(1, n + 1):
        adj[i].sort()

    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n
    # Iterate b from neighbors of a where b > a
    # Find the smallest c from neighbors of both a and b where c > b
    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_a = [v for v in adj[a] if v > a]
        
        # Pre-calculate the bitmask for neighbors of a that are > a
        # Actually, we can just use the full adj_bitset[a] and mask it
        mask_a = adj_bitset[a]
        
        for b in neighbors_a:
            # We need c such that c > b and c is a neighbor of both a and b
            # Intersection of neighbors of a and neighbors of b
            common = mask_a & adj_bitset[b]
            
            # We need the smallest c > b. 
            # We can find the lowest set bit of (common >> (b + 1))
            # shift right to remove bits <= b
            candidates = common >> (b + 1)
            if candidates:
                # Find the index of the lowest set bit
                # (candidates & -candidates).bit_length() - 1 gives the position of the first 1
                # The actual value of c is (position + b + 1)
                lsb = candidates & -candidates
                c = lsb.bit_length() - 1 + b + 1
                print(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
