```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    idx = 2
    adj = [set() for _ in range(n+1)]
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    for a in range(1, n+1):
        la = adj[a]
        for b in la:
            if b > a:
                common = la & adj[b]
                for c in common:
                    if c > b:
                        sys.stdout.write(f"{a} {b} {c}\n")
                        return

solve()
```
