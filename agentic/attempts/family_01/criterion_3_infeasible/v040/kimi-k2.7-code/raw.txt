 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for a in range(n):
        S = adj[a] & ~((1 << a) - 1)
        s = S
        while s:
            low = s & -s
            b = low.bit_length() - 1
            cand = adj[b] & S & ~((1 << (b + 1)) - 1)
            if cand:
                c = (cand & -cand).bit_length() - 1
                sys.stdout.write(f"{a + 1} {b + 1} {c + 1}\n")
                return
            s ^= low

if __name__ == "__main__":
    solve()
```
