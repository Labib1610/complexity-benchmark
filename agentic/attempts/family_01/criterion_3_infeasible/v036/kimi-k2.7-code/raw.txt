 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    masks = [0] * (n + 1)
    out = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        if (masks[u] >> v) & 1:
            continue
        masks[u] |= 1 << v
        masks[v] |= 1 << u
        if u < v:
            out[u].append(v)
        else:
            out[v].append(u)
    for u in range(1, n + 1):
        out[u].sort()
    for a in range(1, n - 1):
        ma = masks[a]
        for b in out[a]:
            inter = ma & masks[b]
            higher = inter & (-1 << (b + 1))
            if higher:
                c = (higher & -higher).bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
```
