def find_incompatible_trio(n, m, pairs):
    from collections import defaultdict
    
    incompatibilities = defaultdict(set)

    for u, v in pairs:
        incompatibilities[u].add(v)
        incompatibilities[v].add(u)
    
    for a in range(1, n + 1):
        for b in incompatibilities[a]:
            for c in incompatibilities[b]:
                if c > b and c in incompatibilities[a]:
                    return a, b, c

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
pairs = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = find_incompatible_trio(n, m, pairs)
print(*result)
