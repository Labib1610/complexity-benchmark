```python
import sys

def solve():
    # Using fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency representation
    # We use sets for O(1) average lookup time
    adj = [set() for _ in range(n + 1)]
    # We also keep a sorted list of neighbors for each node to iterate in order
    adj_list = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        adj_list[u].append(v)
        adj_list[v].append(u)
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate a from 1 to n.
    # For a fixed a, we iterate through its neighbors b where b > a.
    # For a fixed a and b, we look for neighbors c of both a and b where c > b.
    
    # Pre-sort the adjacency lists to ensure we check neighbors in increasing order
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    for a in range(1, n + 1):
        # Get neighbors of a that are greater than a
        neighbors_a = [b for b in adj_list[a] if b > a]
        
        for b in neighbors_a:
            # Now we need to find c such that c > b and c is connected to both a and b.
            # To minimize c, we can iterate through the neighbors of b that are > b
            # and check if they are also neighbors of a.
            # Or iterate through neighbors of a that are > b and check if they are neighbors of b.
            
            # Optimization: iterate through the smaller of the two neighbor sets
            # But since we need the smallest c, iterating through sorted neighbors of b is easier.
            for c in adj_list[b]:
                if c > b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == "__main__":
    solve()
```
