def find_triangle(n, m, edges):
    from collections import defaultdict
    
    # Create an adjacency set for fast lookup
    adj = defaultdict(set)
    
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Iterate over all pairs (u, v) and check for a common neighbor
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:  # Check if there's a link between a and b
                # Find common neighbors of a and b
                common_neighbors = adj[a].intersection(adj[b])
                for c in common_neighbors:
                    if c > b:  # Ensure a < b < c
                        return a, b, c

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = find_triangle(n, m, edges)
print(*result)
