 ```python
import sys
from bisect import bisect_right

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    bits = [0] * (n + 1)

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
        bits[u] |= 1 << (v - 1)
        bits[v] |= 1 << (u - 1)

    for i in range(1, n + 1):
        adj[i].sort()

    for a in range(1, n + 1):
        lst = adj[a]
        idx = bisect_right(lst, a)
        ba = bits[a]
        for b in lst[idx:]:
            mask = ba & bits[b]
            t = mask >> b
            if t:
                c = b + (t & -t).bit_length()
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
```
