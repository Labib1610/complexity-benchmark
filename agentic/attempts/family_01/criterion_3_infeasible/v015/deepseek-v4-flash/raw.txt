We can solve this by checking each possible triple in increasing order and stopping at the first valid triangle.  
Since \( n \leq 2000 \), a naive \( O(n^3) \) check would be too slow, but we can optimize using adjacency sets and early exit.

```python
import sys

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # Check triples in lexicographic order
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b not in adj[a]:
                continue
            # Intersection of neighbors of a and b gives possible c
            common = adj[a] & adj[b]
            for c in sorted(common):
                if c > b:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
