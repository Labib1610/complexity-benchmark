 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_mask = [0] * n
    neighbors = [[] for _ in range(n)]

    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        neighbors[u].append(v)
        neighbors[v].append(u)

    for lst in neighbors:
        lst.sort()

    all_mask = (1 << n) - 1
    suffix = [0] * n
    for b in range(n):
        suffix[b] = all_mask ^ ((1 << (b + 1)) - 1)

    for a in range(n):
        mask_a = adj_mask[a]
        for b in neighbors[a]:
            if b <= a:
                continue
            cand = mask_a & adj_mask[b] & suffix[b]
            if cand:
                c = (cand & -cand).bit_length() - 1
                sys.stdout.write(f"{a + 1} {b + 1} {c + 1}")
                return

if __name__ == "__main__":
    solve()
```
