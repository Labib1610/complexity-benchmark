import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list using sets for O(1) lookup
    adj = defaultdict(set)
    edges = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # For each node, sort its neighbors
    for u in adj:
        adj[u] = sorted(adj[u])
    
    # Iterate through all edges to find a triangle
    found = False
    result = None
    
    # Sort edges to ensure we check in lexicographical order
    edges.sort()
    
    for u, v in edges:
        if found:
            break
        # Find common neighbors of u and v
        # Iterate through the smaller neighbor list
        if len(adj[u]) > len(adj[v]):
            u, v = v, u
        
        for w in adj[u]:
            if w > v and w in adj[v]:
                # Found a triangle: u, v, w where u < v < w
                print(u, v, w)
                found = True
                break

if __name__ == "__main__":
    main()
