```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    p = 2
    for _ in range(m):
        u = int(data[p])
        v = int(data[p + 1])
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full = (1 << (n + 1)) - 1
    high_mask = [0] * (n + 1)
    for i in range(1, n + 1):
        high_mask[i] = full ^ ((1 << (i + 1)) - 1)

    for a in range(1, n + 1):
        mask_a = adj[a]
        sub = mask_a >> (a + 1)
        while sub:
            lsb = sub & -sub
            b = a + 1 + (lsb.bit_length() - 1)
            cand = mask_a & adj[b] & high_mask[b]
            if cand:
                c = (cand & -cand).bit_length() - 1
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
