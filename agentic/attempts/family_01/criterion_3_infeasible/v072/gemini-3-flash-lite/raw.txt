import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through possible a from 1 to n.
    # For a fixed a, we need to find b > a and c > b such that
    # (a, b), (a, c), and (b, c) are all edges.
    
    # Sort neighbors of each node to optimize searching
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        # We need to find b, c in neighbors_a such that b < c and (b, c) is an edge.
        # To ensure lexicographical order, we iterate b in neighbors_a, then c in neighbors_a.
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Check for c > b
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if edge (b, c) exists. Since we stored edges as (u, v) with u < v:
                # b < c is guaranteed by the loop order.
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
