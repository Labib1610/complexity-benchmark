import sys

# Increase recursion depth just in case, though not needed here
sys.setrecursionlimit(2000)

def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return
    
    # Adjacency list for iteration order
    adj = [[] for _ in range(n + 1)]
    # Adjacency mask for bitwise operations
    adj_mask = [0] * (n + 1)
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
        except StopIteration:
            break
            
        adj[u].append(v)
        adj[v].append(u)
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
        
    # Sort adjacency lists to ensure we process neighbors in increasing order
    for i in range(1, n + 1):
        adj[i].sort()
        
    # Iterate through each employee 'a'
    for a in range(1, n + 1):
        # Iterate through neighbors 'b' of 'a'
        # Since adj[a] is sorted, we encounter b's in increasing order
        for b in adj[a]:
            # We need a < b < c
            if b <= a:
                continue
            
            # We need to find the smallest c > b such that c is connected to both a and b
            # This is equivalent to finding the lowest set bit in (adj_mask[a] & adj_mask[b]) 
            # that corresponds to an index > b.
            
            # Calculate common neighbors
            common = adj_mask[a] & adj_mask[b]
            
            # We need to ignore bits corresponding to nodes <= b.
            # The mask ~((1 << (b + 1)) - 1) has 0s at positions 0 to b and 1s elsewhere.
            # This clears bits 0..b.
            mask = ~((1 << (b + 1)) - 1)
            candidates = common & mask
            
            if candidates:
                # Extract the lowest set bit
                # candidates & -candidates gives the value of the lowest set bit
                lowest_bit = candidates & -candidates
                
                # Find the index of the bit (node index)
                # bit_length() of a power of 2^k is k+1. So index is k = bit_length() - 1.
                c = lowest_bit.bit_length() - 1
                
                # Output the lexicographically smallest triple found
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
