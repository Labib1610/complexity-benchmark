 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj_mask = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        adj[u].append(v)
        adj[v].append(u)

    for i in range(1, n + 1):
        adj[i].sort()

    mask_after = [0] * (n + 1)
    for b in range(1, n + 1):
        mask_after[b] = ~((1 << (b + 1)) - 1)

    for a in range(1, n + 1):
        ma = adj_mask[a]
        for b in adj[a]:
            if b <= a:
                continue
            common = ma & adj_mask[b]
            if common:
                cand = common & mask_after[b]
                if cand:
                    c = (cand & -cand).bit_length() - 1
                    sys.stdout.write(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
