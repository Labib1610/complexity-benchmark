 ```python
import sys
import bisect

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)
        neigh[u].append(v)
        neigh[v].append(u)

    for i in range(1, n + 1):
        lst = neigh[i]
        lst.sort()
        uniq = []
        prev = 0
        for x in lst:
            if x != prev:
                uniq.append(x)
                prev = x
        neigh[i] = uniq

    full = (1 << n) - 1
    masks = [0] * (n + 1)
    for b in range(1, n + 1):
        masks[b] = full ^ ((1 << b) - 1)

    for a in range(1, n + 1):
        lst = neigh[a]
        idx = bisect.bisect_right(lst, a)
        for b in lst[idx:]:
            common = adj[a] & adj[b] & masks[b]
            if common:
                c = (common & -common).bit_length()
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
```
