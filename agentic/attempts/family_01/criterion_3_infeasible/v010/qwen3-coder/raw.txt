```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to help with finding lexicographically smallest triple
    edges.sort()
    
    # For each edge (u,v), look for a node w such that (u,w) and (v,w) are also edges
    for u, v in edges:
        if u >= v:
            continue
        # Find common neighbors of u and v
        common_neighbors = adj[u] & adj[v]
        if common_neighbors:
            # Get the smallest such neighbor
            w = min(common_neighbors)
            # Ensure we return in sorted order
            triple = sorted([u, v, w])
            print(triple[0], triple[1], triple[2])
            return

if __name__ == "__main__":
    main()
```
