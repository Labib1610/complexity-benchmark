import sys
from collections import defaultdict

def find_unresolvable_trio(n, edges):
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # Convert to sorted lists for deterministic iteration
    for u in range(1, n + 1):
        adj[u] = sorted(adj[u])
    
    # Look for a triangle (three nodes all connected to each other)
    # Iterate through all combinations with u < v < w for deterministic order
    for u in range(1, n + 1):
        for v in adj[u]:
            if v <= u:
                continue
            for w in adj[v]:
                if w <= v:
                    continue
                # Check if u and w are also connected
                if w in adj[u]:
                    return [u, v, w]
    
    return None

# Read input
input_lines = sys.stdin.read().splitlines()
n, m = map(int, input_lines[0].split())

edges = []
for i in range(1, m + 1):
    u, v = map(int, input_lines[i].split())
    edges.append((u, v))

# Find and print the result
result = find_unresolvable_trio(n, edges)
print(" ".join(map(str, result)))
