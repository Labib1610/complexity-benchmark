import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for lexicographically smallest result
    sorted_adj = [sorted(s) for s in adj]
    
    best = None
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        # Try all pairs (b, c) from neighbors of a with b < c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    # Found a triangle
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
                        # Since we iterate in sorted order, first found is optimal for this a
                        # But we need to check all a to find global minimum
                        # Optimization: if we found (1,2,3), we can stop
                        if best == (1, 2, 3):
                            print("1 2 3")
                            return
    
    if best:
        print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == "__main__":
    solve()
