```python
import sys

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Use adjacency bitmasks and adjacency lists
    # adj[i] is an integer where the j-th bit is set if i and j are connected
    adj = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # Sort adjacency lists to process neighbors in increasing order
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate through each node as the first node 'a'
    for a in range(1, n + 1):
        # Iterate through neighbors of 'a' as the second node 'b'
        # We only care about b > a to satisfy a < b < c condition and avoid duplicates
        # Since adj_list is sorted, b will be visited in increasing order
        for b in adj_list[a]:
            if b <= a:
                continue
            
            # Check for a common neighbor 'c' such that c > b
            # Common neighbors are given by the intersection of bitmasks
            common = adj[a] & adj[b]
            
            # We need the smallest c > b.
            # This corresponds to the lowest set bit in 'common' at a position > b.
            # We can shift common right by (b + 1) to remove bits <= b.
            shifted = common >> (b + 1)
            
            if shifted:
                # Find the lowest set bit in shifted
                # lsb = shifted & -shifted gives the lowest set bit as a power of 2
                lsb = shifted & -shifted
                
                # The index of the bit is bit_length() - 1
                # Since we shifted by (b + 1), we add (b + 1) back
                c = lsb.bit_length() - 1 + (b + 1)
                
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
```

```python
import sys

# Increase recursion depth just in case, though not used
sys.setrecursionlimit(2000)

def solve():
    # Efficiently read all input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency bitmasks
    # adj[i] will store an integer representing the neighbors of i
    # The j-th bit is 1 if there is an edge between i and j
    adj = [0] * (n + 1)
    
    # Adjacency lists to iterate neighbors in order
    adj_list = [[] for _ in range(n + 1)]
    
    # Parse edges
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        
        # Update bitmasks
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
        # Update lists
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # Sort adjacency lists to ensure we process neighbors in increasing order
    # This is crucial for lexicographical order of (a, b, c)
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate to find the lexicographically smallest triangle (a, b, c)
    # a < b < c
    for a in range(1, n + 1):
        # Iterate over neighbors b of a
        for b in adj_list[a]:
            # We need b > a
            if b <= a:
                continue
            
            # We need to find a neighbor c of a such that c > b
            # AND c is a neighbor of b.
            # So c must be in adj[a] & adj[b]
            
            common_neighbors = adj[a] & adj[b]

            shifted = common_neighbors >> (b + 1)

                # The original index is k + (b + 1)
                c = lsb.bit_length() - 1 + (b + 1)
                
                # Found the solution
                print(f"{a} {b} {c}")
                return
```

```python
        return
```

```python
    # adj[i] is an integer where the j-th bit is set if station i interferes with station j

    # Sorting these lists allows us to process neighbors in increasing order
    adj_list = [[] for _ in range(n + 1)]

        u = int(next(iterator))
        v = int(next(iterator))
        
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # Sort adjacency lists to ensure we check neighbors in increasing order
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate through each possible value for the first station 'a'
    for a in range(1, n + 1):
        mask_a = adj[a]
        # Iterate through neighbors of 'a' to find 'b'
        # Since adj_list[a] is sorted, we encounter b in increasing order
        for b in adj_list[a]:
```

```python
        v = int(next(iterator))
        
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # Sort adjacency lists to ensure we check neighbors in increasing order
    # This is crucial to find the lexicographically smallest triple
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate through each possible value for the first station 'a'
    for a in range(1, n + 1):
        mask_a = adj[a]
        # Iterate through neighbors of 'a' to find 'b'
        # Since adj_list[a] is sorted, we encounter b in increasing order
        for b in adj_list[a]:
            # We need b > a to maintain order a < b < c
            if b <= a:
                continue
            
            # Find common neighbors of 'a' and 'b'
            # This is the bitwise AND of their adjacency masks
            common = mask_a & adj[b]
            
            # We need to find a common neighbor 'c' such that c > b
            # Shift the common mask right by (b + 1) to filter out neighbors <= b
            shifted = common >> (b + 1)
            
            if shifted:
                # If shifted is non-zero, there exists at least one common neighbor > b
                # We need the smallest such 'c'
                # lsb = shifted & -shifted isolates the lowest set bit in shifted
                lsb = shifted & -shifted
                
                # Calculate the index of the bit. 
                # bit_length() - 1 gives the index of the lowest set bit relative to 0.
                # We add (b + 1) to get the original index.
                c = lsb.bit_length() - 1 + (b + 1)
                
                # Since we iterate a, then b, then find smallest c, this is the lexicographically smallest triple
                print(f"{a} {b} {c}")
                return
```
