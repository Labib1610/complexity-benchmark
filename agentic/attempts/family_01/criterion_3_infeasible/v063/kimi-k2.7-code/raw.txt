 ```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ans = None
    for u in range(n):
        nb = adj[u] >> (u + 1)
        while nb:
            low = nb & -nb
            offset = low.bit_length() - 1
            v = u + 1 + offset
            common = adj[u] & adj[v]
            shifted = common >> (v + 1)
            if shifted:
                lowc = shifted & -shifted
                c = v + lowc.bit_length()
                ans = (u, v, c)
                break
            nb ^= low
        if ans is not None:
            break

    sys.stdout.write(f"{ans[0]+1} {ans[1]+1} {ans[2]+1}")

if __name__ == "__main__":
    solve()
```
