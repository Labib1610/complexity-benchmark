```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # We need to find a triangle in the graph (three nodes all connected to each other)
    # That's because if we have three nodes a, b, c such that:
    # a-b, b-c, c-a are all edges, then it's impossible to 2-color them.
    
    # To find the lexicographically smallest triangle:
    # We'll iterate through all edges and for each edge (u,v), 
    # we look for a node w such that (u,w) and (v,w) are also edges.
    # We want the smallest such triangle in lex order.
    
    # First, let's sort adjacency lists for consistent iteration
    for i in range(1, n+1):
        adj[i] = list(adj[i])
        adj[i].sort()
    
    # Create a set of edges for O(1) lookup
    edge_set = set()
    for u, v in edges:
        if u > v:
            u, v = v, u
        edge_set.add((u, v))
    
    # Now iterate over all edges (u, v) with u < v
    # For each such edge, look for a node w such that w is connected to both u and v
    # And u < v < w to ensure lexicographical order
    
    best = None
    
    for u in range(1, n+1):
        for v in adj[u]:
            if v <= u:
                continue
            # Now look for w > v such that (u,w) and (v,w) exist
            # Iterate through neighbors of v that are > v
            for w in adj[v]:
                if w <= v:
                    continue
                # Check if (u, w) is an edge
                if w in adj[u]:
                    # Found triangle u-v-w
                    candidate = (u, v, w)
                    if best is None or candidate < best:
                        best = candidate
                        break  # Since we're going in order, first one for this u,v is smallest w
            if best is not None and u >= best[0]:
                # No point checking further if we already found something better
                break
        if best is not None and u >= best[0]:
            break
                
    print(best[0], best[1], best[2])

if __name__ == "__main__":
    main()
```
