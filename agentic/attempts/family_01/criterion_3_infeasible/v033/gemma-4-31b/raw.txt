import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    # To keep the adjacency list sorted for lexicographical smallest search
    # we will use sets for O(1) lookup and sorted lists for iteration.
    edges = []
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        ptr += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # Pre-sort adjacency lists to ensure lexicographical smallest search
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]

    # We need the lexicographically smallest (a, b, c) such that a < b < c
    # and (a,b), (a,c), (b,c) are edges.
    # The strategy: iterate a from 1 to n, then b from a+1 to n, 
    # then check for common neighbors c of a and b where c > b.
    
    for a in range(1, n + 1):
        # Only iterate through neighbors of a that are > a
        neighbors_a = sorted_adj[a]
        # Filter neighbors of a that are > a
        valid_b = [v for v in neighbors_a if v > a]
        
        for b in valid_b:
            # We need to find c such that c > b and c is a neighbor of both a and b.
            # To minimize c, we can iterate through the neighbors of b that are > b
            # and check if they are also neighbors of a.
            # Since sorted_adj[b] is sorted, the first c we find will be the smallest.
            
            # Optimization: check the smaller degree vertex to find common neighbors
            # However, for lexicographical smallest (a, b, c), we must check c > b.
            # We iterate through sorted_adj[b] and check if c in adj[a].
            for c in sorted_adj[b]:
                if c > b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == "__main__":
    solve()
