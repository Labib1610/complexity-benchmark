import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in lexicographical order
    edges.sort()
    
    # For each edge (u, v), find the smallest w such that w is adjacent to both u and v
    # We iterate w in increasing order to find the lexicographically smallest triangle
    for u, v in edges:
        # We need to find w where u < w < v or w > v
        # To get lexicographically smallest (u, v, w), we should check w in increasing order
        # But we also need to consider the case where w < u (handled by earlier edges)
        
        # Check neighbors of u that are > u (to maintain a < b < c)
        # We iterate through sorted neighbors for efficiency
        neighbors_u = sorted([x for x in adj[u] if x > u])
        for w in neighbors_u:
            if w > v and w in adj[v]:
                print(u, v, w)
                return
        
        # Also check if there's w such that u < w < v and w is adjacent to both
        # This is covered by iterating neighbors_u which includes all neighbors > u
    
    # If not found by iterating edges in order, we need a more thorough search
    # This handles cases where the smallest triangle might not be found by the above
    # due to the order of edges (e.g., when smallest triangle has edges not processed in order)
    
    # Alternative approach: for each vertex u, check pairs of its neighbors
    for u in range(1, n + 1):
        neighbors = sorted([x for x in adj[u] if x > u])
        for i in range(len(neighbors)):
            for j in range(i + 1, len(neighbors)):
                v = neighbors[i]
                w = neighbors[j]
                if w in adj[v]:
                    print(u, v, w)
                    return
    
    # Should not reach here given problem guarantees
    print(-1)

if __name__ == "__main__":
    solve()
