import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    deg = [0] * (n + 1)
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        deg[u] += 1
        deg[v] += 1
        edges.append((u, v))
    
    threshold = int((2 * m) ** 0.5)
    
    best = None
    
    # High-degree vertices: check all triples among them
    high = [i for i in range(1, n + 1) if deg[i] > threshold]
    L = len(high)
    for i in range(L):
        a = high[i]
        adj_a = adj[a]
        for j in range(i + 1, L):
            b = high[j]
            if b not in adj_a:
                continue
            adj_b = adj[b]
            for k in range(j + 1, L):
                c = high[k]
                if c in adj_a and c in adj_b:
                    if best is None or (a, b, c) < best:
                        best = (a, b, c)
    
    # Low-degree vertices: for each edge, check common neighbors with low degree
    for u, v in edges:
        if deg[u] > threshold and deg[v] > threshold:
            continue
        if deg[u] <= deg[v]:
            small, big = u, v
        else:
            small, big = v, u
        adj_small = adj[small]
        adj_big = adj[big]
        for w in adj_small:
            if w > big and w in adj_big:
                a, b, c = small, big, w
                if best is None or (a, b, c) < best:
                    best = (a, b, c)
    
    if best is not None:
        print(best[0], best[1], best[2])

solve()
