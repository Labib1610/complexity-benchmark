```python
import sys
from collections import defaultdict

input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
edges = defaultdict(set)

for i in range(1, m + 1):
    u, v = map(int, data[i].split())
    edges[u].add(v)
    edges[v].add(u)

result = None

for a in range(1, n + 1):
    for b in edges[a]:
        if b <= a:
            continue
        for c in edges[b]:
            if c <= b or c not in edges[a]:
                continue
            if result is None or (a, b, c) < result:
                result = (a, b, c)

print(*result)
```
