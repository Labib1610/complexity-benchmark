 ```python
import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    masks = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    p = 2
    for _ in range(m):
        u = data[p]
        v = data[p + 1]
        p += 2
        adj[u].append(v)
        adj[v].append(u)
        masks[u] |= 1 << (v - 1)
        masks[v] |= 1 << (u - 1)
    for i in range(1, n + 1):
        adj[i].sort()
    for a in range(1, n + 1):
        ma = masks[a]
        for b in adj[a]:
            if b <= a:
                continue
            shifted = (ma & masks[b]) >> b
            if shifted:
                c = b + (shifted & -shifted).bit_length()
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
