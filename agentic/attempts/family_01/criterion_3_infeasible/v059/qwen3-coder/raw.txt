```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    graph = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        edges.append((u, v))
        graph[u].add(v)
        graph[v].add(u)
    
    # Sort edges to process in order, helps with finding lexicographically smallest triple
    edges.sort()
    
    # For each edge (u,v), look for a third node w such that (u,w) and (v,w) are also edges
    found = False
    result = None
    
    # We'll iterate through all edges and for each edge (u,v), we check neighbors of u and v
    # to see if they have a common neighbor that forms a triangle
    
    for u, v in edges:
        if u > v:
            u, v = v, u
        # Get neighbors of u and v, and find common neighbors
        neighbors_u = graph[u]
        neighbors_v = graph[v]
        
        # Iterate through the smaller set for efficiency
        if len(neighbors_u) > len(neighbors_v):
            neighbors_u, neighbors_v = neighbors_v, neighbors_u
            
        for w in neighbors_u:
            if w in neighbors_v:
                # Found a triangle: u, v, w
                # Ensure all are distinct and get sorted order
                triple = sorted([u, v, w])
                if not found or triple < result:
                    result = triple
                    found = True
                    
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
```
