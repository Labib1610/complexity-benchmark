```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]); v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    best = None
    
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        if not neighbors_a:
            continue
        
        if best is not None:
            upper_b = best[1]
        else:
            upper_b = n + 1
        
        for b in neighbors_a:
            if b <= a:
                continue
            if best is not None and b >= upper_b:
                break
            
            if best is not None:
                upper_c = best[2]
            else:
                upper_c = n + 1
            
            for c in neighbors_a:
                if c <= b:
                    continue
                if c >= upper_c:
                    break
                if c in adj[b]:
                    best = (a, b, c)
                    upper_b = b
                    upper_c = c
                    break
    
    sys.stdout.write(f"{best[0]} {best[1]} {best[2]}")

solve()
```
