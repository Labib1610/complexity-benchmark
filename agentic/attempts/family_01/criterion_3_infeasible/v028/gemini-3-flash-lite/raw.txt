```python
import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) edge lookups
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        # Store edges such that u < v
        if u < v:
            edges.append((u, v))
        else:
            edges.append((v, u))
            
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through each edge (a, b) with a < b, 
    # then check for any c > b such that c is connected to both a and b.
    # To ensure lexicographical order, we process potential 'a' values in increasing order.
    
    best_triple = None
    
    # Sort edges to help find smallest a, then b
    edges.sort()
    
    for a, b in edges:
        # We need a c such that b < c <= n and c is in both adj[a] and adj[b]
        # To find the smallest such c, intersect the sets
        common = adj[a].intersection(adj[b])
        
        for c in common:
            if c > b:
                # Potential triple (a, b, c) found
                if best_triple is None or (a, b, c) < best_triple:
                    best_triple = (a, b, c)
        
        # Optimization: if we have a best_triple, we can potentially prune
        # But given n=2000, the O(m * sqrt(m)) or similar complexity is fine.
        
    if best_triple:
        print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")

if __name__ == '__main__':
    solve()
```
