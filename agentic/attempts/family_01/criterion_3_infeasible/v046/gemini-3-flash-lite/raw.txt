```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, c from b+1 to n.
    # To optimize, we only consider b such that b is in adj[a],
    # and c such that c is in adj[a] and adj[b].
    
    # Sort adjacency lists to ensure we pick the smallest indices first
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # We only care about neighbors of a to be b and c
        neighbors_a = adj_list[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can look for c in neighbors_a[i+1:]
            # and check if c is also in adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c > b and c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
