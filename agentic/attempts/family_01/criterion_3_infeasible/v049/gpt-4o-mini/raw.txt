def find_restricted_triplet(n, m, restrictions):
    from collections import defaultdict

    # Create a graph to store the restrictions
    graph = defaultdict(set)

    for u, v in restrictions:
        graph[u].add(v)
        graph[v].add(u)

    # Check for each pair (u, v) if there exists a common third element w
    for a in range(1, n + 1):
        # Check all neighbors of a
        neighbors = list(graph[a])
        k = len(neighbors)
        for i in range(k):
            for j in range(i + 1, k):
                b = neighbors[i]
                c = neighbors[j]
                if b in graph[c]:  # Check if b and c are also restricted
                    return a, min(b, c), max(b, c)

# Read input
import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
restrictions = [tuple(map(int, line.split())) for line in data[1:m + 1]]

# Find and print the result
result = find_restricted_triplet(n, m, restrictions)
print(*result)
