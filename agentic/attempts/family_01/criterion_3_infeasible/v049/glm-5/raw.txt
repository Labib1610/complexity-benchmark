```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
    
    for a in range(1, n + 1):
        neighbors = sorted(adj[a])
        size = len(neighbors)
        for i in range(size):
            b = neighbors[i]
            if b < a:
                continue
            for j in range(i + 1, size):
                c = neighbors[j]
                if c < a:
                    continue
                if c in adj[b]:
                    sys.stdout.write(f"{a} {b} {c}\n")
                    return

solve()
```
