import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through possible values of a, then b, then c.
    # a ranges from 1 to n-2
    # b ranges from a+1 to n-1
    # c ranges from b+1 to n
    
    # Pre-sorting adjacency lists helps in checking existence faster
    # but with n=2000, O(n^3) is too slow. However, we only check 
    # pairs (a, b) that are connected.
    
    # Sort adjacency lists to potentially optimize
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b must be connected to a
        adj_a = sorted_adj[a]
        for b in adj_a:
            if b <= a:
                continue
            
            # Candidates for c must be connected to both a and b
            # We look for the intersection of adj[a] and adj[b]
            # Since we need the smallest c, we iterate through the smaller set
            adj_b = adj[b]
            
            # Find common neighbors of a and b that are > b
            # Intersection of two sorted lists is efficient
            for c in adj_a:
                if c <= b:
                    continue
                if c in adj_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
