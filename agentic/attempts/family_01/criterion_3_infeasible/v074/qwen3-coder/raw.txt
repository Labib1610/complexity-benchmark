import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for incompatibilities
    incompatible = [set() for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        incompatible[u].add(v)
        incompatible[v].add(u)
        index += 2
    
    # We need to find a triangle (3 nodes all connected to each other)
    # Enumerate all triangles by checking triplets (i,j,k) with i<j<k
    # But we want the lexicographically smallest one
    
    # To optimize, for each node i, look at its neighbors
    # For each pair of neighbors (j,k) of node i where j<k and j>i
    # Check if j and k are also incompatible (i.e., form a triangle with i)
    
    found = False
    result = None
    
    for i in range(1, n - 1):
        neighbors = [x for x in incompatible[i] if x > i]
        neighbors.sort()
        # Now check all pairs (j,k) with j<k from neighbors
        for idx_j in range(len(neighbors)):
            j = neighbors[idx_j]
            for idx_k in range(idx_j + 1, len(neighbors)):
                k = neighbors[idx_k]
                if k in incompatible[j]:
                    # Found triangle (i,j,k)
                    print(i, j, k)
                    return

if __name__ == "__main__":
    main()
