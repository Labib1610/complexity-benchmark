import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    edges = []
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        adj[u].add(v)
        adj[v].add(u)
    
    # For each node, sort its adjacency list for consistency
    for i in range(1, n + 1):
        adj[i] = sorted(adj[i])
    
    # We want to find a triangle (3-clique) that is lexicographically smallest
    # Iterate through all combinations (i, j, k) with i < j < k
    # To ensure lexicographical order, we iterate i from 1 to n,
    # then j from i+1 to n, then k from j+1 to n
    
    for i in range(1, n - 1):
        for j in range(i + 1, n):
            if j not in adj[i]:
                continue
            for k in range(j + 1, n + 1):
                if k not in adj[i]:
                    continue
                if k not in adj[j]:
                    continue
                # Found a triangle: i-j, i-k, j-k
                print(i, j, k)
                return

if __name__ == "__main__":
    main()
