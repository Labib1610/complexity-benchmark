import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    suffix = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suffix[i] = suffix[i + 1] | (1 << i)

    out = sys.stdout.write
    for a in range(n):
        nb = adj[a] & suffix[a + 1]
        adj_a = adj[a]
        while nb:
            low = nb & -nb
            b = low.bit_length() - 1
            nb &= nb - 1
            inter = adj_a & adj[b] & suffix[b + 1]
            if inter:
                c = (inter & -inter).bit_length() - 1
                out(f"{a + 1} {b + 1} {c + 1}\n")
                return

if __name__ == "__main__":
    solve()
