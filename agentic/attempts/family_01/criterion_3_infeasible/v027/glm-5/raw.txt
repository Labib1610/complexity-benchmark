import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort adjacency lists for lexicographical processing
    # We iterate u from 1 to n, then v > u, then w > v.
    # To check efficiently, we need to test if w is in adj[u] and adj[v].
    # We iterate u, v in order. For each pair, we iterate w in sorted order of adj[u] intersect adj[v].
    # To get w in sorted order, we can iterate through the smaller set and check membership in the larger.
    
    for u in range(1, n + 1):
        # We want v > u. 
        # adj[u] contains neighbors. We only care about v > u.
        neighbors_u = adj[u]
        
        # We can iterate v from u+1 to n, but that's O(N) per u.
        # Better to iterate v in neighbors_u if degree is small, or skip if degree is large?
        # Actually, we just iterate v in neighbors_u.
        # If v < u, we skip (to avoid duplicates and ensure u < v).
        # If v > u, we proceed.
        
        for v in neighbors_u:
            if v < u:
                continue
            
            # Now we have u < v. We need w > v such that w is neighbor of both.
            # We want the smallest such w.
            # We iterate through candidates w from the intersection of neighbors.
            # To find the smallest w > v, we can iterate w starting from v+1 up to n,
            # but that's O(N) check.
            # Alternatively, iterate the smaller adjacency set.
            
            set1 = neighbors_u
            set2 = adj[v]
            
            # Iterate the smaller set to find intersection
            if len(set1) > len(set2):
                set1, set2 = set2, set1
                
            for w in set1:
                if w in set2:
                    if w > v:
                        # Found the lexicographically smallest triplet
                        print(u, v, w)
                        return

if __name__ == '__main__':
    solve()
