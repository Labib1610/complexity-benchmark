Algorithm:

for a in range(n):
  for b in neighbors[a]:
    if b <= a: continue
    common = mask[a] & mask[b]
    # zero out bits <= b
    trimmed = common >> (b+1)
    if trimmed == 0: continue
    # find lowest set bit
    low = trimmed & -trimmed
    tz = low.bit_length() - 1
    c = b+1 + tz
