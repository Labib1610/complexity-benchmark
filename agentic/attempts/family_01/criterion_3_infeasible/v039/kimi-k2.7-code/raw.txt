```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # mask with bits > i set to 1
    hi = [~((1 << (i + 1)) - 1) for i in range(n)]

    out = sys.stdout
    for a in range(n):
        aa = adj[a]
        for b in range(a + 1, n):
            if (aa >> b) & 1:
                mask = aa & adj[b] & hi[b]
                if mask:
                    c = (mask & -mask).bit_length() - 1
                    out.write(f"{a + 1} {b + 1} {c + 1}\n")
                    return

solve()
```
