```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(set)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # Sort edges to process in order
    edges.sort()
    
    # For each edge (u,v), look for a common neighbor w
    # such that (u,w) and (v,w) are also edges
    for u, v in edges:
        if u >= v:
            continue
        # Get neighbors of both u and v
        neighbors_u = adj[u]
        neighbors_v = adj[v]
        # Find common neighbors
        common = neighbors_u & neighbors_v
        if common:
            # Found a triangle
            w = min(common)
            if w > v:
                # Output the trio in sorted order
                trio = sorted([u, v, w])
                print(trio[0], trio[1], trio[2])
                return

if __name__ == "__main__":
    main()
```
