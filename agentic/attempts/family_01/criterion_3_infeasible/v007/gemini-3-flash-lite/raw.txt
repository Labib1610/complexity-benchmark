```python
import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # We iterate a from 1 to n-2, b from a+1 to n-1, c from b+1 to n.
    # To optimize:
    # 1. Sort adjacency lists for predictable order.
    # 2. Iterate a, then b, then check if any node in adj[a] and adj[b] is the same.
    
    # Pre-sorting adjacency lists helps in iterating in order.
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Candidates for b must be neighbors of a.
        # To keep lexicographical order, iterate through sorted neighbors.
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need a c > b such that c is in adj[a] AND c is in adj[b].
            # This is equivalent to finding the smallest c in (adj[a] intersection adj[b])
            # such that c > b.
            
            # Intersection of two sorted lists can be done efficiently.
            # But since we only need the smallest c > b, we can iterate.
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c > b and c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
