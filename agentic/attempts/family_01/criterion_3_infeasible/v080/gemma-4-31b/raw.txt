```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    # We only care about edges where u < v to iterate efficiently
    edges_from = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges_from[u].append(v)

    # Lexicographically smallest (a, b, c) means:
    # 1. Smallest a
    # 2. Smallest b (where b > a)
    # 3. Smallest c (where c > b)
    
    # Iterate through all possible 'a' in increasing order
    for a in range(1, n + 1):
        # Get all neighbors of a that are > a
        neighbors_a = sorted(edges_from[a])
        
        # If a has fewer than 2 neighbors > a, it can't be the smallest in a triplet
        if len(neighbors_a) < 2:
            continue
        
        # Iterate through pairs (b, c) from the neighbors of a
        # Since we sorted neighbors_a, iterating i then j ensures b < c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
