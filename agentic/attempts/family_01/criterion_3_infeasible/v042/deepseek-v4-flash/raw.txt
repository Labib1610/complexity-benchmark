import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [set() for _ in range(n)]
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # For each edge, find a common neighbor
    for u, v in edges:
        # Intersection of neighbors
        common = adj[u] & adj[v]
        if common:
            w = min(common)
            a, b, c = sorted([u + 1, v + 1, w + 1])
            print(a, b, c)
            return

if __name__ == "__main__":
    solve()
