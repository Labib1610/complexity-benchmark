 ```python
import sys
import bisect

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    bits = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            continue
        if (bits[u] >> (v - 1)) & 1:
            continue
        bits[u] |= 1 << (v - 1)
        bits[v] |= 1 << (u - 1)
        adj[u].append(v)
        adj[v].append(u)

    for v in range(1, n + 1):
        adj[v].sort()

    masks = [0] * (n + 1)
    for b in range(1, n + 1):
        masks[b] = ~((1 << b) - 1)

    for a in range(1, n + 1):
        ba = bits[a]
        lst = adj[a]
        idx = bisect.bisect_right(lst, a)
        for j in range(idx, len(lst)):
            b = lst[j]
            common = ba & bits[b]
            if common:
                cand = common & masks[b]
                if cand:
                    c = (cand & -cand).bit_length()
                    sys.stdout.write(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
