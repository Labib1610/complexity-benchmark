import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Find lexicographically smallest triangle (a < b < c)
    for a in range(1, n + 1):
        neighbors = adj[a]
        if len(neighbors) < 2:
            continue
        # We need b, c in neighbors with b < c and edge (b, c) exists.
        # To get lexicographically smallest (a, b, c):
        #   minimize a (outer loop does this)
        #   then minimize b
        #   then minimize c
        # For a fixed a, iterate b in increasing order.
        # For each b, iterate c in increasing order and check edge.
        # But checking all pairs could be O(deg^2). With m < n^2 and
        # guaranteed triangle, we use degree heuristic to bound work.
        sorted_neighbors = sorted(neighbors)
        # Only consider neighbors > a for b and c to ensure a < b < c
        # Actually b and c just need to be > a and b < c.
        # But if b < a, then the triangle (b, a, c) would have been found
        # when outer loop was at b (since b < a). So we can restrict
        # to neighbors > a without missing the lex smallest triangle.
        cand = [x for x in sorted_neighbors if x > a]
        if len(cand) < 2:
            continue
        # Iterate b in increasing order
        found = False
        for i in range(len(cand)):
            b = cand[i]
            adj_b = adj[b]
            # c must be > b and in both adj[a] (already in cand) and adj[b]
            for j in range(i + 1, len(cand)):
                c = cand[j]
                if c in adj_b:
                    print(a, b, c)
                    return
            # Degree heuristic: if deg(b) is large, checking membership is O(1)
            # but iterating all c could be expensive. However, since we
            # must find the lex smallest, we cannot skip.
            # To speed up, we can break early if we found one.
        # If we reach here, no triangle with this a and b < c where both > a.
        # But there might be triangle with b < a < c, which would have
        # been found earlier. So it's fine.
    
    # Fallback (should not reach due to guarantee)
    # Brute force without restriction
    for a in range(1, n + 1):
        neighbors = adj[a]
        sorted_neighbors = sorted(neighbors)
        for i in range(len(sorted_neighbors)):
            b = sorted_neighbors[i]
            if b <= a:
                continue
            adj_b = adj[b]
            for j in range(i + 1, len(sorted_neighbors)):
                c = sorted_neighbors[j]
                if c <= b:
                    continue
                if c in adj_b:
                    print(a, b, c)
                    return

solve()
