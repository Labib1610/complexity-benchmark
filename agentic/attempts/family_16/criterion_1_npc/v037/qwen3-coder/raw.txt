import sys
from collections import defaultdict

def solve():
    j, m = map(int, input().split())
    
    jobs = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        operations = []
        for i in range(o):
            machine = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            operations.append((machine, duration))
        jobs.append(operations)
    
    # We'll use a backtracking approach with pruning
    # For each job, we keep track of when each operation can start
    # For each machine, we keep track of the time intervals when it's busy
    
    # This is a classic job shop scheduling problem
    # We'll try a branch and bound approach
    
    # First, let's compute a lower bound using critical path method
    def compute_makespan(schedule):
        # schedule[i][j] = start time of job i operation j
        machine_end_times = defaultdict(int)  # For each machine, when it becomes free
        job_end_times = [0] * j  # For each job, when it finishes
        
        # We need to process operations in a valid order
        # Let's build a list of all operations with their job and op indices
        all_ops = []
        for job_idx, job in enumerate(jobs):
            for op_idx, (machine, duration) in enumerate(job):
                all_ops.append((job_idx, op_idx, machine, duration))
        
        # We'll simulate the schedule
        # For each job, track which operation we're on
        job_op_index = [0] * j
        # For each machine, track which time slots are occupied
        machine_schedule = [[] for _ in range(m+1)]  # 1-indexed
        
        # Create events: (time, event_type, job, op, machine, duration)
        # event_type: 0 = operation starts, 1 = operation ends
        events = []
        
        # Initialize: all operations that can start (first operation of each job)
        for job_idx in range(j):
            if len(jobs[job_idx]) > 0:
                machine, duration = jobs[job_idx][0]
                events.append((0, 0, job_idx, 0, machine, duration))  # Start at time 0
        
        current_time = 0
        job_next_start_time = [0] * j  # For each job, earliest time its next op can start
        machine_free_time = [0] * (m+1)  # For each machine, when it becomes free
        
        while events or any(job_op_index[job_idx] < len(jobs[job_idx]) for job_idx in range(j)):
            # Sort events by time
            events.sort()
            
            if not events:
                # Find the next possible start time
                next_time = float('inf')
                next_events = []
                
                for job_idx in range(j):
                    if job_op_index[job_idx] < len(jobs[job_idx]):
                        op_idx = job_op_index[job_idx]
                        # This job's next operation can start after its previous operation ends
                        # and after the machine is free
                        machine, duration = jobs[job_idx][op_idx]
                        earliest_start = job_next_start_time[job_idx]
                        earliest_start = max(earliest_start, machine_free_time[machine])
                        next_time = min(next_time, earliest_start)
                        next_events.append((earliest_start, 0, job_idx, op_idx, machine, duration))
                
                # Filter events that happen at next_time
                events = [e for e in next_events if e[0] == next_time]
                current_time = next_time
            
            # Process the earliest event
            time, event_type, job_idx, op_idx, machine, duration = events.pop(0)
            current_time = time
            
            if event_type == 0:  # Operation starts
                # Schedule this operation
                start_time = current_time
                end_time = start_time + duration
                
                # Update machine schedule
                machine_schedule[machine].append((start_time, end_time))
                
                # Schedule the end event
                events.append((end_time, 1, job_idx, op_idx, machine, duration))
                
                # Update when this job's next operation can start
                job_next_start_time[job_idx] = end_time
                
                # Update when this machine will be free
                machine_free_time[machine] = end_time
                
            else:  # event_type == 1, operation ends
                # This operation is done
                job_op_index[job_idx] = op_idx + 1
                
                # If there's a next operation for this job, schedule it
                if job_op_index[job_idx] < len(jobs[job_idx]):
                    next_op_idx = job_op_index[job_idx]
                    next_machine, next_duration = jobs[job_idx][next_op_idx]
                    # It can start when both the job is ready and the machine is free
                    earliest_start = max(job_next_start_time[job_idx], machine_free_time[next_machine])
                    events.append((earliest_start, 0, job_idx, next_op_idx, next_machine, next_duration))
        
        return max(machine_free_time)
    
    # Let's try a simpler approach: use a greedy heuristic with backtracking
    # But given the constraints (j,m <= 10, total ops <= 50), we can try a more systematic approach
    
    # Use a recursive backtracking with pruning
    def backtrack(job_op_index, machine_free_time, job_next_start_time, current_makespan):
        # job_op_index: for each job, which operation we're on
        # machine_free_time: for each machine, when it becomes free
        # job_next_start_time: for each job, earliest time its next op can start
        # current_makespan: current maximum completion time
        
        # If we've processed all operations, return the makespan
        if all(job_op_index[i] == len(jobs[i]) for i in range(j)):
            return current_makespan
        
        # Pruning: if current makespan is already worse than best found, return infinity
        global best_makespan
        if current_makespan >= best_makespan:
            return float('inf')
        
        # Find all possible next operations
        next_options = []
        for job_idx in range(j):
            if job_op_index[job_idx] < len(jobs[job_idx]):
                op_idx = job_op_index[job_idx]
                machine, duration = jobs[job_idx][op_idx]
                earliest_start = max(job_next_start_time[job_idx], machine_free_time[machine])
                end_time = earliest_start + duration
                next_options.append((earliest_start, end_time, job_idx, op_idx, machine, duration))
        
        # Sort by earliest start time to try sooner operations first
        next_options.sort()
        
        result = float('inf')
        for _, end_time, job_idx, op_idx, machine, duration in next_options:
            # Make a copy of the state
            new_job_op_index = job_op_index[:]
            new_machine_free_time = machine_free_time[:]
            new_job_next_start_time = job_next_start_time[:]
            
            # Update the state
            new_job_op_index[job_idx] = op_idx + 1
            new_machine_free_time[machine] = end_time
            new_job_next_start_time[job_idx] = end_time
            
            # Recurse
            new_makespan = max(current_makespan, end_time)
            res = backtrack(new_job_op_index, new_machine_free_time, new_job_next_start_time, new_makespan)
            result = min(result, res)
            
            # Pruning: if we found a good solution, we might be able to prune more
            if res < best_makespan:
                best_makespan = res
        
        return result
    
    # Initialize
    global best_makespan
    best_makespan = float('inf')
    initial_job_op_index = [0] * j
    initial_machine_free_time = [0] * (m+1)
    initial_job_next_start_time = [0] * j
    
    result = backtrack(initial_job_op_index, initial_machine_free_time, initial_job_next_start_time, 0)
    print(result)

solve()
