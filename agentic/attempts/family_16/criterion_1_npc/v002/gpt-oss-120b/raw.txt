import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    J = int(next(it))
    M = int(next(it))
    jobs = []
    total_ops = 0
    for _ in range(J):
        o = int(next(it))
        ops = []
        for __ in range(o):
            off = int(next(it)) - 1
            dur = int(next(it))
            ops.append((off, dur))
        jobs.append(ops)
        total_ops += o

    # precompute remaining time per job from each position
    rem_job = [[0]*(len(jobs[i])+1) for i in range(J)]
    for i in range(J):
        s = 0
        for k in range(len(jobs[i])-1, -1, -1):
            s += jobs[i][k][1]
            rem_job[i][k] = s

    # remaining time per machine (sum of durations of unscheduled ops on that machine)
    # we will compute on the fly for bound

    best = [float('inf')]
    from functools import lru_cache

    @lru_cache(maxsize=None)
    def dfs(pos_tuple, mach_tuple):
        # pos_tuple: job next op index
        # mach_tuple: machine available times
        # compute current makespan so far
        cur_time = 0
        for p in pos_tuple:
            # time when job finished its last scheduled op is stored in job_finish list (we'll compute)
            pass
        # Instead we keep job finish times in separate array passed via closure; to avoid complexity,
        # we store them in mach_tuple? Not needed: we can recompute from schedule order.
        # We'll instead pass job_finish as part of state via a tuple.
        return

    # To simplify, we implement explicit recursion with mutable arrays and memo dict.
    from collections import defaultdict

    memo = {}
    job_pos = [0]*J
    job_finish = [0]*J
    mach_avail = [0]*M

    def lower_bound():
        # max of each job's earliest possible finish and each machine's load
        lb = 0
        for i in range(J):
            lb = max(lb, job_finish[i] + rem_job[i][job_pos[i]])
        # machine load
        mach_load = [0]*M
        for i in range(J):
            for idx in range(job_pos[i], len(jobs[i])):
                off, dur = jobs[i][idx]
                mach_load[off] += dur
        for m in range(M):
            lb = max(lb, mach_avail[m] + mach_load[m])
        return lb

    def dfs_scheduled(scheduled):
        # scheduled: number of ops already scheduled
        if scheduled == total_ops:
            # all done
            cur = max(job_finish)
            if cur < best[0]:
                best[0] = cur
            return
        # bound
        lb = lower_bound()
        if lb >= best[0]:
            return
        # memo key
        key = (tuple(job_pos), tuple(mach_avail))
        if key in memo and memo[key] <= scheduled:
            return
        memo[key] = scheduled

        # choose any job whose next op is ready
        for i in range(J):
            if job_pos[i] >= len(jobs[i]):
                continue
            off, dur = jobs[i][job_pos[i]]
            start = max(job_finish[i], mach_avail[off])
            # schedule this op
            old_job_finish = job_finish[i]
            old_mach = mach_avail[off]
            job_finish[i] = start + dur
            mach_avail[off] = start + dur
            job_pos[i] += 1

            dfs_scheduled(scheduled+1)

            # backtrack
            job_pos[i] -= 1
            job_finish[i] = old_job_finish
            mach_avail[off] = old_mach

    dfs_scheduled(0)
    print(best[0])

if __name__ == "__main__":
    solve()
