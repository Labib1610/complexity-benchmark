    # But we need to consider the dependencies properly
    
    # Let's think of this differently:
    # We have a set of jobs (reviews) with dependencies
    # Each job has a duration and needs a specific resource (official)
    # This is a classic scheduling problem - we can model it as 
    # a job shop scheduling problem and solve with simulation
    
    # For each document, track which review we're on
    doc_review_index = [0] * j
    doc_review_start_times = [0] * j  # When current review of doc can start

    # Events: (time, event_type, doc_id, review_idx)
    # But simpler: just simulate time step by step would be too slow
    
    # Better: process reviews in order, tracking when officials are free
    
    # For each document, we process its reviews sequentially
    # For each review, it can start only when:
    # 1. Previous review of same document is done
    # 2. Required official is free
    
    # So we can process documents independently, but track official availability
    
    # Let's reorganize:
    # For each document, calculate when each of its reviews can start and end
    
    max_end_time = 0
    
    # For each official, keep track of their schedule
    official_schedules = [[] for _ in range(m + 1)]  # List of (start, end) times
    
    # Process documents one by one
    for doc_id in range(j):
        reviews = documents[doc_id]
        current_time = 0
        
        for official, duration in reviews:
            # This review can start only after:
            # 1. Previous review is done (current_time)
            # 2. Official is free
            
            # Find the earliest time this official is free
            official_free = current_time
            for start, end in official_schedules[official]:
                if start <= current_time < end:
                    official_free = end
                elif current_time < start:
                    break
            
            # But we also need to check all intervals for gaps
            # Sort the official's schedule
            sched = sorted(official_schedules[official])
            
            # Find earliest slot where official is free and >= current_time
            actual_start = current_time
            for start, end in sched:
                if actual_start < start:
                    break
                elif actual_start < end:
                    actual_start = end
                elif actual_start >= start and actual_start < end:
                    actual_start = end
            
            # Now official can work from actual_start
            start_time = actual_start
            end_time = start_time + duration
            
            # Add this interval to official's schedule
            official_schedules[official].append((start_time, end_time))
            
            # Update current_time for next review in this document
            current_time = end_time
    
    # Find maximum end time among all schedules
    result = 0
    for i in range(1, m + 1):
        for start, end in official_schedules[i]:
            result = max(result, end)
    
    print(result)

solve()
