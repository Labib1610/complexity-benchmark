We can model this as a job-shop scheduling problem with small constraints, so we can use a search or dynamic programming approach.  
Since the total number of operations is at most 50, we can use a priority-queue-based simulation with backtracking or a branch-and-bound search.

Here’s a possible solution:

```python
import sys
from heapq import heappush, heappop

def solve():
    input_data = sys.stdin.read().strip().split()
    idx = 0
    j = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    
    jobs = []
    for _ in range(j):
        o = int(input_data[idx]); idx += 1
        ops = []
        for _ in range(o):
            machine = int(input_data[idx]); idx += 1
            duration = int(input_data[idx]); idx += 1
            ops.append((machine - 1, duration))
        jobs.append(ops)
    
    # Total operations
    total_ops = sum(len(job) for job in jobs)
    
    # For each job, track next operation index
    next_op = [0] * j
    # For each machine, track when it becomes free
    machine_free = [0] * m
    # For each job, track when it becomes free
    job_free = [0] * j
    
    # We'll do a priority queue simulation: 
    # Each event is (time, job_index, op_index)
    # But we need to decide order when multiple ops can start.
    # Since constraints are small, we can use backtracking search.
    
    best = [float('inf')]
    
    # Precompute operation lists
    ops_list = []
    for job_idx, job in enumerate(jobs):
        for op_idx, (machine, dur) in enumerate(job):
            ops_list.append((job_idx, op_idx, machine, dur))
    
    # We'll use DFS with pruning
    # State: current time, next_op per job, machine_free, job_free
    # We'll try to schedule operations in order of earliest possible start
    
    def dfs(time, next_op, machine_free, job_free, remaining):
        if remaining == 0:
            best[0] = min(best[0], time)
            return
        # Prune: if current time >= best, stop
        if time >= best[0]:
            return
        
        # Find all operations that can start now
        possible = []
        for job_idx in range(j):
            if next_op[job_idx] < len(jobs[job_idx]):
                machine, dur = jobs[job_idx][next_op[job_idx]]
                start_time = max(time, machine_free[machine], job_free[job_idx])
                possible.append((start_time, job_idx, next_op[job_idx], machine, dur))
        
        # Sort by start time to try earliest first
        possible.sort()
        
        # Try each possible operation
        for start_time, job_idx, op_idx, machine, dur in possible:
            # Update state
            old_machine_free = machine_free[machine]
            old_job_free = job_free[job_idx]
            old_next_op = next_op[job_idx]
            
            machine_free[machine] = start_time + dur
            job_free[job_idx] = start_time + dur
            next_op[job_idx] += 1
            
            dfs(max(time, start_time + dur), next_op, machine_free, job_free, remaining - 1)
            
            # Backtrack
            machine_free[machine] = old_machine_free
            job_free[job_idx] = old_job_free
            next_op[job_idx] = old_next_op
    
    dfs(0, next_op, machine_free, job_free, total_ops)
    
    print(best[0])

if __name__ == "__main__":
    solve()
```
