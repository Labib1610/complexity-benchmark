import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    J = int(next(it))
    M = int(next(it))
    jobs = []
    total_ops = 0
    for _ in range(J):
        o = int(next(it))
        ops = []
        for __ in range(o):
            mach = int(next(it)) - 1
            dur = int(next(it))
            ops.append((mach, dur))
        jobs.append(ops)
        total_ops += o

    # precompute remaining total duration per job and per machine
    rem_job = [sum(d for _, d in job) for job in jobs]
    rem_mach = [0] * M
    for job in jobs:
        for mach, dur in job:
            rem_mach[mach] += dur

    # initial upper bound via simple list scheduling (greedy)
    job_pos = [0] * J
    job_time = [0] * J
    mach_time = [0] * M
    ops_done = 0
    while ops_done < total_ops:
        # find all ready operations
        ready = []
        for j in range(J):
            p = job_pos[j]
            if p < len(jobs[j]):
                mach, dur = jobs[j][p]
                start = max(job_time[j], mach_time[mach])
                ready.append((start + dur, start, j, mach, dur))
        # pick the one with earliest finish
        ready.sort()
        fin, st, j, mach, dur = ready[0]
        job_time[j] = fin
        mach_time[mach] = fin
        job_pos[j] += 1
        ops_done += 1
    best = max(max(job_time), max(mach_time))

    # state for DFS
    job_pos = [0] * J
    job_time = [0] * J
    mach_time = [0] * M

    # memoization with tuple of positions and machine times could be huge; skip

    from functools import lru_cache

    # precompute suffix sums for each job for fast remaining
    suffix_job = []
    for job in jobs:
        s = [0] * (len(job) + 1)
        for i in range(len(job) - 1, -1, -1):
            s[i] = s[i + 1] + job[i][1]
        suffix_job.append(s)

    # remaining per machine can be recomputed from global rem_mach minus done durations
    # we track done per machine in array
    mach_done = [0] * M

    @lru_cache(maxsize=None)
    def dfs(pos_key, job_time_key, mach_time_key, mach_done_key):
        nonlocal best
        # decode
        pos = list(pos_key)
        jt = list(job_time_key)
        mt = list(mach_time_key)
        md = list(mach_done_key)

        ops_done = sum(pos)
        if ops_done == total_ops:
            cur = max(jt + mt)
            if cur < best:
                best = cur
            return

        # lower bound
        cur_makespan = max(jt + mt)
        lb = cur_makespan
        for j in range(J):
            lb = max(lb, jt[j] + suffix_job[j][pos[j]])
        for m in range(M):
            lb = max(lb, mt[m] + (rem_mach[m] - md[m]))
        if lb >= best:
            return

        # generate candidates
        candidates = []
        for j in range(J):
            p = pos[j]
            if p < len(jobs[j]):
                mach, dur = jobs[j][p]
                start = max(jt[j], mt[mach])
                finish = start + dur
                candidates.append((finish, start, j, mach, dur))
        # order by earliest finish to improve pruning
        candidates.sort()
        for finish, start, j, mach, dur in candidates:
            # apply
            pos2 = pos.copy()
            jt2 = jt.copy()
            mt2 = mt.copy()
            md2 = md.copy()
            pos2[j] += 1
            jt2[j] = finish
            mt2[mach] = finish
            md2[mach] += dur
            dfs(tuple(pos2), tuple(jt2), tuple(mt2), tuple(md2))

    dfs(tuple(job_pos), tuple(job_time), tuple(mach_time), tuple(mach_done))
    print(best)

if __name__ == "__main__":
    solve()
