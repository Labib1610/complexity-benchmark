from collections import defaultdict, deque

def solve():
    j, m = map(int, input().split())
    
    # Store all steps
    products = []
    total_steps = 0
    
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        steps = []
        for i in range(o):
            workstation = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            steps.append((workstation, duration))
        products.append(steps)
        total_steps += o
    
    # For each workstation, we need to track when it becomes free
    workstation_free_time = [0] * (m + 1)  # 1-indexed
    
    # For each product, we need to track when its previous step finishes
    product_step_finish = [0] * j
    
    # Create a list of all steps with their product index and step index
    all_steps = []
    for prod_idx, steps in enumerate(products):
        for step_idx, (ws, dur) in enumerate(steps):
            all_steps.append((prod_idx, step_idx, ws, dur))
    
    # We'll use a simulation approach
    # At each time, we check which steps can start
    
    # For each product, keep track of which step it's on
    current_step = [0] * j
    
    # Keep track of completion times
    completion_times = [0] * j
    
    # Keep going until all products are done
    while any(current_step[p] < len(products[p]) for p in range(j)):
        # Find the next possible start times
        next_start = None
        candidates = []
        
        for p in range(j):
            if current_step[p] < len(products[p]):
                step_info = products[p][current_step[p]]
                workstation, duration = step_info
                # This step can start at max(product_prev_finish, workstation_free_time)
                ready_time = max(product_step_finish[p], workstation_free_time[workstation])
                if next_start is None or ready_time < next_start:
                    next_start = ready_time
                    candidates = [(p, workstation, duration)]
                elif ready_time == next_start:
                    candidates.append((p, workstation, duration))
        
        # Process one of the candidates (we can process any, but we'll take the first)
        p, workstation, duration = candidates[0]
        start_time = next_start
        end_time = start_time + duration
        
        # Update tracking information
        workstation_free_time[workstation] = end_time
        product_step_finish[p] = end_time
        current_step[p] += 1
        if current_step[p] == len(products[p]):
            completion_times[p] = end_time
    
    print(max(completion_times))

solve()
