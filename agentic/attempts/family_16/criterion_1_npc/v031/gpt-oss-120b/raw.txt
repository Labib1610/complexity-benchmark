import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = list(map(int, sys.stdin.read().strip().split()))
    if not data:
        return
    it = iter(data)
    j = next(it)
    m = next(it)
    jobs = []
    for _ in range(j):
        o = next(it)
        steps = [(next(it)-1, next(it)) for _ in range(o)]  # zero‑based workstation
        jobs.append(steps)

    # simple upper bound: schedule jobs one after another
    ub = 0
    machine_time = [0]*m
    for job in jobs:
        t = 0
        for ws, d in job:
            start = max(t, machine_time[ws])
            t = start + d
            machine_time[ws] = t
        ub = max(ub, t)

    best = ub
    from functools import lru_cache

    # pre‑compute remaining total time per job for lower bound
    rem_sum = [ [0]*(len(job)+1) for job in jobs ]
    for idx, job in enumerate(jobs):
        s = 0
        for p in range(len(job)-1, -1, -1):
            s += job[p][1]
            rem_sum[idx][p] = s

    @lru_cache(maxsize=None)
    def dfs(state, mach_times, job_times):
        nonlocal best
        # state: tuple of next step index for each job
        # mach_times, job_times are tuples of current finish times
        cur_mach = list(mach_times)
        cur_job = list(job_times)

        # lower bound: max of current makespan and each job's earliest possible finish
        cur_mk = max(cur_mach + cur_job)
        lb = cur_mk
        for idx, pos in enumerate(state):
            if pos < len(jobs[idx]):
                lb = max(lb, cur_job[idx] + rem_sum[idx][pos])
        if lb >= best:
            return

        # if all done
        if all(pos == len(jobs[i]) for i, pos in enumerate(state)):
            best = cur_mk
            return

        # generate ready operations
        ready = []
        for i, pos in enumerate(state):
            if pos < len(jobs[i]):
                ws, d = jobs[i][pos]
                est = max(cur_job[i], cur_mach[ws])
                ready.append((est + d, est, i, ws, d))  # sort by earliest finish

        ready.sort()
        for _, est, i, ws, d in ready:
            # schedule this operation
            new_state = list(state)
            new_state[i] += 1
            new_mach = cur_mach[:]
            new_job = cur_job[:]
            finish = est + d
            new_mach[ws] = finish
            new_job[i] = finish
            dfs(tuple(new_state), tuple(new_mach), tuple(new_job))

    init_state = tuple(0 for _ in range(j))
    dfs(init_state, tuple([0]*m), tuple([0]*j))
    print(best)

if __name__ == "__main__":
    solve()
