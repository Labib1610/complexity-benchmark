import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    
    ships = []
    for _ in range(j):
        p = int(next(it))
        steps = []
        for _ in range(p):
            bay = int(next(it))
            dur = int(next(it))
            steps.append((bay - 1, dur))  # 0-index bays
        ships.append(steps)
    
    # State: tuple of current step indices for each ship
    start_state = tuple(0 for _ in range(j))
    
    # dist[state] = earliest completion time for last finished step
    dist = {start_state: 0}
    dq = deque([start_state])
    
    while dq:
        state = dq.popleft()
        cur_t = dist[state]
        
        # Check if all ships finished
        all_done = True
        for i in range(j):
            if state[i] < len(ships[i]):
                all_done = False
                break
        if all_done:
            print(cur_t)
            return
        
        # Try advancing each ship by one step
        for i in range(j):
            step_idx = state[i]
            if step_idx >= len(ships[i]):
                continue
            
            bay, dur = ships[i][step_idx]
            
            # Earliest start time for this step:
            # 1. Previous step of same ship must be done (cur_t)
            # 2. Bay must be free (no other ship using it at that time)
            start_time = cur_t
            for k in range(j):
                if k == i:
                    continue
                other_idx = state[k]
                if other_idx < len(ships[k]):
                    other_bay, other_dur = ships[k][other_idx]
                    if other_bay == bay:
                        # Other ship occupies bay until dist[state]
                        start_time = max(start_time, dist[state])
            
            finish_time = start_time + dur
            
            # Create new state
            new_state = list(state)
            new_state[i] += 1
            new_state = tuple(new_state)
            
            if new_state not in dist or finish_time < dist[new_state]:
                dist[new_state] = finish_time
                dq.appendleft(new_state)  # 0-1 BFS style (all edges have weight 0 in terms of priority)
    
    print(0)

if __name__ == "__main__":
    solve()
