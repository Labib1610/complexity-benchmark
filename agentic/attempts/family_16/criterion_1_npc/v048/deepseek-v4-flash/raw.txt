We can model this as a scheduling problem with precedence constraints and resource (equipment) limits.  
Since the number of tasks is small (≤ 50), we can use a priority queue–based simulation or a DP over task states.

Here’s the Python solution:  

```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().strip().split()
    idx = 0
    j = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    
    tasks = []  # (song, order, equipment, duration)
    for song in range(j):
        o = int(input_data[idx]); idx += 1
        for order in range(o):
            eq = int(input_data[idx]); idx += 1
            dur = int(input_data[idx]); idx += 1
            tasks.append((song, order, eq, dur))
    
    n = len(tasks)
    # Precedence: for each song, task order must be sequential
    # We'll track next task index per song
    next_task = [0] * j
    # For each song, list of tasks in order
    song_tasks = [[] for _ in range(j)]
    for song, order, eq, dur in tasks:
        song_tasks[song].append((eq, dur))
    
    # Min-heap for scheduling: (end_time, equipment, song, order)
    heap = []
    # Equipment availability time
    eq_free = [0] * (m + 1)
    
    # Initially push first task of each song
    for song in range(j):
        if song_tasks[song]:
            eq, dur = song_tasks[song][0]
            heapq.heappush(heap, (0, eq, song, 0))
    
    max_end = 0
    while heap:
        start, eq, song, order = heapq.heappop(heap)
        # Start time is max(equipment free, song free)
        # song free time is end of previous task in same song
        # We track song free time implicitly: start is when we pushed it
        # But we need to ensure song's previous task finished.
        # Since we push next task only after current finishes, start is valid.
        dur = song_tasks[song][order][1]
        end = max(start, eq_free[eq]) + dur
        eq_free[eq] = end
        max_end = max(max_end, end)
        
        # Push next task of same song if exists
        if order + 1 < len(song_tasks[song]):
            next_eq, next_dur = song_tasks[song][order + 1]
            heapq.heappush(heap, (end, next_eq, song, order + 1))
    
    print(max_end)

if __name__ == "__main__":
    solve()
```
