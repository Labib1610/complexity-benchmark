import sys
from collections import defaultdict

def solve():
    j, m = map(int, input().split())
    
    compositions = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        segments = []
        for i in range(o):
            instrument = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            segments.append((instrument, duration))
        compositions.append(segments)
    
    # Binary search on the answer
    def can_finish_in_time(limit):
        # For each instrument, we keep track of when it becomes free
        instrument_free_time = [0] * (m + 1)  # 1-indexed
        
        # For each composition, we track when we can start it
        composition_start_times = [0] * j
        
        # We simulate the process
        # At each step, we try to schedule the next segment of some composition
        # We use a greedy approach: always try to schedule the earliest available segment
        
        # Keep track of which segment we're at in each composition
        current_segment = [0] * j
        
        # Keep track of when each composition finishes
        composition_end_times = [0] * j
        
        # While there are still segments to schedule
        while True:
            # Find a composition that has unscheduled segments
            next_composition = -1
            earliest_start = float('inf')
            
            for comp_idx in range(j):
                if current_segment[comp_idx] < len(compositions[comp_idx]):
                    # This composition has more segments
                    # The earliest we can start this segment is:
                    # max(current composition progress, instrument free time)
                    segment_idx = current_segment[comp_idx]
                    instrument, duration = compositions[comp_idx][segment_idx]
                    
                    earliest_possible_start = max(
                        composition_start_times[comp_idx],
                        instrument_free_time[instrument]
                    )
                    
                    if earliest_possible_start < earliest_start:
                        earliest_start = earliest_possible_start
                        next_composition = comp_idx
            
            # If no more segments, we're done
            if next_composition == -1:
                break
                
            # Schedule the segment
            comp_idx = next_composition
            segment_idx = current_segment[comp_idx]
            instrument, duration = compositions[comp_idx][segment_idx]
            
            start_time = max(
                composition_start_times[comp_idx],
                instrument_free_time[instrument]
            )
            
            end_time = start_time + duration
            
            # Check if we exceed the limit
            if end_time > limit:
                return False
                
            # Update state
            instrument_free_time[instrument] = end_time
            composition_start_times[comp_idx] = end_time
            current_segment[comp_idx] += 1
            if segment_idx == len(compositions[comp_idx]) - 1:
                # This was the last segment of this composition
                composition_end_times[comp_idx] = end_time
        
        # All compositions finished within the limit
        return True
    
    # Binary search for the minimum time
    left, right = 0, 50000  # Upper bound: 50 segments * 1000 max duration
    result = right
    
    while left <= right:
        mid = (left + right) // 2
        if can_finish_in_time(mid):
            result = mid
            right = mid - 1
        else:
            left = mid + 1
    
    print(result)

solve()
