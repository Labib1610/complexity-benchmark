import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    J = int(next(it))
    M = int(next(it))
    jobs = []
    total_ops = 0
    for _ in range(J):
        o = int(next(it))
        ops = []
        for __ in range(o):
            inst = int(next(it)) - 1
            dur = int(next(it))
            ops.append((inst, dur))
        jobs.append(ops)
        total_ops += o

    # precompute remaining total duration per job from each position
    rem_job = [ [0]*(len(jobs[i])+1) for i in range(J) ]
    for i in range(J):
        acc = 0
        for pos in range(len(jobs[i])-1, -1, -1):
            acc += jobs[i][pos][1]
            rem_job[i][pos] = acc
        rem_job[i][len(jobs[i])] = 0

    # remaining duration per machine from each job position (for bound)
    rem_machine = [ [0]*M for _ in range(total_ops+1) ]  # not used, compute on fly

    best = [float('inf')]

    # initial states
    job_pos = [0]*J
    job_ready = [0]*J
    mach_ready = [0]*M

    def lower_bound():
        # optimistic makespan ignoring machine conflicts
        lb = 0
        for i in range(J):
            lb = max(lb, job_ready[i] + rem_job[i][job_pos[i]])
        for m in range(M):
            # sum remaining durations on this machine
            s = mach_ready[m]
            for i in range(J):
                pos = job_pos[i]
                ops = jobs[i]
                while pos < len(ops) and ops[pos][0] != m:
                    pos += 1
                # accumulate all remaining ops of this machine
                while pos < len(ops):
                    if ops[pos][0] == m:
                        s += ops[pos][1]
                    pos += 1
            lb = max(lb, s)
        return lb

    def dfs(done):
        if done == total_ops:
            cur = max(job_ready)
            if cur < best[0]:
                best[0] = cur
            return
        # prune by lower bound
        if lower_bound() >= best[0]:
            return

        # collect ready operations
        ready = []
        for i in range(J):
            p = job_pos[i]
            if p < len(jobs[i]):
                inst, dur = jobs[i][p]
                start = max(job_ready[i], mach_ready[inst])
                ready.append((start, i, inst, dur))
        # order by earliest start (heuristic)
        ready.sort()
        for start, i, inst, dur in ready:
            # schedule this operation
            finish = start + dur
            # save state
            old_job_ready = job_ready[i]
            old_mach_ready = mach_ready[inst]
            old_pos = job_pos[i]

            job_ready[i] = finish
            mach_ready[inst] = finish
            job_pos[i] += 1

            dfs(done + 1)

            # restore
            job_ready[i] = old_job_ready
            mach_ready[inst] = old_mach_ready
            job_pos[i] = old_pos

    dfs(0)
    print(best[0])

if __name__ == "__main__":
    solve()
