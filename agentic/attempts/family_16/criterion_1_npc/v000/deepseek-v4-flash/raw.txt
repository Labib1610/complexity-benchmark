We can model this as a job-shop scheduling problem. Since the number of jobs is small (≤10) and total operations ≤50, we can use a branch-and-bound search with a lower bound to prune.

Here’s the Python solution:  

```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    
    jobs = []
    for _ in range(j):
        o = int(next(it))
        ops = []
        for _ in range(o):
            machine = int(next(it)) - 1
            duration = int(next(it))
            ops.append((machine, duration))
        jobs.append(ops)
    
    # Precompute total work per machine for lower bound
    machine_load = [0] * m
    for ops in jobs:
        for machine, dur in ops:
            machine_load[machine] += dur
    max_load = max(machine_load)
    
    total_ops = sum(len(ops) for ops in jobs)
    
    # State: for each job, next operation index; for each machine, next free time
    # We'll do DFS with pruning.
    
    best = float('inf')
    
    # job_next_op[j] = index of next operation to schedule for job j
    job_next_op = [0] * j
    # machine_free[m] = time when machine m becomes free
    machine_free = [0] * m
    # job_finish[j] = finish time of last scheduled op of job j
    job_finish = [0] * j
    
    # Precompute operation durations for quick access
    op_dur = [[dur for _, dur in ops] for ops in jobs]
    op_mach = [[mach for mach, _ in ops] for ops in jobs]
    
    def dfs(scheduled_count, current_time):
        nonlocal best
        if current_time >= best:
            return
        if scheduled_count == total_ops:
            best = current_time
            return
        
        # Lower bound: max of current time, max machine load, and any job's remaining time
        # Simple lower bound: max(current_time, max_load)
        if max(current_time, max_load) >= best:
            return
        
        # Try to schedule next operation of each job
        for job in range(j):
            if job_next_op[job] >= len(jobs[job]):
                continue
            op_idx = job_next_op[job]
            machine = op_mach[job][op_idx]
            dur = op_dur[job][op_idx]
            start_time = max(job_finish[job], machine_free[machine])
            finish_time = start_time + dur
            
            # Prune if finish_time >= best
            if finish_time >= best:
                continue
            
            # Save state
            old_job_finish = job_finish[job]
            old_machine_free = machine_free[machine]
            old_next = job_next_op[job]
            
            # Update
            job_finish[job] = finish_time
            machine_free[machine] = finish_time
            job_next_op[job] = op_idx + 1
            
            dfs(scheduled_count + 1, max(current_time, finish_time))
            
            # Restore
            job_finish[job] = old_job_finish
            machine_free[machine] = old_machine_free
            job_next_op[job] = old_next
    
    dfs(0, 0)
    print(best)

if __name__ == "__main__":
    solve()
```
