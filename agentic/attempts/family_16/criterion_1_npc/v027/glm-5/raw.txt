import sys
from heapq import heappush, heappop

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    j = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    proposals = []
    for _ in range(j):
        o = int(data[idx]); idx += 1
        stages = []
        for _ in range(o):
            e = int(data[idx]); idx += 1
            d = int(data[idx]); idx += 1
            stages.append((e - 1, d))
        proposals.append(stages)
    
    # Greedy event-driven simulation
    # expert_free[e] = earliest time expert e is free
    # proposal_ready[p] = earliest time proposal p can start its next stage
    # proposal_idx[p] = index of next stage for proposal p
    
    expert_free = [0] * m
    proposal_ready = [0] * j
    proposal_idx = [0] * j
    
    # Priority queue of events: (time, type, ...)
    # type 0: proposal becomes ready (proposal_ready updated)
    # type 1: expert becomes free (expert_free updated)
    # We process events in time order
    
    # Actually simpler: repeatedly find the proposal that can start its next stage earliest
    # But with events we can be more efficient
    
    # Let's use a simple approach: at each step, find the earliest possible start
    # among all proposals that haven't finished, considering both proposal_ready and expert_free
    
    remaining = j
    while remaining > 0:
        best_time = float('inf')
        best_p = -1
        for p in range(j):
            if proposal_idx[p] < len(proposals[p]):
                s = proposal_idx[p]
                e, d = proposals[p][s]
                start = max(proposal_ready[p], expert_free[e])
                if start < best_time:
                    best_time = start
                    best_p = p
        if best_p == -1:
            break
        p = best_p
        s = proposal_idx[p]
        e, d = proposals[p][s]
        start = best_time
        end = start + d
        expert_free[e] = end
        proposal_ready[p] = end
        proposal_idx[p] += 1
        if proposal_idx[p] == len(proposals[p]):
            remaining -= 1
    
    print(max(proposal_ready) if proposal_ready else 0)

solve()
