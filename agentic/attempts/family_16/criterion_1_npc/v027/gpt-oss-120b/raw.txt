import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    jobs = []
    for _ in range(j):
        o = int(next(it))
        ops = []
        for __ in range(o):
            expert = int(next(it)) - 1
            dur = int(next(it))
            ops.append((expert, dur))
        jobs.append(ops)

    # pre‑compute remaining sums per job and per machine for fast lower bound
    total_ops = sum(len(job) for job in jobs)
    job_rem_sum = [ [0]*(len(job)+1) for job in jobs ]
    for idx, job in enumerate(jobs):
        s = 0
        for p in range(len(job)-1, -1, -1):
            s += job[p][1]
            job_rem_sum[idx][p] = s

    # machine remaining sums for each state will be computed on the fly
    # but we can store total remaining per machine for each job position combination
    # given small limits we compute each time.

    best = [float('inf')]
    from functools import lru_cache

    @lru_cache(maxsize=None)
    def dfs(pos_tuple, job_ready_tuple, mach_ready_tuple):
        # pos_tuple: tuple of positions per job
        # job_ready_tuple, mach_ready_tuple: current completion times
        # compute current makespan
        cur_mk = max(mach_ready_tuple) if mach_ready_tuple else 0
        if cur_mk >= best[0]:
            return float('inf')
        # check if all done
        if all(pos == len(jobs[i]) for i, pos in enumerate(pos_tuple)):
            best[0] = cur_mk
            return cur_mk

        # lower bound
        lb = cur_mk
        # job based
        for i, pos in enumerate(pos_tuple):
            if pos < len(jobs[i]):
                lb = max(lb, job_ready_tuple[i] + job_rem_sum[i][pos])
        # machine based
        mach_rem = [0]*m
        for i, pos in enumerate(pos_tuple):
            for p in range(pos, len(jobs[i])):
                mach, dur = jobs[i][p]
                mach_rem[mach] += dur
        for k in range(m):
            if mach_rem[k]:
                lb = max(lb, mach_ready_tuple[k] + mach_rem[k])
        if lb >= best[0]:
            return float('inf')

        # generate successors
        min_finish = float('inf')
        # list of ready operations
        for i, pos in enumerate(pos_tuple):
            if pos == len(jobs[i]):
                continue
            mach, dur = jobs[i][pos]
            start = max(job_ready_tuple[i], mach_ready_tuple[mach])
            finish = start + dur
            # build new state
            new_pos = list(pos_tuple)
            new_pos[i] += 1
            new_job_ready = list(job_ready_tuple)
            new_job_ready[i] = finish
            new_mach_ready = list(mach_ready_tuple)
            new_mach_ready[mach] = finish
            res = dfs(tuple(new_pos), tuple(new_job_ready), tuple(new_mach_ready))
            if res < min_finish:
                min_finish = res
        return min_finish

    init_pos = tuple(0 for _ in range(j))
    init_job_ready = tuple(0 for _ in range(j))
    init_mach_ready = tuple(0 for _ in range(m))
    ans = dfs(init_pos, init_job_ready, init_mach_ready)
    print(ans)

if __name__ == "__main__":
    solve()
