import heapq
from collections import defaultdict

def solve():
    j, m = map(int, input().split())
    
    # Read all jobs
    jobs = []
    for _ in range(j):
        line = list(map(int, input().split()))
        o = line[0]
        steps = []
        for i in range(o):
            station = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            steps.append((station, duration))
        jobs.append(steps)
    
    # For each job, we track which step it's currently on and when it can start that step
    # For each machine, we track when it becomes free
    
    # Current step index for each job
    current_step = [0] * j
    
    # When each job can start its current step
    job_ready_time = [0] * j
    
    # When each machine becomes free
    machine_free_time = [0] * (m + 1)  # 1-indexed
    
    # Priority queue: (time, event_type, job_id)
    # event_type: 0 = job ready, 1 = machine free
    pq = []
    
    # Initialize: all jobs are ready at time 0
    for job_id in range(j):
        heapq.heappush(pq, (0, 0, job_id))
    
    finished_jobs = 0
    total_steps = sum(len(job) for job in jobs)
    completed_steps = 0
    
    while pq and completed_steps < total_steps:
        current_time, event_type, job_id = heapq.heappop(pq)
        
        if event_type == 0:  # Job is ready to schedule next step
            if current_step[job_id] >= len(jobs[job_id]):
                continue  # Job already finished
            
            station, duration = jobs[job_id][current_step[job_id]]
            
            # When can this job start on this station?
            start_time = max(current_time, job_ready_time[job_id], machine_free_time[station])
            
            # Schedule the step
            end_time = start_time + duration
            
            # Update when machine becomes free
            machine_free_time[station] = end_time
            
            # Update when job can start next step
            job_ready_time[job_id] = end_time
            
            # Move to next step
            current_step[job_id] += 1
            completed_steps += 1
            
            # If job is finished, increment counter
            if current_step[job_id] >= len(jobs[job_id]):
                finished_jobs += 1
                if finished_jobs == j:
                    # All jobs finished, but we need to make sure all steps are done
                    pass
            else:
                # Schedule next step availability
                heapq.heappush(pq, (end_time, 0, job_id))
                
        # For machine free events, we don't need separate handling since we track machine_free_time
        
    # Find the maximum time any machine is busy until
    result = max(machine_free_time)
    print(result)

solve()
