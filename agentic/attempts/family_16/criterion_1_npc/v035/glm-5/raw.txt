```python
import sys
from collections import defaultdict

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    
    workflows = []
    for _ in range(j):
        o = int(next(it))
        ops = []
        for _ in range(o):
            machine = int(next(it))
            duration = int(next(it))
            ops.append((machine - 1, duration))  # 0-indexed machines
        workflows.append(ops)
    
    # State: (op_idx for each workflow)
    # For each state, track: (min_time, machine_free_times, workflow_finish_times)
    initial_state = tuple(0 for _ in range(j))
    initial_machines = tuple(0 for _ in range(m))
    initial_finishes = tuple(0 for _ in range(j))
    
    # Priority queue: (time, state, machine_times, finish_times)
    import heapq
    pq = [(0, initial_state, initial_machines, initial_finishes)]
    
    # Best time for each state
    best = defaultdict(lambda: float('inf'))
    best[initial_state] = 0
    
    result = float('inf')
    
    while pq:
        time, state, machines, finishes = heapq.heappop(pq)
        
        if time >= result:
            continue
            
        if best[state] < time:
            continue
        
        # Check if all workflows are done
        all_done = True
        for i in range(j):
            if state[i] < len(workflows[i]):
                all_done = False
                break
        
        if all_done:
            result = min(result, max(finishes))
            continue
        
        # Try to schedule each workflow's next operation
        for wf in range(j):
            op_idx = state[wf]
            if op_idx >= len(workflows[wf]):
                continue
            
            machine, duration = workflows[wf][op_idx]
            
            # Earliest start time for this operation
            earliest_start = finishes[wf]  # After previous operation in workflow
            earliest_start = max(earliest_start, machines[machine])  # After machine is free
            
            start_time = earliest_start
            end_time = start_time + duration
            
            new_state = list(state)
            new_state[wf] = op_idx + 1
            new_state = tuple(new_state)
            
            new_machines = list(machines)
            new_machines[machine] = end_time
            new_machines = tuple(new_machines)
            
            new_finishes = list(finishes)
            new_finishes[wf] = end_time
            new_finishes = tuple(new_finishes)
            
            new_time = max(time, end_time)
            
            if new_time < best[new_state]:
                best[new_state] = new_time
                heapq.heappush(pq, (new_time, new_state, new_machines, new_finishes))
    
    print(result)

if __name__ == "__main__":
    solve()
```
