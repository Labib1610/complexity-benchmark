import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    jobs = []
    total_ops = 0
    for _ in range(j):
        o = int(next(it))
        ops = []
        for __ in range(o):
            machine = int(next(it)) - 1
            dur = int(next(it))
            ops.append((machine, dur))
        jobs.append(ops)
        total_ops += o

    # precompute remaining sums for pruning
    rem_job_sum = [[0]*(len(job)+1) for job in jobs]  # suffix sums
    for idx, job in enumerate(jobs):
        s = 0
        for k in range(len(job)-1, -1, -1):
            s += job[k][1]
            rem_job_sum[idx][k] = s
        rem_job_sum[idx][len(job)] = 0

    # for each machine, list of (job, pos) to quickly sum remaining
    machine_ops = [[] for _ in range(m)]
    for jid, job in enumerate(jobs):
        for pos, (mach, dur) in enumerate(job):
            machine_ops[mach].append((jid, pos))

    # suffix sums per machine per job position
    rem_mach_sum = {}
    for mach in range(m):
        # for each job, we need sum of remaining durations on this machine after a given position
        # we can compute per job prefix of that machine
        per_job = {}
        for jid, pos in machine_ops[mach]:
            # compute sum of durations of this machine for this job from pos onward
            s = 0
            for p in range(pos, len(jobs[jid])):
                if jobs[jid][p][0] == mach:
                    s += jobs[jid][p][1]
            per_job[(jid, pos)] = s
        rem_mach_sum[mach] = per_job

    best = [float('inf')]
    from functools import lru_cache

    @lru_cache(maxsize=None)
    def dfs(state, mach_ready):
        # state: tuple of completed counts per job
        # mach_ready: tuple of machine ready times
        # returns minimal possible makespan from this state (or updates global best)
        # compute current max time
        cur_max = max(mach_ready) if mach_ready else 0
        # lower bound
        lb = cur_max
        # job based
        for jid, done in enumerate(state):
            if done < len(jobs[jid]):
                lb = max(lb, mach_ready[0] + 0)  # placeholder
                # job ready time is when its last finished operation completed
                # we can compute it as max of machine ready times of its previous ops? easier: track separately?
        # Instead we maintain job ready times implicitly via state and machine ready times not enough.
        # We'll pass job ready times as part of state via separate tuple.
        return

    # To simplify, we keep explicit job ready times array.
    start_job_ready = tuple(0 for _ in range(j))
    start_mach_ready = tuple(0 for _ in range(m))

    @lru_cache(maxsize=None)
    def rec(state, job_ready, mach_ready):
        # state: tuple of completed counts per job
        # job_ready: tuple of time when each job is ready for next op
        # mach_ready: tuple of machine ready times
        cur_max = max(job_ready + mach_ready)
        if cur_max >= best[0]:
            return
        if all(state[i] == len(jobs[i]) for i in range(j)):
            best[0] = cur_max
            return
        # lower bound: max of each job's earliest finish and each machine's earliest finish
        lb = cur_max
        for jid in range(j):
            done = state[jid]
            if done < len(jobs[jid]):
                lb = max(lb, job_ready[jid] + rem_job_sum[jid][done])
        for mach in range(m):
            # sum remaining durations on this machine
            s = 0
            for (jid, pos) in machine_ops[mach]:
                done = state[jid]
                # if this operation not yet done and its position >= done
                if pos >= done:
                    # need to check if this specific operation is still pending
                    # we sum all remaining on this machine
                    # approximate by total remaining on machine across all jobs
                    # compute per job remaining on this machine
                    # use precomputed per job suffix for this machine
                    # find first occurrence >= done
                    # we just add all remaining durations of this machine for each job
                    # which is safe upper bound
                    # compute sum of durations of this machine for this job from its current position
                    # find next position of this machine in job
                    # iterate through remaining ops of job
                    for p in range(done, len(jobs[jid])):
                        if jobs[jid][p][0] == mach:
                            s += jobs[jid][p][1]
            lb = max(lb, mach_ready[mach] + s)
        if lb >= best[0]:
            return

        # generate possible next operations
        for jid in range(j):
            done = state[jid]
            if done == len(jobs[jid]):
                continue
            mach, dur = jobs[jid][done]
            start = max(job_ready[jid], mach_ready[mach])
            finish = start + dur
            # create new state
            new_state = list(state)
            new_state[jid] = done + 1
            new_state = tuple(new_state)
            new_job_ready = list(job_ready)
            new_job_ready[jid] = finish
            new_job_ready = tuple(new_job_ready)
            new_mach_ready = list(mach_ready)
            new_mach_ready[mach] = finish
            new_mach_ready = tuple(new_mach_ready)
            rec(new_state, new_job_ready, new_mach_ready)

    rec(tuple(0 for _ in range(j)), start_job_ready, start_mach_ready)
    print(best[0])

if __name__ == "__main__":
    solve()
