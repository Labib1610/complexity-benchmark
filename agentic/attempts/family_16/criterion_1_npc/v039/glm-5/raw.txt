from heapq import heappush, heappop

    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    components = []
    for _ in range(j):
        o = int(next(it))
        tests = []
        for _ in range(o):
            f = int(next(it)) - 1
            d = int(next(it))
            tests.append((f, d))
        components.append(tests)

    # Best time found
    best = float('inf')
    
    # Visited: (comp_idx, test_idx, facility_free_times) -> min time
    visited = {}
    
    pq = [(0, 0, 0, init_facility)]
    
    while pq:
        time, ci, ti, fac = heappop(pq)
        
        if time >= best:
            continue
        
        key = (ci, ti, fac)
        if key in visited and visited[key] <= time:
            continue
        visited[key] = time
        
        # If all components done
        if ci == j:
            if time < best:
                best = time
            continue
        
        # Option 1: Schedule test (ci, ti) if possible
        f, d = components[ci][ti]
        fac_free = fac[f]
        start = max(time, fac_free)
        end = start + d
        
        if end < best:
            new_fac = list(fac)
            new_fac[f] = end
            new_fac = tuple(new_fac)
            new_ti = ti + 1
            new_ci = ci
            if new_ti == len(components[ci]):
                new_ci = ci + 1
                new_ti = 0
            heappush(pq, (end, new_ci, new_ti, new_fac))
        
        # Option 2: Skip this test for now (try scheduling later component's test)
        # We need to allow interleaving

        # Actually, the state should track progress of ALL components

    # Redo with proper state: (progress of each component, facility free times)
    
    # State: (comp_progress_tuple, facility_free_times)

    init_progress = tuple([0] * j)
