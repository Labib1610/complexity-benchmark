import sys
from heapq import heappush, heappop

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    petitions = []
    for _ in range(j):
        o = int(next(it))
        steps = []
        for _ in range(o):
            mag = int(next(it))
            hrs = int(next(it))
            steps.append((mag, hrs))
        petitions.append(steps)
    
    # State: (time, magistrate_free_times tuple, petition_progress tuple, petition_remaining tuple)
    # Use A* / Dijkstra with pruning
    
    # For each petition, precompute remaining time from each step
    rem = []
    for steps in petitions:
        r = [0] * (len(steps) + 1)
        for k in range(len(steps) - 1, -1, -1):
            r[k] = r[k+1] + steps[k][1]
        rem.append(r)
    
    # Initial state
    init_mag = tuple(0 for _ in range(m + 1))  # magistrate 1..m
    init_prog = tuple(0 for _ in range(j))
    init_rem = tuple(rem[i][0] for i in range(j))
    
    # Lower bound: max over petitions of remaining time, and max magistrate free time + remaining work on that magistrate
    # Simple lower bound: max(petition remaining times)
    # Better lower bound: for each magistrate, sum of work remaining on that magistrate + magistrate free time
    
    # Precompute for each petition and each step, the magistrate
    # For lower bound, compute remaining work per magistrate from current progress
    
    def lower_bound(mag_free, prog):
        lb = 0
        # remaining time per petition
        for i in range(j):
            lb = max(lb, rem[i][prog[i]])
        # remaining work per magistrate
        mag_work = [0] * (m + 1)
        for i in range(j):
            p = prog[i]
            for k in range(p, len(petitions[i])):
                mag_work[petitions[i][k][0]] += petitions[i][k][1]
        for mag_id in range(1, m + 1):
            lb = max(lb, mag_free[mag_id] + mag_work[mag_id])
        return lb
    
    # Dijkstra
    # State: (lb, time, mag_free, prog)
    # We use a dict to store best time for each (mag_free, prog)
    
    start = (lower_bound(init_mag, init_prog), 0, init_mag, init_prog)
    heap = [start]
    best = {}
    best[(init_mag, init_prog)] = 0
    
    while heap:
        lb, time, mag_free, prog = heappop(heap)
        if time > best.get((mag_free, prog), float('inf')):
            continue
        
        # Check if all done
        all_done = True
        for i in range(j):
            if prog[i] < len(petitions[i]):
                all_done = False
                break
        if all_done:
            print(time)
            return
        
        # Find the earliest available petition step
        # We can choose any petition that has a next step, and schedule it at max(time, mag_free[mag])
        # But to reduce branching, we can use a greedy: pick the petition that can start earliest?
        # Actually, we need to consider all possibilities for correctness.
        # But branching factor could be up to j=10, and depth up to 50, so 10^50 is too much.
        # We need better approach.
        
        # Key insight: we should only consider scheduling the step that can start earliest among all possible next steps.
        # But that's not necessarily optimal because delaying a step might allow another step to go first.
        # However, if we always schedule the step with the earliest possible start time, we might miss optimal.
        # 
        # Alternative: we can use the fact that at any point, the next event is either:
        # - A magistrate becomes free
        # - We start a step on a magistrate that is already free
        # 
        # Actually, the optimal schedule can be assumed to be "non-delay": no magistrate is idle when there's work available for them.
        # But a magistrate might be idle if the petition is waiting for another magistrate.
        # 
        # Let's think differently: we can model this as a shortest path problem where we always advance the "earliest possible" step.
        # But we need to consider which step to advance when multiple are available.
        # 
        # Given the constraints (j<=10, m<=10, total steps <=50), maybe we can do a more careful search.
        # 
        # Let's try: at each state, find all petitions that can start their next step immediately (magistrate is free at current time).
        # If there are multiple, we need to try all orderings? That's too many.
        # 
        # Actually, if multiple petitions can start on different magistrates, they can all start simultaneously.
        # If multiple petitions want the same magistrate, we need to order them.
        # 
        # So the branching only happens when multiple petitions compete for the same magistrate.
        # 
        # Let's implement this: at each state, find the set of petitions that can start their next step.
        # Group by magistrate. For each magistrate, if multiple petitions want it, we need to try all permutations? No, we can just try all possible next steps for that magistrate.
        # But if different magistrates are free, we can start all of them simultaneously.
        # 
        # Wait, but the current time is the time of the state. Some magistrates might be free at time < current time, but we've already advanced past that.
        # Actually, in our state, time is the current "global" time, and mag_free[mag] is when that magistrate becomes free.
        # 
        # Hmm, this is getting complicated. Let's think of a cleaner formulation.
        # 
        # Alternative: event-based simulation. The state is (mag_free, prog). The "time" is min(mag_free[mag] for mag that has a petition waiting).
        # Actually, the time should be the minimum over all magistrates that have work available.
        # 
        # Let's redefine: the state is (mag_free, prog). The "current time" is the minimum mag_free among magistrates that have at least one petition ready to be processed by them.
        # 
        # At current time, we can start any petition whose next step's magistrate is free at current time.
        # We should start all such petitions simultaneously (on different magistrates).
        # If multiple petitions want the same magistrate, we need to choose one (or more, sequentially).
        # 
        # But this is still complex. Let's just do Dijkstra with the state (mag_free, prog) and transition by picking one petition and scheduling its next step.
        # The time of the new state is max(time, mag_free[mag]) + hours.
        # But we need to be careful: the "time" in the state should represent the time up to which we've scheduled.
        # 
        # Actually, let's use a different approach: the state is (mag_free, prog), and the cost is the maximum completion time.
        # When we schedule a step, the new mag_free for that magistrate becomes max(mag_free[mag], ...) + hours.
        # But what is the "..."? It's the time when the petition is ready, which depends on when its previous step finished.
        # 
        # So we also need to track when each petition's current step finished. Let's add that to the state.
        # 
        # State: (mag_free, prog, petition_ready_time)
        # petition_ready_time[i] = time when petition i is ready to start its next step (prog[i]).
        # 
        # When we schedule step k of petition i:
        #   start_time = max(mag_free[mag], petition_ready_time[i])
        #   end_time = start_time + hours
        #   new_mag_free[mag] = end_time
        #   new_prog[i] = k + 1
        #   new_petition_ready_time[i] = end_time
        # 
        # The answer is max(petition_ready_time) when all prog == len.
        # 
        # This is cleaner. Let's implement this.
        
        pass
    
    # Reimplement with the cleaner state
    init_ready = tuple(0 for _ in range(j))
    start = (lower_bound(init_mag, init_prog), 0, init_mag, init_prog, init_ready)
    heap = [start]
    best = {}
    best[(init_mag, init_prog, init_ready)] = 0
    
    while heap:
        lb, time, mag_free, prog, ready = heappop(heap)
        key = (mag_free, prog, ready)
        if time > best.get(key, float('inf')):
            continue
        
        all_done = True
        for i in range(j):
            if prog[i] < len(petitions[i]):
                all_done = False
                break
        if all_done:
            print(time)
            return
        
        # Try scheduling each petition's next step
        for i in range(j):
            p = prog[i]
            if p >= len(petitions[i]):
                continue
            mag, hrs = petitions[i][p]
            start_time = max(mag_free[mag], ready[i])
            end_time = start_time + hrs
            
            new_mag = list(mag_free)
            new_mag[mag] = end_time
            new_mag = tuple(new_mag)
            
            new_prog = list(prog)
            new_prog[i] = p + 1
            new_prog = tuple(new_prog)
            
            new_ready = list(ready)
            new_ready[i] = end_time
            new_ready = tuple(new_ready)
            
            new_time = max(time, end_time)
            new_key = (new_mag, new_prog, new_ready)
            
            if new_time < best.get(new_key, float('inf')):
                best[new_key] = new_time
                new_lb = lower_bound(new_mag, new_prog)
                heappush(heap, (max(new_time, new_lb), new_time, new_mag, new_prog, new_ready))

solve()
