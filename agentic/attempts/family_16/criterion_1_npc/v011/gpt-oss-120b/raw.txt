import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    J = int(next(it))
    M = int(next(it))
    jobs = []
    total_ops = 0
    for _ in range(J):
        o = int(next(it))
        ops = []
        for __ in range(o):
            mach = int(next(it)) - 1
            dur = int(next(it))
            ops.append((mach, dur))
        jobs.append(ops)
        total_ops += o

    # precompute remaining processing time per job from each position
    rem_job = [[0]*(len(job)+1) for job in jobs]
    for idx, job in enumerate(jobs):
        s = 0
        for i in range(len(job)-1, -1, -1):
            s += job[i][1]
            rem_job[idx][i] = s

    # precompute remaining processing time per machine from each position
    # we will compute on the fly: sum of durations of all unscheduled ops on each machine
    # but we can also precompute total per machine
    total_per_machine = [0]*M
    for job in jobs:
        for mach, dur in job:
            total_per_machine[mach] += dur

    best = [float('inf')]

    # initial state
    job_pos = [0]*J          # next op index for each job
    mach_free = [0]*M        # when each machine becomes free
    job_finish = [0]*J       # when each job finishes its last done op

    # remaining per machine can be derived from job_pos
    def dfs(done_ops, cur_makespan):
        # pruning by current best
        if cur_makespan >= best[0]:
            return
        if done_ops == total_ops:
            best[0] = cur_makespan
            return

        # lower bound: max of current makespan, each job's earliest possible finish,
        # each machine's earliest possible finish
        lb = cur_makespan
        # job lower bounds
        for j in range(J):
            if job_pos[j] < len(jobs[j]):
                earliest = max(job_finish[j], mach_free[jobs[j][job_pos[j]][0]])
                # plus remaining processing of this job after this op
                lb = max(lb, earliest + rem_job[j][job_pos[j]])
        # machine lower bounds
        # compute remaining per machine
        rem_mach = [0]*M
        for j in range(J):
            for p in range(job_pos[j], len(jobs[j])):
                mach, dur = jobs[j][p]
                rem_mach[mach] += dur
        for m in range(M):
            lb = max(lb, mach_free[m] + rem_mach[m])
        if lb >= best[0]:
            return

        # generate list of ready operations
        ready = []
        for j in range(J):
            if job_pos[j] < len(jobs[j]):
                mach, dur = jobs[j][job_pos[j]]
                start = max(job_finish[j], mach_free[mach])
                ready.append((start, j, mach, dur))
        # sort to try earlier start first (helps pruning)
        ready.sort()
        for start, j, mach, dur in ready:
            # schedule this operation
            old_mfree = mach_free[mach]
            old_jfinish = job_finish[j]
            new_finish = start + dur
            mach_free[mach] = new_finish
            job_finish[j] = new_finish
            job_pos[j] += 1
            dfs(done_ops + 1, max(cur_makespan, new_finish))
            # backtrack
            job_pos[j] -= 1
            mach_free[mach] = old_mfree
            job_finish[j] = old_jfinish

    dfs(0,0)
    print(best[0])

if __name__ == "__main__":
    solve()
