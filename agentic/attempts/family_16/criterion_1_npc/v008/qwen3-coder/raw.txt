import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    m = int(data[1])
    
    idx = 2
    jobs = []
    
    for _ in range(j):
        o = int(data[idx])
        idx += 1
        operations = []
        for _ in range(o):
            machine = int(data[idx])
            duration = int(data[idx + 1])
            operations.append((machine, duration))
            idx += 2
        jobs.append(operations)
    
    # We'll use branch and bound with priority queue
    # State: (time, job_positions, machine_end_times)
    # But for simplicity and given the constraints, we can use backtracking with pruning
    
    # Let's try a different approach: since j<=10 and total ops <= 50, we can enumerate all possible schedules
    # But even that might be too much. Let's think of a smarter way.
    
    # Best approach for this constraint: Branch and bound with a good heuristic
    # We'll represent the state as follows:
    # For each job, we track the earliest time it can start its next operation
    # For each machine, we track when it becomes free
    
    # Initial state
    job_op_index = [0] * j  # Which operation each job is on
    machine_end_time = [0] * (m + 1)  # 1-indexed
    job_end_times = [0] * j  # When each job finished its last operation
    
    # Precompute job operation sequences
    job_ops = []
    for job in jobs:
        job_ops.append(job)
    
    # We'll use a recursive backtracking approach with pruning
    # Since the search space isn't too large, this should work
    
    global best_makespan
    best_makespan = float('inf')
    
    def dfs(current_time):
        global best_makespan
        
        # Pruning
        if current_time >= best_makespan:
            return
            
        # Check if all jobs are done
        all_done = True
        for i in range(j):
            if job_op_index[i] < len(job_ops[i]):
                all_done = False
                break
                
        if all_done:
            best_makespan = min(best_makespan, current_time)
            return
            
        # Find all possible next operations that can be scheduled now
        # An operation can be scheduled if:
        # 1. Its job has completed previous operations
        # 2. Its required machine is free at or after job's ready time
        
        candidates = []
        for job_id in range(j):
            if job_op_index[job_id] >= len(job_ops[job_id]):
                continue
                
            op_machine, op_duration = job_ops[job_id][job_op_index[job_id]]
            job_ready_time = job_end_times[job_id]
            machine_free_time = machine_end_time[op_machine]
            
            start_time = max(job_ready_time, machine_free_time)
            end_time = start_time + op_duration
            
            candidates.append((end_time, start_time, job_id, op_machine, op_duration))
        
        # Sort by end_time to try earlier-finishing operations first (heuristic)
        candidates.sort()
        
        for end_time, start_time, job_id, op_machine, op_duration in candidates:
            # Pruning
            if end_time >= best_makespan:
                break
                
            # Apply the operation
            original_job_op_idx = job_op_index[job_id]
            original_machine_end = machine_end_time[op_machine]
            original_job_end = job_end_times[job_id]
            
            job_op_index[job_id] += 1
            machine_end_time[op_machine] = end_time
            job_end_times[job_id] = end_time
            
            # Recurse
            dfs(end_time)
            
            # Backtrack
            job_op_index[job_id] = original_job_op_idx
            machine_end_time[op_machine] = original_machine_end
            job_end_times[job_id] = original_job_end
    
    # Start DFS
    dfs(0)
    print(best_makespan)

solve()
