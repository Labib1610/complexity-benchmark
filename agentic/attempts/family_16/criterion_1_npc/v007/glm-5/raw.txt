import sys
from heapq import heappush, heappop

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    jobs = []
    for _ in range(j):
        o = int(next(it))
        ops = []
        for _ in range(o):
            machine = int(next(it)) - 1
            duration = int(next(it))
            ops.append((machine, duration))
        jobs.append(ops)
    
    # State: (time, machine_free_times tuple, job_progress tuple)
    # job_progress[i] = index of next operation to run for job i
    start_state = (0, tuple([0]*m), tuple([0]*len(jobs)))
    
    # Dijkstra-like search
    # Use a dict to store best machine_free_times for each job_progress
    best = {}
    best[tuple([0]*len(jobs))] = tuple([0]*m)
    
    heap = [(0, tuple([0]*m), tuple([0]*len(jobs)))]
    
    final_time = float('inf')
    
    while heap:
        time, mft, jp = heappop(heap)
        
        if time >= final_time:
            continue
            
        # Check if all jobs are done
        all_done = True
        for i, ops in enumerate(jobs):
            if jp[i] < len(ops):
                all_done = False
                break
        if all_done:
            final_time = time
            continue
        
        # Check if this state is outdated
        if best.get(jp, None) != mft:
            continue
        
        # Find schedulable operations
        schedulable = []
        for i, ops in enumerate(jobs):
            if jp[i] < len(ops):
                machine, duration = ops[jp[i]]
                schedulable.append((i, machine, duration))
        
        # Try all non-empty subsets of schedulable operations
        n = len(schedulable)
        for mask in range(1, 1 << n):
            new_mft = list(mft)
            new_jp = list(jp)
            max_end = time
            
            valid = True
            for k in range(n):
                if mask & (1 << k):
                    i, machine, duration = schedulable[k]
                    start = max(time, mft[machine])
                    end = start + duration
                    if end > final_time:
                        valid = False
                        break
                    new_mft[machine] = end
                    new_jp[i] = jp[i] + 1
                    if end > max_end:
                        max_end = end
            
            if not valid:
                continue
            
            new_mft_t = tuple(new_mft)
            new_jp_t = tuple(new_jp)
            
            old_best = best.get(new_jp_t)
            if old_best is None or new_mft_t < old_best:
                best[new_jp_t] = new_mft_t
                heappush(heap, (max_end, new_mft_t, new_jp_t))
    
    print(final_time)

solve()
