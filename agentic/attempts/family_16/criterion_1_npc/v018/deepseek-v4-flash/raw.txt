We can model this as a scheduling problem where each order must go through its required machines in sequence, and each machine can handle only one order at a time.  

We can simulate the process using a priority queue (min-heap) for each machine, tracking when each machine becomes free.  

Here’s the implementation:  

```python
import sys
import heapq

def main():
    data = sys.stdin.read().strip().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    
    orders = []
    for _ in range(n):
        k = int(data[idx]); idx += 1
        steps = []
        for _ in range(k):
            machine = int(data[idx]) - 1
            time = int(data[idx+1])
            idx += 2
            steps.append((machine, time))
        orders.append(steps)
    
    # Each machine's availability time
    machine_free = [0] * m
    
    # Each order's current step index and start time
    order_step = [0] * n
    order_time = [0] * n
    
    # Min-heap: (time, order_index)
    heap = []
    for i in range(n):
        heapq.heappush(heap, (0, i))
    
    while heap:
        t, oi = heapq.heappop(heap)
        if order_step[oi] >= len(orders[oi]):
            continue
        machine, dur = orders[oi][order_step[oi]]
        start = max(t, machine_free[machine])
        finish = start + dur
        machine_free[machine] = finish
        order_step[oi] += 1
        order_time[oi] = finish
        if order_step[oi] < len(orders[oi]):
            heapq.heappush(heap, (finish, oi))
    
    print(max(order_time))

if __name__ == "__main__":
    main()
```
