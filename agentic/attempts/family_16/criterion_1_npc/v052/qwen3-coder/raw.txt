import sys
from collections import defaultdict

def solve():
    j, m = map(int, sys.stdin.readline().split())
    
    sequences = []
    for _ in range(j):
        line = list(map(int, sys.stdin.readline().split()))
        o = line[0]
        phases = []
        for i in range(o):
            emitter_id = line[1 + 2*i]
            duration = line[1 + 2*i + 1]
            phases.append((emitter_id, duration))
        sequences.append(phases)
    
    # For each emitter, track when it becomes free
    emitter_free_time = [0] * (m + 1)  # 1-indexed
    
    # For each sequence, track when it can start its next phase
    sequence_times = [0] * j
    
    # Process each sequence phase by phase
    for seq_idx, sequence in enumerate(sequences):
        current_time = sequence_times[seq_idx]
        for emitter_id, duration in sequence:
            # This phase can start only after:
            # 1. The previous phase in this sequence finishes (current_time)
            # 2. The emitter becomes free from other sequences (emitter_free_time[emitter_id])
            start_time = max(current_time, emitter_free_time[emitter_id])
            end_time = start_time + duration
            
            # Update when this emitter will be free
            emitter_free_time[emitter_id] = end_time
            
            # Update when this sequence can start its next phase
            current_time = end_time
        
        # Update the final time for this sequence
        sequence_times[seq_idx] = current_time
    
    # The answer is the maximum completion time across all sequences
    print(max(sequence_times))

solve()
