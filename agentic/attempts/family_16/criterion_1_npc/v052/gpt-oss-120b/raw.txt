import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    J = int(next(it))
    M = int(next(it))
    jobs = []
    total_ops = 0
    for _ in range(J):
        o = int(next(it))
        ops = []
        for __ in range(o):
            machine = int(next(it)) - 1
            dur = int(next(it))
            ops.append((machine, dur))
        jobs.append(ops)
        total_ops += o

    # precompute remaining total time per job and per machine for lower bound
    job_remain = [sum(d for _, d in job) for job in jobs]
    machine_remain = [0] * M
    for job in jobs:
        for m, d in job:
            machine_remain[m] += d

    best = [float('inf')]

    # state arrays
    job_pos = [0] * J          # next operation index for each job
    job_ready = [0] * J        # time when current job is ready for its next op
    machine_free = [0] * M     # time when each machine becomes free

    # remaining sums that will be updated during recursion
    job_rem = job_remain[:]
    mach_rem = machine_remain[:]

    def lower_bound(cur_max):
        # bound from jobs
        lb = cur_max
        for j in range(J):
            if job_pos[j] < len(jobs[j]):
                lb = max(lb, job_ready[j] + job_rem[j])
        # bound from machines
        for m in range(M):
            if mach_rem[m]:
                lb = max(lb, machine_free[m] + mach_rem[m])
        return lb

    def dfs(done_ops, cur_max):
        if cur_max >= best[0]:
            return
        if done_ops == total_ops:
            best[0] = cur_max
            return

        # simple heuristic: try jobs in order of earliest possible start
        candidates = []
        for j in range(J):
            p = job_pos[j]
            if p < len(jobs[j]):
                m, d = jobs[j][p]
                start = max(job_ready[j], machine_free[m])
                candidates.append((start, j, m, d))
        # sort by earliest start time
        candidates.sort()
        for start, j, m, d in candidates:
            # schedule this operation
            finish = start + d

            # update state
            prev_job_ready = job_ready[j]
            prev_machine_free = machine_free[m]
            prev_job_pos = job_pos[j]
            prev_job_rem = job_rem[j]
            prev_mach_rem = mach_rem[m]

            job_ready[j] = finish
            machine_free[m] = finish
            job_pos[j] += 1
            job_rem[j] -= d
            mach_rem[m] -= d

            dfs(done_ops + 1, max(cur_max, finish))

            # backtrack
            job_ready[j] = prev_job_ready
            machine_free[m] = prev_machine_free
            job_pos[j] = prev_job_pos
            job_rem[j] = prev_job_rem
            mach_rem[m] = prev_mach_rem

            # prune if best already reached
            if best[0] <= cur_max:
                return

    dfs(0, 0)
    print(best[0])

if __name__ == "__main__":
    solve()
