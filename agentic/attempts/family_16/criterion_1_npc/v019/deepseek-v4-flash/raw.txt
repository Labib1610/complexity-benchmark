import sys

sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))

    jobs = []
    for _ in range(j):
        o = int(next(it))
        ops = []
        for _ in range(o):
            mach = int(next(it)) - 1
            dur = int(next(it))
            ops.append((mach, dur))
        jobs.append(ops)

    total_ops = sum(len(ops) for ops in jobs)

    # Greedy heuristic to get an initial upper bound
    def greedy_makespan():
        progress = [0] * j
        finish_job = [0] * j
        machine_free = [0] * m

        while True:
            ready = []
            for i in range(j):
                if progress[i] < len(jobs[i]):
                    mach, dur = jobs[i][progress[i]]
                    start = max(finish_job[i], machine_free[mach])
                    ready.append((start, i, mach, dur))
            if not ready:
                break
            ready.sort()
            start, i, mach, dur = ready[0]
            end = start + dur
            progress[i] += 1
            finish_job[i] = end
            machine_free[mach] = end

        return max(machine_free)

    best = greedy_makespan()

    def lower_bound(progress, finish_job, machine_free):
        lb = 0

        # Bound per job: current finish + remaining durations
        for i in range(j):
            if progress[i] < len(jobs[i]):
                rem = sum(d for _, d in jobs[i][progress[i]:])
                lb = max(lb, finish_job[i] + rem)

        # Bound per machine: current free time + total remaining work on it
        rem_machine = [0] * m
        for i in range(j):
            for k in range(progress[i], len(jobs[i])):
                mach, d = jobs[i][k]
                rem_machine[mach] += d
        for mm in range(m):
            lb = max(lb, machine_free[mm] + rem_machine[mm])

        return lb

    def dfs(progress, finish_job, machine_free, scheduled):
        nonlocal best

        if scheduled == total_ops:
            makespan = max(machine_free)
            if makespan < best:
                best = makespan
            return

        if lower_bound(progress, finish_job, machine_free) >= best:
            return

        ready = []
        for i in range(j):
            if progress[i] < len(jobs[i]):
                mach, dur = jobs[i][progress[i]]
                start = max(finish_job[i], machine_free[mach])
                ready.append((start, i, mach, dur))

        # Try ordering by earliest possible start (non-delay schedules are sufficient)
        ready.sort()

        for start, i, mach, dur in ready:
            end = start + dur

            old_prog = progress[i]
            old_finish = finish_job[i]
            old_free = machine_free[mach]

            progress[i] += 1
            finish_job[i] = end
            machine_free[mach] = end

            dfs(progress, finish_job, machine_free, scheduled + 1)

            progress[i] = old_prog
            finish_job[i] = old_finish
            machine_free[mach] = old_free

    dfs([0] * j, [0] * j, [0] * m, 0)
    print(best)

if __name__ == "__main__":
    solve()
