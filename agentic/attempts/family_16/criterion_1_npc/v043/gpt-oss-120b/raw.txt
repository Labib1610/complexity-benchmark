```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    m = int(next(it))
    jobs = []
    total_ops = 0
    for _ in range(j):
        o = int(next(it))
        ops = []
        for __ in range(o):
            clerk = int(next(it)) - 1
            dur = int(next(it))
            ops.append((clerk, dur))
        jobs.append(ops)
        total_ops += o

    # precompute remaining sums
    rem_job = [sum(d for _, d in job) for job in jobs]
    rem_machine = [0] * m
    for job in jobs:
        for clerk, dur in job:
            rem_machine[clerk] += dur

    job_pos = [0] * j          # next op index to schedule
    job_finish = [0] * j       # finish time of last scheduled op in each job
    machine_time = [0] * m     # when each machine becomes free

    best = [float('inf')]

    # lower bound function
    def lower_bound(cur_mk):
        lb = cur_mk
        for i in range(j):
            if job_pos[i] < len(jobs[i]):
                lb = max(lb, job_finish[i] + rem_job[i])
        for k in range(m):
            if rem_machine[k]:
                lb = max(lb, machine_time[k] + rem_machine[k])
        return lb

    def dfs(scheduled, cur_mk):
        if scheduled == total_ops:
            if cur_mk < best[0]:
                best[0] = cur_mk
            return
        if cur_mk >= best[0]:
            return
        if lower_bound(cur_mk) >= best[0]:
            return

        # collect all ready operations
        ready = []
        for i in range(j):
            p = job_pos[i]
            if p < len(jobs[i]):
                clerk, dur = jobs[i][p]
                ready.append((i, clerk, dur))

        # simple heuristic: try operations with larger duration first
        ready.sort(key=lambda x: -x[2])

        for i, clerk, dur in ready:
            start = max(job_finish[i], machine_time[clerk])
            finish = start + dur

            # apply
            job_pos[i] += 1
            prev_job_finish = job_finish[i]
            job_finish[i] = finish
            prev_machine_time = machine_time[clerk]
            machine_time[clerk] = finish
            rem_job[i] -= dur
            rem_machine[clerk] -= dur

            dfs(scheduled + 1, max(cur_mk, finish))

            # backtrack
            job_pos[i] -= 1
            job_finish[i] = prev_job_finish
            machine_time[clerk] = prev_machine_time
            rem_job[i] += dur
            rem_machine[clerk] += dur

    dfs(0, 0)
    print(best[0])

if __name__ == "__main__":
    solve()
```
