import sys
import threading

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    jobs = []
    index = 1
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        jobs.append((p1, p2))
        index += 2
    
    # Sort jobs using Johnson's rule
    # Split jobs into two groups:
    # Group1: p1 <= p2
    # Group2: p1 > p2
    group1 = []  # jobs with p1 <= p2, sort by p1 ascending
    group2 = []  # jobs with p1 > p2, sort by p2 descending
    
    for i in range(j):
        p1, p2 = jobs[i]
        if p1 <= p2:
            group1.append((p1, p2, i))
        else:
            group2.append((p1, p2, i))
    
    group1.sort(key=lambda x: x[0])  # Sort by p1 ascending
    group2.sort(key=lambda x: x[1], reverse=True)  # Sort by p2 descending
    
    # Combine the groups: group1 first, then group2
    ordered_jobs = group1 + group2
    
    # Simulate the processing
    time_m1 = 0  # Current time on machine 1
    time_m2 = 0  # Current time on machine 2
    
    for p1, p2, idx in ordered_jobs:
        time_m1 += p1
        # Machine 2 can start processing this job only after:
        # 1. This job finishes on machine 1 (time_m1)
        # 2. Machine 2 is free (time_m2)
        # So it starts at max(time_m1, time_m2) and finishes at max(time_m1, time_m2) + p2
        start_m2 = max(time_m1, time_m2)
        time_m2 = start_m2 + p2
    
    print(time_m2)

# Use threading to increase recursion limit if needed
threading.Thread(target=main).start()
