We can solve this using Johnson’s rule for two-machine flow shop scheduling.  
The idea is to split jobs into two groups:  
- Jobs where `p1 <= p2` go first, sorted by `p1` ascending.  
- Jobs where `p1 > p2` go last, sorted by `p2` descending.  

Then simulate the schedule to compute the makespan.

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    j = int(data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        jobs.append((p1, p2))
    
    group1 = []
    group2 = []
    for p1, p2 in jobs:
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    group1.sort(key=lambda x: x[0])
    group2.sort(key=lambda x: x[1], reverse=True)
    
    sequence = group1 + group2
    
    t1 = 0
    t2 = 0
    for p1, p2 in sequence:
        t1 += p1
        if t2 < t1:
            t2 = t1
        t2 += p2
    
    print(t2)

if __name__ == "__main__":
    main()
```
