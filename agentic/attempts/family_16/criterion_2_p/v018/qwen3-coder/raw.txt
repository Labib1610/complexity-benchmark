import sys
from operator import itemgetter

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    jobs = []
    index = 1
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        jobs.append((p1, p2))
        index += 2
    
    # Sort jobs by p1 increasing, then by p2 decreasing
    # This is equivalent to sorting by p1 - p2 ascending
    jobs.sort(key=lambda x: (x[0] - x[1]))
    
    primary_time = 0
    secondary_time = 0
    
    for p1, p2 in jobs:
        primary_time += p1
        secondary_time = max(secondary_time, primary_time) + p2
    
    print(secondary_time)

if __name__ == "__main__":
    main()
