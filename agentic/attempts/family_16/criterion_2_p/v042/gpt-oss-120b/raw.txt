```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    front = []
    back = []
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        if p1 <= p2:
            front.append((p1, p2))
        else:
            back.append((p1, p2))
    front.sort(key=lambda x: x[0])          # increasing wash time
    back.sort(key=lambda x: x[1], reverse=True)  # decreasing bake time
    seq = front + back

    wash_done = 0
    bake_done = 0
    for p1, p2 in seq:
        wash_done += p1
        bake_done = max(bake_done, wash_done) + p2
    sys.stdout.write(str(bake_done))

if __name__ == "__main__":
    solve()
```
