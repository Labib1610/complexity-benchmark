```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    items = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        items.append((p1, p2))
    
    # Johnson's rule: sort by min(p1, p2)
    # Items with p1 <= p2 go first, sorted by p1 ascending
    # Items with p1 > p2 go after, sorted by p2 descending
    group1 = []  # p1 <= p2
    group2 = []  # p1 > p2
    for p1, p2 in items:
        if p1 <= p2:
            group1.append((p1, p2))
        else:
            group2.append((p1, p2))
    
    group1.sort(key=lambda x: x[0])  # sort by p1 ascending
    group2.sort(key=lambda x: x[1])  # sort by p2 ascending, then reverse
    
    order = group1 + group2[::-1]
    
    time_p1 = 0
    time_p2 = 0
    for p1, p2 in order:
        time_p1 += p1
        time_p2 = max(time_p1, time_p2) + p2
    
    print(time_p2)

solve()
```
