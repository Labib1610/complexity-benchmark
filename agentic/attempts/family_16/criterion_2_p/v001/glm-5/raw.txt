```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    requests = []
    sum_a = 0
    for _ in range(j):
        a = int(next(it))
        b = int(next(it))
        requests.append((a, b))
        sum_a += a
    
    # Sort by Johnson's rule:
    # If a < b, sort by a ascending (smaller a first)
    # If a >= b, sort by b descending (larger b first)
    # Stable sort is not required for correctness but Python's sort is stable.
    requests.sort(key=lambda x: (x[0] < x[1], x[0] if x[0] < x[1] else -x[1]))
    
    # Simulate the process to find the makespan
    # time_a tracks when Phase A finishes the current request
    # time_b tracks when Phase B finishes the current request
    time_a = 0
    time_b = 0
    for a, b in requests:
        time_a += a
        # Phase B can start at max(time_a, time_b)
        if time_a > time_b:
            time_b = time_a + b
        else:
            time_b = time_b + b
            
    print(time_b)

if __name__ == '__main__':
    solve()
```
