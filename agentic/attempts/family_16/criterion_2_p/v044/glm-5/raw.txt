```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    j = int(next(it))
    containers = []
    for _ in range(j):
        p1 = int(next(it))
        p2 = int(next(it))
        containers.append((p1, p2))
    
    # Sort by p2 descending (Johnson's rule for minimizing makespan in 2-machine flow shop)
    containers.sort(key=lambda x: -x[1])
    
    time_scanner = 0
    time_loader = 0
    
    for p1, p2 in containers:
        time_scanner += p1
        # Loader starts at max(time_loader, time_scanner) and finishes after p2
        time_loader = max(time_loader, time_scanner) + p2
    
    print(time_loader)

if __name__ == "__main__":
    solve()
```
