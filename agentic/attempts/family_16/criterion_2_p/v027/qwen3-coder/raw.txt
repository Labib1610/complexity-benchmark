import sys
from functools import cmp_to_key

def compare_rituals(a, b):
    # a = (p1, p2, index), b = (p1, p2, index)
    # Compare p1_a + min(p2_a, p1_b) vs p1_b + min(p2_b, p1_a)
    lhs = a[0] + min(a[1], b[0])
    rhs = b[0] + min(b[1], a[0])
    if lhs < rhs:
        return -1
    elif lhs > rhs:
        return 1
    else:
        return 0

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    rituals = []
    index = 1
    for i in range(j):
        p1 = int(data[index])
        p2 = int(data[index+1])
        rituals.append((p1, p2, i))
        index += 2
    
    # Sort using the custom comparator
    rituals.sort(key=cmp_to_key(compare_rituals))
    
    inscriber_time = 0
    channeler_time = 0
    
    for p1, p2, idx in rituals:
        inscriber_time += p1
        channeler_time = max(channeler_time, inscriber_time) + p2
    
    print(channeler_time)

if __name__ == "__main__":
    main()
