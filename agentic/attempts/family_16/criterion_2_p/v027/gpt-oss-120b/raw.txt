```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    jobs = []
    for _ in range(n):
        a = int(next(it))
        b = int(next(it))
        jobs.append((a, b))
    # Johnson's rule
    first = []
    second = []
    for a, b in jobs:
        if a <= b:
            first.append((a, b))
        else:
            second.append((a, b))
    first.sort(key=lambda x: x[0])          # ascending a
    second.sort(key=lambda x: x[1], reverse=True)  # descending b
    ordered = first + second

    t1 = 0  # completion time on machine 1 (Inscriber)
    t2 = 0  # completion time on machine 2 (Channeler)
    for a, b in ordered:
        t1 += a
        if t2 < t1:
            t2 = t1
        t2 += b
    print(t2)

if __name__ == "__main__":
    solve()
```
