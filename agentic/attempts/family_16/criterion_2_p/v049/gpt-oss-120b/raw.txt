```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    jobs = []
    for _ in range(n):
        p1 = int(next(it)); p2 = int(next(it))
        if p1 <= p2:
            jobs.append((0, p1, p2))   # early group, sort by p1 asc
        else:
            jobs.append((1, -p2, p1, p2))  # later group, sort by p2 desc
    # sort according to Johnson's rule
    jobs.sort(key=lambda x: (x[0], x[1]))
    # compute makespan
    t1 = 0  # completion time on machine 1
    t2 = 0  # completion time on machine 2
    for job in jobs:
        if job[0] == 0:
            _, p1, p2 = job
        else:
            _, _, p1, p2 = job
        t1 += p1
        if t2 < t1:
            t2 = t1
        t2 += p2
    sys.stdout.write(str(t2))

if __name__ == "__main__":
    solve()
```
