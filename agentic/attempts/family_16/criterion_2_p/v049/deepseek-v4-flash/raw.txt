We can solve this using the classic **Johnson’s rule** for two-machine flow shop scheduling.  
The idea is to sort the jobs so that:

- Jobs with `p1 <= p2` are placed first, sorted by `p1` ascending.
- Jobs with `p1 > p2` are placed after, sorted by `p2` descending.

Then simulate the processing to get the total completion time.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    jobs = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        jobs.append((p1, p2))
    
    first = []
    second = []
    for p1, p2 in jobs:
        if p1 <= p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    
    first.sort(key=lambda x: x[0])
    second.sort(key=lambda x: -x[1])
    
    ordered = first + second
    
    t1 = 0
    t2 = 0
    for p1, p2 in ordered:
        t1 += p1
        if t2 < t1:
            t2 = t1
        t2 += p2
    
    print(t2)

if __name__ == "__main__":
    main()
```
