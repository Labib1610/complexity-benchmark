```python
import sys
import threading

def main():
    import sys
    j = int(sys.stdin.readline())
    effects = []
    for _ in range(j):
        p1, p2 = map(int, sys.stdin.readline().split())
        effects.append((p1, p2))
    
    # Sort according to Johnson's rule:
    # - If min(p1_a, p2_b) <= min(p1_b, p2_a), then a should come before b
    # Alternatively, we can sort by:
    # - if p1 <= p2: key = (p1, 0)
    # - else: key = (p2, 1)
    # But more precisely, Johnson's rule for F2||Cmax says:
    # Partition jobs into two sets:
    # S1 = {j : p1_j <= p2_j}
    # S2 = {j : p1_j > p2_j}
    # In S1, sort by p1 increasing
    # In S2, sort by p2 decreasing
    
    S1 = [(p1, p2) for p1, p2 in effects if p1 <= p2]
    S2 = [(p1, p2) for p1, p2 in effects if p1 > p2]
    
    S1.sort(key=lambda x: x[0])  # increasing p1
    S2.sort(key=lambda x: x[1], reverse=True)  # decreasing p2
    
    schedule = S1 + S2
    
    # Now simulate the schedule
    solver_time = 0
    tracer_time = 0
    
    for p1, p2 in schedule:
        solver_time += p1
        tracer_time = max(tracer_time, solver_time) + p2
    
    print(tracer_time)

threading.Thread(target=main).start()
```
