import sys
from functools import cmp_to_key

def compare(system1, system2):
    # system = (chart_time, extract_time, index)
    # To minimize total time, we use the Johnson's rule comparison
    # Compare (extract1, chart1) vs (chart2, extract2)
    # If extract1 <= chart2, keep order, else swap
    if system1[1] <= system2[0]:
        return -1
    elif system2[1] <= system1[0]:
        return 1
    else:
        # When neither condition helps, sort by (extract_time - chart_time) descending
        diff1 = system1[1] - system1[0]
        diff2 = system2[1] - system2[0]
        if diff1 > diff2:
            return -1
        elif diff1 < diff2:
            return 1
        else:
            return 0

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    j = int(data[0])
    systems = []
    idx = 1
    for i in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        systems.append((p1, p2))
        idx += 2
    
    # Sort systems based on comparison function
    # Johnson's rule: sort by min(extract_time, next_chart_time)
    # But more precisely, we want to minimize the makespan
    # We'll use a greedy approach with custom sorting
    
    # Create list with indices for stable sorting
    indexed_systems = [(systems[i][0], systems[i][1], i) for i in range(j)]
    
    # Sort using Johnson's rule heuristic
    # Split into two groups:
    # Group 1: chart_time <= extract_time 
    # Group 2: chart_time > extract_time
    group1 = []  # chart_time <= extract_time
    group2 = []  # chart_time > extract_time
    
    for i in range(j):
        chart, extract = systems[i]
        if chart <= extract:
            group1.append((chart, extract))
        else:
            group2.append((chart, extract))
    
    # Sort group1 by chart_time ascending
    group1.sort(key=lambda x: x[0])
    # Sort group2 by extract_time descending
    group2.sort(key=lambda x: x[1], reverse=True)
    
    # Combine the groups
    ordered_systems = group1 + group2
    
    # Calculate the total time
    surveyor_end_time = 0
    harvester_end_time = 0
    
    for chart_time, extract_time in ordered_systems:
        surveyor_end_time += chart_time
        harvester_end_time = max(harvester_end_time, surveyor_end_time) + extract_time
    
    print(harvester_end_time)

if __name__ == "__main__":
    main()
