```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    j = int(data[0])
    A = []
    B = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx]); p2 = int(data[idx+1])
        idx += 2
        if p1 <= p2:
            A.append((p1, p2))
        else:
            B.append((p1, p2))
    A.sort(key=lambda x: x[0])
    B.sort(key=lambda x: x[1], reverse=True)
    order = A + B
    t = 0
    finish = 0
    for p1, p2 in order:
        t += p1
        if t > finish:
            finish = t
        finish += p2
    print(finish)

solve()
```
