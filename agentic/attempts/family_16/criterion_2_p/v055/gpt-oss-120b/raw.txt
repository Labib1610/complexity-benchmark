```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    group_a = []  # p1 <= p2
    group_b = []  # p1 > p2
    for _ in range(n):
        p1 = int(next(it))
        p2 = int(next(it))
        if p1 <= p2:
            group_a.append((p1, p2))
        else:
            group_b.append((p1, p2))
    group_a.sort(key=lambda x: x[0])          # ascending p1
    group_b.sort(key=lambda x: -x[1])         # descending p2
    order = group_a + group_b

    t1 = 0  # completion time on oven
    t2 = 0  # completion time on frosting applicator
    for p1, p2 in order:
        t1 += p1
        if t2 < t1:
            t2 = t1
        t2 += p2
    sys.stdout.write(str(t2))

if __name__ == "__main__":
    solve()
```
