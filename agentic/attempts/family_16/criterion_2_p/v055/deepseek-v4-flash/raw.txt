We can solve this by sorting the cakes using a greedy rule:  
A cake with a longer frosting time should go earlier if it has a shorter oven time, but the standard optimal rule for two-machine flow shop scheduling is:

- Sort by **decreasing** frosting time if oven time ≤ frosting time (Johnson’s rule),  
  otherwise sort by **increasing** oven time.

Here’s the implementation:

```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    j = int(data[0])
    cakes = []
    idx = 1
    for _ in range(j):
        p1 = int(data[idx])
        p2 = int(data[idx+1])
        idx += 2
        cakes.append((p1, p2))
    
    # Johnson's rule for two-machine flow shop
    first = []
    second = []
    for p1, p2 in cakes:
        if p1 <= p2:
            first.append((p1, p2))
        else:
            second.append((p1, p2))
    
    # Sort first group by increasing oven time
    first.sort(key=lambda x: x[0])
    # Sort second group by decreasing frosting time
    second.sort(key=lambda x: -x[1])
    
    sequence = first + second
    
    oven_end = 0
    frost_end = 0
    for p1, p2 in sequence:
        oven_end += p1
        if frost_end < oven_end:
            frost_end = oven_end
        frost_end += p2
    
    print(frost_end)

if __name__ == "__main__":
    solve()
```
