import sys
from array import array

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

n, m, B = data[0], data[1], data[2]
adj = [[] for _ in range(n)]
idx = 3
for _ in range(m):
    u = data[idx] - 1
    v = data[idx + 1] - 1
    idx += 2
    adj[u].append(v)
    adj[v].append(u)

if n == 1:
    print("YES")
    print(1)
    sys.exit()

deg = [len(x) for x in adj]


def arrangement_cost(order):
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i
    total = 0
    for u in range(n):
        pu = pos[u]
        for v in adj[u]:
            if u < v:
                total += abs(pu - pos[v])
    return total


def improve_order(order, max_passes=30):
    order = order[:]
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    for _ in range(max_passes):
        changed = False
        i = 0
        while i + 1 < n:
            x = order[i]
            y = order[i + 1]
            delta = 0

            for z in adj[x]:
                if z == y:
                    continue
                if pos[z] < i:
                    delta += 1
                else:
                    delta -= 1

            for z in adj[y]:
                if z == x:
                    continue
                if pos[z] < i:
                    delta -= 1
                else:
                    delta += 1

            if delta < 0:
                order[i], order[i + 1] = y, x
                pos[x], pos[y] = i + 1, i
                changed = True
                if i > 0:
                    i -= 1
                    continue
            i += 1

        if not changed:
            break

    return order


def tree_centers():
    if n <= 2:
        return list(range(n))
    d = deg[:]
    leaves = [i for i in range(n) if d[i] <= 1]
    remaining = n
    while remaining > 2:
        remaining -= len(leaves)
        new_leaves = []
        for u in leaves:
            for v in adj[u]:
                if d[v] > 0:
                    d[v] -= 1
                    if d[v] == 1:
                        new_leaves.append(v)
            d[u] = 0
        leaves = new_leaves
    return leaves


def dfs_order(root, reverse_neighbors=False, post=False):
    parent = [-1] * n
    stack = [(root, 0)]
    result = []

    while stack:
        u, state = stack.pop()
        if state == 0:
            if not post:
                result.append(u)
            else:
                stack.append((u, 1))

            children = [v for v in adj[u] if v != parent[u]]
            children.sort(key=lambda x: (deg[x], x), reverse=reverse_neighbors)
            for v in reversed(children):
                parent[v] = u
                stack.append((v, 0))
        else:
            result.append(u)

    return result


def bfs_order(root):
    parent = [-1] * n
    parent[root] = root
    q = [root]
    for u in q:
        children = [v for v in adj[u] if parent[v] == -1]
        children.sort(key=lambda x: (deg[x], x))
        for v in children:
            parent[v] = u
            q.append(v)
    return q


def centroid_order():
    blocked = [False] * n

    def build(start):
        nodes = []
        parent = {start: -1}
        stack = [start]

        while stack:
            u = stack.pop()
            nodes.append(u)
            for v in adj[u]:
                if not blocked[v] and v != parent[u]:
                    parent[v] = u
                    stack.append(v)

        size = {}
        for u in reversed(nodes):
            s = 1
            for v in adj[u]:
                if not blocked[v] and parent.get(v, -2) == u:
                    s += size[v]
            size[u] = s

        total = len(nodes)
        centroid = nodes[0]
        best = total + 1
        for u in nodes:
            largest = total - size[u]
            for v in adj[u]:
                if not blocked[v] and parent.get(v, -2) == u:
                    if size[v] > largest:
                        largest = size[v]
            if largest < best:
                best = largest
                centroid = u

        blocked[centroid] = True
        parts = []
        for v in adj[centroid]:
            if blocked[v]:
                continue
            child_order = build(v)
            p = child_order.index(v)
            parts.append([len(child_order), v, child_order, p])

        parts.sort(key=lambda x: (x[0], x[1]))

        left_inner = []
        right_inner = []
        left_load = 0
        right_load = 0

        for s, attach, seq, p in parts:
            if left_load <= right_load:
                if p < s - 1 - p:
                    seq.reverse()
                left_inner.append(seq)
                left_load += s
            else:
                if p > s - 1 - p:
                    seq.reverse()
                right_inner.append(seq)
                right_load += s

        result = []
        for seq in reversed(left_inner):
            result.extend(seq)
        result.append(centroid)
        for seq in right_inner:
            result.extend(seq)
        return result

    return build(0)


candidates = []
centers = tree_centers()

for c in centers:
    candidates.append(dfs_order(c, False, False))
    candidates.append(dfs_order(c, True, False))
    candidates.append(dfs_order(c, False, True))
    candidates.append(dfs_order(c, True, True))
    candidates.append(bfs_order(c))

candidates.append(centroid_order())

best_order = None
best_cost = 10**30

for cand in candidates:
    cand = improve_order(cand)
    value = arrangement_cost(cand)
    if value < best_cost:
        best_cost = value
        best_order = cand

if best_cost <= B:
    positions = [0] * n
    for i, v in enumerate(best_order, 1):
        positions[v] = i
    print("YES")
    print(*positions)
    sys.exit()

lower_sum = 0
lower_max = 0
for d in deg:
    local = ((d + 1) * (d + 1)) // 4
    lower_sum += local
    if local > lower_max:
        lower_max = local

lower_bound = max(n - 1, lower_max, (lower_sum + 1) // 2)
if lower_bound > B:
    print("NO")
    sys.exit()

# Exact subset dynamic programming for moderate n.
if n <= 21:
    total_masks = 1 << n
    INF = (1 << 60)
    crossing = array('i', [0]) * total_masks
    dp = array('q', [INF]) * total_masks
    dp[0] = 0

    adjmask = [0] * n
    for u in range(n):
        mask = 0
        for v in adj[u]:
            mask |= 1 << v
        adjmask[u] = mask

    for mask in range(1, total_masks):
        bit = mask & -mask
        v = bit.bit_length() - 1
        prev = mask ^ bit
        crossing[mask] = (
            crossing[prev] + deg[v] -
            2 * ((adjmask[v] & prev).bit_count())
        )

        best = INF
        bits = mask
        while bits:
            b = bits & -bits
            val = dp[mask ^ b]
            if val < best:
                best = val
            bits ^= b
        dp[mask] = best + crossing[mask]

    full = total_masks - 1
    optimum = dp[full]

    if optimum > B:
        print("NO")
        sys.exit()

    reverse_order = []
    mask = full
    while mask:
        target = dp[mask] - crossing[mask]
        bits = mask
        chosen = -1
        while bits:
            b = bits & -bits
            if dp[mask ^ b] == target:
                chosen = b.bit_length() - 1
                break
            bits ^= b
        reverse_order.append(chosen)
        mask ^= 1 << chosen

    order = reverse_order[::-1]
    positions = [0] * n
    for i, v in enumerate(order, 1):
        positions[v] = i
    print("YES")
    print(*positions)
    sys.exit()

# Exact branch-and-bound decision search for larger ambiguous instances.
adjmask = [0] * n
for u in range(n):
    mask = 0
    for v in adj[u]:
        mask |= 1 << v
    adjmask[u] = mask

full_mask = (1 << n) - 1
rank = [0] * n
for i, v in enumerate(best_order):
    rank[v] = i
vertex_order = sorted(range(n), key=lambda x: rank[x])

leaf_parent = [-1] * n
for u in range(n):
    if deg[u] == 1:
        leaf_parent[u] = adj[u][0]

memo = {}
path = []
answer = None


def search(mask, cut, spent):
    global answer

    used = mask.bit_count()
    remaining = n - used

    if remaining == 0:
        answer = path[:]
        return True

    if spent + remaining - 1 > B:
        return False

    old = memo.get(mask)
    if old is not None and old <= spent:
        return False
    memo[mask] = spent

    seen_leaf_parents = set()

    for v in vertex_order:
        bit = 1 << v
        if mask & bit:
            continue

        if leaf_parent[v] != -1:
            p = leaf_parent[v]
            if p in seen_leaf_parents:
                continue
            seen_leaf_parents.add(p)

        inside = (adjmask[v] & mask).bit_count()
        new_cut = cut + deg[v] - 2 * inside
        new_spent = spent + new_cut
        new_remaining = remaining - 1

        if new_spent > B:
            continue
        if new_spent + max(0, new_remaining - 1) > B:
            continue

        path.append(v)
        if search(mask | bit, new_cut, new_spent):
            return True
        path.pop()

    return False


if search(0, 0, 0):
    positions = [0] * n
    for i, v in enumerate(answer, 1):
        positions[v] = i
    print("YES")
    print(*positions)
else:
    print("NO")
