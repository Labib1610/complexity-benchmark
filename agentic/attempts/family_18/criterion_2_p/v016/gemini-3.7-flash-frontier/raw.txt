```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    if n == 1:
        if B >= 0:
            print("YES")
            print("1")
        else:
            print("NO")
        return

    # Find subtree sizes to find centroids
    sz = [0] * (n + 1)
    def get_sz(u, p):
        s = 1
        for v in adj[u]:
            if v != p:
                s += get_sz(v, u)
        sz[u] = s
        return s

    get_sz(1, 0)

    centroids = []
    def get_centroids(u, p):
        is_centroid = True
        for v in adj[u]:
            if v != p:
                if sz[v] > n // 2:
                    is_centroid = False
                get_centroids(v, u)
        if is_centroid and (n - sz[u]) <= n // 2:
            centroids.append(u)

    get_centroids(1, 0)

    def layout_from_root(root):
        # Compute sizes rooted at root
        sub_sz = [0] * (n + 1)
        def calc_sz(u, p):
            s = 1
            for v in adj[u]:
                if v != p:
                    s += calc_sz(v, u)
            sub_sz[u] = s
            return s
        calc_sz(root, 0)

        # Build order
        def build(u, p, root_at_left):
            children = [v for v in adj[u] if v != p]
            children.sort(key=lambda v: sub_sz[v], reverse=True)

            if u == root:
                # Root can place children on both left and right
                left_children = []
                right_children = []
                for i, c in enumerate(children):
                    if i % 2 == 0:
                        left_children.append(c)
                    else:
                        right_children.append(c)

                # left_children: largest closest to root, so reverse order of placement
                # left side goes from left to right, root of child should be on right
                res = []
                for c in reversed(left_children):
                    res.extend(build(c, u, root_at_left=False))
                res.append(u)
                for c in right_children:
                    res.extend(build(c, u, root_at_left=True))
                return res
            else:
                if root_at_left:
                    # u is at the very left, all its children are to the right
                    res = [u]
                    for c in children:
                        res.extend(build(c, u, root_at_left=True))
                    return res
                else:
                    # u is at the very right, all its children are to the left
                    res = []
                    for c in children:
                        res.extend(build(c, u, root_at_left=False))
                    res.append(u)
                    return res

        return build(root, 0, True)

    def evaluate(order):
        pos = [0] * (n + 1)
        for p, u in enumerate(order, 1):
            pos[u] = p
        cost = 0
        for u in range(1, n + 1):
            for v in adj[u]:
                if u < v:
                    cost += abs(pos[u] - pos[v])
        return cost, pos

    best_cost = float('inf')
    best_pos = None

    candidates = list(centroids)
    # Add a few more candidates if tree is small or centroid count is small
    if len(candidates) < 10 and n <= 2000:
        candidates = list(range(1, n + 1))
    elif len(candidates) < 5:
        # Add neighbors of centroids
        for c in list(centroids):
            for v in adj[c]:
                if v not in candidates:
                    candidates.append(v)

    for r in candidates:
        order = layout_from_root(r)
        cost, pos = evaluate(order)
        if cost < best_cost:
            best_cost = cost
            best_pos = pos
            if best_cost <= B:
                break

    if best_cost <= B:
        print("YES")
        print(*(best_pos[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
