import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # BFS to find bipartition
    color = [0] * (n + 1)
    color[1] = 1
    queue = deque([1])
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if color[v] == 0:
                color[v] = 3 - color[u]
                queue.append(v)
    
    A = [i for i in range(1, n + 1) if color[i] == 1]
    C = [i for i in range(1, n + 1) if color[i] == 2]
    
    # Assign slots: A gets 1..|A|, C gets |A|+1..n
    pos = [0] * (n + 1)
    for idx, u in enumerate(A):
        pos[u] = idx + 1
    for idx, u in enumerate(C):
        pos[u] = len(A) + idx + 1
    
    # Compute total cable length
    total = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if v > u:
                total += abs(pos[u] - pos[v])
    
    if total <= B:
        print("YES")
        print(' '.join(str(pos[i]) for i in range(1, n + 1)))
    else:
        print("NO")

solve()
