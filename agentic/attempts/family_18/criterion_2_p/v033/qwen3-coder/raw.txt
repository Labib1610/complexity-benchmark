import sys
from collections import defaultdict, deque

def solve():
    n, m, B = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Find center of the tree using BFS
    def find_center():
        # First BFS from node 1 to find the farthest node
        visited = [False] * (n + 1)
        queue = deque()
        queue.append((1, 0))
        visited[1] = True
        farthest_node = 1
        max_dist = 0
        
        while queue:
            node, dist = queue.popleft()
            if dist > max_dist:
                max_dist = dist
                farthest_node = node
                
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    queue.append((neighbor, dist + 1))
        
        # Second BFS from farthest_node to find diameter and center
        visited = [False] * (n + 1)
        parent = [-1] * (n + 1)
        queue = deque()
        queue.append((farthest_node, 0))
        visited[farthest_node] = True
        farthest_node_2 = farthest_node
        max_dist = 0
        distances = [0] * (n + 1)
        
        while queue:
            node, dist = queue.popleft()
            distances[node] = dist
            if dist > max_dist:
                max_dist = dist
                farthest_node_2 = node
                
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    parent[neighbor] = node
                    queue.append((neighbor, dist + 1))
        
        # Find the path between the two farthest nodes
        path = []
        current = farthest_node_2
        while current != -1:
            path.append(current)
            current = parent[current]
        
        # The center is the middle of the path
        center_idx = len(path) // 2
        return path[center_idx]
    
    # BFS to get tree structure from center
    def bfs_tree(root):
        visited = [False] * (n + 1)
        queue = deque([(root, 0)])
        visited[root] = True
        levels = defaultdict(list)
        level_of_node = {}
        level_of_node[root] = 0
        levels[0].append(root)
        
        while queue:
            node, level = queue.popleft()
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    level_of_node[neighbor] = level + 1
                    levels[level + 1].append(neighbor)
                    queue.append((neighbor, level + 1))
        
        return levels, level_of_node
    
    center = find_center()
    levels, level_of_node = bfs_tree(center)
    
    # Calculate the cost of placing nodes in a line according to BFS levels
    # We'll place nodes level by level, trying to minimize the total distance
    
    # Create a placement array
    placement = [0] * (n + 1)
    
    # Place the center at position (n+1)//2 to balance left and right
    positions = list(range(1, n + 1))
    
    # Assign positions level by level
    pos_index = 0
    max_level = max(levels.keys()) if levels else 0
    
    # For each level, assign positions consecutively
    current_pos = 1
    queue = deque([center])
    visited = [False] * (n + 1)
    visited[center] = True
    placement[center] = current_pos
    current_pos += 1
    
    # BFS traversal to assign positions
    level_positions = {0: [center]}
    level_num = 0
    
    while any(not visited[v] for u in adj for v in adj[u]):
        next_queue = deque()
        level_num += 1
        level_positions[level_num] = []
        
        # Process current level
        temp_queue = deque()
        if queue:
            for node in queue:
                for neighbor in sorted(adj[node]):  # Sort for deterministic behavior
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        temp_queue.append(neighbor)
        
        # Assign positions to this level
        level_nodes = list(temp_queue)
        for node in level_nodes:
            if current_pos <= n:
                placement[node] = current_pos
                current_pos += 1
                level_positions[level_num].append(node)
                next_queue.append(node)
        
        queue = next_queue
        if not queue:
            # Find unvisited nodes and continue BFS
            for i in range(1, n + 1):
                if not visited[i]:
                    queue.append(i)
                    visited[i] = True
                    placement[i] = current_pos
                    current_pos += 1
                    break
    
    # If we didn't fill all positions, do a final assignment
    for i in range(1, n + 1):
        if placement[i] == 0:
            placement[i] = current_pos
            current_pos += 1
    
    # Calculate total cost
    total_cost = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:  # Count each edge once
                total_cost += abs(placement[u] - placement[v])
    
    # If within budget, output the solution
    if total_cost <= B:
        print("YES")
        print(*placement[1:])
        return
    
    # Try a different approach - linear arrangement
    # Place nodes in order of a DFS traversal
    visited = [False] * (n + 1)
    order = []
    
    def dfs(node):
        visited[node] = True
        order.append(node)
        for neighbor in sorted(adj[node]):
            if not visited[neighbor]:
                dfs(neighbor)
    
    dfs(1)
    
    # Assign positions 1 to n to the DFS order
    placement_dfs = [0] * (n + 1)
    for i, node in enumerate(order):
        placement_dfs[node] = i + 1
    
    # Calculate cost for DFS arrangement
    total_cost_dfs = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:
                total_cost_dfs += abs(placement_dfs[u] - placement_dfs[v])
    
    if total_cost_dfs <= B:
        print("YES")
        print(*placement_dfs[1:])
        return
    
    # Last resort: try simple 1,2,3,...,n arrangement
    placement_simple = list(range(1, n + 1))
    total_cost_simple = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:
                total_cost_simple += abs(placement_simple[u-1] - placement_simple[v-1])
    
    if total_cost_simple <= B:
        print("YES")
        print(*placement_simple)
        return
    
    print("NO")

solve()
