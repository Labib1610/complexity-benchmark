import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # BFS to find tree centers
    deg = [0] * (n + 1)
    leaves = []
    for i in range(1, n + 1):
        deg[i] = len(adj[i])
        if deg[i] <= 1:
            leaves.append(i)
    
    temp_deg = deg[:]
    temp_adj = [lst[:] for lst in adj]
    remaining = n
    q = list(leaves)
    
    while remaining > 2:
        nq = []
        for u in q:
            for v in temp_adj[u]:
                temp_deg[v] -= 1
                if temp_deg[v] == 1:
                    nq.append(v)
            remaining -= 1
        q = nq
    
    centers = q
    
    # BFS from centers to get depths
    depth = [-1] * (n + 1)
    q2 = []
    for c in centers:
        depth[c] = 0
        q2.append(c)
    
    for u in q2:
        for v in adj[u]:
            if depth[v] == -1:
                depth[v] = depth[u] + 1
                q2.append(v)
    
    # Sort nodes by depth descending
    nodes = list(range(1, n + 1))
    nodes.sort(key=lambda x: -depth[x])
    
    # Greedy assignment
    ans = [0] * (n + 1)
    L = 1
    R = n
    for u in nodes:
        if depth[u] % 2 == 0:
            ans[u] = L
            L += 1
        else:
            ans[u] = R
            R -= 1
    
    # Calculate total length
    total = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if v > u:
                total += abs(ans[u] - ans[v])
    total //= 2
    
    if total <= B:
        print("YES")
        print(' '.join(map(str, ans[1:])))
    else:
        print("NO")

solve()
