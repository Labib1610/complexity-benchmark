```python
import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    if n == 1:
        if B >= 0:
            print("YES")
            print("1")
        else:
            print("NO")
        return

    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # Find centroids of the tree
    sz = [0] * (n + 1)
    def get_sizes(u, p):
        s = 1
        for v in adj[u]:
            if v != p:
                s += get_sizes(v, u)
        sz[u] = s
        return s

    get_sizes(1, 0)

    centroids = []
    def find_centroids(u, p):
        is_centroid = True
        for v in adj[u]:
            if v != p:
                if sz[v] > n // 2:
                    is_centroid = False
                find_centroids(v, u)
        if is_centroid and (n - sz[u]) <= n // 2:
            centroids.append(u)

    find_centroids(1, 0)

    def compute_layout(root):
        sub_sz = [0] * (n + 1)
        def calc_sz(u, p):
            s = 1
            for v in adj[u]:
                if v != p:
                    s += calc_sz(v, u)
            sub_sz[u] = s
            return s

        calc_sz(root, 0)

        def layout_subtree(u, p, attach_side):
            children = [v for v in adj[u] if v != p]
            children.sort(key=lambda x: sub_sz[x])

            if attach_side == 'RIGHT':
                # u is rightmost, children to the left
                res = []
                for c in children:
                    res.extend(layout_subtree(c, u, 'RIGHT'))
                res.append(u)
                return res
            else:
                # u is leftmost, children to the right
                res = [u]
                for c in reversed(children):
                    res.extend(layout_subtree(c, u, 'LEFT'))
                return res

        # Partition children of root
        root_children = list(adj[root])
        root_children.sort(key=lambda x: sub_sz[x], reverse=True)

        L = root_children[0::2]
        R = root_children[1::2]

        order = []
        for c in reversed(L):
            order.extend(layout_subtree(c, root, 'RIGHT'))
        order.append(root)
        for c in R:
            order.extend(layout_subtree(c, root, 'LEFT'))

        return order

    def evaluate(order):
        pos = [0] * (n + 1)
        for i, u in enumerate(order):
            pos[u] = i + 1
        cost = 0
        for u, v in edges:
            cost += abs(pos[u] - pos[v])
        return cost, pos

    # Candidates to try: centroids first, then if n is small, all nodes
    candidates = list(centroids)
    if n <= 1000:
        seen = set(candidates)
        for i in range(1, n + 1):
            if i not in seen:
                candidates.append(i)

    for r in candidates:
        order = compute_layout(r)
        cost, pos = evaluate(order)
        if cost <= B:
            print("YES")
            print(*(pos[1:]))
            return

    print("NO")

if __name__ == '__main__':
    solve()
```
