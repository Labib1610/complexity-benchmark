```python
import sys
import itertools

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))

    # simple backtracking with pruning, works for small n (≈10)
    pos = [-1] * n          # position of each node (0‑based)
    used = [False] * n      # which positions are taken
    edge_adj = [[] for _ in range(n)]
    for u, v in edges:
        edge_adj[u].append(v)
        edge_adj[v].append(u)

    # pre‑compute for each edge an index for quick access
    edge_idx = {}
    for idx, (u, v) in enumerate(edges):
        edge_idx[(u, v)] = edge_idx[(v, u)] = idx

    # current sum of lengths for edges whose both ends are placed
    cur_sum = 0
    # number of edges not yet fully placed
    remaining_edges = m

    # order nodes by degree descending to improve pruning
    order = sorted(range(n), key=lambda x: -len(edge_adj[x]))

    sys.setrecursionlimit(10000)

    def dfs(depth: int) -> bool:
        nonlocal cur_sum, remaining_edges
        if depth == n:
            return cur_sum <= B

        node = order[depth]
        for p in range(n):
            if used[p]:
                continue
            # tentative addition to cur_sum
            add = 0
            newly_done = 0
            for nb in edge_adj[node]:
                if pos[nb] != -1:          # neighbor already placed
                    add += abs(p - pos[nb])
                    newly_done += 1
            # prune
            min_possible = cur_sum + add + (remaining_edges - newly_done) * 1
            if min_possible > B:
                continue

            # commit
            pos[node] = p
            used[p] = True
            cur_sum += add
            remaining_edges -= newly_done

            if dfs(depth + 1):
                return True

            # backtrack
            cur_sum -= add
            remaining_edges += newly_done
            used[p] = False
            pos[node] = -1
        return False

    if dfs(0):
        # convert positions to 1‑based permutation: output position of each unit
        ans = [p + 1 for p in pos]
        print("YES")
        print(' '.join(map(str, ans)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
