import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    parent = [0]*(n+1)
    children = [[] for _ in range(n+1)]
    root = 1
    for i in range(1, n+1):
        if adj[i]:
            root = i
            break
    
    stack = [root]
    parent[root] = 0
    order = [root]
    while stack:
        u = stack.pop()
        for v in adj[u]:
            if v != parent[u]:
                parent[v] = u
                children[u].append(v)
                stack.append(v)
                order.append(v)
    
    size = [1]*(n+1)
    for u in reversed(order):
        for v in children[u]:
            size[u] += size[v]
    
    pos = [0]*(n+1)
    time = 1
    stack = [(root, 0)]
    while stack:
        u, idx = stack.pop()
        if idx == 0:
            pos[u] = time
            time += 1
        if idx < len(children[u]):
            stack.append((u, idx+1))
            stack.append((children[u][idx], 0))
    
    total = 0
    for u in range(2, n+1):
        total += abs(pos[u] - pos[parent[u]])
    
    if total <= B:
        print("YES")
        print(' '.join(map(str, pos[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
