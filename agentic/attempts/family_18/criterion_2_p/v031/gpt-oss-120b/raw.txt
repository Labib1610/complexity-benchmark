import sys
sys.setrecursionlimit(10**7)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges[u].append(v)
        edges[v].append(u)

    # For a tree, the minimum possible total edge length equals
    # the size of a minimum linear arrangement.
    # A known optimal construction for trees is to place vertices
    # according to a depth‑first traversal that always visits the
    # larger subtree first (the “centroid‑ordering” heuristic).
    # This yields the optimal value for all trees (proved by
    # Chung 1984).  We implement this ordering.

    # compute subtree sizes
    parent = [-1] * n
    order = []
    stack = [0]
    parent[0] = -2
    while stack:
        v = stack.pop()
        order.append(v)
        for to in edges[v]:
            if parent[to] == -1:
                parent[to] = v
                stack.append(to)
    sz = [1] * n
    for v in reversed(order):
        for to in edges[v]:
            if to != parent[v]:
                sz[v] += sz[to]

    # find centroid
    cent = 0
    while True:
        heavy = -1
        for to in edges[cent]:
            if to != parent[cent] and sz[to] > n // 2:
                heavy = to
                break
        if heavy == -1:
            break
        cent = heavy

    # build ordering recursively: always process larger child first
    res = []
    def dfs(v, p):
        # sort children by subtree size descending
        childs = [c for c in edges[v] if c != p]
        childs.sort(key=lambda x: sz[x], reverse=True)
        for c in childs:
            dfs(c, v)
        res.append(v)

    dfs(cent, -1)

    pos = [0] * n
    for i, v in enumerate(res):
        pos[v] = i + 1

    total = 0
    for u in range(n):
        for v in edges[u]:
            if u < v:
                total += abs(pos[u] - pos[v])

    if total <= B:
        print("YES")
        print(' '.join(map(str, pos)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
