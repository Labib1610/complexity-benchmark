```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # BFS to get parent pointers and BFS order
    parent = [0] * (n + 1)
    depth = [0] * (n + 1)
    order = []
    visited = [False] * (n + 1)
    root = 1
    visited[root] = True
    q = deque([root])
    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                depth[v] = depth[u] + 1
                q.append(v)
    
    # Build children list
    children = [[] for _ in range(n + 1)]
    for v in range(2, n + 1):
        children[parent[v]].append(v)
    
    # Compute subtree sizes
    size = [1] * (n + 1)
    for u in reversed(order):
        for v in children[u]:
            size[u] += size[v]
    
    # Heavy-light decomposition
    top = [0] * (n + 1)
    top[root] = root
    for u in order:
        max_size = 0
        heavy = -1
        for v in children[u]:
            if size[v] > max_size:
                max_size = size[v]
                heavy = v
        for v in children[u]:
            if v == heavy:
                top[v] = top[u]
            else:
                top[v] = v
    
    # Position in DFS order (heavy path first)
    pos = [0] * (n + 1)
    dfs_order = []
    stack = [(root, iter(children[root]))]
    pos[root] = 0
    dfs_order.append(root)
    while stack:
        u, it_children = stack[-1]
        found = False
        for v in it_children:
            pos[v] = len(dfs_order)
            dfs_order.append(v)
            stack.append((v, iter(children[v])))
            found = True
            break
        if not found:
            stack.pop()
    
    # Build segment tree for range add, range sum
    size_seg = 1
    while size_seg < n:
        size_seg *= 2
    seg = [0] * (2 * size_seg)
    lazy = [0] * (2 * size_seg)
    
    def apply_lazy(node, l, r, val):
        seg[node] += val * (r - l + 1)
        lazy[node] += val
    
    def push_down(node, l, r):
        if lazy[node] != 0 and l < r:
            mid = (l + r) // 2
            apply_lazy(2 * node, l, mid, lazy[node])
            apply_lazy(2 * node + 1, mid + 1, r, lazy[node])
            lazy[node] = 0
    
    def update_range(node, l, r, ql, qr, val):
        if qr < l or r < ql:
            return
        if ql <= l and r <= qr:
            apply_lazy(node, l, r, val)
            return
        push_down(node, l, r)
        mid = (l + r) // 2
        update_range(2 * node, l, mid, ql, qr, val)
        update_range(2 * node + 1, mid + 1, r, ql, qr, val)
        seg[node] = seg[2 * node] + seg[2 * node + 1]
    
    def query_sum(node, l, r, ql, qr):
        if qr < l or r < ql:
            return 0
        if ql <= l and r <= qr:
            return seg[node]
        push_down(node, l, r)
        mid = (l + r) // 2
        return query_sum(2 * node, l, mid, ql, qr) + query_sum(2 * node + 1, mid + 1, r, ql, qr)
    
    def update_path(u, v, val):
        while top[u] != top[v]:
            if depth[top[u]] < depth[top[v]]:
                u, v = v, u
            update_range(1, 0, size_seg - 1, pos[top[u]], pos[u], val)
            u = parent[top[u]]
        if depth[u] > depth[v]:
            u, v = v, u
        if u != v:
            update_range(1, 0, size_seg - 1, pos[u] + 1, pos[v], val)
    
    def query_path(u, v):
        res = 0
        while top[u] != top[v]:
            if depth[top[u]] < depth[top[v]]:
                u, v = v, u
            res += query_sum(1, 0, size_seg - 1, pos[top[u]], pos[u])
            u = parent[top[u]]
        if depth[u] > depth[v]:
            u, v = v, u
        if u != v:
            res += query_sum(1, 0, size_seg - 1, pos[u] + 1, pos[v])
        return res
    
    # Process edges in reverse BFS order
    assign = [0] * (n + 1)
    assign[root] = 1
    
    for u in reversed(order):
        if u == root:
            continue
        p = parent[u]
        # Try assign[u] = assign[p] + size[u]
        update_path(u, root, 1)
        cost = query_path(p, root)
        if cost <= B:
            assign[u] = assign[p] + size[u]
        else:
            update_path(u, root, -1)
            assign[u] = assign[p] + size[u] - 1
    
    # Verify total cost
    total = 0
    for u, v in edges:
        total += abs(assign[u] - assign[v])
    
    if total <= B:
        print("YES")
        print(' '.join(str(assign[i]) for i in range(1, n + 1)))
    else:
        print("NO")

solve()
```
