```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    def try_order(order):
        pos = [0] * (n + 1)
        for idx, node in enumerate(order, 1):
            pos[node] = idx
        total = 0
        for u in range(1, n + 1):
            for v in adj[u]:
                if u < v:
                    total += abs(pos[u] - pos[v])
        return total, pos

    # BFS order from node 1
    bfs = []
    q = deque([1])
    seen = [False] * (n + 1)
    seen[1] = True
    while q:
        u = q.popleft()
        bfs.append(u)
        for v in adj[u]:
            if not seen[v]:
                seen[v] = True
                q.append(v)

    total, pos = try_order(bfs)
    if total <= B:
        print("YES")
        print(' '.join(str(pos[i]) for i in range(1, n + 1)))
        return

    # try reverse BFS
    total, pos = try_order(list(reversed(bfs)))
    if total <= B:
        print("YES")
        print(' '.join(str(pos[i]) for i in range(1, n + 1)))
        return

    # try simple identity order
    identity = list(range(1, n + 1))
    total, pos = try_order(identity)
    if total <= B:
        print("YES")
        print(' '.join(str(pos[i]) for i in range(1, n + 1)))
        return

    print("NO")

if __name__ == "__main__":
    solve()
```
