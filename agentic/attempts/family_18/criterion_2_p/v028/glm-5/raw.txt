```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Find a root (node with degree >= 1, or 1 if n=1)
    root = 1
    while root <= n and not adj[root]:
        root += 1
    
    # BFS to get traversal order
    order = []
    q = deque([root])
    visited = [False] * (n + 1)
    visited[root] = True
    
    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                q.append(v)
    
    # Assign bays in BFS order: 1, 2, ..., n
    bay = [0] * (n + 1)
    for i, u in enumerate(order):
        bay[u] = i + 1
    
    # Calculate total cost
    total_cost = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:
                total_cost += abs(bay[u] - bay[v])
    
    if total_cost <= B:
        print("YES")
        print(' '.join(map(str, bay[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
