Looking at this problem, we need to place n stations (forming a tree with n-1 edges) on n consecutive pads such that the total belt length is at most B.

**Key insight:** The minimum possible total length is achieved by finding an optimal linear arrangement of the tree. For a tree, the minimum linear arrangement problem can be solved optimally.

**Approach:** 
- The minimum total length for a tree equals the sum over all edges of the number of nodes on the smaller side when the edge is cut. This is because in any arrangement, an edge (u,v) contributes at least `min(size_subtree_u, n - size_subtree_u)` to the total (nodes from one side must be separated from nodes on the other side).
- This lower bound is achievable by a DFS-based arrangement.

**Algorithm:**
1. Root the tree at its centroid (node where all subtrees have size ≤ n/2).
2. Do DFS from centroid, placing nodes in DFS order.
3. For each edge, the subtree on the child side has all its nodes consecutive, and the parent side nodes are on the other side, achieving exactly `min(subtree_size, n - subtree_size)`.

    edges = []
    for _ in range(m):

        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    sys.setrecursionlimit(n + 1000)
    
    size = [0]*(n+1)
    def dfs_size(u, p):
        size[u] = 1
        for v in adj[u]:
            if v != p:
                dfs_size(v, u)
                size[u] += size[v]
    
    dfs_size(1, 0)
    
    def find_centroid(u, p):
        for v in adj[u]:
            if v != p and size[v] > n//2:
                return find_centroid(v, u)
        return u
    
    centroid = find_centroid(1, 0)
    dfs_size(centroid, 0)
    
    min_cost = 0
    def compute_cost(u, p):
        nonlocal min_cost
        for v in adj[u]:
            if v != p:
                min_cost += min(size[v], n - size[v])
                compute_cost(v, u)
    compute_cost(centroid, 0)
    
    if min_cost > B:
        print("NO")
        return
    
    pos = [0] * (n + 1)
    
    def subset_partition(children, target):
        """Find subset of children summing to target. Returns set of child indices."""
        # dp[s] = index of child that was last added to achieve sum s, or -2 if not achievable, -1 if s=0
        dp = [-2] * (target + 1)
        dp[0] = -1
        for idx, (v, s) in enumerate(children):
            for s_val in range(target, s - 1, -1):
                if dp[s_val] == -2 and dp[s_val - s] != -2:
                    dp[s_val] = idx
        if dp[target] == -2:
            return None
        # Reconstruct
        result = set()
        cur = target
        while cur > 0:
            idx = dp[cur]
            result.add(idx)
            cur -= children[idx][1]
        return result
    
    def arrange(u, p, lo, hi):
        sz = hi - lo + 1
        c = lo + (sz - 1) // 2
        pos[u] = c
        
        children = [(v, size[v]) for v in adj[u] if v != p]
        if not children:
            return
        
        left_cap = c - lo
        right_cap = hi - c
        
        left_indices = subset_partition(children, left_cap)
        if left_indices is None:
            left_indices = subset_partition(children, right_cap)
            if left_indices is not None:
                # Swap: these go right, rest go left
                left_indices = set(range(len(children))) - left_indices
        
        if left_indices is None:
            # Fallback greedy
            left_indices = set()
            used = 0
            for idx, (v, s) in enumerate(children):
                if used + s <= left_cap:
                    left_indices.add(idx)
                    used += s
        
        left_children = [children[i] for i in range(len(children)) if i in left_indices]
        right_children = [children[i] for i in range(len(children)) if i not in left_indices]
        
        cur = lo
        for v, s in left_children:
            arrange(v, u, cur, cur + s - 1)
            cur += s
        
        cur = c + 1
        for v, s in right_children:
            arrange(v, u, cur, cur + s - 1)
            cur += s
    
    arrange(centroid, 0, 1, n)
    
    actual = 0
    for u, v in edges:
        actual += abs(pos[u] - pos[v])
    
    if actual <= B:
        print("YES")
        print(' '.join(str(pos[i]) for i in range(1, n+1)))
    else:
        print("NO")

            else:

                left_indices.add(i)
                cur -= s

        if target == right_cap and (dp >> left_cap) & 1:
