```python
import sys
from array import array
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, B = data[0], data[1], data[2]
    adj = [[] for _ in range(n)]
    adj_mask = [0] * n
    p = 3

    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u].append(v)
        adj[v].append(u)
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u

    if n == 1:
        if B >= 0:
            print("YES")
            print(1)
        else:
            print("NO")
        return

    if B < n - 1:
        print("NO")
        return

    deg = [len(x) for x in adj]

    local_lb_sum = 0
    for d in deg:
        q = d // 2
        if d & 1:
            local_lb_sum += (q + 1) * (q + 1)
        else:
            local_lb_sum += q * (q + 1)
    lower_bound = max(n - 1, (local_lb_sum + 1) // 2)

    if B < lower_bound:
        print("NO")
        return

    def cost(order):
        pos = [0] * n
        for i, v in enumerate(order):
            pos[v] = i
        total = 0
        for u in range(n):
            pu = pos[u]
            for v in adj[u]:
                if u < v:
                    total += abs(pu - pos[v])
        return total

    def output_order(order):
        pos = [0] * n
        for i, v in enumerate(order, 1):
            pos[v] = i
        print("YES")
        print(*pos)

    def improve(order, rounds=8):
        order = order[:]
        pos = [0] * n
        for i, v in enumerate(order):
            pos[v] = i

        for _ in range(rounds):
            changed = False
            i = 0
            while i + 1 < n:
                a = order[i]
                b = order[i + 1]
                delta = 0

                for x in adj[a]:
                    if x != b:
                        px = pos[x]
                        delta += abs(i + 1 - px) - abs(i - px)

                for x in adj[b]:
                    if x != a:
                        px = pos[x]
                        delta += abs(i - px) - abs(i + 1 - px)

                if delta < 0:
                    order[i], order[i + 1] = b, a
                    pos[a], pos[b] = i + 1, i
                    changed = True
                i += 1

            if not changed:
                break

        return order

    parent = [-1] * n
    traversal = [0]
    for v in traversal:
        for w in adj[v]:
            if w != parent[v]:
                parent[w] = v
                traversal.append(w)

    subtree = [1] * n
    for v in reversed(traversal[1:]):
        subtree[parent[v]] += subtree[v]

    centroid = 0
    best_balance = n
    for v in range(n):
        balance = n - subtree[v]
        for w in adj[v]:
            if parent[w] == v:
                balance = max(balance, subtree[w])
        if balance < best_balance:
            best_balance = balance
            centroid = v

    max_deg_vertex = max(range(n), key=lambda x: deg[x])
    leaf = next((i for i in range(n) if deg[i] == 1), 0)
    roots = []
    for r in (0, centroid, max_deg_vertex, leaf):
        if r not in roots:
            roots.append(r)

    def dfs_order(root, reverse_key=False):
        result = []
        stack = [(root, -1)]
        while stack:
            v, par = stack.pop()
            result.append(v)
            children = [w for w in adj[v] if w != par]
            children.sort(key=lambda x: deg[x], reverse=not reverse_key)
            for w in children:
                stack.append((w, v))
        return result

    def bfs_order(root):
        result = []
        seen = bytearray(n)
        seen[root] = 1
        q = deque([root])
        while q:
            v = q.popleft()
            result.append(v)
            children = [w for w in adj[v] if not seen[w]]
            children.sort(key=lambda x: deg[x], reverse=True)
            for w in children:
                seen[w] = 1
                q.append(w)
        return result

    candidates = [list(range(n))]
    for r in roots:
        candidates.append(dfs_order(r, False))
        candidates.append(dfs_order(r, True))
        candidates.append(bfs_order(r))

    seen_orders = set()
    for candidate in candidates:
        candidate = improve(candidate)
        key = tuple(candidate)
        if key in seen_orders:
            continue
        seen_orders.add(key)
        if cost(candidate) <= B:
            output_order(candidate)
            return

    size = 1 << n
    cap = min(B + 1, n * n + 1)
    cuts = array('I', [0]) * size
    dp = array('I', [0]) * size

    for mask in range(1, size):
        bit = mask & -mask
        prev = mask ^ bit
        v = bit.bit_length() - 1

        c = cuts[prev] + deg[v] - 2 * ((adj_mask[v] & prev).bit_count())
        cuts[mask] = c

        best = cap
        bits = mask
        while bits:
            b = bits & -bits
            value = dp[mask ^ b]
            if value < best:
                best = value
            bits ^= b

        value = best + c
        if value > cap:
            value = cap
        dp[mask] = value

    full = size - 1
    if dp[full] > B:
        print("NO")
        return

    reversed_order = []
    mask = full
    while mask:
        target = dp[mask] - cuts[mask]
        bits = mask
        chosen = 0
        while bits:
            b = bits & -bits
            if dp[mask ^ b] == target:
                chosen = b
                break
            bits ^= b
        reversed_order.append(chosen.bit_length() - 1)
        mask ^= chosen

    order = reversed_order[::-1]
    output_order(order)

if __name__ == "__main__":
    main()
```
