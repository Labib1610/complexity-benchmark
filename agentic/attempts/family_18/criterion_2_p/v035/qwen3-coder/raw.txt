from collections import defaultdict, deque

def main():

    sys.setrecursionlimit(1000000)

    idx += 1
    m = int(data[idx])
    idx += 1
    B = int(data[idx])
    idx += 1
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)

    # we can do a BFS to orient edges.
    
    # First, let's find the root by checking degrees or just pick node 0
    root = 0
    # Build parent relationships using BFS
    parent = [-1] * n
    children = [[] for _ in range(n)]
    visited = [False] * n
    queue = deque()
    queue.append(root)
    visited[root] = True
    
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                children[u].append(v)
                queue.append(v)

    # Since n can be up to 1000, we can't try all permutations.
    # But we can try a greedy approach or specific layouts.

    # 2. Doing a DFS from centroid and laying out nodes

    pos_order = []

        pos_order.append(u)
        for v in children[u]:
            dfs(v)
    
    dfs(root)

    for i in range(n):
        position[pos_order[i]] = i + 1
    
    # Calculate total cost
    total_cost = 0
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
    
    if total_cost <= B:
        print("YES")

        print(' '.join(map(str, position2)))
        return
    
    # Try reverse of post-order
    position2_rev = [n + 1 - position2[i] for i in range(n)]
    total_cost2_rev = 0
    for u, v in edges:
        total_cost2_rev += abs(position2_rev[u] - position2_rev[v])
    
    if total_cost2_rev <= B:
        print("YES")
        print(' '.join(map(str, position2_rev)))
        return

    # but for large n, we might need to conclude it's impossible.
    # Given the constraints (n up to 1000), and that we only need to find one solution,

    # But this is getting complex. Let's see if our previous attempts work for the example.

    # Cost = |1-2| + |1-3| = 1 + 2 = 3 > 2 (budget)
    # Reverse: [3,2,1] -> Cost = |3-2| + |3-1| = 1 + 2 = 3 > 2
    # That's why it didn't work.
    
    # The example output is [2,1,3] which means positions: node 0->2, node 1->1, node 2->3
    # Cost = |2-1| + |2-3| = 1 + 1 = 2 <= 2. This works.
    
    # So our heuristics are not good enough. We need a better way.
    
    # For trees, there's a polynomial algorithm to find the optimal linear arrangement.
    # It's based on dynamic programming with tree decomposition.
    # But that's too complex for this problem.
    
    # Let's try a greedy improvement: local search.
    # But with n up to 1000, even that might be too slow.
    
    # Let's try a simple O(n) heuristic that often works well:
    # 1. Find a centroid of the tree
    # 2. Root the tree at the centroid
    # 3. Do a DFS to assign positions
    
    # But first, let's just try all nodes as root with our simple DFS approach,

                order_bfs.append(v)
