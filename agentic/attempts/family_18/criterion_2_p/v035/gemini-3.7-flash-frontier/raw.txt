```python
import sys
import time
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])
    
    if n == 1:
        if B >= 0:
            print("YES")
            print(1)
        else:
            print("NO")
        return

    edges = []
    adj_list = [[] for _ in range(n)]
    adj_mask = [0] * n
    deg = [0] * n
    
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        edges.append((u, v))
        adj_list[u].append(v)
        adj_list[v].append(u)
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
        deg[u] += 1
        deg[v] += 1

    if n <= 20:
        size = 1 << n
        dp = [10**9] * size
        parent = [0] * size
        cut = [0] * size
        dp[0] = 0
        
        for mask in range(size):
            c = cut[mask]
            d = dp[mask]
            for u in range(n):
                if not (mask & (1 << u)):
                    nmask = mask | (1 << u)
                    ncut = c + deg[u] - 2 * (adj_mask[u] & mask).bit_count()
                    cost = d + ncut
                    if cost < dp[nmask]:
                        dp[nmask] = cost
                        parent[nmask] = u
                        cut[nmask] = ncut

        if dp[size - 1] <= B:
            print("YES")
            curr = size - 1
            order = []
            while curr:
                u = parent[curr]
                order.append(u)
                curr ^= (1 << u)
            order.reverse()
            pos = [0] * n
            for i, u in enumerate(order):
                pos[u] = i + 1
            print(*(pos))
        else:
            print("NO")
        return

    # For n > 20: Heuristic + Simulated Annealing
    start_time = time.time()
    
    # Root tree at centroid to get a good initial ordering
    sz = [0] * n
    def get_sizes(u, p):
        s = 1
        for v in adj_list[u]:
            if v != p:
                s += get_sizes(v, u)
        sz[u] = s
        return s

    get_sizes(0, -1)
    
    def get_centroid(u, p, total):
        for v in adj_list[u]:
            if v != p and sz[v] > total // 2:
                return get_centroid(v, u, total)
        return u

    centroid = get_centroid(0, -1, n)

    # DFS order from centroid, placing children around
    visited = [False] * n
    dfs_order = []
    
    def dfs(u):
        visited[u] = True
        dfs_order.append(u)
        for v in adj_list[u]:
            if not visited[v]:
                dfs(v)
                
    dfs(centroid)

    order = list(dfs_order)
    pos = [0] * n
    for i, u in enumerate(order):
        pos[u] = i

    def compute_cost(p):
        return sum(abs(p[u] - p[v]) for u, v in edges)

    current_cost = compute_cost(pos)
    best_cost = current_cost
    best_pos = list(pos)

    if best_cost <= B:
        print("YES")
        print(*(p + 1 for p in best_pos))
        return

    # Simulated Annealing
    T = 5.0
    cooling_rate = 0.99995
    
    while time.time() - start_time < 4.5:
        i = random.randint(0, n - 1)
        j = random.randint(0, n - 1)
        if i == j:
            continue
        
        u = order[i]
        v = order[j]
        
        delta = 0
        for w in adj_list[u]:
            if w != v:
                delta += abs(j - pos[w]) - abs(i - pos[w])
        for w in adj_list[v]:
            if w != u:
                delta += abs(i - pos[w]) - abs(j - pos[w])

        if delta < 0 or random.random() < pow(2.718281828459045, -delta / max(T, 1e-6)):
            order[i], order[j] = v, u
            pos[u], pos[v] = j, i
            current_cost += delta
            if current_cost < best_cost:
                best_cost = current_cost
                best_pos = list(pos)
                if best_cost <= B:
                    print("YES")
                    print(*(p + 1 for p in best_pos))
                    return
        T *= cooling_rate

    if best_cost <= B:
        print("YES")
        print(*(p + 1 for p in best_pos))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
