import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, B = map(int, sys.stdin.readline().split())
    
    if m != n - 1:
        print("NO")
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    if n == 1:
        if 0 <= B:
            print("YES")
            print("1")
        else:
            print("NO")
        return
    
    # Find root: we'll pick the node with the smallest label for consistency
    # But actually, since it's a tree, we can root it at any node. Let's root at node 1.
    root = 1
    
    # Build tree structure with parent-child relationships
    # We'll do BFS to establish parent relationships
    parent = {root: None}
    children = defaultdict(list)
    queue = deque([root])
    
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if v not in parent:
                parent[v] = u
                children[u].append(v)
                queue.append(v)
    
    # Now we need to try different layouts to minimize total cable length
    # Since it's a tree, we can use dynamic programming approach or try to find an optimal layout
    
    # A good heuristic is to perform a DFS and assign positions in-order
    # This is similar to an in-order traversal of a binary tree
    
    # Let's try a different approach: since we want to minimize sum of distances,
    # we should place nodes that are connected close to each other
    
    # One effective method is to do a DFS and assign positions consecutively
    # But we need to be smarter about the order
    
    # Let's try doing a post-order DFS and placing nodes optimally
    
    # Actually, let's think differently:
    # We can model this as finding a linear arrangement of a tree that minimizes
    # the sum of edge lengths. This is known as the Minimum Linear Arrangement problem for trees.
    
    # For trees, a good heuristic is to use a DFS-based approach where we place
    # subtrees consecutively.
    
    # Let's implement a simple greedy approach:
    # Root the tree at node 1, then do a DFS and assign positions
    
    position = [0] * (n + 1)
    pos_counter = [0]  # Use list to allow modification in nested function
    
    def dfs_assign(node, start_pos):
        """Assign positions via DFS, returns (start, end) of assigned range"""
        children_list = children[node]
        if not children_list:
            # Leaf node
            pos_counter[0] += 1
            position[node] = pos_counter[0]
            return pos_counter[0], pos_counter[0]
        
        child_ranges = []
        current_start = start_pos
        for child in children_list:
            child_start, child_end = dfs_assign(child, current_start)
            child_ranges.append((child_start, child_end))
            current_start = child_end + 1
            
        # Place the current node somewhere among its children
        # A simple approach: place it at the beginning of its range
        pos_counter[0] += 1
        position[node] = pos_counter[0]
        
        # Adjust all positions to be consecutive starting from start_pos
        # Actually, let's re-think this
        
        return start_pos, current_start - 1
    
    # Reset counter
    pos_counter[0] = 0
    
    # Better approach: do an in-order-like assignment
    # Let's do a DFS and assign positions in the order we visit leaves first
    
    def assign_positions_inorder(node):
        for child in children[node]:
            assign_positions_inorder(child)
        pos_counter[0] += 1
        position[node] = pos_counter[0]
        
    assign_positions_inorder(root)
    
    # Calculate total cost of this arrangement
    total_cost = 0
    for u in range(1, n+1):
        for v in adj[u]:
            if v > u:  # Count each edge once
                total_cost += abs(position[u] - position[v])
    
    if total_cost <= B:
        print("YES")
        print(' '.join(map(str, position[1:])))
        return
    
    # If the simple DFS order doesn't work, let's try a different rooting
    # Try rooting at center node (centroid)
    
    # Find centroid of the tree
    def find_centroid():
        # First, find a leaf to start size computation
        size = [0] * (n + 1)
        
        def compute_size(node, par):
            sz = 1
            for neighbor in adj[node]:
                if neighbor != par:
                    sz += compute_size(neighbor, node)
            size[node] = sz
            return sz
        
        compute_size(1, -1)
        
        def find_centroid_node(node, par, total_size):
            for neighbor in adj[node]:
                if neighbor != par:
                    if size[neighbor] > total_size // 2:
                        return find_centroid_node(neighbor, node, total_size)
            
            # Check if current node is centroid
            max_subtree_size = 0
            for neighbor in adj[node]:
                if neighbor != par:
                    max_subtree_size = max(max_subtree_size, size[neighbor])
            
            if max_subtree_size <= total_size // 2:
                return node
            return -1
        
        centroid = find_centroid_node(1, -1, n)
        return centroid
    
    centroid = find_centroid()
    
    # Rebuild parent/children with new root
    parent = {centroid: None}
    children = defaultdict(list)
    queue = deque([centroid])
    
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if v not in parent:
                parent[v] = u
                children[u].append(v)
                queue.append(v)
    
    # Reset and assign positions again
    position = [0] * (n + 1)
    pos_counter[0] = 0
    
    def assign_positions_inorder_v2(node):
        for child in children[node]:
            assign_positions_inorder_v2(child)
        pos_counter[0] += 1
        position[node] = pos_counter[0]
        
    assign_positions_inorder_v2(centroid)
    
    # Calculate total cost
    total_cost = 0
    for u in range(1, n+1):
        for v in adj[u]:
            if v > u:  # Count each edge once
                total_cost += abs(position[u] - position[v])
    
    if total_cost <= B:
        print("YES")
        print(' '.join(map(str, position[1:])))
        return
    
    # Try yet another approach: sort nodes by degree and place high-degree nodes in middle
    
    # Simple fallback: try a breadth-first assignment
    position = [0] * (n + 1)
    queue = deque([centroid])
    visited = [False] * (n + 1)
    visited[centroid] = True
    pos_counter[0] = 0
    
    while queue:
        u = queue.popleft()
        pos_counter[0] += 1
        position[u] = pos_counter[0]
        
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                queue.append(v)
    
    # Calculate total cost
    total_cost = 0
    for u in range(1, n+1):
        for v in adj[u]:
            if v > u:  # Count each edge once
                total_cost += abs(position[u] - position[v])
                
    if total_cost <= B:
        print("YES")
        print(' '.join(map(str, position[1:])))
        return
    
    # If all heuristics fail, it might be impossible
    print("NO")

if __name__ == "__main__":
    main()
