```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        edges.append((u-1, v-1))
    # simple identity permutation
    perm = list(range(1, n+1))
    pos = [i for i in range(n)]  # 0‑based positions
    total = 0
    for u, v in edges:
        total += abs(pos[u] - pos[v])
    if total <= B:
        print("YES")
        print(' '.join(map(str, perm)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
