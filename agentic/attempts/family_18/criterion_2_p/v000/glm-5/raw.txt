import sys
from sys import setrecursionlimit

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    B = int(data[idx]); idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    # Find centroid(s)
    parent = [0] * (n + 1)
    size = [0] * (n + 1)
    order = []
    root = 1
    stack = [root]
    parent[root] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for w in adj[u]:
            if w != parent[u]:
                parent[w] = u
                stack.append(w)
    for u in reversed(order):
        size[u] = 1
        for w in adj[u]:
            if w != parent[u]:
                size[u] += size[w]

    centroid = root
    half = n // 2
    for u in range(1, n + 1):
        ok = True
        if n - size[u] > half:
            ok = False
        if ok:
            for w in adj[u]:
                if w != parent[u] and size[w] > half:
                    ok = False
                    break
        if ok:
            centroid = u
            break

    # BFS from centroid to get depths
    depth = [0] * (n + 1)
    visited = [False] * (n + 1)
    visited[centroid] = True
    queue = [centroid]
    head = 0
    while head < len(queue):
        u = queue[head]; head += 1
        for w in adj[u]:
            if not visited[w]:
                visited[w] = True
                depth[w] = depth[u] + 1
                queue.append(w)

    total = sum(depth[1:n+1])
    if total > B:
        print("NO")
        return

    # Assign positions: centroid at center, others by depth then BFS order
    pos = [0] * (n + 1)
    center = (n + 1) // 2
    pos[centroid] = center

    # Collect nodes by depth
    max_depth = max(depth[1:n+1])
    by_depth = [[] for _ in range(max_depth + 1)]
    for u in queue:
        if u != centroid:
            by_depth[depth[u]].append(u)

    left = center - 1
    right = center + 1
    for d in range(1, max_depth + 1):
        nodes = by_depth[d]
        # Split: half go left, half go right
        mid = len(nodes) // 2
        left_nodes = nodes[:mid]
        right_nodes = nodes[mid:]
        # Place left_nodes from right to left
        for u in reversed(left_nodes):
            pos[u] = left
            left -= 1
        # Place right_nodes from left to right
        for u in right_nodes:
            pos[u] = right
            right += 1

    print("YES")
    print(' '.join(str(pos[i]) for i in range(1, n + 1)))

setrecursionlimit(1 << 25)
solve()
