import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    B = int(data[idx])
    idx += 1
    
    if m != n - 1:
        print("NO")
        return
    
    edges = []
    adj = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    if n == 1:
        if 0 <= B:
            print("YES")
            print(1)
        else:
            print("NO")
        return
    
    # Find center of the tree
    def find_center(adj, n):
        degree = [0] * n
        leaves = deque()
        for i in range(n):
            degree[i] = len(adj[i])
            if degree[i] <= 1:
                leaves.append(i)
        
        remaining = n
        while remaining > 2:
            size = len(leaves)
            remaining -= size
            for _ in range(size):
                u = leaves.popleft()
                for v in adj[u]:
                    degree[v] -= 1
                    if degree[v] == 1:
                        leaves.append(v)
        return list(leaves)
    
    centers = find_center(adj, n)
    root = centers[0]
    
    # Build tree structure from root
    tree = defaultdict(list)
    visited = [False] * n
    stack = [root]
    visited[root] = True
    
    while stack:
        u = stack.pop()
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                tree[u].append(v)
                stack.append(v)
    
    # Compute subtree sizes
    size = [0] * n
    def dfs_size(u):
        sz = 1
        for v in tree[u]:
            sz += dfs_size(v)
        size[u] = sz
        return sz
    
    dfs_size(root)
    
    # Try to place nodes to minimize total stretch
    # We'll use a greedy approach: place subtrees consecutively
    
    pos = [-1] * n
    assigned = [False] * n
    
    def dfs_place(u, start, end):
        sz = size[u]
        if sz == 1:
            pos[u] = start
            assigned[u] = True
            return
        
        current = start
        for v in tree[u]:
            dfs_place(v, current, current + size[v] - 1)
            current += size[v]
        pos[u] = (start + end) // 2  # Place parent in the middle
        assigned[u] = True
    
    dfs_place(root, 1, n)
    
    # Calculate total stretch
    total = 0
    for u, v in edges:
        total += abs(pos[u] - pos[v])
    
    if total <= B:
        print("YES")
        print(' '.join(map(str, pos)))
    else:
        # Try different root
        if len(centers) > 1:
            root2 = centers[1]
            # Rebuild tree with new root
            tree2 = defaultdict(list)
            visited2 = [False] * n
            stack2 = [root2]
            visited2[root2] = True
            
            while stack2:
                u = stack2.pop()
                for v in adj[u]:
                    if not visited2[v]:
                        visited2[v] = True
                        tree2[u].append(v)
                        stack2.append(v)
            
            def dfs_size2(u):
                sz = 1
                for v in tree2[u]:
                    sz += dfs_size2(v)
                size[u] = sz
                return sz
            
            for i in range(n):
                size[i] = 0
            dfs_size2(root2)
            
            pos2 = [-1] * n
            def dfs_place2(u, start, end):
                sz = size[u]
                if sz == 1:
                    pos2[u] = start
                    return
                
                current = start
                for v in tree2[u]:
                    dfs_place2(v, current, current + size[v] - 1)
                    current += size[v]
                pos2[u] = (start + end) // 2
              
            dfs_place2(root2, 1, n)
            
            total2 = 0
            for u, v in edges:
                total2 += abs(pos2[u] - pos2[v])
            
            if total2 <= B:
                print("YES")
                print(' '.join(map(str, pos2)))
            else:
                print("NO")
        else:
            print("NO")

if __name__ == "__main__":
    main()
