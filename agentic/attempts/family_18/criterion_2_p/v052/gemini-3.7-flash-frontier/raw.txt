import sys

# Increase recursion depth for deep trees
sys.setrecursionlimit(300000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    if n == 1:
        print("YES")
        print("1")
        return

    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # Find subtree sizes and centroids
    sz = [0] * (n + 1)

    def get_sizes(u, p):
        s = 1
        for v in adj[u]:
            if v != p:
                s += get_sizes(v, u)
        sz[u] = s
        return s

    get_sizes(1, 0)

    centroids = []

    def get_centroids(u, p):
        is_centroid = True
        for v in adj[u]:
            if v != p:
                get_centroids(v, u)
                if sz[v] > n // 2:
                    is_centroid = False
        if n - sz[u] > n // 2:
            is_centroid = False
        if is_centroid:
            centroids.append(u)

    get_centroids(1, 0)

    def build_layout(root):
        # 1. Compute subtree sizes rooted at `root`
        sub_sz = [0] * (n + 1)

        def calc_sz(u, p):
            s = 1
            for v in adj[u]:
                if v != p:
                    s += calc_sz(v, u)
            sub_sz[u] = s
            return s

        calc_sz(root, 0)

        # 2. Recursive construction
        # dir = 'L': root is placed at the rightmost end of its interval, children extend to the left.
        # dir = 'R': root is placed at the leftmost end of its interval, children extend to the right.
        def layout_subtree(u, p, direction):
            children = [v for v in adj[u] if v != p]
            children.sort(key=lambda v: sub_sz[v], reverse=True)

            if direction == "R":
                res = [u]
                for v in children:
                    res.extend(layout_subtree(v, u, "R"))
                return res
            else:
                res = []
                for v in children:
                    res.extend(layout_subtree(v, u, "L"))
                res.append(u)
                return res

        # For global root, distribute children to Left and Right
        root_children = [v for v in adj[root]]
        root_children.sort(key=lambda v: sub_sz[v], reverse=True)

        left_nodes = []
        right_nodes = []

        # Alternately assign to left and right (largest subtrees closest to root)
        left_children = []
        right_children = []
        for i, v in enumerate(root_children):
            if i % 2 == 0:
                left_children.append(v)
            else:
                right_children.append(v)

        # For left children: reverse order so larger subtrees are closer to root
        for v in reversed(left_children):
            left_nodes.extend(layout_subtree(v, root, "L"))

        for v in right_children:
            right_nodes.extend(layout_subtree(v, root, "R"))

        return left_nodes + [root] + right_nodes

    def evaluate(order):
        pos = [0] * (n + 1)
        for slot, u in enumerate(order, start=1):
            pos[u] = slot
        total_len = 0
        for u, v in edges:
            total_len += abs(pos[u] - pos[v])
        return total_len, pos

    # First check centroids
    candidates = list(centroids)
    if n <= 500:
        candidates = list(range(1, n + 1))
    else:
        # Add a few other nodes (e.g. root 1)
        if 1 not in candidates:
            candidates.append(1)

    best_len = float("inf")
    best_pos = None

    for r in candidates:
        order = build_layout(r)
        total_len, pos = evaluate(order)
        if total_len <= B:
            print("YES")
            print(*(pos[i] for i in range(1, n + 1)))
            return
        if total_len < best_len:
            best_len = total_len
            best_pos = pos

    print("NO")


if __name__ == "__main__":
    solve()
