```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1]); B = int(data[2])
    adj = [[] for _ in range(n+1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    # BFS to get parent pointers and order
    parent = [0]*(n+1)
    order = []
    visited = [False]*(n+1)
    root = 1
    visited[root] = True
    q = deque([root])
    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                q.append(v)
    
    # Compute subtree sizes
    size = [1]*(n+1)
    for u in reversed(order):
        if parent[u] != 0:
            size[parent[u]] += size[u]
    
    # Greedy assignment: assign slots 1..n in DFS order
    slot = [0]*(n+1)
    cur = 1
    stack = [root]
    while stack:
        u = stack.pop()
        slot[u] = cur
        cur += 1
        # Push children in reverse order so first child gets next slot
        children = [v for v in adj[u] if parent[v] == u]
        for v in reversed(children):
            stack.append(v)
    
    # Compute total wire length
    total = 0
    for u in range(2, n+1):
        total += abs(slot[u] - slot[parent[u]])
    
    if total <= B:
        out = ['YES']
        out.append(' '.join(str(slot[i]) for i in range(1, n+1)))
        sys.stdout.write('\n'.join(out))
    else:
        sys.stdout.write('NO')

solve()
```
