```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    B = int(data[idx])
    idx += 1
    
    edges = []
    adj = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Check if the graph is a tree
    if m != n - 1:
        print("NO")
        return
        
    # Check if connected
    visited = [False] * n
    queue = deque()
    queue.append(0)
    visited[0] = True
    node_count = 1
    
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                node_count += 1
                queue.append(v)
                
    if node_count != n:
        print("NO")
        return
    
    # Try to find a layout with total cost <= B
    # We'll do BFS from each node as root and try to layout optimally
    
    # For a tree, we can try to find a layout that minimizes the total edge length
    # One approach: find a centroid, layout it in the middle, and recurse
    
    # But for simplicity and given constraints, let's try a greedy approach:
    # Find a leaf node, make it one end, and build from there
    
    # Actually, let's try to find a linear layout that minimizes total length
    # This is essentially finding a linear arrangement of a tree that minimizes sum of edge lengths
    
    # A good heuristic is to do a DFS and assign positions in-order
    # But we need to be more careful
    
    # Let's try a different approach:
    # Find the diameter of the tree, layout the diameter nodes optimally,
    # then layout the rest optimally
    
    def bfs_farthest(start):
        visited = [-1] * n
        queue = deque()
        queue.append((start, 0))
        visited[start] = 0
        farthest_node = start
        max_dist = 0
        
        while queue:
            u, dist = queue.popleft()
            if dist > max_dist:
                max_dist = dist
                farthest_node = u
            for v in adj[u]:
                if visited[v] == -1:
                    visited[v] = dist + 1
                    queue.append((v, dist + 1))
        return farthest_node, max_dist, visited
    
    # Find one end of the diameter
    end1, _, _ = bfs_farthest(0)
    end2, diameter, dists_from_end1 = bfs_farthest(end1)
    _, _, dists_from_end2 = bfs_farthest(end2)
    
    # Now we have the diameter path from end1 to end2
    # Let's reconstruct the path
    path = []
    visited_path = [False] * n
    current = end2
    while current != end1:
        path.append(current)
        visited_path[current] = True
        for neighbor in adj[current]:
            # The predecessor on the shortest path from end1 to current
            # is the one with dists_from_end1[neighbor] = dists_from_end1[current] - 1
            if dists_from_end1[neighbor] == dists_from_end1[current] - 1:
                current = neighbor
                break
    path.append(end1)
    path.reverse()
    
    # Assign positions to the path nodes
    positions = [-1] * n
    for i in range(len(path)):
        pos = i + 1  # positions are 1-indexed
        positions[path[i]] = pos
    
    # Now assign positions to the remaining nodes
    # Do a BFS from all path nodes to assign the rest
    queue = deque()
    visited_assign = [False] * n
    for node in path:
        queue.append(node)
        visited_assign[node] = True
    
    next_pos = len(path) + 1
    
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if not visited_assign[v]:
                positions[v] = next_pos
                next_pos += 1
                visited_assign[v] = True
                queue.append(v)
    
    # Calculate total cost
    total_cost = 0
    for u, v in edges:
        total_cost += abs(positions[u] - positions[v])
    
    if total_cost <= B:
        print("YES")
        print(' '.join(map(str, positions)))
        return
    
    # If the above didn't work, let's try a different layout
    # Try to layout as a caterpillar or in a way that minimizes cost
    
    # Another approach: try all permutations for small n, but that's too slow
    # Let's try a greedy assignment
    
    # Reset positions
    positions = [-1] * n
    
    # Try a DFS-based assignment
    def dfs_assign(start):
        stack = [start]
        visited_dfs = [False] * n
        visited_dfs[start] = True
        pos = 1
        positions[start] = pos
        pos += 1
        parent = [-1] * n
        
        while stack:
            u = stack[-1]
            # Find unvisited neighbor
            found = False
            for v in adj[u]:
                if not visited_dfs[v]:
                    visited_dfs[v] = True
                    positions[v] = pos
                    pos += 1
                    parent[v] = u
                    stack.append(v)
                    found = True
                    break
            if not found:
                stack.pop()
        
        return positions[:]
    
    # Try DFS from each node
    found = False
    best_positions = None
    min_cost = float('inf')
    
    # For small n, try all starting points
    # For larger n, try a few
    start_nodes = range(min(n, 10)) if n > 10 else range(n)
    
    for start in start_nodes:
        pos_trial = dfs_assign(start)
        cost = 0
        for u, v in edges:
            cost += abs(pos_trial[u] - pos_trial[v])
        if cost < min_cost:
            min_cost = cost
            best_positions = pos_trial[:]
        if cost <= B:
            print("YES")
            print(' '.join(map(str, pos_trial)))
            return
    
    if min_cost <= B:
        print("YES")
        print(' '.join(map(str, best_positions)))
        return
            
    print("NO")

if __name__ == "__main__":
    main()
```
