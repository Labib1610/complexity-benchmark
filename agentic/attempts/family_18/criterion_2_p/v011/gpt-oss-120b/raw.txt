```python
import sys
sys.setrecursionlimit(10**6)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # simple DFS ordering
    order = []
    visited = [False] * n
    stack = [0]
    visited[0] = True
    while stack:
        u = stack.pop()
        order.append(u)
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                stack.append(v)

    def check(order):
        pos = [0] * n
        for i, v in enumerate(order):
            pos[v] = i + 1  # positions 1..n
        total = 0
        for u in range(n):
            for v in adj[u]:
                if u < v:
                    total += abs(pos[u] - pos[v])
        return total, pos

    total, pos = check(order)
    if total <= B:
        print("YES")
        print(' '.join(map(str, pos)))
        return

    # try reverse order
    order_rev = list(reversed(order))
    total, pos = check(order_rev)
    if total <= B:
        print("YES")
        print(' '.join(map(str, pos)))
        return

    print("NO")

if __name__ == "__main__":
    solve()
```
