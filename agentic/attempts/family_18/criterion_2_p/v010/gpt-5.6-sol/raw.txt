```python
import sys
from array import array

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, budget = data[0], data[1], data[2]
    adj_mask = [0] * n
    degree = [0] * n

    idx = 3
    for _ in range(m):
        u = data[idx] - 1
        v = data[idx + 1] - 1
        idx += 2
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        degree[u] += 1
        degree[v] += 1

    if n == 1:
        if budget >= 0:
            print("YES")
            print(1)
        else:
            print("NO")
        return

    if budget < n - 1:
        print("NO")
        return

    total_states = 1 << n

    if n <= 255:
        cut = array('B', [0]) * total_states
    elif n <= 65535:
        cut = array('H', [0]) * total_states
    else:
        cut = array('I', [0]) * total_states

    INF = 0xFFFFFFFF
    dp = array('I', [INF]) * total_states
    dp[0] = 0

    cuts = cut
    dps = dp
    adjs = adj_mask
    degs = degree

    for mask in range(1, total_states):
        bit = mask & -mask
        v = bit.bit_length() - 1
        prev = mask ^ bit
        c = cuts[prev] + degs[v] - 2 * ((adjs[v] & prev).bit_count())
        cuts[mask] = c

        x = mask
        best = INF
        while x:
            b = x & -x
            val = dps[mask ^ b]
            if val < best:
                best = val
            x ^= b
        dps[mask] = best + c

    full = total_states - 1
    if dps[full] > budget:
        print("NO")
        return

    reverse_order = []
    mask = full
    while mask:
        target = dps[mask] - cuts[mask]
        x = mask
        while x:
            bit = x & -x
            prev = mask ^ bit
            if dps[prev] == target:
                reverse_order.append(bit.bit_length() - 1)
                mask = prev
                break
            x ^= bit

    order = reverse_order[::-1]
    position = [0] * n
    for pos, vertex in enumerate(order, 1):
        position[vertex] = pos

    print("YES")
    print(*position)

if __name__ == "__main__":
    main()
```
