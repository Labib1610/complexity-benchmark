```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, B = map(int, sys.stdin.readline().split())
    
    edges = []
    adj = [[] for _ in range(n)]
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Since m = n-1 and connected with no cycles, it's a tree
    # We need to find a layout of the tree on a line such that 
    # the sum of distances equals at most B
    
    # Try all possible layouts? No, that's too many.
    # Key insight: this is about finding a linear arrangement of a tree
    # that minimizes the sum of edge lengths.
    # This sum equals sum over edges (distances between mapped nodes)
    # This is exactly the minimum linear arrangement problem for trees.
    # For trees, there's a polynomial algorithm, but let's try a different approach.
    
    # Let's root the tree and do a post-order traversal to compute subtree sizes
    # Then we can use dynamic programming or greedy placement
    
    # Actually, since it's a tree, we can think of embedding it into a line
    # One good heuristic is the dfs-ordering or bfs-ordering
    # But we want the optimal solution
    
    # Let's model this properly:
    # We have a tree T=(V,E), |V|=n, |E|=n-1
    # We want to find a bijection f: V -> {1,2,...,n}
    # That minimizes sum_{(u,v) in E} |f(u) - f(v)|
    # And check if this sum <= B
    
    # This is the Minimum Linear Arrangement problem for trees
    # There exists a polynomial time algorithm using DP on trees
    # But for simplicity, let's implement a greedy approach first
    
    # Greedy approach: Use DFS to get an ordering
    def dfs_order(start):
        order = []
        visited = [False]*n
        stack = [start]
        while stack:
            node = stack.pop()
            if not visited[node]:
                visited[node] = True
                order.append(node)
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        stack.append(neighbor)
        return order
    
    # Try different starting points for better ordering
    best_cost = float('inf')
    best_assignment = None
    
    # Try BFS-based approach which often gives good results for linear arrangement
    from collections import deque
    
    def bfs_layout(start_node):
        queue = deque([start_node])
        visited = [False] * n
        visited[start_node] = True
        order = [start_node]
        
        while queue:
            node = queue.popleft()
            neighbors = []
            for nei in adj[node]:
                if not visited[nei]:
                    visited[nei] = True
                    neighbors.append(nei)
            # Add neighbors in some order (here just as they come)
            for nei in neighbors:
                order.append(nei)
                queue.append(nei)
                
        return order
    
    # Try multiple starting nodes
    for start in range(min(10, n)):  # Try first few nodes as roots
        order = bfs_layout(start)
        # Assign positions based on order
        pos = [0] * n
        for i in range(n):
            pos[order[i]] = i + 1  # Offices numbered from 1 to n
            
        # Calculate total cost
        total = 0
        for u, v in edges:
            total += abs(pos[u] - pos[v])
            
        if total < best_cost:
            best_cost = total
            best_assignment = pos[:]
    
    # Now let's also try DFS orderings
    def dfs_recursive(node, parent, visited, order):
        visited[node] = True
        order.append(node)
        for nei in adj[node]:
            if not visited[nei] and nei != parent:
                dfs_recursive(nei, node, visited, order)
    
    for start in range(min(10, n)):
        order = []
        visited = [False] * n
        dfs_recursive(start, -1, visited, order)
        
        pos = [0] * n
        for i in range(n):
            pos[order[i]] = i + 1
            
        total = 0
        for u, v in edges:
            total += abs(pos[u] - pos[v])
            
        if total < best_cost:
            best_cost = total
            best_assignment = pos[:]
    
    # If still no solution found or exceeds B
    if best_cost > B:
        print("NO")
    else:
        print("YES")
        print(*best_assignment)

threading.Thread(target=main).start()
```
