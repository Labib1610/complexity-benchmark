import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Find root (node with degree 1, or node 1 if n=1)
    root = 1
    for i in range(1, n + 1):
        if len(adj[i]) == 1:
            root = i
            break
    
    # BFS to get parent-child relationships
    parent = [0] * (n + 1)
    children = [[] for _ in range(n + 1)]
    order = []
    q = deque([root])
    parent[root] = -1
    
    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            if parent[v] == 0 and v != parent[u]:
                parent[v] = u
                children[u].append(v)
                q.append(v)
    
    # Compute subtree sizes
    size = [1] * (n + 1)
    for u in reversed(order):
        for v in children[u]:
            size[u] += size[v]
    
    # DP: dp[u][k] = min cost to place subtree of u in k consecutive sockets
    # We use a dictionary for each node
    dp = [{} for _ in range(n + 1)]
    
    # Process nodes in reverse BFS order (bottom-up)
    for u in reversed(order):
        if not children[u]:
            # Leaf node
            dp[u][1] = 0
        else:
            # Internal node
            # Initialize with first child
            first_child = children[u][0]
            for sz, cost in dp[first_child].items():
                dp[u][sz] = cost
            
            # Merge remaining children
            for v in children[u][1:]:
                new_dp = {}
                for sz1, cost1 in dp[u].items():
                    for sz2, cost2 in dp[v].items():
                        new_sz = sz1 + sz2
                        new_cost = cost1 + cost2 + sz1 * sz2
                        if new_sz not in new_dp or new_cost < new_dp[new_sz]:
                            new_dp[new_sz] = new_cost
                dp[u] = new_dp
    
    # Check if any size n has cost <= B
    if n in dp[root] and dp[root][n] <= B:
        print("YES")
        
        # Reconstruct the assignment
        assignment = [0] * (n + 1)
        available = list(range(1, n + 1))
        
        def assign(u, sockets):
            # Assign subtree rooted at u to the given list of sockets
            assignment[u] = sockets[0]
            remaining = sockets[1:]
            
            if not children[u]:
                return
            
            # Distribute remaining sockets to children
            # We need to find a valid distribution
            # Use greedy approach based on subtree sizes
            child_sockets = {}
            idx = 0
            for v in children[u]:
                child_size = size[v]
                child_sockets[v] = remaining[idx:idx + child_size]
                idx += child_size
            
            for v in children[u]:
                assign(v, child_sockets[v])
        
        assign(root, available)
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
