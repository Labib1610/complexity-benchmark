import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, B = map(int, sys.stdin.readline().split())
    
    # Build the tree from the cables
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Since it's a tree (n nodes, n-1 edges, connected), we can root it arbitrarily
    # We'll try to find an optimal embedding of the tree into a line to minimize sum of distances
    
    # The optimal way to embed a tree into a line to minimize sum of distances
    # is to find a centroid of the tree and then do a DFS ordering
    
    # First, find a centroid of the tree
    # A vertex is a centroid if when removed, all components have size <= n/2
    
    size = [0] * (n + 1)
    
    def dfs_size(u, parent):
        sz = 1
        for v in adj[u]:
            if v != parent:
                sz += dfs_size(v, u)
        size[u] = sz
        return sz
    
    dfs_size(1, -1)
    
    def find_centroid(u, parent, total):
        for v in adj[u]:
            if v != parent and size[v] > total // 2:
                return find_centroid(v, u, total)
        return u
    
    centroid = find_centroid(1, -1, n)
    
    # Now, we root the tree at the centroid and do a DFS to get an ordering
    # This ordering when placed on the line minimizes the sum of distances
    
    order = []
    
    def dfs_order(u, parent):
        order.append(u)
        for v in adj[u]:
            if v != parent:
                dfs_order(v, u)
    
    dfs_order(centroid, -1)
    
    # Now we assign the order to slots 1 to n
    assignment = [0] * (n + 1)
    for i in range(n):
        assignment[order[i]] = i + 1
    
    # Calculate the total cable length
    total_length = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:  # Count each edge only once
                total_length += abs(assignment[u] - assignment[v])
    
    if total_length <= B:
        print("YES")
        print(*assignment[1:])
    else:
        print("NO")

if __name__ == "__main__":
    main()
