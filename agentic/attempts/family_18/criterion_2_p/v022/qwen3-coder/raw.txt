```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    graph = defaultdict(list)
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-indexed
        v = int(data[idx+1]) - 1
        idx += 2
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
    
    # Since m = n-1 and the graph is connected, it's a tree
    # We can use BFS to find a spanning tree (it already is one)
    
    # Try to assign coordinates by placing the tree on a line
    # One strategy: find a centroid or just try a DFS ordering
    
    # Let's do a DFS traversal to get an ordering
    visited = [False] * n
    order = []
    
    def dfs(u):
        visited[u] = True
        order.append(u)
        for v in graph[u]:
            if not visited[v]:
                dfs(v)
    
    # Start DFS from node 0
    dfs(0)
    
    # Assign coordinates based on this order
    coord = [0] * n
    for i in range(n):
        coord[order[i]] = i + 1  # 1-indexed coordinates
    
    # Calculate the cost
    cost = 0
    for u, v in edges:
        cost += abs(coord[u] - coord[v])
    
    if cost <= B:
        print("YES")
        result = [0] * n
        for i in range(n):
            result[order[i]] = i + 1
        print(' '.join(map(str, result)))
        return
    
    # If the DFS order doesn't work, try BFS order
    visited = [False] * n
    order_bfs = []
    queue = deque()
    queue.append(0)
    visited[0] = True
    
    while queue:
        u = queue.popleft()
        order_bfs.append(u)
        for v in graph[u]:
            if not visited[v]:
                visited[v] = True
                queue.append(v)
    
    coord = [0] * n
    for i in range(n):
        coord[order_bfs[i]] = i + 1
    
    cost = 0
    for u, v in edges:
        cost += abs(coord[u] - coord[v])
    
    if cost <= B:
        print("YES")
        result = [0] * n
        for i in range(n):
            result[order_bfs[i]] = i + 1
        print(' '.join(map(str, result)))
        return
    
    # Try to optimize by finding a better layout
    # For trees, one good heuristic is to root at centroid
    # But let's try a simple approach: try different starting nodes
    
    for start_node in range(n):
        visited = [False] * n
        order_dfs = []
        
        def dfs_local(u):
            visited[u] = True
            order_dfs.append(u)
            for v in graph[u]:
                if not visited[v]:
                    dfs_local(v)
        
        visited = [False] * n
        dfs_local(start_node)
        
        coord = [0] * n
        for i in range(n):
            coord[order_dfs[i]] = i + 1
        
        cost = 0
        for u, v in edges:
            cost += abs(coord[u] - coord[v])
        
        if cost <= B:
            print("YES")
            result = [0] * n
            for i in range(n):
                result[order_dfs[i]] = i + 1
            print(' '.join(map(str, result)))
            return
    
    # If still not working, try BFS from different nodes
    for start_node in range(n):
        visited = [False] * n
        order_bfs_local = []
        queue = deque([start_node])
        visited[start_node] = True
        
        while queue:
            u = queue.popleft()
            order_bfs_local.append(u)
            for v in graph[u]:
                if not visited[v]:
                    visited[v] = True
                    queue.append(v)
        
        coord = [0] * n
        for i in range(n):
            coord[order_bfs_local[i]] = i + 1
        
        cost = 0
        for u, v in edges:
            cost += abs(coord[u] - coord[v])
        
        if cost <= B:
            print("YES")
            result = [0] * n
            for i in range(n):
                result[order_bfs_local[i]] = i + 1
            print(' '.join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
