import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Find a root (node with degree 1 if exists, otherwise node 1)
    root = 1
    for i in range(1, n + 1):
        if len(adj[i]) == 1:
            root = i
            break
    
    # BFS to get parent and order
    parent = [0] * (n + 1)
    order = []
    stack = [root]
    parent[root] = -1  # mark as visited
    while stack:
        u = stack.pop()
        order.append(u)
        for v in adj[u]:
            if parent[v] == 0:
                parent[v] = u
                stack.append(v)
    
    # Process in reverse order (leaves first)
    # cnt[u] = number of nodes in subtree rooted at u
    cnt = [1] * (n + 1)
    for u in reversed(order):
        for v in adj[u]:
            if v != parent[u]:
                cnt[u] += cnt[v]
    
    # Calculate minimum cost
    min_cost = 0
    for u in order:
        for v in adj[u]:
            if v != parent[u]:
                min_cost += cnt[v] * (n - cnt[v])
    
    if min_cost > B:
        print("NO")
        return
    
    # Greedy assignment
    # For each node, we need to assign a position in its subtree's range
    # The subtree of u occupies positions [start[u], start[u] + cnt[u] - 1]
    # We assign u to start[u] + offset[u] where offset[u] is determined by children
    
    start = [0] * (n + 1)
    offset = [0] * (n + 1)
    
    # Process in order (root first)
    start[root] = 1
    
    for u in order:
        # Assign positions to children
        # Children subtrees are placed consecutively
        # We need to decide the order of children
        # To minimize cost, we should place children in some order
        # For now, just place them in the order they appear in adj
        
        # But we need to be careful: the position of u affects the cost
        # u should be placed such that it's close to its children
        
        # Let's compute the position of u first
        # u's position = start[u] + offset[u]
        # offset[u] is determined by the sizes of children placed before u
        
        # Initially, offset[u] = 0 (u is at the start of its range)
        # But we can adjust this based on children
        
        # Actually, let's think differently:
        # For each node u, we have a range [start[u], end[u]] where end[u] = start[u] + cnt[u] - 1
        # We need to place u somewhere in this range
        # The children subtrees are placed in the remaining positions
        
        # Let's use a greedy approach:
        # For each node, place it at the start of its range
        # Then place children subtrees after it
        
        # This gives us the minimum cost assignment
        
        pos_u = start[u] + offset[u]
        
        # Now assign start positions to children
        child_start = start[u]
        # If u is at position pos_u, then children are placed around it
        # Let's place children before u first, then after u
        
        # Actually, let's just place children in order after u
        # This is a simple greedy that works for minimum cost
        
        child_pos = start[u]
        for v in adj[u]:
            if v != parent[u]:
                start[v] = child_pos
                child_pos += cnt[v]
    
    # Now compute the actual positions
    # For each node u, its position is start[u] + offset[u]
    # But we need to compute offset[u] based on the structure
    
    # Let's recompute more carefully
    # We'll use a different approach: assign positions in DFS order
    
    pos = [0] * (n + 1)
    current = 1
    
    # DFS from root
    stack = [(root, 0)]  # (node, state) state 0 = enter, 1 = exit
    while stack:
        u, state = stack.pop()
        if state == 0:
            # Enter node: assign position
            pos[u] = current
            current += 1
            # Push children in reverse order for correct DFS order
            for v in reversed(adj[u]):
                if v != parent[u]:
                    stack.append((v, 1))  # exit marker
                    stack.append((v, 0))  # enter
        # state == 1: exit, do nothing
    
    # Verify cost
    cost = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if v > u:  # count each edge once
                cost += abs(pos[u] - pos[v])
    
    if cost <= B:
        print("YES")
        print(' '.join(map(str, pos[1:])))
    else:
        print("NO")

solve()
