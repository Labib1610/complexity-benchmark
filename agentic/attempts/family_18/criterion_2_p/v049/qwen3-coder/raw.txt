import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    graph = defaultdict(list)
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
    
    # Since it's a tree (n nodes, n-1 edges, connected, no cycles)
    # We can try all possible layouts by rooting the tree and doing DFS
    
    # Try to find a layout with cost <= B
    # A good heuristic is to do a BFS/DFS layout
    
    # Let's root at node 1
    root = 1
    parent = {root: None}
    order = []
    queue = deque([root])
    while queue:
        node = queue.popleft()
        order.append(node)
        for neighbor in graph[node]:
            if neighbor not in parent:
                parent[neighbor] = node
                queue.append(neighbor)
    
    # Now we have a BFS ordering. Assign windows based on this order
    window_assignment = {}
    for i in range(n):
        window_assignment[order[i]] = i + 1
    
    # Calculate the cost of this assignment
    total_cost = 0
    for u, v in edges:
        w_u = window_assignment[u]
        w_v = window_assignment[v]
        total_cost += abs(w_u - w_v)
    
    if total_cost <= B:
        result = [0] * (n + 1)
        for node in range(1, n + 1):
            result[node] = window_assignment[node]
        print("YES")
        print(' '.join(map(str, result[1:])))
        return
    
    # If BFS doesn't work, let's try to optimize using local search or other methods
    # But given time constraints, let's try a few permutations
    
    # For small n, we could try all permutations, but that's too slow
    # Instead, let's try a different approach: centroid decomposition might help,
    # but for now, let's just try a simple greedy method
    
    # Alternative idea: since it's a tree, we can do a post-order traversal and assign optimally
    
    # Let's build the tree properly and try a better assignment
    
    # Build adjacency list again
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    # Function to get DFS order
    def dfs_order(start):
        visited = set()
        order = []
        stack = [start]
        while stack:
            node = stack.pop()
            if node in visited:
                continue
            visited.add(node)
            order.append(node)
            for nei in adj[node]:
                if nei not in visited:
                    stack.append(nei)
        return order
    
    # Try multiple starting points
    found = False
    best_assignment = None
    best_cost = float('inf')
    
    # Try all nodes as potential roots for better layout
    for start_node in range(1, min(n+1, 10)):  # Limit to first 10 for speed
        order = []
        visited = set()
        queue = deque([start_node])
        visited.add(start_node)
        order.append(start_node)
        while queue:
            node = queue.popleft()
            for nei in adj[node]:
                if nei not in visited:
                    visited.add(nei)
                    order.append(nei)
                    queue.append(nei)
        
        # Assign windows
        window_map = {}
        for i in range(len(order)):
            window_map[order[i]] = i + 1
            
        # Compute cost
        cost = 0
        for u, v in edges:
            cost += abs(window_map[u] - window_map[v])
            
        if cost < best_cost:
            best_cost = cost
            best_assignment = window_map
            
        if cost <= B:
            result = [0] * (n + 1)
            for i in range(1, n + 1):
                result[i] = window_map[i]
            print("YES")
            print(' '.join(map(str, result[1:])))
            return
    
    # If even best found so far is within budget
    if best_cost <= B:
        result = [0] * (n + 1)
        for i in range(1, n + 1):
            result[i] = best_assignment[i]
        print("YES")
        print(' '.join(map(str, result[1:])))
        return
        
    # Last resort: check if any assignment works by trying to minimize cost
    # One way is to sort nodes by degree and place high-degree nodes in middle
    
    degrees = [(len(adj[node]), node) for node in range(1, n+1)]
    degrees.sort(reverse=True)
    
    # Place highest degree in center
    nodes_by_degree = [node for deg, node in degrees]
    
    # Assign windows based on this order
    window_map = {}
    for i in range(n):
        window_map[nodes_by_degree[i]] = i + 1
        
    cost = 0
    for u, v in edges:
        cost += abs(window_map[u] - window_map[v])
        
    if cost <= B:
        result = [0] * (n + 1)
        for i in range(1, n + 1):
            result[i] = window_map[i]
        print("YES")
        print(' '.join(map(str, result[1:])))
        return

if __name__ == "__main__":
    main()
