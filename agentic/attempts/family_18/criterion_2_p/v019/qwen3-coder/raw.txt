import sys
from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    index = 3
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-indexed
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Since the graph is a tree (n nodes, n-1 edges, connected), we can build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # We'll try all possible permutations for small n, but since n can be up to 1000,
    # we need a smarter approach. However, given the structure is a tree, we can use
    # the fact that the optimal layout is when the tree is laid out as a path.
    # But actually, we just need to find one valid assignment.
    
    # A greedy approach: find a BFS ordering of the tree and place nodes accordingly.
    # This ensures that the distances are minimized.
    
    # Let's root the tree at node 0 and do BFS to get an ordering
    parent = [-1] * n
    visited = [False] * n
    queue = deque()
    queue.append(0)
    visited[0] = True
    order = [0]
    
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                queue.append(v)
                order.append(v)
    
    # Now assign positions based on this BFS order
    position = [0] * n
    for i in range(n):
        position[order[i]] = i + 1  # Bay numbers are 1-indexed
    
    # Calculate total cost
    total_cost = 0
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
    
    if total_cost <= B:
        print("YES")
        print(*position)
        return
    
    # If BFS doesn't work, let's try DFS linearization
    # Reset and do DFS
    visited = [False] * n
    dfs_order = []
    
    def dfs(u, par):
        dfs_order.append(u)
        for v in adj[u]:
            if v != par:
                dfs(v, u)
    
    dfs(0, -1)
    
    position = [0] * n
    for i in range(n):
        position[dfs_order[i]] = i + 1
    
    total_cost = 0
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
    
    if total_cost <= B:
        print("YES")
        print(*position)
        return
    
    # For a tree, another good heuristic is to use centroid decomposition or
    # simply try placing nodes in a way that minimizes edge lengths.
    # But for now, let's just check if the DFS order works.
    # If not, we might need a more sophisticated approach.
    
    # Actually, let's think differently. The problem guarantees it's a tree.
    # One optimal strategy is to perform a post-order traversal and assign
    # positions greedily to minimize the total distance.
    
    # But given time constraints, let's implement a simple greedy:
    # Place the tree nodes in a line according to a DFS traversal.
    
    # Since both BFS and DFS didn't work in our simple assignment,
    # let's consider that maybe we need to try a different rooting or approach.
    
    # However, for trees, a simple DFS ordering often gives a good result.
    # Let's recheck our calculation with DFS.
    
    # Another idea: since it's a tree, we can root it arbitrarily and do a DFS.
    # Then assign positions 1, 2, ..., n to the order of visitation.
    
    # Let's try rooting at the node with highest degree for better distribution
    degrees = [len(adj[i]) for i in range(n)]
    max_degree_node = degrees.index(max(degrees))
    
    visited = [False] * n
    dfs_order = []
    
    def dfs_from_node(u):
        dfs_order.append(u)
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs_from_node(v)
    
    dfs_from_node(max_degree_node)
    
    position = [0] * n
    for i in range(n):
        position[dfs_order[i]] = i + 1
    
    total_cost = 0
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
    
    if total_cost <= B:
        print("YES")
        print(*position)
        return
    
    # If still not working, let's try a simple brute force for small n
    if n <= 8:
        from itertools import permutations
        for perm in permutations(range(1, n+1)):
            pos_map = {i: perm[i] for i in range(n)}
            cost = 0
            for u, v in edges:
                cost += abs(pos_map[u] - pos_map[v])
            if cost <= B:
                print("YES")
                print(*perm)
                return
        print("NO")
        return
    
    # If none of the heuristics work, we have to conclude it's not possible
    # But given the problem nature, one of the above should work.
    # Let's just output NO if we haven't found a solution yet.
    print("NO")

solve()
