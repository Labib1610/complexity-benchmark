```python
import sys
from array import array

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, budget = data[0], data[1], data[2]
    adj = [0] * n
    degree = [0] * n
    edges = []

    idx = 3
    identity_cost = 0
    for _ in range(m):
        u = data[idx] - 1
        v = data[idx + 1] - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        degree[u] += 1
        degree[v] += 1
        edges.append((u, v))
        identity_cost += abs(u - v)

    if n == 1:
        if budget >= 0:
            print("YES")
            print(1)
        else:
            print("NO")
        return

    if budget < n - 1:
        print("NO")
        return

    if identity_cost <= budget:
        print("YES")
        print(*range(1, n + 1))
        return

    total_masks = 1 << n
    max_cost = (n - 1) * (n - 1)
    cap = min(budget + 1, max_cost + 1)

    cut_type = 'B' if n <= 256 else ('H' if n <= 65536 else 'I')
    cut = array(cut_type, [0]) * total_masks

    for mask in range(1, total_masks):
        bit = mask & -mask
        v = bit.bit_length() - 1
        previous = mask ^ bit
        cut[mask] = (
            cut[previous]
            + degree[v]
            - 2 * ((adj[v] & previous).bit_count())
        )

    dp = array('I', [cap]) * total_masks
    parent_type = 'B' if n <= 255 else 'H'
    no_parent = 255 if parent_type == 'B' else 65535
    parent = array(parent_type, [no_parent]) * total_masks
    dp[0] = 0

    for mask in range(1, total_masks):
        x = mask
        best = cap
        chosen = no_parent

        while x:
            bit = x & -x
            value = dp[mask ^ bit]
            if value < best:
                best = value
                chosen = bit.bit_length() - 1
            x ^= bit

        if best < cap:
            value = best + cut[mask]
            if value > cap:
                value = cap
            dp[mask] = value
            if value < cap:
                parent[mask] = chosen

    full = total_masks - 1
    if dp[full] > budget:
        print("NO")
        return

    reverse_order = []
    mask = full
    while mask:
        v = parent[mask]
        reverse_order.append(v)
        mask ^= 1 << v

    order = reverse_order[::-1]
    position = [0] * n
    for pos, v in enumerate(order, 1):
        position[v] = pos

    print("YES")
    print(*position)

if __name__ == "__main__":
    main()
```
