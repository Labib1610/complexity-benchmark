import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Find root (node with degree != 1, or node 1 if n=1)
    root = 1
    if n > 1:
        for i in range(1, n + 1):
            if len(adj[i]) > 1:
                root = i
                break
    
    # BFS to get parent-child relationships
    parent = [0] * (n + 1)
    children = [[] for _ in range(n + 1)]
    order = []
    stack = [root]
    parent[root] = -1
    while stack:
        u = stack.pop()
        order.append(u)
        for v in adj[u]:
            if parent[v] == 0 and v != parent[u]:
                parent[v] = u
                children[u].append(v)
                stack.append(v)
    
    # Compute subtree sizes
    size = [1] * (n + 1)
    for u in reversed(order):
        for v in children[u]:
            size[u] += size[v]
    
    # Place nodes on line
    pos = [0] * (n + 1)
    cur = 1
    
    def dfs(u):
        nonlocal cur
        pos[u] = cur
        cur += 1
        # Sort children by subtree size (largest first)
        sorted_children = sorted(children[u], key=lambda x: -size[x])
        for v in sorted_children:
            dfs(v)
    
    dfs(root)
    
    # Calculate total cost
    total = 0
    for u in range(1, n + 1):
        for v in children[u]:
            total += abs(pos[u] - pos[v])
    
    if total <= B:
        print("YES")
        print(' '.join(map(str, pos[1:])))
    else:
        print("NO")

solve()
