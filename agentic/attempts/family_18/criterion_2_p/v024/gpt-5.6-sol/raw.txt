import sys
from array import array

def output_order(order, n):
    pos = [0] * n
    for i, v in enumerate(order, 1):
        pos[v] = i
    sys.stdout.write("YES\n" + " ".join(map(str, pos)) + "\n")
    sys.exit(0)

def order_cost(order, adj):
    pos = [0] * len(order)
    for i, v in enumerate(order):
        pos[v] = i
    total = 0
    for u in range(len(order)):
        pu = pos[u]
        for v in adj[u]:
            if u < v:
                total += abs(pu - pos[v])
    return total, pos

def improve_order(order, adj, passes=20):
    n = len(order)
    cost, pos = order_cost(order, adj)

    for _ in range(passes):
        changed = False
        for i in range(n - 1):
            a = order[i]
            b = order[i + 1]
            delta = 0

            for x in adj[a]:
                if x != b:
                    delta += abs(i + 1 - pos[x]) - abs(i - pos[x])
            for x in adj[b]:
                if x != a:
                    delta += abs(i - pos[x]) - abs(i + 1 - pos[x])

            if delta < 0:
                order[i], order[i + 1] = b, a
                pos[a], pos[b] = i + 1, i
                cost += delta
                changed = True

        if not changed:
            break

    return order, cost

def rooted_orders(root, adj):
    n = len(adj)
    parent = [-2] * n
    parent[root] = -1
    traversal = []
    stack = [root]

    while stack:
        v = stack.pop()
        traversal.append(v)
        for u in adj[v]:
            if parent[u] == -2:
                parent[u] = v
                stack.append(u)

    size = [1] * n
    for v in reversed(traversal):
        if parent[v] >= 0:
            size[parent[v]] += size[v]

    children = [[] for _ in range(n)]
    for v in traversal:
        if parent[v] >= 0:
            children[parent[v]].append(v)

    results = []
    for mode in range(3):
        if mode == 0:
            for ch in children:
                ch.sort(key=lambda x: size[x])
        elif mode == 1:
            for ch in children:
                ch.sort(key=lambda x: size[x], reverse=True)

        order = []
        stack = [root]
        while stack:
            v = stack.pop()
            order.append(v)
            ch = children[v]
            for u in reversed(ch):
                stack.append(u)
        results.append(order)
        results.append(order[::-1])

    return results, size, parent

def find_centroid(adj):
    n = len(adj)
    parent = [-1] * n
    traversal = [0]
    for v in traversal:
        for u in adj[v]:
            if u != parent[v]:
                parent[u] = v
                traversal.append(u)

    size = [1] * n
    for v in reversed(traversal):
        for u in adj[v]:
            if parent[u] == v:
                size[v] += size[u]

    best = 0
    best_part = n
    for v in range(n):
        largest = n - size[v]
        for u in adj[v]:
            if parent[u] == v and size[u] > largest:
                largest = size[u]
        if largest < best_part:
            best_part = largest
            best = v
    return best

def degree_lower_bound(adj):
    s = 0
    for neighbors in adj:
        d = len(neighbors)
        if d & 1:
            q = d // 2
            s += (q + 1) * (q + 1)
        else:
            q = d // 2
            s += q * (q + 1)
    return (s + 1) // 2

def exact_subset_dp(n, adj_masks, degrees, budget):
    total_states = 1 << n
    max_cost = (n - 1) * (n - 1)
    inf = max_cost + 1

    cuts = array('H', [0]) * total_states
    dp = array('I', [inf]) * total_states
    dp[0] = 0

    for mask in range(1, total_states):
        bit = mask & -mask
        v = bit.bit_length() - 1
        prev = mask ^ bit
        cuts[mask] = cuts[prev] + degrees[v] - 2 * ((adj_masks[v] & prev).bit_count())

        bits = mask
        best = inf
        while bits:
            b = bits & -bits
            val = dp[mask ^ b]
            if val < best:
                best = val
            bits ^= b
        dp[mask] = best + cuts[mask]

    full = total_states - 1
    if dp[full] > budget:
        return None

    reversed_order = []
    mask = full
    while mask:
        target = dp[mask] - cuts[mask]
        bits = mask
        chosen = -1
        while bits:
            b = bits & -bits
            if dp[mask ^ b] == target:
                chosen = b.bit_length() - 1
                mask ^= b
                reversed_order.append(chosen)
                break
            bits ^= b
        if chosen < 0:
            return None

    return reversed_order[::-1]

def sparse_exact_dp(n, adj_masks, degrees, budget):
    full = (1 << n) - 1
    current = {0: (0, 0)}
    chosen_parent = {}

    for k in range(n):
        nxt = {}
        remaining_after = n - k - 2

        for mask, (cost, cut) in current.items():
            available = full ^ mask
            while available:
                bit = available & -available
                available ^= bit
                v = bit.bit_length() - 1
                new_mask = mask | bit
                new_cut = cut + degrees[v] - 2 * ((adj_masks[v] & mask).bit_count())
                new_cost = cost + new_cut

                if new_cost > budget:
                    continue
                if remaining_after > 0 and new_cost + remaining_after > budget:
                    continue

                old = nxt.get(new_mask)
                if old is None or new_cost < old[0]:
                    nxt[new_mask] = (new_cost, new_cut)
                    chosen_parent[new_mask] = v

        current = nxt
        if not current:
            return None

    if full not in current or current[full][0] > budget:
        return None

    rev = []
    mask = full
    while mask:
        v = chosen_parent[mask]
        rev.append(v)
        mask ^= 1 << v
    return rev[::-1]

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, budget = data[0], data[1], data[2]
    adj = [[] for _ in range(n)]
    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u].append(v)
        adj[v].append(u)

    if n == 1:
        if budget >= 0:
            output_order([0], n)
        print("NO")
        return

    if budget < n - 1:
        print("NO")
        return

    degrees = [len(x) for x in adj]

    # Paths attain the universal lower bound n - 1.
    if max(degrees) <= 2:
        start = next(i for i, d in enumerate(degrees) if d == 1)
        order = []
        prev = -1
        cur = start
        while cur != -1:
            order.append(cur)
            nxt = -1
            for u in adj[cur]:
                if u != prev:
                    nxt = u
                    break
            prev, cur = cur, nxt
        output_order(order, n)

    # Handle stars exactly.
    center = max(range(n), key=lambda x: degrees[x])
    if degrees[center] == n - 1:
        minimum = (n * n) // 4
        if minimum > budget:
            print("NO")
            return
        leaves = [v for v in range(n) if v != center]
        left_count = (n - 1) // 2
        order = leaves[:left_count] + [center] + leaves[left_count:]
        output_order(order, n)

    if degree_lower_bound(adj) > budget:
        print("NO")
        return

    # Fast constructive attempts.
    roots = {0, center, find_centroid(adj)}
    best_order = None
    best_cost = 10**30

    for root in roots:
        candidates, _, _ = rooted_orders(root, adj)
        for candidate in candidates:
            candidate, cost = improve_order(candidate, adj)
            if cost < best_cost:
                best_cost = cost
                best_order = candidate[:]
            if cost <= budget:
                output_order(candidate, n)

    adj_masks = [0] * n
    for v in range(n):
        mask = 0
        for u in adj[v]:
            mask |= 1 << u
        adj_masks[v] = mask

    if n <= 24:
        order = exact_subset_dp(n, adj_masks, degrees, budget)
    else:
        order = sparse_exact_dp(n, adj_masks, degrees, budget)

    if order is None:
        print("NO")
    else:
        output_order(order, n)

if __name__ == "__main__":
    main()
