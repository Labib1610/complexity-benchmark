import sys, itertools

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [(int(next(it))-1, int(next(it))-1) for _ in range(m)]
    # simple brute-force for small n
    if n > 10:
        # fallback heuristic: place nodes by degree descending in middle
        deg = [0]*n
        for u,v in edges:
            deg[u]+=1; deg[v]+=1
        order = sorted(range(n), key=lambda x:-deg[x])
        pos = [0]*n
        left = (n+1)//2
        right = left+1
        for i, node in enumerate(order):
            if i%2==0:
                pos[node]=left
                left-=1
            else:
                pos[node]=right
                right+=1
        total = sum(abs(pos[u]-pos[v]) for u,v in edges)
        if total<=B:
            print("YES")
            print(' '.join(str(p) for p in pos))
        else:
            print("NO")
        return

    nodes = list(range(n))
    for perm in itertools.permutations(range(1, n+1)):
        pos = [0]*n
        for i, p in enumerate(perm):
            pos[i]=p
        total = 0
        for u,v in edges:
            total += abs(pos[u]-pos[v])
            if total > B:
                break
        if total <= B:
            print("YES")
            print(' '.join(map(str, pos)))
            return
    print("NO")

if __name__ == "__main__":
    solve()
