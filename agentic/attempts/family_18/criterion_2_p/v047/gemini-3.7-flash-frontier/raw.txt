```python
import sys
import time

def solve():
    sys.setrecursionlimit(300000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    if n == 1:
        if B >= 0:
            print("YES")
            print("1")
        else:
            print("NO")
        return

    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)

    # Subtree sizes and ordering generator
    sz = [0] * (n + 1)
    def get_sizes(u, p):
        s = 1
        for v in adj[u]:
            if v != p:
                s += get_sizes(v, u)
        sz[u] = s
        return s

    def build_order(u, p, direction):
        # direction: 0 = root (spread both sides), +1 = u at left, -1 = u at right
        children = [v for v in adj[u] if v != p]
        children.sort(key=lambda v: sz[v])

        if direction == 0:
            left_children = children[0::2]
            right_children = children[1::2]
            
            res = []
            for v in reversed(left_children):
                res.extend(build_order(v, u, -1))
            res.append(u)
            for v in right_children:
                res.extend(build_order(v, u, 1))
            return res
        elif direction == 1:
            res = [u]
            for v in children:
                res.extend(build_order(v, u, 1))
            return res
        else:
            res = []
            for v in children:
                res.extend(build_order(v, u, -1))
            res.append(u)
            return res

    def compute_cost(pos):
        cost = 0
        for u in range(1, n + 1):
            pu = pos[u]
            for v in adj[u]:
                if u < v:
                    cost += abs(pu - pos[v])
        return cost

    # Find centroids
    get_sizes(1, 0)
    centroids = []
    def find_centroids(u, p):
        is_centroid = True
        for v in adj[u]:
            if v != p:
                find_centroids(v, u)
                if sz[v] > n // 2:
                    is_centroid = False
        if n - sz[u] > n // 2:
            is_centroid = False
        if is_centroid:
            centroids.append(u)

    find_centroids(1, 0)

    # Candidate roots
    candidates = list(centroids)
    if len(candidates) < 5:
        # Add high degree nodes
        nodes_by_deg = sorted(range(1, n + 1), key=lambda x: len(adj[x]), reverse=True)
        for u in nodes_by_deg:
            if u not in candidates:
                candidates.append(u)
            if len(candidates) >= 15:
                break

    best_cost = float('inf')
    best_order = None

    for root in candidates:
        get_sizes(root, 0)
        order = build_order(root, 0, 0)
        pos = [0] * (n + 1)
        for i, u in enumerate(order):
            pos[u] = i + 1
        c = compute_cost(pos)
        if c < best_cost:
            best_cost = c
            best_order = order

    # Optimization: Median heuristic and Local search
    order = list(best_order)
    pos = [0] * (n + 1)
    for i, u in enumerate(order):
        pos[u] = i + 1

    start_time = time.time()

    def median_heuristic(order, pos):
        medians = []
        for u in range(1, n + 1):
            n_pos = sorted([pos[v] for v in adj[u]])
            k = len(n_pos)
            if k % 2 == 1:
                med = n_pos[k // 2]
            else:
                med = (n_pos[k // 2 - 1] + n_pos[k // 2]) / 2.0
            medians.append((med, pos[u], u))
        medians.sort()
        new_order = [u for _, _, u in medians]
        new_pos = [0] * (n + 1)
        for i, u in enumerate(new_order):
            new_pos[u] = i + 1
        return new_order, new_pos

    # Run median heuristic a few times
    for _ in range(5):
        if best_cost <= B:
            break
        new_order, new_pos = median_heuristic(order, pos)
        c = compute_cost(new_pos)
        if c < best_cost:
            best_cost = c
            order = new_order
            pos = new_pos
        else:
            order = new_order
            pos = new_pos

    # Adjacent swaps hill climbing
    improved = True
    iterations = 0
    while improved and iterations < 50:
        if best_cost <= B:
            break
        improved = False
        iterations += 1
        for i in range(n - 1):
            if time.time() - start_time > 4.0:
                break
            u = order[i]
            v = order[i + 1]
            # Calculate delta if we swap order[i] (pos i+1) and order[i+1] (pos i+2)
            pos_u = i + 1
            pos_v = i + 2
            
            delta = 0
            for w in adj[u]:
                if w == v:
                    continue
                pw = pos[w]
                # old: |pos_u - pw|, new: |pos_v - pw|
                if pw < pos_u:
                    delta += 1
                elif pw > pos_v:
                    delta -= 1

            for w in adj[v]:
                if w == u:
                    continue
                pw = pos[w]
                # old: |pos_v - pw|, new: |pos_u - pw|
                if pw < pos_u:
                    delta -= 1
                elif pw > pos_v:
                    delta += 1

            if delta < 0:
                order[i], order[i + 1] = v, u
                pos[u] = pos_v
                pos[v] = pos_u
                best_cost += delta
                improved = True

    if best_cost <= B:
        print("YES")
        ans = [pos[i] for i in range(1, n + 1)]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
