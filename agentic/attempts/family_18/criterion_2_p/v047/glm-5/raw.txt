import sys
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    stack = [1]
    visited[1] = True
    color[1] = 0
    
    while stack:
        u = stack.pop()
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                color[v] = 1 - color[u]
                stack.append(v)
    
    set0 = [u for u in range(1, n + 1) if color[u] == 0]
    set1 = [u for u in range(1, n + 1) if color[u] == 1]
    
    cnt0 = len(set0)
    cnt1 = len(set1)
    
    ans = [0] * (n + 1)
    
    # Strategy: pair length-1 edges by interleaving sets
    # We match the first min(cnt0, cnt1) elements of each set
    # and place them at positions 1,2, ..., 2*min(cnt0,cnt1)
    # with values 1,2, ..., 2*min(cnt0,cnt1)
    
    pair_len = min(cnt0, cnt1)
    pos0_head = 1
    pos1_head = 2
    for i in range(pair_len):
        ans[set0[i]] = pos0_head
        pos0_head += 2
        ans[set1[i]] = pos1_head
        pos1_head += 2
    
    # Remaining elements (only one set has them)
    # Place at positions 2*pair_len + 1 ... n with values same order
    rem_start = 2 * pair_len + 1
    val = rem_start
    
    if cnt0 > pair_len:
        for i in range(pair_len, cnt0):
            ans[set0[i]] = val
            val += 1
    elif cnt1 > pair_len:
        for i in range(pair_len, cnt1):
            ans[set1[i]] = val
            val += 1
    
    total_length = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:
                total_length += abs(ans[u] - ans[v])
    
    # Each edge counted twice, divide by 2
    total_length //= 2
    
    if total_length <= B:
        print("YES")
        print(' '.join(map(str, ans[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
