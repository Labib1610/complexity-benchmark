```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, B = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        edges.append((u, v))
    
    # The graph is a tree since it's connected and m = n - 1 with no cycles
    
    # Since the tree can be bipartitioned (property 1), it's a bipartite graph.
    # We can 2-color the tree.
    
    color = [-1] * (n + 1)
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0
        while queue:
            node = queue.popleft()
            for neighbor in adj[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - color[node]
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    # Should not happen in a bipartite graph
                    return False
        return True
    
    # Find all components and color them
    for i in range(1, n + 1):
        if color[i] == -1:
            if not bfs(i):
                # This shouldn't happen according to problem statement
                print("NO")
                return
    
    # Count nodes in each color class
    count_0 = sum(1 for i in range(1, n + 1) if color[i] == 0)
    count_1 = n - count_0
    
    # Try to minimize the total edge length
    # In a path layout, total length is minimized when we place nodes optimally.
    # But we have a tree structure.
    
    # Strategy:
    # Place all nodes of one color set on the left side and the other on the right side
    # to minimize the sum of distances.
    
    # Let's sort the positions: 1, 2, ..., n
    # If we place k nodes of color 0 and (n-k) nodes of color 1,
    # we want to minimize sum of |pos_u - pos_v| over all edges.
    
    # A good heuristic is to place nodes of one color consecutively on the left
    # and the other color consecutively on the right.
    
    # Let's assign positions:
    # Color 0 gets positions 1 .. count_0
    # Color 1 gets positions count_0+1 .. n
    
    # Check if this satisfies the budget B
    
    # Compute total cost for this assignment
    total_cost = 0
    # For each edge (u,v), compute distance between their assigned positions
    
    # Assign positions based on color
    position = [0] * (n + 1)
    idx_0 = 1
    idx_1 = count_0 + 1
    
    for i in range(1, n + 1):
        if color[i] == 0:
            position[i] = idx_0
            idx_0 += 1
        else:
            position[i] = idx_1
            idx_1 += 1
    
    # Calculate total cost
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
    
    if total_cost <= B:
        print("YES")
        result = [str(position[i]) for i in range(1, n + 1)]
        print(' '.join(result))
        return
    
    # Otherwise, try the reverse assignment
    # Swap colors
    position = [0] * (n + 1)
    idx_0 = 1
    idx_1 = count_1 + 1
    
    for i in range(1, n + 1):
        if color[i] == 1:  # previous color 1 now gets first positions
            position[i] = idx_0
            idx_0 += 1
        else:
            position[i] = idx_1
            idx_1 += 1
            
    # Recalculate total cost
    total_cost = 0
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
        
    if total_cost <= B:
        print("YES")
        result = [str(position[i]) for i in range(1, n + 1)]
        print(' '.join(result))
        return
        
    # If neither works, let's try placing nodes optimally on a line
    # This is essentially finding a layout of a tree on a line minimizing sum of edge lengths
    
    # This is NP-hard in general, but we can use BFS/DFS ordering or centroid decomposition
    
    # Let's do DFS traversal to get an ordering and place nodes accordingly
    
    visited = [False] * (n + 1)
    order = []
    
    def dfs(u):
        visited[u] = True
        order.append(u)
        for v in adj[u]:
            if not visited[v]:
                dfs(v)
                
    # Start DFS from node 1
    dfs(1)
    
    # Now assign positions 1 to n to nodes in order
    position = [0] * (n + 1)
    for i in range(n):
        position[order[i]] = i + 1
        
    # Calculate total cost
    total_cost = 0
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
        
    if total_cost <= B:
        print("YES")
        result = [str(position[i]) for i in range(1, n + 1)]
        print(' '.join(result))
        return
        
    # Last resort: check if even optimal linear arrangement exceeds B
    # For trees, there exist polynomial algorithms, but let's just try another simple method
    
    # Try placing nodes by doing BFS to get an ordering
    visited = [False] * (n + 1)
    order = []
    queue = deque()
    queue.append(1)
    visited[1] = True
    order.append(1)
    
    front = 0
    while front < len(queue):
        u = queue[front]
        front += 1
        for v in adj[u]:
            if not visited[v]:
                visited[v] = True
                queue.append(v)
                order.append(v)
                
    # Assign positions
    position = [0] * (n + 1)
    for i in range(n):
        position[order[i]] = i + 1
        
    # Calculate total cost
    total_cost = 0
    for u, v in edges:
        total_cost += abs(position[u] - position[v])
        
    if total_cost <= B:
        print("YES")
        result = [str(position[i]) for i in range(1, n + 1)]
        print(' '.join(result))
        return
        
    print("NO")

if __name__ == "__main__":
    main()
```
