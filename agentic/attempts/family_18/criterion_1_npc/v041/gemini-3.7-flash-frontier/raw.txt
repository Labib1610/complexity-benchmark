import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])
    
    adj_mask = [0] * n
    deg = [0] * n
    
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj_mask[u] |= (1 << v)
            adj_mask[v] |= (1 << u)
            deg[u] += 1
            deg[v] += 1

    size = 1 << n
    cut = [0] * size
    for mask in range(1, size):
        v = (mask & -mask).bit_length() - 1
        prev = mask ^ (1 << v)
        cut[mask] = cut[prev] + deg[v] - 2 * (adj_mask[v] & prev).bit_count()

    dp = [float('inf')] * size
    parent = [-1] * size
    dp[0] = 0

    for mask in range(size):
        c_dp = dp[mask]
        if c_dp == float('inf'):
            continue
        # Try adding any unused vertex
        rem = ((1 << n) - 1) ^ mask
        while rem:
            v_bit = rem & -rem
            v = v_bit.bit_length() - 1
            nxt = mask | v_bit
            new_cost = c_dp + cut[nxt]
            if new_cost < dp[nxt]:
                dp[nxt] = new_cost
                parent[nxt] = v
            rem ^= v_bit

    if dp[-1] <= B:
        print("YES")
        # Reconstruct order of artifacts placed in bays 1, 2, ..., n
        order = []
        curr = size - 1
        while curr > 0:
            v = parent[curr]
            order.append(v)
            curr ^= (1 << v)
        order.reverse()
        
        # order[k] is the artifact placed at bay (k + 1)
        # We need output where i-th integer is the bay index assigned to artifact i
        ans = [0] * n
        for bay_idx, art_idx in enumerate(order, 1):
            ans[art_idx] = bay_idx
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
