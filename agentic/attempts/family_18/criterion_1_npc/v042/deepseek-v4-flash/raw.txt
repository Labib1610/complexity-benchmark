We need to assign each reaction a distinct step from 1 to n, minimizing the sum of distances for given pairs.  

This is equivalent to finding a permutation of vertices that minimizes the sum of absolute differences for given edges.  
A known optimal arrangement for minimizing sum of absolute differences is to place connected vertices as close as possible — often achieved by ordering vertices by their degree or using a BFS/DFS layout.  

A simple heuristic:  
- Put the most connected vertex in the middle.  
- Then place its neighbors around it, then their neighbors, etc.  

We can try a greedy BFS ordering starting from the highest-degree vertex, assigning steps from the center outward.  

Let’s implement that and check if the total cost ≤ B.  

```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Greedy BFS ordering from highest degree vertex
    deg = [len(adj[i]) for i in range(n)]
    start = max(range(n), key=lambda i: deg[i])
    
    order = []
    visited = [False] * n
    q = deque([start])
    visited[start] = True
    while q:
        u = q.popleft()
        order.append(u)
        # sort neighbors by degree descending for better locality
        for v in sorted(adj[u], key=lambda x: -deg[x]):
            if not visited[v]:
                visited[v] = True
                q.append(v)
    
    # Assign steps: place first in middle, then alternate left/right
    pos = [0] * n
    left = (n - 1) // 2
    right = left + 1
    for i, node in enumerate(order):
        if i % 2 == 0:
            pos[node] = left
            left -= 1
        else:
            pos[node] = right
            right += 1
    
    # Compute total cost
    total = 0
    for u, v in edges:
        total += abs(pos[u] - pos[v])
    
    if total <= B:
        print("YES")
        print(" ".join(str(pos[i] + 1) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
