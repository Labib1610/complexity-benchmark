```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    if n == 0:
        return

    # 0-indexed vertices
    adj_matrix = [[0] * n for _ in range(n)]
    deg = [0] * n
    adj_mask = [0] * n

    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj_matrix[u][v] += 1
            adj_matrix[v][u] += 1
            deg[u] += 1
            deg[v] += 1

    for i in range(n):
        mask = 0
        for j in range(n):
            if adj_matrix[i][j] > 0:
                mask |= 1 << j
        adj_mask[i] = mask

    # DP over subsets
    # dp[mask] = min cost to arrange vertices in mask
    # cut[mask] = number of edges between mask and V \ mask
    num_states = 1 << n
    INF = 10**15
    dp = [INF] * num_states
    parent = [-1] * num_states
    parent_vertex = [-1] * num_states
    cut = [0] * num_states

    dp[0] = 0
    cut[0] = 0

    for mask in range(num_states):
        d = dp[mask]
        if d > B:
            continue
        c = cut[mask]

        # Try adding each unused vertex
        rem = ((1 << n) - 1) ^ mask
        while rem:
            u = (rem & -rem).bit_length() - 1
            rem ^= 1 << u

            nmask = mask | (1 << u)

            # Compute cut for nmask if not already set or compute on the fly
            # cut[nmask] = cut[mask] + deg[u] - 2 * (edges between u and mask)
            edges_u_mask = 0
            for v in range(n):
                if (mask & (1 << v)) and adj_matrix[u][v]:
                    edges_u_mask += adj_matrix[u][v]

            ncut = c + deg[u] - 2 * edges_u_mask
            cut[nmask] = ncut

            new_cost = d + c
            if new_cost < dp[nmask]:
                dp[nmask] = new_cost
                parent[nmask] = mask
                parent_vertex[nmask] = u

    target = (1 << n) - 1
    if dp[target] <= B:
        print("YES")
        order = []
        curr = target
        while curr > 0:
            u = parent_vertex[curr]
            order.append(u)
            curr = parent[curr]
        order.reverse()

        # order[step - 1] = vertex
        # we need step for each vertex (1-indexed)
        pos = [0] * n
        for step, u in enumerate(order, 1):
            pos[u] = step
        print(*(pos[i] for i in range(n)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
