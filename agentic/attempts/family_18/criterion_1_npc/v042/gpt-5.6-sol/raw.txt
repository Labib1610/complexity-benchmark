import sys
from array import array
from collections import deque


def positions_from_order(order, n):
    pos = [0] * n
    for i, v in enumerate(order, 1):
        pos[v] = i
    return pos


def exact_component(vertices, graph, limit):
    k = len(vertices)
    if k <= 1:
        return vertices[:], 0

    index = {v: i for i, v in enumerate(vertices)}
    adj = [[] for _ in range(k)]
    adj_mask = [0] * k
    simple = True

    for v in vertices:
        i = index[v]
        for to, w in graph[v].items():
            j = index[to]
            adj[i].append((j, w))
            adj_mask[i] |= 1 << j
            if w != 1:
                simple = False

    full = (1 << k) - 1
    inf = limit + 1

    if inf <= 0xFF:
        typecode = 'B'
    elif inf <= 0xFFFF:
        typecode = 'H'
    elif inf <= 0xFFFFFFFF:
        typecode = 'I'
    else:
        typecode = 'Q'

    dp = array(typecode, [inf]) * (full + 1)
    dp[0] = 0

    for mask in range(1, full + 1):
        best = inf
        bits = mask
        while bits:
            bit = bits & -bits
            prev = mask ^ bit
            value = dp[prev]
            if value < best:
                best = value
            bits ^= bit

        if best >= inf:
            continue

        if mask == full:
            cut = 0
        elif simple:
            outside = full ^ mask
            cut = 0
            bits = mask
            while bits:
                bit = bits & -bits
                i = bit.bit_length() - 1
                cut += (adj_mask[i] & outside).bit_count()
                if cut >= inf:
                    break
                bits ^= bit
        else:
            outside = full ^ mask
            cut = 0
            bits = mask
            while bits:
                bit = bits & -bits
                i = bit.bit_length() - 1
                for j, w in adj[i]:
                    if outside & (1 << j):
                        cut += w
                if cut >= inf:
                    break
                bits ^= bit

        value = best + cut
        if value < inf:
            dp[mask] = value

    optimum = dp[full]
    if optimum >= inf:
        return None, optimum

    reverse_order = []
    mask = full
    while mask:
        add = 0
        if mask != full:
            if simple:
                outside = full ^ mask
                bits = mask
                while bits:
                    bit = bits & -bits
                    i = bit.bit_length() - 1
                    add += (adj_mask[i] & outside).bit_count()
                    bits ^= bit
            else:
                outside = full ^ mask
                bits = mask
                while bits:
                    bit = bits & -bits
                    i = bit.bit_length() - 1
                    for j, w in adj[i]:
                        if outside & (1 << j):
                            add += w
                    bits ^= bit

        bits = mask
        chosen = -1
        while bits:
            bit = bits & -bits
            prev = mask ^ bit
            if dp[prev] < inf and dp[prev] + add == dp[mask]:
                chosen = bit.bit_length() - 1
                break
            bits ^= bit

        if chosen < 0:
            return None, inf

        reverse_order.append(chosen)
        mask ^= 1 << chosen

    reverse_order.reverse()
    return [vertices[i] for i in reverse_order], optimum


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 3:
        return

    n, m, budget = data[0], data[1], data[2]
    graph = [dict() for _ in range(n)]
    edges = {}
    nonloop_count = 0

    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2

        if u == v:
            continue

        if u > v:
            u, v = v, u
        key = (u, v)
        edges[key] = edges.get(key, 0) + 1
        nonloop_count += 1

    if nonloop_count > budget:
        print("NO")
        return

    for (u, v), w in edges.items():
        graph[u][v] = w
        graph[v][u] = w

    if not edges:
        print("YES")
        print(*range(1, n + 1))
        return

    components = []
    isolated = []
    seen = [False] * n

    for start in range(n):
        if not graph[start]:
            isolated.append(start)
            seen[start] = True
            continue
        if seen[start]:
            continue

        stack = [start]
        seen[start] = True
        component = []

        while stack:
            v = stack.pop()
            component.append(v)
            for to in graph[v]:
                if not seen[to]:
                    seen[to] = True
                    stack.append(to)

        components.append(component)

    lower_bound = 0
    for comp in components:
        edge_weight = 0
        comp_set = set(comp)
        for v in comp:
            for to, w in graph[v].items():
                if v < to and to in comp_set:
                    edge_weight += w
        lower_bound += max(len(comp) - 1, edge_weight)

    if lower_bound > budget:
        print("NO")
        return

    # Fast candidate: original numbering.
    identity_order = list(range(n))
    identity_cost = sum(w * abs(u - v) for (u, v), w in edges.items())
    if identity_cost <= budget:
        pos = positions_from_order(identity_order, n)
        print("YES")
        print(*(x for x in pos))
        return

    # Fast candidate: Cuthill-McKee-style BFS in every component.
    bfs_order = []
    for comp in components:
        start = min(comp, key=lambda v: (len(graph[v]), v))
        used = {start}
        q = deque([start])
        while q:
            v = q.popleft()
            bfs_order.append(v)
            neighbors = [to for to in graph[v] if to not in used]
            neighbors.sort(key=lambda x: (len(graph[x]), x))
            for to in neighbors:
                used.add(to)
                q.append(to)
    bfs_order.extend(isolated)

    bfs_pos = positions_from_order(bfs_order, n)
    bfs_cost = sum(w * abs(bfs_pos[u] - bfs_pos[v]) for (u, v), w in edges.items())
    if bfs_cost <= budget:
        print("YES")
        print(*bfs_pos)
        return

    remaining = budget
    final_order = []

    components.sort(key=len, reverse=True)

    for comp in components:
        order, cost = exact_component(comp, graph, remaining)
        if order is None or cost > remaining:
            print("NO")
            return
        final_order.extend(order)
        remaining -= cost

    final_order.extend(isolated)
    pos = positions_from_order(final_order, n)

    total = sum(w * abs(pos[u] - pos[v]) for (u, v), w in edges.items())
    if total > budget:
        print("NO")
        return

    print("YES")
    print(*pos)


if __name__ == "__main__":
    main()
