```python
import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    for i in range(m):
        u = int(data[3 + 2*i]) - 1  # Convert to 0-based indexing
        v = int(data[4 + 2*i]) - 1
        edges.append((u, v))
    
    # Try all permutations of zone assignments (positions 0 to n-1)
    for perm in permutations(range(n)):
        total_length = 0
        for u, v in edges:
            pos_u = perm[u]
            pos_v = perm[v]
            total_length += abs(pos_u - pos_v)
            if total_length > B:
                break
        if total_length <= B:
            # Found a valid assignment
            # Convert back to 1-based zone numbering
            assignment = [str(perm[i] + 1) for i in range(n)]
            print("YES")
            print(" ".join(assignment))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
