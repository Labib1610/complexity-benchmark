import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
    
    # Greedy smallest-last ordering
    remaining_degree = degree[:]
    removed = [False] * (n + 1)
    order = []
    
    for _ in range(n):
        # Find vertex with minimum degree among remaining
        min_deg = n + 1
        best_v = -1
        for v in range(1, n + 1):
            if not removed[v] and remaining_degree[v] < min_deg:
                min_deg = remaining_degree[v]
                best_v = v
        v = best_v
        order.append(v)
        removed[v] = True
        for u in adj[v]:
            if not removed[u]:
                remaining_degree[u] -= 1
    
    # Assign positions: first in order gets earliest available position
    order_rev = order[::-1]
    position = [0] * (n + 1)
    available_pos = list(range(1, n + 1))
    for v in order_rev:
        position[v] = available_pos.pop(0)
    
    # Compute total pipe length
    total = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if u < v:
                total += abs(position[u] - position[v])
    
    if total <= B:
        print("YES")
        print(' '.join(map(str, position[1:])))
    else:
        print("NO")

solve()
