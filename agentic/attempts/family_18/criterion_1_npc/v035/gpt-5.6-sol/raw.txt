import sys
from array import array

sys.setrecursionlimit(1_000_000)


def arrangement_cost(order, edges):
    pos = [0] * len(order)
    for i, v in enumerate(order):
        pos[v] = i
    return sum(w * abs(pos[u] - pos[v]) for u, v, w in edges)


def improve_order(order, adj, edges):
    k = len(order)
    if k <= 2:
        return order, arrangement_cost(order, edges)

    pos = [0] * k
    for i, v in enumerate(order):
        pos[v] = i

    for _ in range(8):
        changed = False
        i = 0
        while i + 1 < k:
            a, b = order[i], order[i + 1]
            delta = 0
            for x, w in adj[a]:
                delta += w * (abs(i + 1 - pos[x]) - abs(i - pos[x]))
            for x, w in adj[b]:
                delta += w * (abs(i - pos[x]) - abs(i + 1 - pos[x]))
            if delta < 0:
                order[i], order[i + 1] = b, a
                pos[a], pos[b] = i + 1, i
                changed = True
                if i:
                    i -= 1
                    continue
            i += 1
        if not changed:
            break

    return order, arrangement_cost(order, edges)


def heuristic_order(k, adj, edges):
    if k <= 1:
        return list(range(k)), 0

    degree = [sum(w for _, w in adj[v]) for v in range(k)]
    starts = [
        list(range(k)),
        sorted(range(k), key=lambda v: (-degree[v], v)),
        sorted(range(k), key=lambda v: (degree[v], v)),
    ]

    # BFS-like order from a high-degree vertex.
    root = max(range(k), key=lambda v: (degree[v], -v))
    seen = [False] * k
    seen[root] = True
    q = [root]
    bfs = []
    head = 0
    while head < len(q):
        v = q[head]
        head += 1
        bfs.append(v)
        ns = sorted((x for x, _ in adj[v] if not seen[x]),
                    key=lambda x: (-degree[x], x))
        for x in ns:
            if not seen[x]:
                seen[x] = True
                q.append(x)
    starts.append(bfs)

    best_order = starts[0]
    best_cost = arrangement_cost(best_order, edges)

    for initial in starts:
        order = initial[:]
        for _ in range(8):
            pos = [0] * k
            for i, v in enumerate(order):
                pos[v] = i

            keys = []
            for v in range(k):
                den = degree[v]
                if den:
                    num = sum(w * pos[x] for x, w in adj[v])
                    keys.append((num / den, pos[v], v))
                else:
                    keys.append((pos[v], pos[v], v))
            new_order = [v for _, _, v in sorted(keys)]
            if new_order == order:
                break
            order = new_order

        order, cost = improve_order(order, adj, edges)
        if cost < best_cost:
            best_cost = cost
            best_order = order[:]

    return best_order, best_cost


def special_order(k, adj, edges):
    if k <= 1:
        return list(range(k)), 0

    structural_degree = [len(adj[v]) for v in range(k)]

    # A weighted path: placing vertices in path order attains the edgewise lower bound.
    if len(edges) == k - 1 and max(structural_degree) <= 2:
        start = next(v for v in range(k) if structural_degree[v] == 1)
        order = []
        prev = -1
        cur = start
        while cur != -1:
            order.append(cur)
            nxt = -1
            for x, _ in adj[cur]:
                if x != prev:
                    nxt = x
                    break
            prev, cur = cur, nxt
        return order, sum(w for _, _, w in edges)

    # Weighted star.
    centers = [v for v in range(k) if structural_degree[v] == k - 1]
    if len(edges) == k - 1 and centers:
        c = centers[0]
        leaves = [(w, v) for v, w in adj[c]]
        leaves.sort(reverse=True)

        best_cost = None
        best_order = None
        for cp in range(k):
            slots = [p for p in range(k) if p != cp]
            slots.sort(key=lambda p: (abs(p - cp), p))
            order = [-1] * k
            order[cp] = c
            cost = 0
            for (w, v), p in zip(leaves, slots):
                order[p] = v
                cost += w * abs(p - cp)
            if best_cost is None or cost < best_cost:
                best_cost = cost
                best_order = order
        return best_order, best_cost

    return None


def exact_subset_dp(k, adj, heuristic_cost):
    size = 1 << k
    max_u32 = (1 << 32) - 1

    cut_type = 'I' if sum(sum(w for _, w in adj[v]) for v in range(k)) // 2 <= max_u32 else 'Q'
    dp_type = 'I' if heuristic_cost + 1 <= max_u32 else 'Q'
    inf = heuristic_cost + 1

    cuts = array(cut_type, [0]) * size
    dp = array(dp_type, [inf]) * size
    dp[0] = 0
    degrees = [sum(w for _, w in adj[v]) for v in range(k)]

    for mask in range(1, size):
        bit = mask & -mask
        v = bit.bit_length() - 1
        prev = mask ^ bit

        inside = 0
        for x, w in adj[v]:
            if prev & (1 << x):
                inside += w
        c = cuts[prev] + degrees[v] - 2 * inside
        cuts[mask] = c

        best = inf
        bits = mask
        while bits:
            b = bits & -bits
            val = dp[mask ^ b]
            if val < best:
                best = val
            bits ^= b

        val = best + c
        dp[mask] = val if val <= heuristic_cost else inf

    mask = size - 1
    optimum = dp[mask]
    reverse_order = []
    while mask:
        c = cuts[mask]
        bits = mask
        chosen = None
        while bits:
            b = bits & -bits
            if dp[mask ^ b] + c == dp[mask]:
                chosen = b
                break
            bits ^= b
        if chosen is None:
            # This can only happen if all values were saturated, while the known
            # heuristic arrangement itself is optimal at the saturation boundary.
            return None
        reverse_order.append(chosen.bit_length() - 1)
        mask ^= chosen

    reverse_order.reverse()
    return reverse_order, optimum


def branch_and_bound(k, adj, edges, initial_order, initial_cost):
    best_cost = initial_cost
    best_order = initial_order[:]
    full = (1 << k) - 1
    memo = {}
    prefix = []

    def lower_bound(mask, cost):
        remaining = [v for v in range(k) if not (mask >> v) & 1]
        if not remaining:
            return cost

        crossing = []
        internal_twice = 0
        for v in remaining:
            cv = 0
            for x, w in adj[v]:
                if (mask >> x) & 1:
                    cv += w
                elif not (mask >> x) & 1:
                    internal_twice += w
            crossing.append(cv)

        crossing.sort(reverse=True)
        extra = internal_twice // 2
        for rank, value in enumerate(crossing):
            extra += rank * value
        return cost + extra

    def dfs(mask, cost, cut):
        nonlocal best_cost, best_order

        if cost >= best_cost:
            return
        old = memo.get(mask)
        if old is not None and old <= cost:
            return
        memo[mask] = cost

        if mask == full:
            best_cost = cost
            best_order = prefix[:]
            return

        if lower_bound(mask, cost) >= best_cost:
            return

        candidates = []
        remaining = full ^ mask
        bits = remaining
        while bits:
            b = bits & -bits
            v = b.bit_length() - 1
            to_inside = 0
            for x, w in adj[v]:
                if mask & (1 << x):
                    to_inside += w
            new_cut = cut + sum(w for _, w in adj[v]) - 2 * to_inside
            candidates.append((new_cut, -to_inside, v, b))
            bits ^= b

        candidates.sort()
        for new_cut, _, v, b in candidates:
            new_mask = mask | b
            new_cost = cost + (0 if new_mask == full else new_cut)
            if new_cost >= best_cost:
                continue
            prefix.append(v)
            dfs(new_mask, new_cost, new_cut)
            prefix.pop()

    dfs(0, 0, 0)
    return best_order, best_cost


def solve_component(vertices, edge_records):
    k = len(vertices)
    if k <= 1:
        return vertices[:], 0

    index = {v: i for i, v in enumerate(vertices)}
    weight_map = {}
    for u, v, w in edge_records:
        a, b = index[u], index[v]
        if a > b:
            a, b = b, a
        weight_map[(a, b)] = weight_map.get((a, b), 0) + w

    edges = [(a, b, w) for (a, b), w in weight_map.items()]
    adj = [[] for _ in range(k)]
    for a, b, w in edges:
        adj[a].append((b, w))
        adj[b].append((a, w))

    special = special_order(k, adj, edges)
    if special is not None:
        local_order, cost = special
    else:
        local_order, cost = heuristic_order(k, adj, edges)

        if k <= 23:
            exact = exact_subset_dp(k, adj, cost)
            if exact is not None:
                local_order, cost = exact
        else:
            local_order, cost = branch_and_bound(
                k, adj, edges, local_order, cost
            )

    return [vertices[x] for x in local_order], cost


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, budget = data[0], data[1], data[2]
    raw = data[3:]

    pair_weights = {}
    graph = [[] for _ in range(n)]

    for i in range(m):
        u = raw[2 * i] - 1
        v = raw[2 * i + 1] - 1
        if u == v:
            continue
        if u > v:
            u, v = v, u
        pair_weights[(u, v)] = pair_weights.get((u, v), 0) + 1

    weighted_edges = [(u, v, w) for (u, v), w in pair_weights.items()]
    for u, v, _ in weighted_edges:
        graph[u].append(v)
        graph[v].append(u)

    component_id = [-1] * n
    components = []

    for start in range(n):
        if component_id[start] != -1:
            continue
        cid = len(components)
        stack = [start]
        component_id[start] = cid
        vertices = []
        while stack:
            v = stack.pop()
            vertices.append(v)
            for x in graph[v]:
                if component_id[x] == -1:
                    component_id[x] = cid
                    stack.append(x)
        components.append(vertices)

    component_edges = [[] for _ in components]
    for u, v, w in weighted_edges:
        component_edges[component_id[u]].append((u, v, w))

    total_cost = 0
    global_order = []

    items = list(range(len(components)))
    items.sort(key=lambda cid: len(components[cid]))

    solved = {}
    for cid in items:
        order, cost = solve_component(components[cid], component_edges[cid])
        solved[cid] = order
        total_cost += cost
        if total_cost > budget:
            # Continue: later components may still have zero cost, but no solved
            # component can be improved because each is solved optimally.
            pass

    if total_cost > budget:
        print("NO")
        return

    for cid in range(len(components)):
        global_order.extend(solved[cid])

    position = [0] * n
    for p, v in enumerate(global_order, 1):
        position[v] = p

    print("YES")
    print(*position)


if __name__ == "__main__":
    main()
