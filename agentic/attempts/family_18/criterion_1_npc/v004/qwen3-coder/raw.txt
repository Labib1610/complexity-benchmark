from itertools import permutations

                total += abs(pos[u] - pos[v])
                if total > B:
                    break

        deg = [0] * n

            adj[u].append(v)
            adj[v].append(u)
        
        # Sort modules by degree descending
        modules_sorted = sorted(range(n), key=lambda x: deg[x], reverse=True)

        pos = [0] * n
        assigned = [False] * n
        placement = [-1] * n

        queue = [modules_sorted[0]]
        assigned[modules_sorted[0]] = True
        placement[n//2] = modules_sorted[0]
        pos[modules_sorted[0]] = n//2
        
        # BFS-like assignment for neighbors
        placed = 1
        current_positions = [n//2]
        next_queue = []
        
        for u in queue:
            for v in adj[u]:
                if not assigned[v]:
                    next_queue.append(v)
                    assigned[v] = True
            if not next_queue:
                next_queue = [i for i in range(n) if not assigned[i]]
        
        # Assign remaining nodes greedily
        available_positions = [i for i in range(n) if placement[i] == -1]
        unassigned_modules = [i for i in range(n) if not assigned[i]]
        
        import heapq
        for mod in unassigned_modules:
            best_pos = -1
            best_cost = float('inf')
            for p in available_positions:
                # Estimate cost if we put mod at position p
                cost = 0
                for u,v in edges:
                    if u == mod and placement[pos[v]] != -1:
                        cost += abs(p - pos[v])
                    elif v == mod and placement[pos[u]] != -1:
                        cost += abs(p - pos[u])
                if cost < best_cost:
                    best_cost = cost
                    best_pos = p
            if best_pos != -1:
                placement[best_pos] = mod
                pos[mod] = best_pos
                available_positions.remove(best_pos)
        
        # Check total cost
        total = 0
        for u,v in edges:
            total += abs(pos[u] - pos[v])
