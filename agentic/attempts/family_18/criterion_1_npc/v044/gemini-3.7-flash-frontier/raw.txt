```python
import sys
import time
import random
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])
    
    if m == 0:
        print("YES")
        print(*(i + 1 for i in range(n)))
        return
        
    edges = []
    adj = [[] for _ in range(n)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)

    # If simple theoretical lower bound exceeds B, we can't do it
    # Minimum cost for m edges on n vertices: at least m (if no multigraph, min is m)
    if m > B:
        print("NO")
        return

    # Small n exact DP
    if n <= 18:
        # Bitmask DP
        # cut[mask] = cut between mask and V \ mask
        num_states = 1 << n
        deg = [len(adj[i]) for i in range(n)]
        adj_mask = [0] * n
        for u in range(n):
            for v in adj[u]:
                adj_mask[u] |= (1 << v)

        dp = [float('inf')] * num_states
        parent = [-1] * num_states
        dp[0] = 0
        
        cut = [0] * num_states
        for mask in range(1, num_states):
            # lowest set bit
            lsb = (mask & -mask).bit_length() - 1
            prev_mask = mask ^ (1 << lsb)
            # cut(mask) = cut(prev_mask) + deg[lsb] - 2 * count_edges(lsb, prev_mask)
            edges_to_prev = (adj_mask[lsb] & prev_mask).bit_count()
            cut[mask] = cut[prev_mask] + deg[lsb] - 2 * edges_to_prev

        for mask in range(num_states - 1):
            cur_d = dp[mask]
            if cur_d > B:
                continue
            rem = (~mask) & (num_states - 1)
            while rem:
                v = (rem & -rem).bit_length() - 1
                rem ^= (1 << v)
                nxt_mask = mask | (1 << v)
                nxt_cost = cur_d + cut[nxt_mask]
                if nxt_cost < dp[nxt_mask]:
                    dp[nxt_mask] = nxt_cost
                    parent[nxt_mask] = v

        if dp[num_states - 1] <= B:
            # Reconstruct
            order = []
            curr = num_states - 1
            while curr > 0:
                v = parent[curr]
                order.append(v)
                curr ^= (1 << v)
            order.reverse()
            # stage[var] = 1-based position
            stages = [0] * n
            for pos, var in enumerate(order):
                stages[var] = pos + 1
            print("YES")
            print(*stages)
        else:
            print("NO")
        return

    # For larger n, use multi-restart Simulated Annealing
    start_time = time.time()
    time_limit = 4.5  # seconds

    def compute_cost(perm):
        # perm[pos] = var
        pos = [0] * n
        for p, v in enumerate(perm):
            pos[v] = p
        c = 0
        for u, v in edges:
            c += abs(pos[u] - pos[v])
        return c, pos

    best_cost = float('inf')
    best_perm = None

    # Initial heuristic using degree / BFS ordering
    # BFS order
    visited = [False] * n
    bfs_order = []
    for i in range(n):
        if not visited[i]:
            queue = [i]
            visited[i] = True
            for u in queue:
                bfs_order.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        queue.append(v)
    
    init_orders = [list(range(n)), bfs_order]
    
    # Sort by degree
    deg_order = sorted(range(n), key=lambda x: len(adj[x]), reverse=True)
    # Put high degree in the middle
    mid_order = [0] * n
    l, r = (n - 1) // 2, (n - 1) // 2 + 1
    for i, v in enumerate(deg_order):
        if i % 2 == 0:
            mid_order[l] = v
            l -= 1
        else:
            mid_order[r] = v
            r += 1
    init_orders.append(mid_order)

    # Try initial candidates
    for p in init_orders:
        c, _ = compute_cost(p)
        if c < best_cost:
            best_cost = c
            best_perm = list(p)
        if best_cost <= B:
            break

    # SA Loop
    restart_count = 0
    while time.time() - start_time < time_limit and best_cost > B:
        restart_count += 1
        if restart_count <= len(init_orders):
            perm = list(init_orders[restart_count - 1])
        else:
            perm = list(range(n))
            random.shuffle(perm)
        
        cur_cost, pos = compute_cost(perm)
        if cur_cost < best_cost:
            best_cost = cur_cost
            best_perm = list(perm)
            if best_cost <= B:
                break

        T_start = max(1.0, float(n))
        T_end = 0.01
        steps = 40000
        decay = (T_end / T_start) ** (1.0 / steps)
        T = T_start

        for step in range(steps):
            if step % 500 == 0:
                if time.time() - start_time >= time_limit:
                    break

            # Choose move type: swap two or insert
            # 80% swap, 20% insert
            if random.random() < 0.7:
                # Swap two random positions
                i = random.randint(0, n - 1)
                j = random.randint(0, n - 1)
                if i == j:
                    continue
                u, v = perm[i], perm[j]
                
                # Compute delta
                delta = 0
                for w in adj[u]:
                    if w != v:
                        pw = pos[w]
                        delta += abs(j - pw) - abs(i - pw)
                for w in adj[v]:
                    if w != u:
                        pw = pos[w]
                        delta += abs(i - pw) - abs(j - pw)
                
                if delta <= 0 or math.exp(-delta / T) > random.random():
                    cur_cost += delta
                    perm[i], perm[j] = perm[j], perm[i]
                    pos[u], pos[v] = j, i
                    if cur_cost < best_cost:
                        best_cost = cur_cost
                        best_perm = list(perm)
                        if best_cost <= B:
                            break
            else:
                # Insert move: move element at pos i to pos j
                i = random.randint(0, n - 1)
                j = random.randint(0, n - 1)
                if i == j:
                    continue
                u = perm[i]
                # If we move u from i to j
                # Simulating delta:
                delta = 0
                step_dir = 1 if i < j else -1
                range_min, range_max = min(i, j), max(i, j)
                
                for w in adj[u]:
                    pw = pos[w]
                    # new position of w:
                    new_pw = pw
                    if range_min <= pw <= range_max:
                        if i < j and pw > i:
                            new_pw = pw - 1
                        elif i > j and pw < i:
                            new_pw = pw + 1
                    delta += abs(j - new_pw) - abs(i - pw)
                
                # For vertices between i and j, their pos shifts by -step_dir
                # Check their incident edges to other vertices not u
                # (For speed, full delta for insert is heavier, do approximate or full if small deg)
                # Actually, simpler to just do swap of adjacent elements or standard 2-opt
                # Let's revert to swap or reverse
                pass

            T *= decay

    if best_cost <= B:
        stages = [0] * n
        for p, v in enumerate(best_perm):
            stages[v] = p + 1
        print("YES")
        print(*stages)
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
