import sys
from array import array
from collections import deque

def output_order(order, n):
    position = [0] * n
    for i, v in enumerate(order, 1):
        position[v] = i
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, position)))

def order_cost(order, adjacency):
    pos = {v: i for i, v in enumerate(order)}
    total = 0
    for u in order:
        pu = pos[u]
        for v, w in adjacency[u].items():
            if u < v:
                total += w * abs(pu - pos[v])
    return total

def solve_component_dp(component, adjacency, budget):
    k = len(component)
    if k <= 1:
        return 0, component[:]

    vertices = sorted(component, key=lambda v: (len(adjacency[v]), v))
    index = {v: i for i, v in enumerate(vertices)}

    degree = [0] * k
    masks = [0] * k
    weighted_neighbors = [[] for _ in range(k)]
    unit_weights = True

    for i, u in enumerate(vertices):
        mask = 0
        for v, w in adjacency[u].items():
            if v in index:
                j = index[v]
                degree[i] += w
                mask |= 1 << j
                weighted_neighbors[i].append((j, w))
                if w != 1:
                    unit_weights = False
        masks[i] = mask

    size = 1 << k
    cuts = array('q', [0]) * size

    if unit_weights:
        for s in range(1, size):
            bit = s & -s
            i = bit.bit_length() - 1
            t = s ^ bit
            cuts[s] = cuts[t] + degree[i] - 2 * ((masks[i] & t).bit_count())
    else:
        for s in range(1, size):
            bit = s & -s
            i = bit.bit_length() - 1
            t = s ^ bit
            inside_weight = 0
            for j, w in weighted_neighbors[i]:
                if (t >> j) & 1:
                    inside_weight += w
            cuts[s] = cuts[t] + degree[i] - 2 * inside_weight

    cap = budget + 1
    dp = array('q', [cap]) * size
    dp[0] = 0

    for s in range(1, size):
        best = cap
        bits = s
        while bits:
            bit = bits & -bits
            t = s ^ bit
            value = dp[t] + cuts[t]
            if value < best:
                best = value
            bits ^= bit
        dp[s] = best

    full = size - 1
    minimum = dp[full]
    if minimum > budget:
        return minimum, None

    reversed_order = []
    s = full
    while s:
        bits = s
        chosen = -1
        while bits:
            bit = bits & -bits
            t = s ^ bit
            if dp[t] + cuts[t] == dp[s]:
                chosen = bit.bit_length() - 1
                s = t
                break
            bits ^= bit
        if chosen < 0:
            return cap, None
        reversed_order.append(vertices[chosen])

    reversed_order.reverse()
    return minimum, reversed_order

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 3:
        return

    n, m, budget = data[0], data[1], data[2]
    adjacency = [dict() for _ in range(n)]
    edges = []
    nonloop_edges = 0

    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u == v:
            continue
        nonloop_edges += 1
        edges.append((u, v))
        adjacency[u][v] = adjacency[u].get(v, 0) + 1
        adjacency[v][u] = adjacency[v].get(u, 0) + 1

    if budget < nonloop_edges:
        print("NO")
        return

    identity_cost = sum(abs(u - v) for u, v in edges)
    if identity_cost <= budget:
        output_order(list(range(n)), n)
        return

    visited = [False] * n
    components = []

    for start in range(n):
        if visited[start]:
            continue
        visited[start] = True
        queue = [start]
        component = []
        for u in queue:
            component.append(u)
            for v in adjacency[u]:
                if not visited[v]:
                    visited[v] = True
                    queue.append(v)
        components.append(component)

    # If every component is a path (or an isolated vertex), the lower bound
    # of one unit per non-loop dependency is attainable.
    path_order = []
    all_paths = True

    for component in components:
        if len(component) == 1:
            path_order.append(component[0])
            continue

        unique_edge_count = sum(len(adjacency[v]) for v in component) // 2
        endpoints = [v for v in component if len(adjacency[v]) == 1]

        if (unique_edge_count != len(component) - 1 or
                any(len(adjacency[v]) > 2 for v in component) or
                len(endpoints) != 2):
            all_paths = False
            break

        previous = -1
        current = endpoints[0]
        for _ in range(len(component)):
            path_order.append(current)
            nxt = -1
            for v in adjacency[current]:
                if v != previous:
                    nxt = v
                    break
            previous, current = current, nxt

    if all_paths:
        output_order(path_order, n)
        return

    # Try inexpensive layouts first, which also handles large easy instances.
    heuristic_order = []
    heuristic_total = 0

    for component in components:
        if len(component) <= 1:
            heuristic_order.extend(component)
            continue

        candidates = []
        candidates.append(sorted(component))
        candidates.append(sorted(component, key=lambda v: (len(adjacency[v]), v)))
        candidates.append(sorted(component, key=lambda v: (-len(adjacency[v]), v)))

        starts = sorted(component, key=lambda v: (len(adjacency[v]), v))[:min(8, len(component))]
        for start in starts:
            seen = {start}
            q = deque([start])
            bfs_order = []
            while q:
                u = q.popleft()
                bfs_order.append(u)
                neighbors = sorted(
                    (v for v in adjacency[u] if v not in seen),
                    key=lambda v: (len(adjacency[v]), v)
                )
                for v in neighbors:
                    seen.add(v)
                    q.append(v)
            candidates.append(bfs_order)
            candidates.append(list(reversed(bfs_order)))

        best_order = None
        best_cost = None
        for candidate in candidates:
            cost = order_cost(candidate, adjacency)
            if best_cost is None or cost < best_cost:
                best_cost = cost
                best_order = candidate

        heuristic_total += best_cost
        heuristic_order.extend(best_order)

    if heuristic_total <= budget:
        output_order(heuristic_order, n)
        return

    total_minimum = 0
    final_order = []

    for component in components:
        remaining = budget - total_minimum
        if remaining < 0:
            print("NO")
            return

        minimum, component_order = solve_component_dp(component, adjacency, remaining)
        if component_order is None:
            print("NO")
            return

        total_minimum += minimum
        final_order.extend(component_order)

    if total_minimum <= budget:
        output_order(final_order, n)
    else:
        print("NO")

if __name__ == "__main__":
    main()
