```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    B = int(data[idx]); idx += 1

    deg = [0] * (n + 1)
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        deg[u] += 1
        deg[v] += 1

    # Sort variables by degree descending; assign to stages 1..n in that order.
    # This is a heuristic that tends to minimize sum of |x-y| over edges.
    order = sorted(range(1, n + 1), key=lambda x: -deg[x])
    pos = [0] * (n + 1)
    for stage, var in enumerate(order, start=1):
        pos[var] = stage

    # Compute total delay
    total = 0
    idx2 = 3
    for _ in range(m):
        u = int(data[idx2]); idx2 += 1
        v = int(data[idx2]); idx2 += 1
        total += abs(pos[u] - pos[v])

    if total <= B:
        out = ['YES']
        out.append(' '.join(str(pos[i]) for i in range(1, n + 1)))
        sys.stdout.write('\n'.join(out))
    else:
        sys.stdout.write('NO')

solve()
```
