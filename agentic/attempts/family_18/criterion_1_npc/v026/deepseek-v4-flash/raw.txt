We can model this as a problem of assigning numbers 1…n to vertices so that the sum of absolute differences over edges is ≤ B.

A simple valid approach is to use a **minimum linear arrangement** heuristic: we can sort vertices by their graph degree and assign sectors in that order. This often works, but we may need a backtracking or branch‑and‑bound search to be sure.

Here’s an efficient constructive search that uses **branch and bound** to try assignments and prune when cost exceeds B, while exploiting the fact that n is small enough in many cases.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n, m, B = map(int, input_data[:3])
    edges = []
    adj = [[] for _ in range(n)]  # 0-indexed
    for i in range(m):
        u = int(input_data[3 + 2*i]) - 1
        v = int(input_data[4 + 2*i]) - 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)

    # Precompute edge list as pairs of indices
    edges = [(u, v) for u, v in edges]

    # Greedy starting order: by degree descending, to place high‑degree near center
    degree = [len(adj[i]) for i in range(n)]
    order = sorted(range(n), key=lambda x: degree[x], reverse=True)

    # We'll try assigning variables sequentially, but reorder for search
    assigned = [0] * n
    pos = [0] * n
    used = [False] * (n + 1)

    best_assign = None
    best_cost = float('inf')

    # Branch and bound DFS
    def dfs(idx, current_cost):
        nonlocal best_assign, best_cost
        if current_cost > B:
            return
        if idx == n:
            if current_cost < best_cost:
                best_cost = current_cost
                best_assign = assigned[:]
            return

        var = order[idx]
        # Try positions 1..n
        for p in range(1, n + 1):
            if not used[p]:
                # compute additional cost
                add = 0
                placed_neighbors = []
                for nb in adj[var]:
                    if assigned[nb] != 0:
                        add += abs(p - assigned[nb])
                if current_cost + add > B:
                    continue
                # assign
                assigned[var] = p
                used[p] = True
                dfs(idx + 1, current_cost + add)
                used[p] = False
                assigned[var] = 0
                # optional: if found exact B, can early break
                if best_cost <= B:
                    return

    dfs(0, 0)

    if best_assign is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(str(best_assign[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
