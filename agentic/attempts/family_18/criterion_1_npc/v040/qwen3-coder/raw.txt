```python
import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-based
        v = int(data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
    
    # Try all permutations
    for perm in permutations(range(n)):
        total = 0
        pos = [0] * n
        for i in range(n):
            pos[perm[i]] = i
        
        for u, v in edges:
            total += abs(pos[u] - pos[v])
            if total > B:
                break
        if total <= B:
            print("YES")
            print(' '.join(str(perm[i] + 1) for i in range(n)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
