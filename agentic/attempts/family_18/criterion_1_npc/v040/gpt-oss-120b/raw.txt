import sys, itertools

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    # pre‑compute adjacency list for possible pruning (optional)
    # brute‑force all permutations
    positions = list(range(1, n + 1))  # bay numbers
    for perm in itertools.permutations(positions):
        total = 0
        for u, v in edges:
            total += abs(perm[u] - perm[v])
            if total > B:          # early stop
                break
        if total <= B:
            print("YES")
            print(' '.join(map(str, perm)))
            return
    print("NO")

if __name__ == "__main__":
    solve()
