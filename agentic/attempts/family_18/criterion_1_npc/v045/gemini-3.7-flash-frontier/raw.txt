import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    adj = [0] * n
    deg = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u
            deg[u] += 1
            deg[v] += 1

    # DP over subsets
    # dp[mask] = (min_cost, last_vertex_added)
    # cut[mask] = number of edges between mask and V \ mask
    num_states = 1 << n
    dp = [10**9] * num_states
    parent = [-1] * num_states
    cut = [0] * num_states

    dp[0] = 0

    # Precompute popcount / bit lengths or iterate in order of mask
    for mask in range(num_states - 1):
        cur_cost = dp[mask]
        if cur_cost > B:
            continue

        c = cut[mask]

        # Try adding each vertex not in mask
        available = (~mask) & (num_states - 1)
        temp = available
        while temp:
            v = (temp & -temp).bit_length() - 1
            temp &= temp - 1

            next_mask = mask | (1 << v)
            # cut of next_mask = cut(mask) + deg(v) - 2 * (edges from v to mask)
            edges_to_mask = (adj[v] & mask).bit_count()
            next_cut = c + deg[v] - 2 * edges_to_mask
            cut[next_mask] = next_cut

            next_cost = cur_cost + next_cut
            if next_cost < dp[next_mask]:
                dp[next_mask] = next_cost
                parent[next_mask] = v

    final_mask = num_states - 1
    if dp[final_mask] <= B:
        print("YES")
        # Reconstruct the order of operations in slots 1..n
        order = []
        curr = final_mask
        while curr > 0:
            v = parent[curr]
            order.append(v)
            curr ^= 1 << v
        order.reverse()

        # order[k] is the operation assigned to slot (k + 1)
        # We need the time slot assigned to operation i
        ans = [0] * n
        for slot, op in enumerate(order, 1):
            ans[op] = slot

        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
