```python
import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    for i in range(m):
        u = int(data[3 + 2*i]) - 1  # Convert to 0-indexed
        v = int(data[4 + 2*i]) - 1
        edges.append((u, v))
    
    # Try all permutations
    for perm in permutations(range(n)):
        total = 0
        pos = [0] * n
        for i in range(n):
            pos[perm[i]] = i
        
        for u, v in edges:
            total += abs(pos[u] - pos[v])
        
        if total <= B:
            print("YES")
            result = [0] * n
            for i in range(n):
                result[perm[i]] = i + 1  # Convert back to 1-indexed
            print(' '.join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
