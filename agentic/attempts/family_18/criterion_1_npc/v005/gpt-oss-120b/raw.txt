```python
import sys
import itertools

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    # compute total length for a given permutation (list of positions for nodes 0..n-1)
    def total_len(pos):
        s = 0
        for u, v in edges:
            s += abs(pos[u] - pos[v])
            if s > B:  # early break
                return s
        return s

    # try all permutations if feasible
    if n <= 8:
        nodes = list(range(n))
        for perm in itertools.permutations(range(1, n + 1)):
            # perm[i] is position of node i (1‑based)
            pos = [0] * n
            for idx, p in enumerate(perm):
                pos[idx] = p
            if total_len(pos) <= B:
                print("YES")
                print(' '.join(map(str, perm)))
                return
        print("NO")
        return

    # for larger n use simple order 1..n as a heuristic
    perm = list(range(1, n + 1))
    pos = perm[:]  # position of node i is i+1
    if total_len(pos) <= B:
        print("YES")
        print(' '.join(map(str, perm)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
