import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # BFS from node 0 to get distances
    dist = [-1] * n
    dist[0] = 0
    q = deque([0])
    while q:
        u = q.popleft()
        for v in adj[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                q.append(v)

    # Check if graph is connected
    if any(d == -1 for d in dist):
        print("NO")
        return

    # Sort nodes by distance from node 0
    nodes = list(range(n))
    nodes.sort(key=lambda x: dist[x])

    # Place nodes in order of increasing distance
    pos = [0] * n
    for i, node in enumerate(nodes):
        pos[node] = i + 1

    # Calculate total harness length
    total = 0
    for u in range(n):
        for v in adj[u]:
            if u < v:
                total += abs(pos[u] - pos[v])

    if total <= B:
        print("YES")
        print(' '.join(map(str, pos)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
