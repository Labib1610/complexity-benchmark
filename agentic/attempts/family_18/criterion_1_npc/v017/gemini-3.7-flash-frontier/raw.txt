import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    adj = [0] * n
    deg = [0] * n

    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u
            deg[u] += 1
            deg[v] += 1

    num_states = 1 << n
    cut = [0] * num_states

    # Precompute cut sizes
    for mask in range(1, num_states):
        # find lowest set bit
        lsb = mask & -mask
        v = lsb.bit_length() - 1
        prev = mask ^ lsb
        cut[mask] = cut[prev] + deg[v] - 2 * (adj[v] & prev).bit_count()

    INF = float("inf")
    dp = [INF] * num_states
    dp[0] = 0

    for mask in range(num_states - 1):
        cur_dp = dp[mask]
        if cur_dp > B:
            continue
        # Try adding each vertex not in mask
        avail = ((1 << n) - 1) ^ mask
        while avail:
            lsb = avail & -avail
            nxt = mask | lsb
            cost = cur_dp + cut[nxt]
            if cost < dp[nxt]:
                dp[nxt] = cost
            avail ^= lsb

    full_mask = num_states - 1
    if dp[full_mask] > B:
        print("NO")
        return

    print("YES")

    # Reconstruct the permutation
    assignment = [0] * n
    cur_mask = full_mask

    for pos in range(n, 0, -1):
        # Find which vertex was placed at position `pos`
        best_v = -1
        best_prev_dp = INF

        temp = cur_mask
        while temp:
            lsb = temp & -temp
            v = lsb.bit_length() - 1
            prev = cur_mask ^ lsb
            if dp[prev] < best_prev_dp:
                best_prev_dp = dp[prev]
                best_v = v
            temp ^= lsb

        assignment[best_v] = pos
        cur_mask ^= 1 << best_v

    print(*(assignment[i] for i in range(n)))


if __name__ == "__main__":
    solve()
