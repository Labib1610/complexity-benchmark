import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    index = 3
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Try all permutations if n is small
    if n <= 8:
        for perm in permutations(range(n)):
            total = 0
            for u, v in edges:
                pos_u = perm[u]
                pos_v = perm[v]
                total += abs(pos_u - pos_v)
                if total > B:
                    break
            if total <= B:
                # Found a valid permutation
                result = [0] * n
                for i in range(n):
                    result[perm[i]] = i + 1  # Convert back to 1-based indexing
                print("YES")
                print(' '.join(map(str, result)))
                return
        print("NO")
