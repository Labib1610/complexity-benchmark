import sys
from array import array
from collections import deque

def ordering_cost(order, adj):
    pos = {v: i for i, v in enumerate(order)}
    total = 0
    for u in order:
        pu = pos[u]
        for v, w in adj[u].items():
            if u < v:
                total += w * abs(pu - pos[v])
    return total

def heuristic_order(component, adj):
    if len(component) <= 1:
        return component[:]

    component_set = set(component)
    degree = {v: sum(adj[v].values()) for v in component}

    def bfs_order(start):
        seen = {start}
        q = deque([start])
        result = []
        while q:
            u = q.popleft()
            result.append(u)
            neighbors = [v for v in adj[u] if v in component_set and v not in seen]
            neighbors.sort(key=lambda x: (degree[x], x))
            for v in neighbors:
                seen.add(v)
                q.append(v)
        return result

    def farthest(start):
        dist = {start: 0}
        q = deque([start])
        last = start
        while q:
            u = q.popleft()
            if dist[u] > dist[last]:
                last = u
            for v in adj[u]:
                if v in component_set and v not in dist:
                    dist[v] = dist[u] + 1
                    q.append(v)
        return last

    minimum_degree_vertex = min(component, key=lambda x: (degree[x], x))
    endpoint1 = farthest(minimum_degree_vertex)
    endpoint2 = farthest(endpoint1)

    candidates = [
        bfs_order(minimum_degree_vertex),
        bfs_order(endpoint1),
        bfs_order(endpoint2),
    ]

    for start in (minimum_degree_vertex, endpoint1, endpoint2):
        seen = {start}
        stack = [start]
        result = []
        while stack:
            u = stack.pop()
            result.append(u)
            neighbors = [v for v in adj[u] if v in component_set and v not in seen]
            neighbors.sort(key=lambda x: (degree[x], x), reverse=True)
            for v in neighbors:
                seen.add(v)
                stack.append(v)
        candidates.append(result)

    best = candidates[0]
    best_cost = ordering_cost(best, adj)
    for candidate in candidates[1:]:
        cost = ordering_cost(candidate, adj)
        if cost < best_cost:
            best = candidate
            best_cost = cost
    return best

def path_order_if_possible(component, adj):
    if len(component) <= 1:
        return component[:]

    unique_edge_count = sum(len(adj[v]) for v in component) // 2
    if unique_edge_count != len(component) - 1:
        return None
    if any(len(adj[v]) > 2 for v in component):
        return None

    start = next((v for v in component if len(adj[v]) == 1), None)
    if start is None:
        return None

    order = []
    previous = -1
    current = start
    while True:
        order.append(current)
        nxt = None
        for v in adj[current]:
            if v != previous:
                nxt = v
                break
        if nxt is None:
            break
        previous, current = current, nxt

    return order if len(order) == len(component) else None

def exact_array(component, adj, limit):
    k = len(component)
    index = {v: i for i, v in enumerate(component)}
    rows = [dict() for _ in range(k)]
    degrees = [0] * k
    simple = True

    for i, u in enumerate(component):
        for v, w in adj[u].items():
            j = index[v]
            rows[i][j] = w
            degrees[i] += w
            if w != 1:
                simple = False

    neighbor_masks = [0] * k
    if simple:
        for i in range(k):
            mask = 0
            for j in rows[i]:
                mask |= 1 << j
            neighbor_masks[i] = mask

    size = 1 << k
    full = size - 1
    total_weight = sum(degrees) // 2
    max_cut = total_weight
    use_int32 = limit + 1 <= 2_147_483_647 and max_cut <= 2_147_483_647
    typecode = 'i' if use_int32 else 'q'
    inf = limit + 1

    cuts = array(typecode, [0]) * size
    dp = array(typecode, [inf]) * size
    dp[0] = 0

    for mask in range(1, size):
        bit = mask & -mask
        i = bit.bit_length() - 1
        previous = mask ^ bit

        if simple:
            connected = (neighbor_masks[i] & previous).bit_count()
        else:
            connected = 0
            for j, w in rows[i].items():
                if previous & (1 << j):
                    connected += w

        cut = cuts[previous] + degrees[i] - 2 * connected
        cuts[mask] = cut

        remaining = mask
        best = inf
        while remaining:
            b = remaining & -remaining
            value = dp[mask ^ b]
            if value < best:
                best = value
            remaining ^= b

        value = best + cut
        if value > inf:
            value = inf
        dp[mask] = value

    if dp[full] > limit:
        return None, None

    reverse_order = []
    mask = full
    while mask:
        target = dp[mask] - cuts[mask]
        remaining = mask
        chosen = 0
        while remaining:
            bit = remaining & -remaining
            if dp[mask ^ bit] == target:
                chosen = bit
                break
            remaining ^= bit
        i = chosen.bit_length() - 1
        reverse_order.append(component[i])
        mask ^= chosen

    order = reverse_order[::-1]
    return order, dp[full]

def exact_sparse(component, adj, limit):
    k = len(component)
    index = {v: i for i, v in enumerate(component)}
    rows = [dict() for _ in range(k)]
    degrees = [0] * k
    simple = True

    for i, u in enumerate(component):
        for v, w in adj[u].items():
            j = index[v]
            rows[i][j] = w
            degrees[i] += w
            if w != 1:
                simple = False

    total_weight = sum(degrees) // 2
    if total_weight > limit:
        return None, None

    neighbor_masks = [0] * k
    if simple:
        for i in range(k):
            mask = 0
            for j in rows[i]:
                mask |= 1 << j
            neighbor_masks[i] = mask

    full = (1 << k) - 1
    current = {0: (0, 0, 0)}  # mask -> (cost, cut, internal edge weight)
    parent = {}

    for _ in range(k):
        nxt = {}
        for mask, (cost, cut, internal) in current.items():
            available = full ^ mask
            while available:
                bit = available & -available
                i = bit.bit_length() - 1

                if simple:
                    connected = (neighbor_masks[i] & mask).bit_count()
                else:
                    connected = 0
                    for j, w in rows[i].items():
                        if mask & (1 << j):
                            connected += w

                new_mask = mask | bit
                new_cut = cut + degrees[i] - 2 * connected
                new_internal = internal + connected
                new_cost = cost + (0 if new_mask == full else new_cut)

                remaining_internal = total_weight - new_internal - new_cut
                if new_cost + remaining_internal <= limit:
                    old = nxt.get(new_mask)
                    if old is None or new_cost < old[0]:
                        nxt[new_mask] = (new_cost, new_cut, new_internal)
                        parent[new_mask] = mask

                available ^= bit

        current = nxt
        if not current:
            return None, None

    if full not in current:
        return None, None

    cost = current[full][0]
    reverse_order = []
    mask = full
    while mask:
        previous = parent[mask]
        bit = mask ^ previous
        reverse_order.append(component[bit.bit_length() - 1])
        mask = previous

    return reverse_order[::-1], cost

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, budget = data[0], data[1], data[2]
    adj = [dict() for _ in range(n)]
    p = 3

    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u == v:
            continue
        adj[u][v] = adj[u].get(v, 0) + 1
        adj[v][u] = adj[v].get(u, 0) + 1

    visited = [False] * n
    components = []
    isolated = []

    for start in range(n):
        if visited[start]:
            continue
        visited[start] = True
        if not adj[start]:
            isolated.append(start)
            continue

        stack = [start]
        component = []
        while stack:
            u = stack.pop()
            component.append(u)
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    stack.append(v)
        components.append(component)

    lower_bounds = []
    total_lower_bound = 0
    for component in components:
        edge_weight = sum(sum(adj[v].values()) for v in component) // 2
        lower_bounds.append(edge_weight)
        total_lower_bound += edge_weight

    if total_lower_bound > budget:
        print("NO")
        return

    heuristic_orders = []
    heuristic_total = 0
    for component in components:
        path_order = path_order_if_possible(component, adj)
        if path_order is not None:
            order = path_order
        else:
            order = heuristic_order(component, adj)
        heuristic_orders.append(order)
        heuristic_total += ordering_cost(order, adj)

    if heuristic_total <= budget:
        final_order = []
        for order in heuristic_orders:
            final_order.extend(order)
        final_order.extend(isolated)

        positions = [0] * n
        for i, v in enumerate(final_order, 1):
            positions[v] = i
        print("YES")
        print(*positions)
        return

    indices = list(range(len(components)))
    indices.sort(key=lambda i: len(components[i]), reverse=True)

    remaining_lower = total_lower_bound
    used_cost = 0
    exact_orders = [None] * len(components)

    for idx in indices:
        component = components[idx]
        component_lower = lower_bounds[idx]
        remaining_lower -= component_lower
        component_limit = budget - used_cost - remaining_lower

        if component_limit < component_lower:
            print("NO")
            return

        path_order = path_order_if_possible(component, adj)
        if path_order is not None:
            order = path_order
            cost = component_lower
        elif len(component) <= 22:
            order, cost = exact_array(component, adj, component_limit)
        else:
            order, cost = exact_sparse(component, adj, component_limit)

        if order is None:
            print("NO")
            return

        exact_orders[idx] = order
        used_cost += cost

    if used_cost > budget:
        print("NO")
        return

    final_order = []
    for order in exact_orders:
        final_order.extend(order)
    final_order.extend(isolated)

    positions = [0] * n
    for i, v in enumerate(final_order, 1):
        positions[v] = i

    print("YES")
    print(*positions)

if __name__ == "__main__":
    solve()
