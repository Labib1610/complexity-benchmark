import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    dependencies = []
    for i in range(m):
        u = int(data[3 + 2*i]) - 1  # Convert to 0-based indexing
        v = int(data[4 + 2*i]) - 1
        dependencies.append((u, v))
    
    # If m is 0, any permutation works
    if m == 0:
        print("YES")
        print(" ".join(str(i) for i in range(1, n+1)))
        return
    
    # Try all permutations for small n
    if n <= 8:
        for perm in permutations(range(n)):
            total = 0
            pos = [0] * n
            for i in range(n):
                pos[perm[i]] = i + 1  # 1-based phase number
            for u, v in dependencies:
                total += abs(pos[u] - pos[v])
                if total > B:
                    break
            if total <= B:
                print("YES")
                result = [0] * n
                for i in range(n):
                    result[perm[i]] = i + 1
                print(" ".join(map(str, result)))
                return
        print("NO")
        return
    
    # For larger n, use a heuristic approach
    # Try to place dependent nodes close to each other
    
    # Build a graph of dependencies
    from collections import defaultdict, deque
    adj = defaultdict(list)
    degree = [0] * n
    for u, v in dependencies:
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
    
    # Start with nodes of highest degree
    # Use a greedy approach: place high-degree nodes first, then their neighbors
    order = list(range(n))
    order.sort(key=lambda x: degree[x], reverse=True)
    
    # Try a few permutations around this order
    import random
    random.seed(42)
    
    best_perm = None
    min_score = float('inf')
    
    # Generate permutations by swapping adjacent elements in the sorted order
    def get_neighbors(perm):
        res = []
        for i in range(len(perm)-1):
            new_perm = list(perm)
            new_perm[i], new_perm[i+1] = new_perm[i+1], new_perm[i]
            res.append(tuple(new_perm))
        return res
    
    # Start with the degree-based order
    current_order = order[:]
    pos = [0] * n
    for i in range(n):
        pos[current_order[i]] = i + 1
    total = 0
    for u, v in dependencies:
        total += abs(pos[u] - pos[v])
    if total <= B:
        print("YES")
        result = [0] * n
        for i in range(n):
            result[current_order[i]] = i + 1
        print(" ".join(map(str, result)))
        return
    
    # Try local search
    visited = set()
    queue = deque()
    queue.append((tuple(current_order), total))
    visited.add(tuple(current_order))
    
    for _ in range(10000):  # Limit iterations
        if not queue:
            break
        perm_tuple, current_score = queue.popleft()
        if current_score < min_score:
            min_score = current_score
            best_perm = list(perm_tuple)
        if current_score <= B:
            print("YES")
            pos_result = [0] * n
            for i in range(n):
                pos_result[perm_tuple[i]] = i + 1
            print(" ".join(map(str, pos_result)))
            return
        # Get neighbors
        for neighbor_tuple in get_neighbors(perm_tuple):
            if neighbor_tuple in visited:
                continue
            visited.add(neighbor_tuple)
            # Calculate score
            pos_n = [0] * n
            for i in range(n):
                pos_n[neighbor_tuple[i]] = i + 1
            score = 0
            for u, v in dependencies:
                score += abs(pos_n[u] - pos_n[v])
                if score > B + 100:  # Early pruning
                    break
            if score <= B + 100:
                queue.append((neighbor_tuple, score))
    
    if min_score <= B and best_perm is not None:
        print("YES")
        pos_result = [0] * n
        for i in range(n):
            pos_result[best_perm[i]] = i + 1
        print(" ".join(map(str, pos_result)))
        return
    
    # If still not found, try random permutations for small n
    if n <= 10:
        for _ in range(10000):
            perm = list(range(n))
            random.shuffle(perm)
            pos_r = [0] * n
            for i in range(n):
                pos_r[perm[i]] = i + 1
            total_r = 0
            for u, v in dependencies:
                total_r += abs(pos_r[u] - pos_r[v])
            if total_r <= B:
                print("YES")
                result = [0] * n
                for i in range(n):
                    result[perm[i]] = i + 1
                print(" ".join(map(str, result)))
                return
    
    print("NO")

if __name__ == "__main__":
    main()
