```python
import sys
import time
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])
    
    adj = [[] for _ in range(n)]
    adj_mat = [[0] * n for _ in range(n)]
    deg = [0] * n
    edges = []
    
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            adj_mat[u][v] += 1
            adj_mat[v][u] += 1
            deg[u] += 1
            deg[v] += 1
            edges.append((u, v))
            
    if m == 0:
        print("YES")
        print(*(i + 1 for i in range(n)))
        return

    # If n <= 20, exact DP
    if n <= 20:
        adj_mask = [0] * n
        for u in range(n):
            for v in adj[u]:
                adj_mask[u] |= (1 << v)

        # cut[mask] = number of edges between mask and V \ mask
        # cut[mask | (1 << v)] = cut[mask] + deg[v] - 2 * (neighbors of v in mask)
        num_masks = 1 << n
        dp = [float('inf')] * num_masks
        parent = [-1] * num_masks
        cut = [0] * num_masks
        dp[0] = 0

        for mask in range(num_masks):
            c = cut[mask]
            d = dp[mask]
            for v in range(n):
                if not (mask & (1 << v)):
                    nmask = mask | (1 << v)
                    if cut[nmask] == 0 and nmask != 0:
                        in_mask_neighbors = (mask & adj_mask[v]).bit_count()
                        cut[nmask] = c + deg[v] - 2 * in_mask_neighbors
                    
                    val = d + c
                    if val < dp[nmask]:
                        dp[nmask] = val
                        parent[nmask] = v

        full = (1 << n) - 1
        if dp[full] <= B:
            print("YES")
            # Reconstruct order
            curr = full
            order = []
            while curr > 0:
                v = parent[curr]
                order.append(v)
                curr ^= (1 << v)
            order.reverse()
            # order[phase - 1] = step
            ans = [0] * n
            for phase_0, step in enumerate(order):
                ans[step] = phase_0 + 1
            print(*ans)
        else:
            print("NO")
        return

    # If n > 20, use Simulated Annealing + Local Search
    start_time = time.time()
    time_limit = 4.5

    def compute_cost(order):
        pos = [0] * n
        for p, u in enumerate(order):
            pos[u] = p
        cost = 0
        for u, v in edges:
            cost += abs(pos[u] - pos[v])
        return cost, pos

    # Initial solution: BFS / Spectral / Degree heuristic
    best_order = list(range(n))
    best_order.sort(key=lambda x: -deg[x])
    
    # Try BFS ordering as well
    visited = [False] * n
    bfs_order = []
    for i in range(n):
        if not visited[i]:
            queue = [i]
            visited[i] = True
            while queue:
                curr = queue.pop(0)
                bfs_order.append(curr)
                for neighbor in adj[curr]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
                        
    c_bfs, pos_bfs = compute_cost(bfs_order)
    c_best, pos_best = compute_cost(best_order)
    
    if c_bfs < c_best:
        best_order = bfs_order[:]
        c_best = c_bfs

    if c_best <= B:
        print("YES")
        ans = [pos_best[i] + 1 for i in range(n)]
        print(*ans)
        return

    # SA Setup
    current_order = best_order[:]
    current_pos = [0] * n
    for p, u in enumerate(current_order):
        current_pos[u] = p
    current_cost = c_best

    T_start = float(max(1.0, current_cost * 0.1))
    T_end = 0.001
    
    iters = 0
    while time.time() - start_time < time_limit:
        iters += 1
        # Annealing schedule
        elapsed = (time.time() - start_time) / time_limit
        T = T_start * ((T_end / T_start) ** elapsed)
        
        # Pick two indices to swap
        i = random.randint(0, n - 1)
        j = random.randint(0, n - 1)
        if i == j:
            continue
        
        u = current_order[i]
        v = current_order[j]
        
        # Calculate delta cost
        delta = 0
        for w in adj[u]:
            if w != v:
                delta += abs(j - current_pos[w]) - abs(i - current_pos[w])
        for w in adj[v]:
            if w != u:
                delta += abs(i - current_pos[w]) - abs(j - current_pos[w])
                
        if delta < 0 or (T > 0 and random.random() < 2.718281828459045 ** (-delta / T)):
            current_order[i], current_order[j] = v, u
            current_pos[u] = j
            current_pos[v] = i
            current_cost += delta
            
            if current_cost < c_best:
                c_best = current_cost
                best_order = current_order[:]
                if c_best <= B:
                    print("YES")
                    ans = [current_pos[k] + 1 for k in range(n)]
                    print(*ans)
                    return

    if c_best <= B:
        print("YES")
        _, best_pos = compute_cost(best_order)
        ans = [best_pos[k] + 1 for k in range(n)]
        print(*ans)
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
