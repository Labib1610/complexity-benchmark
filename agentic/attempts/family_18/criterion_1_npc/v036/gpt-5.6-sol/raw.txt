import sys
from collections import deque

def arrangement_cost(order, edges):
    pos = [0] * len(order)
    for i, v in enumerate(order):
        pos[v] = i
    return sum(abs(pos[a] - pos[b]) for a, b in edges)

def improve_order(order, adj):
    k = len(order)
    if k <= 1:
        return order

    order = order[:]
    pos = [0] * k
    for i, v in enumerate(order):
        pos[v] = i

    for _ in range(6):
        changed = False
        directions = (range(k - 1), range(k - 2, -1, -1))
        for indices in directions:
            for p in indices:
                a = order[p]
                b = order[p + 1]
                delta = 0

                for x in adj[a]:
                    if x != b:
                        delta += abs((p + 1) - pos[x]) - abs(p - pos[x])
                for x in adj[b]:
                    if x != a:
                        delta += abs(p - pos[x]) - abs((p + 1) - pos[x])

                if delta < 0:
                    order[p], order[p + 1] = b, a
                    pos[a], pos[b] = p + 1, p
                    changed = True
        if not changed:
            break

    return order

def heuristic_order(adj, edges):
    k = len(adj)
    if k <= 1:
        return list(range(k)), 0

    degrees = [len(x) for x in adj]
    candidates = [
        list(range(k)),
        sorted(range(k), key=lambda x: (degrees[x], x)),
        sorted(range(k), key=lambda x: (-degrees[x], x)),
    ]

    roots = {
        0,
        min(range(k), key=lambda x: (degrees[x], x)),
        max(range(k), key=lambda x: (degrees[x], -x)),
    }

    for root in roots:
        seen = [False] * k
        seen[root] = True
        q = deque([root])
        order = []
        while q:
            v = q.popleft()
            order.append(v)
            nxt = [u for u in adj[v] if not seen[u]]
            nxt.sort(key=lambda x: (degrees[x], x))
            for u in nxt:
                if not seen[u]:
                    seen[u] = True
                    q.append(u)
        candidates.append(order)
        candidates.append(order[::-1])

    best_order = None
    best_cost = None
    for order in candidates:
        order = improve_order(order, adj)
        cost = arrangement_cost(order, edges)
        if best_cost is None or cost < best_cost:
            best_cost = cost
            best_order = order

    return best_order, best_cost

def exact_order(adj, edge_count, cap):
    k = len(adj)
    if k <= 1:
        return list(range(k)), 0
    if edge_count > cap:
        return None

    adj_masks = [0] * k
    degrees = [0] * k
    for v in range(k):
        mask = 0
        for u in adj[v]:
            mask |= 1 << u
        adj_masks[v] = mask
        degrees[v] = len(adj[v])

    full = (1 << k) - 1
    base = 2 * edge_count + 1

    # Packed value: ((cost * base) + cut) * base + degree_sum
    layers = [{0: 0}]
    current = layers[0]

    for size in range(k):
        nxt = {}
        remaining_after = k - size - 1

        for mask, packed in current.items():
            t = packed
            degree_sum = t % base
            t //= base
            cut = t % base
            cost = t // base

            missing = full ^ mask
            while missing:
                bit = missing & -missing
                missing -= bit
                v = bit.bit_length() - 1

                neighbors_inside = (adj_masks[v] & mask).bit_count()
                new_cut = cut + degrees[v] - 2 * neighbors_inside
                new_degree_sum = degree_sum + degrees[v]
                new_cost = cost + new_cut

                if new_cost > cap:
                    continue

                # Every edge wholly inside the unplaced set must contribute
                # at least one more unit to the objective.
                remaining_edges = (
                    2 * edge_count - new_degree_sum - new_cut
                ) // 2
                if new_cost + remaining_edges > cap:
                    continue

                # A connected remaining process has at least one crossing cut
                # for each non-final placement when this is stronger.
                if remaining_after > 0 and new_cost + remaining_after > cap:
                    continue

                new_mask = mask | bit
                new_packed = (
                    (new_cost * base + new_cut) * base + new_degree_sum
                )

                old = nxt.get(new_mask)
                if old is None or new_packed < old:
                    nxt[new_mask] = new_packed

        if not nxt:
            return None
        layers.append(nxt)
        current = nxt

    packed = current.get(full)
    if packed is None:
        return None

    minimum_cost = packed // base // base
    mask = full
    reverse_order = []

    for size in range(k, 0, -1):
        packed = layers[size][mask]
        t = packed // base
        current_cut = t % base
        current_cost = t // base
        target_previous_cost = current_cost - current_cut

        bits = mask
        chosen = -1
        while bits:
            bit = bits & -bits
            bits -= bit
            previous_mask = mask ^ bit
            previous_packed = layers[size - 1].get(previous_mask)
            if previous_packed is not None:
                previous_cost = previous_packed // base // base
                if previous_cost == target_previous_cost:
                    chosen = bit.bit_length() - 1
                    break

        if chosen < 0:
            return None

        reverse_order.append(chosen)
        mask ^= 1 << chosen

    reverse_order.reverse()
    return reverse_order, minimum_cost

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 3:
        return

    n, m_input, budget = data[0], data[1], data[2]
    graph = [set() for _ in range(n)]

    idx = 3
    for _ in range(m_input):
        x = data[idx] - 1
        y = data[idx + 1] - 1
        idx += 2
        if x != y:
            graph[x].add(y)
            graph[y].add(x)

    edge_count_total = sum(len(s) for s in graph) // 2
    if edge_count_total > budget:
        print("NO")
        return

    visited = [False] * n
    components = []
    isolated = []

    for start in range(n):
        if visited[start]:
            continue
        if not graph[start]:
            visited[start] = True
            isolated.append(start)
            continue

        visited[start] = True
        stack = [start]
        vertices = []

        while stack:
            v = stack.pop()
            vertices.append(v)
            for u in graph[v]:
                if not visited[u]:
                    visited[u] = True
                    stack.append(u)

        local_id = {v: i for i, v in enumerate(vertices)}
        k = len(vertices)
        local_adj = [[] for _ in range(k)]
        local_edges = []

        for i, v in enumerate(vertices):
            for u in graph[v]:
                j = local_id[u]
                local_adj[i].append(j)
                if i < j:
                    local_edges.append((i, j))

        heuristic, heuristic_cost = heuristic_order(local_adj, local_edges)
        components.append({
            "vertices": vertices,
            "adj": local_adj,
            "edges": local_edges,
            "m": len(local_edges),
            "heuristic": heuristic,
            "heuristic_cost": heuristic_cost,
            "order": None,
            "cost": None,
        })

    heuristic_total = sum(c["heuristic_cost"] for c in components)

    if heuristic_total <= budget:
        for c in components:
            c["order"] = c["heuristic"]
            c["cost"] = c["heuristic_cost"]
    else:
        processing = sorted(
            range(len(components)),
            key=lambda i: len(components[i]["vertices"]),
            reverse=True
        )

        reserved_lower_bound = sum(c["m"] for c in components)
        spent = 0

        for ci in processing:
            c = components[ci]
            reserved_lower_bound -= c["m"]
            cap = budget - spent - reserved_lower_bound

            if cap < c["m"]:
                print("NO")
                return

            if c["heuristic_cost"] == c["m"]:
                c["order"] = c["heuristic"]
                c["cost"] = c["m"]
            else:
                result = exact_order(c["adj"], c["m"], cap)
                if result is None:
                    print("NO")
                    return
                c["order"], c["cost"] = result

            spent += c["cost"]

        if spent > budget:
            print("NO")
            return

    position = [0] * n
    rail_order = []

    for c in components:
        vertices = c["vertices"]
        rail_order.extend(vertices[x] for x in c["order"])
    rail_order.extend(isolated)

    for pos, instrument in enumerate(rail_order, 1):
        position[instrument] = pos

    print("YES")
    print(*position)

if __name__ == "__main__":
    main()
