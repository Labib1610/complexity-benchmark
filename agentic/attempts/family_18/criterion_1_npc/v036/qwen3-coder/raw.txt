```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    B = int(data[idx])
    idx += 1
    
    edges = []
    for _ in range(m):
        x = int(data[idx]) - 1  # Convert to 0-based indexing
        idx += 1
        y = int(data[idx]) - 1
        idx += 1
        edges.append((x, y))
    
    # We'll try to place instruments on the rail to minimize total cable usage
    # A good heuristic is to place connected instruments close to each other
    # This is essentially a graph layout problem on a line minimizing sum of distances
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    degree = [0] * n
    for x, y in edges:
        adj[x].append(y)
        adj[y].append(x)
        degree[x] += 1
        degree[y] += 1
    
    # Try to find a placement that uses at most B cables
    # We can use a greedy approach or backtracking, but since n <= 8, we can try all permutations
    
    if n <= 8:
        import itertools
        # Try all permutations of positions 0 to n-1 (0-based)
        for perm in itertools.permutations(range(n)):
            total_cable = 0
            pos = [0] * n
            for i in range(n):
                pos[perm[i]] = i
            
            for x, y in edges:
                total_cable += abs(pos[x] - pos[y])
            
            if total_cable <= B:
                print("YES")
                result = [0] * n
                for i in range(n):
                    result[perm[i]] = i + 1  # Convert to 1-based indexing
                print(' '.join(map(str, result)))
                return
        
        print("NO")
        return
    
    # For larger n, we need a more sophisticated approach
    # But according to constraints, n <= 8, so the above should work
    
    # However, let's also handle the example case properly
    # Actually, looking at the example:
    # n=4, instruments 1,2,3,4
    # edges: (1,2), (1,3), (1,4) 
    # In 0-based: (0,1), (0,2), (0,3)
    # Assignment given: 2 1 3 4 (in 1-based) -> 1 0 2 3 (in 0-based)
    # So instrument 0->pos1, 1->pos0, 2->pos2, 3->pos3
    # Distances: |1-0|=1, |1-2|=1, |1-3|=2. Total = 4. OK.
    
    # Since n <= 8 in typical contest settings, brute force with permutations works
    # But we already handled that above
    
    # If we reach here for some reason with n > 8, we might need a different strategy
    # But based on problem constraints, we should be fine
    
    # Let's just implement the permutation solution properly
    import itertools
    for perm in itertools.permutations(range(n)):
        total_cable = 0
        pos = [0] * n
        for i in range(n):
            pos[perm[i]] = i
        
        for x, y in edges:
            total_cable += abs(pos[x] - pos[y])
        
        if total_cable <= B:
            print("YES")
            result = [0] * n
            for i in range(n):
                result[perm[i]] = i + 1
            print(' '.join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
