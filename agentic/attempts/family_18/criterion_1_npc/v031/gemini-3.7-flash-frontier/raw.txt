import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    adj = [0] * n
    deg = [0] * n

    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u
            deg[u] += 1
            deg[v] += 1

    size = 1 << n
    dp = [0] * size
    cut = [0] * size
    parent = [0] * size

    # Precompute bit positions for single-bit masks
    bit_pos = {1 << i: i for i in range(n)}

    for mask in range(1, size):
        lsb = mask & -mask
        u = bit_pos[lsb]
        prev_mask = mask ^ lsb
        cut[mask] = (
            cut[prev_mask] + deg[u] - 2 * (prev_mask & adj[u]).bit_count()
        )

        best_val = 10**15
        best_u = -1

        temp = mask
        while temp:
            b = temp & -temp
            v = bit_pos[b]
            val = dp[mask ^ b]
            if val < best_val:
                best_val = val
                best_u = v
            temp ^= b

        dp[mask] = best_val + cut[mask]
        parent[mask] = best_u

    min_cost = dp[size - 1]
    if min_cost > B:
        print("NO")
        return

    print("YES")
    pos = [0] * n
    curr_mask = size - 1
    for p in range(n, 0, -1):
        u = parent[curr_mask]
        pos[u] = p
        curr_mask ^= 1 << u

    print(*(pos[i] for i in range(n)))


if __name__ == "__main__":
    solve()
