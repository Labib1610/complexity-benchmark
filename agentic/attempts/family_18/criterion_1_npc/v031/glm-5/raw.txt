```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # BFS from node 0 to find farthest node
    def bfs_farthest(start):
        dist = [-1] * n
        q = deque([start])
        dist[start] = 0
        farthest = start
        while q:
            u = q.popleft()
            for v in adj[u]:
                if dist[v] == -1:
                    dist[v] = dist[u] + 1
                    q.append(v)
                    if dist[v] > dist[farthest]:
                        farthest = v
        return farthest, dist
    
    end1, _ = bfs_farthest(0)
    end2, dist1 = bfs_farthest(end1)
    _, dist2 = bfs_farthest(end2)
    
    # Eccentricity of each node
    ecc = [max(dist1[i], dist2[i]) for i in range(n)]
    
    # Sort nodes by eccentricity
    order = sorted(range(n), key=lambda x: ecc[x])
    perm = [0] * n
    for i, node in enumerate(order):
        perm[node] = i + 1
    
    # Calculate total ribbon length
    total = 0
    for u in range(n):
        for v in adj[u]:
            if u < v:
                total += abs(perm[u] - perm[v])
    
    if total <= B:
        print("YES")
        print(' '.join(map(str, perm)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
