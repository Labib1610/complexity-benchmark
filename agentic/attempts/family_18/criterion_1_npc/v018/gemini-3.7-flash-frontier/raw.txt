An optimal and efficient solution in Python 3 using bitmask dynamic programming (based on cut values) to solve the Minimum Linear Arrangement problem.

```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    if n == 0:
        print("YES\n")
        return

    adj = [0] * n
    deg = [0] * n

    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u
            deg[u] += 1
            deg[v] += 1

    total_masks = 1 << n
    dp = [float("inf")] * total_masks
    cut = [0] * total_masks
    parent = [-1] * total_masks

    dp[0] = 0

    for mask in range(1, total_masks):
        # Pick the lowest set bit to compute cut[mask] in O(1) from cut[prev]
        lsb = mask & -mask
        v = lsb.bit_length() - 1
        prev_mask = mask ^ lsb
        cut[mask] = cut[prev_mask] + deg[v] - 2 * (adj[v] & prev_mask).bit_count()

        c = 0 if mask == total_masks - 1 else cut[mask]

        best_cost = float("inf")
        best_v = -1

        sub = mask
        while sub:
            b = sub & -sub
            u = b.bit_length() - 1
            cost = dp[mask ^ b]
            if cost < best_cost:
                best_cost = cost
                best_v = u
            sub ^= b

        dp[mask] = best_cost + c
        parent[mask] = best_v

    min_cost = dp[total_masks - 1]

    if min_cost <= B:
        print("YES")
        # Reconstruct positions
        # The module chosen at step with mask has position = popcount(mask)
        ans = [0] * n
        curr_mask = total_masks - 1
        while curr_mask > 0:
            pos = curr_mask.bit_count()
            u = parent[curr_mask]
            ans[u] = pos
            curr_mask ^= 1 << u

        print(*(ans[i] for i in range(n)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
