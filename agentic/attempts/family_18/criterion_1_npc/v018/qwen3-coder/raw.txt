```python
import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-based indexing
        v = int(data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
    
    # Try all permutations of bay assignments
    for perm in permutations(range(n)):
        total_length = 0
        positions = list(perm)
        for u, v in edges:
            pos_u = positions[u]
            pos_v = positions[v]
            total_length += abs(pos_u - pos_v)
            if total_length > B:
                break
        if total_length <= B:
            # Found a valid assignment
            result = [pos + 1 for pos in positions]  # Convert back to 1-based indexing
            print("YES")
            print(' '.join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
