```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    L = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Greedy placement: place nodes in order 1..n
    # Node i is placed at position i
    # Total cord length = sum of |i - j| for all edges (i, j) with i < j
    
    total_cord = 0
    for u in range(1, n + 1):
        for v in adj[u]:
            if v > u:
                total_cord += (v - u)
    
    if total_cord <= L:
        print("YES")
        print(' '.join(str(i) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
