```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    K = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    # order vertices by degree descending for better pruning
    vertices = list(range(1, n + 1))
    vertices.sort(key=lambda x: -len(adj[x]))

    pos = [0] * (n + 1)          # position of each vertex (1‑based)
    used = [False] * (n + 1)
    answer = None
    found = False

    def dfs(depth: int, cur_sum: int):
        nonlocal answer, found
        if found:
            return
        if cur_sum > K:
            return
        if depth > n:
            # all placed
            answer = pos[1:].copy()
            found = True
            return
        # try each unused vertex
        for v in vertices:
            if not used[v]:
                # place v at position 'depth'
                add = 0
                for nb in adj[v]:
                    if used[nb]:
                        add += abs(depth - pos[nb])
                if cur_sum + add > K:
                    continue
                used[v] = True
                pos[v] = depth
                dfs(depth + 1, cur_sum + add)
                used[v] = False
                pos[v] = 0
                if found:
                    return

    dfs(1, 0)

    out = []
    if found:
        out.append("YES")
        out.append(" ".join(map(str, answer)))
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```

This program searches for a permutation of the components (stations 1…n) whose total sum of distances for all required links does not exceed the given limit K. It uses depth‑first search with pruning based on the current accumulated length and processes vertices in decreasing degree order to reach a solution quickly for the intended problem sizes.
