import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    N = int(input_data[0])
    M = int(input_data[1])
    L = int(input_data[2])

    adj = [0] * N
    deg = [0] * N
    idx = 3
    for _ in range(M):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        deg[u] += 1
        deg[v] += 1

    total_states = 1 << N
    cut = [0] * total_states
    for mask in range(1, total_states):
        lsb = (mask & -mask).bit_length() - 1
        prev = mask ^ (1 << lsb)
        cut[mask] = cut[prev] + deg[lsb] - 2 * (adj[lsb] & prev).bit_count()

    INF = 10**9
    dp = [INF] * total_states
    parent = [0] * total_states
    dp[0] = 0

    for mask in range(total_states - 1):
        c = dp[mask]
        if c > L:
            continue
        for u in range(N):
            if not (mask & (1 << u)):
                nmask = mask | (1 << u)
                nc = c + cut[nmask]
                if nc < dp[nmask]:
                    dp[nmask] = nc
                    parent[nmask] = u

    if dp[total_states - 1] <= L:
        order = []
        curr = total_states - 1
        while curr > 0:
            u = parent[curr]
            order.append(u)
            curr ^= 1 << u
        order.reverse()

        ans = [0] * N
        for station, u in enumerate(order, 1):
            ans[u] = station

        print("YES")
        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
