import sys, itertools

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
    # brute‑force all permutations (feasible for small n)
    for perm in itertools.permutations(range(n)):
        # position of each module (1‑based)
        pos = [0] * n
        for idx, mod in enumerate(perm):
            pos[mod] = idx + 1
        total = 0
        for u, v in edges:
            total += abs(pos[u] - pos[v])
            if total > B:   # early break
                break
        if total <= B:
            # output positions in order of modules 1..n
            out = ' '.join(str(pos[i]) for i in range(n))
            sys.stdout.write("YES\n" + out + "\n")
            return
    sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
