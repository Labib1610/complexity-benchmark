import sys
import time
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])
    
    edges = []
    adj_mask = [0] * n
    adj_list = [[] for _ in range(n)]
    
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        edges.append((u, v))
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    deg = [bin(adj_mask[i]).count('1') for i in range(n)]
    
    if m == 0:
        print("YES")
        print(*(list(range(1, n + 1))))
        return

    # If n <= 20, exact DP
    if n <= 20:
        size = 1 << n
        dp = [10**9] * size
        parent = [-1] * size
        cut = [0] * size
        
        dp[0] = 0
        
        for mask in range(1, size):
            # lowest set bit
            lsb = mask & -mask
            u = (lsb).bit_length() - 1
            prev = mask ^ lsb
            
            # cut size for mask
            cut[mask] = cut[prev] + deg[u] - 2 * (adj_mask[u] & prev).bit_count()
            
            best_cost = 10**9
            best_p = -1
            
            # Try all u in mask
            temp = mask
            while temp:
                bit = temp & -temp
                v = bit.bit_length() - 1
                cost = dp[mask ^ bit]
                if cost < best_cost:
                    best_cost = cost
                    best_p = v
                temp ^= bit
            
            dp[mask] = best_cost + (cut[mask] if mask != size - 1 else 0)
            parent[mask] = best_p
            
        if dp[size - 1] <= B:
            print("YES")
            order = []
            curr = size - 1
            while curr > 0:
                p = parent[curr]
                order.append(p)
                curr ^= (1 << p)
            order.reverse()
            
            pos = [0] * n
            for bay, mod in enumerate(order, 1):
                pos[mod] = bay
            print(*(pos))
        else:
            print("NO")
        return

    # For larger n, Simulated Annealing
    start_time = time.time()
    
    def calc_cost(order):
        pos = [0] * n
        for i, u in enumerate(order):
            pos[u] = i
        c = 0
        for u, v in edges:
            c += abs(pos[u] - pos[v])
        return c, pos

    best_cost = float('inf')
    best_order = None

    while time.time() - start_time < 4.2:
        # Initial greedy / random order
        order = list(range(n))
        random.shuffle(order)
        
        pos = [0] * n
        for i, u in enumerate(order):
            pos[u] = i
            
        cur_cost = 0
        for u, v in edges:
            cur_cost += abs(pos[u] - pos[v])
            
        if cur_cost <= B:
            print("YES")
            res_pos = [p + 1 for p in pos]
            print(*res_pos)
            return
            
        T = 50.0
        T_min = 1e-3
        alpha = 0.995
        
        while T > T_min and time.time() - start_time < 4.2:
            for _ in range(2 * n):
                i = random.randint(0, n - 1)
                j = random.randint(0, n - 1)
                if i == j:
                    continue
                u = order[i]
                v = order[j]
                
                # delta cost calculation
                delta = 0
                for w in adj_list[u]:
                    if w != v:
                        delta += abs(j - pos[w]) - abs(i - pos[w])
                for w in adj_list[v]:
                    if w != u:
                        delta += abs(i - pos[w]) - abs(j - pos[w])
                
                if delta <= 0 or random.random() < (2.718281828459045 ** (-delta / T)):
                    order[i], order[j] = order[j], order[i]
                    pos[u], pos[v] = j, i
                    cur_cost += delta
                    if cur_cost <= B:
                        print("YES")
                        res_pos = [p + 1 for p in pos]
                        print(*res_pos)
                        return
            T *= alpha
            
        if cur_cost < best_cost:
            best_cost = cur_cost
            best_order = list(order)

    if best_cost <= B:
        print("YES")
        pos = [0] * n
        for i, u in enumerate(best_order):
            pos[u] = i + 1
        print(*pos)
    else:
        print("NO")

if __name__ == '__main__':
    solve()
