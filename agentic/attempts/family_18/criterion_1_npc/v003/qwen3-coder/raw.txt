```python
import sys
from itertools import permutations

def solve():
    n, m, B = map(int, sys.stdin.readline().split())
    
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-based indexing
    
    # Try all permutations
    for perm in permutations(range(n)):
        total_length = 0
        for u, v in edges:
            pos_u = perm[u]
            pos_v = perm[v]
            total_length += abs(pos_u - pos_v)
        
        if total_length <= B:
            print("YES")
            result = [0] * n
            for i in range(n):
                result[perm[i]] = i + 1  # Convert back to 1-based indexing
            print(*result)
            return
    
    print("NO")

solve()
```
