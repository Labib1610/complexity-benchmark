```python
import sys
import threading

def solve():
    import sys, itertools
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [(int(next(it))-1, int(next(it))-1) for _ in range(m)]
    # Precompute adjacency list for quick sum
    # Brute force permutations
    positions = list(range(n))
    for perm in itertools.permutations(positions):
        # perm[i] = position of module i (0-indexed)
        total = 0
        for u, v in edges:
            total += abs(perm[u] - perm[v])
            if total > B:
                break
        if total <= B:
            # output 1-indexed positions
            out = [str(p+1) for p in perm]
            sys.stdout.write("YES\n")
            sys.stdout.write(" ".join(out))
            return
    sys.stdout.write("NO")

def main():
    solve()

if __name__ == "__main__":
    threading.Thread(target=main).start()
```
