```python
import sys
from itertools import permutations

def read_input():
    n, m, B = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-based indexing
    return n, m, B, edges

def calculate_cost(perm, edges):
    cost = 0
    for u, v in edges:
        pos_u = perm[u]
        pos_v = perm[v]
        cost += abs(pos_u - pos_v)
    return cost

def solve():
    n, m, B, edges = read_input()
    
    # Try all permutations
    for perm in permutations(range(n)):
        cost = calculate_cost(perm, edges)
        if cost <= B:
            # Convert back to 1-based indexing for output
            result = [str(p + 1) for p in perm]
            print("YES")
            print(" ".join(result))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
```
