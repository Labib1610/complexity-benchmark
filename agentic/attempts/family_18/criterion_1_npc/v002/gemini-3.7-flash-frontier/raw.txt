import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    adj = [0] * n
    deg = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u
            deg[u] += 1
            deg[v] += 1

    total_states = 1 << n
    cut = [0] * total_states

    for S in range(1, total_states):
        lsb = S & -S
        u = lsb.bit_length() - 1
        S_prev = S ^ lsb
        cut[S] = cut[S_prev] + deg[u] - 2 * (S_prev & adj[u]).bit_count()

    INF = 10**18
    dp = [INF] * total_states
    parent = [-1] * total_states
    dp[0] = 0

    for S in range(total_states - 1):
        d = dp[S]
        if d > B:
            continue
        cost = d + cut[S]
        if cost > B:
            continue

        rem = (total_states - 1) ^ S
        while rem:
            lsb = rem & -rem
            u = lsb.bit_length() - 1
            rem ^= lsb
            nxt = S | lsb
            if cost < dp[nxt]:
                dp[nxt] = cost
                parent[nxt] = u

    full_mask = total_states - 1
    if dp[full_mask] <= B:
        order = []
        curr = full_mask
        while curr > 0:
            u = parent[curr]
            order.append(u)
            curr ^= 1 << u
        order.reverse()

        pos = [0] * n
        for p, u in enumerate(order, 1):
            pos[u] = p

        print("YES")
        print(*(pos[i] for i in range(n)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
