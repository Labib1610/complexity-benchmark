import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = []
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)

    pos = [-1] * n          # position of each node (1..n)
    used = [False] * (n + 1)  # positions used
    answer = None

    # precompute edge list per node for quick update
    edge_idx = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        edge_idx[u].append((idx, v))
        edge_idx[v].append((idx, u))

    # current contribution of each edge, -1 if not yet both ends placed
    edge_contrib = [-1] * m
    cur_sum = 0

    def dfs(v: int) -> bool:
        nonlocal cur_sum, answer
        if v == n:
            if cur_sum <= B:
                answer = pos.copy()
                return True
            return False

        for p in range(1, n + 1):
            if used[p]:
                continue
            # place node v at position p
            added = []
            inc = 0
            ok = True
            for idx, nb in edge_idx[v]:
                if pos[nb] != -1:          # neighbor already placed
                    d = abs(p - pos[nb])
                    inc += d
                    added.append(idx)
                # else not yet counted
            if cur_sum + inc > B:
                continue
            # commit
            pos[v] = p
            used[p] = True
            cur_sum += inc
            for idx in added:
                edge_contrib[idx] = inc  # store just for completeness

            if dfs(v + 1):
                return True

            # backtrack
            for idx in added:
                edge_contrib[idx] = -1
            cur_sum -= inc
            used[p] = False
            pos[v] = -1
        return False

    if dfs(0):
        print("YES")
        # convert positions to 1‑based list per module order
        print(' '.join(str(x) for x in answer))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
