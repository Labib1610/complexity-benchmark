```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    
    deg = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        deg[u] += 1
        deg[v] += 1
    
    modules = list(range(1, n + 1))
    modules.sort(key=lambda x: deg[x], reverse=True)
    
    pos = [0] * (n + 1)
    left = 1
    right = n
    for i, mod in enumerate(modules):
        if i % 2 == 0:
            pos[mod] = left
            left += 1
        else:
            pos[mod] = right
            right -= 1
    
    total = 0
    for mod in modules:
        d = deg[mod]
        p = pos[mod]
        total += d * min(p - 1, n - p)
    
    if total <= B:
        out = ['YES']
        out.append(' '.join(str(pos[i]) for i in range(1, n + 1)))
        sys.stdout.write('\n'.join(out))
    else:
        sys.stdout.write('NO')

solve()
```
