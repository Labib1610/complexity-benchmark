```python
import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    index = 3
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2

    # Try all permutations of bay assignments (0 to n-1)
    for perm in permutations(range(n)):
        total_cost = 0
        positions = [0] * n
        for i in range(n):
            positions[perm[i]] = i  # module perm[i] is placed at position i
        
        for u, v in edges:
            pos_u = positions[u]
            pos_v = positions[v]
            total_cost += abs(pos_u - pos_v)
        
        if total_cost <= B:
            # Found a valid assignment
            result = [0] * n
            for mod in range(n):
                result[mod] = positions[mod] + 1  # Convert back to 1-based indexing
            print("YES")
            print(' '.join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
