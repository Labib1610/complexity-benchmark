```python
import sys
def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); B = int(next(it))
    edges = [(int(next(it))-1, int(next(it))-1) for _ in range(m)]
    deg = [0]*n
    for u,v in edges:
        deg[u]+=1; deg[v]+=1
    # generate slot order: middle outward
    slots = []
    mid = (n-1)//2
    left = mid
    right = mid+1
    while left>=0 or right<n:
        if left>=0:
            slots.append(left+1)
            left-=1
        if right<n:
            slots.append(right+1)
            right+=1
    # order nodes by degree descending
    nodes = list(range(n))
    nodes.sort(key=lambda x: -deg[x])
    perm = [0]*n
    for node, slot in zip(nodes, slots):
        perm[node]=slot
    total = 0
    for u,v in edges:
        total += abs(perm[u]-perm[v])
        if total > B:
            break
    if total<=B:
        print("YES")
        print(' '.join(map(str, perm)))
        return
    # try opposite slot order
    slots_rev = list(reversed(slots))
    for node, slot in zip(nodes, slots_rev):
        perm[node]=slot
    total=0
    for u,v in edges:
        total+=abs(perm[u]-perm[v])
        if total>B: break
    if total<=B:
        print("YES")
        print(' '.join(map(str, perm)))
    else:
        print("NO")
if __name__ == "__main__":
    solve()
```
