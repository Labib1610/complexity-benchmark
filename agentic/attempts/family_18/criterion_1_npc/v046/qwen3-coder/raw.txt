from itertools import permutations

def solve():
    n, m, B = map(int, input().split())
    
    # Read all the links
    links = []
    for _ in range(m):
        x, y = map(int, input().split())
        links.append((x-1, y-1))  # Convert to 0-indexed
    
    # Try all possible permutations of bay assignments
    for perm in permutations(range(n)):
        # Calculate total wire length for this permutation
        total_length = 0
        valid = True
        
        for x, y in links:
            # Get the bay positions for modules x and y
            pos_x = perm[x]
            pos_y = perm[y]
            # Add the wire length between these positions
            total_length += abs(pos_x - pos_y)
            
            # Early termination if we already exceed the budget
            if total_length > B:
                valid = False
                break
        
        if valid and total_length <= B:
            print("YES")
            # Convert back to 1-indexed and print the assignment
            result = [str(pos + 1) for pos in perm]
            print(" ".join(result))
            return
    
    print("NO")

solve()
