```python
import sys, random, time

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    def total_cost(p):
        s = 0
        for u, v in edges:
            s += abs(p[u] - p[v])
        return s

    # start with identity permutation (1..n)
    best_perm = list(range(n))
    best_cost = total_cost(best_perm)
    if best_cost <= B:
        print("YES")
        print(' '.join(str(x + 1) for x in best_perm))
        return

    start_time = time.time()
    TIME_LIMIT = 4.8  # a bit less than overall limit

    # try several random restarts with simple hill climbing
    while time.time() - start_time < TIME_LIMIT:
        perm = list(range(n))
        random.shuffle(perm)
        cur_cost = total_cost(perm)

        improved = True
        while improved and time.time() - start_time < TIME_LIMIT:
            improved = False
            # try random pair swaps, accept if better
            for _ in range(200):
                i, j = random.sample(range(n), 2)
                perm[i], perm[j] = perm[j], perm[i]
                new_cost = total_cost(perm)
                if new_cost < cur_cost:
                    cur_cost = new_cost
                    improved = True
                    if cur_cost <= B:
                        print("YES")
                        print(' '.join(str(x + 1) for x in perm))
                        return
                else:
                    perm[i], perm[j] = perm[j], perm[i]  # revert

        if cur_cost < best_cost:
            best_cost = cur_cost
            best_perm = perm[:]

    # if not found feasible, output NO
    print("NO")

if __name__ == "__main__":
    solve()
```
