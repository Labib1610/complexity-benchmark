import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    B = int(data[2])
    
    edges = []
    for i in range(m):
        u = int(data[3 + 2*i]) - 1  # Convert to 0-based indexing
        v = int(data[3 + 2*i + 1]) - 1
        edges.append((u, v))
    
    # We'll try a greedy approach first: place nodes connected to node 0 near the center
    # But since that may not work, we'll also try small examples with brute force
    
    if n <= 8:
        # For small n, try all permutations
        for perm in permutations(range(n)):
            total = 0
            pos = [0] * n
            for idx, p in enumerate(perm):
                pos[p] = idx + 1  # sector numbering from 1 to n
            for u, v in edges:
                total += abs(pos[u] - pos[v])
                if total > B:
                    break
            if total <= B:
                print("YES")
                print(' '.join(map(str, perm)))
                return
        print("NO")
        return
    
    # For larger n, use heuristic: BFS/DFS or greedy placement
    # Let's try a greedy approach based on degrees
    degree = [0] * n
    adj = [[] for _ in range(n)]
    for u,v in edges:
        degree[u] +=1
        degree[v] +=1
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to place high-degree nodes near the center
    # Simple greedy: sort nodes by degree descending, place them in order near center
    nodes_sorted = list(range(n))
    nodes_sorted.sort(key=lambda x: degree[x], reverse=True)
    
    # Initialize positions array
    pos = [0] * n
    sector_used = [False] * (n+1)  # 1-based indexing
    
    # Place first node in the middle
    mid = (n+1)//2
    first_node = nodes_sorted[0]
    pos[first_node] = mid
    sector_used[mid] = True
    
    # Now, for each remaining node, try to place it in the best available spot
    queue = [first_node]
    placed = {first_node}
    
    while queue:
        current = queue.pop(0)
        neighbors = adj[current]
        for neighbor in neighbors:
            if neighbor not in placed:
                # Find best position for neighbor adjacent to current
                current_pos = pos[current]
                best_pos = -1
                best_score = float('inf')
                for s in range(1, n+1):
                    if not sector_used[s]:
                        score = abs(s - current_pos)
                        if score < best_score:
                            best_score = score
                            best_pos = s
                if best_pos != -1:
                    pos[neighbor] = best_pos
                    sector_used[best_pos] = True
                    placed.add(neighbor)
                    queue.append(neighbor)
    
    # If some nodes are still unplaced, place them anywhere available
    if len(placed) < n:
        available_sectors = [s for s in range(1,n+1) if not sector_used[s]]
        unplaced_nodes = [node for node in range(n) if node not in placed]
        for i, node in enumerate(unplaced_nodes):
            pos[node] = available_sectors[i]
    
    # Calculate total cost
    total = 0
    for u,v in edges:
        total += abs(pos[u] - pos[v])
    
    if total <= B:
        print("YES")
        result = [0] * n
        for i in range(n):
            result[i] = pos[i]
        print(' '.join(map(str, result)))
        return
    
    # If greedy fails, let's try another approach: simulated annealing or random sampling
    # But given time constraints, we'll just do a few random permutations for larger n
    import random
    random.seed(42)
    for _ in range(min(10000, 100000//n)):
        perm = list(range(1,n+1))
        random.shuffle(perm)
        pos_random = [0] * n
        for idx, p in enumerate(perm):
            pos_random[idx] = p
        total_rand = 0
        for u,v in edges:
            total_rand += abs(perm[u] - perm[v])
        if total_rand <= B:
            print("YES")
            print(' '.join(map(str, perm)))
            return
            
    print("NO")

if __name__ == "__main__":
    main()
