import sys
from array import array
from collections import deque


def score_order(order, adj_masks):
    k = len(order)
    pos = [0] * k
    for i, v in enumerate(order):
        pos[v] = i

    total = 0
    for u in range(k):
        mask = adj_masks[u] & ~((1 << (u + 1)) - 1)
        while mask:
            bit = mask & -mask
            v = bit.bit_length() - 1
            total += abs(pos[u] - pos[v])
            mask ^= bit
    return total


def heuristic_order(adj_masks):
    k = len(adj_masks)
    if k <= 1:
        return list(range(k)), 0

    degrees = [x.bit_count() for x in adj_masks]
    candidates = [list(range(k))]

    vertices = sorted(range(k), key=lambda x: (-degrees[x], x))
    slots = sorted(range(k), key=lambda x: (abs(2 * x - (k - 1)), x))
    centered = [0] * k
    for v, p in zip(vertices, slots):
        centered[p] = v
    candidates.append(centered)

    starts = {
        min(range(k), key=lambda x: (degrees[x], x)),
        max(range(k), key=lambda x: (degrees[x], -x)),
    }

    for start in starts:
        seen = [False] * k
        seen[start] = True
        q = deque([start])
        bfs = []
        while q:
            u = q.popleft()
            bfs.append(u)
            neighbors = []
            mask = adj_masks[u]
            while mask:
                bit = mask & -mask
                v = bit.bit_length() - 1
                mask ^= bit
                if not seen[v]:
                    seen[v] = True
                    neighbors.append(v)
            neighbors.sort(key=lambda x: (degrees[x], x))
            for v in neighbors:
                seen[v] = True
                q.append(v)
        candidates.append(bfs)

    best = candidates[0]
    best_cost = score_order(best, adj_masks)
    for candidate in candidates[1:]:
        value = score_order(candidate, adj_masks)
        if value < best_cost:
            best_cost = value
            best = candidate
    return best, best_cost


def special_solution(adj_masks, edge_count):
    k = len(adj_masks)
    if k <= 1:
        return list(range(k)), 0

    degrees = [mask.bit_count() for mask in adj_masks]

    if edge_count == k * (k - 1) // 2:
        cost = k * (k * k - 1) // 6
        return list(range(k)), cost

    if edge_count == k - 1 and max(degrees) <= 2:
        start = degrees.index(1)
        order = []
        previous = -1
        current = start
        while current != -1:
            order.append(current)
            nxt = -1
            mask = adj_masks[current]
            while mask:
                bit = mask & -mask
                v = bit.bit_length() - 1
                mask ^= bit
                if v != previous:
                    nxt = v
                    break
            previous, current = current, nxt
        return order, k - 1

    if edge_count == k and all(d == 2 for d in degrees):
        order = []
        previous = -1
        current = 0
        for _ in range(k):
            order.append(current)
            mask = adj_masks[current]
            nxt = -1
            while mask:
                bit = mask & -mask
                v = bit.bit_length() - 1
                mask ^= bit
                if v != previous and (len(order) == 1 or v != order[0]):
                    nxt = v
                    break
            previous, current = current, nxt
        return order, 2 * k - 2

    centers = [i for i, d in enumerate(degrees) if d == k - 1]
    if edge_count == k - 1 and centers and all(
        degrees[i] == 1 or i == centers[0] for i in range(k)
    ):
        center = centers[0]
        leaves = [i for i in range(k) if i != center]
        center_position = (k - 1) // 2
        order = leaves[:center_position] + [center] + leaves[center_position:]
        cost = sum(abs(i - center_position) for i in range(k) if i != center_position)
        return order, cost

    return None


def exact_solution(adj_masks, edge_count, budget):
    k = len(adj_masks)
    if k <= 1:
        return list(range(k)), 0

    maximum_cost = edge_count * (k - 1)
    cap = min(budget + 1, maximum_cost + 1)
    cut_bits = max(1, edge_count.bit_length())
    cut_mask = (1 << cut_bits) - 1

    size = 1 << k
    use_array = ((cap << cut_bits) | edge_count) < (1 << 64)
    if use_array:
        packed = array('Q', [0]) * size
    else:
        packed = [0] * size

    for mask in range(1, size):
        low = mask & -mask
        v = low.bit_length() - 1
        rest = mask ^ low

        previous_cut = packed[rest] & cut_mask
        cut = previous_cut + adj_masks[v].bit_count() - 2 * (
            adj_masks[v] & rest
        ).bit_count()

        best = cap
        choices = mask
        while choices:
            bit = choices & -choices
            previous_dp = packed[mask ^ bit] >> cut_bits
            if previous_dp < best:
                best = previous_dp
            choices ^= bit

        value = cut + best
        if value > cap:
            value = cap
        packed[mask] = cut | (value << cut_bits)

    full = size - 1
    optimum = packed[full] >> cut_bits
    if optimum > budget:
        return None

    reverse_order = []
    mask = full
    while mask:
        current_dp = packed[mask] >> cut_bits
        current_cut = packed[mask] & cut_mask
        choices = mask
        found = False
        while choices:
            bit = choices & -choices
            rest = mask ^ bit
            previous_dp = packed[rest] >> cut_bits
            if current_dp == current_cut + previous_dp:
                reverse_order.append(bit.bit_length() - 1)
                mask = rest
                found = True
                break
            choices ^= bit
        if not found:
            return None

    reverse_order.reverse()
    return reverse_order, optimum


def output_allocation(component_data, local_orders, n):
    position = [0] * n
    sector = 1
    for data, order in zip(component_data, local_orders):
        vertices = data[0]
        for local_vertex in order:
            position[vertices[local_vertex]] = sector
            sector += 1

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, position)) + "\n")


def main():
    input_data = list(map(int, sys.stdin.buffer.read().split()))
    if not input_data:
        return

    n, m, budget = input_data[0], input_data[1], input_data[2]
    adjacency = [[] for _ in range(n)]
    edges = []
    identity_cost = 0

    index = 3
    for _ in range(m):
        u = input_data[index] - 1
        v = input_data[index + 1] - 1
        index += 2
        adjacency[u].append(v)
        adjacency[v].append(u)
        edges.append((u, v))
        identity_cost += abs(u - v)

    if budget < m:
        print("NO")
        return

    if identity_cost <= budget:
        print("YES")
        print(*range(1, n + 1))
        return

    visited = [False] * n
    components = []

    for start in range(n):
        if visited[start]:
            continue
        visited[start] = True
        stack = [start]
        vertices = []
        while stack:
            u = stack.pop()
            vertices.append(u)
            for v in adjacency[u]:
                if not visited[v]:
                    visited[v] = True
                    stack.append(v)
        components.append(vertices)

    component_data = []
    for vertices in components:
        k = len(vertices)
        local_index = {v: i for i, v in enumerate(vertices)}
        adj_masks = [0] * k
        degree_sum = 0

        for i, u in enumerate(vertices):
            mask = 0
            for v in adjacency[u]:
                j = local_index.get(v)
                if j is not None:
                    mask |= 1 << j
            adj_masks[i] = mask
            degree_sum += mask.bit_count()

        edge_count = degree_sum // 2
        component_data.append((vertices, adj_masks, edge_count))

    heuristic_orders = []
    heuristic_total = 0
    for _, adj_masks, _ in component_data:
        order, cost = heuristic_order(adj_masks)
        heuristic_orders.append(order)
        heuristic_total += cost

    if heuristic_total <= budget:
        output_allocation(component_data, heuristic_orders, n)
        return

    exact_orders = []
    exact_total = 0

    for _, adj_masks, edge_count in component_data:
        remaining = budget - exact_total
        if remaining < 0:
            print("NO")
            return

        result = special_solution(adj_masks, edge_count)
        if result is None:
            result = exact_solution(adj_masks, edge_count, remaining)

        if result is None:
            print("NO")
            return

        order, cost = result
        exact_total += cost
        if exact_total > budget:
            print("NO")
            return
        exact_orders.append(order)

    output_allocation(component_data, exact_orders, n)


if __name__ == "__main__":
    main()
