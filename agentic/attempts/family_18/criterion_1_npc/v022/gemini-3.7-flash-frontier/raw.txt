import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    adj = [0] * n
    deg = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        deg[u] += 1
        deg[v] += 1

    size = 1 << n
    dp = [float("inf")] * size
    parent = [0] * size
    cut = [0] * size

    dp[0] = 0
    for mask in range(1, size):
        v = (mask & -mask).bit_length() - 1
        prev_mask = mask ^ (1 << v)
        cut[mask] = (
            cut[prev_mask] + deg[v] - 2 * (adj[v] & prev_mask).bit_count()
        )

    for mask in range(size - 1):
        cur_d = dp[mask]
        if cur_d > B:
            continue
        c = cut[mask]
        rem = ((1 << n) - 1) ^ mask
        while rem:
            v = (rem & -rem).bit_length() - 1
            rem ^= 1 << v
            nxt_mask = mask | (1 << v)
            val = cur_d + c
            if val < dp[nxt_mask]:
                dp[nxt_mask] = val
                parent[nxt_mask] = v

    full = size - 1
    if dp[full] <= B:
        print("YES")
        ans = [0] * n
        curr = full
        for pos in range(n, 0, -1):
            v = parent[curr]
            ans[v] = pos
            curr ^= 1 << v
        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
