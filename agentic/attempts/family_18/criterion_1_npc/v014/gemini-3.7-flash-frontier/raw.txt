import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    adj = [0] * n
    deg = [0] * n

    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        deg[u] += 1
        deg[v] += 1

    INF = 10**9
    size = 1 << n
    dp = [INF] * size
    parent = [0] * size
    cut = [0] * size

    dp[0] = 0

    for mask in range(1, size):
        lowbit = mask & -mask
        u = lowbit.bit_length() - 1
        prev_mask = mask ^ lowbit
        cut[mask] = cut[prev_mask] + deg[u] - 2 * (prev_mask & adj[u]).bit_count()

    full_mask = size - 1
    for mask in range(size):
        c = dp[mask]
        if c > B:
            continue
        avail = full_mask ^ mask
        while avail:
            lowbit = avail & -avail
            avail ^= lowbit
            u = lowbit.bit_length() - 1
            next_mask = mask | lowbit
            val = c + cut[next_mask]
            if val < dp[next_mask]:
                dp[next_mask] = val
                parent[next_mask] = u

    if dp[full_mask] <= B:
        print("YES")
        mask = full_mask
        order = []
        while mask > 0:
            u = parent[mask]
            order.append(u)
            mask ^= 1 << u
        order.reverse()

        slots = [0] * n
        for slot_idx, mod in enumerate(order):
            slots[mod] = slot_idx + 1
        print(*(slots[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
