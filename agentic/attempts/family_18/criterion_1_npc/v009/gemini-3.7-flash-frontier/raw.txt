import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    B = int(input_data[2])

    if n == 0:
        print("YES\n")
        return
    if n == 1:
        if B >= 0:
            print("YES")
            print("1")
        else:
            print("NO")
        return

    adj_list = [[] for _ in range(n)]
    deg = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj_list[u].append(v)
            adj_list[v].append(u)
            deg[u] += 1
            deg[v] += 1

    size = 1 << n
    dp = [10**15] * size
    parent = [-1] * size
    cut = [0] * size

    dp[0] = 0

    for mask in range(size - 1):
        c_dp = dp[mask]
        c_cut = cut[mask]

        avail = ((size - 1) ^ mask)
        while avail:
            v = (avail & -avail).bit_length() - 1
            avail &= avail - 1
            nmask = mask | (1 << v)

            cnt = 0
            for u in adj_list[v]:
                if (mask >> u) & 1:
                    cnt += 1

            ncut = c_cut + deg[v] - 2 * cnt
            cut[nmask] = ncut

            ncost = c_dp + ncut
            if ncost < dp[nmask]:
                dp[nmask] = ncost
                parent[nmask] = v

    if dp[size - 1] <= B:
        print("YES")
        order = []
        curr = size - 1
        while curr > 0:
            v = parent[curr]
            order.append(v)
            curr ^= 1 << v
        order.reverse()

        pos = [0] * n
        for p, v in enumerate(order):
            pos[v] = p + 1
        print(*(pos[i] for i in range(n)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
