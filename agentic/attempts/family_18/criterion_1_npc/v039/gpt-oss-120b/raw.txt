```python
import sys, itertools

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    B = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    # quick lower bound: each edge at least distance 1
    if B < m:
        print("NO")
        return

    # try identity permutation first
    perm = list(range(n))
    total = 0
    for u, v in edges:
        total += abs(perm[u] - perm[v])
    if total <= B:
        print("YES")
        print(' '.join(str(p + 1) for p in perm))
        return

    # if n is small, brute force
    if n <= 9:
        for p in itertools.permutations(range(n)):
            total = 0
            for u, v in edges:
                total += abs(p[u] - p[v])
                if total > B:
                    break
            if total <= B:
                print("YES")
                print(' '.join(str(p[i] + 1) for i in range(n)))
                return
        print("NO")
        return

    # for larger n, use a simple heuristic: sort by degree (high degree in middle)
    deg = [0] * n
    for u, v in edges:
        deg[u] += 1
        deg[v] += 1
    order = sorted(range(n), key=lambda x: -deg[x])
    # place them alternating from center outwards
    pos = [0] * n
    left = (n - 1) // 2
    right = left + 1
    for idx, node in enumerate(order):
        if idx % 2 == 0:
            pos[node] = left
            left -= 1
        else:
            pos[node] = right
            right += 1
    total = sum(abs(pos[u] - pos[v]) for u, v in edges)
    if total <= B:
        print("YES")
        # convert positions to permutation: we need slot of each server
        # pos gives slot index (0-based). Need to output slot numbers 1..n
        print(' '.join(str(pos[i] + 1) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
