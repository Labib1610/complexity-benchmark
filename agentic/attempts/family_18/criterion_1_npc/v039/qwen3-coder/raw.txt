```python
import sys
from itertools import permutations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx+1])
    B = int(data[idx+2])
    idx += 3
    
    connections = []
    for _ in range(m):
        u = int(data[idx]) - 1  # Convert to 0-based
        v = int(data[idx+1]) - 1
        connections.append((u, v))
        idx += 2

    # Try all permutations
    for perm in permutations(range(n)):
        total_length = 0
        for u, v in connections:
            pos_u = perm[u]
            pos_v = perm[v]
            total_length += abs(pos_u - pos_v)
            if total_length > B:
                break
        if total_length <= B:
            # Found a valid permutation
            result = [str(x + 1) for x in perm]  # Convert back to 1-based
            print("YES")
            print(" ".join(result))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
