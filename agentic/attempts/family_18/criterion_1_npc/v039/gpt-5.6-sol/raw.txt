```python
import sys
from collections import deque
from array import array

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 3:
        return

    n, m, budget = data[0], data[1], data[2]

    parent_dsu = list(range(n))
    size_dsu = [1] * n

    def find(x):
        while parent_dsu[x] != x:
            parent_dsu[x] = parent_dsu[parent_dsu[x]]
            x = parent_dsu[x]
        return x

    def union(a, b):
        a = find(a)
        b = find(b)
        if a == b:
            return
        if size_dsu[a] < size_dsu[b]:
            a, b = b, a
        parent_dsu[b] = a
        size_dsu[a] += size_dsu[b]

    adj = [dict() for _ in range(n)]
    nonloop_edges = 0

    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u == v:
            continue
        nonloop_edges += 1
        adj[u][v] = adj[u].get(v, 0) + 1
        adj[v][u] = adj[v].get(u, 0) + 1
        union(u, v)

    if nonloop_edges > budget:
        print("NO")
        return

    groups = {}
    isolated = []
    for v in range(n):
        if not adj[v]:
            isolated.append(v)
        else:
            r = find(v)
            groups.setdefault(r, []).append(v)

    components = list(groups.values())

    def order_cost(order):
        pos = {v: i for i, v in enumerate(order)}
        total = 0
        for v in order:
            pv = pos[v]
            for u, w in adj[v].items():
                if v < u:
                    total += w * abs(pv - pos[u])
        return total

    def improve_order(order):
        k = len(order)
        if k <= 2:
            return order[:]

        order = order[:]
        pos = {v: i for i, v in enumerate(order)}

        for _ in range(12):
            changed = False
            i = 0
            while i + 1 < k:
                a = order[i]
                b = order[i + 1]
                delta = 0

                for x, w in adj[a].items():
                    if x != b:
                        px = pos[x]
                        delta += w * (abs(i + 1 - px) - abs(i - px))

                for x, w in adj[b].items():
                    if x != a:
                        px = pos[x]
                        delta += w * (abs(i - px) - abs(i + 1 - px))

                if delta < 0:
                    order[i], order[i + 1] = b, a
                    pos[a], pos[b] = i + 1, i
                    changed = True
                    if i:
                        i -= 1
                        continue
                i += 1

            if not changed:
                break

        return order

    def heuristic_order(vertices):
        start = min(vertices, key=lambda x: (sum(adj[x].values()), len(adj[x]), x))
        seen = {start}
        q = deque([start])
        bfs = []

        while q:
            v = q.popleft()
            bfs.append(v)
            neighbors = sorted(
                adj[v],
                key=lambda x: (sum(adj[x].values()), len(adj[x]), x)
            )
            for u in neighbors:
                if u not in seen:
                    seen.add(u)
                    q.append(u)

        candidates = [
            bfs,
            list(reversed(bfs)),
            sorted(vertices, key=lambda x: (sum(adj[x].values()), x)),
            sorted(vertices),
        ]

        best_order = None
        best_cost = None
        for candidate in candidates:
            candidate = improve_order(candidate)
            c = order_cost(candidate)
            if best_cost is None or c < best_cost:
                best_cost = c
                best_order = candidate

        return best_order, best_cost

    heuristic_results = []
    heuristic_total = 0
    for comp in components:
        order, cost = heuristic_order(comp)
        heuristic_results.append((comp, order, cost))
        heuristic_total += cost

    if heuristic_total <= budget:
        final_order = []
        for _, order, _ in heuristic_results:
            final_order.extend(order)
        final_order.extend(isolated)

        positions = [0] * n
        for i, v in enumerate(final_order, 1):
            positions[v] = i

        print("YES")
        print(*positions)
        return

    def exact_dense(vertices, upper_limit):
        k = len(vertices)
        index = {v: i for i, v in enumerate(vertices)}
        degrees = [0] * k
        local_rows = [[] for _ in range(k)]
        simple = True

        for i, v in enumerate(vertices):
            for u, w in adj[v].items():
                j = index[u]
                local_rows[i].append((j, w))
                degrees[i] += w
                if w != 1:
                    simple = False

        adj_masks = None
        if simple:
            adj_masks = [0] * k
            for i in range(k):
                mask = 0
                for j, _ in local_rows[i]:
                    mask |= 1 << j
                adj_masks[i] = mask

        total_masks = 1 << k
        values = array('q', [0]) * total_masks

        if simple:
            for mask in range(1, total_masks):
                bit = mask & -mask
                i = bit.bit_length() - 1
                prev = mask ^ bit
                inside = (adj_masks[i] & prev).bit_count()
                values[mask] = values[prev] + degrees[i] - 2 * inside
        else:
            for mask in range(1, total_masks):
                bit = mask & -mask
                i = bit.bit_length() - 1
                prev = mask ^ bit
                inside = 0
                for j, w in local_rows[i]:
                    if prev & (1 << j):
                        inside += w
                values[mask] = values[prev] + degrees[i] - 2 * inside

        choice = bytearray(total_masks)
        inf = upper_limit + 1

        for mask in range(1, total_masks):
            cut = values[mask]
            bits = mask
            best = inf
            best_i = 0

            while bits:
                bit = bits & -bits
                i = bit.bit_length() - 1
                candidate = values[mask ^ bit]
                if candidate < best:
                    best = candidate
                    best_i = i
                bits ^= bit

            val = best + cut
            if val > inf:
                val = inf
            values[mask] = val
            choice[mask] = best_i

        full = total_masks - 1
        best_cost = values[full]
        if best_cost > upper_limit:
            return None, best_cost

        reversed_order = []
        mask = full
        while mask:
            i = choice[mask]
            reversed_order.append(vertices[i])
            mask ^= 1 << i

        reversed_order.reverse()
        return reversed_order, best_cost

    def exact_branch(vertices, heuristic, heuristic_cost, upper_limit):
        k = len(vertices)
        index = {v: i for i, v in enumerate(vertices)}
        degrees = [0] * k
        rows = [[] for _ in range(k)]
        simple = True
        edge_count = 0

        for i, v in enumerate(vertices):
            for u, w in adj[v].items():
                j = index[u]
                rows[i].append((j, w))
                degrees[i] += w
                if i < j:
                    edge_count += w
                if w != 1:
                    simple = False

        masks = [0] * k
        if simple:
            for i in range(k):
                am = 0
                for j, _ in rows[i]:
                    am |= 1 << j
                masks[i] = am

        rank = [0] * k
        for r, v in enumerate(heuristic):
            rank[index[v]] = r

        best = min(heuristic_cost, upper_limit + 1)
        best_path = [index[v] for v in heuristic] if heuristic_cost <= upper_limit else None

        if best == edge_count:
            return heuristic[:], best

        full = (1 << k) - 1
        memo = {}
        path = []

        sys.setrecursionlimit(max(1000, k + 100))

        def weight_to_mask(i, mask):
            if simple:
                return (masks[i] & mask).bit_count()
            total = 0
            for j, w in rows[i]:
                if mask & (1 << j):
                    total += w
            return total

        def dfs(mask, cut, accumulated, remaining_internal):
            nonlocal best, best_path

            if accumulated + remaining_internal >= best:
                return

            old = memo.get(mask)
            if old is not None and old <= accumulated:
                return
            memo[mask] = accumulated

            if mask == full:
                best = accumulated
                best_path = path[:]
                return

            remaining = full ^ mask
            candidates = []

            bits = remaining
            while bits:
                bit = bits & -bits
                i = bit.bit_length() - 1
                new_cut = cut + degrees[i] - 2 * weight_to_mask(i, mask)
                rem_without = remaining ^ bit
                removed_internal = weight_to_mask(i, rem_without)
                new_internal = remaining_internal - removed_internal
                new_acc = accumulated + new_cut
                if new_acc + new_internal < best:
                    candidates.append((new_cut, rank[i], i, new_acc, new_internal))
                bits ^= bit

            candidates.sort()

            for _, _, i, new_acc, new_internal in candidates:
                path.append(i)
                dfs(mask | (1 << i),
                    cut + degrees[i] - 2 * weight_to_mask(i, mask),
                    new_acc,
                    new_internal)
                path.pop()
                if best == edge_count:
                    return

        dfs(0, 0, 0, edge_count)

        if best_path is None:
            return None, upper_limit + 1
        return [vertices[i] for i in best_path], best

    exact_results = []
    exact_total = 0

    # Larger components first can expose impossibility earlier.
    heuristic_results.sort(key=lambda item: len(item[0]), reverse=True)

    for comp, heuristic, heuristic_cost in heuristic_results:
        component_edges = sum(sum(adj[v].values()) for v in comp) // 2

        if exact_total + component_edges > budget:
            print("NO")
            return

        if heuristic_cost == component_edges:
            order, cost = heuristic, heuristic_cost
        elif len(comp) <= 21:
            order, cost = exact_dense(comp, budget - exact_total)
        else:
            order, cost = exact_branch(
                comp, heuristic, heuristic_cost, budget - exact_total
            )

        if order is None or exact_total + cost > budget:
            print("NO")
            return

        exact_results.append(order)
        exact_total += cost

    if exact_total > budget:
        print("NO")
        return

    final_order = []
    for order in exact_results:
        final_order.extend(order)
    final_order.extend(isolated)

    positions = [0] * n
    for i, v in enumerate(final_order, 1):
        positions[v] = i

    print("YES")
    print(*positions)

if __name__ == "__main__":
    main()
```
