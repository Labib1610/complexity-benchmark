import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    triples = []
    for _ in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        triples.append((a, b, c))
        index += 3
    
    n = 3 * q
    
    # Build graph: for each element, list of triples containing it
    element_to_triples = defaultdict(list)
    for i, (a, b, c) in enumerate(triples):
        element_to_triples[a].append(i)
        element_to_triples[b].append(i)
        element_to_triples[c].append(i)
    
    # Try to find a matching that covers all elements exactly once
    # This is essentially a exact cover problem, we can use backtracking
    
    def backtrack(pos, selected, used_elements, used_triples):
        if pos > n:
            if len(selected) == q:
                return selected[:]
            return None
        
        # If element pos is already covered, move to next
        if pos in used_elements:
            return backtrack(pos + 1, selected, used_elements, used_triples)
        
        # Try each triple containing pos
        for triple_index in element_to_triples.get(pos, []):
            if triple_index in used_triples:
                continue
            a, b, c = triples[triple_index]
            # Check if all elements of this triple are still available
            if a in used_elements or b in used_elements or c in used_elements:
                continue
            
            # Use this triple
            selected.append(triple_index)
            used_triples.add(triple_index)
            used_elements.add(a)
            used_elements.add(b)
            used_elements.add(c)
            
            res = backtrack(pos + 1, selected, used_elements, used_triples)
            if res is not None:
                return res
            
            # Backtrack
            selected.pop()
            used_triples.remove(triple_index)
            used_elements.remove(a)
            used_elements.remove(b)
            used_elements.remove(c)
        
        return None
    
    # Optimization: start with elements that have the least number of triples
    # This can help prune the search space early
    element_count = [len(element_to_triples[i]) for i in range(1, n+1)]
    sorted_elements = sorted(range(1, n+1), key=lambda x: element_count[x-1])
    
    # However, for simplicity and given the constraints, we'll proceed with normal backtracking
    # starting from 1
    result = backtrack(1, [], set(), set())
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for idx in result:
            a, b, c = triples[idx]
            print(a, b, c)

if __name__ == "__main__":
    main()
