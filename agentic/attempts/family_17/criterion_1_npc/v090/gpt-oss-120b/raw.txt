import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        triples.append((a, b, c))

    n_cols = 3 * q          # each person is a column
    n_rows = len(triples)   # each triple is a row

    # ---------- DLX structures ----------
    max_nodes = n_rows * 3 + n_cols + 1
    U = [0] * max_nodes
    D = [0] * max_nodes
    L = [0] * max_nodes
    R = [0] * max_nodes
    C = [0] * max_nodes          # column header for each node
    S = [0] * (n_cols + 1)       # size of each column (header 0 is root)
    row_id = [0] * max_nodes     # which row a node belongs to

    # header nodes: 0 is root, 1..n_cols are column headers
    for i in range(n_cols + 1):
        U[i] = D[i] = i
        L[i] = i - 1
        R[i] = i + 1
    L[0] = n_cols
    R[n_cols] = 0

    node_cnt = n_cols + 1  # next free index

    def add_node(row, col):
        nonlocal node_cnt
        idx = node_cnt
        node_cnt += 1
        C[idx] = col
        row_id[idx] = row

        # vertical links
        U[idx] = U[col]
        D[idx] = col
        D[U[col]] = idx
        U[col] = idx
        S[col] += 1

        # horizontal links (will be linked later)
        L[idx] = R[idx] = idx
        return idx

    # build rows
    row_nodes = []   # store one node of each row to later close the horizontal circle
    for r, (a, b, c) in enumerate(triples):
        first = add_node(r, a + 1)
        second = add_node(r, b + 1)
        third = add_node(r, c + 1)

        # link horizontally
        L[second] = first
        R[first] = second
        L[third] = second
        R[second] = third
        L[first] = third
        R[third] = first

        row_nodes.append(first)

    # ---------- DLX algorithm ----------
    solution_rows = []

    def cover(col):
        # remove column header
        L[R[col]] = L[col]
        R[L[col]] = R[col]
        # remove rows
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[col]] = col
        R[L[col]] = col

    def search(k):
        if R[0] == 0:          # all columns covered
            return True
        # choose column with minimal size
        c = R[0]
        min_s = S[c]
        j = R[c]
        while j != 0:
            if S[j] < min_s:
                min_s = S[j]
                c = j
                if min_s == 0:
                    break
            j = R[j]
        if min_s == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution_rows.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            if search(k + 1):
                return True
            # backtrack
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution_rows.pop()
            r = D[r]
        uncover(c)
        return False

    possible = search(0)

    if not possible:
        print("NO")
        return

    print("YES")
    # output selected triples
    used = set()
    for rid in solution_rows:
        a, b, c = triples[rid]
        # convert back to 1‑based
        print(a + 1, b + 1, c + 1)

if __name__ == "__main__":
    solve()
