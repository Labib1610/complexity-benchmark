import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    blueprints = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        blueprints.append((a, b, c))
    
    n = 3 * q
    
    # Build adjacency list: for each pair (u, v), store list of third vertices w
    # such that (u, v, w) is a valid blueprint (with u < v < w)
    adj = {}
    for a, b, c in blueprints:
        # Store all orderings for easy lookup
        # (a, b) -> c
        if (a, b) not in adj:
            adj[(a, b)] = []
        adj[(a, b)].append(c)
        # (a, c) -> b
        if (a, c) not in adj:
            adj[(a, c)] = []
        adj[(a, c)].append(b)
        # (b, c) -> a
        if (b, c) not in adj:
            adj[(b, c)] = []
        adj[(b, c)].append(a)
    
    # Sort the lists for deterministic behavior
    for key in adj:
        adj[key].sort()
    
    used = [False] * (n + 1)
    result = []
    
    def backtrack(idx):
        # Find the smallest unused vertex
        while idx <= n and used[idx]:
            idx += 1
        if idx > n:
            return True
        
        used[idx] = True
        # Try to find a valid triplet starting with idx
        # We need to find j, k such that idx < j < k and (idx, j, k) is a blueprint
        for j in range(idx + 1, n + 1):
            if used[j]:
                continue
            if (idx, j) not in adj:
                continue
            for k in adj[(idx, j)]:
                if k > j and not used[k]:
                    used[j] = True
                    used[k] = True
                    result.append((idx, j, k))
                    if backtrack(idx + 1):
                        return True
                    result.pop()
                    used[j] = False
                    used[k] = False
        used[idx] = False
        return False
    
    if backtrack(1):
        print("YES")
        for triplet in result:
            print(triplet[0], triplet[1], triplet[2])
    else:
        print("NO")

if __name__ == "__main__":
    solve()
