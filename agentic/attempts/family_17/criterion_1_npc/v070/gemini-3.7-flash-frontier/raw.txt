import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    if 3 * q == 0:
        print("YES")
        return

    triples = []
    seen = set()
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        # Normalize
        tup = tuple(sorted((a, b, c)))
        if tup not in seen:
            seen.add(tup)
            triples.append(tup)

    n = 3 * q
    adj = [[] for _ in range(n + 1)]

    for a, b, c in triples:
        mask = (1 << a) | (1 << b) | (1 << c)
        item = (mask, a, b, c)
        adj[a].append(item)
        adj[b].append(item)
        adj[c].append(item)

    # Initial check: if any element has 0 triples, impossible
    for u in range(1, n + 1):
        if not adj[u]:
            print("NO")
            return

    covered = 0
    solution = []

    def backtrack(num_covered):
        nonlocal covered
        if num_covered == n:
            return True

        best_u = -1
        min_c = 10**9
        best_candidates = None

        for u in range(1, n + 1):
            if (covered >> u) & 1:
                continue
            
            # Filter valid candidates for uncovered element u
            valid = [item for item in adj[u] if not (covered & item[0])]
            count = len(valid)
            
            if count == 0:
                return False
            
            if count < min_c:
                min_c = count
                best_u = u
                best_candidates = valid
                if min_c == 1:
                    break

        for mask, a, b, c in best_candidates:
            covered |= mask
            solution.append((a, b, c))
            if backtrack(num_covered + 3):
                return True
            solution.pop()
            covered ^= mask

        return False

    if backtrack(0):
        print("YES")
        for a, b, c in solution:
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
