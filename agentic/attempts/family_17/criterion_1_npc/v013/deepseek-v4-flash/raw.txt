We can model this as an exact cover problem: we need to select exactly \( q \) trios from the given list such that every operator from 1 to \( 3q \) appears exactly once.

Since \( q \le 200 \), the total number of operators is at most 600. The number of trios can be up to 20,000.  
We can solve this with backtracking + pruning, or use a bitmask-based exact cover approach.

A straightforward backtracking that picks the next unassigned operator and tries trios containing it works well with pruning.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q

    trios = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        trios.append((a, b, c))

    # Group trios by the smallest unassigned operator they contain
    by_first = [[] for _ in range(n)]
    for idx, (a, b, c) in enumerate(trios):
        by_first[a].append(idx)
        by_first[b].append(idx)
        by_first[c].append(idx)

    used = [False] * n
    chosen = []
    trio_list = trios

    def backtrack(start):
        if len(chosen) == q:
            return True
        # find first unused operator
        while start < n and used[start]:
            start += 1
        if start == n:
            return False
        # try trios containing this operator
        for idx in by_first[start]:
            a, b, c = trio_list[idx]
            if used[a] or used[b] or used[c]:
                continue
            used[a] = used[b] = used[c] = True
            chosen.append((a + 1, b + 1, c + 1))
            if backtrack(start + 1):
                return True
            chosen.pop()
            used[a] = used[b] = used[c] = False
        return False

    if backtrack(0):
        print("YES")
        for trio in chosen:
            print(*trio)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
