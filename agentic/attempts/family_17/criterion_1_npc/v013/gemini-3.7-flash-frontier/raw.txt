import sys

def solve():
    sys.setrecursionlimit(2000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    trios = []
    idx = 2
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        trios.append((a, b, c))
        
    # DLX setup
    # 0 is root, 1..n are column headers
    max_nodes = n + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    sz = [0] * (n + 1)
    
    for i in range(n + 1):
        L[i] = (i - 1 + n + 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        
    node_cnt = n + 1
    
    for r_idx, (a, b, c) in enumerate(trios):
        first = node_cnt
        cols = (a, b, c)
        for i, c_idx in enumerate(cols):
            cur = node_cnt
            node_cnt += 1
            col[cur] = c_idx
            row[cur] = r_idx
            sz[c_idx] += 1
            
            # Vertical linking
            U[cur] = U[c_idx]
            D[cur] = c_idx
            D[U[c_idx]] = cur
            U[c_idx] = cur
            
            # Horizontal linking
            if i == 0:
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = L[first]
                R[cur] = first
                R[L[first]] = cur
                L[first] = cur

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    sol = []

    def search():
        if R[0] == 0:
            return True
        # Pick column with minimum size
        c = R[0]
        best_c = c
        min_sz = sz[c]
        curr = R[c]
        while curr != 0:
            if sz[curr] < min_sz:
                min_sz = sz[curr]
                best_c = curr
            curr = R[curr]
        c = best_c
        
        if min_sz == 0:
            return False
            
        cover(c)
        r = D[c]
        while r != c:
            sol.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]
            if search():
                return True
            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            sol.pop()
            r = D[r]
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in sol:
            a, b, c = trios[r_idx]
            out.append(f"{a} {b} {c}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
