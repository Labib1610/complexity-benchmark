import sys

def solve():
    sys.setrecursionlimit(2000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    clusters = []
    idx = 2
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        clusters.append((a, b, c))
        
    elem_to_clusters = [[] for _ in range(n + 1)]
    for c_id, (a, b, c) in enumerate(clusters):
        elem_to_clusters[a].append(c_id)
        elem_to_clusters[b].append(c_id)
        elem_to_clusters[c].append(c_id)
        
    for i in range(1, n + 1):
        if not elem_to_clusters[i]:
            print("NO")
            return

    covered = [False] * (n + 1)
    cluster_removed = [0] * t  # ref count: how many of its elements are covered
    
    solution = []
    
    def cover_cluster(c_id):
        a, b, c = clusters[c_id]
        for elem in (a, b, c):
            covered[elem] = True
            for other_cid in elem_to_clusters[elem]:
                cluster_removed[other_cid] += 1

    def uncover_cluster(c_id):
        a, b, c = clusters[c_id]
        for elem in (a, b, c):
            covered[elem] = False
            for other_cid in elem_to_clusters[elem]:
                cluster_removed[other_cid] -= 1

    def search(rem_q):
        if rem_q == 0:
            return True
        
        # MRV heuristic: find uncovered element with minimum valid clusters
        min_candidates = None
        best_elem = -1
        
        for elem in range(1, n + 1):
            if not covered[elem]:
                valid_count = 0
                for c_id in elem_to_clusters[elem]:
                    if cluster_removed[c_id] == 0:
                        valid_count += 1
                if valid_count == 0:
                    return False
                if min_candidates is None or valid_count < min_candidates:
                    min_candidates = valid_count
                    best_elem = elem
                    if valid_count == 1:
                        break
        
        cands = [c_id for c_id in elem_to_clusters[best_elem] if cluster_removed[c_id] == 0]
        for c_id in cands:
            cover_cluster(c_id)
            solution.append(c_id)
            
            if search(rem_q - 1):
                return True
                
            solution.pop()
            uncover_cluster(c_id)
            
        return False

    if search(q):
        print("YES")
        for c_id in solution:
            a, b, c = clusters[c_id]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
