import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(100000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triples.append((a, b, c))

    num_cols = 3 * q

    # Quick check: if t < q, exact cover is impossible
    if t < q:
        print("NO")
        return

    max_nodes = num_cols + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row = [0] * max_nodes
    col_size = [0] * (num_cols + 1)

    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = (i - 1) % (num_cols + 1)
        R[i] = (i + 1) % (num_cols + 1)
        U[i] = i
        D[i] = i
        C[i] = i

    node_cnt = num_cols + 1
    for r_idx, (a, b, c) in enumerate(triples):
        nodes = []
        for col in (a, b, c):
            curr = node_cnt
            node_cnt += 1
            col_size[col] += 1
            C[curr] = col
            row[curr] = r_idx

            U[curr] = U[col]
            D[curr] = col
            D[U[col]] = curr
            U[col] = curr
            nodes.append(curr)

        L[nodes[0]] = nodes[2]
        R[nodes[0]] = nodes[1]
        L[nodes[1]] = nodes[0]
        R[nodes[1]] = nodes[2]
        L[nodes[2]] = nodes[1]
        R[nodes[2]] = nodes[0]

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size (MRV heuristic)
        c = R[0]
        min_size = col_size[c]
        curr = R[c]
        while curr != 0:
            if col_size[curr] < min_size:
                min_size = col_size[curr]
                c = curr
            curr = R[curr]

        if min_size == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            ans.append(row[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            ans.pop()
            r = D[r]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in ans:
            a, b, c = triples[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
