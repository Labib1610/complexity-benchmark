```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx + 1]) - 1
        c = int(input_data[idx + 2]) - 1
        idx += 3
        triples.append((a, b, c))

    triples = list(set(triples))

    if len(triples) < q:
        print("NO")
        return

    n = 3 * q
    triples_of = [[] for _ in range(n)]
    for i, (a, b, c) in enumerate(triples):
        triples_of[a].append(i)
        triples_of[b].append(i)
        triples_of[c].append(i)

    for i in range(n):
        if not triples_of[i]:
            print("NO")
            return

    covered = [False] * n
    selected = []
    uncovered_count = n

    def search():
        nonlocal uncovered_count
        if uncovered_count == 0:
            return True

        min_candidates = 999999
        best_triples = []

        for u in range(n):
            if not covered[u]:
                cand = []
                for tr_idx in triples_of[u]:
                    a, b, c = triples[tr_idx]
                    if not covered[a] and not covered[b] and not covered[c]:
                        cand.append(tr_idx)
                if len(cand) < min_candidates:
                    min_candidates = len(cand)
                    best_triples = cand
                    if min_candidates <= 1:
                        break

        if min_candidates == 0:
            return False

        for tr_idx in best_triples:
            a, b, c = triples[tr_idx]
            if not covered[a] and not covered[b] and not covered[c]:
                covered[a] = covered[b] = covered[c] = True
                uncovered_count -= 3
                selected.append(tr_idx)

                if search():
                    return True

                selected.pop()
                covered[a] = covered[b] = covered[c] = False
                uncovered_count += 3

        return False

    sys.setrecursionlimit(2000)
    if search():
        print("YES")
        for tr_idx in selected:
            a, b, c = triples[tr_idx]
            print(f"{a + 1} {b + 1} {c + 1}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
