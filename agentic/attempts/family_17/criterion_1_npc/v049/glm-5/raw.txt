```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        
        adj[a].append((b, c))
        adj[b].append((a, c))
        adj[c].append((a, b))
        
    for i in range(1, n + 1):
        adj[i].sort()

    match_to = [0] * (n + 1)
    result = []
    
    for u in range(1, n + 1):
        if match_to[u] == 0:
            found = False
            for v, w in adj[u]:
                if match_to[v] == 0 and match_to[w] == 0:
                    match_to[u] = v
                    match_to[v] = u
                    match_to[w] = w
                    result.append((u, v, w))
                    found = True
                    break
            
            if not found:
                print("NO")
                return

    print("YES")
    for triplet in result:
        print(triplet[0], triplet[1], triplet[2])

if __name__ == '__main__':
    solve()
```
