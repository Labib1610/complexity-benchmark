```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return
    
    n = 3 * q
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(t):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            w = int(next(iterator))
        except StopIteration:
            break
        adj[u].append((v, w))
        adj[v].append((u, w))
        adj[w].append((u, v))
    
    match = [0] * (n + 1)
    
    for u in range(1, n + 1):
        if match[u] != 0:
            continue
        
        queue = deque()
        queue.append(u)
        parent = {}
        found = -1
        
        while queue and found == -1:
            curr = queue.popleft()
            for v, w in adj[curr]:
                if match[v] == 0:
                    found = curr
                    parent[v] = (curr, w)
                    break
                elif v not in parent:
                    parent[v] = (curr, w)
                    queue.append(match[v])
        
        if found == -1:
            print("NO")
            return
        
        path = []
        curr_v = v
        while curr_v != u:
            p, w = parent[curr_v]
            path.append((p, curr_v, w))
            curr_v = p
        
        for p, v, w in path:
            match[p] = w
            match[w] = p
            match[v] = v
    
    if any(match[i] == 0 for i in range(1, n + 1)):
        print("NO")
    else:
        print("YES")
        for i in range(1, n + 1):
            if i < match[i] and i < match[match[i]]:
                print(i, match[i], match[match[i]])

if __name__ == "__main__":
    solve()
```
