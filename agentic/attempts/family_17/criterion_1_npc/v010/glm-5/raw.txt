import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    n = 3 * q
    adj = [[] for _ in range(n + 1)]
    configs = []
    
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        configs.append((a, b, c))
        adj[a].append((b, c))
        adj[b].append((a, c))
        adj[c].append((a, b))
    
    match = [0] * (n + 1)
    
    for u in range(1, n + 1):
        if match[u] != 0:
            continue
        
        found = False
        for v, w in adj[u]:
            if match[v] == 0 and match[w] == 0:
                match[u] = v
                match[v] = w
                match[w] = u
                found = True
                break
        
        if found:
            continue
        
        for v, w in adj[u]:
            if match[v] == 0:
                prev_w = match[w]
                if prev_w == 0:
                    continue
                match[prev_w] = 0
                match[w] = 0
                match[v] = 0
                
                match[u] = v
                match[v] = w
                match[w] = u
                found = True
                break
        
        if found:
            continue
        
        for v, w in adj[u]:
            prev_v = match[v]
            prev_w = match[w]
            if prev_v == 0 or prev_w == 0:
                continue
            
            match[prev_v] = 0
            match[prev_w] = 0
            match[v] = 0
            match[w] = 0
            
            match[u] = v
            match[v] = w
            match[w] = u
            found = True
            break
    
    for i in range(1, n + 1):
        if match[i] == 0:
            print("NO")
            return
    
    print("YES")
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            v = match[i]
            w = match[v]
            print(i, v, w)
            visited[i] = True
            visited[v] = True
            visited[w] = True

if __name__ == "__main__":
    solve()
