from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    configs = []
    person_to_configs = defaultdict(list)
    
    for i in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        index += 3
        config = (a, b, c)
        configs.append(config)
        person_to_configs[a].append(i)
        person_to_configs[b].append(i)
        person_to_configs[c].append(i)
    
    # Try to find a matching that covers all 3q persons
    # This is essentially a exact cover problem, we can model it as finding a maximum matching
    # We'll use backtracking with pruning
    
    # Since q <= 200 and t <= 20000, and we need exactly q triplets,
    # we can try to build the solution incrementally
    
    # Let's model this as a graph matching problem:
    # Each configuration is a node on the right side
    # Each person is a node on the left side
    # We want a perfect matching that uses exactly q edges (configurations)
    
    # More efficient approach: since we must select exactly q configurations to cover 3q people,
    # and each configuration covers 3 people, this is equivalent to finding a matching of size q
    # that covers all vertices.
    
    # Use backtracking with bitmasks for efficiency
    selected = [False] * t
    used_persons = [False] * (3*q + 1)
    result = []
    
    def backtrack(config_idx, count):
        if count == q:
            # Check if all persons are covered
            for i in range(1, 3*q + 1):
                if not used_persons[i]:
                    return False
            return True
            
        if config_idx == t:
            return False
            
        # Pruning: if remaining configs can't fill the needed count
        if count + (t - config_idx) < q:
            return False
            
        # Try including current configuration
        a, b, c = configs[config_idx]
        if not used_persons[a] and not used_persons[b] and not used_persons[c]:
            selected[config_idx] = True
            used_persons[a] = used_persons[b] = used_persons[c] = True
            result.append(config_idx)
            
            if backtrack(config_idx + 1, count + 1):
                return True
                
            # Backtrack
            selected[config_idx] = False
            used_persons[a] = used_persons[b] = used_persons[c] = False
            result.pop()
        
        # Try not including current configuration
        if backtrack(config_idx + 1, count):
            return True
            
        return False
    
    # Optimization: sort configurations by the smallest person id to try "earlier" configurations first
    # This might help in pruning
    indexed_configs = list(enumerate(configs))
    indexed_configs.sort(key=lambda x: min(x[1]))
    sorted_indices = [x[0] for x in indexed_configs]
    configs_sorted = [x[1] for x in indexed_configs]
    configs[:] = configs_sorted
    # Rebuild person_to_configs mapping based on new indices
    person_to_configs = defaultdict(list)
    for i, (a,b,c) in enumerate(configs):
        person_to_configs[a].append(i)
        person_to_configs[b].append(i)
        person_to_configs[c].append(i)
    
    if backtrack(0, 0):
        print("YES")
        for idx in result:
            a, b, c = configs[idx]
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    main()
