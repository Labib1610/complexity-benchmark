import sys
input = sys.stdin.readline

def solve():
    q, t = map(int, input().split())
    n = 3 * q
    
    # Read all triads
    triads = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triads.append((a, b, c))
    
    # If t is small, use recursive backtracking
    if t <= 20:
        used = [False] * (n + 1)
        selected = []
        
        def backtrack(count):
            if count == q:
                return True
            
            # Find first unused artifact
            start = 1
            while start <= n and used[start]:
                start += 1
            
            # Try all triads containing 'start'
            for a, b, c in triads:
                if a == start or b == start or c == start:
                    if not used[a] and not used[b] and not used[c]:
                        used[a] = used[b] = used[c] = True
                        selected.append((a, b, c))
                        if backtrack(count + 1):
                            return True
                        selected.pop()
                        used[a] = used[b] = used[c] = False
            return False
        
        if backtrack(0):
            print("YES")
            for a, b, c in selected:
                print(a, b, c)
        else:
            print("NO")
        return
    
    # For larger t, use greedy with backtracking
    # Build adjacency: for each pair, which triads contain it
    pair_to_triads = {}
    for idx, (a, b, c) in enumerate(triads):
        pairs = [(a, b), (a, c), (b, c)]
        for p in pairs:
            if p not in pair_to_triads:
                pair_to_triads[p] = []
            pair_to_triads[p].append(idx)
    
    # For each artifact, which triads contain it
    artifact_to_triads = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(triads):
        artifact_to_triads[a].append(idx)
        artifact_to_triads[b].append(idx)
        artifact_to_triads[c].append(idx)
    
    used = [False] * (n + 1)
    triad_used = [False] * t
    selected = []
    
    def backtrack(count):
        if count == q:
            return True
        
        # Find first unused artifact
        start = 1
        while start <= n and used[start]:
            start += 1
        
        # Sort triads by degree (number of unused artifacts in triad)
        candidates = []
        for idx in artifact_to_triads[start]:
            if not triad_used[idx]:
                a, b, c = triads[idx]
                if not used[a] and not used[b] and not used[c]:
                    # Count how many other triads each artifact is in
                    degree = len(artifact_to_triads[a]) + len(artifact_to_triads[b]) + len(artifact_to_triads[c])
                    candidates.append((degree, idx, a, b, c))
        
        # Sort by degree (prefer triads with artifacts that have fewer options)
        candidates.sort()
        
        for _, idx, a, b, c in candidates:
            used[a] = used[b] = used[c] = True
            triad_used[idx] = True
            selected.append((a, b, c))
            if backtrack(count + 1):
                return True
            selected.pop()
            triad_used[idx] = False
            used[a] = used[b] = used[c] = False
        
        return False
    
    if backtrack(0):
        print("YES")
        for a, b, c in selected:
            print(a, b, c)
    else:
        print("NO")

solve()
