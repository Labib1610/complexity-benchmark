import sys
from itertools import combinations

def find_exhibition_layout():
    q, t = map(int, input().split())
    n = 3 * q
    
    triads = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triads.append((a, b, c))
    
    # Try all combinations of q triads
    for combo in combinations(range(t), q):
        used = [False] * (n + 1)  # Index 0 unused, artifacts are 1-indexed
        selected_triads = []
        
        valid = True
        for i in combo:
            a, b, c = triads[i]
            if used[a] or used[b] or used[c]:
                valid = False
                break
            used[a] = used[b] = used[c] = True
            selected_triads.append((a, b, c))
        
        if valid and all(used[1:]):
            print("YES")
            for triad in selected_triads:
                print(triad[0], triad[1], triad[2])
            return
    
    print("NO")

find_exhibition_layout()
