import sys

# Increase recursion depth just in case
sys.setrecursionlimit(1000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    idx = 2
    triples = []
    tri_by_elem = [[] for _ in range(n + 1)]
    
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
        tri_by_elem[a].append(i)
        tri_by_elem[b].append(i)
        tri_by_elem[c].append(i)
        
    covered = [False] * (n + 1)
    ans = []
    
    def backtrack(k):
        if k == 0:
            return True
        
        # Find uncovered element with minimum number of available valid triples (MRV)
        best_u = -1
        min_options = 10**9
        best_cand = []
        
        for u in range(1, n + 1):
            if not covered[u]:
                cand = []
                for tri_idx in tri_by_elem[u]:
                    a, b, c = triples[tri_idx]
                    if not covered[a] and not covered[b] and not covered[c]:
                        cand.append(tri_idx)
                
                if len(cand) < min_options:
                    min_options = len(cand)
                    best_u = u
                    best_cand = cand
                    if min_options == 0:
                        return False
                    if min_options == 1:
                        break
        
        if best_u == -1:
            return True
        
        for tri_idx in best_cand:
            a, b, c = triples[tri_idx]
            if covered[a] or covered[b] or covered[c]:
                continue
            covered[a] = covered[b] = covered[c] = True
            ans.append(tri_idx)
            
            if backtrack(k - 1):
                return True
            
            ans.pop()
            covered[a] = covered[b] = covered[c] = False
            
        return False

    if backtrack(q):
        print("YES")
        out = []
        for tri_idx in ans:
            a, b, c = triples[tri_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
