import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        triples.append((a, b, c))

    n_cols = 3 * q

    # Dancing Links structures
    size = n_cols + 1  # column headers 0..n_cols-1, root at n_cols
    U = list(range(size))
    D = list(range(size))
    L = list(range(size))
    R = list(range(size))
    C = [i for i in range(size)]  # column of node i (header points to itself)
    S = [0] * size                # column sizes, only for headers
    row_id = []                   # row index for each node (excluding headers)

    root = n_cols
    # link column headers horizontally
    for i in range(n_cols):
        L[i] = i - 1 if i > 0 else root
        R[i] = i + 1 if i < n_cols - 1 else root
    L[root] = n_cols - 1
    R[root] = 0

    # add rows
    for rid, (a, b, c) in enumerate(triples):
        first = None
        for col in (a, b, c):
            node = len(C)
            C.append(col)
            row_id.append(rid)
            # vertical links
            U.append(U[col])
            D.append(col)
            D[U[col]] = node
            U[col] = node
            S[col] += 1
            # horizontal links within the row
            if first is None:
                first = node
                L.append(node)
                R.append(node)
            else:
                L.append(L[first])
                R.append(first)
                R[L[first]] = node
                L[first] = node

    solution_rows = []

    def cover(col):
        # remove column header
        L[R[col]] = L[col]
        R[L[col]] = R[col]
        # remove rows
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[col]] = col
        R[L[col]] = col

    def search(k):
        if R[root] == root:  # all columns covered
            return True
        # choose column with minimal size
        c = R[root]
        min_s = S[c]
        j = R[c]
        while j != root:
            if S[j] < min_s:
                c = j
                min_s = S[j]
                if min_s == 0:
                    break
            j = R[j]
        if min_s == 0:
            return False
        cover(c)
        r = D[c]
        while r != c:
            solution_rows.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            if search(k + 1):
                return True
            # backtrack
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution_rows.pop()
            r = D[r]
        uncover(c)
        return False

    if search(0):
        print("YES")
        # output triples in any order, ensure each printed once
        used = set()
        for rid in solution_rows:
            a, b, c = triples[rid]
            print(a + 1, b + 1, c + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
