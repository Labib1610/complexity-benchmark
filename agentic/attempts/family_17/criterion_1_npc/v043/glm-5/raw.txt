Looking at this problem, we need to find q disjoint triples from the given logs that cover all 3q personnel. This is an exact cover problem with 3-element subsets.

Given the constraints (q ≤ 200, so 3q ≤ 600 personnel, and t ≤ 20000 logs), I'll use a backtracking approach with good pruning. The key insight is that we can process personnel in order and always try to cover the smallest uncovered personnel ID first - this dramatically reduces the search space since each step has limited choices.

For efficiency, I'll:
1. Precompute which logs contain each personnel
2. Always pick the smallest uncovered personnel and try all logs containing them
3. Use bitmasks for fast coverage tracking

```python
import sys

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    q = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    n = 3 * q
    
    logs = []
    person_to_logs = [[] for _ in range(n + 1)]
    
    for i in range(t):
        a = int(input_data[idx]); idx += 1
        b = int(input_data[idx]); idx += 1
        c = int(input_data[idx]); idx += 1
        logs.append((a, b, c))
        person_to_logs[a].append(i)
        person_to_logs[b].append(i)
        person_to_logs[c].append(i)
    
    if q == 0:
        print("YES")
        return
    
    # Find the person in the fewest logs - start there for better pruning
    # But simpler: always cover smallest uncovered person
    
    used = [False] * (n + 1)
    chosen = []
    
    def backtrack():
        if len(chosen) == q:
            return True
        
        # Find smallest uncovered person
        target = -1
        for p in range(1, n + 1):
            if not used[p]:
                target = p
                break
        
        # Try all logs containing target that don't conflict
        for log_idx in person_to_logs[target]:
            a, b, c = logs[log_idx]
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                chosen.append(log_idx)
                if backtrack():
                    return True
                chosen.pop()
                used[a] = used[b] = used[c] = False
        
        return False
    
    if backtrack():
        print("YES")
        for log_idx in chosen:
            a, b, c = logs[log_idx]
            print(f"{a} {b} {c}")
    else:
        print("NO")

solve()
```
