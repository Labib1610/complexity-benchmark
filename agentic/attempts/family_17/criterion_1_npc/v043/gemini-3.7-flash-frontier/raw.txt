import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    # Read triplets
    triplets = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triplets.append((a, b, c))
    
    # DLX structures
    # Columns: 0 is root, 1..n are elements
    max_nodes = n + 1 + 3 * t + 5
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [-1] * max_nodes
    col_size = [0] * (n + 1)
    
    for i in range(n + 1):
        L[i] = (i - 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        C[i] = i
    
    node_count = n + 1
    
    for r, (a, b, c) in enumerate(triplets):
        row_nodes = []
        for col in (a, b, c):
            cur = node_count
            node_count += 1
            
            C[cur] = col
            row_id[cur] = r
            col_size[col] += 1
            
            # Insert into column col (at the bottom)
            U[cur] = U[col]
            D[cur] = col
            D[U[col]] = cur
            U[col] = cur
            
            row_nodes.append(cur)
        
        # Link row_nodes horizontally
        L[row_nodes[0]] = row_nodes[2]
        R[row_nodes[0]] = row_nodes[1]
        L[row_nodes[1]] = row_nodes[0]
        R[row_nodes[1]] = row_nodes[2]
        L[row_nodes[2]] = row_nodes[1]
        R[row_nodes[2]] = row_nodes[0]

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size
        c = R[0]
        min_size = col_size[c]
        curr = R[0]
        while curr != 0:
            if col_size[curr] < min_size:
                min_size = col_size[curr]
                c = curr
            curr = R[curr]
        
        if min_size == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            
            if search():
                return True
            
            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]
        
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r in solution:
            out.append(f"{triplets[r][0]} {triplets[r][1]} {triplets[r][2]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
