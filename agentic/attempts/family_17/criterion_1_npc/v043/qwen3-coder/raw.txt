import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    logs = []
    for i in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        logs.append((a, b, c))
        index += 3
    
    # We need to find a matching that covers all 3q vertices
    # This is a exact cover problem, can be modeled as finding a perfect matching in a hypergraph
    # Since each log covers exactly 3 nodes and we need to cover 3q nodes with q logs,
    # it's equivalent to finding a matching of size q in a 3-uniform hypergraph
    
    # We can model this as a bipartite matching problem or use backtracking
    # Given the constraints (q <= 200), we should use a more efficient approach
    
    # Let's try a greedy approach first, but that might not work
    # Instead, let's use backtracking with pruning
    
    # Build a graph where each person points to the logs they are in
    person_to_logs = defaultdict(list)
    for i, (a, b, c) in enumerate(logs):
        person_to_logs[a].append(i)
        person_to_logs[b].append(i)
        person_to_logs[c].append(i)
    
    # We'll use backtracking
    used_logs = [False] * t
    deployed = [False] * (3*q + 1)
    selected = []
    
    def backtrack(log_idx):
        # Count how many people are deployed
        count = sum(deployed)
        if count == 3*q:
            return True
        if log_idx == t:
            return False
            
        # Pruning: if even if we select all remaining logs, we can't reach 3*q
        # This is a weak pruning, but let's try without for now
        
        # Try not taking this log
        if backtrack(log_idx + 1):
            return True
            
        # Try taking this log if it's not used and none of its members are deployed
        a, b, c = logs[log_idx]
        if not used_logs[log_idx] and not deployed[a] and not deployed[b] and not deployed[c]:
            # Take this log
            used_logs[log_idx] = True
            deployed[a] = True
            deployed[b] = True
            deployed[c] = True
            selected.append(log_idx)
            
            if backtrack(log_idx + 1):
                return True
                
            # Backtrack
            used_logs[log_idx] = False
            deployed[a] = False
            deployed[b] = False
            deployed[c] = False
            selected.pop()
            
        return False
    
    # This might be too slow. Let's try a different approach.
    # Since q <= 200 and t <= 20000, let's try a more optimized backtracking
    
    # Sort logs by the smallest person ID to improve pruning
    # Actually, let's just implement a clean backtracking
    
    def solve():
        # Try all combinations using recursive backtracking
        selected_logs = []
        used_personnel = [False] * (3*q + 1)
        
        def dfs(log_index):
            if len(selected_logs) == q:
                return True
            if log_index == t:
                return False
                
            # Try skipping this log
            if dfs(log_index + 1):
                return True
                
            # Try using this log
            a, b, c = logs[log_index]
            if not used_personnel[a] and not used_personnel[b] and not used_personnel[c]:
                # Use this log
                used_personnel[a] = True
                used_personnel[b] = True
                used_personnel[c] = True
                selected_logs.append(log_index)
                
                if dfs(log_index + 1):
                    return True
                    
                # Backtrack
                used_personnel[a] = False
                used_personnel[b] = False
                used_personnel[c] = False
                selected_logs.pop()
                
            return False
        
        if dfs(0):
            print("YES")
            for log_idx in selected_logs:
                a, b, c = logs[log_idx]
                print(a, b, c)
            return True
        else:
            print("NO")
            return False
    
    solve()

if __name__ == "__main__":
    main()
