We can model this as an exact cover problem: we have `3q` elements (personnel) and `t` subsets (logs), each of size 3. We need to select `q` disjoint subsets that cover all elements.

Since `q ≤ 200` and `t ≤ 20000`, we can use backtracking with pruning, or a simple greedy + backtracking approach. We'll sort logs and use recursion with bitmask or set-based tracking.

Here's the Python solution:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    logs = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        logs.append((a, b, c))
    
    # Sort logs by first element for deterministic order
    logs.sort()
    
    used = [False] * (3*q + 1)
    chosen = []
    
    # Precompute adjacency: for each person, list of logs containing them
    person_to_logs = [[] for _ in range(3*q + 1)]
    for i, (a,b,c) in enumerate(logs):
        person_to_logs[a].append(i)
        person_to_logs[b].append(i)
        person_to_logs[c].append(i)
    
    # Backtracking
    def backtrack(start):
        if len(chosen) == q:
            return True
        # Find first unused person
        first_unused = None
        for p in range(1, 3*q+1):
            if not used[p]:
                first_unused = p
                break
        if first_unused is None:
            return False
        
        # Try logs containing first_unused
        for log_idx in person_to_logs[first_unused]:
            a,b,c = logs[log_idx]
            if not used[a] and not used[b] and not used[c]:
                # choose
                used[a] = used[b] = used[c] = True
                chosen.append((a,b,c))
                if backtrack(start+1):
                    return True
                # backtrack
                chosen.pop()
                used[a] = used[b] = used[c] = False
        return False
    
    if backtrack(0):
        print("YES")
        for a,b,c in chosen:
            print(a,b,c)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
