import sys
input = sys.stdin.readline

def solve():
    q, t = map(int, input().split())
    n = 3 * q
    
    # Read all triples
    triples = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triples.append((a, b, c))
    
    # Build adjacency list: for each pair of vertices, list triples containing them
    # We'll use a dictionary mapping (min, max) -> list of triple indices
    pair_to_triples = {}
    for idx, (a, b, c) in enumerate(triples):
        pairs = [(a, b), (a, c), (b, c)]
        for p in pairs:
            if p not in pair_to_triples:
                pair_to_triples[p] = []
            pair_to_triples[p].append(idx)
    
    # Build adjacency between triples that share a pair
    adj = [[] for _ in range(t)]
    for p, indices in pair_to_triples.items():
        for i in range(len(indices)):
            for j in range(i + 1, len(indices)):
                adj[indices[i]].append(indices[j])
                adj[indices[j]].append(indices[i])
    
    # Find connected components
    visited = [False] * t
    components = []
    
    for i in range(t):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            components.append(comp)
    
    # For each component, check if we can select a valid subset
    # We need to select triples such that each vertex appears exactly once
    
    # Greedy approach: for each component, try to find a valid selection
    # Since each vertex appears in at most 2 triples in a component,
    # we can use a simple greedy approach
    
    selected = [False] * t
    used = [False] * (n + 1)
    
    for comp in components:
        # Get all vertices in this component
        vertices = set()
        for idx in comp:
            a, b, c = triples[idx]
            vertices.add(a)
            vertices.add(b)
            vertices.add(c)
        
        # Check if any vertex is already used
        conflict = False
        for v in vertices:
            if used[v]:
                conflict = True
                break
        
        if conflict:
            continue
        
        # Try to find a valid selection within this component
        # Build a graph where vertices are the vials and edges are triples
        # We need to find a matching that covers all vertices
        
        # Since each vertex has degree at most 2, the graph is a collection of paths and cycles
        # We can use a greedy approach
        
        # Build adjacency for vials within this component
        vial_adj = {}
        vial_to_triples = {}
        for idx in comp:
            a, b, c = triples[idx]
            for v in [a, b, c]:
                if v not in vial_adj:
                    vial_adj[v] = []
                if v not in vial_to_triples:
                    vial_to_triples[v] = []
                vial_to_triples[v].append(idx)
            vial_adj[a].append(b)
            vial_adj[a].append(c)
            vial_adj[b].append(a)
            vial_adj[b].append(c)
            vial_adj[c].append(a)
            vial_adj[c].append(b)
        
        # Find connected components of vials
        vial_visited = {}
        for v in vertices:
            vial_visited[v] = False
        
        vial_components = []
        for v in vertices:
            if not vial_visited[v]:
                v_comp = []
                stack = [v]
                vial_visited[v] = True
                while stack:
                    u = stack.pop()
                    v_comp.append(u)
                    for w in vial_adj[u]:
                        if not vial_visited[w]:
                            vial_visited[w] = True
                            stack.append(w)
                vial_components.append(v_comp)
        
        # For each vial component, try to find a valid selection
        for v_comp in vial_components:
            # Check if size is divisible by 3
            if len(v_comp) % 3 != 0:
                # This component cannot be fully covered
                # But we might still be able to use some triples
                pass
            
            # Try to select triples greedily
            # Sort triples by some heuristic (e.g., smallest vertex)
            comp_triples = []
            for idx in comp:
                a, b, c = triples[idx]
                if a in v_comp or b in v_comp or c in v_comp:
                    comp_triples.append(idx)
            
            # Sort by smallest vertex
            comp_triples.sort(key=lambda x: min(triples[x]))
            
            # Greedy selection
            local_used = set()
            for idx in comp_triples:
                a, b, c = triples[idx]
                if a not in local_used and b not in local_used and c not in local_used:
                    selected[idx] = True
                    local_used.add(a)
                    local_used.add(b)
                    local_used.add(c)
            
            # Mark all vertices in this component as used
            for v in local_used:
                used[v] = True
    
    # Check if we have exactly q triples selected
    selected_indices = [i for i in range(t) if selected[i]]
    
    if len(selected_indices) == q:
        # Verify all vertices are used exactly once
        all_used = set()
        valid = True
        for idx in selected_indices:
            a, b, c = triples[idx]
            if a in all_used or b in all_used or c in all_used:
                valid = False
                break
            all_used.add(a)
            all_used.add(b)
            all_used.add(c)
        
        if valid and len(all_used) == n:
            print("YES")
            for idx in selected_indices:
                a, b, c = triples[idx]
                print(a, b, c)
            return
    
    # If greedy didn't work, try a different approach
    # Reset and try matching-based approach
    
    # Build bipartite graph: triples on left, vertices on right
    # But this is a 3-dimensional matching problem, which is NP-hard in general
    # However, with the constraint that each vertex appears in at most 2 triples,
    # we can use a polynomial-time algorithm
    
    # Let's try a different greedy approach: process components in order of size
    selected = [False] * t
    used = [False] * (n + 1)
    
    # Sort components by size (smallest first)
    components.sort(key=len)
    
    for comp in components:
        # Get all vertices in this component
        vertices = set()
        for idx in comp:
            a, b, c = triples[idx]
            vertices.add(a)
            vertices.add(b)
            vertices.add(c)
        
        # Filter out already used vertices
        available_vertices = [v for v in vertices if not used[v]]
        
        if len(available_vertices) % 3 != 0:
            continue
        
        # Try to find a perfect matching within this component
        # Use backtracking with pruning
        comp_triples = [idx for idx in comp if all(not used[v] for v in triples[idx])]
        
        # Sort by number of available vertices
        comp_triples.sort(key=lambda x: sum(1 for v in triples[x] if not used[v]), reverse=True)
        
        # Simple backtracking
        def backtrack(pos, count, target):
            if count == target:
                return True
            if pos >= len(comp_triples):
                return False
            
            # Try to select current triple
            idx = comp_triples[pos]
            a, b, c = triples[idx]
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                selected[idx] = True
                if backtrack(pos + 1, count + 1, target):
                    return True
                used[a] = used[b] = used[c] = False
                selected[idx] = False
            
            # Try not selecting current triple
            if backtrack(pos + 1, count, target):
                return True
            
            return False
        
        target = len(available_vertices) // 3
        backtrack(0, 0, target)
    
    # Check result
    selected_indices = [i for i in range(t) if selected[i]]
    
    if len(selected_indices) == q:
        all_used = set()
        valid = True
        for idx in selected_indices:
            a, b, c = triples[idx]
            if a in all_used or b in all_used or c in all_used:
                valid = False
                break
            all_used.add(a)
            all_used.add(b)
            all_used.add(c)
        
        if valid and len(all_used) == n:
            print("YES")
            for idx in selected_indices:
                a, b, c = triples[idx]
                print(a, b, c)
            return
    
    print("NO")

solve()
