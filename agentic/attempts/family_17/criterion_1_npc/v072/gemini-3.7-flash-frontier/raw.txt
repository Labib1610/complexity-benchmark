import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    triples = []
    
    idx = 2
    seen = set()
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        # Normalize/sort
        triple = tuple(sorted((a, b, c)))
        if triple not in seen:
            seen.add(triple)
            triples.append(triple)
    
    # Pre-check: all elements 1..n must appear at least once
    element_to_triples = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triples):
        element_to_triples[a].append(i)
        element_to_triples[b].append(i)
        element_to_triples[c].append(i)
        
    for i in range(1, n + 1):
        if not element_to_triples[i]:
            print("NO")
            return

    used = [False] * (n + 1)
    solution = []

    def backtrack(covered_count):
        if covered_count == n:
            return True
        
        # MRV: find uncovered element with minimum available choices
        best_u = -1
        min_choices = float('inf')
        best_valid_triples = []
        
        for u in range(1, n + 1):
            if not used[u]:
                valid_triples = []
                for tr_idx in element_to_triples[u]:
                    a, b, c = triples[tr_idx]
                    if not used[a] and not used[b] and not used[c]:
                        valid_triples.append(tr_idx)
                
                count = len(valid_triples)
                if count == 0:
                    return False
                if count < min_choices:
                    min_choices = count
                    best_u = u
                    best_valid_triples = valid_triples
                    if min_choices == 1:
                        break
        
        for tr_idx in best_valid_triples:
            a, b, c = triples[tr_idx]
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                solution.append(tr_idx)
                if backtrack(covered_count + 3):
                    return True
                solution.pop()
                used[a] = used[b] = used[c] = False
                
        return False

    if backtrack(0):
        print("YES")
        for tr_idx in solution:
            a, b, c = triples[tr_idx]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
