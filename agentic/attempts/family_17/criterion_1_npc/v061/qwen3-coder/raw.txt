from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    components_count = 3 * q
    
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    configurations = []
    component_to_configs = defaultdict(list)
    
    for i in range(t):
        a, b, c = map(int, input().split())
        configurations.append((a, b, c))
        component_to_configs[a].append(i)
        component_to_configs[b].append(i)
        component_to_configs[c].append(i)
    
    # Try to find a perfect matching using backtracking with pruning
    used_configs = [False] * t
    result = []
    component_count = [0] * (components_count + 1)
    
    def backtrack(pos):
        if pos == q:
            return True
            
        # Find the first unused component
        first_unused = -1
        for i in range(1, components_count + 1):
            if component_count[i] == 0:
                first_unused = i
                break
        
        if first_unused == -1:
            return False
            
        # Try all configurations containing first_unused
        for config_idx in component_to_configs[first_unused]:
            if used_configs[config_idx]:
                continue
                
            a, b, c = configurations[config_idx]
            
            # Check if all components in this configuration are available
            if component_count[a] > 0 or component_count[b] > 0 or component_count[c] > 0:
                continue
            
            # Use this configuration
            used_configs[config_idx] = True
            component_count[a] += 1
            component_count[b] += 1
            component_count[c] += 1
            result.append((a, b, c))
            
            if backtrack(pos + 1):
                return True
                
            # Backtrack
            used_configs[config_idx] = False
            component_count[a] -= 1
            component_count[b] -= 1
            component_count[c] -= 1
            result.pop()
        
        return False
    
    if backtrack(0):
        print("YES")
        for triple in result:
            print(triple[0], triple[1], triple[2])
    else:
        print("NO")

solve()
