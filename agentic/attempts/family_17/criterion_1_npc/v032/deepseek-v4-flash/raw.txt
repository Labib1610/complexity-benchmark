We can model this as an exact cover problem: we have \(3q\) operators and a list of triples (units). We need to pick \(q\) triples that cover all operators exactly once.

Since \(q \le 200\), \(3q \le 600\), and \(t \le 20000\), we can use backtracking with pruning. A good approach is to pick the operator with the fewest available units first (minimum remaining values heuristic) and try each unit containing that operator.

We’ll implement a recursive search with bitmasks for efficiency.

```python
import sys

def solve():
    input = sys.stdin.readline
    q, t = map(int, input().split())
    n = 3 * q
    units = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        units.append((a - 1, b - 1, c - 1))
    
    # For each operator, list of unit indices that contain them
    op_units = [[] for _ in range(n)]
    for idx, (a, b, c) in enumerate(units):
        op_units[a].append(idx)
        op_units[b].append(idx)
        op_units[c].append(idx)
    
    used = [False] * t
    chosen = []
    full_mask = (1 << n) - 1
    
    # Precompute mask for each unit
    unit_masks = []
    for a, b, c in units:
        unit_masks.append((1 << a) | (1 << b) | (1 << c))
    
    def backtrack(mask):
        if mask == full_mask:
            return True
        # Find operator with fewest available units
        best_op = -1
        best_list = None
        for op in range(n):
            if not (mask >> op) & 1:
                avail = [i for i in op_units[op] if not used[i]]
                if best_op == -1 or len(avail) < len(best_list):
                    best_op = op
                    best_list = avail
                    if len(best_list) == 0:
                        return False
                    if len(best_list) == 1:
                        break
        for ui in best_list:
            if used[ui]:
                continue
            um = unit_masks[ui]
            if mask & um:
                continue
            used[ui] = True
            chosen.append(ui)
            if backtrack(mask | um):
                return True
            chosen.pop()
            used[ui] = False
        return False
    
    if backtrack(0):
        print("YES")
        for ui in chosen:
            a, b, c = units[ui]
            print(a + 1, b + 1, c + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
