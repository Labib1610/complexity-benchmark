import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    triplets = []
    idx = 2
    for i in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        idx += 3
        triplets.append((a, b, c))
        
    triplets_of = [[] for _ in range(n)]
    for i, (a, b, c) in enumerate(triplets):
        triplets_of[a].append(i)
        triplets_of[b].append(i)
        triplets_of[c].append(i)
        
    deg = [len(triplets_of[i]) for i in range(n)]
    active = [True] * t
    uncovered = set(range(n))
    
    # Check trivial impossibility
    for i in range(n):
        if deg[i] == 0:
            print("NO")
            return

    chosen = []

    def backtrack():
        if not uncovered:
            return True
        
        # Pick element with minimum degree
        best_elem = -1
        min_deg = float('inf')
        for elem in uncovered:
            d = deg[elem]
            if d < min_deg:
                min_deg = d
                best_elem = elem
                if min_deg <= 1:
                    break
        
        if min_deg == 0:
            return False
        
        # Get active candidates for best_elem
        candidates = [cand for cand in triplets_of[best_elem] if active[cand]]
        
        for cand in candidates:
            # Choose this triplet
            chosen.append(cand)
            a, b, c = triplets[cand]
            
            # Deactivate all triplets sharing elements with cand
            deactivated = []
            for elem in (a, b, c):
                uncovered.remove(elem)
                for tr in triplets_of[elem]:
                    if active[tr]:
                        active[tr] = False
                        deactivated.append(tr)
                        ta, tb, tc = triplets[tr]
                        deg[ta] -= 1
                        deg[tb] -= 1
                        deg[tc] -= 1
            
            if backtrack():
                return True
            
            # Undo
            chosen.pop()
            for elem in (a, b, c):
                uncovered.add(elem)
            for tr in deactivated:
                active[tr] = True
                ta, tb, tc = triplets[tr]
                deg[ta] += 1
                deg[tb] += 1
                deg[tc] += 1
                
        return False

    if backtrack():
        print("YES")
        out = []
        for cand in chosen:
            a, b, c = triplets[cand]
            out.append(f"{a + 1} {b + 1} {c + 1}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
