import sys
input = sys.stdin.readline

def solve():
    q_t = input().split()
    if not q_t:
        return
    q = int(q_t[0])
    t = int(q_t[1])
    
    scripts = []
    # scripts_by_module[m] will store indices of scripts containing module m
    scripts_by_module = [[] for _ in range(3 * q + 1)]
    
    for i in range(t):
        line = input().split()
        if not line:
            # handle potential empty lines if any, though unlikely in strict input
            continue
        a, b, c = map(int, line)
        scripts.append((a, b, c))
        scripts_by_module[a].append(i)
        scripts_by_module[b].append(i)
        scripts_by_module[c].append(i)

    n = 3 * q
    # We need to select q scripts covering all modules 1..n exactly once.
    # This is an exact cover problem.
    # Given constraints (n <= 600, t <= 20000), we use Algorithm X with dancing links
    # or a recursive backtracking with efficient pruning.
    # To speed up, we use a heuristic: always pick the module with the fewest covering scripts.
    
    # Pre-calculate script indices for each module for faster access
    # (Already done in scripts_by_module)
    
    # Track state
    used_modules = [False] * (n + 1)
    used_scripts = [False] * t
    
    # For Algorithm X, we need to efficiently disable/enable rows and columns.
    # Disabling a column (module) means removing it from consideration.
    # Disabling a row (script) means marking it as used or invalid (conflicting).
    
    # We can implement a recursive search.
    # To optimize, we maintain a list of available modules.
    # When we pick a script, we mark its 3 modules as used, and mark all scripts
    # containing any of these modules as invalid.
    
    # To efficiently find conflicting scripts:
    # When module m is covered, all scripts in scripts_by_module[m] become invalid.
    
    # We need to restore state upon backtracking.
    
    # Let's use arrays to track validity.
    # valid_script[i] = True if script i is not yet invalid
    # valid_module[m] = True if module m is not yet covered
    
    valid_script = [True] * t
    valid_module = [True] * (n + 1)
    
    # Optimization: count of valid scripts for each module, to pick the one with minimum options.
    # However, maintaining these counts dynamically is expensive.
    # Instead, we can just iterate to find the module with min options on the fly, or use a simpler heuristic.
    # Given n is small (600), iterating to find the module with min options is acceptable.
    
    # Let's store the solution path
    solution = []

    def search():
        # If we have selected q scripts, we are done
        if len(solution) == q:
            return True
        
        # Find the uncovered module with the fewest valid scripts
        # This is the standard heuristic for Algorithm X
        best_module = -1
        min_options = float('inf')
        
        # We can iterate all modules 1..n
        # To speed this up, we could maintain a set of uncovered modules, but n is small.
        for m in range(1, n + 1):
            if valid_module[m]:
                count = 0
                # Count valid scripts covering m
                for script_idx in scripts_by_module[m]:
                    if valid_script[script_idx]:
                        count += 1
                if count == 0:
                    return False # Dead end: module m cannot be covered
                if count < min_options:
                    min_options = count
                    best_module = m
                    if min_options == 1:
                        break # Optimization: can't do better than 1
        
        if best_module == -1:
            return False # Should not happen if logic is correct

        # Try all valid scripts covering best_module
        # We need to iterate over a snapshot of indices because we will modify valid_script
        candidates = []
        for script_idx in scripts_by_module[best_module]:
            if valid_script[script_idx]:
                candidates.append(script_idx)
        
        for script_idx in candidates:
            # Choose this script
            a, b, c = scripts[script_idx]
            
            # Check if modules are still valid (should be, since script is valid)
            # Mark modules as used
            # Mark conflicting scripts as invalid
            
            # Save state for backtracking
            # We only need to save what we change.
            # We change valid_module[a], valid_module[b], valid_module[c]
            # We change valid_script[...] for all scripts touching a, b, or c.
            
            # Optimization: instead of saving all changed states, we can use a "depth" counter
            # or a stack of changes, but simple recursion with local lists is safer/easier.
            # Given constraints, let's try to be efficient.
            
            # Identify scripts to invalidate
            # A script is invalid if it touches a, b, or c.
            # We can iterate scripts_by_module[a], [b], [c].
            
            # Collect all scripts to disable
            to_disable_scripts = set()
            for mod in [a, b, c]:
                for s_idx in scripts_by_module[mod]:
                    if valid_script[s_idx]:
                        to_disable_scripts.add(s_idx)
            
            # Apply changes
            valid_module[a] = False
            valid_module[b] = False
            valid_module[c] = False
            
            changed_scripts_indices = []
            for s_idx in to_disable_scripts:
                valid_script[s_idx] = False
                changed_scripts_indices.append(s_idx)
            
            solution.append(script_idx)
            
            if search():
                return True
            
            # Backtrack
            solution.pop()
            
            valid_module[a] = True
            valid_module[b] = True
            valid_module[c] = True
            
            for s_idx in changed_scripts_indices:
                valid_script[s_idx] = True
                
        return False

    if search():
        print("YES")
        for idx in solution:
            a, b, c = scripts[idx]
            print(a, b, c)
    else:
        print("NO")

if __name__ == '__main__':
    solve()
