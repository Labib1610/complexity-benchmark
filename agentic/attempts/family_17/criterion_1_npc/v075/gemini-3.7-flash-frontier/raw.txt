import sys

# Increase recursion depth for search
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    triples = []
    idx = 2
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    # DLX setup
    # 0 is root
    # 1..n are column headers
    max_nodes = n + 1 + 3 * t + 10
    
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    size = [0] * (n + 1)
    
    # Initialize column headers
    for i in range(n + 1):
        L[i] = (i - 1 + n + 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        C[i] = i
    
    node_count = n + 1
    
    for r_idx, (a, b, c) in enumerate(triples):
        cols = (a, b, c)
        first_node = -1
        for col in cols:
            node = node_count
            node_count += 1
            
            C[node] = col
            row_id[node] = r_idx
            size[col] += 1
            
            # Insert into column col vertically (at the bottom)
            U[node] = U[col]
            D[node] = col
            D[U[col]] = node
            U[col] = node
            
            # Insert into row horizontally
            if first_node == -1:
                first_node = node
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first_node]
                R[node] = first_node
                R[L[first_node]] = node
                L[first_node] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column c with minimum size[c]
        c = R[0]
        min_s = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_s:
                min_s = size[curr]
                c = curr
            curr = R[curr]
            
        if min_s == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            
            if search():
                return True
            
            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]
        
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r_idx in solution:
            a, b, c = triples[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
