We can model this as an exact cover problem, but since \( q \leq 200 \) and \( t \leq 20000 \), we can use a backtracking approach with pruning.

We’ll represent each task as a bit in a bitmask (up to 600 tasks, so we can use Python integers for bitmasks).  
We’ll sort candidate pipelines by their first task to help with pruning.

Here’s the code:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q
    pipelines = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        pipelines.append((a, b, c))
    
    # Sort by first task
    pipelines.sort()
    
    # Group by first task
    by_first = [[] for _ in range(n)]
    for idx, (a, b, c) in enumerate(pipelines):
        by_first[a].append(idx)
    
    used = [False] * t
    chosen = []
    full_mask = (1 << n) - 1
    
    def backtrack(mask, count):
        if count == q:
            return mask == full_mask
        # Find first unassigned task
        # We can find the lowest zero bit
        # Since n <= 600, we can loop
        first_free = 0
        while first_free < n and (mask >> first_free) & 1:
            first_free += 1
        if first_free == n:
            return mask == full_mask
        
        # Try pipelines starting with first_free
        for idx in by_first[first_free]:
            a, b, c = pipelines[idx]
            if (mask >> a) & 1 or (mask >> b) & 1 or (mask >> c) & 1:
                continue
            # Choose
            used[idx] = True
            chosen.append(idx)
            new_mask = mask | (1 << a) | (1 << b) | (1 << c)
            if backtrack(new_mask, count + 1):
                return True
            chosen.pop()
            used[idx] = False
        return False
    
    if backtrack(0, 0):
        print("YES")
        for idx in chosen:
            a, b, c = pipelines[idx]
            print(a + 1, b + 1, c + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
