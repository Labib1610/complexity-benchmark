import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        adj[a].append((b, c))
        
    for a in range(1, n + 1):
        adj[a].sort()
        
    dp = [0] * (1 << n)
    parent = [0] * (1 << n)
    
    for mask in range(1 << n):
        if dp[mask] == 0 and mask != 0:
            continue
        if dp[mask] == q:
            continue
            
        first = -1
        for i in range(n):
            if not (mask & (1 << i)):
                first = i
                break
                
        if first == -1:
            continue
            
        a = first + 1
        for b, c in adj[a]:
            if not (mask & (1 << (b - 1))) and not (mask & (1 << (c - 1))):
                new_mask = mask | (1 << (a - 1)) | (1 << (b - 1)) | (1 << (c - 1))
                if dp[new_mask] < dp[mask] + 1:
                    dp[new_mask] = dp[mask] + 1
                    parent[new_mask] = mask
                    
    full_mask = (1 << n) - 1
    if dp[full_mask] == q:
        print("YES")
        curr = full_mask
        res = []
        while curr != 0:
            prev = parent[curr]
            diff = curr ^ prev
            a = (diff & -diff).bit_length()
            diff ^= (1 << (a - 1))
            b = (diff & -diff).bit_length()
            diff ^= (1 << (b - 1))
            c = (diff & -diff).bit_length()
            res.append(f"{a} {b} {c}")
            curr = prev
        print("\n".join(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
