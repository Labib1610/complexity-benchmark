from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    pipelines = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        pipelines.append((a, b, c))
    
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    # Try to find a matching using backtracking with optimizations
    # This is essentially a exact cover problem
    
    # For small constraints, we can use a recursive approach with pruning
    if q <= 200 and t <= 20000:
        # Build a graph where each pipeline is a node and there's an edge 
        # between two pipelines if they don't share any tasks
        
        # But actually, we want to select q pipelines such that all 3*q tasks are covered exactly once
        # This is a set cover problem where each set has size 3 and we need exactly q sets to cover 3*q elements
        
        def can_extend(selected, new_pipeline):
            # Check if new_pipeline conflicts with any already selected pipeline
            tasks_in_selected = set()
            for p in selected:
                tasks_in_selected.update(p)
            
            new_tasks = set(new_pipeline)
            return len(tasks_in_selected.intersection(new_tasks)) == 0
        
        def backtrack(selected, used_pipelines, next_index):
            if len(selected) == q:
                return selected[:]
            
            if next_index >= t:
                return None
            
            # Pruning: if even selecting all remaining pipelines won't give us enough
            if len(selected) + (t - next_index) < q:
                return None
            
            # Try including current pipeline
            current_pipeline = pipelines[next_index]
            if can_extend(selected, current_pipeline):
                selected.append(current_pipeline)
                result = backtrack(selected, used_pipelines, next_index + 1)
                if result is not None:
                    return result
                selected.pop()
            
            # Try not including current pipeline
            result = backtrack(selected, used_pipelines, next_index + 1)
            return result
        
        # Optimization: sort pipelines to try more "promising" ones first
        # But for now, let's just do basic backtracking
        result = backtrack([], set(), 0)
        
        if result is not None:
            print("YES")
            for pipeline in result:
                print(*pipeline)
        else:
            print("NO")
    else:
        print("NO")

def solve_efficient():
    q, t = map(int, input().split())
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    pipelines = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        pipelines.append((a, b, c))
    
    # Create a bipartite graph
    # Left side: tasks (1 to 3q)
    # Right side: pipelines (0 to t-1)
    # Edge exists if task is in pipeline
    
    # But we actually want a matching that covers all tasks exactly once
    # And uses exactly q pipelines
    
    # This is equivalent to finding a maximum matching in a 3-uniform hypergraph
    # But let's model it as an exact cover problem and use backtracking with optimization
    
    # Use a more efficient approach: for each task, keep track of which pipelines contain it
    task_to_pipelines = defaultdict(list)
    for i, (a, b, c) in enumerate(pipelines):
        task_to_pipelines[a].append(i)
        task_to_pipelines[b].append(i)
        task_to_pipelines[c].append(i)
    
    # Sort tasks by number of available pipelines (ascending)
    tasks = list(range(1, 3*q + 1))
    tasks.sort(key=lambda x: len(task_to_pipelines[x]))
    
    selected_pipelines = []
    used_pipelines = [False] * t
    used_tasks = [False] * (3*q + 1)
    
    def dfs(task_idx):
        if task_idx == len(tasks):
            return len(selected_pipelines) == q
        
        current_task = tasks[task_idx]
        if used_tasks[current_task]:
            return dfs(task_idx + 1)
        
        # Try each pipeline containing this task
        for pipeline_idx in task_to_pipelines[current_task]:
            if used_pipelines[pipeline_idx]:
                continue
                
            a, b, c = pipelines[pipeline_idx]
            if used_tasks[a] or used_tasks[b] or used_tasks[c]:
                continue
            
            # Select this pipeline
            selected_pipelines.append(pipeline_idx)
            used_pipelines[pipeline_idx] = True
            used_tasks[a] = used_tasks[b] = used_tasks[c] = True
            
            if dfs(task_idx + 1):
                return True
            
            # Backtrack
            selected_pipelines.pop()
            used_pipelines[pipeline_idx] = False
            used_tasks[a] = used_tasks[b] = used_tasks[c] = False
        
        return False
    
    if dfs(0):
        print("YES")
        for idx in selected_pipelines:
            print(*pipelines[idx])
    else:
        print("NO")

solve_efficient()
