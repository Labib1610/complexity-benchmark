```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    triples = [tuple(sorted((int(next(it)), int(next(it)), int(next(it))))) for _ in range(t)]
    # Simple heuristic: try to greedily pick triples
    used = [False] * (3 * q + 1)
    ans = []
    for a, b, c in triples:
        if not used[a] and not used[b] and not used[c]:
            used[a] = used[b] = used[c] = True
            ans.append((a, b, c))
            if len(ans) == q:
                break
    if len(ans) == q and all(used[1:]):
        print("YES")
        for tri in ans:
            print(*tri)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
