import sys
from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    groups = []
    index = 2
    for _ in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        groups.append((a, b, c))
        index += 3
    
    # We have 3*q operators numbered from 1 to 3*q
    # We need to select exactly q groups such that each operator appears exactly once
    
    # This is essentially a matching problem: find a perfect matching in a hypergraph
    # where each hyperedge is a group and we want to select q hyperedges that cover
    # all 3*q vertices exactly once.
    
    # Since q <= 200, and each group has size 3, this is manageable with backtracking
    # but let's think of it as an exact cover problem.
    
    # Use backtracking with pruning
    from collections import Counter
    
    def backtrack(selected, used_operators, pos):
        if len(selected) == q:
            return selected[:]
        
        if pos >= len(groups):
            return None
            
        # Pruning: if even if we select all remaining groups, 
        # we won't have enough to reach q, then fail
        if len(selected) + (len(groups) - pos) < q:
            return None
            
        # Try including current group
        group = groups[pos]
        a, b, c = group
        
        # Check if this group conflicts with already used operators
        if a not in used_operators and b not in used_operators and c not in used_operators:
            # Include this group
            selected.append(group)
            used_operators.add(a)
            used_operators.add(b)
            used_operators.add(c)
            
            res = backtrack(selected, used_operators, pos + 1)
            if res is not None:
                return res
                
            # Backtrack
            selected.pop()
            used_operators.remove(a)
            used_operators.remove(b)
            used_operators.remove(c)
        
        # Try not including current group
        return backtrack(selected, used_operators, pos + 1)
    
    # Optimization: sort groups by how "rare" their elements are
    # Count frequency of each operator
    freq = Counter()
    for group in groups:
        for op in group:
            freq[op] += 1
    
    # Sort groups by sum of frequencies (least frequent first)
    groups.sort(key=lambda g: freq[g[0]] + freq[g[1]] + freq[g[2]])
    
    result = backtrack([], set(), 0)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for group in result:
            print(group[0], group[1], group[2])

solve()
