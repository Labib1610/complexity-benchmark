import sys

# Increase recursion depth for search
sys.setrecursionlimit(20000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    m = 3 * q  # number of columns (operators 1 to 3q)

    # Pre-allocate DLX structures
    # Max nodes: m + 1 (headers) + 3 * t
    max_nodes = m + 1 + 3 * t + 10

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    size = [0] * (m + 1)

    # Initialize column headers
    for i in range(m + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        C[i] = i
    L[0] = m
    R[m] = 0

    triples = []
    idx = 2
    node_cnt = m

    for r in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triples.append((a, b, c))

        cols = (a, b, c)
        first = node_cnt + 1
        for i, col in enumerate(cols):
            node_cnt += 1
            cur = node_cnt
            C[cur] = col
            row_id[cur] = r
            size[col] += 1

            # Vertical link: insert at bottom of column col
            u = U[col]
            U[cur] = u
            D[cur] = col
            D[u] = cur
            U[col] = cur

            # Horizontal link
            if i == 0:
                L[cur] = cur + 2
                R[cur] = cur + 1
            elif i == 1:
                L[cur] = cur - 1
                R[cur] = cur + 1
            else:
                L[cur] = cur - 1
                R[cur] = first

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size
        c = R[0]
        min_size = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_size:
                min_size = size[curr]
                c = curr
                if min_size == 0:
                    break
            curr = R[curr]

        if min_size == 0:
            return False

        cover(c)
        i = D[c]
        while i != c:
            ans.append(row_id[i])
            j = R[i]
            while j != i:
                cover(C[j])
                j = R[j]

            if search():
                return True

            ans.pop()
            j = L[i]
            while j != i:
                uncover(C[j])
                j = L[j]
            i = D[i]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r in ans:
            out.append(f"{triples[r][0]} {triples[r][1]} {triples[r][2]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")


if __name__ == "__main__":
    solve()
