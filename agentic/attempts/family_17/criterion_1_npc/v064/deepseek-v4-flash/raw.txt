We can model this as an exact cover problem: we have 3q elements (precursors) and t possible triples (recipes). We need to select q triples that cover all elements exactly once.

Since q ≤ 200, the total number of elements is at most 600, and t ≤ 20000. We can solve this using backtracking with bitmasks and pruning.

Here’s the Python solution:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        idx += 3
        triples.append((a, b, c))
    
    n = 3 * q
    full_mask = (1 << n) - 1
    
    # Precompute mask for each triple
    masks = []
    for a, b, c in triples:
        masks.append((1 << a) | (1 << b) | (1 << c))
    
    # Sort triples by size of mask (all same size 3) but maybe by first element for better pruning
    order = sorted(range(t), key=lambda i: (triples[i][0], triples[i][1], triples[i][2]))
    
    chosen = []
    
    # Backtracking
    def backtrack(mask, start):
        if mask == full_mask:
            return True
        # Find first uncovered element
        # We can find the lowest zero bit
        inv = (~mask) & full_mask
        if inv == 0:
            return False
        # lowest set bit in inv
        lsb = inv & -inv
        first = (lsb.bit_length() - 1)
        
        # Try triples that contain this element
        for i in order:
            if (masks[i] & lsb) == 0:
                continue
            if masks[i] & mask:
                continue
            chosen.append(triples[i])
            if backtrack(mask | masks[i], i+1):
                return True
            chosen.pop()
        return False
    
    if backtrack(0, 0):
        print("YES")
        for a, b, c in chosen:
            print(a+1, b+1, c+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
