import sys

# Increase recursion depth just in case
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triples = []
    idx = 2
    seen = set()
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        trip = tuple(sorted((a, b, c)))
        if trip not in seen:
            seen.add(trip)
            triples.append(trip)

    # item_to_triples
    item_to_triples = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triples):
        item_to_triples[a].append(i)
        item_to_triples[b].append(i)
        item_to_triples[c].append(i)

    # Check if any element has 0 triples
    for i in range(1, n + 1):
        if not item_to_triples[i]:
            print("NO")
            return

    covered = [False] * (n + 1)
    uncovered_items = set(range(1, n + 1))
    solution = []

    def search():
        if not uncovered_items:
            return True

        # Unit propagation & find best item (MRV)
        while True:
            best_item = -1
            min_options = 1000000
            best_candidates = []

            for item in list(uncovered_items):
                valid = []
                for tid in item_to_triples[item]:
                    a, b, c = triples[tid]
                    if not covered[a] and not covered[b] and not covered[c]:
                        valid.append(tid)
                c_len = len(valid)
                if c_len == 0:
                    return False
                if c_len < min_options:
                    min_options = c_len
                    best_item = item
                    best_candidates = valid
                    if min_options == 1:
                        break

            if min_options == 1:
                # Forced choice
                tid = best_candidates[0]
                a, b, c = triples[tid]
                covered[a] = covered[b] = covered[c] = True
                uncovered_items.remove(a)
                uncovered_items.remove(b)
                uncovered_items.remove(c)
                solution.append(tid)

                if search():
                    return True

                solution.pop()
                covered[a] = covered[b] = covered[c] = False
                uncovered_items.add(a)
                uncovered_items.add(b)
                uncovered_items.add(c)
                return False
            else:
                break

        # Branch on best_item
        for tid in best_candidates:
            a, b, c = triples[tid]
            if not covered[a] and not covered[b] and not covered[c]:
                covered[a] = covered[b] = covered[c] = True
                uncovered_items.remove(a)
                uncovered_items.remove(b)
                uncovered_items.remove(c)
                solution.append(tid)

                if search():
                    return True

                solution.pop()
                covered[a] = covered[b] = covered[c] = False
                uncovered_items.add(a)
                uncovered_items.add(b)
                uncovered_items.add(c)

        return False

    if search():
        print("YES")
        for tid in solution:
            a, b, c = triples[tid]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
