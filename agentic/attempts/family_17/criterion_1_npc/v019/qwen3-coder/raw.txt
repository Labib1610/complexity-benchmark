```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    protocols = []
    index = 2
    for _ in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        protocols.append((a, b, c))
        index += 3
    
    # We need to find a matching that covers all 3q nodes exactly once
    # This is a exact cover problem, we can model it as finding a perfect matching in a hypergraph
    # But since each protocol covers exactly 3 nodes and we have 3q nodes, we need exactly q protocols
    # So this is equivalent to finding a matching of size q that covers all nodes
    
    # We'll use backtracking / DFS with pruning
    
    # Build adjacency for each element (which protocols contain this element)
    element_to_protocols = defaultdict(list)
    for idx, (a, b, c) in enumerate(protocols):
        element_to_protocols[a].append(idx)
        element_to_protocols[b].append(idx)
        element_to_protocols[c].append(idx)
    
    # Try to build a solution using backtracking
    selected = []
    used_protocols = [False] * t
    covered = [False] * (3*q + 1)
    
    def backtrack(pos):
        if pos == 3*q + 1:
            return len(selected) == q
        if covered[pos]:
            return backtrack(pos + 1)
        
        # Try each protocol containing 'pos'
        for proto_idx in element_to_protocols[pos]:
            if used_protocols[proto_idx]:
                continue
            a, b, c = protocols[proto_idx]
            if covered[a] or covered[b] or covered[c]:
                continue
            
            # Use this protocol
            selected.append(proto_idx)
            used_protocols[proto_idx] = True
            covered[a] = covered[b] = covered[c] = True
            
            if backtrack(pos + 1):
                return True
                
            # Backtrack
            selected.pop()
            used_protocols[proto_idx] = False
            covered[a] = covered[b] = covered[c] = False
        
        return False
    
    # Optimization: start with elements that have fewest options
    # Count how many protocols each element is in
    element_count = [0] * (3*q + 1)
    for i in range(1, 3*q + 1):
        element_count[i] = len(element_to_protocols[i])
    
    # Sort elements by count (but we'll iterate through them in order in backtrack)
    
    if backtrack(1):
        print("YES")
        for proto_idx in selected:
            a, b, c = protocols[proto_idx]
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
