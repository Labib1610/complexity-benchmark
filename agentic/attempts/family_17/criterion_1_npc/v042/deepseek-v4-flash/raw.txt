We need to find a perfect matching of size q among given triads covering all 3q artifacts. This is essentially an exact cover problem, which can be solved via Algorithm X (Dancing Links) or via reduction to bipartite matching by treating each triad as a hyperedge.

Since q ≤ 200 and t ≤ 20000, a backtracking with pruning (or DLX) is feasible.

Here’s a Python implementation using depth-first search with set-based pruning:

```python
import sys

def solve():
    input = sys.stdin.readline
    q, t = map(int, input().split())
    triads = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triads.append((a, b, c))
    
    # We'll do backtracking: choose an artifact that is not covered yet,
    # try all triads containing it.
    # Precompute which triads contain each artifact.
    artifact_to_triads = [[] for _ in range(3*q + 1)]
    for idx, (a,b,c) in enumerate(triads):
        artifact_to_triads[a].append(idx)
        artifact_to_triads[b].append(idx)
        artifact_to_triads[c].append(idx)
    
    used = [False] * (3*q + 1)
    chosen = []
    
    # For faster, we can try to pick artifact with fewest possible triads
    def backtrack():
        if len(chosen) == q:
            return True
        # find smallest uncovered artifact
        best_art = -1
        best_count = float('inf')
        for art in range(1, 3*q+1):
            if not used[art]:
                cnt = 0
                for tidx in artifact_to_triads[art]:
                    a,b,c = triads[tidx]
                    if not used[a] and not used[b] and not used[c]:
                        cnt += 1
                if cnt < best_count:
                    best_count = cnt
                    best_art = art
                    if cnt == 0:
                        break
        if best_count == 0:
            return False
        # try triads containing best_art
        for tidx in artifact_to_triads[best_art]:
            a,b,c = triads[tidx]
            if used[a] or used[b] or used[c]:
                continue
            # choose
            used[a] = used[b] = used[c] = True
            chosen.append((a,b,c))
            if backtrack():
                return True
            # backtrack
            chosen.pop()
            used[a] = used[b] = used[c] = False
        return False
    
    if backtrack():
        sys.stdout.write("YES\n")
        for a,b,c in chosen:
            sys.stdout.write(f"{a} {b} {c}\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
```
