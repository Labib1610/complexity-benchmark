import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triples.append((a, b, c))
        idx += 3

    # Quick sanity check: if number of available elements in triples < n
    present = [False] * (n + 1)
    for a, b, c in triples:
        present[a] = present[b] = present[c] = True
    if sum(present[1:]) < n:
        print("NO")
        return

    # Dancing Links (DLX) Setup
    max_nodes = n + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row_id = [0] * max_nodes
    sz = [0] * (n + 1)

    # Initialize column headers
    for i in range(n + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        col[i] = i
        sz[i] = 0
    L[0] = n
    R[n] = 0

    node_count = n

    for r_idx, (a, b, c) in enumerate(triples):
        cols = (a, b, c)
        first_node = node_count + 1
        for i, c_idx in enumerate(cols):
            node_count += 1
            cur = node_count
            col[cur] = c_idx
            row_id[cur] = r_idx

            sz[c_idx] += 1
            # Vertical linking
            U[cur] = U[c_idx]
            D[cur] = c_idx
            D[U[c_idx]] = cur
            U[c_idx] = cur

            # Horizontal linking
            if i == 0:
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = node_count - 1
                R[cur] = first_node
                R[node_count - 1] = cur
                L[first_node] = cur

    def cover(c):
        R[L[c]] = R[c]
        L[R[c]] = L[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                D[U[j]] = D[j]
                U[D[j]] = U[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                D[U[j]] = j
                U[D[j]] = j
                j = L[j]
            i = U[i]
        R[L[c]] = c
        L[R[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True
        # Pick column with minimum size (MRV heuristic)
        c = R[0]
        min_sz = sz[c]
        cur = R[c]
        while cur != 0:
            if sz[cur] < min_sz:
                min_sz = sz[cur]
                c = cur
            cur = R[cur]

        if min_sz == 0:
            return False

        cover(c)
        i = D[c]
        while i != c:
            ans.append(row_id[i])
            j = R[i]
            while j != i:
                cover(col[j])
                j = R[j]
            if search():
                return True
            j = L[i]
            while j != i:
                uncover(col[j])
                j = L[j]
            ans.pop()
            i = D[i]
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r in ans:
            a, b, c = triples[r]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
