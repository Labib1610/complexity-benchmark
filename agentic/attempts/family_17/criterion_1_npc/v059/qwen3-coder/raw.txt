from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    formulas = []
    index = 2
    for i in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        formulas.append((a, b, c))
        index += 3
    
    # Build graph: for each material, list all formulas that contain it
    graph = defaultdict(list)
    for idx, (a, b, c) in enumerate(formulas):
        graph[a].append(idx)
        graph[b].append(idx)
        graph[c].append(idx)
    
    # Try to find a matching that covers exactly q formulas
    # This is essentially a maximum matching problem in a 3-uniform hypergraph
    # We'll use a greedy approach with backtracking
    
    def can_match(formula_indices_used, materials_used, formula_count):
        if formula_count == q:
            return True, []
        
        for i in range(t):
            if i in formula_indices_used:
                continue
                
            a, b, c = formulas[i]
            if a in materials_used or b in materials_used or c in materials_used:
                continue
            
            # Try using this formula
            formula_indices_used.add(i)
            materials_used.add(a)
            materials_used.add(b)
            materials_used.add(c)
            
            found, result = can_match(formula_indices_used, materials_used, formula_count + 1)
            if found:
                return True, [(a, b, c)] + result
            
            # Backtrack
            formula_indices_used.remove(i)
            materials_used.remove(a)
            materials_used.remove(b)
            materials_used.remove(c)
        
        return False, []
    
    # Optimization: if we can't even select q formulas without conflicts, impossible
    # But let's just try the recursive solution with pruning
    
    found, result = can_match(set(), set(), 0)
    if found:
        print("YES")
        for a, b, c in reversed(result):
            print(a, b, c)
    else:
        print("NO")

solve()
