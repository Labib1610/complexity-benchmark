```python
import sys
input = sys.stdin.readline

def solve():
    line = input().split()
    if not line:
        return
    q = int(line[0])
    t = int(line[1])
    n = 3 * q
    
    cases = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        cases.append((a, b, c))
    
    # Build adjacency list: for each pair (u, v), list of third vertices w
    adj = {}
    for a, b, c in cases:
        adj.setdefault((a, b), []).append(c)
        adj.setdefault((a, c), []).append(b)
        adj.setdefault((b, c), []).append(a)
    
    # Sort for deterministic tie-breaking
    for key in adj:
        adj[key].sort()
    
    used = [False] * (n + 1)
    result = []
    
    # Greedy: find first unused vertex, try to form a triangle
    for _ in range(q):
        # Find first unused vertex
        u = 1
        while u <= n and used[u]:
            u += 1
        if u > n:
            print("NO")
            return
        
        found = False
        # Try to find v, w such that (u, v, w) is a valid case
        for v in range(u + 1, n + 1):
            if used[v]:
                continue
            key = (u, v)
            if key in adj:
                for w in adj[key]:
                    if w > v and not used[w]:
                        result.append((u, v, w))
                        used[u] = used[v] = used[w] = True
                        found = True
                        break
                if found:
                    break
        if not found:
            print("NO")
            return
    
    print("YES")
    for a, b, c in result:
        print(a, b, c)

if __name__ == "__main__":
    solve()
```
