```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0]); t = int(data[1])
    n = 3 * q
    triples = []
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1]); c = int(data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    # Build adjacency: for each pair, list of triple indices containing that pair
    pair_to_triples = {}
    for i, (a, b, c) in enumerate(triples):
        for x, y in ((a, b), (a, c), (b, c)):
            key = (x, y) if x < y else (y, x)
            if key not in pair_to_triples:
                pair_to_triples[key] = []
            pair_to_triples[key].append(i)
    
    # For each triple, find conflicting triples (sharing at least one pair)
    conflicts = [set() for _ in range(t)]
    for key, lst in pair_to_triples.items():
        L = len(lst)
        for i in range(L):
            ti = lst[i]
            for j in range(L):
                if i != j:
                    conflicts[ti].add(lst[j])
    
    # Convert to lists
    conflict_list = [list(s) for s in conflicts]
    
    # Greedy with conflict removal
    removed = [False] * t
    chosen = []
    remaining = t
    
    # Process triples in order; when we pick one, remove all conflicting
    for i in range(t):
        if removed[i]:
            continue
        # Pick triple i
        chosen.append(i)
        removed[i] = True
        remaining -= 1
        for j in conflict_list[i]:
            if not removed[j]:
                removed[j] = True
                remaining -= 1
    
    if len(chosen) == q:
        out = ["YES"]
        for i in chosen:
            a, b, c = triples[i]
            out.append(f"{a} {b} {c}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

solve()
```
