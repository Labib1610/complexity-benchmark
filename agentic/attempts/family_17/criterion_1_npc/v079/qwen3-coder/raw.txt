from collections import defaultdict
import sys

def find_triad_allocation():
    q, t = map(int, input().split())
    n = 3 * q
    
    if t == 0:
        if q == 0:
            print("YES")
            return
        else:
            print("NO")
            return
    
    triads = []
    person_to_triads = defaultdict(list)
    
    for i in range(t):
        a, b, c = map(int, input().split())
        triads.append((a, b, c))
        person_to_triads[a].append(i)
        person_to_triads[b].append(i)
        person_to_triads[c].append(i)
    
    # Try to find a valid allocation using backtracking
    selected = [False] * t
    assigned = [False] * (n + 1)
    result = []
    
    def backtrack(person):
        if person > n:
            return True
            
        if assigned[person]:
            return backtrack(person + 1)
            
        # Try each triad containing this person
        for triad_idx in person_to_triads[person]:
            if selected[triad_idx]:
                continue
                
            a, b, c = triads[triad_idx]
            
            # Check if all members are available
            if assigned[a] or assigned[b] or assigned[c]:
                continue
            
            # Select this triad
            selected[triad_idx] = True
            assigned[a] = assigned[b] = assigned[c] = True
            result.append(triad_idx)
            
            # Continue with next unassigned person
            next_person = person + 1
            while next_person <= n and assigned[next_person]:
                next_person += 1
                
            if backtrack(next_person):
                return True
                
            # Backtrack
            selected[triad_idx] = False
            assigned[a] = assigned[b] = assigned[c] = False
            result.pop()
            
        return False
    
    # Start with person 1
    if backtrack(1):
        print("YES")
        for idx in result:
            a, b, c = triads[idx]
            print(a, b, c)
    else:
        print("NO")

find_triad_allocation()
