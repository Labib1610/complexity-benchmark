import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triads = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triads.append((a, b, c))

    # DLX (Dancing Links) data structures
    max_nodes = n + 1 + 3 * t + 10

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    Row = [0] * max_nodes
    S = [0] * (n + 1)

    for i in range(n + 1):
        L[i] = (i - 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        C[i] = i
        S[i] = 0

    node_cnt = n + 1

    for r_idx, (a, b, c) in enumerate(triads):
        first = node_cnt
        for col in (a, b, c):
            curr = node_cnt
            node_cnt += 1
            C[curr] = col
            Row[curr] = r_idx
            S[col] += 1

            # Insert into column col
            U[curr] = U[col]
            D[curr] = col
            D[U[col]] = curr
            U[col] = curr

            # Insert into row
            if curr == first:
                L[curr] = curr
                R[curr] = curr
            else:
                L[curr] = L[first]
                R[curr] = first
                R[L[first]] = curr
                L[first] = curr

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with the minimum remaining candidate count
        c = R[0]
        min_s = S[c]
        curr = R[c]
        while curr != 0:
            if S[curr] < min_s:
                min_s = S[curr]
                c = curr
            curr = R[curr]

        if min_s == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            ans.append(Row[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            ans.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            r = D[r]

        uncover(c)
        return False

    if search():
        print("YES")
        for r_idx in ans:
            print(*(triads[r_idx]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
