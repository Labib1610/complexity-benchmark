import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q
    
    # Build adjacency list: for each pair of vertices, which third vertex forms a trio
    adj = [[] for _ in range(n + 1)]
    trios = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        trios.append((a, b, c))
        adj[a].append((b, c))
        adj[b].append((a, c))
        adj[c].append((a, b))
    
    # Sort adjacency lists for deterministic behavior
    for i in range(1, n + 1):
        adj[i].sort()
    
    used = [False] * (n + 1)
    result = []
    
    def dfs():
        if len(result) == q:
            return True
        
        # Find first unused vertex
        u = 1
        while u <= n and used[u]:
            u += 1
        
        if u > n:
            return False
        
        used[u] = True
        
        # Try all trios containing u
        for v, w in adj[u]:
            if not used[v] and not used[w]:
                used[v] = True
                used[w] = True
                result.append((u, v, w))
                
                if dfs():
                    return True
                
                result.pop()
                used[v] = False
                used[w] = False
        
        used[u] = False
        return False
    
    if dfs():
        print("YES")
        for trio in result:
            print(trio[0], trio[1], trio[2])
    else:
        print("NO")

solve()
