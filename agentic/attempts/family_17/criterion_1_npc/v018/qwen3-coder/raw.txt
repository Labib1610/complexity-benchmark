from collections import defaultdict
import sys

def solve():
    q, t = map(int, input().split())
    n = 3 * q
    
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    trios = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        trios.append((a, b, c))
    
    # Try all possible combinations of q trios
    # This is a matching problem - we want to select q trios such that
    # each operator appears exactly once
    
    def backtrack(index, selected, used):
        if len(selected) == q:
            return selected[:]
        
        if index >= t:
            return None
            
        # Try not taking current trio
        result = backtrack(index + 1, selected, used)
        if result is not None:
            return result
            
        # Try taking current trio
        a, b, c = trios[index]
        if a not in used and b not in used and c not in used:
            selected.append(index)
            used.add(a)
            used.add(b)
            used.add(c)
            
            result = backtrack(index + 1, selected, used)
            
            used.remove(a)
            used.remove(b)
            used.remove(c)
            selected.pop()
            
            if result is not None:
                return result
        
        return None
    
    # Optimization: if q is large, try a greedy approach first
    solution = backtrack(0, [], set())
    
    if solution is not None:
        print("YES")
        for idx in solution:
            a, b, c = trios[idx]
            print(a, b, c)
    else:
        print("NO")

solve()
