import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        triples.append((a, b, c))
        idx += 3

    # Dancing Links setup
    # Column headers 1..n, Root is 0
    max_nodes = n + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_id = [0] * max_nodes
    size = [0] * (n + 1)

    for i in range(n + 1):
        L[i] = (i - 1 + n + 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i

    node_cnt = n

    for r, (a, b, c) in enumerate(triples):
        cols = (a, b, c)
        first = node_cnt + 1
        for i, col in enumerate(cols):
            node_cnt += 1
            cur = node_cnt
            C[cur] = col
            row_id[cur] = r
            size[col] += 1

            # Up / Down link into column
            U[cur] = U[col]
            D[cur] = col
            D[U[col]] = cur
            U[col] = cur

            # Left / Right link within row
            if i == 0:
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = node_cnt - 1
                R[cur] = first
                R[node_cnt - 1] = cur
                L[first] = cur

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size
        c = R[0]
        min_sz = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_sz:
                min_sz = size[curr]
                c = curr
            curr = R[curr]

        if min_sz == 0:
            return False

        cover(c)
        row_node = D[c]
        while row_node != c:
            ans.append(row_id[row_node])

            right_node = R[row_node]
            while right_node != row_node:
                cover(C[right_node])
                right_node = R[right_node]

            if search():
                return True

            ans.pop()
            left_node = L[row_node]
            while left_node != row_node:
                uncover(C[left_node])
                left_node = L[left_node]

            row_node = D[row_node]

        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r in ans:
            a, b, c = triples[r]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
