import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    if t < q:
        print("NO")
        return
    
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    # Remove duplicate triples
    triples = sorted(list(set(triples)))
    
    # freq_to_triples: for each frequency 1..3q, list of triple indices
    freq_to_triples = [[] for _ in range(3 * q + 1)]
    for i, (a, b, c) in enumerate(triples):
        freq_to_triples[a].append(i)
        freq_to_triples[b].append(i)
        freq_to_triples[c].append(i)
        
    # Check if any frequency has 0 triples
    for f in range(1, 3 * q + 1):
        if not freq_to_triples[f]:
            print("NO")
            return

    # DLX / Backtracking state
    # We maintain active status for triples and covered status for frequencies.
    # To quickly find the frequency with min active triples, we can maintain the count of active triples per frequency.
    
    active_triple = [True] * len(triples)
    active_count = [len(freq_to_triples[f]) for f in range(3 * q + 1)]
    covered = [False] * (3 * q + 1)
    
    solution = []
    
    def search(uncovered_count):
        if uncovered_count == 0:
            return True
        
        # Choose uncovered frequency with minimum active_count
        best_f = -1
        min_c = float('inf')
        for f in range(1, 3 * q + 1):
            if not covered[f]:
                c = active_count[f]
                if c < min_c:
                    min_c = c
                    best_f = f
                    if min_c == 0:
                        return False
                    if min_c == 1:
                        break  # Can't get better than 1 candidate
        
        if best_f == -1 or min_c == 0:
            return False
        
        # Try each active triple for best_f
        # Make a copy of candidates to avoid mutation during iteration
        candidates = [tr_idx for tr_idx in freq_to_triples[best_f] if active_triple[tr_idx]]
        
        for tr_idx in candidates:
            a, b, c = triples[tr_idx]
            solution.append(tr_idx)
            
            # Cover a, b, c
            covered[a] = covered[b] = covered[c] = True
            
            # Deactivate all triples that share any element with (a, b, c)
            deactivated = []
            for f in (a, b, c):
                for other_tr in freq_to_triples[f]:
                    if active_triple[other_tr]:
                        active_triple[other_tr] = False
                        deactivated.append(other_tr)
                        oa, ob, oc = triples[other_tr]
                        active_count[oa] -= 1
                        active_count[ob] -= 1
                        active_count[oc] -= 1
            
            if search(uncovered_count - 3):
                return True
            
            # Backtrack
            solution.pop()
            covered[a] = covered[b] = covered[c] = False
            for other_tr in deactivated:
                active_triple[other_tr] = True
                oa, ob, oc = triples[other_tr]
                active_count[oa] += 1
                active_count[ob] += 1
                active_count[oc] += 1
                
        return False

    if search(3 * q):
        print("YES")
        for tr_idx in solution:
            print(f"{triples[tr_idx][0]} {triples[tr_idx][1]} {triples[tr_idx][2]}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
