We can model this as an exact cover problem: we have 3q frequencies and need to pick q triples that cover all frequencies exactly once.

Since q ≤ 200, the total number of frequencies is at most 600.  
We can use a backtracking approach with pruning, or use Algorithm X (Dancing Links) for exact cover.

Here’s a Python solution using backtracking with bitmasks for efficiency:

```python
import sys

def solve():
    input = sys.stdin.readline
    q, t = map(int, input().split())
    triples = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triples.append((a-1, b-1, c-1))
    
    n = 3 * q
    # Precompute bitmask for each triple
    masks = []
    for a, b, c in triples:
        masks.append((1 << a) | (1 << b) | (1 << c))
    
    full_mask = (1 << n) - 1
    
    # Backtracking
    chosen = []
    used = 0
    
    # Sort triples by some heuristic (e.g., by first frequency) to speed up
    order = sorted(range(t), key=lambda i: (triples[i][0], triples[i][1], triples[i][2]))
    
    def backtrack(start, used, chosen):
        if used == full_mask:
            return True
        # Find first uncovered frequency
        # We can find the lowest bit not set
        uncovered = (~used) & full_mask
        # lowest set bit
        lsb = uncovered & -uncovered
        idx = (lsb.bit_length() - 1)
        
        for i in range(start, t):
            idx2 = order[i]
            m = masks[idx2]
            if (m & used) == 0 and (m & lsb) != 0:
                chosen.append(triples[idx2])
                if backtrack(i+1, used | m, chosen):
                    return True
                chosen.pop()
        return False
    
    if backtrack(0, 0, chosen):
        print("YES")
        for a, b, c in chosen:
            print(a+1, b+1, c+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
