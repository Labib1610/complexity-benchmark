import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    # Read triads
    idx = 2
    triads = []
    elem_to_triads = [[] for _ in range(n + 1)]
    
    for i in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triads.append((a, b, c))
        elem_to_triads[a].append(i)
        elem_to_triads[b].append(i)
        elem_to_triads[c].append(i)

    # Check if every element appears at least once
    for i in range(1, n + 1):
        if not elem_to_triads[i]:
            print("NO")
            return

    # DLX structures
    # Columns are 1 to n (representing elements to be covered)
    # Rows are triads
    # Column headers: 0 is root, 1..n are column headers
    MAX_NODES = (n + 1) + 3 * t + 5
    L = [0] * MAX_NODES
    R = [0] * MAX_NODES
    U = [0] * MAX_NODES
    D = [0] * MAX_NODES
    col = [0] * MAX_NODES
    row = [0] * MAX_NODES
    size = [0] * (n + 1)

    # Initialize column headers
    for i in range(n + 1):
        L[i] = (i - 1 + (n + 1)) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        col[i] = i
        row[i] = -1

    node_count = n + 1

    for r_idx, (a, b, c) in enumerate(triads):
        cols = (a, b, c)
        first_node = node_count
        for i, c_idx in enumerate(cols):
            curr = node_count
            node_count += 1
            col[curr] = c_idx
            row[curr] = r_idx
            size[c_idx] += 1

            # Insert into column list (at the bottom)
            U[curr] = U[c_idx]
            D[curr] = c_idx
            D[U[c_idx]] = curr
            U[c_idx] = curr

            # Insert into row list
            if i == 0:
                L[curr] = curr
                R[curr] = curr
            else:
                L[curr] = L[first_node]
                R[curr] = first_node
                R[L[first_node]] = curr
                L[first_node] = curr

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size
        min_size = float('inf')
        c = -1
        curr = R[0]
        while curr != 0:
            if size[curr] < min_size:
                min_size = size[curr]
                c = curr
                if min_size == 0:
                    break
            curr = R[curr]

        if min_size == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            solution.pop()
            r = D[r]

        uncover(c)
        return False

    sys.setrecursionlimit(20000)
    if search():
        print("YES")
        for tid in solution:
            print(triads[tid][0], triads[tid][1], triads[tid][2])
    else:
        print("NO")

if __name__ == '__main__':
    solve()
