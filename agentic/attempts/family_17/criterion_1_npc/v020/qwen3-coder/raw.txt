from collections import defaultdict
import sys

def find_exhibition():
    q, t = map(int, input().split())
    
    triads = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triads.append((a, b, c))
    
    # If no triads, check if q=0 (but q>=1 per constraints)
    if t == 0:
        if q == 0:
            print("YES")
            return
        else:
            print("NO")
            return
    
    # Try all combinations of q triads from t triads
    # Since q <= 200 and t <= 20000, we need an efficient approach
    # This is essentially a matching problem: find q disjoint triads covering all 3q elements
    
    # Use backtracking with pruning
    def backtrack(index, selected, used):
        if len(selected) == q:
            return selected[:]
        
        if index >= t:
            return None
            
        # Pruning: if even selecting all remaining triads won't give us enough
        if len(selected) + (t - index) < q:
            return None
            
        # Try not taking current triad
        result = backtrack(index + 1, selected, used)
        if result is not None:
            return result
            
        # Try taking current triad if possible
        a, b, c = triads[index]
        if a not in used and b not in used and c not in used:
            selected.append(index)
            used.add(a)
            used.add(b)
            used.add(c)
            
            result = backtrack(index + 1, selected, used)
            
            if result is not None:
                return result
                
            selected.pop()
            used.remove(a)
            used.remove(b)
            used.remove(c)
        
        return None
    
    # Optimization: sort triads by earliest element to improve pruning
    # Also, build adjacency information
    artifact_to_triads = defaultdict(list)
    for i, (a, b, c) in enumerate(triads):
        artifact_to_triads[a].append(i)
        artifact_to_triads[b].append(i)
        artifact_to_triads[c].append(i)
    
    # More sophisticated approach: try to build incrementally
    # But given the constraints, let's try a different backtracking
    
    # Even better: model as exact cover problem and use DLX, but overkill here
    # Let's optimize our backtracking:
    
    # Precompute which triads conflict with each other (share artifacts)
    conflicts = [set() for _ in range(t)]
    for i in range(t):
        for j in range(i+1, t):
            # Check if triads i and j share any artifacts
            set_i = set(triads[i])
            set_j = set(triads[j])
            if set_i & set_j:  # If intersection is non-empty
                conflicts[i].add(j)
                conflicts[j].add(i)
    
    # Now do backtracking with this conflict information
    def backtrack_v2(selected_indices):
        if len(selected_indices) == q:
            return selected_indices[:]
            
        # Find next triad to consider
        start = selected_indices[-1] + 1 if selected_indices else 0
        
        for i in range(start, t):
            # Check if triad i conflicts with any already selected triad
            conflict = False
            for sel_idx in selected_indices:
                if i in conflicts[sel_idx]:
                    conflict = True
                    break
            
            if not conflict:
                selected_indices.append(i)
                result = backtrack_v2(selected_indices)
                if result is not None:
                    return result
                selected_indices.pop()
                
        return None
    
    # This still might be too slow. Let's go back to first approach but with better pruning
    
    solution = backtrack(0, [], set())
    
    if solution is None:
        print("NO")
    else:
        print("YES")
        for idx in solution:
            a, b, c = triads[idx]
            print(a, b, c)

find_exhibition()
