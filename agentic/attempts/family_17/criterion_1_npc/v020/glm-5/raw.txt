import sys
from collections import defaultdict

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    
    n = 3 * q
    
    # Map each triad to its index
    triad_to_idx = {}
    # For each artifact, list of triads containing it
    artifact_to_triads = defaultdict(list)
    
    for i in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        triad = (a, b, c)
        triad_to_idx[triad] = i
        artifact_to_triads[a].append(i)
        artifact_to_triads[b].append(i)
        artifact_to_triads[c].append(i)
    
    # Store triads for index lookup
    triads = [None] * t
    for triad, idx in triad_to_idx.items():
        triads[idx] = triad
    
    # For each artifact, track which triads contain it
    artifact_triad_list = [[] for _ in range(n + 1)]
    for artifact in range(1, n + 1):
        artifact_triad_list[artifact] = artifact_to_triads[artifact][:]
    
    # Matching state
    match = [-1] * t  # Which artifact set is matched to which triad (-1 = unmatched)
    used_artifact = [False] * (n + 1)  # Which artifacts are used
    match_count = 0
    
    # Try to match each artifact set (1,2,3), (4,5,6), ...
    for set_id in range(q):
        base = set_id * 3
        a1, a2, a3 = base + 1, base + 2, base + 3
        
        # Find triads that contain all three artifacts
        candidates = []
        for ti in artifact_triad_list[a1]:
            triad = triads[ti]
            if a2 in triad and a3 in triad:
                candidates.append(ti)
        
        if candidates:
            # Pick any valid triad
            ti = candidates[0]
            match[ti] = set_id
            match_count += 1
            continue
        
        # Need to find a triad that covers these artifacts through augmentation
        # BFS to find augmenting path
        parent = [-1] * t
        parent_set = [-1] * t  # Which set we're trying to steal for
        queue = []
        
        # Start with triads containing a1
        for ti in artifact_triad_list[a1]:
            if match[ti] == -1:
                parent[ti] = -2  # Mark as start
                parent_set[ti] = set_id
                queue.append(ti)
            else:
                # This triad is matched, we might steal it
                pass
        
        found = -1
        head = 0
        while head < len(queue) and found == -1:
            ti = queue[head]
            head += 1
            triad = triads[ti]
            
            # From this triad, we can try to match its artifacts to other triads
            for art in triad:
                target_set = None
                if art == a1:
                    target_set = set_id
                elif art <= base + 3 and art >= base + 1:
                    # This is one of our target artifacts
                    target_set = set_id
                else:
                    # This artifact belongs to another set
                    target_set = (art - 1) // 3
                
                if target_set is None or target_set == set_id:
                    continue
                
                # Look for triads that can cover target_set's artifacts
                tbase = target_set * 3
                ta1, ta2, ta3 = tbase + 1, tbase + 2, tbase + 3
                
                for tj in artifact_triad_list[ta1]:
                    if parent[tj] != -1:
                        continue
                    ttriad = triads[tj]
                    if ta2 in ttriad and ta3 in ttriad:
                        # This triad can cover target_set
                        if match[tj] == -1:
                            # Unmatched, we can use it
                            parent[tj] = ti
                            parent_set[tj] = target_set
                            found = tj
                            break
                        else:
                            # Matched, add to queue
                            parent[tj] = ti
                            parent_set[tj] = target_set
                            queue.append(tj)
                
                if found != -1:
                    break
        
        if found == -1:
            # Try different starting points
            for start_art in [a2, a3]:
                if found != -1:
                    break
                parent = [-1] * t
                parent_set = [-1] * t
                queue = []
                
                for ti in artifact_triad_list[start_art]:
                    if match[ti] == -1:
                        parent[ti] = -2
                        parent_set[ti] = set_id
                        queue.append(ti)
                
                head = 0
                while head < len(queue) and found == -1:
                    ti = queue[head]
                    head += 1
                    triad = triads[ti]
                    
                    for art in triad:
                        if art >= base + 1 and art <= base + 3:
                            # This is one of our target artifacts
                            # Check if we can complete
                            remaining = [a1, a2, a3]
                            for x in triad:
                                if x >= base + 1 and x <= base + 3 and x in remaining:
                                    remaining.remove(x)
                            
                            if len(remaining) == 0:
                                found = ti
                                break
                            
                            # Try to find triads for remaining artifacts
                            for rart in remaining:
                                for tj in artifact_triad_list[rart]:
                                    if parent[tj] != -1:
                                        continue
                                    if match[tj] == -1:
                                        parent[tj] = ti
                                        parent_set[tj] = set_id
                                        found = tj
                                        break
                                    else:
                                        parent[tj] = ti
                                        parent_set[tj] = set_id
                                        queue.append(tj)
                                if found != -1:
                                    break
                    
                    if found != -1:
                        break
        
        if found == -1:
            print("NO")
            return
        
        # Backtrack and update matches
        tj = found
        while tj >= 0:
            p = parent[tj]
            if p == -2:
                match[tj] = set_id
                break
            # Swap: tj takes over parent_set[tj], parent takes over parent's set
            old_match = match[tj]
            match[tj] = parent_set[tj]
            if old_match != -1:
                # Need to find new match for old_match
                set_id = old_match
                # This is handled by continuing the backtrack
            tj = p
        
        match_count += 1
    
    # Verify and output
    used = [False] * (n + 1)
    result = []
    matched_sets = set()
    
    for ti in range(t):
        if match[ti] != -1:
            triad = triads[ti]
            for art in triad:
                if used[art]:
                    print("NO")
                    return
                used[art] = True
            matched_sets.add(match[ti])
            result.append((match[ti], triad))
    
    if len(matched_sets) != q:
        print("NO")
        return
    
    for i in range(n + 1):
        if i >= 1 and not used[i]:
            print("NO")
            return
    
    result.sort()
    print("YES")
    for _, triad in result:
        print(triad[0], triad[1], triad[2])

def solve_simple():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    
    n = 3 * q
    
    triads = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        triads.append((a, b, c))
    
    # Simple greedy with backtracking
    used = [False] * (n + 1)
    result = []
    
    def backtrack(idx, count):
        if count == q:
            return True
        
        # Find next unused artifact
        start = 1
        while start <= n and used[start]:
            start += 1
        
        if start > n:
            return count == q
        
        # Try all triads containing 'start'
        for triad in triads:
            if start in triad:
                a, b, c = triad
                if not used[a] and not used[b] and not used[c]:
                    used[a] = used[b] = used[c] = True
                    result.append(triad)
                    if backtrack(idx + 1, count + 1):
                        return True
                    result.pop()
                    used[a] = used[b] = used[c] = False
        
        return False
    
    if backtrack(0, 0):
        print("YES")
        for triad in result:
            print(triad[0], triad[1], triad[2])
    else:
        print("NO")

solve_simple()
