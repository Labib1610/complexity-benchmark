```python
import sys

def main():
    sys.setrecursionlimit(5000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    by_first = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        by_first[a].append((b, c))
        
    used = bytearray(n + 1)
    chosen = []
    
    def dfs(first_uncovered):
        while first_uncovered <= n and used[first_uncovered]:
            first_uncovered += 1
        if first_uncovered > n:
            return True
            
        u = first_uncovered
        for b, c in by_first[u]:
            if not used[b] and not used[c]:
                used[u] = used[b] = used[c] = 1
                chosen.append((u, b, c))
                if dfs(u + 1):
                    return True
                chosen.pop()
                used[u] = used[b] = used[c] = 0
                
        return False

    if dfs(1):
        print("YES")
        out = []
        for a, b, c in chosen:
            out.append(f"{a} {b} {c}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        print("NO")

if __name__ == '__main__':
    main()
```
