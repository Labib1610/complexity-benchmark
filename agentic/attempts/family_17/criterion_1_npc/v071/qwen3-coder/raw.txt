from collections import defaultdict
import sys

def find_deployment(q, t, cells):
    if t < q:
        return "NO"
    
    # Create adjacency list for graph representation
    # Each operator is a node, each cell is a hyperedge
    # We'll model this as a matching problem
    
    # Try to find a perfect matching that covers all 3*q operators
    # This is equivalent to finding q disjoint cells
    
    # Use backtracking with pruning
    def backtrack(cell_idx, selected_cells, covered_operators):
        if len(selected_cells) == q:
            return selected_cells[:]
        
        if cell_idx >= len(cells):
            return None
            
        # Pruning: if remaining cells + current selection < q, impossible
        if len(selected_cells) + (len(cells) - cell_idx) < q:
            return None
            
        # Try including current cell
        cell = cells[cell_idx]
        if not any(op in covered_operators for op in cell):
            # Add cell
            selected_cells.append(cell_idx)
            for op in cell:
                covered_operators.add(op)
            
            result = backtrack(cell_idx + 1, selected_cells, covered_operators)
            if result is not None:
                return result
                
            # Backtrack
            selected_cells.pop()
            for op in cell:
                covered_operators.remove(op)
        
        # Try skipping current cell
        result = backtrack(cell_idx + 1, selected_cells, covered_operators)
        return result
    
    # Sort cells by smallest operator number for better pruning
    indexed_cells = [(i, cell) for i, cell in enumerate(cells)]
    indexed_cells.sort(key=lambda x: min(x[1]))
    sorted_indices = [x[0] for x in indexed_cells]
    sorted_cells = [cells[i] for i in sorted_indices]
    
    # Try to find solution
    solution_indices = backtrack(0, [], set())
    
    if solution_indices is None:
        return "NO"
    
    # Map back to original indices
    original_indices = [sorted_indices[i] for i in solution_indices]
    
    result = ["YES"]
    for idx in original_indices:
        result.append(f"{cells[idx][0]} {cells[idx][1]} {cells[idx][2]}")
    
    return "\n".join(result)

def main():
    input_lines = []
    for line in sys.stdin:
        input_lines.append(line.strip())
    
    first_line = list(map(int, input_lines[0].split()))
    q, t = first_line[0], first_line[1]
    
    cells = []
    for i in range(1, t + 1):
        if i < len(input_lines):
            cell_data = list(map(int, input_lines[i].split()))
            cells.append(tuple(cell_data))
    
    # Special case handling for efficiency
    if q == 0:
        print("YES")
        return
        
    if t == 0:
        print("NO")
        return
    
    # For small cases, try exhaustive search
    def solve_small():
        # Try all combinations of q cells
        from itertools import combinations
        
        for cell_combination in combinations(range(len(cells)), q):
            used_operators = set()
            valid = True
            
            for cell_idx in cell_combination:
                cell = cells[cell_idx]
                for op in cell:
                    if op in used_operators:
                        valid = False
                        break
                    used_operators.add(op)
                if not valid:
                    break
                    
            if valid and len(used_operators) == 3 * q:
                result = ["YES"]
                for cell_idx in cell_combination:
                    cell = cells[cell_idx]
                    result.append(f"{cell[0]} {cell[1]} {cell[2]}")
                return "\n".join(result)
                
        return "NO"
    
    # For larger cases, use greedy approach first
    def solve_greedy():
        # Greedy: pick cells that cover new operators
        selected = []
        covered = set()
        
        available_cells = list(range(len(cells)))
        
        for _ in range(q):
            best_cell = -1
            for i in available_cells:
                cell = cells[i]
                new_ops = sum(1 for op in cell if op not in covered)
                if new_ops == 3:  # Cell covers 3 new operators
                    best_cell = i
                    break
            
            if best_cell == -1:
                # Find cell with maximum new operators
                max_new = -1
                for i in available_cells:
                    cell = cells[i]
                    new_ops = sum(1 for op in cell if op not in covered)
                    if new_ops > max_new:
                        max_new = new_ops
                        best_cell = i
            
            if best_cell == -1:
                return None
                
            cell = cells[best_cell]
            selected.append(best_cell)
            for op in cell:
                covered.add(op)
            if best_cell in available_cells:
                available_cells.remove(best_cell)
        
        if len(covered) == 3 * q:
            return selected
        return None
    
    # Try greedy approach first
    if q <= 50:  # For smaller instances, try exact algorithm
        result = solve_small()
        print(result)
    else:
        greedy_result = solve_greedy()
        if greedy_result is not None:
            result = ["YES"]
            for idx in greedy_result:
                cell = cells[idx]
                result.append(f"{cell[0]} {cell[1]} {cell[2]}")
            print("\n".join(result))
        else:
            # Fallback to backtracking for medium cases
            result = find_deployment(q, t, cells)
            print(result)

if __name__ == "__main__":
    main()
