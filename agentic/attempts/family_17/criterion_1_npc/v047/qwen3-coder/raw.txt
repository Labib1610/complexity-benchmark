from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    if t == 0:
        print("NO")
        return
    
    teams = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        teams.append((a, b, c))
    
    # Build graph: each team is a node, edges connect teams that share members
    # We want to find a matching that covers all 3q people
    
    # This is essentially a maximum matching problem where we want to match
    # exactly q teams (nodes) such that no two teams share a person
    
    # Let's model this as finding a maximum independent set of size q in the 
    # conflict graph where nodes are teams and edges represent conflicts
    
    # Two teams conflict if they share at least one member
    def conflict(team1, team2):
        s1 = set(team1)
        s2 = set(team2)
        return len(s1 & s2) > 0
    
    # Build adjacency list for conflicts
    adj = [[] for _ in range(t)]
    for i in range(t):
        for j in range(i+1, t):
            if conflict(teams[i], teams[j]):
                adj[i].append(j)
                adj[j].append(i)
    
    # Now we need to find if there's an independent set of size q
    # This is equivalent to finding a clique of size q in the complement graph
    # But let's approach it differently - use backtracking with pruning
    
    # More direct approach: try all combinations but prune early
    # Actually, let's think of it as a set packing problem
    
    # Use bitmasks for small q, but q can be up to 200...
    # Let's try a greedy + backtracking approach
    
    # Better yet, model as exact cover problem and use DLX, but that might be overkill
    
    # Let's try a different approach: maximum matching in a 3-uniform hypergraph
    # But even simpler: since we want exactly q disjoint teams covering all 3q people,
    # this is like partitioning the complete set into q parts, each part being a team
    
    # Try backtracking with optimization
    used_person = [False] * (3*q + 1)
    selected_teams = []
    
    def backtrack(index):
        # If we've selected q teams, check if all people are covered
        if len(selected_teams) == q:
            # Check if all people are used
            for i in range(1, 3*q + 1):
                if not used_person[i]:
                    return False
            return True
        
        # Pruning: if remaining positions can't be filled, return
        remaining_teams = t - index
        needed_teams = q - len(selected_teams)
        if remaining_teams < needed_teams:
            return False
            
        # Try teams starting from index
        for i in range(index, t):
            a, b, c = teams[i]
            # If any member is already used, skip this team
            if used_person[a] or used_person[b] or used_person[c]:
                continue
                
            # Use this team
            used_person[a] = used_person[b] = used_person[c] = True
            selected_teams.append((a, b, c))
            
            # Recurse
            if backtrack(i + 1):
                return True
                
            # Backtrack
            used_person[a] = used_person[b] = used_person[c] = False
            selected_teams.pop()
        
        return False
    
    if backtrack(0):
        print("YES")
        for team in selected_teams:
            print(*team)
    else:
        print("NO")

solve()
