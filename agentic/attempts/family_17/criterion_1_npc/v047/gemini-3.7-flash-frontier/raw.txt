```python
import sys

def main():
    sys.setrecursionlimit(10000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        if 1 <= a < b < c <= n:
            adj[a].append((b, c))
            
    used = [False] * (n + 1)
    ans = []
    
    def dfs(x):
        while x <= n and used[x]:
            x += 1
        if x > n:
            return True
        
        for b, c in adj[x]:
            if not used[b] and not used[c]:
                used[b] = True
                used[c] = True
                ans.append((x, b, c))
                if dfs(x + 1):
                    return True
                ans.pop()
                used[b] = False
                used[c] = False
        return False

    if dfs(1):
        out = ["YES"]
        for triplet in ans:
            out.append(f"{triplet[0]} {triplet[1]} {triplet[2]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    main()
```
