import sys

def main():
    sys.setrecursionlimit(5000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    max_nodes = n + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    size = [0] * (n + 1)
    
    # Initialize column headers
    for i in range(n + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        col[i] = i
    L[0] = n
    R[n] = 0
    
    triples_list = []
    ptr = 2
    node_cnt = n + 1
    
    for idx in range(t):
        a = int(input_data[ptr])
        b = int(input_data[ptr + 1])
        c = int(input_data[ptr + 2])
        ptr += 3
        triples_list.append((a, b, c))
        
        n1 = node_cnt
        n2 = node_cnt + 1
        n3 = node_cnt + 2
        node_cnt += 3
        
        cols = (a, b, c)
        nodes = (n1, n2, n3)
        
        for k in range(3):
            nd = nodes[k]
            c_idx = cols[k]
            row[nd] = idx
            col[nd] = c_idx
            size[c_idx] += 1
            
            U[nd] = U[c_idx]
            D[nd] = c_idx
            D[U[c_idx]] = nd
            U[c_idx] = nd
            
        L[n1] = n3; R[n1] = n2
        L[n2] = n1; R[n2] = n3
        L[n3] = n2; R[n3] = n1

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True
        
        c = R[0]
        min_s = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_s:
                min_s = size[curr]
                c = curr
                if min_s == 0:
                    break
            curr = R[curr]
        
        if min_s == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            ans.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]
            if search():
                return True
            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            ans.pop()
            r = D[r]
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for idx in ans:
            a, b, c = triples_list[idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    main()
