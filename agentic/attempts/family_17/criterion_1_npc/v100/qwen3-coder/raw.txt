from itertools import combinations

def solve():
    q, t = map(int, input().split())
    configs = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        configs.append((a, b, c))
    
    # Try all combinations of q tri-bonds
    for combo in combinations(range(t), q):
        used_atoms = set()
        selected_bonds = []
        valid = True
        
        for i in combo:
            a, b, c = configs[i]
            if a in used_atoms or b in used_atoms or c in used_atoms:
                valid = False
                break
            used_atoms.add(a)
            used_atoms.add(b)
            used_atoms.add(c)
            selected_bonds.append((a, b, c))
        
        if valid and len(used_atoms) == 3 * q:
            print("YES")
            for bond in selected_bonds:
                print(bond[0], bond[1], bond[2])
            return
    
    print("NO")

solve()
