import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    n = 3 * q
    
    adj = [[] for _ in range(n + 1)]
    configs = []
    
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        configs.append((a, b, c))
        adj[a].append(len(configs) - 1)
        adj[b].append(len(configs) - 1)
        adj[c].append(len(configs) - 1)
    
    used = [False] * (n + 1)
    result = []
    
    # Greedy selection with backtracking
    # We process atoms in order and try to assign them to a tri-bond
    
    stack = []
    # State: (current_atom_index, result_list, used_array_snapshot)
    # To optimize, we use a recursive approach with manual stack
    
    # Let's use a simple backtracking approach
    # Since n <= 600, and branching factor is limited, this should work
    
    def backtrack(start_node):
        # Find first unused atom
        u = -1
        for i in range(start_node, n + 1):
            if not used[i]:
                u = i
                break
        
        if u == -1:
            return True
        
        # Try all configurations containing u
        for conf_idx in adj[u]:
            a, b, c = configs[conf_idx]
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                result.append((a, b, c))
                if backtrack(u + 1):
                    return True
                result.pop()
                used[a] = used[b] = used[c] = False
        
        return False
    
    if backtrack(1):
        sys.stdout.write("YES\n")
        for bond in result:
            sys.stdout.write(f"{bond[0]} {bond[1]} {bond[2]}\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
