```python
from collections import defaultdict
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    q, t = map(int, sys.stdin.readline().split())
    n = 3 * q
    
    triplets = []
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(t):
        a, b, c = map(int, sys.stdin.readline().split())
        triplets.append((a, b, c))
        adj[a].append(len(triplets) - 1)
        adj[b].append(len(triplets) - 1)
        adj[c].append(len(triplets) - 1)
    
    # Try all matchings using backtracking with pruning
    selected = [False] * t
    result = []
    used = [False] * (n + 1)
    
    def solve(pos):
        if pos == q:
            return True
            
        for i in range(t):
            if selected[i]:
                continue
            a, b, c = triplets[i]
            if used[a] or used[b] or used[c]:
                continue
                
            selected[i] = True
            used[a] = used[b] = used[c] = True
            result.append((a, b, c))
            
            if solve(pos + 1):
                return True
                
            selected[i] = False
            used[a] = used[b] = used[c] = False
            result.pop()
            
        return False
    
    # Optimization: if any node has degree 0, impossible
    for i in range(1, n + 1):
        if not adj[i]:
            print("NO")
            return
    
    if solve(0):
        print("YES")
        for triplet in result:
            print(*triplet)
    else:
        print("NO")

threading.Thread(target=main).start()
```
