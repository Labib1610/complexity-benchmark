```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(5000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triplets = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triplets.append((a, b, c))

    # Remove duplicates
    triplets = list(set(triplets))

    item_to_triplets = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(triplets):
        item_to_triplets[a].append(i)
        item_to_triplets[b].append(i)
        item_to_triplets[c].append(i)

    # Quick check: if any element cannot be covered, impossible
    for i in range(1, n + 1):
        if not item_to_triplets[i]:
            print("NO")
            return

    covered = [False] * (n + 1)
    ans = []

    def search(num_covered):
        if num_covered == n:
            return True

        best_cand = None
        min_count = 1000000

        for i in range(1, n + 1):
            if not covered[i]:
                cands = []
                for tr_idx in item_to_triplets[i]:
                    a, b, c = triplets[tr_idx]
                    if not covered[a] and not covered[b] and not covered[c]:
                        cands.append(tr_idx)
                cnt = len(cands)
                if cnt == 0:
                    return False
                if cnt < min_count:
                    min_count = cnt
                    best_cand = cands
                    if cnt == 1:
                        break

        for tr_idx in best_cand:
            a, b, c = triplets[tr_idx]
            if not covered[a] and not covered[b] and not covered[c]:
                covered[a] = covered[b] = covered[c] = True
                ans.append(tr_idx)
                if search(num_covered + 3):
                    return True
                ans.pop()
                covered[a] = covered[b] = covered[c] = False

        return False

    if search(0):
        print("YES")
        out = []
        for tr_idx in ans:
            a, b, c = triplets[tr_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
