import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    idx = 2
    triplets = []
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1]); c = int(data[idx+2])
        idx += 3
        triplets.append((a, b, c))
    
    n = 3 * q
    # Build adjacency: for each pair, list of triplets containing that pair
    pair_to_triplets = {}
    for i, (a, b, c) in enumerate(triplets):
        for x, y in ((a, b), (a, c), (b, c)):
            key = (x, y) if x < y else (y, x)
            pair_to_triplets.setdefault(key, []).append(i)
    
    # For each triplet, find candidate "partner" triplets sharing exactly 2 modules
    adj = [[] for _ in range(t)]
    for pair, tlist in pair_to_triplets.items():
        L = len(tlist)
        for i in range(L):
            ti = tlist[i]
            for j in range(i+1, L):
                tj = tlist[j]
                adj[ti].append(tj)
                adj[tj].append(ti)
    
    # Greedy matching with lookahead
    used_triplets = [False] * t
    used_modules = [False] * (n + 1)
    result = []
    
    for _ in range(q):
        best = None
        best_score = -1
        for i in range(t):
            if used_triplets[i]:
                continue
            a, b, c = triplets[i]
            if used_modules[a] or used_modules[b] or used_modules[c]:
                continue
            # Count available partners
            cnt = 0
            for j in adj[i]:
                if not used_triplets[j]:
                    x, y, z = triplets[j]
                    if not used_modules[x] and not used_modules[y] and not used_modules[z]:
                        cnt += 1
            if cnt > best_score:
                best_score = cnt
                best = i
                if cnt >= 2:
  #                  break  # good enough
                    pass
        
        if best is None:
            print("NO")
            return
        
        used_triplets[best] = True
        a, b, c = triplets[best]
        used_modules[a] = True
        used_modules[b] = True
        used_modules[c] = True
        result.append((a, b, c))
    
    print("YES")
    out = []
    for a, b, c in result:
        out.append(f"{a} {b} {c}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
