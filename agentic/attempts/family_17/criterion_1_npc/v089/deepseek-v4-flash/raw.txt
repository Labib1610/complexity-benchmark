We can model this as an exact cover problem: we need to select exactly \( q \) triplets from the catalog such that every module from 1 to \( 3q \) appears exactly once.

Since \( q \leq 200 \), the total number of modules is at most 600, and \( t \) can be up to 20000. A backtracking approach with pruning can work if we process modules in order and only try triplets that contain the smallest unused module.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    triplets = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triplets.append((a, b, c))
    
    n = 3 * q
    # Group triplets by their smallest element for quick lookup
    by_min = [[] for _ in range(n + 1)]
    for tri in triplets:
        by_min[tri[0]].append(tri)
    
    used = [False] * (n + 1)
    chosen = []
    
    def backtrack(start):
        # Find smallest unused module
        while start <= n and used[start]:
            start += 1
        if start > n:
            return True
        # Try triplets starting with this module
        for tri in by_min[start]:
            a, b, c = tri
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                chosen.append(tri)
                if backtrack(start + 1):
                    return True
                chosen.pop()
                used[a] = used[b] = used[c] = False
        return False
    
    if backtrack(1):
        print("YES")
        for tri in chosen:
            print(tri[0], tri[1], tri[2])
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
