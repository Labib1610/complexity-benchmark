from collections import defaultdict
import sys

def solve():
    q, t = map(int, input().split())
    
    triplets = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triplets.append((a, b, c))
    
    # If no triplets, impossible unless q=0
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    # Try all combinations of q triplets
    # Since q <= 200 and t <= 20000, we need an efficient approach
    # This is essentially a matching problem - find q triplets that cover all 3q elements exactly once
    
    # Use backtracking with pruning
    def backtrack(index, used, selected):
        # If we've selected q triplets, check if we've used all elements
        if len(selected) == q:
            if len(used) == 3 * q:
                return selected[:]
            return None
        
        # If remaining triplets can't fill the gap, prune
        if len(selected) + (t - index) < q:
            return None
            
        # Try including current triplet
        for i in range(index, t):
            a, b, c = triplets[i]
            if a not in used and b not in used and c not in used:
                used.add(a)
                used.add(b)
                used.add(c)
                selected.append(i)
                
                result = backtrack(i + 1, used, selected)
                if result is not None:
                    return result
                    
                selected.pop()
                used.remove(a)
                used.remove(b)
                used.remove(c)
        
        return None
    
    # Optimization: if q is large relative to t, impossible
    if q > t:
        print("NO")
        return
    
    result = backtrack(0, set(), [])
    
    if result is not None:
        print("YES")
        for idx in result:
            a, b, c = triplets[idx]
            print(a, b, c)
    else:
        print("NO")

solve()
