from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    setups = []
    component_to_setups = defaultdict(list)
    
    for i in range(t):
        a, b, c = map(int, input().split())
        setups.append((a, b, c))
        component_to_setups[a].append(i)
        component_to_setups[b].append(i)
        component_to_setups[c].append(i)
    
    # Try to find a matching using backtracking with optimizations
    used_setups = [False] * t
    selected_setups = []
    component_count = 3 * q
    covered = [False] * (component_count + 1)
    covered_count = 0
    
    def backtrack(pos):
        nonlocal covered_count
        
        if len(selected_setups) == q:
            return True
            
        if pos > component_count:
            return False
            
        if covered[pos]:
            return backtrack(pos + 1)
            
        # Try all setups containing component 'pos'
        for setup_idx in component_to_setups[pos]:
            if used_setups[setup_idx]:
                continue
                
            a, b, c = setups[setup_idx]
            
            # Check if all components in this setup are not yet covered
            if covered[a] or covered[b] or covered[c]:
                continue
                
            # Use this setup
            used_setups[setup_idx] = True
            selected_setups.append(setup_idx)
            covered[a] = covered[b] = covered[c] = True
            covered_count += 3
            
            if backtrack(pos + 1):
                return True
                
            # Backtrack
            used_setups[setup_idx] = False
            selected_setups.pop()
            covered[a] = covered[b] = covered[c] = False
            covered_count -= 3
            
        return False
    
    # Also try a greedy approach first
    def greedy():
        temp_used = [False] * t
        temp_covered = [False] * (component_count + 1)
        temp_selected = []
        
        # Priority queue: prioritize setups that cover components with fewer options
        component_options = [len(component_to_setups[i]) for i in range(1, component_count + 1)]
        
        for _ in range(q):
            best_setup = -1
            best_score = float('inf')
            
            # Find the uncovered component with fewest options
            target_component = -1
            min_options = float('inf')
            for i in range(1, component_count + 1):
                if not temp_covered[i] and component_options[i-1] < min_options:
                    min_options = component_options[i-1]
                    target_component = i
            
            if target_component == -1:
                break
                
            # Find best setup for this component
            for setup_idx in component_to_setups[target_component]:
                if temp_used[setup_idx]:
                    continue
                a, b, c = setups[setup_idx]
                if temp_covered[a] or temp_covered[b] or temp_covered[c]:
                    continue
                    
                # Score based on how many other components this would cover
                score = sum(1 for x in [a,b,c] if not temp_covered[x])
                if score < best_score:
                    best_score = score
                    best_setup = setup_idx
                    
            if best_setup == -1:
                return None
                
            temp_used[best_setup] = True
            temp_selected.append(best_setup)
            a, b, c = setups[best_setup]
            temp_covered[a] = temp_covered[b] = temp_covered[c] = True
            
        return temp_selected if len(temp_selected) == q else None
    
    # Try greedy first
    result = greedy()
    if result is not None:
        print("YES")
        for idx in result:
            a, b, c = setups[idx]
            print(a, b, c)
        return
    
    # If greedy fails, try backtracking from component 1
    if backtrack(1):
        print("YES")
        for idx in selected_setups:
            a, b, c = setups[idx]
            print(a, b, c)
    else:
        print("NO")

solve()
