import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    groups = []
    group_index = {}
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        groups.append((a, b, c))
        group_index[(a, b, c)] = len(groups) - 1
    
    n = 3 * q
    adj = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(groups):
        adj[a].append((b, c, idx))
        adj[b].append((a, c, idx))
        adj[c].append((a, b, idx))
    
    used_groups = [False] * t
    result = []
    
    # Process: find a group containing the smallest unassigned artifact
    assigned = [False] * (n + 1)
    
    for target in range(1, n + 1):
        if assigned[target]:
            continue
        # Find a group containing target
        found = False
        for b, c, idx in adj[target]:
            if not used_groups[idx] and not assigned[b] and not assigned[c]:
                used_groups[idx] = True
                assigned[target] = True
                assigned[b] = True
                assigned[c] = True
                result.append(groups[idx])
                found = True
                break
        if not found:
            print("NO")
            return
    
    print("YES")
    for g in result:
        print(g[0], g[1], g[2])

solve()
