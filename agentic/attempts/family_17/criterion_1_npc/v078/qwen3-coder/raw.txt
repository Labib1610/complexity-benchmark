import sys
from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    schematics = []
    for _ in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        schematics.append((a, b, c))
        index += 3
    
    # We need to find a perfect matching of q triplets from 3q elements
    # This is equivalent to finding a partition of the set {1,2,...,3q} into q triplets
    # where each triplet is in the given list of schematics
    
    # This is essentially a exact cover problem, which can be solved with backtracking
    
    # First, let's check if it's even possible by checking degrees etc.
    # But we'll just do backtracking
    
    # Build a graph: for each element, which schematics contain it
    element_to_schematics = defaultdict(list)
    for i, (a, b, c) in enumerate(schematics):
        element_to_schematics[a].append(i)
        element_to_schematics[b].append(i)
        element_to_schematics[c].append(i)
    
    # Try to find a matching
    selected = [False] * t
    result = []
    
    def backtrack(pos):
        if pos > 3*q:
            return len(result) == q
        if pos not in element_to_schematics:
            return False
            
        # For this position, try all schematics that contain it and haven't been used
        for i in element_to_schematics[pos]:
            if selected[i]:
                continue
            a, b, c = schematics[i]
            # Check if all elements of this schematic are available
            # We need to track which positions we've assigned
            # Actually, let's think differently
            
        # Let's try a different approach: try to build up the solution step by step
        # At each step, pick an unused element and try to put it in a triplet
        
        return False
    
    # Even better approach: try all possible combinations with pruning
    # Since q <= 200 and t <= 20000, and we're looking for exactly q triplets,
    # we can model this as finding a matching in a hypergraph
    
    # But let's think of it as a search problem:
    # State: which schematics are selected (bitmask or list)
    # Constraint: no element appears twice, total q schematics, all elements covered
    
    # Let's use recursive backtracking with optimization
    
    selected_schematics = []
    used_elements = [False] * (3*q + 1)
    
    def dfs(schematic_index):
        # If we've selected q schematics, check if we covered all elements
        if len(selected_schematics) == q:
            # Check if all elements from 1 to 3*q are used
            for i in range(1, 3*q + 1):
                if not used_elements[i]:
                    return False
            return True
            
        # If we already have too many schematics
        if len(selected_schematics) > q:
            return False
            
        # Pruning: if remaining elements can't form enough triplets
        count_unused = sum(1 for i in range(1, 3*q + 1) if not used_elements[i])
        if count_unused != 3 * (q - len(selected_schematics)):
            return False
            
        # Try next schematic
        for i in range(schematic_index, t):
            a, b, c = schematics[i]
            # Check if this schematic can be added
            if used_elements[a] or used_elements[b] or used_elements[c]:
                continue
                
            # Add this schematic
            selected_schematics.append(i)
            used_elements[a] = used_elements[b] = used_elements[c] = True
            
            # Recurse
            if dfs(i + 1):
                return True
                
            # Backtrack
            selected_schematics.pop()
            used_elements[a] = used_elements[b] = used_elements[c] = False
            
        return False
    
    # Special case: if q=0
    if q == 0:
        if t >= 0:
            print("YES")
            return
        else:
            print("NO")
            return
    
    if dfs(0):
        print("YES")
        for i in selected_schematics:
            a, b, c = schematics[i]
            print(a, b, c)
    else:
        print("NO")

solve()
