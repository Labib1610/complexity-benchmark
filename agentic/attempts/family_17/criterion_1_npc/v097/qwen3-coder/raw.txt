from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    configs = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        configs.append((a, b, c))
    
    if q == 0:
        print("YES")
        return
    
    # Build a graph for matching
    # Left side: people (1 to 3*q)
    # Right side: configurations (0 to t-1)
    # Edges represent that a person is in a configuration
    
    adj = defaultdict(list)  # person -> list of configuration indices
    rev_adj = [[] for _ in range(t)]  # config index -> list of people
    
    for i, (a, b, c) in enumerate(configs):
        adj[a].append(i)
        adj[b].append(i)
        adj[c].append(i)
        rev_adj[i] = [a, b, c]
    
    # Try to find a matching that covers all 3*q people
    # This is essentially a maximum matching problem
    # We want a matching of size q (since each config covers 3 people, q configs cover 3*q people)
    
    # Use backtracking with pruning
    # For each person, try to assign them to one of their available configurations
    
    # But let's think differently - we want to select exactly q configurations
    # such that all 3*q people are covered exactly once
    # This is a set cover problem where each set has size 3 and we want exactly q sets
    
    # Let's model this as finding a matching in a tripartite hypergraph
    # Or more simply, use backtracking with bitmasks
    
    # Since q <= 200, and we need to track which people are used,
    # let's use a different approach
    
    # Model as exact cover problem and use backtracking
    used_configs = [False] * t
    used_people = [False] * (3 * q + 1)
    result = []
    
    def backtrack(config_idx):
        # Count how many people we've covered
        count = sum(used_people)
        if count == 3 * q:
            return True
        if len(result) == q:
            return False
            
        # Try configurations starting from config_idx
        for i in range(config_idx, t):
            a, b, c = configs[i]
            if not used_people[a] and not used_people[b] and not used_people[c]:
                # Use this configuration
                used_people[a] = used_people[b] = used_people[c] = True
                result.append((a, b, c))
                
                if backtrack(i + 1):
                    return True
                    
                # Backtrack
                used_people[a] = used_people[b] = used_people[c] = False
                result.pop()
                
        return False
    
    # Optimization: if any person appears in 0 configurations, impossible
    person_count = [0] * (3 * q + 1)
    for a, b, c in configs:
        person_count[a] += 1
        person_count[b] += 1
        person_count[c] += 1
    
    for i in range(1, 3 * q + 1):
        if person_count[i] == 0:
            print("NO")
            return
    
    if backtrack(0):
        print("YES")
        for a, b, c in result:
            print(a, b, c)
    else:
        print("NO")

solve()
