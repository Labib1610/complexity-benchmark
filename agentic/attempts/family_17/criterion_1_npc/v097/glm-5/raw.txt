```python
import sys
input = sys.stdin.readline

def solve():
    q, t = map(int, input().split())
    n = 3 * q
    
    # Read all configurations
    configs = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        configs.append((a, b, c))
    
    # Greedy approach: repeatedly find a configuration that covers exactly one uncovered element
    # This works because if a valid partition exists, there's always at least one element
    # that appears in exactly one configuration among those in the partition
    
    used = [False] * (n + 1)
    selected = []
    
    # Build adjacency list for each element
    adj = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(configs):
        adj[a].append(i)
        adj[b].append(i)
        adj[c].append(i)
    
    # Track which configurations are still usable (all three elements unused)
    config_active = [True] * t
    
    # Track count of active configurations for each element
    active_count = [0] * (n + 1)
    for i in range(n + 1):
        active_count[i] = len(adj[i])
    
    # Process elements one by one
    for _ in range(q):
        # Find an element with exactly one active configuration
        found = False
        for elem in range(1, n + 1):
            if not used[elem] and active_count[elem] == 1:
                # Find the unique active configuration containing this element
                for ci in adj[elem]:
                    if config_active[ci]:
                        a, b, c = configs[ci]
                        # Use this configuration
                        selected.append(configs[ci])
                        used[a] = used[b] = used[c] = True
                        
                        # Deactivate all configurations that share elements
                        for x in [a, b, c]:
                            for cj in adj[x]:
                                if config_active[cj]:
                                    config_active[cj] = False
                                    # Update active counts
                                    a2, b2, c2 = configs[cj]
                                    active_count[a2] -= 1
                                    active_count[b2] -= 1
                                    active_count[c2] -= 1
                        
                        found = True
                        break
                if found:
                    break
        
        if not found:
            # No element with exactly one active configuration
            # Try any available configuration
            for ci in range(t):
                if config_active[ci]:
                    a, b, c = configs[ci]
                    if not used[a] and not used[b] and not used[c]:
                        selected.append(configs[ci])
                        used[a] = used[b] = used[c] = True
                        
                        # Deactivate all configurations that share elements
                        for x in [a, b, c]:
                            for cj in adj[x]:
                                if config_active[cj]:
                                    config_active[cj] = False
                                    a2, b2, c2 = configs[cj]
                                    active_count[a2] -= 1
                                    active_count[b2] -= 1
                                    active_count[c2] -= 1
                        
                        found = True
                        break
            
            if not found:
                print("NO")
                return
    
    if len(selected) == q:
        print("YES")
        for a, b, c in selected:
            print(a, b, c)
    else:
        print("NO")

solve()
```
