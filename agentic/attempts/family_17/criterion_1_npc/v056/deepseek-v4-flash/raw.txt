We can model this as a matching problem: we have \(3q\) gears, and each valid configuration is a triple. We need to select exactly \(q\) triples that are disjoint and cover all gears.

Since \(q \le 200\), the total number of gears is at most 600. We can use a backtracking or maximum matching approach.  
A good approach is to treat it as an exact cover problem and solve with Algorithm X (Dancing Links) or a simple recursive backtracking with pruning.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        # Check stability constraints
        if c - a >= 2 and a + b + c <= 9*q - 3 and a*b*c < 27*q**3:
            triples.append((a, b, c))
    
    n = 3 * q
    # Build adjacency: gear -> list of triple indices
    gear_to_triples = [[] for _ in range(n+1)]
    for i, (a, b, c) in enumerate(triples):
        gear_to_triples[a].append(i)
        gear_to_triples[b].append(i)
        gear_to_triples[c].append(i)
    
    used_triple = [False] * len(triples)
    used_gear = [False] * (n+1)
    chosen = []
    
    # Recursive backtracking
    def backtrack(start_gear):
        if len(chosen) == q:
            return True
        # Find first unused gear
        g = start_gear
        while g <= n and used_gear[g]:
            g += 1
        if g > n:
            return False
        # Try triples containing g
        for idx in gear_to_triples[g]:
            if used_triple[idx]:
                continue
            a, b, c = triples[idx]
            if used_gear[a] or used_gear[b] or used_gear[c]:
                continue
            # Choose
            used_triple[idx] = True
            used_gear[a] = used_gear[b] = used_gear[c] = True
            chosen.append((a, b, c))
            if backtrack(g+1):
                return True
            # Undo
            chosen.pop()
            used_triple[idx] = False
            used_gear[a] = used_gear[b] = used_gear[c] = False
        return False
    
    if backtrack(1):
        print("YES")
        for a, b, c in chosen:
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
