```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    ptr = 2
    raw_triplets = []
    max_sum = 9 * q - 3
    max_prod = 27 * (q ** 3)
    
    for _ in range(t):
        a = int(input_data[ptr])
        b = int(input_data[ptr + 1])
        c = int(input_data[ptr + 2])
        ptr += 3
        
        # Physical constraints
        if c - a >= 2 and (a + b + c) <= max_sum and (a * b * c) < max_prod:
            raw_triplets.append((a, b, c))
            
    n = 3 * q
    triplets_by_elem = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(raw_triplets):
        triplets_by_elem[a].append(idx)
        triplets_by_elem[b].append(idx)
        triplets_by_elem[c].append(idx)
        
    used = [False] * (n + 1)
    uncovered = set(range(1, n + 1))
    solution = []
    
    def search():
        if not uncovered:
            return True
        
        # Choose uncovered element with minimum valid triplets
        best_u = -1
        min_count = float('inf')
        best_options = []
        
        for u in uncovered:
            valid_opts = []
            for tid in triplets_by_elem[u]:
                a, b, c = raw_triplets[tid]
                if not used[a] and not used[b] and not used[c]:
                    valid_opts.append(tid)
            cnt = len(valid_opts)
            if cnt < min_count:
                min_count = cnt
                best_u = u
                best_options = valid_opts
                if min_count == 0:
                    return False
                if min_count == 1:
                    break
        
        for tid in best_options:
            a, b, c = raw_triplets[tid]
            if used[a] or used[b] or used[c]:
                continue
            
            used[a] = used[b] = used[c] = True
            uncovered.remove(a)
            uncovered.remove(b)
            uncovered.remove(c)
            solution.append(tid)
            
            if search():
                return True
            
            solution.pop()
            used[a] = used[b] = used[c] = False
            uncovered.add(a)
            uncovered.add(b)
            uncovered.add(c)
            
        return False

    if search():
        print("YES")
        for tid in solution:
            a, b, c = raw_triplets[tid]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
