import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q
    rows = []
    for _ in range(t):
        a = int(next(it)); b = int(next(it)); c = int(next(it))
        # constraints
        if c - a < 2:
            continue
        if a + b + c > 9 * q - 3:
            continue
        if a * b * c >= 27 * q * q * q:
            continue
        rows.append((a, b, c))

    # DLX structures
    max_nodes = len(rows) * 3 + n + 1
    U = [0] * max_nodes
    D = [0] * max_nodes
    L = [0] * max_nodes
    R = [0] * max_nodes
    C = [0] * max_nodes
    S = [0] * (n + 1)          # column sizes
    row_id = [0] * max_nodes   # which row a node belongs to
    col_head = [i for i in range(n + 1)]

    # header node 0
    for i in range(n + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = D[i] = i
    L[0] = n
    R[n] = 0

    node_cnt = n
    for r_idx, (a, b, c) in enumerate(rows, 1):
        first = None
        for col in (a, b, c):
            node_cnt += 1
            C[node_cnt] = col
            row_id[node_cnt] = r_idx
            # vertical link
            U[node_cnt] = U[col]
            D[node_cnt] = col
            D[U[col]] = node_cnt
            U[col] = node_cnt
            S[col] += 1
            # horizontal link
            if first is None:
                first = node_cnt
                L[node_cnt] = R[node_cnt] = node_cnt
            else:
                L[node_cnt] = L[first]
                R[node_cnt] = first
                R[L[first]] = node_cnt
                L[first] = node_cnt

    solution_rows = []

    sys.setrecursionlimit(10000)

    def cover(col):
        # remove column header
        L[R[col]] = L[col]
        R[L[col]] = R[col]
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[col]] = col
        R[L[col]] = col

    def search(k):
        if R[0] == 0:
            return True
        # choose column with minimal size
        c = R[0]
        min_s = S[c]
        j = R[c]
        while j != 0:
            if S[j] < min_s:
                c = j
                min_s = S[j]
                if min_s == 0:
                    break
            j = R[j]
        if min_s == 0:
            return False
        cover(c)
        r = D[c]
        while r != c:
            solution_rows.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            if search(k + 1):
                return True
            # backtrack
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution_rows.pop()
            r = D[r]
        uncover(c)
        return False

    if search(0):
        # map row ids back to triples
        used = set(solution_rows)
        out = []
        for idx, triple in enumerate(rows, 1):
            if idx in used:
                out.append(triple)
        print("YES")
        for a, b, c in out:
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
