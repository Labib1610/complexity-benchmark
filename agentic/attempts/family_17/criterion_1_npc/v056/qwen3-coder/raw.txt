from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    configs = []
    valid_configs = []
    
    # Read configurations
    for _ in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        index += 3
        configs.append((a, b, c))
        
        # Check stability conditions
        if c - a >= 2 and a + b + c <= 9*q - 3 and a * b * c < 27*q*q*q:
            valid_configs.append((a, b, c))
    
    # Create a graph where each configuration is a node and edges represent non-conflicting configurations
    # This is essentially a matching problem - we want to find a perfect matching that covers all 3q gears
    
    # We'll use a backtracking approach with pruning
    # Since q <= 200, and we need exactly q groups, let's try to find a valid partition
    
    # First, check if it's even possible by counting gear usage
    gear_count = [0] * (3*q + 1)
    for config in valid_configs:
        a, b, c = config
        gear_count[a] += 1
        gear_count[b] += 1
        gear_count[c] += 1
    
    # Build adjacency list for conflicts (which configurations share gears)
    config_gears = {}
    gear_to_configs = defaultdict(list)
    
    for i, config in enumerate(valid_configs):
        config_gears[i] = set(config)
        for gear in config:
            gear_to_configs[gear].append(i)
    
    # Conflict graph: two configs conflict if they share a gear
    conflicts = defaultdict(set)
    for gear in range(1, 3*q + 1):
        confs = gear_to_configs[gear]
        for i in range(len(confs)):
            for j in range(i+1, len(confs)):
                conflicts[confs[i]].add(confs[j])
                conflicts[confs[j]].add(confs[i])
    
    # Try to find a valid set of q non-conflicting configurations that use all gears
    selected = []
    used_gears = [False] * (3*q + 1)
    
    def backtrack(config_idx):
        # If we've selected q configurations, check if all gears are used
        if len(selected) == q:
            # Check if all gears are used
            for i in range(1, 3*q + 1):
                if not used_gears[i]:
                    return False
            return True
        
        # Pruning: if remaining configs can't fill the needed slots, return False
        remaining_slots = q - len(selected)
        if len(valid_configs) - config_idx < remaining_slots:
            return False
            
        # Try each configuration starting from config_idx
        for i in range(config_idx, len(valid_configs)):
            config = valid_configs[i]
            a, b, c = config
            
            # If any gear in this config is already used, skip
            if used_gears[a] or used_gears[b] or used_gears[c]:
                continue
                
            # If this config conflicts with any already selected config, skip
            conflict = False
            for sel_idx in selected:
                if i in conflicts[sel_idx]:
                    conflict = True
                    break
            if conflict:
                continue
            
            # Select this configuration
            selected.append(i)
            used_gears[a] = True
            used_gears[b] = True
            used_gears[c] = True
            
            # Recurse
            if backtrack(i + 1):
                return True
                
            # Backtrack
            selected.pop()
            used_gears[a] = False
            used_gears[b] = False
            used_gears[c] = False
            
        return False
    
    # Run backtracking
    if backtrack(0):
        print("YES")
        for idx in selected:
            a, b, c = valid_configs[idx]
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    main()
