import sys

# Increase recursion depth for deep backtracking if needed
sys.setrecursionlimit(50000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])

    N = 3 * q
    triples = []
    idx = 2
    seen = set()
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        trip = (a, b, c)
        if trip not in seen:
            seen.add(trip)
            triples.append(trip)

    num_triples = len(triples)
    # Total nodes: 1 (root) + N (columns) + 3 * num_triples
    max_nodes = 1 + N + 3 * num_triples + 10

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    S = [0] * (N + 1)
    row_of = [0] * max_nodes

    # Initialize column headers
    for i in range(N + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        C[i] = i
        S[i] = 0
    L[0] = N
    R[N] = 0

    node_count = N + 1

    for r_idx, (a, b, c) in enumerate(triples):
        cols = (a, b, c)
        first_node = node_count
        row_nodes = []
        for col in cols:
            curr_node = node_count
            node_count += 1
            row_nodes.append(curr_node)

            C[curr_node] = col
            row_of[curr_node] = r_idx
            S[col] += 1

            # Insert curr_node above col (at the bottom of column col)
            U[curr_node] = U[col]
            D[curr_node] = col
            D[U[col]] = curr_node
            U[col] = curr_node

        # Link row_nodes horizontally
        n1, n2, n3 = row_nodes
        L[n1] = n3
        R[n1] = n2
        L[n2] = n1
        R[n2] = n3
        L[n3] = n2
        R[n3] = n1

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        # MRV: Choose column c with minimum size S[c]
        c = R[0]
        min_s = S[c]
        curr = R[0]
        while curr != 0:
            if S[curr] < min_s:
                min_s = S[curr]
                c = curr
                if min_s == 0:
                    break
            curr = R[curr]

        if min_s == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_of[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            solution.pop()
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]

            r = D[r]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for trip_idx in solution:
            a, b, c = triples[trip_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
