from collections import defaultdict
import sys
sys.setrecursionlimit(100000)

def find_matching(q, t, chords):
    if t == 0:
        return "NO"
    
    # Build graph: each frequency -> list of chords that contain it
    freq_to_chords = defaultdict(list)
    for i in range(t):
        a, b, c = chords[i]
        freq_to_chords[a].append(i)
        freq_to_chords[b].append(i)
        freq_to_chords[c].append(i)
    
    # Try to find a matching using backtracking
    # We'll use a greedy approach with backtracking
    
    # chords_used[i] = True if chord i is used
    chords_used = [False] * t
    # freq_matched[f] = True if frequency f is matched
    freq_matched = [False] * (3*q + 1)
    result = []
    
    def backtrack(chord_count):
        if chord_count == q:
            return True
            
        # Find the first unmatched frequency
        first_unmatched = -1
        for f in range(1, 3*q + 1):
            if not freq_matched[f]:
                first_unmatched = f
                break
                
        if first_unmatched == -1:
            return False
            
        # Try all chords that contain this frequency
        for chord_idx in freq_to_chords[first_unmatched]:
            if chords_used[chord_idx]:
                continue
                
            a, b, c = chords[chord_idx]
            
            # Check if all frequencies in this chord are free
            if freq_matched[a] or freq_matched[b] or freq_matched[c]:
                continue
                
            # Use this chord
            chords_used[chord_idx] = True
            freq_matched[a] = True
            freq_matched[b] = True
            freq_matched[c] = True
            result.append((a, b, c))
            
            if backtrack(chord_count + 1):
                return True
                
            # Backtrack
            chords_used[chord_idx] = False
            freq_matched[a] = False
            freq_matched[b] = False
            freq_matched[c] = False
            result.pop()
            
        return False
    
    if backtrack(0):
        output = ["YES"]
        for chord in result:
            output.append(f"{chord[0]} {chord[1]} {chord[2]}")
        return "\n".join(output)
    else:
        return "NO"

# Read input
line = input().split()
q = int(line[0])
t = int(line[1])

chords = []
for _ in range(t):
    line = input().split()
    a, b, c = int(line[0]), int(line[1]), int(line[2])
    chords.append((a, b, c))

print(find_matching(q, t, chords))
