import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q
    
    chords = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        chords.append((a, b, c))
    
    # Build adjacency list: for each frequency, list of chord indices containing it
    freq_to_chords = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(chords):
        freq_to_chords[a].append(idx)
        freq_to_chords[b].append(idx)
        freq_to_chords[c].append(idx)
    
    # Hopcroft-Karp for maximum matching on bipartite graph
    # Left side: frequencies 1..n, Right side: chords 0..t-1
    # Edge: frequency i is in chord j
    
    INF = float('inf')
    
    # pairU[i] = chord matched to frequency i (1-indexed), -1 if unmatched
    pairU = [-1] * (n + 1)
    # pairV[j] = frequency matched to chord j (0-indexed), -1 if unmatched
    pairV = [-1] * t
    dist = [0] * (n + 1)
    
    from collections import deque
    
    def bfs():
        queue = deque()
        for u in range(1, n + 1):
            if pairU[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist_null = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist_null:
                for j in freq_to_chords[u]:
                    v = pairV[j]
                    if v == -1:
                        dist_null = dist[u] + 1
                    elif dist[v] == INF:
                        dist[v] = dist[u] + 1
                        queue.append(v)
        return dist_null != INF
    
    def dfs(u):
        for j in freq_to_chords[u]:
            v = pairV[j]
            if v == -1 or (dist[v] == dist[u] + 1 and dfs(v)):
                pairU[u] = j
                pairV[j] = u
                return True
        dist[u] = INF
        return False
    
    matching = 0
    while bfs():
        for u in range(1, n + 1):
            if pairU[u] == -1:
                if dfs(u):
                    matching += 1
    
    if matching == n:
        print("YES")
        for u in range(1, n + 1):
            j = pairU[u]
            a, b, c = chords[j]
            print(f"{a} {b} {c}")
    else:
        print("NO")

solve()
