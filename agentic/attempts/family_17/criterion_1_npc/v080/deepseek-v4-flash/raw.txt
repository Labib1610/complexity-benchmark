 We need to determine if we can select exactly \( q \) triples from the given list such that every ID from 1 to \( 3q \) appears exactly once. This is a perfect matching problem in a 3-uniform hypergraph.

We can model this as an exact cover problem and solve it using backtracking with bitmasking or Algorithm X (Dancing Links). Since \( q \leq 200 \), the number of vertices is at most 600, and the number of triples is at most 20000. We need an efficient approach.

We can use recursion with pruning:
- Maintain a list of unused vertices.
- Pick the smallest unused vertex, try all triples containing it, and recurse.
- Use bitmasks for fast operations.

Let's implement it.

```python
import sys
sys.setrecursionlimit(10000000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        triples.append((a, b, c))
    
    n = 3 * q
    # adjacency list: for each vertex, list of triple indices
    adj = [[] for _ in range(n)]
    for idx, (a, b, c) in enumerate(triples):
        adj[a].append(idx)
        adj[b].append(idx)
        adj[c].append(idx)
    
    used = [False] * n
    chosen = []
    triple_used = [False] * t
    
    # Precompute triple masks
    triple_mask = []
    for a, b, c in triples:
        triple_mask.append((1 << a) | (1 << b) | (1 << c))
    
    # For pruning: sort triples by size of adjacency? Not needed.
    
    def backtrack():
        if len(chosen) == q:
            return True
        # find first unused vertex
        v = -1
        for i in range(n):
            if not used[i]:
                v = i
                break
        if v == -1:
            return False
        # try triples containing v
        for idx in adj[v]:
            if triple_used[idx]:
                continue
            a, b, c = triples[idx]
            if used[a] or used[b] or used[c]:
                continue
            # choose
            used[a] = used[b] = used[c] = True
            triple_used[idx] = True
            chosen.append((a+1, b+1, c+1))
            if backtrack():
                return True
            # backtrack
            chosen.pop()
            triple_used[idx] = False
            used[a] = used[b] = used[c] = False
        return False
    
    if backtrack():
        print("YES")
        for a, b, c in chosen:
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
