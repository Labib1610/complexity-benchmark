import sys

# Increase recursion depth for search
sys.setrecursionlimit(3000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n_cols = 3 * q
    
    # DLX structures
    # Max nodes: 0 (root) + n_cols + 3 * t
    max_nodes = n_cols + 3 * t + 5
    
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    sz = [0] * (n_cols + 1)
    
    # Initialize column headers
    for i in range(n_cols + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
    L[0] = n_cols
    R[n_cols] = 0
    
    node_idx = n_cols + 1
    triples = []
    
    ptr = 2
    for r in range(t):
        a = int(input_data[ptr])
        b = int(input_data[ptr + 1])
        c = int(input_data[ptr + 2])
        ptr += 3
        
        triples.append((a, b, c))
        
        first = node_idx
        for k, c_idx in enumerate((a, b, c)):
            idx = node_idx + k
            col[idx] = c_idx
            row[idx] = r
            sz[c_idx] += 1
            
            # Insert into column list (at the bottom)
            U[idx] = U[c_idx]
            D[idx] = c_idx
            D[U[c_idx]] = idx
            U[c_idx] = idx
            
        L[first] = first + 2
        R[first] = first + 1
        L[first + 1] = first
        R[first + 1] = first + 2
        L[first + 2] = first + 1
        R[first + 2] = first
        
        node_idx += 3

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size
        c = R[0]
        min_sz = sz[c]
        curr = R[c]
        while curr != 0:
            if sz[curr] < min_sz:
                min_sz = sz[curr]
                c = curr
            curr = R[curr]
            
        if min_sz == 0:
            return False
            
        cover(c)
        i = D[c]
        while i != c:
            ans.append(row[i])
            j = R[i]
            while j != i:
                cover(col[j])
                j = R[j]
            if search():
                return True
            ans.pop()
            j = L[i]
            while j != i:
                uncover(col[j])
                j = L[j]
            i = D[i]
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in ans:
            a, b, c = triples[r_idx]
            out.append(f"{a} {b} {c}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
