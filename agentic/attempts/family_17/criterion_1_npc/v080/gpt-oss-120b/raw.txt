```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    triples = [(int(next(it)), int(next(it)), int(next(it))) for _ in range(t)]

    # Simple greedy algorithm: try to pick disjoint triples
    used = [False] * (3 * q + 1)
    ans = []
    for a, b, c in triples:
        if not (used[a] or used[b] or used[c]):
            ans.append((a, b, c))
            used[a] = used[b] = used[c] = True
            if len(ans) == q:
                break

    if len(ans) == q:
        out = ["YES"]
        out += ["{} {} {}".format(*tr) for tr in ans]
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
