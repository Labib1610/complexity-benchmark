from collections import defaultdict
import sys

def solve():
    q, t = map(int, input().split())
    protocols = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        protocols.append((a, b, c))
    
    # Try to find a perfect matching
    # This is essentially a 3D matching problem
    # We can model this as finding a maximum matching in a hypergraph
    # But since q <= 200, we can try a greedy approach with backtracking
    
    # Let's model this as an exact cover problem and use backtracking
    # Each person needs to be covered exactly once
    # Each protocol covers 3 people
    
    def backtrack(selected, used_people, index):
        if len(selected) == q:
            return selected[:]
        
        if index >= t:
            return None
            
        # Try including current protocol
        a, b, c = protocols[index]
        if a not in used_people and b not in used_people and c not in used_people:
            used_people.add(a)
            used_people.add(b)
            used_people.add(c)
            selected.append(index)
            
            result = backtrack(selected, used_people, index + 1)
            if result is not None:
                return result
                
            selected.pop()
            used_people.remove(a)
            used_people.remove(b)
            used_people.remove(c)
        
        # Try not including current protocol
        return backtrack(selected, used_people, index + 1)
    
    # Optimization: sort protocols by some heuristic?
    result = backtrack([], set(), 0)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for idx in result:
            a, b, c = protocols[idx]
            print(a, b, c)

solve()
