import sys


def solve():
    sys.setrecursionlimit(2000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    if t < q:
        print("NO")
        return

    N = 3 * q
    triples = []
    idx = 2
    for i in range(1, t + 1):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triples.append((a, b, c))

    max_nodes = N + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    size = [0] * (N + 1)

    # Initialize root and column headers
    L[0] = N
    R[0] = 1 if N > 0 else 0
    for i in range(1, N + 1):
        L[i] = i - 1
        R[i] = (i + 1) if i < N else 0
        U[i] = i
        D[i] = i
        col[i] = i
        size[i] = 0

    node_cnt = N + 1
    for r_idx, (c1, c2, c3) in enumerate(triples, 1):
        first_node = node_cnt
        cols = (c1, c2, c3)
        for k in range(3):
            nid = node_cnt + k
            c = cols[k]
            col[nid] = c
            row[nid] = r_idx

            # Link vertically
            U[nid] = U[c]
            D[nid] = c
            D[U[c]] = nid
            U[c] = nid
            size[c] += 1

        L[first_node] = first_node + 2
        R[first_node] = first_node + 1
        L[first_node + 1] = first_node
        R[first_node + 1] = first_node + 2
        L[first_node + 2] = first_node + 1
        R[first_node + 2] = first_node

        node_cnt += 3

    # If any column has size 0, no cover is possible
    for i in range(1, N + 1):
        if size[i] == 0:
            print("NO")
            return

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size
        c = R[0]
        min_size = size[c]
        cur = R[c]
        while cur != 0:
            if size[cur] < min_size:
                min_size = size[cur]
                c = cur
            cur = R[cur]

        if min_size == 0:
            return False

        cover(c)
        i = D[c]
        while i != c:
            solution.append(row[i])
            j = R[i]
            while j != i:
                cover(col[j])
                j = R[j]

            if search():
                return True

            j = L[i]
            while j != i:
                uncover(col[j])
                j = L[j]
            solution.pop()
            i = D[i]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r in solution:
            a, b, c = triples[r - 1]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
