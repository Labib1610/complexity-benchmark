from collections import defaultdict
import sys

def solve():
    q, t = map(int, input().split())
    triples = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triples.append((a, b, c))
    
    # Try all combinations of q triples
    # Use backtracking with pruning
    
    def can_form_complete_partition():
        # This is essentially a exact cover problem
        # We can model this as finding a matching in a hypergraph
        # But given small constraints, we can try backtracking
        
        if t < q:
            return None
            
        # Build adjacency for faster lookup
        used = [False] * t
        selected = []
        server_used = [False] * (3 * q + 1)
        
        def backtrack(index):
            if len(selected) == q:
                return True
                
            if index >= t:
                return False
                
            # Try including this triple
            a, b, c = triples[index]
            if not server_used[a] and not server_used[b] and not server_used[c]:
                server_used[a] = server_used[b] = server_used[c] = True
                selected.append((a, b, c))
                if backtrack(index + 1):
                    return True
                selected.pop()
                server_used[a] = server_used[b] = server_used[c] = False
            
            # Try not including this triple
            if backtrack(index + 1):
                return True
                
            return False
        
        if backtrack(0):
            return selected
        return None
    
    # Better approach: since q <= 200, and each server must be used exactly once,
    # this is a exact cover problem. We can solve it with backtracking but more efficiently.
    
    def solve_exact_cover():
        # Build graph where nodes are triples and edges connect overlapping triples
        # We want to find an independent set of size q that covers all 3*q servers
        
        if t < q:
            return None
            
        # More systematic backtracking
        selected_indices = []
        used_servers = set()
        
        def is_valid_add(triple_idx):
            a, b, c = triples[triple_idx]
            return a not in used_servers and b not in used_servers and c not in used_servers
        
        def add_triple(triple_idx):
            a, b, c = triples[triple_idx]
            used_servers.add(a)
            used_servers.add(b)
            used_servers.add(c)
            selected_indices.append(triple_idx)
            
        def remove_triple(triple_idx):
            a, b, c = triples[triple_idx]
            used_servers.remove(a)
            used_servers.remove(b)
            used_servers.remove(c)
            selected_indices.remove(triple_idx)
        
        def backtrack(index):
            if len(selected_indices) == q:
                return True
                
            if index >= t:
                return False
                
            # Pruning: if even selecting all remaining triples won't give us enough
            if len(selected_indices) + (t - index) < q:
                return False
                
            # Try including current triple
            if is_valid_add(index):
                add_triple(index)
                if backtrack(index + 1):
                    return True
                remove_triple(index)
            
            # Try not including current triple
            if backtrack(index + 1):
                return True
                
            return False
        
        if backtrack(0):
            return [triples[i] for i in selected_indices]
        return None
    
    result = solve_exact_cover()
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for triple in result:
            print(triple[0], triple[1], triple[2])

solve()
