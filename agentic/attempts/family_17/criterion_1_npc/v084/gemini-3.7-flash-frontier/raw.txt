import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    
    triplets = []
    seen = set()
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        
        # 0-indexed internally
        a -= 1
        b -= 1
        c -= 1
        trip = (a, b, c)
        if trip not in seen:
            seen.add(trip)
            triplets.append(trip)
            
    m = len(triplets)
    triplet_masks = [(1 << a) | (1 << b) | (1 << c) for a, b, c in triplets]
    
    by_element = [[] for _ in range(n)]
    for i, (a, b, c) in enumerate(triplets):
        by_element[a].append(i)
        by_element[b].append(i)
        by_element[c].append(i)
        
    # Quick check: any element with 0 triplets?
    for i in range(n):
        if not by_element[i]:
            print("NO")
            return

    solution = []
    full_mask = (1 << n) - 1

    def search(used_mask):
        if used_mask == full_mask:
            return True
        
        # Choose the uncovered element with minimum remaining valid candidates (MRV)
        best_u = -1
        min_candidates = None
        min_deg = 1000000
        
        # Find first uncovered element, and scan uncovered elements to find MRV
        temp = ~used_mask & full_mask
        while temp > 0:
            # get lowest set bit
            lsb = temp & -temp
            u = lsb.bit_length() - 1
            temp ^= lsb
            
            # Count valid triplets for u
            candidates = [tr for tr in by_element[u] if not (triplet_masks[tr] & used_mask)]
            deg = len(candidates)
            if deg == 0:
                return False
            if deg < min_deg:
                min_deg = deg
                best_u = u
                min_candidates = candidates
                if min_deg == 1:
                    break
        
        for tr in min_candidates:
            mask = triplet_masks[tr]
            solution.append(tr)
            if search(used_mask | mask):
                return True
            solution.pop()
            
        return False

    if search(0):
        print("YES")
        for tr_idx in solution:
            a, b, c = triplets[tr_idx]
            print(f"{a+1} {b+1} {c+1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
