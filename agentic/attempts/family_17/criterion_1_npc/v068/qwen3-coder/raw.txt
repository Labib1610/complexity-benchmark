import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2

    for _ in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        triples.append((a,b,c))
        index += 3
    
    # Build graph: each triple is a node, connect if they share a crate
    # But actually, we want to find a matching that covers all crates exactly once
    # This is a exact cover problem. We can model it as a graph matching or use backtracking.
    
    # Let's model this as an exact cover problem with Algorithm X / Dancing Links
    # But given the constraints (q <= 200), we can try a simpler backtracking approach
    
    # Crates are numbered 1 to 3*q
    # Each triple uses 3 distinct crates
    # Need to select exactly q triples such that each crate is used exactly once
    
    # Use backtracking with bitmasks or sets
    crates_needed = 3 * q
    triples_mask = [0] * t
    crate_to_triples = defaultdict(list)
    
    for i, (a,b,c) in enumerate(triples):
        mask = (1 << (a-1)) | (1 << (b-1)) | (1 << (c-1))
        triples_mask[i] = mask
        crate_to_triples[a].append(i)
        crate_to_triples[b].append(i)
        crate_to_triples[c].append(i)
    
    # We need to select q triples such that total mask has exactly 3*q bits set
    target_mask = (1 << (crates_needed)) - 1
    used = [False] * t
    selected = []
    
    def backtrack(pos, current_mask):
        if len(selected) == q:
            if current_mask == target_mask:
                return True
            else:
                return False
        if pos == t:
            return False
            
        # Pruning: if remaining triples + current selection < q, prune
        if len(selected) + (t - pos) < q:
            return False
            
        # Try including this triple
        mask = triples_mask[pos]
        if not (current_mask & mask): # No overlap
            selected.append(pos)
            if backtrack(pos + 1, current_mask | mask):
                return True
            selected.pop()
            
        # Try not including this triple
        if backtrack(pos + 1, current_mask):
            return True
            
        return False
    
    # Optimization: sort triples by how constrained they are?
    # Or start with triples that contain rare crates
    
    # For small inputs, simple backtracking might work
    # But let's try a more systematic approach
    
    # Alternative: build conflict graph and find maximum independent set of size q
    # But that's also hard
    
    # Let's go with the backtracking but optimize it
    
    # Sort triples by earliest crate to make search tree better
    # triples_with_index = list(enumerate(triples))
    # triples_with_index.sort(key=lambda x: min(x[1]))
    # order = [x[0] for x in triples_with_index]
    
    found = [False]
    result = []
    
    def dfs(pos, current_mask):
        if found[0]:
            return
        if len(selected) == q:
            if current_mask == target_mask:
                found[0] = True
                result.extend(selected[:])
            return
        if pos == t:
            return
        # Prune
        needed = q - len(selected)
        remaining = t - pos
        if remaining < needed:
            return
            
        # Try including current triple
        mask = triples_mask[pos]
        if (current_mask & mask) == 0:
            selected.append(pos)
            dfs(pos + 1, current_mask | mask)
            if found[0]:
                return
            selected.pop()
        
        # Try skipping current triple
        dfs(pos + 1, current_mask)
    
    dfs(0, 0)
    
    if found[0]:

            print(a, b, c)

if __name__ == "__main__":
    main()
