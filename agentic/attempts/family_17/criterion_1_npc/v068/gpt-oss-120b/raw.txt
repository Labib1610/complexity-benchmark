import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q  # number of items to cover
    rows = []
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        c = int(next(it)) - 1
        rows.append((a, b, c))

    # ---- DLX structures ----
    # column header nodes: 0..n-1
    # each node: left, right, up, down, column, row_id
    max_nodes = t * 3 + n + 1
    left = [0] * max_nodes
    right = [0] * max_nodes
    up = [0] * max_nodes
    down = [0] * max_nodes
    col = [0] * max_nodes
    row_id = [-1] * max_nodes
    size = [0] * n  # column sizes

    header = max_nodes - 1  # sentinel

    # init header links
    for i in range(n + 1):
        left[i] = i - 1
        right[i] = i + 1
        up[i] = down[i] = i
    left[0] = header
    right[header] = 0
    left[header] = n
    right[n] = header

    node_ptr = n + 1  # next free node index

    for r_idx, (a, b, c) in enumerate(rows):
        for item in (a, b, c):
            nd = node_ptr
            node_ptr += 1
            col[nd] = item
            row_id[nd] = r_idx
            # vertical links
            up[nd] = up[item]
            down[nd] = item
            down[up[item]] = nd
            up[item] = nd
            size[item] += 1
            # horizontal links (will close later)
            if left[nd] == 0 and right[nd] == 0:
                left[nd] = right[nd] = nd
            else:
                # insert to the left of the column header's first node
                left[nd] = left[item]
                right[nd] = item
                right[left[item]] = nd
                left[item] = nd

    solution = []

    def cover(c):
        # remove column header
        left[right[c]] = left[c]
        right[left[c]] = right[c]
        # remove rows
        i = down[c]
        while i != c:
            j = right[i]
            while j != i:
                down[up[j]] = down[j]
                up[down[j]] = up[j]
                size[col[j]] -= 1
                j = right[j]
            i = down[i]

    def uncover(c):
        i = up[c]
        while i != c:
            j = left[i]
            while j != i:
                size[col[j]] += 1
                down[up[j]] = j
                up[down[j]] = j
                j = left[j]
            i = up[i]
        left[right[c]] = c
        right[left[c]] = c

    def search(k):
        if right[header] == header:
            return True
        # choose column with minimal size
        c = right[header]
        min_sz = size[c]
        cc = c
        j = right[c]
        while j != header:
            if size[j] < min_sz:
                min_sz = size[j]
                cc = j
                if min_sz == 0:
                    break
            j = right[j]
        c = cc
        if min_sz == 0:
            return False
        cover(c)
        r = down[c]
        while r != c:
            solution.append(row_id[r])
            j = right[r]
            while j != r:
                cover(col[j])
                j = right[j]
            if search(k + 1):
                return True
            # backtrack
            solution.pop()
            j = left[r]
            while j != r:
                uncover(col[j])
                j = left[j]
            r = down[r]
        uncover(c)
        return False

    if search(0):
        out = ["YES"]
        for idx in solution:
            a, b, c = rows[idx]
            out.append(f"{a+1} {b+1} {c+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
