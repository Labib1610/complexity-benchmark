import sys

# Increase recursion depth just in case
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q
    
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    # Remove duplicate triples if any
    unique_triples = list(set(triples))
    t = len(unique_triples)
    
    # triples_of[u] contains indices of triples containing element u
    triples_of = [[] for _ in range(n + 1)]
    for i, (a, b, c) in enumerate(unique_triples):
        triples_of[a].append(i)
        triples_of[b].append(i)
        triples_of[c].append(i)
        
    degree = [len(triples_of[u]) for u in range(n + 1)]
    active = [True] * t
    uncovered = set(range(1, n + 1))
    solution = []
    
    def search():
        if not uncovered:
            return True
        
        # Pick element with minimum remaining valid triples
        min_u = -1
        min_deg = float('inf')
        for u in uncovered:
            d = degree[u]
            if d < min_deg:
                min_deg = d
                min_u = u
                if min_deg == 0:
                    return False
                if min_deg == 1:
                    break
        
        u = min_u
        
        # Try all active triples containing u
        candidate_triples = [tr_idx for tr_idx in triples_of[u] if active[tr_idx]]
        
        for tr_idx in candidate_triples:
            a, b, c = unique_triples[tr_idx]
            
            # Deactivate all triples sharing elements with (a, b, c)
            deactivated = []
            for elem in (a, b, c):
                uncovered.remove(elem)
                for other_tr in triples_of[elem]:
                    if active[other_tr]:
                        active[other_tr] = False
                        deactivated.append(other_tr)
                        oa, ob, oc = unique_triples[other_tr]
                        degree[oa] -= 1
                        degree[ob] -= 1
                        degree[oc] -= 1
            
            solution.append(tr_idx)
            if search():
                return True
            solution.pop()
            
            # Undo deactivations
            for elem in (a, b, c):
                uncovered.add(elem)
            for other_tr in deactivated:
                active[other_tr] = True
                oa, ob, oc = unique_triples[other_tr]
                degree[oa] += 1
                degree[ob] += 1
                degree[oc] += 1
                
        return False

    if search():
        print("YES")
        for tr_idx in solution:
            a, b, c = unique_triples[tr_idx]
            print(f"{a} {b} {c}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
