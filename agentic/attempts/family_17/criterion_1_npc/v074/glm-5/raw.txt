import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    triples = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        triples.append((a, b, c))
    
    n = 3 * q
    # Build adjacency list: for each pair (u, v), list of triples containing both
    adj = {}
    triple_set = set(triples)
    
    for i, (a, b, c) in enumerate(triples):
        pairs = [(a, b), (a, c), (b, c)]
        for u, v in pairs:
            key = (u, v)
            if key not in adj:
                adj[key] = []
            adj[key].append(i)
    
    # DP: dp[mask] = triple index used to achieve this mask, or -1 if not achievable
    # mask is bitmask of used artifacts
    # We use dictionary to store reachable states
    
    dp = {0: -1}  # mask -> triple index that led here (or -1 for start)
    parent = {}   # mask -> previous mask
    
    # Process states in order of increasing number of bits
    states = [0]
    state_idx = 0
    
    found_mask = None
    
    while state_idx < len(states):
        mask = states[state_idx]
        state_idx += 1
        
        if bin(mask).count('1') == n:
            found_mask = mask
            break
        
        # Find first unused artifact
        first_unused = -1
        for i in range(1, n + 1):
            if not (mask & (1 << i)):
                first_unused = i
                break
        
        if first_unused == -1:
            continue
        
        # Try all triples containing first_unused
        # We need to find triples that contain first_unused
        # For efficiency, iterate through triples
        for idx, (a, b, c) in enumerate(triples):
            if a == first_unused or b == first_unused or c == first_unused:
                # Check if all three are unused
                if not (mask & (1 << a)) and not (mask & (1 << b)) and not (mask & (1 << c)):
                    new_mask = mask | (1 << a) | (1 << b) | (1 << c)
                    if new_mask not in dp:
                        dp[new_mask] = idx
                        parent[new_mask] = mask
                        states.append(new_mask)
                        if bin(new_mask).count('1') == n:
                            found_mask = new_mask
                            break
        if found_mask is not None:
            break
    
    if found_mask is None:
        print("NO")
        return
    
    # Reconstruct solution
    print("YES")
    result = []
    mask = found_mask
    while mask != 0:
        idx = dp[mask]
        result.append(triples[idx])
        mask = parent[mask]
    
    result.reverse()
    for triple in result:
        print(triple[0], triple[1], triple[2])

solve()
