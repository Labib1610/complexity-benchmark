from collections import defaultdict

def solve():
    q, t = map(int, input().split())
    groups = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        groups.append((a, b, c))
    
    # Try all combinations of q groups
    # Since q <= 200 and t <= 20000, we need a more efficient approach
    
    # This is essentially a matching problem - we want to find q groups
    # that cover all 3q people exactly once
    
    # Use backtracking with pruning
    def backtrack(selected, used_people, index):
        if len(selected) == q:
            return selected[:]
        
        if index >= t:
            return None
            
        # Pruning: if even selecting all remaining groups won't give us enough
        if len(selected) + (t - index) < q:
            return None
            
        # Try including current group
        a, b, c = groups[index]
        if a not in used_people and b not in used_people and c not in used_people:
            selected.append(index)
            used_people.add(a)
            used_people.add(b)
            used_people.add(c)
            
            result = backtrack(selected, used_people, index + 1)
            if result is not None:
                return result
                
            selected.pop()
            used_people.remove(a)
            used_people.remove(b)
            used_people.remove(c)
        
        # Try not including current group
        return backtrack(selected, used_people, index + 1)
    
    # For larger inputs, we might need a different approach
    # Let's use a greedy approach with backtracking optimization
    
    if t < q:
        print("NO")
        return
        
    result = backtrack([], set(), 0)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for idx in result:
            a, b, c = groups[idx]
            print(a, b, c)

solve()
