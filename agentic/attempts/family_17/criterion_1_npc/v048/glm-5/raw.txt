import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    groups = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        groups.append((a, b, c))
    
    n = 3 * q
    
    # Build adjacency list: for each person, list of group indices containing them
    adj = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(groups):
        adj[a].append(idx)
        adj[b].append(idx)
        adj[c].append(idx)
    
    # visited[i] = True if group i has been selected
    visited = [False] * t
    # used[p] = True if person p has been assigned
    used = [False] * (n + 1)
    
    # We'll process persons in order, trying to match them
    # Use a stack for DFS-like matching
    result = []
    
    # For each unassigned person, try to find a group
    for person in range(1, n + 1):
        if used[person]:
            continue
        
        # DFS to find a matching starting from this person
        stack = [(person, 0)]  # (current person, index in adj[person])
        parent = {}  # group_idx -> (prev_person, prev_group_idx, index_in_adj)
        found = False
        target_group = -1
        
        while stack:
            p, idx = stack[-1]
            
            if idx < len(adj[p]):
                stack[-1] = (p, idx + 1)
                g_idx = adj[p][idx]
                
                if visited[g_idx]:
                    continue
                
                g = groups[g_idx]
                # Check if this group can be used (all members free)
                can_use = True
                conflict_person = -1
                for member in g:
                    if used[member]:
                        can_use = False
                        conflict_person = member
                        break
                
                if can_use:
                    # We can use this group - trace back and mark
                    visited[g_idx] = True
                    for member in g:
                        used[member] = True
                    result.append(g)
                    found = True
                    target_group = g_idx
                    break
                else:
                    # Try to find alternate assignment for conflict_person
                    if conflict_person not in [x[0] for x in stack]:
                        stack.append((conflict_person, 0))
            else:
                stack.pop()
        
        if not found:
            print("NO")
            return
    
    print("YES")
    for g in result:
        print(g[0], g[1], g[2])

solve()
