def main():
    data = sys.stdin.buffer.read().split()
    q = int(data[0]); t = int(data[1])
    n = 3 * q
    ptr = 2
    seen = set()
    triples = []
    for _ in range(t):
        a = int(data[ptr]); b = int(data[ptr+1]); c = int(data[ptr+2])
        ptr += 3
        key = (a, b, c)
        if key not in seen:
            seen.add(key)
            triples.append(key)
    
    parent = list(range(n + 1))
    def find(x):
        r = x
        while parent[r] != r:
            r = parent[r]
        while parent[x] != r:
            parent[x], x = r, parent[x]
        return r
    
    for (a, b, c) in triples:
        ra = find(a)
        rb = find(b)
        if ra != rb:
            parent[rb] = ra
        rc = find(c)
        if ra != rc:
            parent[rc] = ra
    
    comp_elems = {}
    for e in range(1, n + 1):
        r = find(e)
        if r in comp_elems:
            comp_elems[r].append(e)
        else:
            comp_elems[r] = [e]
    
    for elems in comp_elems.values():

    out = ["YES"]
