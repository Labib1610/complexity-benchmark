import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    n = 3 * q
    teams = []
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1]); c = int(data[idx+2])
        idx += 3
        teams.append((a, b, c))
    
    # Build adjacency: for each pair of vertices, list of third vertices forming a team
    # We'll use a dictionary keyed by (min, max) of two vertices
    from collections import defaultdict
    pair_to_third = defaultdict(list)
    for (a, b, c) in teams:
        pair_to_third[(a, b)].append(c)
        pair_to_third[(a, c)].append(b)
        pair_to_third[(b, c)].append(a)
    
    # We need exact cover of vertices 1..n with q triples from the given teams.
    # This is exact cover / 3-dimensional matching. With n up to 600, t up to 20000.
    # We'll use backtracking with the most-constrained-variable heuristic.
    
    # For quick lookup, convert pair_to_third lists to sets for fast membership
    pair_to_third_set = {}
    for k, v in pair_to_third.items():
        pair_to_third_set[k] = set(v)
    
    # State: set of uncovered vertices. We want to cover all.
    # Heuristic: pick the smallest uncovered vertex, try all teams containing it.
    # To find teams containing a vertex v, we can precompute vertex_to_teams.
    vertex_to_teams = [[] for _ in range(n+1)]
    for i, (a, b, c) in enumerate(teams):
        vertex_to_teams[a].append(i)
        vertex_to_teams[b].append(i)
        vertex_to_teams[c].append(i)
    
    used = [False] * (n+1)
    chosen = []
    
    # Precompute team membership as sets for quick checks
    team_members = [set(t) for t in teams]
    
    def backtrack(count):
        if count == q:
            return True
        # Find the smallest uncovered vertex
        v = -1
        for i in range(1, n+1):
            if not used[i]:
                v = i
                break
        if v == -1:
            return count == q
        # Try all teams containing v
        for ti in vertex_to_teams[v]:
            a, b, c = teams[ti]
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                chosen.append((a, b, c))
                if backtrack(count + 1):
                    return True
                chosen.pop()
                used[a] = used[b] = used[c] = False
        return False
    
    if backtrack(0):
        out = ["YES"]
        for team in chosen:
            out.append(f"{team[0]} {team[1]} {team[2]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

solve()
