from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    teams = []
    for i in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        teams.append((a, b, c))
        index += 3
    
    # Build graph for matching
    # Each person is a node on the left, each team is a node on the right
    # Edge exists if person is in team
    
    # But actually, this is a 3D matching problem. 
    # We want to select q teams such that each person appears exactly once.
    
    # This is equivalent to finding a perfect matching in a 3-uniform hypergraph
    # where each hyperedge is a team and we want to select q disjoint hyperedges
    # covering all 3q vertices.
    
    # We can model this as an exact cover problem and use backtracking or DLX
    # But given the small constraints (q <= 200), let's try a greedy + backtracking approach
    
    # Actually, let's model it as a matching problem:
    # Create a bipartite graph where:
    # Left side: 3q persons
    # Right side: t teams
    # Edge exists if person is in team
    # We want to select exactly q teams such that each person is covered exactly once
    # This is a variant of the set cover problem with exact coverage.
    
    # Use backtracking with optimizations
    # For each person, keep track of which teams contain this person
    
    person_to_teams = defaultdict(list)
    for i, (a, b, c) in enumerate(teams):
        person_to_teams[a].append(i)
        person_to_teams[b].append(i)
        person_to_teams[c].append(i)
    
    # Sort persons by number of teams they're in (for optimization)
    persons = list(range(1, 3*q + 1))
    persons.sort(key=lambda x: len(person_to_teams[x]))
    
    selected_teams = [False] * t
    used_persons = [False] * (3*q + 1)
    result = []
    
    def backtrack(person_index):
        if person_index == 3*q:
            return len(result) == q
        
        person = persons[person_index]
        if used_persons[person]:
            return backtrack(person_index + 1)
        
        # Try each team containing this person
        for team_index in person_to_teams[person]:
            if selected_teams[team_index]:
                continue
            
            a, b, c = teams[team_index]
            if used_persons[a] or used_persons[b] or used_persons[c]:
                continue
            
            # Select this team
            selected_teams[team_index] = True
            used_persons[a] = True
            used_persons[b] = True
            used_persons[c] = True
            result.append((a, b, c))
            
            if backtrack(person_index + 1):
                return True
            
            # Backtrack
            selected_teams[team_index] = False
            used_persons[a] = False
            used_persons[b] = False
            used_persons[c] = False
            result.pop()
        
        return False
    
    # Optimization: if any person is not in any team, impossible
    for person in range(1, 3*q + 1):
        if not person_to_teams[person]:
            print("NO")
            return
    
    if backtrack(0):
        print("YES")
        for team in result:
            print(team[0], team[1], team[2])
    else:
        print("NO")

if __name__ == "__main__":
    main()
