We need to select exactly \( q \) disjoint triples from the given \( t \) triples, covering all \( 3q \) people exactly once.  

This is essentially a **3-dimensional matching** problem, which is NP-hard in general, but here \( q \le 200 \) and \( t \le 20000 \), so we can use a backtracking approach with pruning, or reduce it to **exact cover** and solve with Algorithm X (Dancing Links).  

We’ll implement a backtracking search with bitmasks for efficiency.  

```python
import sys

def solve():
    input = sys.stdin.readline
    q, t = map(int, input().split())
    triples = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triples.append((a-1, b-1, c-1))  # 0-index

    n = 3 * q
    # Precompute bitmask for each triple
    masks = []
    for a, b, c in triples:
        masks.append((1 << a) | (1 << b) | (1 << c))

    # Sort triples by size of their smallest element to help pruning
    order = sorted(range(t), key=lambda i: (triples[i][0], triples[i][1], triples[i][2]))
    masks = [masks[i] for i in order]
    triples = [triples[i] for i in order]

    full_mask = (1 << n) - 1
    chosen = []
    used = 0

    # For pruning: for each position, list of triple indices that can be used
    # We'll do simple recursive backtracking with bitmask

    # Precompute for each person the list of triple indices containing them
    person_to_triples = [[] for _ in range(n)]
    for idx, (a, b, c) in enumerate(triples):
        person_to_triples[a].append(idx)
        person_to_triples[b].append(idx)
        person_to_triples[c].append(idx)

    # Recursive search
    def dfs(start_person, used_mask, chosen_list):
        if len(chosen_list) == q:
            return used_mask == full_mask
        # Find first unused person
        p = start_person
        while p < n and (used_mask >> p) & 1:
            p += 1
        if p == n:
            return False
        # Try triples containing p
        for idx in person_to_triples[p]:
            m = masks[idx]
            if used_mask & m:
                continue
            # Prune: if this triple contains already used person, skip
            chosen_list.append(idx)
            if dfs(p + 1, used_mask | m, chosen_list):
                return True
            chosen_list.pop()
        return False

    if dfs(0, 0, chosen):
        print("YES")
        for idx in chosen:
            a, b, c = triples[idx]
            print(a+1, b+1, c+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
