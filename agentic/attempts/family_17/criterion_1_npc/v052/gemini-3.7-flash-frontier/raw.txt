```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    if q == 0:
        print("YES")
        return

    blueprints = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx + 1]) - 1
        c = int(input_data[idx + 2]) - 1
        idx += 3
        blueprints.append((a, b, c))

    n = 3 * q
    adj = [[] for _ in range(n)]
    for i, bp in enumerate(blueprints):
        adj[bp[0]].append(i)
        adj[bp[1]].append(i)
        adj[bp[2]].append(i)

    used = [False] * n
    selected = []

    def backtrack(rem_q):
        if rem_q == 0:
            return True

        best_u = -1
        min_candidates = 999999
        best_cand_list = None

        for u in range(n):
            if not used[u]:
                cand = []
                for bp_idx in adj[u]:
                    a, b, c = blueprints[bp_idx]
                    if not used[a] and not used[b] and not used[c]:
                        cand.append(bp_idx)
                cnt = len(cand)
                if cnt == 0:
                    return False
                if cnt < min_candidates:
                    min_candidates = cnt
                    best_u = u
                    best_cand_list = cand
                    if cnt == 1:
                        break

        for bp_idx in best_cand_list:
            a, b, c = blueprints[bp_idx]
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                selected.append(bp_idx)
                if backtrack(rem_q - 1):
                    return True
                selected.pop()
                used[a] = used[b] = used[c] = False

        return False

    if backtrack(q):
        print("YES")
        for bp_idx in selected:
            a, b, c = blueprints[bp_idx]
            print(f"{a + 1} {b + 1} {c + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
