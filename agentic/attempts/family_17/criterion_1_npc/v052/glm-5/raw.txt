import sys
from collections import defaultdict

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return
    
    n = 3 * q
    adj = defaultdict(list)
    edges = []
    
    for i in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        edges.append((u, v, w))
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        adj[w].append((u, v, i))
    
    # Greedy matching approach
    used_edges = set()
    matched = [False] * (n + 1)
    
    # Sort nodes by degree to prioritize constrained nodes
    nodes = list(range(1, n + 1))
    nodes.sort(key=lambda x: len(adj[x]))
    
    for u in nodes:
        if matched[u]:
            continue
        # Find an available edge containing u
        found = False
        for v, w, idx in adj[u]:
            if not matched[v] and not matched[w]:
                used_edges.add(idx)
                matched[u] = True
                matched[v] = True
                matched[w] = True
                found = True
                break
        if found:
            continue
            
    if len(used_edges) == q:
        print("YES")
        for i in used_edges:
            u, v, w = edges[i]
            print(u, v, w)
        return
    
    # If greedy fails, use backtracking with pruning
    # This handles cases where greedy makes wrong choices
    sys.setrecursionlimit(10000)
    
    edge_list = edges
    edge_used = [False] * t
    node_used = [False] * (n + 1)
    result_indices = []
    
    def backtrack(count):
        if count == q:
            return True
        # Find first unused node
        start = 1
        while start <= n and node_used[start]:
            start += 1
        if start > n:
            return False
        
        # Try all edges containing start
        for v, w, idx in adj[start]:
            if edge_used[idx]:
                continue
            if node_used[v] or node_used[w]:
                continue
            
            edge_used[idx] = True
            node_used[start] = True
            node_used[v] = True
            node_used[w] = True
            result_indices.append(idx)
            
            if backtrack(count + 1):
                return True
            
            result_indices.pop()
            node_used[start] = False
            node_used[v] = False
            node_used[w] = False
            edge_used[idx] = False
        
        return False
    
    if backtrack(0):
        print("YES")
        for i in result_indices:
            u, v, w = edges[i]
            print(u, v, w)
    else:
        print("NO")

solve()
