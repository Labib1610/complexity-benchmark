We can model this as an exact cover problem: we have \(3q\) items and need to pick \(q\) triples that cover all items exactly once.  
Since \(q \le 200\) and \(t \le 20000\), we can use backtracking with pruning.

Here’s the Python solution:

```python
import sys

def solve():
    input = sys.stdin.readline
    q, t = map(int, input().split())
    triples = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triples.append((a, b, c))
    
    n = 3 * q
    used = [False] * (n + 1)
    chosen = []
    
    # Precompute adjacency: for each relic, list of triples containing it
    relic_to_triples = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(triples):
        relic_to_triples[a].append(idx)
        relic_to_triples[b].append(idx)
        relic_to_triples[c].append(idx)
    
    # Sort triples by sum of relic codes for deterministic behavior
    triples_sorted = sorted(range(t), key=lambda i: sum(triples[i]))
    
    def backtrack():
        if len(chosen) == q:
            return True
        # Find first unused relic
        first = 1
        while first <= n and used[first]:
            first += 1
        if first > n:
            return False
        
        # Try triples containing this relic
        for idx in relic_to_triples[first]:
            a, b, c = triples[idx]
            if used[a] or used[b] or used[c]:
                continue
            # Choose this triple
            used[a] = used[b] = used[c] = True
            chosen.append((a, b, c))
            if backtrack():
                return True
            # Backtrack
            chosen.pop()
            used[a] = used[b] = used[c] = False
        return False
    
    if backtrack():
        print("YES")
        for a, b, c in chosen:
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
