```python
import sys
input = sys.stdin.readline

def solve():
    line1 = input().split()
    if not line1:
        return
    q = int(line1[0])
    t = int(line1[1])
    
    n = 3 * q  # Total number of personnel
    
    if q == 0:
        print("YES")
        return
    
    # triangles[i] will store list of triangles containing vertex i+1
    triangles = [[] for _ in range(n + 1)]
    # Store all triangles (1-indexed IDs)
    all_triangles = []
    
    for _ in range(t):
        parts = list(map(int, input().split()))
        a, b, c = parts[0], parts[1], parts[2]
        all_triangles.append((a, b, c))
        triangles[a].append((a, b, c))
        triangles[b].append((a, b, c))
        triangles[c].append((a, b, c))
    
    # DP state: match[i] stores list of triangles matched for first i personnel
    # match[i] = None if impossible
    match = [None] * (n + 1)
    match[0] = []
    
    for i in range(n):
        if match[i] is None:
            continue
        
        # If i+1 is already matched via a previous triangle that covers it, skip
        # Actually i represents the number of matched personnel, so we try to match person i+1 next
        
        u = i + 1
        
        if u + 2 > n:
            continue
            
        # Try all triangles containing u that could cover persons u, v, w
        # We need v > u and w > v for this to be valid in our greedy/DP ordering
        # But triangles might not be sorted. Let's check all.
        
        for tri in triangles[u]:
            a, b, c = tri
            # We need this triangle to cover exactly persons i+1, some j, some k
            # where i+1 is the smallest uncovered.
            # So one of a,b,c must be u = i+1.
            # The other two must be > i (i.e., not already matched).
            # Since we process i from 0 to n, matched set is {1..i}.
            # So other two must be > i.
            
            # Check if this triangle is valid for transition from i
            others = []
            if a == u:
                others = [b, c]
            elif b == u:
                others = [a, c]
            elif c == u:
                others = [a, b]
            else:
                continue
            
            v, w = min(others), max(others)
            
            # Both v and w must be > i (not yet matched)
            # i is the count of matched, so if we take this triangle, we match {u, v, w}
            # indices u, v, w are 1-based.
            # Valid if v > i and w > i? 
            # Actually valid if v > i and w > i.
            # u is i+1, so u > i is automatic.
            # We just need v > i and w > i.
            
            if v > i and w > i:
                new_mask = i + 3
                if match[new_mask] is None:
                    # We can store the new match
                    # But we need to recover the triangles later.
                    # For now, just store the list.
                    match[new_mask] = match[i] + [tri]
    
    result = match[n]
    if result is not None and len(result) == q:
        print("YES")
        for tri in result:
            print(tri[0], tri[1], tri[2])
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
