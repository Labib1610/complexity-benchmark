import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    num_elements = 3 * q
    if t < q:
        print("NO")
        return
        
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        triples.append((a, b, c))
        idx += 3

    # DLX Implementation
    # Columns are 1 to num_elements
    # Column 0 is the root node
    
    # We allocate arrays for DLX nodes:
    # Max nodes: 1 (root) + num_elements (column headers) + t * 3 (for each cell in triples)
    max_nodes = 1 + num_elements + t * 3 + 10
    
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col_of = [0] * max_nodes
    row_of = [0] * max_nodes
    col_size = [0] * (num_elements + 1)
    
    # Initialize column headers
    for i in range(num_elements + 1):
        L[i] = (i - 1) % (num_elements + 1)
        R[i] = (i + 1) % (num_elements + 1)
        U[i] = i
        D[i] = i
        col_of[i] = i
        col_size[i] = 0
        
    node_cnt = num_elements + 1
    
    for r_idx, (a, b, c) in enumerate(triples):
        row_nodes = []
        for c_idx in (a, b, c):
            cur = node_cnt
            node_cnt += 1
            
            col_of[cur] = c_idx
            row_of[cur] = r_idx
            col_size[c_idx] += 1
            
            # Insert cur into column c_idx list (at the bottom, above c_idx)
            U[cur] = U[c_idx]
            D[cur] = c_idx
            D[U[c_idx]] = cur
            U[c_idx] = cur
            
            row_nodes.append(cur)
            
        # Link row horizontally
        L[row_nodes[0]] = row_nodes[2]
        R[row_nodes[0]] = row_nodes[1]
        
        L[row_nodes[1]] = row_nodes[0]
        R[row_nodes[1]] = row_nodes[2]
        
        L[row_nodes[2]] = row_nodes[1]
        R[row_nodes[2]] = row_nodes[0]

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_size[col_of[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_size[col_of[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size
        c = R[0]
        min_size = col_size[c]
        temp = R[c]
        while temp != 0:
            if col_size[temp] < min_size:
                min_size = col_size[temp]
                c = temp
            temp = R[temp]
            
        if min_size == 0:
            return False
            
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_of[r])
            j = R[r]
            while j != r:
                cover(col_of[j])
                j = R[j]
                
            if search():
                return True
                
            solution.pop()
            j = L[r]
            while j != r:
                uncover(col_of[j])
                j = L[j]
            r = D[r]
            
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in solution:
            out.append(f"{triples[r_idx][0]} {triples[r_idx][1]} {triples[r_idx][2]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
