We can model this as an exact cover problem: we have `3q` operators and need to cover them with disjoint triples from the allowed list.

Since `q` can be up to 200, the number of operators is at most 600. The number of triples can be up to 20,000.  
We can solve this with backtracking + bitmask representation for efficiency.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    q = int(input_data[0])
    t = int(input_data[1])
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        idx += 3
        triples.append((a, b, c))
    
    n = 3 * q
    full_mask = (1 << n) - 1
    
    # Precompute mask for each triple
    masks = []
    for a, b, c in triples:
        masks.append((1 << a) | (1 << b) | (1 << c))
    
    # Sort triples by number of remaining possibilities? We'll just backtrack.
    # Use memoization for visited states
    from functools import lru_cache
    
    # We'll store chosen triples indices
    chosen = []
    
    # For speed, we can group triples by their first element
    by_first = [[] for _ in range(n)]
    for i, (a, b, c) in enumerate(triples):
        by_first[a].append(i)
    
    # Backtracking
    def backtrack(mask):
        if mask == full_mask:
            return True
        # Find first uncovered operator
        # Use bit trick to find lowest zero bit
        # invert mask and get lowest set bit
        inv = (~mask) & full_mask
        if inv == 0:
            return False
        # lowest set bit
        first = (inv & -inv).bit_length() - 1
        for idx in by_first[first]:
            m = masks[idx]
            if mask & m == 0:
                chosen.append(idx)
                if backtrack(mask | m):
                    return True
                chosen.pop()
        return False
    
    if backtrack(0):
        print("YES")
        for idx in chosen:
            a, b, c = triples[idx]
            print(a+1, b+1, c+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
