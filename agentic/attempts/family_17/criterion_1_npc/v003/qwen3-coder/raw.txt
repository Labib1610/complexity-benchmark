from itertools import combinations

def solve():
    q, t = map(int, input().split())
    triples = []
    for _ in range(t):
        a, b, c = map(int, input().split())
        triples.append((a, b, c))
    
    # We need to select exactly q triples to cover all 3*q operators
    # This is essentially a matching problem - we want to find a perfect matching
    # where each operator appears exactly once
    
    # Try all combinations of q triples from the t available triples
    for combo_indices in combinations(range(t), q):
        selected_triples = [triples[i] for i in combo_indices]
        
        # Check if this selection covers all operators exactly once
        operators = []
        for triple in selected_triples:
            operators.extend(triple)
        
        # Check if we have exactly 3*q operators and they're all distinct
        if len(operators) == 3 * q and len(set(operators)) == 3 * q:
            print("YES")
            for triple in selected_triples:
                print(triple[0], triple[1], triple[2])
            return
    
    print("NO")

solve()
