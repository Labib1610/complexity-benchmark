import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    q = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    n = 3 * q

    triples = []
    for _ in range(t):
        a = int(data[idx]) - 1; idx += 1
        b = int(data[idx]) - 1; idx += 1
        c = int(data[idx]) - 1; idx += 1
        triples.append((a, b, c))

    if q == 0:
        print("YES")
        return

    if t == 0:
        print("NO")
        return

    # Build adjacency: for each pair, list of triples containing it
    pair_to_triples = {}
    for i, (a, b, c) in enumerate(triples):
        for x, y in ((a, b), (a, c), (b, c)):
            key = (x, y) if x < y else (y, x)
            if key not in pair_to_triples:
                pair_to_triples[key] = []
            pair_to_triples[key].append(i)

    # Build graph: nodes are triples, edges connect triples sharing exactly 2 vertices
    adj = [[] for _ in range(t)]
    for key, lst in pair_to_triples.items():
        for i in range(len(lst)):
            for j in range(i + 1, len(lst)):
                adj[lst[i]].append(lst[j])
                adj[lst[j]].append(lst[i])

    # Find connected components
    comp_id = [-1] * t
    comp_list = []
    for i in range(t):
        if comp_id[i] == -1:
            comp = []
            stack = [i]
            comp_id[i] = len(comp_list)
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if comp_id[v] == -1:
                        comp_id[v] = len(comp_list)
                        stack.append(v)
            comp_list.append(comp)

    # For each component, find a perfect matching
    # Use greedy + backtracking with pruning
    result = [None] * q
    found = False

    for comp in comp_list:
        if found:
            break
        comp_set = set(comp)
        comp_size = len(comp)

        if comp_size % 2 != 0:
            print("NO")
            return

        # Sort by degree (ascending) for better pruning
        comp_sorted = sorted(comp, key=lambda x: len(adj[x]))

        # Precompute adjacency within component
        comp_adj = {}
        for u in comp:
            comp_adj[u] = [v for v in adj[u] if v in comp_set]

        # Greedy: match vertices with only one neighbor first
        temp_matched = {}
        temp_available = set(comp)
        ok = True

        # Process forced matches (degree 1 vertices)
        changed = True
        while changed and ok:
            changed = False
            for u in list(temp_available):
                neighbors = [v for v in comp_adj[u] if v in temp_available]
                if len(neighbors) == 0:
                    ok = False
                    break
                if len(neighbors) == 1:
                    v = neighbors[0]
                    temp_matched[u] = v
                    temp_matched[v] = u
                    temp_available.discard(u)
                    temp_available.discard(v)
                    changed = True

        if not ok:
            print("NO")
            return

        remaining = sorted(temp_available, key=lambda x: len([v for v in comp_adj[x] if v in temp_available]))

        # Backtracking on remaining
        rem_list = remaining
        rem_set = set(remaining)
        rem_adj = {u: [v for v in comp_adj[u] if v in rem_set] for u in rem_list}
        rem_idx = {u: i for i, u in enumerate(rem_list)}
        m = len(rem_list)

        if m == 0:
            # All matched greedily
            for u, v in temp_matched.items():
                if u < v:
                    slot = len([x for x in range(q) if result[x] is not None])
                    if slot >= q:
                        ok = False
                        break
                    result[slot] = triples[u]
            if not ok:
                print("NO")
                return
            continue

        # Bitmask DP for small remaining
        if m <= 22:
            full_mask = (1 << m) - 1
            dp = [None] * (1 << m)
            dp[full_mask] = ()
            for mask in range(full_mask, -1, -1):
                if dp[mask] is None:
                    continue
                if mask == 0:
                    break
                # Find first unmatched
                first = -1
                for i in range(m):
                    if not (mask & (1 << i)):
                        first = i
                        break
                if first == -1:
                    continue
                u = rem_list[first]
                for v in rem_adj[u]:
                    j = rem_idx[v]
                    if not (mask & (1 << j)):
                        new_mask = mask | (1 << first) | (1 << j)
                        if dp[new_mask] is None:
                            dp[new_mask] = (u, v) + dp[mask]
            if dp[0] is None:
                print("NO")
                return
            # Collect all matches
            all_matches = []
            for u, v in temp_matched.items():
                if u < v:
                    all_matches.append((u, v))
            # Extract from dp
            ptr = 0
            arr = dp[0]
            i = 0
            while i < len(arr):
                all_matches.append((arr[i], arr[i+1]))
                i += 2
            for u, v in all_matches:
                slot = 0
                while slot < q and result[slot] is not None:
                    slot += 1
                if slot >= q:
                    print("NO")
                    return
                result[slot] = triples[u]
            found = True
            break
        else:
            # Backtracking
            match_arr = [None] * m
            used = [False] * m

            def backtrack(pos):
                if pos == m:
                    return True
                if used[pos]:
                    return backtrack(pos + 1)
                u = rem_list[pos]
                for v in rem_adj[u]:
                    j = rem_idx[v]
                    if not used[j]:
                        used[pos] = True
                        used[j] = True
                        match_arr[pos] = v
                        match_arr[j] = u
                        if backtrack(pos + 1):
                            return True
                        used[pos] = False
                        used[j] = False
                        match_arr[pos] = None
                        match_arr[j] = None
                return False

            if not backtrack(0):
                print("NO")
                return

            all_matches = []
            for u, v in temp_matched.items():
                if u < v:
                    all_matches.append((u, v))
            for i in range(m):
                if match_arr[i] is not None:
                    u = rem_list[i]
                    v = match_arr[i]
                    if u < v:
                        all_matches.append((u, v))

            for u, v in all_matches:
                slot = 0
                while slot < q and result[slot] is not None:
                    slot += 1
                if slot >= q:
                    print("NO")
                    return
                result[slot] = triples[u]
            found = True
            break

    if not found:
        print("NO")
        return

    # Verify
    seen = set()
    for triple in result:
        for x in triple:
            if x in seen:
                print("NO")
                return
            seen.add(x)
    if len(seen) != n:
        print("NO")
        return

    print("YES")
    for triple in result:
        print(f"{triple[0]+1} {triple[1]+1} {triple[2]+1}")

solve()
