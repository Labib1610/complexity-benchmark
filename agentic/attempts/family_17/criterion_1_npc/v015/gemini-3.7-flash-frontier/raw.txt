import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(5000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])
    n = 3 * q

    triplets = []
    idx = 2
    seen = set()
    for _ in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx + 1]) - 1
        c = int(input_data[idx + 2]) - 1
        idx += 3
        trip = tuple(sorted((a, b, c)))
        if trip not in seen:
            seen.add(trip)
            triplets.append(trip)

    trip_masks = [(1 << a) | (1 << b) | (1 << c) for a, b, c in triplets]

    elem_trips = [[] for _ in range(n)]
    for i, (a, b, c) in enumerate(triplets):
        elem_trips[a].append(i)
        elem_trips[b].append(i)
        elem_trips[c].append(i)

    for i in range(n):
        if not elem_trips[i]:
            print("NO")
            return

    target = (1 << n) - 1
    solution = []

    def search(covered):
        if covered == target:
            return True

        best_elem = -1
        min_cands = float("inf")
        best_cands = None

        uncov = target ^ covered
        curr = uncov
        while curr:
            lsb = curr & -curr
            i = lsb.bit_length() - 1
            curr ^= lsb

            valid = [
                idx for idx in elem_trips[i] if (trip_masks[idx] & covered) == 0
            ]
            c = len(valid)
            if c == 0:
                return False
            if c < min_cands:
                min_cands = c
                best_elem = i
                best_cands = valid
                if c == 1:
                    break

        for cand_idx in best_cands:
            solution.append(cand_idx)
            if search(covered | trip_masks[cand_idx]):
                return True
            solution.pop()

        return False

    if search(0):
        print("YES")
        for cand_idx in solution:
            a, b, c = triplets[cand_idx]
            print(f"{a+1} {b+1} {c+1}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
