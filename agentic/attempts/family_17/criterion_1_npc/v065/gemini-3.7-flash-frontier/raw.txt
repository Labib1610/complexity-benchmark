import sys

# Increase recursion depth for deep search trees
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    q = int(input_data[0])
    t = int(input_data[1])

    n = 3 * q

    triplets = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        triplets.append((a, b, c))

    max_nodes = (n + 1) + 3 * t + 10

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row_id = [0] * max_nodes
    sz = [0] * (n + 1)

    # Initialize column headers
    for i in range(n + 1):
        L[i] = (i - 1 + n + 1) % (n + 1)
        R[i] = (i + 1) % (n + 1)
        U[i] = i
        D[i] = i
        sz[i] = 0

    node_cnt = n + 1

    for r_idx, (c1, c2, c3) in enumerate(triplets):
        first = -1
        for c in (c1, c2, c3):
            node = node_cnt
            node_cnt += 1
            col[node] = c
            row_id[node] = r_idx
            sz[c] += 1

            # Insert vertically at the bottom of column c
            D[node] = c
            U[node] = U[c]
            D[U[c]] = node
            U[c] = node

            # Insert horizontally in current row
            if first == -1:
                first = node
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first]
                R[node] = first
                R[L[first]] = node
                L[first] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        # Choose column c with minimum size
        c = R[0]
        min_sz = sz[c]
        temp = R[0]
        while temp != 0:
            if sz[temp] < min_sz:
                min_sz = sz[temp]
                c = temp
                if min_sz <= 1:
                    break
            temp = R[temp]

        if min_sz == 0:
            return False

        cover(c)
        i = D[c]
        while i != c:
            ans.append(row_id[i])
            j = R[i]
            while j != i:
                cover(col[j])
                j = R[j]

            if search():
                return True

            ans.pop()
            j = L[i]
            while j != i:
                uncover(col[j])
                j = L[j]
            i = D[i]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in ans:
            a, b, c = triplets[r_idx]
            out.append(f"{a} {b} {c}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")


if __name__ == "__main__":
    solve()
