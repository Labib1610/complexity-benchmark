import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    groups = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        groups.append((a, b, c))
    
    n = 3 * q
    
    # Build adjacency: for each pair of operators, which groups contain both
    pair_to_groups = {}
    group_contains = []
    for idx, (a, b, c) in enumerate(groups):
        group_contains.append((a, b, c))
        for x, y in [(a, b), (a, c), (b, c)]:
            key = (x, y)
            if key not in pair_to_groups:
                pair_to_groups[key] = []
            pair_to_groups[key].append(idx)
    
    # For each operator, which groups contain it
    op_to_groups = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(groups):
        op_to_groups[a].append(idx)
        op_to_groups[b].append(idx)
        op_to_groups[c].append(idx)
    
    used = [False] * (n + 1)
    result = []
    
    # Greedy: pick smallest unused operator, try to form a valid triple
    for _ in range(q):
        # Find smallest unused operator
        start = 1
        while start <= n and used[start]:
            start += 1
        if start > n:
            print("NO")
            return
        
        found = False
        # Try each group containing start
        for gidx in op_to_groups[start]:
            a, b, c = groups[gidx]
            # Check if all three are unused
            if not used[a] and not used[b] and not used[c]:
                used[a] = used[b] = used[c] = True
                result.append((a, b, c))
                found = True
                break
        
        if not found:
            print("NO")
            return
    
    print("YES")
    for a, b, c in result:
        print(a, b, c)

if __name__ == "__main__":
    solve()
