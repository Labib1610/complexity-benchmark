import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    groups = []
    for i in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        groups.append((a, b, c))
        index += 3
    
    # We need to find a perfect matching that covers all 3q nodes
    # This is a exact cover problem, we can model it as finding a matching in a hypergraph
    # But since each group has size 3 and we want to cover all nodes exactly once,
    # this is equivalent to finding a partition of the set {1, 2, ..., 3q} into sets of size 3
    # where each set is one of our allowed groups.
    
    # We can solve this using backtracking or by reducing to maximum matching
    # Let's use backtracking with optimizations
    
    # First, let's build a mapping from each element to the list of groups containing it
    elem_to_groups = defaultdict(list)
    for i, (a, b, c) in enumerate(groups):
        elem_to_groups[a].append(i)
        elem_to_groups[b].append(i)
        elem_to_groups[c].append(i)
    
    # Sort groups by the smallest element for consistent processing
    # Also sort the elements within each group
    groups = [tuple(sorted(group)) for group in groups]
    
    selected = [False] * t  # which groups are selected
    used = [False] * (3*q + 1)  # which elements are already covered
    solution = []
    
    def backtrack(pos):
        if pos == 3*q + 1:
            return len(solution) == q
        
        if used[pos]:
            return backtrack(pos + 1)
            
        # Try all groups containing 'pos'
        for i in elem_to_groups[pos]:
            a, b, c = groups[i]
            if not selected[i] and not used[a] and not used[b] and not used[c]:
                selected[i] = True
                used[a] = used[b] = used[c] = True
                solution.append((a, b, c))
                
                if backtrack(pos + 1):
                    return True
                    
                selected[i] = False
                used[a] = used[b] = used[c] = False
                solution.pop()
        
        return False
    
    # Optimization: start with the element that appears in fewest groups
    # Find element with minimum non-zero count
    min_count = float('inf')
    start_elem = 1
    for i in range(1, 3*q + 1):
        count = len(elem_to_groups[i])
        if count > 0 and count < min_count:
            min_count = count
            start_elem = i
        elif count == 0 and i <= 3*q:
            # If any element doesn't appear in any group, impossible
            print("NO")
            return
            
    if backtrack(1):
        print("YES")
        for group in solution:
            print(group[0], group[1], group[2])
    else:
        print("NO")

if __name__ == "__main__":
    main()
