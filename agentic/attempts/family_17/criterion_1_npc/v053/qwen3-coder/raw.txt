from collections import defaultdict
import sys

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    configs = []
    for i in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        configs.append((a, b, c))
        index += 3
    
    # Build a graph where each module is a node and each configuration is an edge
    # This is actually a matching problem: we want to find a perfect matching in a 3-uniform hypergraph
    # But since each configuration has exactly 3 nodes, we can model this as finding a perfect 3-matching
    
    # We'll use backtracking with pruning
    # Each configuration covers exactly 3 modules, and we need to cover all 3*q modules exactly once
    
    # Let's try to model this as an exact cover problem and use backtracking
    
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    # Create a list of which configurations contain each module
    module_to_configs = defaultdict(list)
    for i in range(t):
        a, b, c = configs[i]
        module_to_configs[a].append(i)
        module_to_configs[b].append(i)
        module_to_configs[c].append(i)
    
    # Try to build a solution using backtracking
    selected = []
    used_configs = [False] * t
    used_modules = [False] * (3*q + 1)
    
    def backtrack(module):
        # Find the first unused module
        while module <= 3*q and used_modules[module]:
            module += 1
            
        if module > 3*q:
            # All modules are used
            return len(selected) == q
        
        # Try all configurations that contain this module
        for config_idx in module_to_configs[module]:
            if used_configs[config_idx]:
                continue
                
            a, b, c = configs[config_idx]
            
            # Check if all modules in this configuration are available
            if used_modules[a] or used_modules[b] or used_modules[c]:
                continue
            
            # Use this configuration
            selected.append(config_idx)
            used_configs[config_idx] = True
            used_modules[a] = True
            used_modules[b] = True
            used_modules[c] = True
            
            # Recurse
            if backtrack(module + 1):
                return True
                
            # Backtrack
            selected.pop()
            used_configs[config_idx] = False
            used_modules[a] = False
            used_modules[b] = False
            used_modules[c] = False
        
        return False
    
    # Start backtracking from module 1
    if backtrack(1):
        print("YES")
        for config_idx in selected:
            a, b, c = configs[config_idx]
            print(a, b, c)
    else:
        print("NO")

solve()
