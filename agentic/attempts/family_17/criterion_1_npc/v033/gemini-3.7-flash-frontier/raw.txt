```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(3000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n = 3 * q
    triples = []
    elem_triples = [[] for _ in range(n)]
    
    idx = 2
    for tid in range(t):
        a = int(input_data[idx]) - 1
        b = int(input_data[idx+1]) - 1
        c = int(input_data[idx+2]) - 1
        idx += 3
        triples.append((a, b, c))
        elem_triples[a].append(tid)
        elem_triples[b].append(tid)
        elem_triples[c].append(tid)
        
    triple_conflict = [0] * t
    elem_covered = [False] * n
    active_elements = list(range(n))
    
    solution = []
    
    def search(num_uncovered):
        if num_uncovered == 0:
            return True
        
        # Unit propagation / MRV finding
        best_elem = -1
        min_candidates = 999999
        forced_tid = -1
        
        # We find the element with minimum available triples
        # Also check if any element has 0 candidates (immediate fail)
        # or 1 candidate (can be forced)
        for e in active_elements:
            if elem_covered[e]:
                continue
            cand_count = 0
            last_tid = -1
            for tid in elem_triples[e]:
                if triple_conflict[tid] == 0:
                    cand_count += 1
                    last_tid = tid
            
            if cand_count == 0:
                return False
            if cand_count < min_candidates:
                min_candidates = cand_count
                best_elem = e
                if cand_count == 1:
                    forced_tid = last_tid
                    break  # can't get fewer than 1 without being 0
        
        # If forced by unit propagation or just branching
        if min_candidates == 1:
            candidate_triples = [forced_tid]
        else:
            candidate_triples = [tid for tid in elem_triples[best_elem] if triple_conflict[tid] == 0]
        
        for tid in candidate_triples:
            u, v, w = triples[tid]
            if elem_covered[u] or elem_covered[v] or elem_covered[w]:
                continue
            
            # Apply triple
            elem_covered[u] = True
            elem_covered[v] = True
            elem_covered[w] = True
            
            for tr in elem_triples[u]:
                triple_conflict[tr] += 1
            for tr in elem_triples[v]:
                triple_conflict[tr] += 1
            for tr in elem_triples[w]:
                triple_conflict[tr] += 1
                
            solution.append(tid)
            
            if search(num_uncovered - 3):
                return True
                
            # Rollback
            solution.pop()
            for tr in elem_triples[u]:
                triple_conflict[tr] -= 1
            for tr in elem_triples[v]:
                triple_conflict[tr] -= 1
            for tr in elem_triples[w]:
                triple_conflict[tr] -= 1
                
            elem_covered[u] = False
            elem_covered[v] = False
            elem_covered[w] = False
            
        return False

    # Pre-check: if any element has no triples at all, impossible
    for i in range(n):
        if not elem_triples[i]:
            print("NO")
            return

    if search(n):
        print("YES")
        out = []
        for tid in solution:
            u, v, w = triples[tid]
            out.append(f"{u+1} {v+1} {w+1}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
