import sys
from collections import defaultdict

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    triples = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        triples.append((a, b, c))
    
    n = 3 * q
    
    # Build adjacency: for each pair, which triples contain it
    pair_to_triples = defaultdict(list)
    for idx, (a, b, c) in enumerate(triples):
        pairs = [(a, b), (a, c), (b, c)]
        for p in pairs:
            pair_to_triples[p].append(idx)
    
    # For each triple, precompute its pairs
    triple_pairs = []
    for a, b, c in triples:
        triple_pairs.append([(a, b), (a, c), (b, c)])
    
    # DP: dp[mask] = (prev_mask, triple_idx) or None
    # mask is bitmask of used vertices (0-indexed)
    # We use dictionary for sparse DP
    dp = {0: (-1, -1)}
    
    # Process masks in increasing order of set bits
    masks_by_bits = defaultdict(list)
    for mask in dp:
        masks_by_bits[bin(mask).count('1')].append(mask)
    
    for bits in range(0, n, 3):
        if bits not in masks_by_bits:
            break
        for mask in masks_by_bits[bits]:
            if mask not in dp:
                continue
            # Find first unused vertex
            unused = []
            for v in range(n):
                if not (mask & (1 << v)):
                    unused.append(v)
                    if len(unused) == 2:
                        break
            
            if len(unused) < 2:
                continue
            
            v1, v2 = unused[0], unused[1]
            pair = (v1 + 1, v2 + 1)  # Convert back to 1-indexed
            
            # Find triples containing this pair
            for tidx in pair_to_triples.get(pair, []):
                a, b, c = triples[tidx]
                # Convert to 0-indexed
                va, vb, vc = a - 1, b - 1, c - 1
                # Check if all three are unused
                if (mask >> va) & 1:
                    continue
                if (mask >> vb) & 1:
                    continue
                if (mask >> vc) & 1:
                    continue
                
                new_mask = mask | (1 << va) | (1 << vb) | (1 << vc)
                if new_mask not in dp:
                    dp[new_mask] = (mask, tidx)
                    new_bits = bits + 3
                    masks_by_bits[new_bits].append(new_mask)
    
    full_mask = (1 << n) - 1
    if full_mask not in dp:
        print("NO")
        return
    
    print("YES")
    # Reconstruct solution
    result = []
    mask = full_mask
    while mask != 0:
        prev_mask, tidx = dp[mask]
        result.append(triples[tidx])
        mask = prev_mask
    
    # Output in reverse order (or any order is fine)
    for triple in reversed(result):
        print(f"{triple[0]} {triple[1]} {triple[2]}")

if __name__ == "__main__":
    solve()
