```python
import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    n = 3 * q
    rows = []
    col_to_rows = [[] for _ in range(n+1)]
    for _ in range(t):
        a = int(next(it)); b = int(next(it)); c = int(next(it))
        rows.append((a,b,c))
        idx = len(rows)-1
        col_to_rows[a].append(idx)
        col_to_rows[b].append(idx)
        col_to_rows[c].append(idx)

    # active columns set
    active_cols = set(range(1, n+1))
    # for each column, current list of rows still possible
    cur_col_rows = [set(col_to_rows[i]) for i in range(n+1)]

    solution = []

    # heuristic: choose column with fewest options
    def choose_column():
        min_c = None
        min_len = 10**9
        for c in active_cols:
            l = len(cur_col_rows[c])
            if l < min_len:
                min_len = l
                min_c = c
                if l == 0:
                    break
        return min_c, min_len

    def cover(row_idx):
        a,b,c = rows[row_idx]
        removed = []
        for col in (a,b,c):
            if col in active_cols:
                active_cols.remove(col)
                # for each row that uses this col, remove its other cols
                for r in list(cur_col_rows[col]):
                    if r == row_idx: continue
                    ra,rb,rc = rows[r]
                    for cc in (ra,rb,rc):
                        if cc != col and cc in active_cols:
                            cur_col_rows[cc].discard(r)
                removed.append(col)
        return removed

    def uncover(removed, row_idx):
        a,b,c = rows[row_idx]
        for col in reversed(removed):
            # restore column
            active_cols.add(col)
            # restore rows
            for r in col_to_rows[col]:
                if r == row_idx: continue
                ra,rb,rc = rows[r]
                for cc in (ra,rb,rc):
                    if cc != col and cc in active_cols:
                        cur_col_rows[cc].add(r)

    def dfs():
        if not active_cols:
            return True
        col, cnt = choose_column()
        if cnt == 0:
            return False
        # try each row covering this column
        for row_idx in list(cur_col_rows[col]):
            # select row
            solution.append(row_idx)
            removed = cover(row_idx)
            if dfs():
                return True
            # backtrack
            solution.pop()
            uncover(removed, row_idx)
        return False

    if dfs():
        out = ["YES"]
        for idx in solution:
            out.append(f"{rows[idx][0]} {rows[idx][1]} {rows[idx][2]}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
