import sys

# Increase recursion depth just in case
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    q = int(input_data[0])
    t = int(input_data[1])
    
    n_items = 3 * q
    
    # Read triples
    triples = []
    idx = 2
    for _ in range(t):
        a = int(input_data[idx])
        b = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        triples.append((a, b, c))
    
    # If t < q, impossible
    if t < q:
        print("NO")
        return

    # Check if all elements appear at least once
    item_to_triples = [[] for _ in range(n_items + 1)]
    for i, (a, b, c) in enumerate(triples):
        item_to_triples[a].append(i)
        item_to_triples[b].append(i)
        item_to_triples[c].append(i)
        
    for i in range(1, n_items + 1):
        if len(item_to_triples[i]) == 0:
            print("NO")
            return

    # DLX Implementation
    # Columns: 1 to n_items
    # Root: 0
    # Column headers: 1 to n_items
    
    L = [0] * (n_items + 1)
    R = [0] * (n_items + 1)
    for i in range(n_items + 1):
        L[i] = i - 1
        R[i] = i + 1
    L[0] = n_items
    R[n_items] = 0
    
    # We will build nodes for each element in each triple
    # Max nodes: 3 * t + n_items + 1
    U = [0] * (3 * t + n_items + 1)
    D = [0] * (3 * t + n_items + 1)
    col = [0] * (3 * t + n_items + 1)
    row = [0] * (3 * t + n_items + 1)
    sz = [0] * (n_items + 1)
    
    for i in range(1, n_items + 1):
        U[i] = i
        D[i] = i
        col[i] = i
        
    node_cnt = n_items
    
    for r, (a, b, c) in enumerate(triples):
        row_nodes = []
        for c_idx in (a, b, c):
            node_cnt += 1
            nd = node_cnt
            col[nd] = c_idx
            row[nd] = r
            sz[c_idx] += 1
            
            # Insert into column list
            U[nd] = U[c_idx]
            D[nd] = c_idx
            D[U[c_idx]] = nd
            U[c_idx] = nd
            
            row_nodes.append(nd)
            
        # Connect row nodes circularly via L and R
        # Note: L and R need to be sized appropriately
        # Let's extend L and R arrays
    
    # Let's re-allocate L, R properly
    total_nodes = node_cnt + 1
    L_full = [0] * total_nodes
    R_full = [0] * total_nodes
    for i in range(n_items + 1):
        L_full[i] = L[i]
        R_full[i] = R[i]
    L = L_full
    R = R_full
    
    curr = n_items + 1
    for _ in range(t):
        n1, n2, n3 = curr, curr + 1, curr + 2
        L[n1], R[n1] = n3, n2
        L[n2], R[n2] = n1, n3
        L[n3], R[n3] = n2, n1
        curr += 3

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size (MRV)
        c = R[0]
        min_sz = sz[c]
        j = R[0]
        while j != 0:
            if sz[j] < min_sz:
                min_sz = sz[j]
                c = j
                if min_sz == 0:
                    break
            j = R[j]
            
        if min_sz == 0:
            return False
            
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]
                
            if search():
                return True
                
            solution.pop()
            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            r = D[r]
            
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r_idx in solution:
            a, b, c = triples[r_idx]
            out.append(f"{a} {b} {c}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
