from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    flights = []
    for i in range(t):
        a = int(data[index])
        b = int(data[index+1])
        c = int(data[index+2])
        flights.append((a, b, c))
        index += 3
    
    # Total sum required
    total_zones = 3 * q
    required_sum = total_zones * (total_zones + 1) // 2
    
    # We need to select exactly q flights such that:
    # 1. Each zone from 1 to 3q appears exactly once
    # 2. The sum of all zones in selected flights equals required_sum
    
    # This is essentially a matching problem where we need to find a perfect matching
    # that covers all zones exactly once and satisfies the sum constraint.
    
    # Since q <= 200 and t <= 20000, we can try backtracking or other approaches,
    # but given constraints, let's model this as an exact cover problem with additional constraint.
    
    # Let's use backtracking with pruning.
    
    # Precompute sums of each flight
    flight_sums = [sum(flight) for flight in flights]
    
    # Zones covered by each flight
    flight_zones = [set(flight) for flight in flights]
    
    # Try to build a solution using backtracking
    selected = []
    used_zones = [False] * (total_zones + 1)
    current_sum = 0
    
    def backtrack(idx):
        if len(selected) == q:
            if current_sum == required_sum:
                return True
            else:
                return False
        
        # Pruning: if even selecting the smallest possible remaining sums won't reach required_sum
        # or selecting the largest possible exceeds it, prune.
        
        # But since it's complex to compute, let's just proceed with basic backtracking
        # and hope the constraints allow it.
        
        for i in range(idx, t):
            # Check if this flight can be added
            a, b, c = flights[i]
            if used_zones[a] or used_zones[b] or used_zones[c]:
                continue
            
            # Add this flight
            selected.append(i)
            used_zones[a] = used_zones[b] = used_zones[c] = True
            nonlocal current_sum
            current_sum += flight_sums[i]
            
            if backtrack(i+1):
                return True
            
            # Backtrack
            selected.pop()
            used_zones[a] = used_zones[b] = used_zones[c] = False
            current_sum -= flight_sums[i]
        
        return False
    
    # Optimization: sort flights by sum to try smaller sums first
    # Enumerate flights with their indices and sort by sum
    indexed_flights = list(enumerate(flights))
    indexed_flights.sort(key=lambda x: sum(x[1]))
    sorted_indices = [x[0] for x in indexed_flights]
    # Reorder flights and flight_sums accordingly
    flights_sorted = [flights[i] for i in sorted_indices]
    flight_sums_sorted = [flight_sums[i] for i in sorted_indices]
    flight_zones_sorted = [flight_zones[i] for i in sorted_indices]
    
    # Update references
    flights, flight_sums, flight_zones = flights_sorted, flight_sums_sorted, flight_zones_sorted
    
    # Now perform backtracking
    selected = []
    used_zones = [False] * (total_zones + 1)
    current_sum = 0
    
    def backtrack_opt(idx):
        if len(selected) == q:
            if current_sum == required_sum:
                return True
            else:
                return False
                
        # Pruning based on remaining sum needed
        remaining_flights_needed = q - len(selected)
        if remaining_flights_needed < 0:
            return False
            
        # Estimate minimum and maximum sum we can get from remaining flights
        # This is approximate; for better pruning we could precompute prefix/suffix sums
        
        for i in range(idx, t):
            # Check if this flight can be added
            a, b, c = flights[i]
            if used_zones[a] or used_zones[b] or used_zones[c]:
                continue
            
            # Add this flight
            selected.append(i)
            used_zones[a] = used_zones[b] = used_zones[c] = True
            nonlocal current_sum
            current_sum += flight_sums[i]
            
            if backtrack_opt(i+1):
                return True
            
            # Backtrack
            selected.pop()
            used_zones[a] = used_zones[b] = used_zones[c] = False
            current_sum -= flight_sums[i]
        
        return False
    
    if backtrack_opt(0):
        print("YES")
        # Map back to original indices if we sorted
        for idx in selected:
            a, b, c = flights[idx]
            print(a, b, c)
    else:
        print("NO")

if __name__ == "__main__":
    main()
