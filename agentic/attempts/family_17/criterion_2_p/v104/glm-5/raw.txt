import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    # collectors: 1..q, processors: q+1..2q
    # source = 0, sink = 2q+1
    N = 2 * q + 2
    S = 0
    T = 2 * q + 1
    
    head = [-1] * N
    to = []
    cap = []
    nxt = []
    
    def add_edge(u, v, c):
        head.append(0)  # placeholder
        # forward
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = len(to) - 1
        # backward
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = len(to) - 1
    
    # source to collectors
    for a in range(1, q + 1):
        add_edge(S, a, 1)
    
    # processors to sink
    for b in range(q + 1, 2 * q + 1):
        add_edge(b, T, 1)
    
    # edges from input
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        add_edge(a, b, 1)
    
    # Dinic
    level = [-1] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        level[S] = 0
        dq = deque([S])
        while dq:
            u = dq.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    dq.append(v)
                e = nxt[e]
        return level[T] >= 0
    
    it_ptr = [0] * N
    
    def dfs(u, f):
        if u == T:
            return f
        e = it_ptr[u]
        while e != -1:
            v = to[e]
            if cap[e] > 0 and level[v] == level[u] + 1:
                ret = dfs(v, min(f, cap[e]))
                if ret > 0:
                    cap[e] -= ret
                    cap[e ^ 1] += ret
                    it_ptr[u] = e
                    return ret
            e = nxt[e]
        it_ptr[u] = e
        return 0
    
    flow = 0
    while bfs():
        for i in range(N):
            it_ptr[i] = head[i]
        while True:
            f = dfs(S, 10**9)
            if f == 0:
                break
            flow += f
    
    if flow < q:
        sys.stdout.write("NO\n")
        return
    
    out = ["YES"]
    # find chosen edges: collector -> processor with cap 0 (used)
    # edges from input are after source->collector and processor->sink edges
    # source->collector edges: indices 0,2,4,... (2 per collector)
    # processor->sink edges: next 2*q edges
    # input edges start at index 4*q
    start = 4 * q
    i = start
    cnt = 0
    while i < len(to) and cnt < q:
        if cap[i] == 0:
            # forward edge used
            u = to[i ^ 1]  # collector
            v = to[i]      # processor
            out.append(f"{u} {v}")
            cnt += 1
            i += 2  # skip its reverse
        else:
            i += 2
    
    sys.stdout.write("\n".join(out) + "\n")

solve()
