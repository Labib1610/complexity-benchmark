We need to find a perfect matching between controllers (1..q) and platforms (q+1..2q) using the given edges.  
This is a bipartite matching problem, but with up to 400,000 edges and 100,000 nodes per side, we need an efficient algorithm.

We can use **Hopcroft–Karp** for maximum bipartite matching, which runs in O(E√V).  
If the matching size equals q, we output the matched pairs.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q)]
edges = []
for _ in range(t):
    a, b = map(int, input().split())
    a -= 1
    b -= q + 1  # platform index 0..q-1
    adj[a].append(b)
    edges.append((a, b))

# Hopcroft-Karp
pair_u = [-1] * q
pair_v = [-1] * q
dist = [0] * q

def bfs():
    dq = deque()
    for u in range(q):
        if pair_u[u] == -1:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = -1
    found = False
    while dq:
        u = dq.popleft()
        for v in adj[u]:
            pu = pair_v[v]
            if pu != -1 and dist[pu] == -1:
                dist[pu] = dist[u] + 1
                dq.append(pu)
            elif pu == -1:
                found = True
    return found

def dfs(u):
    for v in adj[u]:
        pu = pair_v[v]
        if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
            pair_u[u] = v
            pair_v[v] = u
            return True
    dist[u] = -1
    return False

matching = 0
while bfs():
    for u in range(q):
        if pair_u[u] == -1 and dfs(u):
            matching += 1

if matching == q:
    print("YES")
    for u in range(q):
        print(u + 1, pair_u[u] + q + 1)
else:
    print("NO")
```
