import sys

# Increase recursion depth for DFS in large graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    # Number of nodes in the flow graph:
    # 0: Source
    # 1 to q: Controllers
    # q+1 to 2q: Platforms
    # 2q+1: Sink
    source = 0
    sink = 2 * q + 1
    num_nodes = 2 * q + 2

    # Adjacency list for Dinic's algorithm
    # Each entry stores [to, capacity, rev_index]
    adj = [[] for _ in range(num_nodes)]

    def add_edge(u, v, cap):
        adj[u].append([v, cap, len(adj[v])])
        adj[v].append([u, 0, len(adj[u]) - 1])

    # Build edges from source to controllers (IDs 1 to q)
    for i in range(1, q + 1):
        add_edge(source, i, 1)

    # Build edges from platforms (IDs q+1 to 2q) to sink
    for i in range(q + 1, 2 * q + 1):
        add_edge(i, sink, 1)

    # Build edges for certified combinations
    for _ in range(t):
        try:
            a = int(next(iterator))
            b = int(next(iterator))
        except StopIteration:
            break
        # a is controller (1..q), b is platform (q+1..2q)
        add_edge(a, b, 1)

    # Dinic's Algorithm Implementation
    level = [0] * num_nodes
    ptr = [0] * num_nodes
    q_deque = [0] * num_nodes

    def bfs():
        for i in range(num_nodes):
            level[i] = -1
        level[source] = 0
        q_deque[0] = source
        head, tail = 0, 1
        while head < tail:
            u = q_deque[head]
            head += 1
            for v, cap, rev in adj[u]:
                if cap > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q_deque[tail] = v
                    tail += 1
        return level[sink] != -1

    def dfs(u, pushed):
        if u == sink:
            return pushed
        for i in range(ptr[u], len(adj[u])):
            ptr[u] = i
            v, cap, rev = adj[u][i]
            if cap > 0 and level[v] == level[u] + 1:
                tr = dfs(v, min(pushed, cap))
                if tr > 0:
                    adj[u][i][1] -= tr
                    adj[v][rev][1] += tr
                    return tr
        return 0

    max_flow = 0
    while bfs():
        ptr = [0] * num_nodes
        while True:
            pushed = dfs(source, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed

    if max_flow == q:
        sys.stdout.write("YES\n")
        # Extract matching
        # Controller IDs are 1 to q
        for u in range(1, q + 1):
            for v, cap, rev in adj[u]:
                # If capacity is 0 and v is a platform (v > q), it's a used edge
                if v > q and v <= 2 * q and cap == 0:
                    sys.stdout.write(f"{u} {v}\n")
                    break
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
