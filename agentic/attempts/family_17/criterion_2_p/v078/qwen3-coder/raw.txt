```python
from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Controllers: 1 to q
    # Platforms: q+1 to 2*q
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm
    # But we'll use a simpler approach with DFS for this problem
    
    # partner[i] represents which platform is matched to controller i
    # partner[j] represents which controller is matched to platform j
    partner = {}
    
    def dfs(u, visited):
        for v in adj[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If v is unmatched or we can find augmenting path
            if v not in partner or dfs(partner[v], visited):
                partner[u] = v
                partner[v] = u
                return True
        return False
    
    # Try to match all controllers
    matched = 0
    for u in range(1, q + 1):
        if dfs(u, set()):
            matched += 1
    
    if matched != q:
        print("NO")
        return
    
    print("YES")
    for u in range(1, q + 1):
        print(u, partner[u])

solve()
```
