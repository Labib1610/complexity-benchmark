We need to solve the bipartite perfect matching problem.  
- Nodes: plugs numbered `1..q` and sockets numbered `q+1..2q`.  
- Edge if pair is in the compatible list.  

We can solve this with **Hopcroft–Karp algorithm** in **O(E√V)** time — sufficient for `q ≤ 1e5`, `t ≤ 4e5`.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q)]
for _ in range(t):
    a, b = map(int, input().split())
    adj[a - 1].append(b - q - 1)  # zero-index sockets

# Hopcroft-Karp
INF = 10**9
pair_u = [-1] * q
pair_v = [-1] * q
dist = [0] * q

def bfs():
    dq = deque()
    for u in range(q):
        if pair_u[u] == -1:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = INF
    found = False
    while dq:
        u = dq.popleft()
        for v in adj[u]:
            if pair_v[v] == -1:
                found = True
            else:
                if dist[pair_v[v]] == INF:
                    dist[pair_v[v]] = dist[u] + 1
                    dq.append(pair_v[v])
    return found

def dfs(u):
    for v in adj[u]:
        if pair_v[v] == -1 or (dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v])):
            pair_u[u] = v
            pair_v[v] = u
            return True
    dist[u] = INF
    return False

matching = 0
while bfs():
    for u in range(q):
        if pair_u[u] == -1 and dfs(u):
            matching += 1

if matching == q:
    print("YES")
    for u in range(q):
        print(u + 1, pair_u[u] + q + 1)
else:
    print("NO")
```
