```python
import sys
from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Plugs: 1 to q
    # Sockets: q+1 to 2*q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        graph[b].append(a)
        edges.append((a, b))
    
    # Use Hopcroft-Karp algorithm implementation for maximum matching
    # But let's use a simpler approach with Ford-Fulkerson style DFS
    
    def dfs(u, match_v, match_u, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            # If v is unmatched or we can find augmenting path
            if match_v[v] == -1 or dfs(match_v[v], match_v, match_u, visited):
                match_v[v] = u
                match_u[u] = v
                return True
        return False
    
    # Initialize matching arrays
    # match_u[u] = matched vertex v for plug u, or -1
    # match_v[v] = matched vertex u for socket v, or -1
    match_u = [-1] * (q + 1)  # Index 0 unused
    match_v = [-1] * (2 * q + 1)  # Index 0 unused
    
    matching = 0
    # Try to match each plug
    for u in range(1, q + 1):
        visited = [False] * (2 * q + 1)
        if dfs(u, match_v, match_u, visited):
            matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_u[u])
    else:
        print("NO")

solve()
```
