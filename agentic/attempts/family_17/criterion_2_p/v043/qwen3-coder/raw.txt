from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Left side: 1 to q, Right side: q+1 to 2q
    graph = defaultdict(list)
    degree = [0] * (2 * q + 1)
    
    edges = []
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        graph[b].append(a)
        degree[a] += 1
        degree[b] += 1
        edges.append((a, b))
    
    # Check if every vertex has at least one edge
    for i in range(1, 2*q + 1):
        if degree[i] == 0:
            print("NO")
            return
    
    # Find maximum matching using Hopcroft-Karp or simpler approach
    # Since this is bipartite matching, we can use a standard algorithm
    
    # Try to find a perfect matching
    match_left = [-1] * (q + 1)  # match_left[i] = j means left node i matches right node j
    match_right = [-1] * (q + 1)  # match_right[j] = i means right node j+q matches left node i
    
    def bfs():
        dist = [-1] * (q + 1)
        queue = deque()
        
        for u in range(1, q + 1):
            if match_left[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = -1
        
        dist[0] = -1  # 0 is the virtual node
        while queue:
            u = queue.popleft()
            if dist[u] != -1:
                for v in graph[u]:
                    v_idx = v - q  # Convert to 1-based index in right part
                    if dist[match_right[v_idx]] == -1:
                        dist[match_right[v_idx]] = dist[u] + 1
                        queue.append(match_right[v_idx])
        return dist[0] != -1
    
    def dfs(u):
        if u != 0:
            for v in graph[u]:
                v_idx = v - q  # Convert to 1-based index in right part
                if dist[match_right[v_idx]] == dist[u] + 1:
                    if dfs(match_right[v_idx]):
                        match_right[v_idx] = u
                        match_left[u] = v
                        return True
            dist[u] = -1
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if match_left[u] == -1:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_left[u])
    else:
        print("NO")

def solve_simple():
    q, t = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b)
    
    # Try to find a matching using a simple greedy approach with backtracking
    # This is essentially finding a maximum bipartite matching
    
    match = [-1] * (2*q + 1)  # match[i] = j means i is matched to j
    matched_right = [False] * (q + 1)  # track which right nodes are matched
    
    def dfs(u, visited):
        for v in adj[u]:
            v_idx = v - q  # Convert to 1-based index for right side
            if visited[v_idx]:
                continue
            visited[v_idx] = True
            
            # If v is unmatched or we can find augmenting path
            if match[v] == -1 or dfs(match[v], visited):
                match[u] = v
                match[v] = u
                return True
        return False
    
    matching = 0
    for u in range(1, q + 1):
        visited = [False] * (q + 1)
        if dfs(u, visited):
            matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match[u])
    else:
        print("NO")

solve_simple()
