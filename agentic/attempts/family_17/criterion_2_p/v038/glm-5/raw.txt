import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    adj = [[] for _ in range(q + 1)]
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
    
    # We need to find a perfect matching in a bipartite graph
    # Left side: nodes 1 to q
    # Right side: nodes q+1 to 2q
    
    pair_U = [0] * (q + 1)
    pair_V = [0] * (2 * q + 1)
    dist = [0] * (q + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = -1
        dist[0] = -1
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_V[v] == 0:
                        dist[0] = dist[u] + 1
                    elif dist[pair_V[v]] == -1:
                        dist[pair_V[v]] = dist[u] + 1
                        queue.append(pair_V[v])
        return dist[0] != -1
    
    def dfs(u):
        for v in adj[u]:
            if pair_V[v] == 0 or (dist[pair_V[v]] == dist[u] + 1 and dfs(pair_V[v])):
                pair_U[u] = v
                pair_V[v] = u
                return True
        dist[u] = -1
        return False
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, pair_U[u])
    else:
        print("NO")

if __name__ == "__main__":
    solve()
