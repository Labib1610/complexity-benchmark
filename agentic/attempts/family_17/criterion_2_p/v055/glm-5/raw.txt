def hopcroft_karp(n, m, edges):
    # n = number of left vertices (modules), m = right (ports)
    # adjacency: adj[u] = list of v
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
    
    # match_left[u] = matched v or 0 (1-indexed right vertices)
    # match_right[v] = matched u or 0
    match_left = [0] * (n + 1)
    match_right = [0] * (m + 1)

        queue = deque()

            if match_left[u] == 0:
                dist[u] = 0
                queue.append(u)
        found = False
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                w = match_right[v]
                if w == 0:
                    found = True
                elif dist[w] == INF:
                    dist[w] = dist[u] + 1
                    queue.append(w)

                    match_right[v] = u

            dist[u] = INF

    for a in range(1, q + 1):
        for p in adj[a]:
            if match_r[p] == 0:
                match_r[p] = a
                match_l[a] = p
                break
    
    INF = 1 << 30
    dist = [INF] * (q + 1)
    ptr = [0] * (q + 1)
    
    matched = sum(1 for a in range(1, q+1) if match_l[a])

        queue = []
