import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(q + 1)]
    # Store original index of each edge
    for i in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append((v, i))

    # We need to find a perfect matching
    # Use Hopcroft-Karp algorithm
    
    # Nodes on left side: 1 to q
    # Nodes on right side: q+1 to 2q, remapped to 1 to q
    
    # pair_U[u] stores the right-node matched with left-node u
    pair_U = [-1] * (q + 1)
    # pair_V[v] stores the left-node matched with right-node v (remapped)
    pair_V = [-1] * (q + 1)
    dist = [0] * (q + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_U[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = -1
        
        found_augmenting_path = False
        
        while queue:
            u = queue.popleft()
            for v_remapped, edge_idx in adj[u]:
                # If v is matched, traverse to its pair
                # pair_V[v_remapped] gives the u' matched to v
                # We want to see if we can reach an unmatched node
                # via alternating paths
                
                # In BFS, we are building layers.
                # From free U, we go to V.
                # From V, if matched, we go to matched U.
                # We stop when we reach a free V.
                
                # Here, we are at u.
                # We try to go to v_remapped.
                # If v_remapped is not matched, we found an augmenting path.
                # If v_remapped is matched, we must go through its match u' = pair_V[v_remapped].
                
                u_prime = pair_V[v_remapped]
                if u_prime != -1:
                    if dist[u_prime] == -1:
                        dist[u_prime] = dist[u] + 1
                        queue.append(u_prime)
                else:
                    # Reached a free node on the right
                    found_augmenting_path = True
        return found_augmenting_path

    def dfs(u):
        for v_remapped, edge_idx in adj[u]:
            u_prime = pair_V[v_remapped]
            if u_prime == -1:
                # Found augmenting path ending in free v
                pair_U[u] = v_remapped
                pair_V[v_remapped] = u
                return True
            elif dist[u_prime] == dist[u] + 1:
                if dfs(u_prime):
                    pair_U[u] = v_remapped
                    pair_V[v_remapped] = u
                    return True
        dist[u] = -1 # Mark as visited in this DFS phase to avoid reprocessing
        return False

    matching_count = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_U[u] == -1:
                if dfs(u):
                    matching_count += 1
    
    if matching_count == q:
        print("YES")
        # Reconstruct edges
        # We need to output the original edge indices
        # We have pair_U[u] = v_remapped.
        # We need to find which edge (u, v) was selected.
        # Since adj[u] contains (v, index), we can iterate to find the edge.
        # To be safe and efficient, we can iterate adj[u] again.
        
        # Optimization: build a map for selected edges
        # Actually, just iterate adj[u] for each u in pair_U
        # pair_U[u] gives the matched v_remapped.
        
        for u in range(1, q + 1):
            v_remapped = pair_U[u]
            # Find the edge in adj[u] that connects to v_remapped
            # adj[u] contains (v_remapped, edge_index)
            # Note: v_remapped is already the remapped index (1..q)
            # corresponding to original node q + v_remapped.
            
            found = False
            for vr, idx in adj[u]:
                if vr == v_remapped:
                    # Output original nodes
                    # u is original, v is q + vr
                    print(f"{u} {q + vr}")
                    found = True
                    break
            if not found:
                # Should not happen if matching is valid
                pass
    else:
        print("NO")

if __name__ == '__main__':
    solve()
