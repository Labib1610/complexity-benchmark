    match_right = [-1] * n_right

                dist[i] = 0
                queue.append(i)
            else:
                dist[i] = INF

            for v in adj[u]:

                if w == -1:
                    found = True

                    dist[w] = dist[u] + 1

                match_left[u] = v
                match_right[v] = u
                return True
        dist[u] = INF
        return False

def scipy_solve(q, t, data):

    from scipy.sparse import csr_matrix

def pure_solve(q, t, vals):
    n = q
    adj = [[] for _ in range(n + 1)]
    for i in range(0, 2*t, 2):
        adj[vals[i]].append(vals[i+1] - q)
    
    matchL = [0] * (n + 1)
    matchR = [0] * (n + 1)
    INF = n + 10
    dist = [0] * (n + 1)
    
    # Greedy init
    for u in range(1, n+1):
        au = adj[u]
        for v in au:
            if matchR[v] == 0:
                matchR[v] = u
                matchL[u] = v
                break
    
    # main HK loop
    while True:
        # free left vertices
        free = [u for u in range(1, n+1) if matchL[u] == 0]
        if not free:
            break  # perfect
        # BFS
        for u in range(1, n+1):
            dist[u] = INF
        queue = free[:]  # dist 0
        for u in free:
            dist[u] = 0
        found = False
        qi = 0
        while qi < len(queue):
            u = queue[qi]; qi += 1
            du1 = dist[u] + 1
            for v in adj[u]:
                w = matchR[v]
                if w == 0:
                    found = True
                elif dist[w] == INF:
                    dist[w] = du1
                    queue.append(w)
        if not found:
            return "NO"
        # DFS phase
        for root in free:
            # iterative DFS
            stack = [(root, iter(adj[root]))]
            pv = [0]
            ok = False
            while stack:
                u = stack[-1][0]
                du1 = dist[u] + 1
                advanced = False
                for v in stack[-1][1]:
                    w = matchR[v]
                    if w == 0:
                        pv[-1] = v
                        # augment along stack
                        # stack[i][0] matched to pv[i]
                        i = 0
                        for frame in stack:
                            uu = frame[0]
                            vv = pv[i]
                            matchL[uu] = vv
                            matchR[vv] = uu
                            i += 1
                        ok = True
                        break
                    if dist[w] == du1:
                        pv[-1] = v
                        stack.append((w, iter(adj[w])))
                        pv.append(0)
                        advanced = True
                        break
                if ok:
                    break
                if not advanced:
                    dist[u] = INF
                    stack.pop()
                    pv.pop()
        # loop back to BFS
    # build result
    return [(u, matchL[u] + q) for u in range(1, n+1)]
