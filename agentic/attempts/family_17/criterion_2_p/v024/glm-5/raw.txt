import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    
    # Build adjacency list for left side (legacy servers 1..q)
    # Right side (virtual machines q+1..2q) mapped to 0..q-1
    adj = [[] for _ in range(q + 1)]
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1])
        idx += 2
        adj[a].append(b - q)  # map b to 0-indexed
    
    # Hopcroft-Karp algorithm for maximum bipartite matching
    INF = q + 1
    pair_u = [0] * (q + 1)  # pair_u[u] = v (0 means unmatched)
    pair_v = [0] * (q + 1)  # pair_v[v] = u
    dist = [0] * (q + 1)
    
    from collections import deque
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        for v in adj[u]:
            if dist[pair_v[v]] == dist[u] + 1:
                if dfs(pair_v[v]):
                    pair_u[u] = v
                    pair_v[v] = u
                    return True
        dist[u] = INF
        return False
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        out = ["YES"]
        for u in range(1, q + 1):
            out.append(f"{u} {pair_u[u] + q}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

solve()
