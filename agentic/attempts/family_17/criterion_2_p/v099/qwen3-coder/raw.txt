from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build graph for bipartite matching
    # Controllers: 1 to q
    # Platforms: q+1 to 2q
    # We'll create a graph where edges go from controllers to platforms
    
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a maximum matching using BFS-based algorithm (Hopcroft-Karp like)
    # But we can use a simpler approach with DFS for this problem
    
    def dfs(u, match_controller, match_platform, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            # If platform v is unmatched or we can find augmenting path
            if match_platform[v] == -1 or dfs(match_platform[v], match_controller, match_platform, visited):
                match_controller[u] = v
                match_platform[v] = u
                return True
        return False
    
    # Initialize matching arrays
    # match_controller[i] stores which platform controller i is matched with
    # match_platform[j] stores which controller platform j is matched with
    match_controller = [-1] * (q + 1)  # 1-indexed
    match_platform = [-1] * (2 * q + 1)  # 1-indexed
    
    matching = 0
    for u in range(1, q + 1):
        visited = [False] * (2 * q + 1)
        if dfs(u, match_controller, match_platform, visited):
            matching += 1
    
    if matching != q:
        print("NO")
        return
    
    print("YES")
    for u in range(1, q + 1):
        print(u, match_controller[u])

solve()
