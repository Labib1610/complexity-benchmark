import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    q = int(next(iterator))
    t = int(next(iterator))

    adj = [[] for _ in range(q + 1)]
    degrees = [0] * (q + 1)
    
    # Store edges to output them later if needed
    edges = []
    
    for _ in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        edges.append((u, v))
        adj[u].append(v)
        degrees[u] += 1

    # We need to find a perfect matching in a bipartite graph
    # Controllers: 1..q, Platforms: q+1..2q
    # Using Hopcroft-Karp algorithm
    
    pair_U = [0] * (q + 1)
    pair_V = [0] * (2 * q + 1)
    dist = [0] * (q + 1)

    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_V[v] == 0:
                        if dist[0] == float('inf'):
                            dist[0] = dist[u] + 1
                    elif dist[pair_V[v]] == float('inf'):
                        dist[pair_V[v]] = dist[u] + 1
                        queue.append(pair_V[v])
        return dist[0] != float('inf')

    def dfs(u):
        for v in adj[u]:
            if pair_V[v] == 0 or (dist[pair_V[v]] == dist[u] + 1 and dfs(pair_V[v])):
                pair_U[u] = v
                pair_V[v] = u
                return True
        dist[u] = float('inf')
        return False

    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, pair_U[u])
    else:
        print("NO")

if __name__ == '__main__':
    solve()
