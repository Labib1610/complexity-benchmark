from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    if t < q:
        print("NO")
        return
    
    # Build bipartite graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        # Convert to 0-indexed
        u = a - 1  # injector: 0 to q-1
        v = b - q - 1  # extractor: 0 to q-1
        graph[u].append(v)
        edges.append((u, v))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm
    pair_u = [-1] * q  # pair_u[u] = v means u is matched to v
    pair_v = [-1] * q  # pair_v[v] = u means v is matched to u
    
    def bfs():
        dist = [-1] * q
        queue = deque()
        
        # Initialize queue with free vertices in U
        for u in range(q):
            if pair_u[u] == -1:
                dist[u] = 0
                queue.append(u)
        
        level = -1
        
        while queue:
            u = queue.popleft()
            
            if dist[u] > level:
                level = dist[u]
            
            for v in graph[u]:
                paired_u = pair_v[v]
                if paired_u == -1:
                    # Found an augmenting path
                    return True
                if dist[paired_u] == -1:
                    dist[paired_u] = dist[u] + 1
                    queue.append(paired_u)
        
        return False
    
    def dfs(u):
        for v in graph[u]:
            paired_u = pair_v[v]
            if paired_u == -1 or (dist[paired_u] == dist[u] + 1 and dfs(paired_u)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        return False
    
    # Hopcroft-Karp main loop
    matching = 0
    while bfs():
        dist = [-1] * q
        for u in range(q):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    
    if matching < q:
        print("NO")
        return
    
    print("YES")
    for u in range(q):
        v = pair_u[u]
        # Convert back to 1-indexed
        print(u + 1, v + q + 1)

solve()
