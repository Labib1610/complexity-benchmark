We need to match each command unit (1..q) to a distinct response unit (q+1..2q) using only allowed links. This is a bipartite matching problem where both sides have size q, and we need a perfect matching.

Since q can be up to 100,000 and t up to 400,000, we need an efficient algorithm. Hopcroft–Karp is suitable here.

We’ll implement Hopcroft–Karp for maximum bipartite matching, and if the matching size equals q, we output YES and the matched pairs.

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q)]
for _ in range(t):
    a, b = map(int, input().split())
    adj[a-1].append(b-1)  # response side index 0..q-1

# Hopcroft-Karp
pair_u = [-1] * q
pair_v = [-1] * q
dist = [0] * q

INF = 10**9

def bfs():
    dq = deque()
    for u in range(q):
        if pair_u[u] == -1:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = INF
    found = False
    while dq:
        u = dq.popleft()
        for v in adj[u]:
            pu = pair_v[v]
            if pu != -1 and dist[pu] == INF:
                dist[pu] = dist[u] + 1
                dq.append(pu)
            elif pu == -1:
                found = True
    return found

def dfs(u):
    for v in adj[u]:
        pu = pair_v[v]
        if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
            pair_u[u] = v
            pair_v[v] = u
            return True
    dist[u] = INF
    return False

matching = 0
while bfs():
    for u in range(q):
        if pair_u[u] == -1 and dfs(u):
            matching += 1

if matching == q:
    print("YES")
    for u in range(q):
        print(u+1, pair_u[u]+q+1)
else:
    print("NO")
```
