We need to find a perfect matching between the low-band stations (1..q) and high-band stations (q+1..2q) using the given edges. This is a bipartite matching problem.  

Since q can be up to 100,000 and t up to 400,000, we need an efficient algorithm — Hopcroft–Karp is suitable here.

We’ll:
- Build adjacency lists for low-band stations.
- Run Hopcroft–Karp to find maximum matching.
- If matching size == q, output YES and the matched pairs; otherwise NO.

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    INF = 10**9
    pair_u = [-1] * n
    pair_v = [-1] * m
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u

def main():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a-1].append(b-1)  # store high-band as 0-indexed

    matching_size, pair_u = hopcroft_karp(q, q, adj)

    if matching_size != q:
        print("NO")
    else:
        print("YES")
        for u in range(q):
            v = pair_u[u]
            print(u+1, v+1+q)

if __name__ == "__main__":
    main()
```
