We need to find a perfect matching in a bipartite graph where Tier A and Tier B each have \( q \) nodes, and there are \( t \) edges.  
This is a classic **maximum bipartite matching** problem, and since \( q \) can be up to \( 100,000 \) and \( t \) up to \( 400,000 \), we need an efficient algorithm — Hopcroft–Karp is suitable.

We’ll:
1. Build adjacency lists for Tier A nodes (1..q).
2. Run Hopcroft–Karp to find maximum matching.
3. If matching size == q, print YES and the matched pairs; otherwise print NO.

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    # n = number of left nodes (Tier A)
    # m = number of right nodes (Tier B)
    INF = 10**9
    pair_u = [-1] * (n + 1)
    pair_v = [-1] * (m + 1)
    dist = [0] * (n + 1)

    def bfs():
        dq = deque()
        for u in range(1, n + 1):
            if pair_u[u] == -1:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        found = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, n + 1):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u

def main():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q + 1)]
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b - q)  # store right node as 1..q

    matching_size, pair_u = hopcroft_karp(q, q, adj)

    if matching_size != q:
        print("NO")
    else:
        print("YES")
        for u in range(1, q + 1):
            v = pair_u[u]
            print(u, v + q)

if __name__ == "__main__":
    main()
```
