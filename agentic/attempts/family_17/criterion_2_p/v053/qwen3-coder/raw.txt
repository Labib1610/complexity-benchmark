```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    if t == 0:
        if q == 0:
            print("YES")
        else:
            print("NO")
        return
    
    pairs = []
    adj_ground = defaultdict(list)   # adj_ground[a] = list of indices of pairs containing ground node a
    adj_orbit = defaultdict(list)    # adj_orbit[b] = list of indices of pairs containing orbit node b
    
    idx = 2
    for i in range(t):
        a = int(data[idx])
        b = int(data[idx+1])
        pairs.append((a, b))
        adj_ground[a].append(i)
        adj_orbit[b].append(i)
        idx += 2
    
    # This is a bipartite matching problem.
    # Ground nodes: 1 to q
    # Orbit nodes: q+1 to 2q
    # Edges are the viable pairs
    # We want a perfect matching that matches all q ground nodes (and thus q orbit nodes)
    
    # Build adjacency list for ground nodes to orbit nodes
    # We'll use a greedy approach with backtracking via DFS (essentially finding maximum matching)
    
    match_orbit = [-1] * (2*q + 1)  # match_orbit[b] = index of pair that matches orbit node b, or -1
    match_ground_pair_index = [-1] * (q + 1)  # match_ground_pair_index[a] = index of pair matched to ground a
    
    def dfs(a, visited):
        for pair_idx in adj_ground[a]:
            b = pairs[pair_idx][1]
            if visited[b]:
                continue
            visited[b] = True
            
            # If b is unmatched or we can find augmenting path
            if match_orbit[b] == -1 or dfs(pairs[match_orbit[b]][0], visited):
                match_orbit[b] = pair_idx
                match_ground_pair_index[a] = pair_idx
                return True
        return False
    
    matched = 0
    for a in range(1, q+1):
        visited = [False] * (2*q + 1)
        if dfs(a, visited):
            matched += 1
    
    if matched == q:
        print("YES")
        for a in range(1, q+1):
            pair_idx = match_ground_pair_index[a]
            print(pairs[pair_idx][0], pairs[pair_idx][1])
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
