```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    adj = [[] for _ in range(q + 1)]
    idx = 2
    for i in range(t):
        a = int(data[idx]); b = int(data[idx + 1])
        idx += 2
        adj[a].append(b - q)
    matchL = [0] * (q + 1)
    matchR = [0] * (q + 1)
    visited = [0] * (q + 1)
    token = 0

    def dfs(u):
        nonlocal token
        for v in adj[u]:
            if visited[v] == token:
                continue
            visited[v] = token
            if matchR[v] == 0 or dfs(matchR[v]):
                matchR[v] = u
                matchL[u] = v
                return True
        return False

    matched = 0
    for u in range(1, q + 1):
        token += 1
        if dfs(u):
            matched += 1
            if matched == q:
                break

    if matched != q:
        sys.stdout.write("NO\n")
        return

    out = ["YES"]
    for u in range(1, q + 1):
        out.append(f"{u} {matchL[u] + q}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
```
