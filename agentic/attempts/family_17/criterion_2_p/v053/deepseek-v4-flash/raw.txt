We need to find a perfect matching in a bipartite graph where  
left side: nodes 1…q (ground arrays)  
right side: nodes q+1…2q (orbital relays)  

This is essentially finding a perfect matching in a bipartite graph.  

We can use Hopcroft–Karp algorithm which runs in O(E√V).

Here’s the code:  

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    INF = 10**9
    pairU = [-1] * n
    pairV = [-1] * m
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pairU[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pairV[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pairV[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pairU[u] == -1 and dfs(u):
                matching += 1
    return matching, pairU, pairV

def main():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q)]
    edges = []
    for _ in range(t):
        a, b = map(int, input().split())
        a -= 1
        b -= q + 1  # shift right side index to 0..q-1
        adj[a].append(b)
        edges.append((a, b))

    matching, pairU, pairV = hopcroft_karp(q, q, adj)

    if matching != q:
        print("NO")
        return

    print("YES")
    result = []
    for u in range(q):
        v = pairU[u]
        result.append(f"{u+1} {q+1+v}")
    sys.stdout.write("\n".join(result))

if __name__ == "__main__":
    main()
```
