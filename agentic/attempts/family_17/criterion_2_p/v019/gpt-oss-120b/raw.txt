```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    adj = [[] for _ in range(q + 1)]          # left side 1..q
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b - q)                  # map right side to 1..q

    NIL = 0
    pair_u = [NIL] * (q + 1)                  # pair for left vertices
    pair_v = [NIL] * (q + 1)                  # pair for right vertices
    dist = [0] * (q + 1)

    def bfs() -> bool:
        dq = deque()
        for u in range(1, q + 1):
            if pair_u[u] == NIL:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = -1
        found = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu == NIL:
                    found = True
                elif dist[pu] == -1:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
        return found

    sys.setrecursionlimit(1 << 25)
    def dfs(u: int) -> bool:
        for v in adj[u]:
            pu = pair_v[v]
            if pu == NIL or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = -1
        return False

    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == NIL and dfs(u):
                matching += 1

    if matching != q:
        sys.stdout.write("NO")
        return

    out_lines = ["YES"]
    for u in range(1, q + 1):
        v = pair_u[u] + q
        out_lines.append(f"{u} {v}")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```
