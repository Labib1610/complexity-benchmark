```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    edges = []
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1])
        idx += 2
        edges.append((a, b))

    # Hopcroft-Karp on bipartite graph: left nodes 1..q, right nodes q+1..2q
    # We map right node b to b - q for internal arrays.
    adj = [[] for _ in range(q + 1)]
    for a, b in edges:
        adj[a].append(b - q)

    INF = 10**9
    pair_u = [0] * (q + 1)   # pair_u[u] = v (1-indexed v) or 0
    pair_v = [0] * (q + 1)   # pair_v[v] = u or 0
    dist = [0] * (q + 1)

    from collections import deque

    def bfs():
        dq = deque()
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while dq:
            u = dq.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        dq.append(pu)
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_u[u] = v
                        pair_v[v] = u
                        return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1

    out = []
    if matching == q:
        out.append("YES")
        for u in range(1, q + 1):
            v = pair_u[u]
            out.append(f"{u} {v + q}")
    else:
        out.append("NO")

    sys.stdout.write("\n".join(out))

solve()
```
