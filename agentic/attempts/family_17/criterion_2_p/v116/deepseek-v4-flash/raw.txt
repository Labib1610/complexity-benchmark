We need to find a perfect matching in a bipartite graph between Alpha (q nodes) and Beta (q nodes). Since q can be up to 100,000 and t up to 400,000, we should use Hopcroft–Karp for efficiency.

Here’s the solution:

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q)]
for _ in range(t):
    a, b = map(int, input().split())
    adj[a - 1].append(b - q - 1)  # convert to 0-index

INF = float('inf')
pair_u = [-1] * q
pair_v = [-1] * q
dist = [0] * q

def bfs():
    dq = deque()
    for u in range(q):
        if pair_u[u] == -1:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = INF
    found = False
    while dq:
        u = dq.popleft()
        for v in adj[u]:
            if pair_v[v] == -1:
                found = True
            else:
                if dist[pair_v[v]] == INF:
                    dist[pair_v[v]] = dist[u] + 1
                    dq.append(pair_v[v])
    return found

def dfs(u):
    for v in adj[u]:
        if pair_v[v] == -1 or (dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v])):
            pair_u[u] = v
            pair_v[v] = u
            return True
    dist[u] = INF
    return False

matching = 0
while bfs():
    for u in range(q):
        if pair_u[u] == -1 and dfs(u):
            matching += 1

if matching != q:
    print("NO")
else:
    print("YES")
    for u in range(q):
        print(u + 1, pair_u[u] + q + 1)
```
