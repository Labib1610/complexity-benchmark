from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    if t < q:
        print("NO")
        return
    
    edges = []
    graph = [[] for _ in range(2*q + 1)]
    edge_indices = []
    
    idx = 2
    for i in range(t):
        a = int(data[idx])
        b = int(data[idx+1])
        idx += 2
        edges.append((a, b))
        graph[a].append((b, i))
        graph[b].append((a, i))
    
    # Try to find a perfect matching using BFS-based augmenting paths
    # This is essentially finding a maximum matching in a bipartite graph
    
    # Build adjacency list for bipartite graph
    adj = defaultdict(list)
    for i, (a, b) in enumerate(edges):
        adj[a].append((b, i))
    
    # Hungarian algorithm / Ford-Fulkerson method for maximum bipartite matching
    def bfs_match():
        pair_U = {}
        pair_V = {}
        dist = {}
        
        def bfs():
            queue = deque()
            for u in range(1, q+1):
                if u not in pair_U:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = float('inf')
            dist[None] = float('inf')
            
            while queue:
                u = queue.popleft()
                if dist[u] < dist[None]:
                    for v, idx in adj[u]:
                        if v in pair_V:
                            u_prime = pair_V[v]
                            if dist[u_prime] == float('inf'):
                                dist[u_prime] = dist[u] + 1
                                queue.append(u_prime)
                        else:
                            dist[None] = dist[u] + 1
                            return True
            return False
        
        def dfs(u):
            if u is not None:
                for v, idx in adj[u]:
                    if v in pair_V:
                        u_prime = pair_V[v]
                        if dist[u_prime] == dist[u] + 1 and dfs(u_prime):
                            pair_U[u] = v
                            pair_V[v] = u
                            return True
                    else:
                        pair_U[u] = v
                        pair_V[v] = u
                        return True
                dist[u] = float('inf')
                return False
            return True
        
        matching = 0
        while bfs():
            for u in range(1, q+1):
                if u not in pair_U and dfs(u):
                    matching += 1
        return matching, pair_U
    
    matching_size, matching_pairs = bfs_match()
    
    if matching_size < q:
        print("NO")
        return
    
    print("YES")
    for u in range(1, q+1):
        v = matching_pairs[u]
        print(u, v)

solve()
