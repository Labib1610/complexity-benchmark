from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build graph for bipartite matching
    # Left side: primary reflectors (1 to q)
    # Right side: secondary digitizers (q+1 to 2*q)
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Hopcroft-Karp implementation for maximum bipartite matching
    # or simpler Ford-Fulkerson/DFS approach
    
    match_right = {}  # digitizer -> reflector
    match_left = {}   # reflector -> digitizer
    
    def dfs(u, visited):
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If v is unmatched or we can find augmenting path
            if v not in match_right or dfs(match_right[v], visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    # Try to match all left vertices (reflectors)
    matching = 0
    for u in range(1, q + 1):
        if dfs(u, set()):
            matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_left[u])
    else:
        print("NO")

solve()
