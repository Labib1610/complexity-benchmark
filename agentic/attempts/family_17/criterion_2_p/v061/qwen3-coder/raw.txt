from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    # Left side: tasks 1 to q
    # Right side: tasks q+1 to 2q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching
    # Using Hopcroft-Karp algorithm implementation
    
    # match_right[v] = u means right node v is matched to left node u
    match_left = [0] * (q + 1)  # 1-indexed
    match_right = [0] * (q + 1)  # 1-indexed, for nodes q+1 to 2q, we use index b-q
    
    def bfs():
        queue = deque()
        dist = [0] * (q + 1)
        
        for u in range(1, q + 1):
            if match_left[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            
            if dist[u] < dist[0]:
                for v in graph[u]:
                    # v is in range q+1 to 2q, so we map it to 1 to q
                    v_idx = v - q
                    
                    if dist[match_right[v_idx]] == float('inf'):
                        dist[match_right[v_idx]] = dist[u] + 1
                        queue.append(match_right[v_idx])
        
        return dist[0] != float('inf')
    
    def dfs(u):
        if u == 0:
            return True
            
        for v in graph[u]:
            v_idx = v - q
            if dist[match_right[v_idx]] == dist[u] + 1:
                if dfs(match_right[v_idx]):
                    match_right[v_idx] = u
                    match_left[u] = v
                    return True
        
        dist[u] = float('inf')
        return False
    
    matching = 0
    dist = [0] * (q + 1)
    
    while bfs():
        for u in range(1, q + 1):
            if match_left[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_left[u])
    else:
        print("NO")

solve()
