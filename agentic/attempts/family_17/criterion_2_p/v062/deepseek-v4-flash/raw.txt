We need to match each species from 1 to q with exactly one species from q+1 to 2q, using only the given edges. This is a perfect matching problem in a bipartite graph.

We can solve it using a maximum bipartite matching algorithm (Hopcroft–Karp) since q can be up to 100,000 and t up to 400,000.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q + 1)]
for _ in range(t):
    a, b = map(int, input().split())
    adj[a].append(b - q)  # store right side as 1..q

# Hopcroft-Karp
INF = 10**9
pairU = [0] * (q + 1)
pairV = [0] * (q + 1)
dist = [0] * (q + 1)

def bfs():
    dq = deque()
    for u in range(1, q + 1):
        if pairU[u] == 0:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = INF
    dist[0] = INF
    while dq:
        u = dq.popleft()
        if dist[u] < dist[0]:
            for v in adj[u]:
                if dist[pairV[v]] == INF:
                    dist[pairV[v]] = dist[u] + 1
                    dq.append(pairV[v])
    return dist[0] != INF

def dfs(u):
    if u != 0:
        for v in adj[u]:
            if dist[pairV[v]] == dist[u] + 1:
                if dfs(pairV[v]):
                    pairV[v] = u
                    pairU[u] = v
                    return True
        dist[u] = INF
        return False
    return True

matching = 0
while bfs():
    for u in range(1, q + 1):
        if pairU[u] == 0:
            if dfs(u):
                matching += 1

if matching == q:
    print("YES")
    for u in range(1, q + 1):
        print(u, pairU[u] + q)
else:
    print("NO")
```
