from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Primary modules: 1 to q
    # Secondary modules: q+1 to 2q (we'll map them to 1 to q for easier handling)
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        # Map secondary module b to index b - q (1-based to 1-based in second set)
        adj[a].append(b - q)
        edges.append((a, b - q))
    
    # Use Hopcroft-Karp algorithm for maximum bipartite matching
    # or simpler Ford-Fulkerson based approach
    
    # pair_u[u] stores the matched vertex in V for vertex u in U (primary)
    # pair_v[v] stores the matched vertex in U for vertex v in V (secondary)
    pair_u = [0] * (q + 1)  # 1-indexed
    pair_v = [0] * (q + 1)  # 1-indexed
    dist = [0] * (q + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == float('inf'):
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != float('inf')
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = float('inf')
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            # Convert back to original secondary module ID
            print(u, pair_u[u] + q)
    else:
        print("NO")

solve()
