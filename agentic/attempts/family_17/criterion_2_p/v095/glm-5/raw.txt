```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    
    # Build bipartite graph: left = 1..q, right = q+1..2q
    # We'll use 0-indexed internally: left 0..q-1, right 0..q-1 (representing q+1..2q)
    head = [-1] * q
    to = []
    nxt = []
    
    for _ in range(t):
        a = int(next(it)) - 1  # 0-indexed left
        b = int(next(it)) - 1 - q  # 0-indexed right
        to.append(b)
        nxt.append(head[a])
        head[a] = len(to) - 1
    
    # Hopcroft-Karp
    pair_u = [-1] * q  # left i matched to right pair_u[i]
    pair_v = [-1] * q  # right j matched to left pair_v[j]
    dist = [0] * q
    
    INF = 10**9
    
    def bfs():
        dq = deque()
        for u in range(q):
            if pair_u[u] == -1:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        found = False
        while dq:
            u = dq.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                pu = pair_v[v]
                if pu == -1:
                    found = True
                elif dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
                e = nxt[e]
        return found
    
    def dfs(u):
        e = head[u]
        while e != -1:
            v = to[e]
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
            e = nxt[e]
        dist[u] = INF
        return False
    
    matching = 0
    while bfs():
        for u in range(q):
            if pair_u[u] == -1:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        out = ["YES"]
        for u in range(q):
            v = pair_u[u]
            out.append(f"{u+1} {v+1+q}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

solve()
```
