```python
from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    # Left side: specimens 1 to q
    # Right side: capsules q+1 to 2*q
    graph = defaultdict(list)
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm approach
    # But we'll use a simpler approach with DFS-based augmenting paths
    
    match_specimen = {}  # specimen -> capsule
    match_capsule = {}   # capsule -> specimen
    
    def dfs(u, visited):
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If capsule v is unmatched or we can find augmenting path
            if v not in match_capsule or dfs(match_capsule[v], visited):
                match_specimen[u] = v
                match_capsule[v] = u
                return True
        return False
    
    matched = 0
    for i in range(1, q + 1):
        visited = set()
        if dfs(i, visited):
            matched += 1
    
    if matched == q:
        print("YES")
        for i in range(1, q + 1):
            print(i, match_specimen[i])
    else:
        print("NO")

solve()
```
