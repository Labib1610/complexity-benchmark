from collections import defaultdict, deque

def find_maximum_matching(graph, mentor_count):
    # Hopcroft-Karp algorithm implementation for bipartite matching
    pair_u = [0] * (mentor_count + 1)  # 1-indexed
    pair_v = [0] * (mentor_count + 1)  # 1-indexed
    dist = [0] * (mentor_count + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, mentor_count + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in graph[u]:
                    if dist[pair_v[v]] == float('inf'):
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != float('inf')
    
    def dfs(u):
        if u != 0:
            for v in graph[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = float('inf')
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, mentor_count + 1):
            if pair_u[u] == 0 and dfs(u):
                matching += 1
    
    if matching == mentor_count:
        print("YES")
        for u in range(1, mentor_count + 1):
            print(u, pair_u[u])
    else:
        print("NO")

# Read input
q, t = map(int, input().split())

# Build bipartite graph
# Mentors: 1 to q
# Apprentices: q+1 to 2*q (but we'll map them to 1 to q for easier handling)
graph = defaultdict(list)

for _ in range(t):
    a, b = map(int, input().split())
    # Map apprentice b to index 1 to q
    apprentice_idx = b - q
    graph[a].append(apprentice_idx)

find_maximum_matching(graph, q)
