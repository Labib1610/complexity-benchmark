        u = int(next(it))
        v = int(next(it))
        v_idx = v - q
        adj[u].append((v_idx, i))
        edges.append((u, v))

    # Hopcroft-Karp Algorithm
    INF = float('inf')
    pair_U = [0] * (q + 1)
    pair_V = [0] * (q + 1)
    dist = [0] * (q + 1)

    from collections import deque

    def bfs():
        dq = deque()
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        
        dist[0] = INF
        
        while dq:
            u = dq.popleft()
            if dist[u] < dist[0]:
                # Iterate neighbors
                for v, _ in adj[u]:
                    if pair_V[v] == 0:
                        # Unmatched in V, so we can reach 0 (NIL)
                        dist[0] = dist[u] + 1
                    elif dist[pair_V[v]] == INF:
                        dist[pair_V[v]] = dist[u] + 1
                        dq.append(pair_V[v])
        
        return dist[0] != INF

    def dfs(u):
        for v, i in adj[u]:
            if pair_V[v] == 0 or (dist[pair_V[v]] == dist[u] + 1 and dfs(pair_V[v])):
                pair_U[u] = v
                pair_V[v] = u

                # For output, we just need to know which u is matched to which v.
                # The actual edge line can be reconstructed.
                return True
        
        dist[u] = INF
        return False

    matching = 0
