```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    
    # Build adjacency list for controllers (1 to q)
    # Platforms are q+1 to 2q
    adj = [[] for _ in range(q + 1)]
    
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
    
    # Sort adjacency lists for deterministic behavior (optional but good)
    for i in range(1, q + 1):
        adj[i].sort()
    
    # match_platform[p] = controller c matched with platform p, 0 if free
    match_platform = [0] * (2 * q + 1)
    
    def dfs(u, visited):
        for v in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match_platform[v] == 0 or dfs(match_platform[v], visited):
                match_platform[v] = u
                return True
        return False
    
    match_count = 0
    for u in range(1, q + 1):
        visited = [False] * (2 * q + 1)
        if dfs(u, visited):
            match_count += 1
    
    if match_count == q:
        print("YES")
        # Output matches: need to list q pairs
        # We have match_platform[p] = c
        # We need to output c p for each match
        # We can iterate platforms q+1 to 2q
        # But we need to output in some order, problem says "q lines each with the two items of one chosen pair"
        # The order of lines doesn't seem strictly specified, but usually any valid matching is fine.
        # Let's output in order of controllers 1 to q for clarity, finding their matched platform.
        # We can build a match_controller array for easy printing.
        match_controller = [0] * (q + 1)
        for p in range(q + 1, 2 * q + 1):
            c = match_platform[p]
            if c != 0:
                match_controller[c] = p
        
        for c in range(1, q + 1):
            print(c, match_controller[c])
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
