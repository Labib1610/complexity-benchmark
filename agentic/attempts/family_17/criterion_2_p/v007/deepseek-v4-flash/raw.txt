We need to check if a perfect matching exists between controllers (1 to q) and platforms (q+1 to 2q) given allowed edges, and output one such matching if it exists.

The problem constraints are large, so we need an efficient matching algorithm — Hopcroft–Karp is suitable given up to 400,000 edges.

```python
import sys
from collections import deque

def hopcroft_karp(n, m, adj):
    # n: number of left nodes (controllers), m: number of right nodes (platforms)
    INF = 10**9
    pair_u = [-1] * n
    pair_v = [-1] * m
    dist = [0] * n

    def bfs():
        dq = deque()
        for u in range(n):
            if pair_u[u] == -1:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        found = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu == -1:
                    found = True
                elif dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u, pair_v

def main():
    data = sys.stdin.buffer.read().split()
    q = int(data[0])
    t = int(data[1])
    adj = [[] for _ in range(q)]
    idx = 2
    for _ in range(t):
        a = int(data[idx]) - 1
        b = int(data[idx+1]) - q - 1  # platform index from 0 to q-1
        idx += 2
        adj[a].append(b)
    
    matching, pair_u, _ = hopcroft_karp(q, q, adj)
    
    if matching != q:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n")
        for u in range(q):
            sys.stdout.write(f"{u+1} {pair_u[u]+q+1}\n")

if __name__ == "__main__":
    main()
```
