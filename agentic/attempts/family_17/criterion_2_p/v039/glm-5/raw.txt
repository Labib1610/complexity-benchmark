import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    n = 2 * q
    head = list(range(n + 1))
    to = [0] * (2 * t)
    nxt = [0] * (2 * t)
    idx = 0
    for i in range(t):
        a = int(data[2 + 2 * i])
        b = int(data[3 + 2 * i])
        to[idx] = b
        nxt[idx] = head[a]
        head[a] = idx
        idx += 1
        to[idx] = a
        nxt[idx] = head[b]
        head[b] = idx
        idx += 1

    match = [0] * (n + 1)
    visited = [0] * (n + 1)
    stamp = 0
    stack = []

    for u in range(1, q + 1):
        if match[u] != 0:
            continue
        stamp += 1
        stack.clear()
        stack.append((u, head[u], 0))
        visited[u] = stamp
        aug = False
        while stack:
            node, e, p = stack[-1]
            if e == 0:
                stack.pop()
                if p != 0 and stack:
                    stack[-1] = (stack[-1][0], nxt[stack[-1][1]], stack[-1][2])
                continue
            v = to[e - 1]
            if visited[v] == stamp:
                stack[-1] = (node, nxt[e - 1], p)
                continue
            visited[v] = stamp
            w = match[v]
            if w == 0:
                match[v] = node
                match[node] = v
                stack.pop()
                aug = True
                break
            else:
                stack.append((w, head[w], e))
        if aug:
            continue

        # second pass: try to find augmenting path through matched vertices
        stack.clear()
        stack.append((u, head[u], 0))
        visited[u] = stamp
        aug = False
        while stack:
            node, e, p = stack[-1]
            if e == 0:
                stack.pop()
                if p != 0 and stack:
                    stack[-1] = (stack[-1][0], nxt[stack[-1][1]], stack[-1][2])
                continue
            v = to[e - 1]
            if visited[v] == stamp:
                stack[-1] = (node, nxt[e - 1], p)
                continue
            visited[v] = stamp
            w = match[v]
            if w == 0:
                # augment
                match[v] = node
                match[node] = v
                stack.pop()
                cur = node
                while stack:
                    _, ep, _ = stack.pop()
                    pv = to[ep - 1]
                    pw = match[pv]
                    match[pv] = cur
                    match[cur] = pv
                    match[pw] = 0
                    cur = pw
                match[cur] = 0
                aug = True
                break
            else:
                stack.append((w, head[w], e))
        if not aug:
            print("NO")
            return

    out = ["YES"]
    for u in range(1, q + 1):
        out.append(f"{u} {match[u]}")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
