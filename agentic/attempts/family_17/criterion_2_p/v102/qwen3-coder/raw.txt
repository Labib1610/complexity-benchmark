from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    if t < q:
        print("NO")
        return
    
    # Build bipartite graph
    graph = defaultdict(list)
    pairs = []
    
    for i in range(t):
        a, b = map(int, input().split())
        graph[a].append((b, i))
        pairs.append((a, b))
    
    # Try to find a perfect matching using BFS-based augmenting paths
    # This is essentially the Hopcroft-Karp algorithm simplified
    
    # match_responder[r] = index of pair that matches responder r, or -1 if unmatched
    match_responder = {}
    match_initiator = {}
    
    # Initialize all as unmatched
    for i in range(1, q + 1):
        match_initiator[i] = -1
    
    for i in range(q + 1, 2 * q + 1):
        match_responder[i] = -1
    
    def bfs():
        # Find shortest augmenting paths
        queue = deque()
        dist = {}
        
        for i in range(1, q + 1):
            dist[i] = float('inf')
            
        for i in range(1, q + 1):
            if match_initiator[i] == -1:
                dist[i] = 0
                queue.append(i)
        
        dist[None] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[None]:
                for v, pair_idx in graph[u]:
                    mv = match_responder[v]
                    if dist[mv] == float('inf'):
                        dist[mv] = dist[u] + 1
                        if mv == -1:
                            dist[None] = dist[mv]
                        else:
                            queue.append(mv)
        
        return dist[None] != float('inf')
    
    def dfs(u):
        if u != -1:
            for v, pair_idx in graph[u]:
                mv = match_responder[v]
                if dist[mv] == dist[u] + 1 and dfs(mv):
                    match_initiator[u] = pair_idx
                    match_responder[v] = pair_idx
                    return True
            dist[u] = float('inf')
            return False
        return True
    
    # Main matching loop
    matching = 0
    while bfs():
        for i in range(1, q + 1):
            if match_initiator[i] == -1 and dfs(i):
                matching += 1
    
    if matching < q:
        print("NO")
        return
    
    print("YES")
    result = [None] * q
    for i in range(1, q + 1):
        pair_idx = match_initiator[i]
        a, b = pairs[pair_idx]
        result[a-1] = f"{a} {b}"
    
    for line in result:
        print(line)

solve()
