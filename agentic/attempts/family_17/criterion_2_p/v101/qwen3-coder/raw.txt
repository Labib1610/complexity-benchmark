from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        # Convert to 0-indexed
        u = a - 1  # low-band nodes: 0 to q-1
        v = b - 1 - q  # high-band nodes: 0 to q-1
        graph[u].append(v)
        edges.append((u, v))
    
    # Try to find maximum matching using Hopcroft-Karp algorithm
    # But since we only need to check if perfect matching exists,
    # we can use a simpler approach with DFS
    
    match_low = [-1] * q  # match_low[u] = v means low node u is matched to high node v
    match_high = [-1] * q  # match_high[v] = u means high node v is matched to low node u
    
    def dfs(u, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            # If high node v is unmatched or we can find augmenting path
            if match_high[v] == -1 or dfs(match_high[v], visited):
                match_low[u] = v
                match_high[v] = u
                return True
        return False
    
    # Find maximum matching
    matching = 0
    for u in range(q):
        visited = [False] * q
        if dfs(u, visited):
            matching += 1
    
    if matching < q:
        print("NO")
        return
    
    print("YES")
    for u in range(q):
        v = match_low[u]
        # Convert back to 1-indexed
        print(u + 1, v + 1 + q)

solve()
