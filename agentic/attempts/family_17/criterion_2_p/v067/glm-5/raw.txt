    for i in range(len(ea)):
        adj[ea[i] - 1].append(eb[i] - bshift)
    # check for empty
    for u in range(n):
        if not adj[u]:
            return None
    pair_u = [-1] * n
    pair_v = [-1] * n
    dist = [INF] * (n + 1)
    
    # Greedy init
    for u in range(n):
        for v in adj[u]:
            if pair_v[v] < 0:
                pair_v[v] = u
                pair_u[u] = v
                break
    
    matched = 0
    for u in range(n):
        if pair_u[u] >= 0:
            matched += 1
    if matched == n:

        queue = []
