```python
import sys
from collections import defaultdict, deque

def solve():
    q, t = map(int, sys.stdin.readline().split())
    
    # Build graph for bipartite matching
    # Controllers: 1 to q
    # Platforms: q+1 to 2*q
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, sys.stdin.readline().split())
        adj[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm implementation
    # But we'll use a simpler approach with DFS for this case
    
    match_controller = [0] * (q + 1)  # match_controller[c] = matched platform
    match_platform = [0] * (2 * q + 1)  # match_platform[p] = matched controller
    
    def dfs(u, visited):
        for v in adj[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If platform v is unmatched or we can find augmenting path
            if match_platform[v] == 0 or dfs(match_platform[v], visited):
                match_controller[u] = v
                match_platform[v] = u
                return True
        return False
    
    matching = 0
    for i in range(1, q + 1):
        if dfs(i, set()):
            matching += 1
    
    if matching == q:
        print("YES")
        for i in range(1, q + 1):
            print(i, match_controller[i])
    else:
        print("NO")

solve()
```
