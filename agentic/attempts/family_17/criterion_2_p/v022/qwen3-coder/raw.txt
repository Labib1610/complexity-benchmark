from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    if t < q:
        print("NO")
        return
    
    # Build adjacency list for bipartite graph
    graph = defaultdict(list)
    edges = []
    
    for i in range(t):
        a, b = map(int, input().split())
        # Convert to 0-indexed
        a -= 1
        b -= 1
        graph[a].append((b, i))
        edges.append((a, b))
    
    # Try to find a perfect matching using BFS-based augmenting paths
    # This is essentially the Hopcroft-Karp algorithm simplified
    
    # match_u[u] stores the matched vertex v for primary u (-1 if unmatched)
    # match_v[v] stores the matched vertex u for secondary v (-1 if unmatched)
    match_u = [-1] * q
    match_v = [-1] * q
    
    def bfs():
        # Returns true if there is an augmenting path
        dist = [-1] * q
        queue = deque()
        
        # Find free vertices in U (primary)
        for u in range(q):
            if match_u[u] == -1:
                dist[u] = 0
                queue.append(u)
        
        found = False
        
        while queue:
            u = queue.popleft()
            
            for v, _ in graph[u]:
                # v is 0-indexed in secondary group, so subtract q
                v_idx = v - q
                
                # If v is not matched, we found an augmenting path
                if match_v[v_idx] == -1:
                    found = True
                elif dist[match_v[v_idx]] == -1:
                    # If v is matched but its partner is not yet visited
                    dist[match_v[v_idx]] = dist[u] + 1
                    queue.append(match_v[v_idx])
        
        return found
    
    def dfs(u):
        # DFS to find augmenting path starting from u
        for v, _ in graph[u]:
            v_idx = v - q
            
            # If v is free, take it
            if match_v[v_idx] == -1:
                match_v[v_idx] = u
                match_u[u] = v
                return True
            
            # If v is matched, try to find alternative for its current match
            elif dist[match_v[v_idx]] == dist[u] + 1 and dfs(match_v[v_idx]):
                match_v[v_idx] = u
                match_u[u] = v
                return True
        
        dist[u] = -1  # Mark as visited
        return False
    
    # Main matching loop
    matching = 0
    while bfs():
        for u in range(q):
            if match_u[u] == -1:
                if dfs(u):
                    matching += 1
    
    if matching < q:
        print("NO")
        return
    
    print("YES")
    for u in range(q):
        # Convert back to 1-indexed
        print(u + 1, match_u[u] + 1)

solve()
