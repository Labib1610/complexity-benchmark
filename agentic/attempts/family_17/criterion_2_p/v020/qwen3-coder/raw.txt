```python
from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Left side: acids 1 to q
    # Right side: alkalis q+1 to 2q
    graph = defaultdict(list)
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm approach
    # But we'll use a simpler approach with DFS-based augmenting paths
    
    match_alkali = {}  # alkali -> acid matched to it
    match_acid = {}    # acid -> alkali matched to it
    
    def dfs(u, visited):
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If v is unmatched or we can find augmenting path from v's match
            if v not in match_alkali or dfs(match_alkali[v], visited):
                match_alkali[v] = u
                match_acid[u] = v
                return True
        return False
    
    # Try to match each acid
    matched = 0
    for acid in range(1, q + 1):
        if dfs(acid, set()):
            matched += 1
    
    if matched == q:
        print("YES")
        for acid in range(1, q + 1):
            print(acid, match_acid[acid])
    else:
        print("NO")

solve()
```
