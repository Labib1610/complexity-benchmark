```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    q = int(data[0])
    t = int(data[1])
    
    index = 2
    pairs = []
    adj = defaultdict(list)
    
    for _ in range(t):
        a = int(data[index])
        b = int(data[index+1])
        pairs.append((a, b))
        adj[a].append(b)
        index += 2
    
    # We need to find a perfect matching such that:
    # 1. Each alpha (1 to q) is matched to exactly one beta (q+1 to 2q)
    # 2. Each beta is matched to exactly one alpha
    # 3. For each pair (a,b), b > a
    # 4. Sum of alphas = q*(q+1)//2
    # 5. Sum of betas = q*(3*q+1)//2
    
    # The ideal pairing would be alpha i with beta (q+i), which gives:
    # sum of alphas = 1+2+...+q = q*(q+1)//2
    # sum of betas = (q+1)+(q+2)+...+(2q) = q*(3*q+1)//2
    # This satisfies conditions 4 and 5.
    # So we just need to check if there exists a perfect matching in the bipartite graph
    # where left side is alphas (1..q) and right side is betas (q+1..2q)
    
    # Build graph for matching
    # Left nodes: 1 to q (alpha)
    # Right nodes: q+1 to 2q (beta), we map them to 1 to q for easier handling
    # So beta j maps to j-q
    
    # Use Hopcroft-Karp or similar. But since q can be up to 1e5, let's use a simpler approach:
    # Try to build a matching using greedy + backtracking or max flow
    
    # Actually, let's model this as a maximum bipartite matching problem
    # and check if we can get a perfect matching of size q
    
    # Let's use a simple augmenting path approach (Hungarian might be overkill)
    
    # Build adjacency list for alphas
    graph = [[] for _ in range(q+1)]  # index 0 unused
    for a in range(1, q+1):
        for b in adj[a]:
            graph[a].append(b - q)  # map beta id to 1-based index in right set
            
    # Now we want a matching from left set (1..q) to right set (1..q)
    match_right = [0] * (q+1)  # match_right[r] = l means right node r is matched to left node l
    match_left = [0] * (q+1)   # match_left[l] = r means left node l is matched to right node r
    
    def dfs(u, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match_right[v] == 0 or dfs(match_right[v], visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    matching = 0
    for u in range(1, q+1):
        visited = [False] * (q+1)
        if dfs(u, visited):
            matching += 1
            
    if matching != q:
        print("NO")
        return
        
    print("YES")
    for u in range(1, q+1):
        v = match_left[u]
        beta_id = v + q
        print(u, beta_id)

if __name__ == "__main__":
    main()
```
