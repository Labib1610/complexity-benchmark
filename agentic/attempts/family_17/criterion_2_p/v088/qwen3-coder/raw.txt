from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Transmitters: 1 to q
    # Receivers: q+1 to 2*q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm approach
    # But we'll use a simpler approach with DFS-based maximum bipartite matching
    
    match_receiver = {}  # receiver -> transmitter
    match_transmitter = {}  # transmitter -> receiver
    
    def dfs(u, visited):
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If receiver v is unmatched or we can find augmenting path
            if v not in match_receiver or dfs(match_receiver[v], visited):
                match_receiver[v] = u
                match_transmitter[u] = v
                return True
        return False
    
    matching = 0
    for u in range(1, q + 1):
        if dfs(u, set()):
            matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_transmitter[u])
    else:
        print("NO")

solve()
