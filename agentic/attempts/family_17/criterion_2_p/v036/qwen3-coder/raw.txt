from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    # Controllers: 1 to q
    # Platforms: q+1 to 2*q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm approach
    # But we'll use a simpler approach with DFS-based augmenting paths
    
    match_controller = {}  # controller -> platform
    match_platform = {}    # platform -> controller
    
    def dfs(u, visited):
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If platform v is unmatched or we can find augmenting path
            if v not in match_platform or dfs(match_platform[v], visited):
                match_controller[u] = v
                match_platform[v] = u
                return True
        return False
    
    matching = 0
    for u in range(1, q + 1):
        if dfs(u, set()):
            matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_controller[u])
    else:
        print("NO")

solve()
