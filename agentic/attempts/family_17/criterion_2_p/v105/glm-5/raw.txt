```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    edges = []
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1])
        idx += 2
        edges.append((a, b - q - 1))

    adj = [[] for _ in range(q + 1)]
    for a, b in edges:
        adj[a].append(b)

    pair = [0] * (q + 1)
    matched = [0] * (q + 1)
    order = list(range(1, q + 1))
    order.sort(key=lambda x: len(adj[x]))

    for u in order:
        visited = [False] * (q + 1)
        stack = [(u, 0)]
        visited[u] = True
        parent = [0] * (q + 1)
        found = -1
        while stack:
            node, pi = stack.pop()
            if node <= 0:
                continue
            i = pi
            while i < len(adj[node]):
                v = adj[node][i]
                i += 1
                if visited[v]:
                    continue
                visited[v] = True
                parent[v] = node
                if matched[v] == 0:
                    found = v
                    stack.append((node, i))
                    break
                else:
                    stack.append((node, i))
                    stack.append((matched[v], 0))
                    break
            else:
                pass
        if found != -1:
            cur = found
            while cur != 0:
                p = parent[cur]
                prev = pair[p]
                pair[p] = cur
                matched[cur] = p
                cur = prev

    if any(pair[i] == 0 for i in range(1, q + 1)):
        sys.stdout.write("NO\n")
        return

    out = ["YES"]
    for i in range(1, q + 1):
        out.append(f"{i} {pair[i] + q + 1}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
```
