from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Primary nodes: 1 to q
    # Secondary nodes: q+1 to 2*q
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching in bipartite graph
    # Using Hopcroft-Karp algorithm implementation
    
    # match_u[u] stores the matched vertex in V for vertex u in U
    # match_v[v] stores the matched vertex in U for vertex v in V
    match_u = [0] * (q + 1)  # 1-indexed for primary assets
    match_v = [0] * (q + 1)  # 1-indexed for secondary assets (relative to q)
    
    def bfs():
        dist = [0] * (q + 1)
        queue = deque()
        
        for u in range(1, q + 1):
            if match_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            
            if dist[u] < dist[0]:
                for v in adj[u]:
                    # v is in range [q+1, 2*q], so we use v-q to index match_v
                    v_idx = v - q
                    if dist[match_v[v_idx]] == float('inf'):
                        dist[match_v[v_idx]] = dist[u] + 1
                        queue.append(match_v[v_idx])
        
        return dist[0] != float('inf')
    
    def dfs(u):
        if u == 0:
            return True
            
        for v in adj[u]:
            v_idx = v - q
            if dist[match_v[v_idx]] == dist[u] + 1:
                if dfs(match_v[v_idx]):
                    match_v[v_idx] = u
                    match_u[u] = v
                    return True
                    
        dist[u] = float('inf')
        return False
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if match_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_u[u])
    else:
        print("NO")

solve()
