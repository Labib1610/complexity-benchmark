```python
from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Left side: Control Modules (1 to q)
    # Right side: Propulsion Units (q+1 to 2q)
    graph = defaultdict(list)
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
    
    # Try to find a perfect matching using Ford-Fulkerson with BFS (Edmonds-Karp)
    # We'll implement Hopcroft-Karp for better performance
    
    # match_u[u] stores the matched vertex in V for vertex u in U (or 0 if unmatched)
    # match_v[v] stores the matched vertex in U for vertex v in V (or 0 if unmatched)
    match_u = [0] * (q + 1)  # Index 0 unused
    match_v = [0] * (q + 1)  # Index 0 unused
    
    def bfs():
        # Find shortest augmenting paths
        dist = [0] * (q + 1)
        queue = deque()
        
        # Initialize queue with free nodes in U
        for u in range(1, q + 1):
            if match_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        
        dist[0] = float('inf')  # 0 represents NIL
        
        while queue:
            u = queue.popleft()
            
            if dist[u] < dist[0]:  # If we haven't reached NIL yet
                for v in graph[u]:
                    # If v is matched to some other node
                    if dist[match_v[v - q]] == float('inf'):
                        dist[match_v[v - q]] = dist[u] + 1
                        queue.append(match_v[v - q])
        
        return dist[0] != float('inf')
    
    def dfs(u):
        if u == 0:
            return True
            
        for v in graph[u]:
            paired_u = match_v[v - q]
            if dist[paired_u] == dist[u] + 1 and dfs(paired_u):
                match_v[v - q] = u
                match_u[u] = v
                return True
                
        dist[u] = float('inf')
        return False
    
    # Hopcroft-Karp main algorithm
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if match_u[u] == 0 and dfs(u):
                matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_u[u])
    else:
        print("NO")

solve()
```
