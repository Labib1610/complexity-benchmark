import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    edges = []
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1])
        idx += 2
        edges.append((a, b))

    # Build bipartite graph: left side 1..q, right side 1..q (b - q)
    adj = [[] for _ in range(q + 1)]
    for a, b in edges:
        adj[a].append(b - q)

    # Hopcroft-Karp
    INF = 10**9
    pair_u = [0] * (q + 1)   # pair_u[u] = v (1-indexed)
    pair_v = [0] * (q + 1)   # pair_v[v] = u
    dist = [0] * (q + 1)

    from collections import deque

    def bfs():
        dq = deque()
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while dq:
            u = dq.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        dq.append(pu)
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_u[u] = v
                        pair_v[v] = u
                        return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1

    if matching != q:
        sys.stdout.write("NO\n")
        return

    out = ["YES"]
    for u in range(1, q + 1):
        v = pair_u[u]
        out.append(f"{u} {v + q}")
    sys.stdout.write("\n".join(out) + "\n")

if __name__ == "__main__":
    main()
