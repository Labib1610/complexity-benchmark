from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Create adjacency list for bipartite graph
    # Left side: input terminals 1 to q
    # Right side: output terminals q+1 to 2q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using Ford-Fulkerson with BFS (Edmonds-Karp)
    # We need to match all q input terminals to q output terminals
    
    # Build residual graph
    # Nodes: 0 (source), 1..q (inputs), (q+1)..(2q) (outputs), 2*q+1 (sink)
    res_graph = defaultdict(lambda: defaultdict(int))
    
    # Connect source to all input terminals
    for i in range(1, q + 1):
        res_graph[0][i] = 1
    
    # Connect all output terminals to sink
    for i in range(q + 1, 2 * q + 1):
        res_graph[i][2 * q + 1] = 1
    
    # Add edges from input to output terminals
    for a, b in edges:
        res_graph[a][b] = 1
    
    # Ford-Fulkerson with BFS to find max flow
    def bfs():
        parent = [-1] * (2 * q + 2)
        visited = [False] * (2 * q + 2)
        queue = deque([0])
        visited[0] = True
        
        while queue:
            u = queue.popleft()
            
            for v in res_graph[u]:
                if not visited[v] and res_graph[u][v] > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    
                    if v == 2 * q + 1:  # Reached sink
                        return parent
        return None
    
    # Find maximum flow
    max_flow = 0
    while True:
        parent = bfs()
        if parent is None:
            break
            
        # Find minimum capacity along the path
        path_flow = float('inf')
        s = 2 * q + 1
        while s != 0:
            path_flow = min(path_flow, res_graph[parent[s]][s])
            s = parent[s]
        
        # Update residual capacities
        v = 2 * q + 1
        while v != 0:
            u = parent[v]
            res_graph[u][v] -= path_flow
            res_graph[v][u] += path_flow
            v = u
            
        max_flow += path_flow
    
    # Check if we have a perfect matching (flow = q)
    if max_flow != q:
        print("NO")
        return
    
    # Reconstruct the matching
    print("YES")
    matching = []
    
    # For each input terminal, find which output terminal it's connected to
    for i in range(1, q + 1):
        for j in range(q + 1, 2 * q + 1):
            # If there's flow from i to j in the residual graph, 
            # it means this edge is in the matching
            if res_graph[j][i] > 0:  # Back edge indicates used edge
                matching.append((i, j))
                break
    
    # Sort by input terminal for consistent output
    matching.sort()
    
    for a, b in matching:
        print(a, b)

solve()
