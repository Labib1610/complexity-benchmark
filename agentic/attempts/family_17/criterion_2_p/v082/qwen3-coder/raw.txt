from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    if t < q:
        print("NO")
        return
    
    # Build adjacency list for bipartite graph
    # Left side: fusion cores 1 to q
    # Right side: cooling towers q+1 to 2q
    graph = defaultdict(list)
    edges = []
    
    for i in range(t):
        a, b = map(int, input().split())
        graph[a].append((b, i))
        edges.append((a, b))
    
    # Try to find a perfect matching using BFS-based augmenting paths
    # This is essentially the Hopcroft-Karp algorithm simplified for our needs
    
    # match_right[b] = index of edge that matches cooling tower b
    match_right = {}
    # match_left[a] = index of edge that matches fusion core a
    match_left = {}
    
    def bfs():
        # Find shortest augmenting paths
        queue = deque()
        dist = {}
        
        for a in range(1, q + 1):
            if a not in match_left:
                dist[a] = 0
                queue.append(a)
            else:
                dist[a] = float('inf')
        dist[None] = float('inf')
        
        while queue:
            a = queue.popleft()
            if dist[a] < dist[None]:
                for b, edge_idx in graph[a]:
                    # b is matched to some a_prime
                    if b in match_right:
                        a_prime = match_right[b]
                        if dist[a_prime] == float('inf'):
                            dist[a_prime] = dist[a] + 1
                            queue.append(a_prime)
                    else:
                        # Found an augmenting path
                        dist[None] = dist[a] + 1
        return dist[None] != float('inf')
    
    def dfs(a):
        if a is None:
            return True
            
        for b, edge_idx in graph[a]:
            if b in match_right:
                a_prime = match_right[b]
                if dist.get(a_prime, float('inf')) == dist[a] + 1:
                    if dfs(a_prime):
                        match_left[a] = edge_idx
                        match_right[b] = a
                        return True
            else:
                # b is free
                match_left[a] = edge_idx
                match_right[b] = a
                return True
        dist[a] = float('inf')
        return False
    
    # Main matching loop
    matching = 0
    while bfs():
        for a in range(1, q + 1):
            if a not in match_left:
                if dfs(a):
                    matching += 1
    
    if matching < q:
        print("NO")
        return
    
    print("YES")
    for a in range(1, q + 1):
        edge_idx = match_left[a]
        print(edges[edge_idx][0], edges[edge_idx][1])

solve()
