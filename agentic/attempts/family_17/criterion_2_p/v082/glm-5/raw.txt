import sys

def solve():
    input = sys.stdin.read().split()
    if not input:
        return
    it = iter(input)
    try:
        q = int(next(it))
        t = int(next(it))
    except StopIteration:
        return

    # Adjacency lists: cores 1..q -> list of towers, towers q+1..2q -> list of cores
    adj_core = [[] for _ in range(q + 1)]
    adj_tower = [[] for _ in range(q + 1)]  # 1-indexed for tower IDs relative to q
    
    # Store original edges to retrieve solution later
    edges = []
    for _ in range(t):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj_core[u].append(v)
        # Tower v maps to index v - q (1 to q)
        adj_tower[v - q].append(u)

    # Check Hall's condition
    # For each core subset S, |N(S)| >= |S|
    # We use a greedy basis-style check for minimality
    basis = [0] * (q + 1)  # basis[k] is a tower mask with k-th lowest set bit at position k
    
    possible = True
    for u in range(1, q + 1):
        # Compute adjacency bitmask for core u
        mask = 0
        # Sort neighbors to ensure deterministic ordering for bit positioning (not strictly necessary for correctness)
        for v in adj_core[u]:
            mask |= (1 << (v - q))
            if mask.bit_length() > q: # Should not happen, but safety
                pass
        
        # Reduce mask with current basis
        current = mask
        # Traverse bits from highest to lowest
        b_len = current.bit_length()
        for k in range(b_len, 0, -1):
            if (current >> (k - 1)) & 1:
                if basis[k]:
                    current ^= basis[k]
                else:
                    # Found a new basis vector
                    basis[k] = current
                    break
        else:
            # current became 0, so u's neighbors are a subset of previous cores' neighbors
            # This implies Hall's violation
            possible = False
            break

    if not possible:
        print("NO")
        return

    # Hall's condition satisfied, a perfect matching exists.
    # Construct matching using bipartite matching algorithm (DFS augmenting paths)
    # match_core[u] = v, match_tower[v-q] = u
    match_core = [0] * (q + 1)
    match_tower = [0] * (q + 1)
    
    def dfs(u, visited):
        for v in adj_core[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match_tower[v - q] == 0 or dfs(match_tower[v - q], visited):
                match_core[u] = v
                match_tower[v - q] = u
                return True
        return False

    for u in range(1, q + 1):
        visited = [0] * (2 * q + 1)
        if not dfs(u, visited):
            # Should not happen given Hall's check passed, but for safety
            print("NO")
            return

    # Output solution
    # To conform to example output style (sorted by core ID), matching is constructed in order.
    # We can output directly.
    print("YES")
    for u in range(1, q + 1):
        print(f"{u} {match_core[u]}")

if __name__ == '__main__':
    solve()
