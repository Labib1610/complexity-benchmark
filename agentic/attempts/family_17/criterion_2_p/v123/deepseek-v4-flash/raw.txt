We can model this as a matching problem.  
We have lower assets `1..q` and higher assets `q+1..2q`.  
We need to select exactly `q` edges such that:

- Every asset appears exactly once.
- Sum of lower IDs = `q(q+1)/2`.

Since each lower asset must be matched exactly once, the sum condition is automatically satisfied if we match all lower assets exactly once.  
So the problem reduces to: **Is there a perfect matching between lower and higher assets using only given edges?**

We can solve this with a maximum bipartite matching algorithm (Hopcroft–Karp) because `q` can be up to 100,000 and `t` up to 400,000.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q + 1)]
for _ in range(t):
    a, b = map(int, input().split())
    adj[a].append(b - q)  # store higher asset as 1..q

# Hopcroft–Karp
pair_u = [0] * (q + 1)
pair_v = [0] * (q + 1)
dist = [0] * (q + 1)
INF = 10**9

def bfs():
    dq = deque()
    for u in range(1, q + 1):
        if pair_u[u] == 0:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = INF
    dist[0] = INF
    while dq:
        u = dq.popleft()
        if dist[u] < dist[0]:
            for v in adj[u]:
                if dist[pair_v[v]] == INF:
                    dist[pair_v[v]] = dist[u] + 1
                    dq.append(pair_v[v])
    return dist[0] != INF

def dfs(u):
    if u != 0:
        for v in adj[u]:
            if dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v]):
                pair_v[v] = u
                pair_u[u] = v
                return True
        dist[u] = INF
        return False
    return True

matching = 0
while bfs():
    for u in range(1, q + 1):
        if pair_u[u] == 0 and dfs(u):
            matching += 1

if matching == q:
    print("YES")
    for u in range(1, q + 1):
        print(u, pair_u[u] + q)
else:
    print("NO")
```
