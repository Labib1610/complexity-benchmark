import sys

# Increase recursion depth for DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    # We need to find a perfect matching in a bipartite graph.
    # Left side: vocalists 1..q
    # Right side: instrumentalists q+1..2q
    # We map instrumentalists to 1..q for array indexing purposes.
    
    adj = [[] for _ in range(q + 1)]
    
    for _ in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        # v is in q+1..2q, map to 1..q
        adj[u].append(v - q)

    # matchR[v] stores the vocalist u matched to instrumentalist v (1-indexed)
    matchR = [0] * (q + 1)
    
    # Standard Hopcroft-Karp Algorithm
    
    # Pair_U[u] = v if matched
    # Pair_V[v] = u if matched
    # We can use matchR as Pair_V, and maintain Pair_U separately or just check matchR.
    # Let's stick to standard variable names for clarity.
    pair_U = [0] * (q + 1)
    pair_V = [0] * (q + 1)
    dist = [0] * (q + 1)
    
    from collections import deque

    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_V[v] == 0:
                        # Unmatched vertex on the right side
                        if dist[0] == float('inf'):
                            dist[0] = dist[u] + 1
                    else:
                        # Matched, check the matched vertex on left
                        if dist[pair_V[v]] == float('inf'):
                            dist[pair_V[v]] = dist[u] + 1
                            queue.append(pair_V[v])
        return dist[0] != float('inf')

    def dfs(u):
        for v in adj[u]:
            if pair_V[v] == 0 or (dist[pair_V[v]] == dist[u] + 1 and dfs(pair_V[v])):
                pair_U[u] = v
                pair_V[v] = u
                return True
        dist[u] = float('inf')
        return False

    matching = 0
    for u in range(1, q + 1):
        # Initial greedy matching (optional but can speed up)
        if pair_U[u] == 0:
            for v in adj[u]:
                if pair_V[v] == 0:
                    pair_U[u] = v
                    pair_V[v] = u
                    matching += 1
                    break
    
    while bfs():
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                if dfs(u):
                    matching += 1

    if matching == q:
        sys.stdout.write("YES\n")
        # Output the pairs
        # We need to output u and v (original IDs)
        # pair_U[u] is the mapped instrumentalist index (1..q)
        # Original ID is pair_U[u] + q
        results = []
        for u in range(1, q + 1):
            results.append(f"{u} {pair_U[u] + q}")
        sys.stdout.write("\n".join(results) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
