We need to find a perfect matching between the left half (1..q) and right half (q+1..2q) given that edges only exist between these halves.

This is a bipartite matching problem. Since q can be up to 100,000 and t up to 400,000, we use Hopcroft–Karp algorithm for maximum bipartite matching to run in O(E√V).

If the maximum matching size equals q, we output YES with edges. Otherwise NO.

Here's the solution:

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    INF = float('inf')
    pair_u = [-1] * (n + 1)
    pair_v = [-1] * (m + 1)
    dist = [0] * (n + 1)

    def bfs():
        dq = deque()
        for u in range(1, n + 1):
            if pair_u[u] == -1:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        found = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu == -1:
                    found = True
                elif dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, n + 1):
            if pair_u[u] == -1:
                if dfs(u):
                    matching += 1
    return matching, pair_u, pair_v

def main():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q + 1)]
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b - q)  # map right side to 1..q

    matching, pair_u, _ = hopcroft_karp(q, q, adj)
    if matching != q:
        print("NO")
    else:
        print("YES")
        for u in range(1, q + 1):
            v = pair_u[u] + q
            print(u, v)

if __name__ == "__main__":
    main()
```
