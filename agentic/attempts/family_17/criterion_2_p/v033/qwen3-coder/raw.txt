```python
import sys
from collections import defaultdict, deque

def solve():
    q, t = map(int, sys.stdin.readline().split())
    
    # Build adjacency list for bipartite graph
    # Controllers: 1 to q
    # Platforms: q+1 to 2*q
    adj = defaultdict(list)
    degree = [0] * (2 * q + 1)
    
    edges = []
    for _ in range(t):
        a, b = map(int, sys.stdin.readline().split())
        adj[a].append(b)
        adj[b].append(a)
        degree[a] += 1
        degree[b] += 1
        edges.append((a, b))
    
    # Check if we can find a perfect matching
    # This is a bipartite graph matching problem
    
    # Use Hopcroft-Karp algorithm implementation
    # But let's use a simpler approach with DFS forHungarian algorithm / augmenting paths
    
    match_controller = [0] * (q + 1)  # match_controller[c] = matched platform
    match_platform = [0] * (q + 1)    # match_platform[p-q] = matched controller
    
    def dfs(u, visited):
        for v in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            # If platform v is unmatched or we can find augmenting path
            if match_platform[v - q] == 0 or dfs(match_platform[v - q], visited):
                match_controller[u] = v
                match_platform[v - q] = u
                return True
        return False
    
    matching = 0
    for controller in range(1, q + 1):
        visited = [False] * (2 * q + 1)
        if dfs(controller, visited):
            matching += 1
    
    if matching != q:
        print("NO")
        return
    
    print("YES")
    for i in range(1, q + 1):
        print(i, match_controller[i])

solve()
```
