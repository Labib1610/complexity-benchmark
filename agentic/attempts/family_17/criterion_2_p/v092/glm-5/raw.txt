import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    q = int(next(iterator))
    t = int(next(iterator))
    
    adj = [[] for _ in range(2 * q + 1)]
    edges = []
    for _ in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    match = [0] * (2 * q + 1)
    dist = [0] * (2 * q + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if match[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = 0
        dist[0] = -1
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if match[v] == 0:
                        dist[0] = dist[u] + 1
                    elif dist[match[v]] == 0:
                        dist[match[v]] = dist[u] + 1
                        queue.append(match[v])
        return dist[0] != -1
    
    def dfs(u):
        for v in adj[u]:
            if match[v] == 0 or (dist[match[v]] == dist[u] + 1 and dfs(match[v])):
                match[u] = v
                match[v] = u
                return True
        dist[u] = 0
        return False
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if match[u] == 0 and dfs(u):
                matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match[u])
    else:
        print("NO")

if __name__ == "__main__":
    solve()
