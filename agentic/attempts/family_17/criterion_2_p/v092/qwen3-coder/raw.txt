from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    # Left side: drones 1 to q (lasers)
    # Right side: drones q+1 to 2q (retroreflectors)
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using BFS-based augmenting paths
    # This is essentially the Hopcroft-Karp algorithm simplified for our needs
    
    # mate[u] = v means u is matched to v
    mate = {}
    dist = {}
    
    def bfs():
        queue = deque()
        dist.clear()
        
        # Initialize distances for free nodes on left side
        for u in range(1, q + 1):
            if u not in mate:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        
        dist[None] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[None]:
                for v in graph[u]:
                    # v is matched to mate[v] (or None if unmatched)
                    mv = mate.get(v)
                    if dist[mv] == float('inf'):
                        dist[mv] = dist[u] + 1
                        queue.append(mv)
        
        return dist[None] != float('inf')
    
    def dfs(u):
        if u is not None:
            for v in graph[u]:
                mv = mate.get(v)
                if dist[mv] == dist[u] + 1 and dfs(mv):
                    mate[v] = u
                    mate[u] = v
                    return True
            dist[u] = float('inf')
            return False
        return True
    
    # Main matching loop
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if u not in mate and dfs(u):
                matching += 1
    
    if matching < q:
        print("NO")
        return
    
    print("YES")
    # Output the matching
    for u in range(1, q + 1):
        v = mate[u]
        print(u, v)

solve()
