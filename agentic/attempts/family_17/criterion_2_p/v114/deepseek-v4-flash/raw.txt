 We can model this as a bipartite matching problem between processors and memory banks.  
Since the graph is large (up to 400,000 edges), we need an efficient algorithm — Hopcroft–Karp is suitable.

We’ll:
1. Build adjacency lists for processors (left side) to memory banks (right side).
2. Run Hopcroft–Karp to find maximum matching.
3. If matching size == q, output YES and the matched pairs; otherwise NO.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q + 1)]

for _ in range(t):
    a, b = map(int, input().split())
    adj[a].append(b - q)  # store memory bank as 1..q

# Hopcroft-Karp
INF = 10**9
pairU = [0] * (q + 1)
pairV = [0] * (q + 1)
dist = [0] * (q + 1)

def bfs():
    dq = deque()
    for u in range(1, q + 1):
        if pairU[u] == 0:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = INF
    dist[0] = INF
    while dq:
        u = dq.popleft()
        if dist[u] < dist[0]:
            for v in adj[u]:
                if dist[pairV[v]] == INF:
                    dist[pairV[v]] = dist[u] + 1
                    dq.append(pairV[v])
    return dist[0] != INF

def dfs(u):
    if u != 0:
        for v in adj[u]:
            if dist[pairV[v]] == dist[u] + 1:
                if dfs(pairV[v]):
                    pairV[v] = u
                    pairU[u] = v
                    return True
        dist[u] = INF
        return False
    return True

matching = 0
while bfs():
    for u in range(1, q + 1):
        if pairU[u] == 0 and dfs(u):
            matching += 1

if matching == q:
    print("YES")
    for u in range(1, q + 1):
        print(u, pairU[u] + q)
else:
    print("NO")
```
