from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    if t < q:
        print("NO")
        return
        
    # Build graph for bipartite matching
    # Processors: 1 to q
    # Memory banks: q+1 to 2*q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Check if each processor and memory bank has at least one connection
    for i in range(1, q + 1):
        if not graph[i]:
            print("NO")
            return
    
    # Count degree of each memory bank
    mem_degree = defaultdict(int)
    for a, b in edges:
        mem_degree[b] += 1
    
    for i in range(q + 1, 2 * q + 1):
        if mem_degree[i] == 0:
            print("NO")
            return
    
    # Find maximum bipartite matching using Ford-Fulkerson with BFS
    def bfs_match(match_proc_to_mem, match_mem_to_proc):
        # Find augmenting paths
        visited = set()
        queue = deque()
        parent = {}
        
        # Start BFS from unmatched processors
        for proc in range(1, q + 1):
            if match_proc_to_mem[proc] == -1:
                queue.append(proc)
                visited.add(proc)
                parent[proc] = None
        
        found_path = False
        sink = -1
        
        while queue and not found_path:
            node = queue.popleft()
            
            # If node is a processor
            if 1 <= node <= q:
                for mem in graph[node]:
                    if mem not in visited:
                        visited.add(mem)
                        parent[mem] = node
                        if match_mem_to_proc[mem] == -1:
                            # Found an augmenting path
                            found_path = True
                            sink = mem
                            break
                        else:
                            next_proc = match_mem_to_proc[mem]
                            queue.append(next_proc)
                            visited.add(next_proc)
                            parent[next_proc] = mem
            else:  # node is a memory bank
                proc = match_mem_to_proc[node]
                if proc != -1 and proc not in visited:
                    visited.add(proc)
                    parent[proc] = node
                    queue.append(proc)
        
        if not found_path:
            return False
            
        # Augment the path
        current = sink
        while parent[current] is not None:
            prev = parent[current]
            if 1 <= prev <= q:  # prev is processor
                match_proc_to_mem[prev] = current
                match_mem_to_proc[current] = prev
            current = prev
            
        return True
    
    # Initialize matching
    match_proc_to_mem = {i: -1 for i in range(1, q + 1)}
    match_mem_to_proc = {i: -1 for i in range(q + 1, 2 * q + 1)}
    
    # Keep finding augmenting paths until no more exist
    while bfs_match(match_proc_to_mem, match_mem_to_proc):
        pass
    
    # Check if we have a perfect matching
    matched_count = sum(1 for i in range(1, q + 1) if match_proc_to_mem[i] != -1)
    
    if matched_count != q:
        print("NO")
        return
    
    print("YES")
    for proc in range(1, q + 1):
        mem = match_proc_to_mem[proc]
        print(proc, mem)

solve()
