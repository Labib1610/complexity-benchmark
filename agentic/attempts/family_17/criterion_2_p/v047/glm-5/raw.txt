import sys

# Set recursion limit higher for safety
sys.setrecursionlimit(3000000)

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    # We will build a bipartite graph between sets A={1..q} and B={q+1..2q}
    # Nodes 1..q are on the left, nodes q+1..2q are on the right.
    # Maximum node label is 2q.
    
    # Capacity arrays for Dinic's algorithm
    # We need to handle up to node 2q.
    # We'll use 1-based indexing internally for simplicity.
    
    # Head array for adjacency list
    # Size: 2q + 5 (plus extra for source/sink if mapped differently, but here sink is 2q+2)
    
    # Actually, let's use 0-indexed arrays for internal representation for speed?
    # Or map source to 0, left nodes 1..q, right nodes q+1..2q, sink to 2q+1?
    # Let's map: source = 2q+1, sink = 2q+2.
    # Total nodes: 2q + 3.
    
    N_nodes = 2 * q + 3
    head = [-1] * N_nodes
    to = []
    nxt = []
    cap = []
    
    def add_edge(u, v, c):
        # Forward edge
        to.append(v)
        nxt.append(head[u])
        cap.append(c)
        head[u] = len(to) - 1
        
        # Backward edge
        to.append(u)
        nxt.append(head[v])
        cap.append(0)
        head[v] = len(to) - 1

    source = 2 * q + 1
    sink = 2 * q + 2
    
    # Build source to controllers
    for i in range(1, q + 1):
        add_edge(source, i, 1)
    
    # Build platforms to sink
    for j in range(q + 1, 2 * q + 1):
        add_edge(j, sink, 1)
        
    # Read pairings
    for _ in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        add_edge(u, v, 1)
        
    # Dinic's Algorithm
    level = [-1] * N_nodes
    ptr = [0] * N_nodes
    
    def bfs():
        for i in range(1, 2 * q + 3):
            level[i] = -1
        queue = []
        queue.append(source)
        level[source] = 0
        idx = 0
        while idx < len(queue):
            u = queue[idx]
            idx += 1
            edge_idx = head[u]
            while edge_idx != -1:
                v = to[edge_idx]
                if cap[edge_idx] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    queue.append(v)
                    if v == sink:
                        return True
                edge_idx = nxt[edge_idx]
        return level[sink] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == sink:
            return pushed
        for i in range(ptr[u], N_nodes): # The loop should use ptr[u] to iterate
            # Manual iteration using ptr
            pass 
        
        # Correct DFS with pointer
        # We need to reset ptr before Dinic loops start usually, but we do it in the flow loop
        
        while ptr[u] != -1: # Iterating edges from ptr[u]
            edge_idx = ptr[u]
            v = to[edge_idx]
            if level[v] != level[u] + 1 or cap[edge_idx] == 0:
                ptr[u] = nxt[edge_idx]
                continue
            
            tr = dfs(v, min(pushed, cap[edge_idx]))
            if tr == 0:
                ptr[u] = nxt[edge_idx]
                continue
            
            cap[edge_idx] -= tr
            cap[edge_idx ^ 1] += tr
            return tr
        return 0

    flow = 0
    while bfs():
        # Reset pointers for gradient descent
        for i in range(1, 2 * q + 3):
            ptr[i] = head[i]
        
        while True:
            pushed = dfs(source, float('inf'))
            if pushed == 0:
                break
            flow += pushed
    
    if flow == q:
        print("YES")
        # Reconstruct answer by checking edges from controllers
        for u in range(1, q + 1):
            edge_idx = head[u]
            while edge_idx != -1:
                # Check if this edge was saturated and is pointing to a platform
                v = to[edge_idx]
                if v >= q + 1 and v <= 2 * q and cap[edge_idx] == 0:
                    print(u, v)
                    break
                edge_idx = nxt[edge_idx]
    else:
        print("NO")

if __name__ == '__main__':
    solve()
