from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    if t < q:
        print("NO")
        return
    
    # Create adjacency list for bipartite graph
    # Left side: 1 to q, Right side: q+1 to 2*q
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b)
        edges.append((a, b))
    
    # Try to find a maximum matching using BFS (Hopcroft-Karp like approach)
    # But we can use a simpler approach since we need a perfect matching
    
    # Use greedy approach with backtracking simulation via max flow concept
    # Actually, let's implement a simple bipartite matching algorithm
    
    match_right = {}  # match_right[b] = a means b is matched to a
    match_left = {}   # match_left[a] = b means a is matched to b
    
    def bfs():
        # Find augmenting paths
        visited = set()
        queue = deque()
        
        # Initialize queue with unmatched left nodes
        for a in range(1, q + 1):
            if a not in match_left:
                queue.append(a)
                visited.add(a)
            else:
                visited.add(a)
        
        parent = {}
        found = False
        target_b = None
        
        while queue and not found:
            node = queue.popleft()
            
            if node <= q:  # It's a left node
                for b in adj[node]:
                    if b not in visited:
                        visited.add(b)
                        parent[b] = node
                        if b not in match_right:
                            found = True
                            target_b = b
                            break
                        else:
                            next_a = match_right[b]
                            queue.append(next_a)
                            parent[next_a] = b
                            visited.add(next_a)
            else:  # It's a right node
                pass  # We only add left nodes to queue initially
                
        if not found:
            return False
            
        # Augment the path
        current = target_b
        while current in parent:
            prev = parent[current]
            if current <= q:  # current is left
                match_left[prev] = current
                match_right[current] = prev
            else:  # current is right
                match_left[prev] = current
                match_right[current] = prev
            current = prev if prev in parent else None
            
        return True
    
    # Alternative implementation using a more standard approach
    def dfs(u, visited):
        for v in adj[u]:
            if v in visited:
                continue
            visited.add(v)
            if v not in match_right or dfs(match_right[v], visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    # Main matching loop
    matching = 0
    for a in range(1, q + 1):
        visited = set()
        if dfs(a, visited):
            matching += 1
    
    if matching == q:
        print("YES")
        for a in range(1, q + 1):
            print(a, match_left[a])
    else:
        print("NO")

solve()
