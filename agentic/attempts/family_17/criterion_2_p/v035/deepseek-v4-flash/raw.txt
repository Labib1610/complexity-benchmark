We can model this as a bipartite graph matching problem:  
- Controllers are on the left (1..q), platforms on the right (q+1..2q).  
- Each allowed pair is an edge.  
- We need a perfect matching covering all vertices.

Since q can be up to 100,000 and edges up to 400,000, we need an efficient algorithm — Hopcroft–Karp is suitable.

```python
import sys
from collections import deque

def hopcroft_karp(n, m, adj):
    INF = 10**9
    pair_u = [-1] * n
    pair_v = [-1] * m
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - q - 1
        adj[a].append(b)

    matching, pair_u = hopcroft_karp(q, q, adj)
    if matching != q:
        sys.stdout.write("NO\n")
    else:
        out = ["YES"]
        for u in range(q):
            v = pair_u[u]
            out.append(f"{u+1} {v+q+1}")
        sys.stdout.write("\n".join(out) + "\n")

if __name__ == "__main__":
    main()
```
