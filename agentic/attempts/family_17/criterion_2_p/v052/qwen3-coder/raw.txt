from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Left side: 1 to q, Right side: q+1 to 2q
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b)
        edges.append((a, b))
    
    # Try to find a maximum matching using BFS (Hopcroft-Karp like approach)
    # But we can use a simpler approach: try to find a maximum matching with augmenting paths
    
    # match_right[b] = a if b is matched to a, 0 otherwise
    match_right = {}
    match_left = {}
    
    def bfs():
        # Find shortest augmenting paths
        queue = deque()
        dist = {}
        
        for a in range(1, q + 1):
            if a not in match_left:
                dist[a] = 0
                queue.append(a)
            else:
                dist[a] = float('inf')
        dist[None] = float('inf')
        
        while queue:
            a = queue.popleft()
            if dist[a] < dist[None]:
                for b in adj[a]:
                    if b not in match_right:
                        if dist[None] == float('inf'):
                            dist[None] = dist[a] + 1
                    else:
                        a_next = match_right[b]
                        if dist[a_next] == float('inf'):
                            dist[a_next] = dist[a] + 1
                            queue.append(a_next)
        return dist[None] != float('inf')
    
    def dfs(a):
        if a == 0:
            return True
        for b in adj[a]:
            if b in match_right:
                a_next = match_right[b]
            else:
                a_next = 0
            if dist.get(a_next, float('inf')) == dist.get(a, -1) + 1:
                if dfs(a_next):
                    match_right[b] = a
                    match_left[a] = b
                    return True
        dist[a] = float('inf')
        return False
    
    # Simplified approach using basic augmenting paths
    def augment():
        visited = set()
        def find_path(a):
            if a in visited:
                return False
            visited.add(a)
            for b in adj[a]:
                if b not in match_right:
                    match_right[b] = a
                    match_left[a] = b
                    return True
                elif find_path(match_right[b]):
                    match_right[b] = a
                    match_left[a] = b
                    return True
            return False
        
        for a in range(1, q + 1):
            if a not in match_left:
                visited.clear()
                if not find_path(a):
                    return False
        return True
    
    # Run the matching algorithm
    success = True
    for _ in range(q):
        if not augment():
            success = False
            break
    
    if success and len(match_left) == q:
        print("YES")
        for a in range(1, q + 1):
            print(a, match_left[a])
    else:
        print("NO")

solve()
