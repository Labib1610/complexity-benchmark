import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    
    # Build adjacency list for left side (1..q)
    # Right side nodes are (b - q) in 1..q
    adj = [[] for _ in range(q + 1)]
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1])
        idx += 2
        adj[a].append(b - q)
    
    # Hopcroft-Karp algorithm for bipartite matching
    pair_u = [0] * (q + 1)  # pair_u[u] = v matched to u (1-indexed)
    pair_v = [0] * (q + 1)  # pair_v[v] = u matched to v (1-indexed)
    dist = [0] * (q + 1)
    
    from collections import deque
    
    INF = 10**9
    
    def bfs():
        dq = deque()
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while dq:
            u = dq.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        dq.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        for v in adj[u]:
            if dist[pair_v[v]] == dist[u] + 1:
                if dfs(pair_v[v]):
                    pair_u[u] = v
                    pair_v[v] = u
                    return True
        dist[u] = INF
        return False
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        out = ["YES"]
        for u in range(1, q + 1):
            out.append(f"{u} {pair_u[u] + q}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

solve()
