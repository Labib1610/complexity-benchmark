```python
from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    # Left side: controllers 1 to q
    # Right side: platforms q+1 to 2q
    graph = defaultdict(list)
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
    
    # Try to find a maximum matching using Hopcroft-Karp algorithm
    # But we'll use a simpler approach with DFS for correctness
    
    # partner[i] = j means controller i is matched to platform j
    # partner[j] = i means platform j is matched to controller i
    partner = {}
    
    def dfs(u, visited):
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If v is unmatched or we can find augmenting path
            if v not in partner or dfs(partner[v], visited):
                partner[u] = v
                partner[v] = u
                return True
        return False
    
    # Try to match all controllers
    matched = 0
    for u in range(1, q + 1):
        if dfs(u, set()):
            matched += 1
    
    if matched == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, partner[u])
    else:
        print("NO")

solve()
```
