We need to find a perfect matching between acids (1..q) and bases (q+1..2q) such that:

- Each edge is in the approved list.
- Sum of labels ≤ 3q.
- Absolute difference ≥ 1 (always true since a ≤ q < b).

So the only real constraints are the approved edges and the sum constraint.

We can model this as a bipartite matching problem.  
Since q can be up to 100,000 and t up to 400,000, we need an efficient algorithm — Hopcroft–Karp is suitable.

We also need to filter edges where a + b > 3q.

Let’s implement Hopcroft–Karp.

```python
import sys
from collections import deque

def hopcroft_karp(n, m, adj):
    # adj: list of lists for left side (size n), each entry is right node index (0-based)
    INF = 10**9
    pairU = [-1] * n
    pairV = [-1] * m
    dist = [0] * n

    def bfs():
        dq = deque()
        for u in range(n):
            if pairU[u] == -1:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        found = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                pu = pairV[v]
                if pu == -1:
                    found = True
                elif dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pairV[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pairU[u] == -1 and dfs(u):
                matching += 1
    return matching, pairU

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    q = int(next(it))
    t = int(next(it))
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - 1  # 0-based for right side
        if (a + 1) + (b + 1) <= 3 * q:
            adj[a].append(b)
    # Remove duplicates
    for i in range(q):
        adj[i] = list(set(adj[i]))
    matching, pairU = hopcroft_karp(q, q, adj)
    if matching != q:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n")
        for u in range(q):
            v = pairU[u]
            sys.stdout.write(f"{u+1} {v+1}\n")

if __name__ == "__main__":
    main()
```
