from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Frontend nodes: 1 to q
    # Backend nodes: q+1 to 2q
    graph = defaultdict(list)
    edges = []
    
    for i in range(t):
        a, b = map(int, input().split())
        graph[a].append((b, i))
        graph[b].append((a, i))
        edges.append((a, b))
    
    # Try to find a perfect matching using BFS-based augmenting paths
    # This is essentially the Hopcroft-Karp algorithm simplified
    
    # mate[i] = j means node i is matched to node j
    mate = [0] * (2*q + 1)
    dist = [0] * (2*q + 1)
    
    def bfs():
        queue = deque()
        for i in range(1, q+1):
            if mate[i] == 0:
                dist[i] = 0
                queue.append(i)
            else:
                dist[i] = float('inf')
        dist[0] = float('inf')
        
        while queue:
            x = queue.popleft()
            if dist[x] < dist[0]:
                for y, _ in graph[x]:
                    if dist[mate[y]] == float('inf'):
                        dist[mate[y]] = dist[x] + 1
                        queue.append(mate[y])
        return dist[0] != float('inf')
    
    def dfs(x):
        if x != 0:
            for y, _ in graph[x]:
                if dist[mate[y]] == dist[x] + 1:
                    if dfs(mate[y]):
                        mate[y] = x
                        mate[x] = y
                        return True
            dist[x] = float('inf')
            return False
        return True
    
    # Find maximum matching
    matching = 0
    while bfs():
        for i in range(1, q+1):
            if mate[i] == 0:
                if dfs(i):
                    matching += 1
    
    if matching < q:
        print("NO")
        return
    
    print("YES")
    for i in range(1, q+1):
        print(i, mate[i])

solve()
