We need to match each frontend to a unique backend and vice versa, meaning this is essentially a **perfect matching** problem in a bipartite graph.  

We can solve it using maximum bipartite matching with the Hopcroft–Karp algorithm for efficiency, since the constraints are large.

```python
import sys
from collections import deque

input = sys.stdin.readline

q, t = map(int, input().split())
adj = [[] for _ in range(q + 1)]
for _ in range(t):
    a, b = map(int, input().split())
    adj[a].append(b - q)  # store backend index 1..q

# Hopcroft-Karp
pair_u = [0] * (q + 1)  # frontend matched to backend
pair_v = [0] * (q + 1)  # backend matched to frontend
dist = [0] * (q + 1)

INF = 10**9

def bfs():
    dq = deque()
    for u in range(1, q + 1):
        if pair_u[u] == 0:
            dist[u] = 0
            dq.append(u)
        else:
            dist[u] = INF
    dist[0] = INF
    while dq:
        u = dq.popleft()
        if dist[u] < dist[0]:
            for v in adj[u]:
                if dist[pair_v[v]] == INF:
                    dist[pair_v[v]] = dist[u] + 1
                    dq.append(pair_v[v])
    return dist[0] != INF

def dfs(u):
    if u != 0:
        for v in adj[u]:
            if dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v]):
                pair_v[v] = u
                pair_u[u] = v
                return True
        dist[u] = INF
        return False
    return True

matching = 0
while bfs():
    for u in range(1, q + 1):
        if pair_u[u] == 0 and dfs(u):
            matching += 1

if matching < q:
    print("NO")
else:
    print("YES")
    for u in range(1, q + 1):
        v = pair_u[u]
        print(u, v + q)
```
