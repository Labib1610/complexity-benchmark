from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        # Convert to 0-indexed
        u = a - 1  # transmitter 0 to q-1
        v = b - 1 - q  # receiver 0 to q-1
        graph[u].append(v)
        edges.append((u, v))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm
    # But we'll use a simpler approach with DFS for this case
    
    match_receiver = [-1] * q  # which transmitter is matched to receiver j
    match_transmitter = [-1] * q  # which receiver is matched to transmitter i
    
    def dfs(u, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            # If receiver v is unmatched or we can find augmenting path
            if match_receiver[v] == -1 or dfs(match_receiver[v], visited):
                match_receiver[v] = u
                match_transmitter[u] = v
                return True
        return False
    
    matching = 0
    for i in range(q):
        visited = [False] * q
        if dfs(i, visited):
            matching += 1
    
    if matching != q:
        print("NO")
        return
    
    print("YES")
    # Output the matching
    for i in range(q):
        # Convert back to 1-indexed
        transmitter = i + 1
        receiver = match_transmitter[i] + 1 + q
        print(transmitter, receiver)

solve()
