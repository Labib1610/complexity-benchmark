from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        # Convert to 0-indexed
        u = a - 1  # transmitter: 0 to q-1
        v = b - q - 1  # receiver: 0 to q-1
        graph[u].append(v)
        edges.append((u, v))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm
    # But we can use a simpler approach with DFS for bipartite matching
    
    match_receiver = [-1] * q  # which transmitter is matched to each receiver
    match_transmitter = [-1] * q  # which receiver is matched to each transmitter
    
    def dfs(u, visited):
        for v in graph[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            # If receiver v is unmatched or we can find augmenting path
            if match_receiver[v] == -1 or dfs(match_receiver[v], visited):
                match_receiver[v] = u
                match_transmitter[u] = v
                return True
        return False
    
    matching = 0
    for u in range(q):
        visited = [False] * q
        if dfs(u, visited):
            matching += 1
    
    if matching != q:
        print("NO")
        return
    
    print("YES")
    for i in range(q):
        # Convert back to 1-indexed
        transmitter = i + 1
        receiver = match_transmitter[i] + q + 1
        print(transmitter, receiver)

solve()
