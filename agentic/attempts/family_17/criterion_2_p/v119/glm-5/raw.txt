```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    
    # Build adjacency list
    # Zone A: 1..q, Zone B: q+1..2q
    # We'll use 0-indexed internally: A: 0..q-1, B: 0..q-1 (mapped from q+1..2q)
    # But let's keep original indexing for output
    
    adj = [[] for _ in range(q + 1)]  # 1-indexed for Zone A
    
    idx = 2
    edges = []
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1])
        idx += 2
        adj[a].append(b)
        edges.append((a, b))
    
    # Hopcroft-Karp algorithm for bipartite matching
    # Left side: Zone A (1..q), Right side: Zone B (q+1..2q)
    
    pair_u = [0] * (q + 1)  # pair_u[a] = b (matched to), 0 means unmatched
    pair_v = [0] * (q + 1)  # pair_v[b-q] = a, 0 means unmatched
    dist = [0] * (q + 1)
    
    from collections import deque
    
    INF = float('inf')
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    v_idx = v - q
                    if pair_v[v_idx] != 0:
                        if dist[pair_v[v_idx]] == INF:
                            dist[pair_v[v_idx]] = dist[u] + 1
                            queue.append(pair_v[v_idx])
                    else:
                        dist[0] = dist[u] + 1
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                v_idx = v - q
                if pair_v[v_idx] == 0 or (dist[pair_v[v_idx]] == dist[u] + 1 and dfs(pair_v[v_idx])):
                    pair_u[u] = v
                    pair_v[v_idx] = u
                    return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        out = []
        for u in range(1, q + 1):
            out.append(f"{u} {pair_u[u]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        print("NO")

solve()
```
