from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build bipartite graph
    # Controllers: 1 to q
    # Platforms: q+1 to 2*q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Use Hopcroft-Karp or simpler Ford-Fulkerson for max matching
    # Since q can be up to 100000, we need an efficient approach
    # Let's use a simple approach with adjacency list and BFS for augmenting paths
    
    # Matching from controller to platform
    match_controller = {}
    match_platform = {}
    
    def bfs():
        # Find shortest augmenting path
        queue = deque()
        dist = {}
        
        for c in range(1, q + 1):
            if c not in match_controller:
                dist[c] = 0
                queue.append(c)
            else:
                dist[c] = float('inf')
        
        dist[None] = float('inf')
        
        while queue:
            c = queue.popleft()
            
            if dist[c] < dist[None]:
                for p in graph[c]:
                    if p not in match_platform:
                        if dist[None] == float('inf'):
                            dist[None] = dist[c] + 1
                    else:
                        nc = match_platform[p]
                        if dist.get(nc, float('inf')) == float('inf'):
                            dist[nc] = dist[c] + 1
                            queue.append(nc)
        
        return dist[None] != float('inf')
    
    def dfs(c):
        if c is None:
            return True
            
        for p in graph[c]:
            if p not in match_platform or \
               (dist.get(match_platform[p], float('inf')) == dist[c] + 1 and dfs(match_platform[p])):
                match_controller[c] = p
                match_platform[p] = c
                return True
                
        dist[c] = float('inf')
        return False
    
    # Hopcroft-Karp main loop
    matching = 0
    while bfs():
        for c in range(1, q + 1):
            if c not in match_controller:
                if dfs(c):
                    matching += 1
    
    if matching == q:
        print("YES")
        for c in range(1, q + 1):
            print(c, match_controller[c])
    else:
        print("NO")

solve()
