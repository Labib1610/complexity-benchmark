from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Left side: modules 1 to q (pins)
    # Right side: modules q+1 to 2q (slots)
    adj = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm
    # But we'll use a simpler approach with DFS for this case
    
    # match_right[b] = a means module b (slot) is matched to module a (pin)
    match_right = {}
    match_left = {}
    
    def dfs(u, visited):
        for v in adj[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If v is unmatched or we can find augmenting path
            if v not in match_right or dfs(match_right[v], visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    # Try to match all left vertices (pins)
    matched = 0
    for u in range(1, q + 1):
        if dfs(u, set()):
            matched += 1
    
    if matched == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, match_left[u])
    else:
        print("NO")

solve()
