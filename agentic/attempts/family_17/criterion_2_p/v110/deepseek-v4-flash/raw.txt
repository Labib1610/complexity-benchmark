We need to find a perfect matching in a bipartite graph where left side is modules 1..q and right side is modules q+1..2q.  
The graph can be large (up to 400k edges), so we need an efficient algorithm.  

We can use Hopcroft-Karp algorithm for maximum bipartite matching.  
If the matching size equals q, we output YES and the matched pairs.  

We must be careful with recursion depth and memory.  
We'll implement iterative BFS/DFS for Hopcroft-Karp.  

```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    
    adj = [[] for _ in range(q)]
    for _ in range(t):
        a = int(next(it)) - 1
        b = int(next(it)) - q - 1
        adj[a].append(b)
    
    # Hopcroft-Karp
    INF = 10**9
    pairU = [-1] * q
    pairV = [-1] * q
    dist = [0] * q
    
    def bfs():
        dq = deque()
        for u in range(q):
            if pairU[u] == -1:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        found = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                pu = pairV[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
                elif pu == -1:
                    found = True
        return found
    
    def dfs(u):
        for v in adj[u]:
            pu = pairV[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False
    
    matching = 0
    while bfs():
        for u in range(q):
            if pairU[u] == -1:
                if dfs(u):
                    matching += 1
        if matching == q:
            break
    
    if matching < q:
        sys.stdout.write("NO\n")
    else:
        out = ["YES\n"]
        for u in range(q):
            v = pairU[u]
            out.append(f"{u+1} {v+q+1}\n")
        sys.stdout.write(''.join(out))

if __name__ == "__main__":
    solve()
```
