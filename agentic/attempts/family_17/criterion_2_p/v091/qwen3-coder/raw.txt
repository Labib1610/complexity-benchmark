from collections import defaultdict, deque

def find_maximum_matching(graph, n):
    """
    Find maximum matching in a bipartite graph using Hopcroft-Karp algorithm
    """
    # Create adjacency list for left and right sets
    adj = defaultdict(list)
    for u, v in graph:
        adj[u].append(v)
    
    # Initialize
    pair_u = [0] * (n + 1)  # pair_u[u] = v means u is matched to v
    pair_v = [0] * (n + 1)  # pair_v[v] = u means v is matched to u
    dist = [0] * (n + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, n + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == float('inf'):
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != float('inf')
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = float('inf')
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, n + 1):
            if pair_u[u] == 0 and dfs(u):
                matching += 1
    
    return matching, pair_u

def solve():
    q, t = map(int, input().split())
    
    if t < q:
        print("NO")
        return
    
    edges = []
    for _ in range(t):
        a, b = map(int, input().split())
        edges.append((a, b))
    
    # Find maximum matching
    matching_size, pair_u = find_maximum_matching(edges, q)
    
    if matching_size != q:
        print("NO")
    else:
        print("YES")
        for i in range(1, q + 1):
            print(i, pair_u[i])

solve()
