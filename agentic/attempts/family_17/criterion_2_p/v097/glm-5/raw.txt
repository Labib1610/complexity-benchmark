import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    
    adj_primary = [[] for _ in range(q + 1)]
    adj_secondary = [[] for _ in range(q + 1)]
    
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        adj_primary[a].append(b - q)
        adj_secondary[b - q].append(a)
    
    # Hopcroft-Karp algorithm
    pair_U = [0] * (q + 1)
    pair_V = [0] * (q + 1)
    dist = [0] * (q + 1)
    NIL = 0
    
    def bfs():
        queue = deque()
        for u in range(1, q + 1):
            if pair_U[u] == NIL:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[NIL] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[NIL]:
                for v in adj_primary[u]:
                    if dist[pair_V[v]] == float('inf'):
                        dist[pair_V[v]] = dist[u] + 1
                        queue.append(pair_V[v])
        return dist[NIL] != float('inf')
    
    def dfs(u):
        if u != NIL:
            for v in adj_primary[u]:
                if dist[pair_V[v]] == dist[u] + 1:
                    if dfs(pair_V[v]):
                        pair_V[v] = u
                        pair_U[u] = v
                        return True
            dist[u] = float('inf')
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, q + 1):
            if pair_U[u] == NIL:
                if dfs(u):
                    matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(f"{u} {pair_U[u] + q}")
    else:
        print("NO")

if __name__ == "__main__":
    solve()
