import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    q = int(next(it))
    t = int(next(it))
    
    adj = [[] for _ in range(q + 1)]
    edges = []
    for _ in range(t):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        edges.append((a, b))
    
    # U = 1..q, V = q+1..2q
    # We use 0-based indexing for arrays internally where needed, but match[u] stores v.
    match_u = [0] * (q + 1)
    match_v = [0] * (2 * q + 1)
    dist = [0] * (q + 1)
    
    def bfs():
        dq = deque()
        for u in range(1, q + 1):
            if match_u[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = -1
        found_augmenting_path = False
        while dq:
            u = dq.popleft()
            for v in adj[u]:
                u2 = match_v[v]
                if u2 != 0 and dist[u2] == -1:
                    dist[u2] = dist[u] + 1
                    dq.append(u2)
                elif u2 == 0:
                    found_augmenting_path = True
        return found_augmenting_path

    def dfs(u):
        for v in adj[u]:
            u2 = match_v[v]
            if u2 == 0 or (dist[u2] == dist[u] + 1 and dfs(u2)):
                match_u[u] = v
                match_v[v] = u
                return True
        dist[u] = -1
        return False

    matching_size = 0
    while bfs():
        for u in range(1, q + 1):
            if match_u[u] == 0 and dfs(u):
                matching_size += 1
    
    if matching_size != q:
        print("NO")
    else:
        print("YES")
        # Output the q edges in the matching
        # match_u[u] contains the matched v
        for u in range(1, q + 1):
            print(u, match_u[u])

if __name__ == "__main__":
    solve()
