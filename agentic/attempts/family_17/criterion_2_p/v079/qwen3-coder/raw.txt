from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build graph for bipartite matching
    # Emitters: 1 to q
    # Receivers: q+1 to 2q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using BFS-based algorithm (Hopcroft-Karp like approach)
    # But we'll use a simpler approach with DFS since we just need any perfect matching
    
    # For each emitter, try to find an augmenting path
    match_receiver = {}  # receiver -> emitter
    match_emitter = {}   # emitter -> receiver
    
    def dfs(emitter, visited):
        for receiver in graph[emitter]:
            if receiver in visited:
                continue
            visited.add(receiver)
            
            # If receiver is unmatched or we can find augmenting path
            if receiver not in match_receiver or dfs(match_receiver[receiver], visited):
                match_receiver[receiver] = emitter
                match_emitter[emitter] = receiver
                return True
        return False
    
    # Try to match each emitter
    matching = 0
    for emitter in range(1, q + 1):
        if dfs(emitter, set()):
            matching += 1
    
    if matching == q:
        print("YES")
        for emitter in range(1, q + 1):
            print(emitter, match_emitter[emitter])
    else:
        print("NO")

solve()
