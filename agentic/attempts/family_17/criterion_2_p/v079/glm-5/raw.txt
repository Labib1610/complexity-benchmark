```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        q = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    N = 2 * q
    adj = [[] for _ in range(N + 1)]
    
    for _ in range(t):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    match = [0] * (N + 1)
    
    for u in range(1, q + 1):
        if match[u] != 0:
            continue
            
        visited = [False] * (N + 1)
        parent = [0] * (N + 1)
        queue = deque()
        found = -1
        
        queue.append(u)
        visited[u] = True
        
        while queue and found == -1:
            curr = queue.popleft()
            
            if curr <= q:
                for v in adj[curr]:
                    if not visited[v]:
                        visited[v] = True
                        parent[v] = curr
                        if match[v] == 0:
                            found = v
                            break
                        queue.append(v)
            else:
                if match[curr] != 0:
                    next_u = match[curr]
                    if not visited[next_u]:
                        visited[next_u] = True
                        parent[next_u] = curr
                        queue.append(next_u)
        
        if found == -1:
            print("NO")
            return
            
        v = found
        while v != 0:
            u = parent[v]
            prev_v = match[u]
            match[u] = v
            match[v] = u
            v = prev_v

    print("YES")
    for u in range(1, q + 1):
        print(u, match[u])

if __name__ == '__main__':
    solve()
```
