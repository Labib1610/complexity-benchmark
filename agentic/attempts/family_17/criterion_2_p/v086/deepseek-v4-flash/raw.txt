We can model this as a bipartite matching problem between bays (1…q) and panels (q+1…2q).  
Since q can be up to 100,000 and t up to 400,000, we need an efficient algorithm.

A simple DFS-based augmenting path algorithm (Hopcroft–Karp) works in O(E√V).  
We’ll implement that.

```python
import sys
from collections import deque

input = sys.stdin.readline

def hopcroft_karp(n, m, adj):
    # n = left side size, m = right side size
    INF = 10**9
    pair_u = [-1] * (n + 1)
    pair_v = [-1] * (m + 1)
    dist = [0] * (n + 1)

    def bfs():
        q = deque()
        for u in range(1, n + 1):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, n + 1):
            if pair_u[u] == -1:
                if dfs(u):
                    matching += 1
    return pair_u, matching

def solve():
    q, t = map(int, input().split())
    adj = [[] for _ in range(q + 1)]
    edges = []
    for _ in range(t):
        a, b = map(int, input().split())
        # left: a, right: b - q
        adj[a].append(b - q)
        edges.append((a, b))

    pair_u, matching = hopcroft_karp(q, q, adj)

    if matching != q:
        print("NO")
        return

    # pair_u[u] = v (panel index minus q)
    selected = []
    for u in range(1, q + 1):
        v = pair_u[u]
        selected.append((u, v + q))

    print("YES")
    for a, b in selected:
        print(a, b)

if __name__ == "__main__":
    solve()
```
