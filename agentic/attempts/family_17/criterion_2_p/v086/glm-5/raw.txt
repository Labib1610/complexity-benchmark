import sys

# Increase recursion limit for DFS in bipartite matching
sys.setrecursionlimit(10 ** 6)

def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        q = int(next(iterator))
        t_count = int(next(iterator))
    except StopIteration:
        return
    
    # Number of instrument bays: q (indices 1 to q)
    # Number of radiator panels: q (indices q+1 to 2q)
    
    # Read conduit edges
    # We will solve using Hopcroft-Karp algorithm for bipartite matching
    # Left set U: instrument bays 1..q
    # Right set V: radiator panels q+1..2q
    
    adj = [[] for _ in range(q + 1)]
    edges_raw = []
    
    for _ in range(t_count):
        try:
            a = int(next(iterator))
            b = int(next(iterator))
        except StopIteration:
            break
        
        # Remap right node b to 0-indexed for the right set: b - q - 1
        # But for matching arrays, it's easier to handle indices carefully.
        # Right set nodes are b (from q+1 to 2q).
        # Let's build adjacency list.
        
        # Edge from a to b
        adj[a].append(b)
        edges_raw.append((a, b))

    # Hopcroft-Karp Algorithm
    # pair_U[u] stores the matched node in V for u in U
    # pair_V[v] stores the matched node in U for v in V
    # We use dictionaries or arrays for pairs.
    
    # pair_U: size q+1 (1-based indexing for bay)
    # pair_V: size 2q+1 (matching the original node indices)
    
    pair_U = [0] * (q + 1)
    pair_V = [0] * (2 * q + 1)
    
    INF = float('inf')
    dist = [0] * (q + 1)
    
    from collections import deque
    
    def bfs():
        dq = deque()
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        
        dist[0] = INF # Sentinel 0 is not a real node
        
        while dq:
            u = dq.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_V[v] == 0:
                        dist[0] = dist[u] + 1
                    elif dist[pair_V[v]] == INF:
                        dist[pair_V[v]] = dist[u] + 1
                        dq.append(pair_V[v])
        
        return dist[0] != INF

    def dfs(u):
        for v in adj[u]:
            if pair_V[v] == 0 or (dist[pair_V[v]] == dist[u] + 1 and dfs(pair_V[v])):
                pair_U[u] = v
                pair_V[v] = u
                return True
        dist[u] = INF
        return False

    matching_count = 0
    
    while bfs():
        for u in range(1, q + 1):
            if pair_U[u] == 0:
                if dfs(u):
                    matching_count += 1
    
    if matching_count != q:
        sys.stdout.write("NO\n")
        return
    
    sys.stdout.write("YES\n")
    for u in range(1, q + 1):
        sys.stdout.write(f"{u} {pair_U[u]}\n")

if __name__ == '__main__':
    solve()
