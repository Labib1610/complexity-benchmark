from collections import defaultdict, deque

def solve():
    q, t = map(int, input().split())
    
    # Build adjacency list for bipartite graph
    # Left side: deprecated servers 1 to q
    # Right side: current servers q+1 to 2q
    graph = defaultdict(list)
    edges = []
    
    for _ in range(t):
        a, b = map(int, input().split())
        graph[a].append(b)
        edges.append((a, b))
    
    # Try to find a perfect matching using Hopcroft-Karp algorithm approach
    # But we'll use a simpler approach with DFS-based augmenting paths
    
    # partner[b] = a means deprecated server a is matched with current server b
    partner = {}
    # matched[a] = b means current server b is matched with deprecated server a
    matched = {}
    
    def dfs(u, visited):
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            
            # If v is unmatched or we can find an augmenting path from v's partner
            if v not in partner or dfs(partner[v], visited):
                partner[v] = u
                matched[u] = v
                return True
        return False
    
    # Try to match each deprecated server
    matching = 0
    for u in range(1, q + 1):
        if dfs(u, set()):
            matching += 1
    
    if matching == q:
        print("YES")
        for u in range(1, q + 1):
            print(u, matched[u])
    else:
        print("NO")

solve()
