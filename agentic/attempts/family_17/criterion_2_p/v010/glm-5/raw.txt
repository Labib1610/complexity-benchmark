import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        q_str = next(iterator)
        t_str = next(iterator)
    except StopIteration:
        return
    
    q = int(q_str)
    t = int(t_str)

    adj = [[] for _ in range(q + 1)]
    edges = []
    
    for _ in range(t):
        u_str = next(iterator)
        v_str = next(iterator)
        u = int(u_str)
        v = int(v_str)
        adj[u].append(v)
        edges.append((u, v))

    # Left side nodes: 1 to q (controllers)
    # Right side nodes: q+1 to 2q (platforms)
    
    pair_u = [0] * (q + 1)
    pair_v = [0] * (2 * q + 1)
    
    INF = q + 2
    dist = [0] * (q + 1)

    def bfs():
        queue = []
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        
        found_augmenting_path = False
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            
            if dist[u] < dist[0]:
                for v in adj[u]:
                    # If v is matched, we can explore its match (pair_v[v])
                    # dist[u] + 1 gives the level.
                    # We only update dist for the matched left node.
                    u_match = pair_v[v]
                    if u_match == 0:
                        # This neighbor is free, we found a path
                        found_augmenting_path = True
                    elif dist[u_match] == INF:
                        dist[u_match] = dist[u] + 1
                        queue.append(u_match)
        
        dist[0] = dist[0] if not found_augmenting_path else 0
        return found_augmenting_path

    def dfs(u):
        for v in adj[u]:
            u_match = pair_v[v]
            if u_match == 0 or (dist[u_match] == dist[u] + 1 and dfs(u_match)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching_count = 0
    
    # BFS to build layer graph
    while bfs():
        for u in range(1, q + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching_count += 1
    
    if matching_count != q:
        print("NO")
        return

    print("YES")
    
    # We must output exactly the edges that form the matching.
    # The order of output does not matter.
    # We iterate through the original edges and output those used in the perfect matching.
    
    output = []
    for u, v in edges:
        if pair_u[u] == v:
            output.append(f"{u} {v}\n")
            # Mark as used to handle potential duplicate edges in input correctly
            pair_u[u] = 0 
            
    sys.stdout.write("".join(output))

if __name__ == '__main__':
    solve()
