 We need to determine if a perfect matching exists between controllers (1..q) and platforms (q+1..2q) using only the given certified pairs, and if so, output one such matching.

This is a classic **bipartite maximum matching** problem.  
We can solve it using **Hopcroft–Karp** algorithm for efficiency, since:

- q ≤ 100,000  
- t ≤ 400,000  

The graph is large, so we need an O(E√V) algorithm.

We'll:

1. Build adjacency list for controllers → platforms.
2. Run Hopcroft–Karp to find maximum matching.
3. If matching size == q, output YES and the pairs; otherwise NO.

Here’s the Python implementation:

```python
import sys
from collections import deque

def hopcroft_karp(n, adj, pair_u, pair_v, dist):
    INF = float('inf')
    q = deque()
    for u in range(1, n+1):
        if pair_u[u] == 0:
            dist[u] = 0
            q.append(u)
        else:
            dist[u] = INF
    dist[0] = INF
    while q:
        u = q.popleft()
        if dist[u] < dist[0]:
            for v in adj[u]:
                if dist[pair_v[v]] == INF:
                    dist[pair_v[v]] = dist[u] + 1
                    q.append(pair_v[v])
    return dist[0] != INF

def dfs(u, adj, pair_u, pair_v, dist):
    if u != 0:
        for v in adj[u]:
            if dist[pair_v[v]] == dist[u] + 1:
                if dfs(pair_v[v], adj, pair_u, pair_v, dist):
                    pair_v[v] = u
                    pair_u[u] = v
                    return True
        dist[u] = float('inf')
        return False
    return True

def main():
    input = sys.stdin.readline
    q, t = map(int, input().split())
    adj = [[] for _ in range(q+1)]
    for _ in range(t):
        a, b = map(int, input().split())
        adj[a].append(b - q)  # store platform index as 1..q

    pair_u = [0] * (q+1)
    pair_v = [0] * (q+1)
    dist = [0] * (q+1)

    matching = 0
    while hopcroft_karp(q, adj, pair_u, pair_v, dist):
        for u in range(1, q+1):
            if pair_u[u] == 0:
                if dfs(u, adj, pair_u, pair_v, dist):
                    matching += 1

    if matching != q:
        print("NO")
    else:
        print("YES")
        for u in range(1, q+1):
            print(u, pair_u[u] + q)

if __name__ == "__main__":
    main()
```
