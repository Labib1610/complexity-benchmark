# Use iterative DFS to avoid recursion depth issues
def solve():

        return

    t = int(next(it))
    
    adj = [[] for _ in range(q + 1)]  # 1-indexed for controllers (1..q)

    def dfs(u, visited, stack):
        # Iterative DFS with backtracking
        # stack contains list of (u, iterator over adj[u])
        iterators = [iter(adj[u])]
        while iterators:
            try:
                v = next(iterators[-1])
                if visited[v]:
                    continue
                visited[v] = True
                # Try to match v
                # If v is unmatched or we can find alternate match
                # We use recursion simulation
                u_match = match_r[v]
                if u_match == -1:
                    # v is free, match it with current path's u
                    match_r[v] = stack[-1] if stack else u
                    result[match_r[v]] = v + q + 1
                    return True
                else:
                    # v is matched, try to find alternate for u_match
                    # Push u_match onto our "stack"
                    stack.append(u_match)
                    iterators.append(iter(adj[u_match]))
            except StopIteration:
                iterators.pop()
                if stack:
                    stack.pop()
                if not iterators:
                    return False
        return False
