```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    q = int(data[0])
    t = int(data[1])
    n = 2 * q
    head = [-1] * (n + 1)
    to = [0] * (2 * t)
    nxt = [0] * (2 * t)
    edge_cnt = 0
    idx = 2
    for _ in range(t):
        a = int(data[idx]); b = int(data[idx+1]); idx += 2
        to[edge_cnt] = b; nxt[edge_cnt] = head[a]; head[a] = edge_cnt; edge_cnt += 1
        to[edge_cnt] = a; nxt[edge_cnt] = head[b]; head[b] = edge_cnt; edge_cnt += 1

    match = [0] * (n + 1)
    vis = [0] * (n + 1)
    stamp = 0

    def augment(u):
        stack = [(u, head[u], False)]
        while stack:
            v, e, processed = stack.pop()
            if processed:
                continue
            stack.append((v, e, True))
            while e != -1:
                w = to[e]
                if vis[w] != stamp:
                    vis[w] = stamp
                    if match[w] == 0:
                        match[v] = w
                        match[w] = v
                        return True
                    else:
                        stack.append((match[w], head[match[w]], False))
                e = nxt[e]
        return False

    for u in range(1, q + 1):
        if match[u] == 0:
            stamp += 1
            if not augment(u):
                print("NO")
                return

    out = ["YES"]
    for u in range(1, q + 1):
        out.append(f"{u} {match[u]}")
    sys.stdout.write("\n".join(out))

solve()
```
