We can model this as a **max-flow** problem:

- Source = production facility `s`
- Sink = distribution hub `t`
- Each shipping lane is a directed edge with capacity `cap`
- We need to check if we can send exactly `d` units of flow from `s` to `t`
- If yes, output the flow on each edge.

We’ll use **Dinic’s algorithm** for max flow, then reconstruct the flow on each edge.

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
        self.edges = []  # (to, rev, cap)

    def add_edge(self, fr, to, cap):
        forward = [to, len(self.graph[to]), cap]
        backward = [fr, len(self.graph[fr]), 0]
        self.graph[fr].append(forward)
        self.graph[to].append(backward)
        self.edges.append((fr, len(self.graph[fr]) - 1))

    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            v = q.popleft()
            for to, rev, cap in self.graph[v]:
                if cap > 0 and self.level[to] < 0:
                    self.level[to] = self.level[v] + 1
                    q.append(to)
        return self.level[t] >= 0

    def dfs(self, v, t, f):
        if v == t:
            return f
        for i in range(self.it[v], len(self.graph[v])):
            self.it[v] = i
            to, rev, cap = self.graph[v][i]
            if cap > 0 and self.level[v] < self.level[to]:
                ret = self.dfs(to, t, min(f, cap))
                if ret > 0:
                    self.graph[v][i][2] -= ret
                    self.graph[to][rev][2] += ret
                    return ret
        return 0

    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

    def get_flow_on_edge(self, idx):
        fr, rev_idx = self.edges[idx]
        # The reverse edge's cap is the flow
        to, rev, cap = self.graph[fr][rev_idx]
        # reverse edge is graph[to][rev]
        return self.graph[to][rev][2]

def main():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1

    dinic = Dinic(n)
    edges_input = []
    for _ in range(m):
        u, v, cap = map(int, input().split())
        u -= 1
        v -= 1
        dinic.add_edge(u, v, cap)
        edges_input.append((u, v, cap))

    flow = dinic.max_flow(s, t)
    if flow < d:
        print("NO")
        return

    # We need exactly d flow, but max flow may be > d.
    # We can just output the flow on each edge from the max flow solution.
    # Since all capacities are integers and flow is integral, this is valid.
    result = []
    for i in range(m):
        result.append(dinic.get_flow_on_edge(i))

    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
```
