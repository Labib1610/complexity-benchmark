import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # number of active contracts, always 1, unused
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))

    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, cap, i))
        adj[v].append((u, cap, i))

    INF = float('inf')
    dist = [INF] * (n + 1)
    dist[s] = 0
    dq = deque([s])
    in_dq = [False] * (n + 1)
    in_dq[s] = True
    while dq:
        u = dq.popleft()
        in_dq[u] = False
        du = dist[u]
        for v, cap, _ in adj[u]:
            if cap > 0 and du + 1 < dist[v]:
                dist[v] = du + 1
                if not in_dq[v]:
                    dq.append(v)
                    in_dq[v] = True

    if dist[t] == INF:
        print("NO")
        return

    adj_flow = [[] for _ in range(n + 1)]
    for i, (u, v, cap) in enumerate(edges):
        adj_flow[u].append([v, cap, 0, i])
        adj_flow[v].append([u, 0, 0, i])
        edges[i] = [u, v, cap]

    def bfs():
        d_local = [INF] * (n + 1)
        d_local[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for v, cap, flow, idx in adj_flow[u]:
                if cap - flow > 0 and d_local[v] == INF:
                    d_local[v] = d_local[u] + 1
                    q.append(v)
        return d_local

    def dfs(u, pushed, d_local, ptr):
        if u == t:
            return pushed
        while ptr[u] < len(adj_flow[u]):
            e = adj_flow[u][ptr[u]]
            v, cap, flow, idx = e
            if cap - flow > 0 and d_local[v] == d_local[u] + 1:
                tr = dfs(v, min(pushed, cap - flow), d_local, ptr)
                if tr > 0:
                    e[2] += tr
                    adj_flow[v][e[3]][2] -= tr
                    return tr
            ptr[u] += 1
        return 0

    flow = 0
    while flow < d:
        d_local = bfs()
        if d_local[t] == INF:
            break
        ptr = [0] * (n + 1)
        while True:
            f = dfs(s, d - flow, d_local, ptr)
            if f == 0:
                break
            flow += f
            if flow >= d:
                break

    if flow < d:
        print("NO")
        return

    ans = [0] * m
    for u in range(n + 1):
        for v, cap, fl, idx in adj_flow[u]:
            if fl > 0:
                ans[idx] = fl

    print("YES")
    print(' '.join(map(str, ans)))

solve()
