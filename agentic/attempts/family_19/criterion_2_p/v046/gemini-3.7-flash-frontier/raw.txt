```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    _ = int(next(iterator)) # active contracts (1)
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    head = [-1] * (n + 1)
    to = []
    cap = []
    nxt = []
    edge_idx = 0
    
    orig_edges = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        
        orig_edges.append(edge_idx)
        
        # Forward edge
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = edge_idx
        edge_idx += 1
        
        # Backward edge
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = edge_idx
        edge_idx += 1
        
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[t] != -1

    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e]
            if level[u] + 1 == level[v] and tr > 0:
                push = pushed if pushed < tr else tr
                tr_pushed = dfs(v, push)
                if tr_pushed > 0:
                    cap[e] -= tr_pushed
                    cap[e ^ 1] += tr_pushed
                    return tr_pushed
            e = nxt[e]
            ptr[u] = e
        return 0

    total_flow = 0
    while total_flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while total_flow < d:
            pushed = dfs(s, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow == d:
        print("YES")
        ans = [cap[e ^ 1] for e in orig_edges]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
