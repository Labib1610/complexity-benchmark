from collections import deque

def solve():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    
    edges = []
    graph = [[] for _ in range(n + 1)]
    capacity = []
    
    for i in range(m):
        u, v, cap = map(int, input().split())
        edges.append((u, v))
        graph[u].append((v, len(capacity)))
        capacity.append(cap)
        graph[v].append((u, len(capacity)))
        capacity.append(0)  # Reverse edge with 0 capacity initially
    
    # Add super source and super sink
    super_s = 0
    super_t = n + 1
    
    # Connect super source to s with capacity d
    graph[super_s].append((s, len(capacity)))
    capacity.append(d)
    graph[s].append((super_s, len(capacity)))
    capacity.append(0)
    
    # Connect t to super sink with capacity d
    graph[t].append((super_t, len(capacity)))
    capacity.append(d)
    graph[super_t].append((t, len(capacity)))
    capacity.append(0)
    
    # Also add constraint that flow out of s minus flow into s equals d
    # And flow into t minus flow out of t equals d
    
    def bfs():
        parent = [-1] * (n + 2)
        visited = [False] * (n + 2)
        queue = deque([super_s])
        visited[super_s] = True
        
        while queue:
            u = queue.popleft()
            for v, edge_idx in graph[u]:
                if not visited[v] and capacity[edge_idx] > 0:
                    visited[v] = True
                    parent[v] = (u, edge_idx)
                    if v == super_t:
                        return parent
                    queue.append(v)
        return None
    
    def edmonds_karp():
        max_flow = 0
        while True:
            parent = bfs()
            if parent is None:
                break
                
            # Find minimum capacity along the path
            path_flow = float('inf')
            s_node = super_t
            while s_node != super_s:
                u, edge_idx = parent[s_node]
                path_flow = min(path_flow, capacity[edge_idx])
                s_node = u
            
            # Update capacities
            v = super_t
            while v != super_s:
                u, edge_idx = parent[v]
                capacity[edge_idx] -= path_flow
                # The reverse edge is edge_idx ^ 1
                capacity[edge_idx ^ 1] += path_flow
                v = u
                
            max_flow += path_flow
            
        return max_flow
    
    max_flow = edmonds_karp()
    
    if max_flow < d:
        print("NO")
        return
    
    print("YES")
    
    # Extract flow on original edges
    result = []
    for i in range(m):
        # The forward edge index is 2*i, reverse is 2*i+1
        # Original capacity was at 2*i, and we added reverse edge at 2*i+1
        # Flow on original edge = original capacity - current capacity
        flow = capacity[2*i + 1]  # This is the flow that went through
        result.append(flow)
    
    print(' '.join(map(str, result)))

solve()
