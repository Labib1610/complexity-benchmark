import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


class Edge:
    __slots__ = ("to", "rev", "cap")

    def __init__(self, to, rev, cap):
        self.to = to
        self.rev = rev
        self.cap = cap


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    demand = next(it)

    graph = [[] for _ in range(n)]
    original_edges = []

    def add_edge(u, v, cap):
        forward_index = len(graph[u])
        reverse_index = len(graph[v]) + (1 if u == v else 0)

        forward = Edge(v, reverse_index, cap)
        reverse = Edge(u, forward_index, 0)

        graph[u].append(forward)
        graph[v].append(reverse)
        original_edges.append((forward, cap))

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        add_edge(u, v, cap)

    if demand == 0:
        print("YES")
        print(" ".join(["0"] * m))
        return

    if s == t:
        print("NO")
        return

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            u = queue.popleft()
            for edge in graph[u]:
                if edge.cap > 0 and level[edge.to] == -1:
                    level[edge.to] = level[u] + 1
                    if edge.to == t:
                        return True
                    queue.append(edge.to)
        return level[t] != -1

    def dfs(u, pushed):
        if u == t:
            return pushed

        while ptr[u] < len(graph[u]):
            edge = graph[u][ptr[u]]
            if edge.cap > 0 and level[edge.to] == level[u] + 1:
                sent = dfs(edge.to, min(pushed, edge.cap))
                if sent:
                    edge.cap -= sent
                    graph[edge.to][edge.rev].cap += sent
                    return sent
            ptr[u] += 1

        return 0

    total_flow = 0

    while total_flow < demand and bfs():
        ptr[:] = [0] * n
        while total_flow < demand:
            sent = dfs(s, demand - total_flow)
            if sent == 0:
                break
            total_flow += sent

    if total_flow < demand:
        print("NO")
        return

    flows = [str(cap - edge.cap) for edge, cap in original_edges]
    print("YES")
    print(" ".join(flows))


if __name__ == "__main__":
    main()
