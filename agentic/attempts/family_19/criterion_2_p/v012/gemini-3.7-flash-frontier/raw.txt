import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        n_str = next(iterator)
    except StopIteration:
        return
        
    n = int(n_str)
    m = int(next(iterator))
    _ = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    head = [-1] * (n + 1)
    to = []
    nxt = []
    cap = []
    flow = []
    
    def add_edge(u, v, c):
        idx = len(to)
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        head[u] = idx
        
        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        head[v] = idx + 1

    edge_indices = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        edge_indices.append(len(to))
        add_edge(u, v, c)
        
    if s == t:
        if d == 0:
            print("YES")
            print(*(0 for _ in range(m)))
        else:
            print("NO")
        return

    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] - flow[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[t] != -1
        
    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            if level[v] == level[u] + 1 and cap[e] - flow[e] > 0:
                tr = dfs(v, min(pushed, cap[e] - flow[e]))
                if tr > 0:
                    flow[e] += tr
                    flow[e ^ 1] -= tr
                    return tr
            e = nxt[e]
            ptr[u] = e
        return 0

    total_flow = 0
    while total_flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while total_flow < d:
            pushed = dfs(s, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed
            
    if total_flow == d:
        print("YES")
        ans = [flow[idx] for idx in edge_indices]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
