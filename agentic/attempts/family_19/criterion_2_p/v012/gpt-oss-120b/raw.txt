```python
import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it); m = next(it); _ = next(it)  # number of orders =1
    s = next(it)-1; t = next(it)-1; d = next(it)
    edges_input = []
    for _ in range(m):
        u = next(it)-1; v = next(it)-1; cap = next(it)
        edges_input.append((u, v, cap))

    # Dinic
    N = n
    g = [[] for _ in range(N)]
    class Edge:
        __slots__ = ('to','rev','cap','orig')
        def __init__(self,to,rev,cap,orig):
            self.to=to; self.rev=rev; self.cap=cap; self.orig=orig

    def add_edge(fr,to,cap,idx):
        fwd = Edge(to, len(g[to]), cap, idx)
        rev = Edge(fr, len(g[fr]), 0, -1)
        g[fr].append(fwd)
        g[to].append(rev)

    for idx,(u,v,cap) in enumerate(edges_input):
        add_edge(u,v,cap,idx)

    level = [0]*N
    itp = [0]*N
    def bfs():
        for i in range(N):
            level[i] = -1
        q = deque([s])
        level[s]=0
        while q:
            v=q.popleft()
            for e in g[v]:
                if e.cap>0 and level[e.to]<0:
                    level[e.to]=level[v]+1
                    q.append(e.to)
        return level[t]>=0

    def dfs(v,f):
        if v==t:
            return f
        for i in range(itp[v], len(g[v])):
            e=g[v][i]
            if e.cap>0 and level[v]<level[e.to]:
                ret=dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap-=ret
                    g[e.to][e.rev].cap+=ret
                    return ret
            itp[v]+=1
        return 0

    flow=0
    INF=10**18
    while bfs():
        itp=[0]*N
        while True:
            f=dfs(s, INF)
            if not f: break
            flow+=f

    if flow<d:
        print("NO")
        return

    # we have flow >= d, need to extract exactly d units.
    # Since capacities are integers, we can just take the current flow and possibly reduce excess.
    # Build result array with flow on each original edge.
    res = [0]*m
    for u in range(N):
        for e in g[u]:
            if e.orig!=-1:
                used = g[e.to][e.rev].cap  # reverse edge capacity = flow sent
                res[e.orig] = used

    # total flow currently = flow, may be > d. Need to reduce by (flow-d) while keeping feasibility.
    excess = flow - d
    if excess>0:
        # try to push back excess along any edge with flow>0 using residual capacities.
        # We'll run a simple DFS from s to t reducing flow.
        # Instead, we can treat it as we need to send excess from t to s in residual graph.
        # Build residual graph again using current capacities.
        # We'll repeatedly find augmenting path from t to s.
        def bfs_back():
            for i in range(N):
                level[i]=-1
            q=deque([t])
            level[t]=0
            while q:
                v=q.popleft()
                for e in g[v]:
                    # we can go forward if reverse capacity >0 (i.e., we can reduce flow)
                    rev_edge = g[e.to][e.rev]
                    if rev_edge.cap>0 and level[e.to]==-1:
                        level[e.to]=level[v]+1
                        q.append(e.to)
            return level[s]!=-1

        def dfs_back(v, f):
            if v==s:
                return f
            for i in range(itp[v], len(g[v])):
                e=g[v][i]
                rev = g[e.to][e.rev]
                if rev.cap>0 and level[v]<level[e.to]:
                    ret=dfs_back(e.to, min(f, rev.cap))
                    if ret:
                        rev.cap-=ret
                        e.cap+=ret
                        # adjust result
                        if e.orig!=-1:
                            # this edge is original forward, reducing flow
                            res[e.orig]-=ret
                        else:
                            # this edge is reverse of original, increasing flow on original
                            # find original edge index
                            # reverse edge's rev points to original forward edge
                            orig_idx = g[e.to][e.rev].orig
                            if orig_idx!=-1:
                                res[orig_idx]+=ret
                        return ret
                itp[v]+=1
            return 0

        while excess>0 and bfs_back():
            itp=[0]*N
            pushed=dfs_back(t, excess)
            if not pushed:
                break
            excess-=pushed

    print("YES")
    print(' '.join(str(res[i]) for i in range(m)))

if __name__ == "__main__":
    solve()
```
