import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # constant 1
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    radj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        c = int(next(it))
        edges.append((u, v, c))
        adj[u].append(i)
        radj[v].append(i)
    
    # Check reachability from s
    reachable_s = [False] * (n + 1)
    q = deque([s])
    reachable_s[s] = True
    while q:
        u = q.popleft()
        for i in adj[u]:
            v = edges[i][1]
            if not reachable_s[v]:
                reachable_s[v] = True
                q.append(v)
    
    # Check reverse reachability to t
    reachable_t = [False] * (n + 1)
    q = deque([t])
    reachable_t[t] = True
    while q:
        v = q.popleft()
        for i in radj[v]:
            u = edges[i][0]
            if not reachable_t[u]:
                reachable_t[u] = True
                q.append(u)
    
    # Build flow network with only useful edges
    # Node indices: 1..n for original, n+1 = super source, n+2 = super sink
    N = n + 2
    S = n + 1
    T = n + 2
    cap = [[0] * (N + 1) for _ in range(N + 1)]
    flow = [[0] * (N + 1) for _ in range(N + 1)]
    graph = [[] for _ in range(N + 1)]
    
    # Add edges
    for i, (u, v, c) in enumerate(edges):
        if reachable_s[u] and reachable_t[v]:
            cap[u][v] += c
            graph[u].append(v)
            graph[v].append(u)
    
    # Add super source/sink edges
    cap[S][s] = d
    graph[S].append(s)
    graph[s].append(S)
    
    cap[t][T] = d
    graph[t].append(T)
    graph[T].append(t)
    
    # Dinic's algorithm
    level = [0] * (N + 1)
    ptr = [0] * (N + 1)
    
    def bfs():
        for i in range(1, N + 1):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            for v in graph[u]:
                if level[v] == -1 and cap[u][v] - flow[u][v] > 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[T] != -1
    
    def dfs(u, pushed):
        if u == T:
            return pushed
        for i in range(ptr[u], len(graph[u])):
            v = graph[u][i]
            if level[v] != level[u] + 1 or cap[u][v] - flow[u][v] == 0:
                continue
            tr = dfs(v, min(pushed, cap[u][v] - flow[u][v]))
            if tr:
                flow[u][v] += tr
                flow[v][u] -= tr
                return tr
            ptr[u] += 1
        return 0
    
    total_flow = 0
    while bfs():
        ptr = [0] * (N + 1)
        while True:
            pushed = dfs(S, float('inf'))
            if not pushed:
                break
            total_flow += pushed
    
    if total_flow != d:
        print("NO")
        return
    
    # Extract flow on original edges
    result = []
    for u, v, c in edges:
        result.append(str(flow[u][v]))
    
    print("YES")
    print(" ".join(result))

solve()
