 We can model this as a **max flow** problem.  
We need to check if we can send exactly `d` units from `s` to `t` while respecting capacities.  
If the max flow is at least `d`, we can output a valid flow assignment.

Here’s the Python solution:

```python
import sys
from collections import deque

input = sys.stdin.readline

class MaxFlow:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]
        self.cap = {}
    
    def add_edge(self, u, v, c):
        self.adj[u].append(v)
        self.adj[v].append(u)
        self.cap[(u, v)] = self.cap.get((u, v), 0) + c
        self.cap[(v, u)] = self.cap.get((v, u), 0)
    
    def bfs(self, s, t, parent):
        visited = [False] * self.n
        q = deque([s])
        visited[s] = True
        while q:
            u = q.popleft()
            for v in self.adj[u]:
                if not visited[v] and self.cap.get((u, v), 0) > 0:
                    visited[v] = True
                    parent[v] = u
                    if v == t:
                        return True
                    q.append(v)
        return False
    
    def max_flow(self, s, t):
        flow = 0
        parent = [-1] * self.n
        while self.bfs(s, t, parent):
            # find bottleneck
            v = t
            path_flow = float('inf')
            while v != s:
                u = parent[v]
                path_flow = min(path_flow, self.cap[(u, v)])
                v = u
            # update capacities
            v = t
            while v != s:
                u = parent[v]
                self.cap[(u, v)] -= path_flow
                self.cap[(v, u)] += path_flow
                v = u
            flow += path_flow
        return flow

def main():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1
    
    edges = []
    mf = MaxFlow(n)
    for _ in range(m):
        u, v, c = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v, c))
        mf.add_edge(u, v, c)
    
    flow = mf.max_flow(s, t)
    if flow < d:
        print("NO")
        return
    
    # Reconstruct flow on each edge
    result = []
    for u, v, c in edges:
        # original capacity minus remaining capacity gives used flow
        used = c - mf.cap[(u, v)]
        result.append(str(used))
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
