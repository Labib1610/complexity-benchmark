import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    required = next(it)

    graph = [[] for _ in range(n)]
    to = []
    capacity = []
    edge_ids = []
    original_capacity = []

    def add_edge(u, v, cap):
        eid = len(to)
        graph[u].append(eid)
        to.append(v)
        capacity.append(cap)

        graph[v].append(eid + 1)
        to.append(u)
        capacity.append(0)

        edge_ids.append(eid)
        original_capacity.append(cap)

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        add_edge(u, v, cap)

    if s == t:
        if required == 0:
            print("YES")
            print(*([0] * m))
        else:
            print("NO")
        return

    sys.setrecursionlimit(max(1_000_000, n + 10))

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            v = queue.popleft()
            next_level = level[v] + 1
            for eid in graph[v]:
                u = to[eid]
                if capacity[eid] > 0 and level[u] == -1:
                    level[u] = next_level
                    queue.append(u)

        return level[t] != -1

    def dfs(v, pushed):
        if v == t:
            return pushed

        edges = graph[v]
        while ptr[v] < len(edges):
            eid = edges[ptr[v]]
            u = to[eid]

            if capacity[eid] > 0 and level[u] == level[v] + 1:
                sent = dfs(u, min(pushed, capacity[eid]))
                if sent:
                    capacity[eid] -= sent
                    capacity[eid ^ 1] += sent
                    return sent

            ptr[v] += 1

        return 0

    flow = 0
    while flow < required and bfs():
        ptr[:] = [0] * n
        while flow < required:
            sent = dfs(s, required - flow)
            if sent == 0:
                break
            flow += sent

    if flow < required:
        print("NO")
        return

    answer = [
        original_capacity[i] - capacity[edge_ids[i]]
        for i in range(m)
    ]

    print("YES")
    print(*answer)

if __name__ == "__main__":
    main()
