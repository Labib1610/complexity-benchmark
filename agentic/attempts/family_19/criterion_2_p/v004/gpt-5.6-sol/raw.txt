import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    required = next(it)

    adj = [[] for _ in range(n)]
    to = []
    residual = []
    original_capacity = []
    input_edges = []

    def add_edge(u, v, capacity):
        idx = len(to)
        to.extend((v, u))
        residual.extend((capacity, 0))
        original_capacity.extend((capacity, 0))
        adj[u].append(idx)
        adj[v].append(idx ^ 1)
        input_edges.append(idx)

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        capacity = next(it)
        add_edge(u, v, capacity)

    if required == 0:
        sys.stdout.write("YES\n" + " ".join("0" for _ in range(m)) + "\n")
        return

    if s == t:
        sys.stdout.write("NO\n")
        return

    sys.setrecursionlimit(max(1_000_000, 2 * n + 10))
    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            v = queue.popleft()
            for edge in adj[v]:
                u = to[edge]
                if residual[edge] > 0 and level[u] == -1:
                    level[u] = level[v] + 1
                    if u == t:
                        return True
                    queue.append(u)
        return level[t] != -1

    def dfs(v, pushed):
        if v == t:
            return pushed

        while ptr[v] < len(adj[v]):
            edge = adj[v][ptr[v]]
            u = to[edge]

            if residual[edge] > 0 and level[u] == level[v] + 1:
                sent = dfs(u, min(pushed, residual[edge]))
                if sent:
                    residual[edge] -= sent
                    residual[edge ^ 1] += sent
                    return sent

            ptr[v] += 1

        return 0

    total_flow = 0
    while total_flow < required and bfs():
        ptr[:] = [0] * n
        while total_flow < required:
            sent = dfs(s, required - total_flow)
            if sent == 0:
                break
            total_flow += sent

    if total_flow < required:
        sys.stdout.write("NO\n")
        return

    flows = [
        str(original_capacity[edge] - residual[edge])
        for edge in input_edges
    ]
    sys.stdout.write("YES\n" + " ".join(flows) + "\n")

if __name__ == "__main__":
    main()
