import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = next(it)  # ignored (the '1' in statement)
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    D = int(next(it))

    edges_input = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges_input.append((u, v, cap))

    # Dinic
    N = n
    g = [[] for _ in range(N)]

    class Edge:
        __slots__ = ('to', 'rev', 'cap', 'orig_cap')
        def __init__(self, to, rev, cap):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.orig_cap = cap

    def add_edge(fr, to, cap):
        fwd = Edge(to, len(g[to]), cap)
        rev = Edge(fr, len(g[fr]), 0)
        g[fr].append(fwd)
        g[to].append(rev)
        return fwd, rev

    # keep reference to forward edges in input order
    forward_refs = []
    for u, v, cap in edges_input:
        fwd, rev = add_edge(u, v, cap)
        forward_refs.append(fwd)

    level = [0]*N
    it_ptr = [0]*N

    def bfs():
        for i in range(N):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in g[v]:
                if e.cap > 0 and level[e.to] < 0:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] >= 0

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(g[v])):
            e = g[v][i]
            if e.cap > 0 and level[v] < level[e.to]:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    g[e.to][e.rev].cap += ret
                    return ret
            it_ptr[v] += 1
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        it_ptr = [0]*N
        while True:
            f = dfs(s, INF)
            if not f:
                break
            flow += f

    if flow < D:
        print("NO")
        return

    # we have a flow of at least D, but we may have sent more than D.
    # Need to adjust to exactly D while respecting capacities.
    # Since capacities are integers, we can simply reduce excess flow along any augmenting path.
    excess = flow - D
    if excess > 0:
        # Build residual graph of current flow to push back excess from t to s.
        # We'll repeatedly find a path from t to s in residual (i.e., reverse edges with flow) and send back.
        while excess > 0:
            # BFS from t to s in residual where we can move flow back (i.e., edges with rev capacity >0)
            prev = [None]*N
            q = deque([t])
            prev[t] = (t, None)  # marker
            while q and prev[s] is None:
                v = q.popleft()
                for e in g[v]:
                    # e is forward edge in residual; we can go if e.cap > 0 (i.e., we can push more flow forward)
                    # To send flow back, we need to traverse reverse edges of original flow.
                    # Those are edges where the reverse edge has cap >0 (meaning flow can be reduced).
                    if e.cap > 0 and prev[e.to] is None:
                        prev[e.to] = (v, e)
                        q.append(e.to)
            if prev[s] is None:
                break  # cannot reduce further (should not happen)
            # find bottleneck
            v = s
            bott = excess
            while v != t:
                pv, e = prev[v]
                bott = min(bott, e.cap)
                v = pv
            # push back
            v = s
            while v != t:
                pv, e = prev[v]
                e.cap -= bott
                g[e.to][e.rev].cap += bott
                v = pv
            excess -= bott

    # Now extract flows on original edges: flow = original_cap - remaining cap
    result = []
    for fwd in forward_refs:
        used = fwd.orig_cap - fwd.cap
        result.append(str(used))

    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    solve()
