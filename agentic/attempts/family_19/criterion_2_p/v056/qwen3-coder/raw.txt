from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx+1])
    k = int(data[idx+2])
    idx += 3
    
    s = int(data[idx])
    t = int(data[idx+1])
    d = int(data[idx+2])
    idx += 3
    
    edges = []
    graph = defaultdict(list)
    indeg = [0] * (n + 1)
    
    for i in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        L = int(data[idx+2])
        idx += 3
        edges.append((u, v, L))
        graph[u].append((v, L, i))
        graph[v].append((u, L, i))
        indeg[v] += 1
    
    # Since k=1, we just need to send d units from s to t
    # This is a max-flow problem with demands
    
    # We model this as a circulation problem with lower bounds 0 and upper bounds L
    # Supply at s is +d, demand at t is -d, others 0
    
    # Let's build the residual graph and use Ford-Fulkerson or similar
    # But since we want to find ANY feasible flow (not necessarily max), 
    # and we have EXACTLY d to send, we can model it as a max-flow problem
    
    # Create super-source and super-sink to handle supply/demand
    # Actually, since we have exactly one source and one sink with equal supply/demand,
    # we can directly run max-flow from s to t with demand d
    
    # Build adjacency list for flow network
    adj = [[] for _ in range(n+1)]
    capacity = {}
    edge_index = {}
    
    for i in range(m):
        u, v, L = edges[i]
        adj[u].append((v, L, i))  # (to, capacity, index)
        capacity[(u,v,i)] = L
        edge_index[(u,v,i)] = i
    
    # Now we want to push exactly d flow from s to t
    # We'll use Ford-Fulkerson/Edmonds-Karp
    
    def bfs_flow():
        parent_edge = [-1] * m
        parent_node = [-1] * (n+1)
        visited = [False] * (n+1)
        q = deque()
        q.append((s, float('inf')))
        visited[s] = True
        
        while q:
            u, flow = q.popleft()
            for v, cap, idx in adj[u]:
                if not visited[v] and cap > 0:
                    visited[v] = True
                    new_flow = min(flow, cap)
                    parent_node[v] = u
                    parent_edge[edge_index[(u,v,idx)]] = edge_index[(u,v,idx)]
                    if v == t:
                        return new_flow, parent_node, parent_edge
                    q.append((v, new_flow))
        return 0, None, None
    
    # To track current flow on each edge
    flow_on_edge = [0] * m
    
    total_flow = 0
    while total_flow < d:
        f, parent_node, parent_edge = bfs_flow()
        if f == 0:
            break
        if total_flow + f > d:
            f = d - total_flow
        # Update residual capacities and flows
        cur = t
        while cur != s:
            prev = parent_node[cur]
            # Find the edge index from prev to cur
            for v, cap, idx in adj[prev]:
                if v == cur:
                    flow_on_edge[idx] += f
                    # Update capacity in residual graph
                    capacity[(prev, cur, idx)] -= f
                    # Add reverse edge if not exists
                    found = False
                    for x, c, i in adj[cur]:
                        if x == prev and i == idx:
                            found = True
                            break
                    if not found:
                        adj[cur].append((prev, 0, idx))  # Reverse edge with 0 initial capacity
                        capacity[(cur, prev, idx)] = 0
                    capacity[(cur, prev, idx)] += f
                    break
            cur = prev
        total_flow += f
    
    if total_flow < d:
        print("NO")
        return
    
    # Check flow conservation at each node
    # For each node, sum of incoming flows should equal sum of outgoing flows
    # Except for s (out - in = d) and t (in - out = d)
    
    node_balance = [0] * (n+1)  # net flow: positive means more out than in
    for i in range(m):
        u, v, _ = edges[i]
        node_balance[u] -= flow_on_edge[i]
        node_balance[v] += flow_on_edge[i]
    
    # Check if balances match requirements
    if node_balance[s] == -d and node_balance[t] == d:
        # All other nodes should have balance 0
        valid = True
        for i in range(1, n+1):
            if i != s and i != t and node_balance[i] != 0:
                valid = False
                break
        if valid:
            print("YES")
            print(' '.join(map(str, flow_on_edge)))
            return
    
    print("NO")

solve()
