import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    outgoing = [[] for _ in range(n + 1)]
    incoming = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        L = int(next(it))
        edges.append((u, v, L))
        outgoing[u].append(i)
        incoming[v].append(i)
    
    # Find a path from s to t using DFS
    path_edges = []
    visited = [False] * (n + 1)
    parent_edge = [-1] * (n + 1)
    
    stack = [s]
    visited[s] = True
    found_path = False
    
    while stack and not found_path:
        u = stack.pop()
        for ei in outgoing[u]:
            v = edges[ei][1]
            if not visited[v]:
                visited[v] = True
                parent_edge[v] = ei
                if v == t:
                    found_path = True
                    break
                stack.append(v)
    
    if found_path:
        # Reconstruct path
        v = t
        while v != s:
            ei = parent_edge[v]
            path_edges.append(ei)
            u = edges[ei][0]
            v = u
        path_edges.reverse()
        
        # Find minimum capacity on path
        min_cap = min(edges[ei][2] for ei in path_edges)
        
        if min_cap >= d:
            # We can route all flow through this path
            flow = [0] * m
            for ei in path_edges:
                flow[ei] = d
            print("YES")
            print(" ".join(map(str, flow)))
            return
    
    # Otherwise, use max flow
    # Build residual graph with edges in both directions
    adj = [[] for _ in range(n + 1)]
    edge_list = []
    
    # Original edges (index 0 to 2*m-1, even indices for forward)
    for i, (u, v, L) in enumerate(edges):
        idx = i * 2
        edge_list.append((u, v, L, idx + 1))  # forward edge
        adj[u].append(idx)
        edge_list.append((v, u, 0, idx))      # reverse edge
        adj[v].append(idx + 1)
    
    def bfs_level(s, t, level):
        dq = deque([s])
        level[s] = 0
        while dq:
            u = dq.popleft()
            for ei in adj[u]:
                v, cap, rev = edge_list[ei][1], edge_list[ei][2], edge_list[ei][3]
                if cap > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    dq.append(v)
        return level[t] != -1
    
    def dfs_flow(u, t, f, level, it):
        if u == t:
            return f
        for i in range(it[u], len(adj[u])):
            ei = adj[u][i]
            v = edge_list[ei][1]
            if edge_list[ei][2] > 0 and level[v] == level[u] + 1:
                pushed = dfs_flow(v, t, min(f, edge_list[ei][2]), level, it)
                if pushed > 0:
                    edge_list[ei] = (edge_list[ei][0], edge_list[ei][1], edge_list[ei][2] - pushed, edge_list[ei][3])
                    rev = edge_list[ei][3]
                    edge_list[rev] = (edge_list[rev][0], edge_list[rev][1], edge_list[rev][2] + pushed, edge_list[rev][3])
                    return pushed
            it[u] += 1
        return 0
    
    total_flow = 0
    level = [-1] * (n + 1)
    while bfs_level(s, t, level):
        it = [0] * (n + 1)
        while True:
            pushed = dfs_flow(s, t, float('inf'), level, it)
            if pushed == 0:
                break
            total_flow += pushed
        level = [-1] * (n + 1)
    
    if total_flow >= d:
        flow = [0] * m
        for i in range(m):
            forward_idx = i * 2
            original_cap = edges[i][2]
            current_cap = edge_list[forward_idx][2]
            flow[i] = original_cap - current_cap
        print("YES")
        print(" ".join(map(str, flow)))
    else:
        print("NO")

solve()
