import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _k = next(it)

    s = next(it) - 1
    t = next(it) - 1
    demand = next(it)

    adj = [[] for _ in range(n)]
    to = []
    capacity = []
    original_capacity = []
    original_edges = []

    def add_edge(u, v, cap):
        eid = len(to)
        to.append(v)
        capacity.append(cap)
        original_capacity.append(cap)
        adj[u].append(eid)

        to.append(u)
        capacity.append(0)
        original_capacity.append(0)
        adj[v].append(eid ^ 1)
        return eid

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        original_edges.append(add_edge(u, v, cap))

    if demand < 0 or (s == t and demand != 0):
        print("NO")
        return

    if demand == 0:
        print("YES")
        print(" ".join(["0"] * m))
        return

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        q = deque([s])

        while q:
            v = q.popleft()
            next_level = level[v] + 1
            for eid in adj[v]:
                u = to[eid]
                if capacity[eid] > 0 and level[u] == -1:
                    level[u] = next_level
                    q.append(u)

        return level[t] != -1

    def send_block(limit):
        vertices = [s]
        requests = [limit]
        sent = [0]
        incoming = [-1]

        while vertices:
            v = vertices[-1]

            if v == t or sent[-1] == requests[-1] or ptr[v] >= len(adj[v]):
                result = requests[-1] if v == t else sent[-1]
                child_request = requests[-1]
                eid = incoming[-1]

                vertices.pop()
                requests.pop()
                sent.pop()
                incoming.pop()

                if not vertices:
                    return result

                if result:
                    capacity[eid] -= result
                    capacity[eid ^ 1] += result
                    sent[-1] += result

                if result < child_request:
                    ptr[vertices[-1]] += 1
                continue

            eid = adj[v][ptr[v]]
            u = to[eid]

            if capacity[eid] <= 0 or level[u] != level[v] + 1:
                ptr[v] += 1
                continue

            child_request = min(requests[-1] - sent[-1], capacity[eid])
            vertices.append(u)
            requests.append(child_request)
            sent.append(0)
            incoming.append(eid)

        return 0

    flow = 0
    while flow < demand and bfs():
        ptr[:] = [0] * n
        pushed = send_block(demand - flow)
        if pushed == 0:
            break
        flow += pushed

    if flow != demand:
        print("NO")
        return

    answer = [
        str(original_capacity[eid] - capacity[eid])
        for eid in original_edges
    ]
    print("YES")
    print(" ".join(answer))

if __name__ == "__main__":
    main()
