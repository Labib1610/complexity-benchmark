import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    iterator = iter(data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    if s == t:
        if d == 0:
            print("YES")
            print(*(0 for _ in range(m)))
        else:
            print("NO")
        return

    # Dinic's Algorithm
    # Nodes 1..n, and supersource 0
    src = 0
    dst = t
    
    # We will add an edge from 0 to s with capacity d
    # Graph representation
    # edge: [to, cap, flow, rev_idx, original_edge_idx]
    
    head = [[] for _ in range(n + 1)]
    edges = []
    
    def add_edge(u, v, cap, orig_idx):
        idx_u = len(head[u])
        idx_v = len(head[v])
        if u == v:
            idx_v += 1
        head[u].append([v, cap, 0, idx_v, orig_idx])
        head[v].append([u, 0, 0, idx_u, -1])

    # Supersource to s
    add_edge(src, s, d, -1)
    
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        add_edge(u, v, cap, i)
        
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[src] = 0
        queue = [src]
        head_ptr = 0
        while head_ptr < len(queue):
            u = queue[head_ptr]
            head_ptr += 1
            for e in head[u]:
                v, cap, flow, _, _ = e
                if cap - flow > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        return level[dst] >= 0

    ptr = [0] * (n + 1)

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == dst:
            return pushed
        for cid in range(ptr[u], len(head[u])):
            ptr[u] = cid
            e = head[u][cid]
            v, cap, flow, rev_idx, _ = e
            tr = cap - flow
            if level[u] + 1 != level[v] or tr == 0:
                continue
            push = dfs(v, min(pushed, tr))
            if push == 0:
                continue
            e[2] += push
            head[v][rev_idx][2] -= push
            return push
        return 0

    sys.setrecursionlimit(max(10000, n + 500))
    
    flow = 0
    while bfs():
        ptr = [0] * (n + 1)
        while True:
            pushed = dfs(src, d - flow)
            if pushed == 0:
                break
            flow += pushed
            if flow == d:
                break
        if flow == d:
            break
            
    if flow != d:
        print("NO")
        return
        
    ans = [0] * m
    for u in range(1, n + 1):
        for e in head[u]:
            orig_idx = e[4]
            if orig_idx != -1:
                ans[orig_idx] = e[2]
                
    print("YES")
    print(*(ans[i] for i in range(m)))

if __name__ == '__main__':
    solve()
