We need to find a feasible flow of exactly \( d \) units from \( s \) to \( t \) in a directed graph where each edge has a capacity (upper bound), and we allowed any non-negative integer flow. This is a maximum flow problem: if the maximum flow from \( s \) to \( t \) is at least \( d \), then we can output a valid flow of \( d \) units.

We can use a standard max-flow algorithm such as Dinic’s.

We also need to output the flow on each edge in the original order.

Here's the Python solution:

```python
import sys
from collections import deque

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
        self.edges = []  # (to, rev, cap)
    
    def add_edge(self, fr, to, cap):
        forward = [to, len(self.graph[to]), cap]
        backward = [fr, len(self.graph[fr]), 0]
        self.graph[fr].append(forward)
        self.graph[to].append(backward)
        self.edges.append((fr, to, forward, backward))
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque()
        q.append(s)
        self.level[s] = 0
        while q:
            v = q.popleft()
            for e in self.graph[v]:
                if e[2] > 0 and self.level[e[0]] < 0:
                    self.level[e[0]] = self.level[v] + 1
                    q.append(e[0])
        return self.level[t] >= 0
    
    def dfs(self, v, t, f):
        if v == t:
            return f
        for i in range(self.it[v], len(self.graph[v])):
            self.it[v] = i
            e = self.graph[v][i]
            if e[2] > 0 and self.level[v] < self.level[e[0]]:
                ret = self.dfs(e[0], t, min(f, e[2]))
                if ret > 0:
                    e[2] -= ret
                    self.graph[e[0]][e[1]][2] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1
    dinic = Dinic(n)
    original_caps = []
    edge_info = []
    for _ in range(m):
        u, v, cap = map(int, input().split())
        u -= 1
        v -= 1
        dinic.add_edge(u, v, cap)
        original_caps.append(cap)
        edge_info.append((u, v))

    maxflow = dinic.max_flow(s, t)
    if maxflow < d:
        print("NO")
        return

    # We want exactly d flow, we may need to reduce if maxflow > d
    # Since Dinic already computes maximal flow, but we need exactly d.
    # We can recompute for desired d by limiting once flow reaches d:
    # We'll construct new flow network and run until d.
    # Simpler: run Dinic again, but stop when flow reaches d.
    
    # Build fresh graph
    dinic2 = Dinic(n)
    for u, v, cap in zip(*zip(*edge_info), original_caps):
        pass
    # Re-add edges with original caps
    for (u, v), cap in zip(edge_info, original_caps):
        dinic2.add_edge(u, v, cap)

    flow = 0
    INF = 10**18
    while flow < d:
        if not dinic2.bfs(s, t):
            print("NO")
            return
        dinic2.it = [0] * n
        while flow < d:
            f = dinic2.dfs(s, t, d - flow)
            if f == 0:
                break
            flow += f
        if f == 0 and flow < d:
            print("NO")
            return

    # Output flow per original edge
    res = []
    for u, v, forward, backward in dinic2.edges:
        # flow on edge = original_cap - forward[2]
        # but we need original cap stored
        # forward[2] is remaining capacity
        pass

    # Easier: store flow during add
    flow_values = []
    for (u, v), cap in zip(edge_info, original_caps):
        for e in dinic2.graph[u]:
            if e[0] == v and e[2] < cap and e[2] >= 0:
                flow_used = cap - e[2]
                flow_values.append(flow_used)
                break
        else:
            flow_values.append(0)

    print("YES")
    print(" ".join(map(str, flow_values)))
```
