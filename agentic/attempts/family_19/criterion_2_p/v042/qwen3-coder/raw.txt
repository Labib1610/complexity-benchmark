from collections import defaultdict, deque

def solve():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    
    edges = []
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    for i in range(m):
        u, v, limit = map(int, input().split())
        edges.append((u, v, limit, i))
        graph[u].append((v, limit, i))
        reverse_graph[v].append((u, limit, i))
    
    # We need to find a flow of value d from s to t
    # Let's use Ford-Fulkerson with BFS (Edmonds-Karp)
    
    # Build adjacency list with capacity and index
    adj = [[] for _ in range(n+1)]
    edge_index_map = {}  # Maps (u,v,idx) -> edge index in adj structure
    
    for idx, (u, v, limit, orig_idx) in enumerate(edges):
        # Forward edge
        adj[u].append([v, limit, 0, len(adj[v])])  # to, capacity, flow, reverse index
        edge_index_map[(u, v, orig_idx)] = len(adj[u]) - 1
        # Backward edge
        adj[v].append([u, 0, 0, len(adj[u]) - 1])
        
    def bfs():
        parent = [-1] * (n + 1)
        parent_edge = [-1] * (n + 1)
        queue = deque()
        queue.append(s)
        
        while queue:
            node = queue.popleft()
            for i, (to, capacity, flow, rev_idx) in enumerate(adj[node]):
                if capacity > flow and parent[to] == -1 and to != s:
                    parent[to] = node
                    parent_edge[to] = i
                    if to == t:
                        return parent, parent_edge
                    queue.append(to)
                    
        return parent, parent_edge
    
    max_flow = 0
    while True:
        parent, parent_edge = bfs()
        if parent[t] == -1:
            break
            
        # Find minimum residual capacity along the path
        path_flow = float('inf')
        cur = t
        while cur != s:
            prev = parent[cur]
            edge_idx = parent_edge[cur]
            to, capacity, flow, rev_idx = adj[prev][edge_idx]
            path_flow = min(path_flow, capacity - flow)
            cur = prev
            
        # Update flows
        cur = t
        while cur != s:
            prev = parent[cur]
            edge_idx = parent_edge[cur]
            # Forward edge
            adj[prev][edge_idx][2] += path_flow
            # Backward edge
            rev_idx = adj[prev][edge_idx][3]
            adj[cur][rev_idx][2] -= path_flow
            cur = prev
            
        max_flow += path_flow
        if max_flow >= d:
            break
    
    if max_flow < d:
        print("NO")
        return
        
    # Now we need to extract the actual flows
    result = [0] * m
    for idx, (u, v, limit, orig_idx) in enumerate(edges):
        # Get the flow on this edge
        if (u, v, orig_idx) in edge_index_map:
            edge_idx = edge_index_map[(u, v, orig_idx)]
            flow = adj[u][edge_idx][2]
            result[orig_idx] = flow
    
    print("YES")
    print(*result)

solve()
