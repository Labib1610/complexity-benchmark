import sys
from collections import deque

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    K = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    if s == t:
        if d == 0:
            print("YES")
            print(*(0 for _ in range(m)))
        else:
            print("NO")
        return

    # Super source to limit total flow to d
    source = 0
    sink = t
    
    # Graph representation for Dinic
    # Nodes: 0 (super-source), 1..n
    head = [-1] * (n + 1)
    to = []
    cap = []
    flow = []
    nxt = []
    
    def add_edge(u, v, c):
        idx = len(to)
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        head[u] = idx
        
        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        head[v] = idx ^ 1
        return idx

    # Add edge from super source to s with capacity d
    add_edge(source, s, d)
    
    edge_indices = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        idx = add_edge(u, v, c)
        edge_indices.append(idx)
        
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[source] = 0
        q = deque([source])
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                u = to[e]
                if cap[e] - flow[e] > 0 and level[u] == -1:
                    level[u] = level[v] + 1
                    q.append(u)
                e = nxt[e]
        return level[sink] != -1

    def dfs(v, pushed):
        if pushed == 0:
            return 0
        if v == sink:
            return pushed
        e = ptr[v]
        while e != -1:
            u = to[e]
            tr = cap[e] - flow[e]
            if level[v] + 1 == level[u] and tr > 0:
                push = dfs(u, min(pushed, tr))
                if push > 0:
                    flow[e] += push
                    flow[e ^ 1] -= push
                    return push
            e = nxt[e]
            ptr[v] = e
        return 0

    tot_flow = 0
    INF = 10**18
    while bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while True:
            pushed = dfs(source, INF)
            if pushed == 0:
                break
            tot_flow += pushed

    if tot_flow < d:
        print("NO")
    else:
        print("YES")
        ans = [flow[idx] for idx in edge_indices]
        print(*(ans))

if __name__ == '__main__':
    solve()
