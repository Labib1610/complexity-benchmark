import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    s = next(it) - 1
    t = next(it) - 1
    demand = next(it)

    if k <= 0 or demand < 0 or demand % k != 0:
        print("NO")
        return

    head = [-1] * n
    to = []
    cap = []
    nxt = []
    edge_indices = []
    original_caps = []

    def add_edge(u, v, capacity):
        idx = len(to)

        to.append(v)
        cap.append(capacity)
        nxt.append(head[u])
        head[u] = idx

        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = idx + 1

        return idx

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        c = next(it)
        scaled_capacity = c // k
        edge_indices.append(add_edge(u, v, scaled_capacity))
        original_caps.append(scaled_capacity)

    required = demand // k

    if required == 0:
        print("YES")
        print(" ".join(["0"] * m))
        return

    if s == t:
        print("NO")
        return

    sys.setrecursionlimit(max(1000, 2 * n + 100))

    level = [-1] * n
    current = [-1] * n

    def bfs():
        for i in range(n):
            level[i] = -1
        level[s] = 0
        q = deque([s])

        while q:
            v = q.popleft()
            e = head[v]
            next_level = level[v] + 1
            while e != -1:
                w = to[e]
                if cap[e] > 0 and level[w] == -1:
                    level[w] = next_level
                    q.append(w)
                e = nxt[e]

        return level[t] != -1

    def dfs(v, pushed):
        if v == t:
            return pushed

        while current[v] != -1:
            e = current[v]
            w = to[e]

            if cap[e] > 0 and level[w] == level[v] + 1:
                sent = dfs(w, min(pushed, cap[e]))
                if sent:
                    cap[e] -= sent
                    cap[e ^ 1] += sent
                    return sent

            current[v] = nxt[e]

        return 0

    total_flow = 0
    while total_flow < required and bfs():
        current[:] = head
        while total_flow < required:
            sent = dfs(s, required - total_flow)
            if sent == 0:
                break
            total_flow += sent

    if total_flow < required:
        print("NO")
        return

    answer = [
        (original_caps[i] - cap[edge_indices[i]]) * k
        for i in range(m)
    ]

    print("YES")
    print(" ".join(map(str, answer)))

if __name__ == "__main__":
    main()
