from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx+1])
    K = int(data[idx+2])
    idx += 3
    
    s = int(data[idx])
    t = int(data[idx+1])
    d = int(data[idx+2])
    idx += 3
    
    edges = []
    graph = defaultdict(list)
    capacity = []
    
    for i in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        c = int(data[idx+2])
        idx += 3
        edges.append((u, v))
        graph[u].append((v, i))
        graph[v].append((u, i))
        capacity.append(c)
    
    # Since K=1, we don't need to worry about modularity beyond integer flows.
    # This becomes a standard max-flow problem with demand d from s to t.
    
    # We'll use a flow network approach. 
    # Add a super source and super sink to model the fixed demand.
    # But actually, since we have only one source and one sink with fixed demand,
    # we can directly find an s-t flow of value d.
    
    # Let's build the residual graph and use Ford-Fulkerson or similar.
    # However, we're not just checking feasibility; we need to find a specific flow.
    
    # Actually, let's think differently. 
    # This is a circulation problem with demands.
    # But it's simpler: we want to send exactly d units from s to t.
    
    # So we can model this as a max-flow problem:
    # - Source is s, Sink is t
    # - Edge capacities are given
    # - We want to push exactly d flow
    
    # If max flow < d, then impossible.
    # Otherwise, extract the flow values.
    
    # Implementing Ford-Fulkerson
    
    # Build adjacency list with edge indices
    adj = [[] for _ in range(n+1)]
    back_adj = [[] for _ in range(n+1)]
    for i in range(m):
        u, v = edges[i]
        adj[u].append((v, i))       # (node, edge_index)
        back_adj[v].append((u, i))
    
    def bfs_flow():
        # BFS to find augmenting path
        parent_edge = [-1] * (n+1)
        parent_node = [-1] * (n+1)
        visited = [False] * (n+1)
        queue = deque()
        queue.append(s)
        visited[s] = True
        
        while queue:
            u = queue.popleft()
            for v, e_idx in adj[u]:
                if not visited[v] and flow[e_idx] < capacity[e_idx]:
                    visited[v] = True
                    parent_node[v] = u
                    parent_edge[v] = e_idx
                    if v == t:
                        return parent_edge
                    queue.append(v)
            # Also check backward edges
            for prev_node, e_idx in back_adj[u]:
                if not visited[prev_node] and flow[e_idx] > 0:
                    visited[prev_node] = True
                    parent_node[prev_node] = u
                    parent_edge[prev_node] = e_idx
                    if prev_node == t:
                        return parent_edge
                    queue.append(prev_node)
        return None
    
    # Initialize flow
    flow = [0] * m
    
    # Max flow loop
    while True:
        parent = bfs_flow()
        if parent[t] == -1:
            break
            
        # Find minimum residual capacity along the path
        path_flow = float('inf')
        curr = t
        path_edges = []
        while curr != s:
            e_idx = parent[curr]
            u = parent_node[curr]
            # Determine if forward or backward edge
            found = False
            for v, idx in adj[u]:
                if v == curr and idx == e_idx:
                    path_flow = min(path_flow, capacity[e_idx] - flow[e_idx])
                    path_edges.append((e_idx, 'forward'))
                    found = True
                    break
            if not found:
                # Must be backward
                path_flow = min(path_flow, flow[e_idx])
                path_edges.append((e_idx, 'backward'))
            curr = parent_node[curr]
        
        # Update flow
        for e_idx, direction in path_edges:
            if direction == 'forward':
                flow[e_idx] += path_flow
            else:
                flow[e_idx] -= path_flow
    
    # Check if we achieved the required demand
    total_out_s = sum(flow[i] for i in range(m) if edges[i][0] == s)
    total_in_s = sum(flow[i] for i in range(m) if edges[i][1] == s)
    net_out_s = total_out_s - total_in_s
    
    if net_out_s != d:
        print("NO")
        return
    
    # Verify conservation at other nodes
    node_net = [0] * (n+1)
    for i in range(m):
        u, v = edges[i]
        node_net[u] += flow[i]
        node_net[v] -= flow[i]
    
    # Conservation check (except s and t)
    valid = True
    for i in range(1, n+1):
        if i != s and i != t and node_net[i] != 0:
            valid = False
            break
    if node_net[s] != -d or node_net[t] != d:
        valid = False
        
    if not valid:
        print("NO")
        return
        
    print("YES")
    print(' '.join(map(str, flow)))

solve()
