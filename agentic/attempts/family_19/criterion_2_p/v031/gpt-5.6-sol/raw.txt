import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    demand = next(it)

    graph = [[] for _ in range(n)]
    original_edges = []

    def add_edge(u, v, capacity):
        forward = [v, len(graph[v]), capacity]
        backward = [u, len(graph[u]), 0]
        graph[u].append(forward)
        graph[v].append(backward)
        original_edges.append((forward, capacity))

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        capacity = next(it)
        if u == v:
            original_edges.append((None, capacity))
        else:
            add_edge(u, v, capacity)

    if demand == 0 or s == t:
        print("YES")
        print(" ".join(["0"] * m))
        return

    sys.setrecursionlimit(max(1_000_000, n * 2 + 10))
    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            u = queue.popleft()
            next_level = level[u] + 1
            for edge in graph[u]:
                if edge[2] > 0 and level[edge[0]] == -1:
                    level[edge[0]] = next_level
                    queue.append(edge[0])

        return level[t] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == t:
            return pushed

        while ptr[u] < len(graph[u]):
            edge = graph[u][ptr[u]]
            v, rev, capacity = edge

            if capacity > 0 and level[v] == level[u] + 1:
                sent = dfs(v, min(pushed, capacity))
                if sent:
                    edge[2] -= sent
                    graph[v][rev][2] += sent
                    return sent

            ptr[u] += 1

        return 0

    total_flow = 0
    while total_flow < demand and bfs():
        ptr[:] = [0] * n
        while total_flow < demand:
            sent = dfs(s, demand - total_flow)
            if sent == 0:
                break
            total_flow += sent

    if total_flow < demand:
        print("NO")
        return

    flows = []
    for edge, capacity in original_edges:
        if edge is None:
            flows.append("0")
        else:
            flows.append(str(capacity - edge[2]))

    print("YES")
    print(" ".join(flows))

if __name__ == "__main__":
    main()
