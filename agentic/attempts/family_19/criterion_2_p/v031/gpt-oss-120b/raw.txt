```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = int(next(it))  # the third number is always 1 per statement
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))

    edges_input = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges_input.append((u, v, cap))

    # Dinic
    class Edge:
        __slots__ = ('to', 'rev', 'cap', 'orig')
        def __init__(self, to, rev, cap, orig):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.orig = orig  # original capacity

    graph = [[] for _ in range(n)]

    # store reference to the forward edge for each original edge
    edge_refs = []

    def add_edge(fr, to, cap):
        fwd = Edge(to, len(graph[to]), cap, cap)
        rev = Edge(fr, len(graph[fr]), 0, 0)
        graph[fr].append(fwd)
        graph[to].append(rev)
        return fwd

    for u, v, cap in edges_input:
        fwd = add_edge(u, v, cap)
        edge_refs.append(fwd)

    level = [0] * n
    it_ptr = [0] * n

    from collections import deque

    def bfs():
        for i in range(n):
            level[i] = -1
        q = deque()
        level[s] = 0
        q.append(s)
        while q:
            v = q.popleft()
            for e in graph[v]:
                if e.cap > 0 and level[e.to] < 0:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] >= 0

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(graph[v])):
            e = graph[v][i]
            if e.cap > 0 and level[v] < level[e.to]:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    graph[e.to][e.rev].cap += ret
                    return ret
            it_ptr[v] += 1
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        it_ptr = [0] * n
        while True:
            pushed = dfs(s, INF)
            if not pushed:
                break
            flow += pushed

    if flow != d:
        print("NO")
        return

    # we have a feasible flow of value d, need to output flows on each original edge
    # The current residual capacities reflect used flow = original_cap - residual_cap
    result = []
    for e in edge_refs:
        used = e.orig - e.cap
        result.append(str(used))

    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    solve()
```
