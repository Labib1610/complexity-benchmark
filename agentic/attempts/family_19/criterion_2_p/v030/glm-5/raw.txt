import sys
from collections import deque

def solve():
    it = iter(sys.stdin.read().strip().split())
    try:
        n = int(next(it))
        m = int(next(it))
        _ = int(next(it))  # number of cargo types, ignored as per problem
    except StopIteration:
        return

    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u - 1, v - 1, cap))
    
    # Build flow graph
    # nodes: original n nodes
    # mapping: 
    # Original capacity edge i -> edge i
    # Reverse edge -> edge m + i
    
    # Adjacency list: stores edge indices
    adj = [[] for _ in range(n)]
    edge_targets = [] # maps edge_idx -> target node
    capacities = []   # maps edge_idx -> capacity
    
    for i, (u, v, cap) in enumerate(edges):
        adj[u].append(i)
        edge_targets.append(v)
        capacities.append(cap)
        
        adj[v].append(m + i)
        edge_targets.append(u)
        capacities.append(0)
    
    # Dinic's algorithm
    level = []
    ptr = []
    
    def bfs():
        nonlocal level
        level = [-1] * n
        q = deque([s])
        level[s] = 0
        while q:
            u = q.popleft()
            for edge_idx in adj[u]:
                v = edge_targets[edge_idx]
                cap = capacities[edge_idx]
                if cap > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] != -1
    
    def dfs(u, flow):
        if u == t:
            return flow
        # global ptr # not needed in python closure read, but write needs nonlocal
        # ptr is modified, let's ensure we access the outer one
        # actually ptr is a list defined in outer scope, so modifying its elements is fine
        # but if we reassign ptr, we need nonlocal.
        # Here we just read and increment elements.
        
        for k in range(ptr[u], len(adj[u])):
            ptr[u] = k
            edge_idx = adj[u][k]
            v = edge_targets[edge_idx]
            cap = capacities[edge_idx]
            
            if cap > 0 and level[v] == level[u] + 1:
                pushed = dfs(v, min(flow, cap))
                if pushed > 0:
                    capacities[edge_idx] -= pushed
                    rev_idx = edge_idx + m if edge_idx < m else edge_idx - m
                    capacities[rev_idx] += pushed
                    return pushed
        return 0
        
    max_flow = 0
    while bfs():
        ptr = [0] * n
        while True:
            pushed = dfs(s, d - max_flow)
            if pushed == 0:
                break
            max_flow += pushed
            if max_flow == d:
                break
        if max_flow == d:
            break
            
    if max_flow == d:
        print("YES")
        flow_vals = []
        for i in range(m):
            # flow on edge i is original capacity - remaining capacity
            # stored at capacities[i]
            original_cap = edges[i][2]
            remaining_cap = capacities[i]
            flow_vals.append(original_cap - remaining_cap)
        print(*(flow_vals))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
