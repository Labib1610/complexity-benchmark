```python
import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
_ = next(it)

s = next(it)
t = next(it)
demand = next(it)

adj = [[] for _ in range(n + 1)]
to = []
cap = []
original_capacity = []
edge_ids = []

def add_edge(u, v, capacity):
    edge_id = len(to)
    to.append(v)
    cap.append(capacity)
    original_capacity.append(capacity)
    adj[u].append(edge_id)

    to.append(u)
    cap.append(0)
    original_capacity.append(0)
    adj[v].append(edge_id + 1)
    return edge_id

for _ in range(m):
    u = next(it)
    v = next(it)
    capacity = next(it)
    edge_ids.append(add_edge(u, v, capacity))

if s == t:
    if demand != 0:
        print("NO")
    else:
        print("YES")
        print(*([0] * m))
    sys.exit()

level = [-1] * (n + 1)
ptr = [0] * (n + 1)

def bfs():
    level[:] = [-1] * (n + 1)
    level[s] = 0
    queue = deque([s])

    while queue:
        u = queue.popleft()
        for edge_id in adj[u]:
            v = to[edge_id]
            if cap[edge_id] > 0 and level[v] == -1:
                level[v] = level[u] + 1
                queue.append(v)

    return level[t] != -1

def dfs(u, pushed):
    if pushed == 0:
        return 0
    if u == t:
        return pushed

    while ptr[u] < len(adj[u]):
        edge_id = adj[u][ptr[u]]
        v = to[edge_id]

        if cap[edge_id] > 0 and level[v] == level[u] + 1:
            sent = dfs(v, min(pushed, cap[edge_id]))
            if sent:
                cap[edge_id] -= sent
                cap[edge_id ^ 1] += sent
                return sent

        ptr[u] += 1

    return 0

flow = 0
while flow < demand and bfs():
    ptr[:] = [0] * (n + 1)
    while flow < demand:
        sent = dfs(s, demand - flow)
        if sent == 0:
            break
        flow += sent

if flow < demand:
    print("NO")
else:
    answer = [
        original_capacity[edge_id] - cap[edge_id]
        for edge_id in edge_ids
    ]
    print("YES")
    print(*answer)
```
