import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]

    def add_edge(self, u, v, cap, idx=None):
        # forward edge
        fwd = [v, cap, None, idx]
        # backward edge
        rev = [u, 0, fwd, None]
        fwd[2] = rev
        self.adj[u].append(fwd)
        self.adj[v].append(rev)

    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = [s]
        self.level[s] = 0
        for u in q:
            for v, cap, rev, idx in self.adj[u]:
                if cap and self.level[v] == -1:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] != -1

    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.adj[u])):
            self.it[u] = i
            v, cap, rev, idx = self.adj[u][i]
            if cap and self.level[v] == self.level[u] + 1:
                ret = self.dfs(v, t, min(f, cap))
                if ret:
                    self.adj[u][i][1] -= ret
                    rev[1] += ret
                    return ret
        return 0

    def max_flow(self, s, t, limit=None):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                pushed = self.dfs(s, t, INF if limit is None else limit - flow)
                if not pushed:
                    break
                flow += pushed
                if limit is not None and flow >= limit:
                    return flow
        return flow

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = int(next(it))  # cargo types ignored
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    edges = []
    dinic = Dinic(n)
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))
        dinic.add_edge(u, v, cap, idx)

    max_possible = dinic.max_flow(s, t)
    if max_possible < d:
        print("NO")
        return

    # recompute flow to get exactly d (reuse same graph but reset capacities)
    # Build fresh graph with same capacities
    dinic = Dinic(n)
    edge_refs = []
    for idx, (u, v, cap) in enumerate(edges):
        dinic.add_edge(u, v, cap, idx)
        # store reference to forward edge for later flow extraction
        edge_refs.append(dinic.adj[u][-2])  # the forward edge just added

    dinic.max_flow(s, t, limit=d)

    # extract flows
    result = [0] * m
    for idx, fwd in enumerate(edge_refs):
        used = edges[idx][2] - fwd[1]  # original cap - residual cap
        result[idx] = used

    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
