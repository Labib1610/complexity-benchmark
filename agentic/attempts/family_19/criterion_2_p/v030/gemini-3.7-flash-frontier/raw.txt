import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    if s == t:
        if d == 0:
            print("YES")
            print(*(0 for _ in range(m)))
        else:
            print("NO")
        return

    # Super source 0, nodes 1..n
    S = 0
    T = t
    
    head = [-1] * (n + 1)
    to = []
    cap = []
    nxt = []
    edge_indices = []
    
    def add_edge(u, v, c, orig_idx=-1):
        # Forward edge
        nxt.append(head[u])
        to.append(v)
        cap.append(c)
        head[u] = len(to) - 1
        edge_indices.append(orig_idx)
        
        # Backward edge
        nxt.append(head[v])
        to.append(u)
        cap.append(0)
        head[v] = len(to) - 1
        edge_indices.append(-1)
        
    add_edge(S, s, d, -1)
    
    orig_caps = []
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        limit = int(next(iterator))
        orig_caps.append(limit)
        add_edge(u, v, limit, i)
        
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[T] != -1

    ptr = [0] * (n + 1)
    
    def dfs(u, pushed):
        if pushed == 0 or u == T:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e]
            if level[u] + 1 == level[v] and tr > 0:
                push = dfs(v, min(pushed, tr))
                if push > 0:
                    cap[e] -= push
                    cap[e ^ 1] += push
                    return push
            e = nxt[e]
            ptr[u] = e
        return 0

    flow = 0
    while bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            flow += pushed

    if flow != d:
        print("NO")
    else:
        print("YES")
        ans = [0] * m
        # In our representation, original edges are added after the S->s edge
        # Edge 0 and 1 are S->s and s->S
        # Subsequent pairs 2*i + 2, 2*i + 3 are the m original edges
        for i in range(m):
            f_edge = 2 + 2 * i
            assigned_flow = orig_caps[i] - cap[f_edge]
            ans[i] = assigned_flow
        print(*(ans))

if __name__ == '__main__':
    solve()
