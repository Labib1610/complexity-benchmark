```python
import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
_ = next(it)

s = next(it) - 1
t = next(it) - 1
required = next(it)

graph = [[] for _ in range(n)]
original_edges = []

def add_edge(u, v, capacity):
    forward_index = len(graph[u])
    forward = [v, capacity, 0]
    graph[u].append(forward)

    reverse_index = len(graph[v])
    reverse = [u, 0, forward_index]
    graph[v].append(reverse)

    forward[2] = reverse_index
    original_edges.append((forward, capacity))

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    capacity = next(it)
    add_edge(u, v, capacity)

if s == t:
    if required != 0:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n" + " ".join("0" for _ in range(m)) + "\n")
    sys.exit()

level = [-1] * n
pointer = [0] * n

def bfs():
    for i in range(n):
        level[i] = -1
    level[s] = 0
    queue = deque([s])

    while queue:
        u = queue.popleft()
        for edge in graph[u]:
            if edge[1] > 0 and level[edge[0]] == -1:
                level[edge[0]] = level[u] + 1
                queue.append(edge[0])

    return level[t] != -1

def dfs(u, pushed):
    if pushed == 0:
        return 0
    if u == t:
        return pushed

    while pointer[u] < len(graph[u]):
        edge = graph[u][pointer[u]]
        v, residual, reverse_index = edge

        if residual > 0 and level[v] == level[u] + 1:
            sent = dfs(v, min(pushed, residual))
            if sent:
                edge[1] -= sent
                graph[v][reverse_index][1] += sent
                return sent

        pointer[u] += 1

    return 0

total_flow = 0

while total_flow < required and bfs():
    pointer[:] = [0] * n
    while total_flow < required:
        sent = dfs(s, required - total_flow)
        if sent == 0:
            break
        total_flow += sent

if total_flow < required:
    sys.stdout.write("NO\n")
else:
    flows = [str(capacity - edge[1]) for edge, capacity in original_edges]
    sys.stdout.write("YES\n" + " ".join(flows) + "\n")
```
