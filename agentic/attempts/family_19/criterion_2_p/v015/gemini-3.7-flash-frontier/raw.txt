```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    _ = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # Supersource to enforce exact flow limit d
    S = 0
    T = t
    
    # Total vertices: 0 to n
    num_nodes = n + 1
    
    head = [-1] * num_nodes
    to = []
    cap = []
    nxt = []
    
    def add_edge(u, v, c):
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = len(to) - 1
        
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = len(to) - 1

    # Edge indices in the graph
    # Original edges are added first
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        add_edge(u, v, c)
        
    add_edge(S, s, d)
    
    level = [-1] * num_nodes
    
    def bfs():
        for i in range(num_nodes):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[T] != -1

    ptr = [0] * num_nodes
    
    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == T:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e]
            if level[u] + 1 == level[v] and tr > 0:
                push = dfs(v, min(pushed, tr))
                if push > 0:
                    cap[e] -= push
                    cap[e ^ 1] += push
                    return push
            ptr[u] = nxt[e]
            e = nxt[e]
        return 0

    max_flow = 0
    while bfs():
        for i in range(num_nodes):
            ptr[i] = head[i]
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed

    if max_flow == d:
        print("YES")
        ans = []
        for i in range(m):
            # The flow on edge i is the capacity of the reverse edge 2*i + 1
            ans.append(str(cap[2 * i + 1]))
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
