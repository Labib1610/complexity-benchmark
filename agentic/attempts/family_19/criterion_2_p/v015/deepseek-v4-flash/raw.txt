We can model this as a max flow problem.  
We need to check if the maximum flow from `s` to `t` is at least `d`.  
If yes, we output `YES` and the flow on each edge.  
Otherwise, output `NO`.

Here’s the Python solution:

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1

    edges = []
    graph = [[] for _ in range(n)]
    edge_index = 0

    for _ in range(m):
        u, v, cap = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append([v, cap, edge_index])
        graph[v].append([u, 0, edge_index])
        edges.append((u, v, cap))
        edge_index += 1

    # Dinic's algorithm
    level = [0] * n
    it = [0] * n

    def bfs():
        for i in range(n):
            level[i] = -1
        q = deque()
        q.append(s)
        level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, idx in graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] >= 0

    def dfs(u, f):
        if u == t:
            return f
        for i in range(it[u], len(graph[u])):
            v, cap, idx = graph[u][i]
            if cap > 0 and level[v] == level[u] + 1:
                ret = dfs(v, min(f, cap))
                if ret > 0:
                    graph[u][i][1] -= ret
                    # find reverse edge
                    for j in range(len(graph[v])):
                        if graph[v][j][2] == idx and graph[v][j][0] == u:
                            graph[v][j][1] += ret
                            break
                    return ret
            it[u] += 1
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        it = [0] * n
        while True:
            f = dfs(s, INF)
            if f == 0:
                break
            flow += f

    if flow < d:
        print("NO")
        return

    print("YES")
    # Build result from residual graph
    result = [0] * m
    for u in range(n):
        for v, cap, idx in graph[u]:
            if idx % 2 == 0:  # original edge
                # original capacity minus remaining capacity
                orig_cap = edges[idx][2]
                result[idx] = orig_cap - cap
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
```
