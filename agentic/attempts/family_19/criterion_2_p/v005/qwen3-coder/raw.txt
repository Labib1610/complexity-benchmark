    rev_adj = [[] for _ in range(n+1)]

        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        c = int(data[idx]); idx += 1

        adj[u].append(i)
        rev_adj[v].append(i)
    
    # Max flow from s to t
    flow = [0] * m
    
    def bfs():
        parent = [-1] * (n + 1)
        parent_edge = [-1] * (n + 1)
        visited = [False] * (n + 1)
        q = deque([s])
        visited[s] = True
        
        while q:
            u = q.popleft()
            if u == t:
                break
            # Forward edges
            for e in adj[u]:
                v = edges[e][1]
                if not visited[v] and flow[e] < cap[e]:
                    visited[v] = True
                    parent[v] = u
                    parent_edge[v] = e
                    q.append(v)
            # Backward edges
            for e in rev_adj[u]:
                v = edges[e][0]
                if not visited[v] and flow[e] > 0:
                    visited[v] = True
                    parent[v] = u
                    parent_edge[v] = ~e  # bitwise complement to mark backward
                    q.append(v)
        
        if not visited[t]:
            return 0, None
        
        # Find bottleneck
        bottleneck = float('inf')
        cur = t
        while cur != s:
            e = parent_edge[cur]
            if e >= 0:
                bottleneck = min(bottleneck, cap[e] - flow[e])
                cur = edges[e][0]
            else:
                e = ~e
                bottleneck = min(bottleneck, flow[e])
                cur = edges[e][1]
        
        return bottleneck, parent_edge
    
    total_flow = 0
    while total_flow < d:
        aug, parent_edge = bfs()

            break

        total_flow += aug
        
        # Augment along path
        cur = t
        while cur != s:

            if e >= 0:
                flow[e] += aug
