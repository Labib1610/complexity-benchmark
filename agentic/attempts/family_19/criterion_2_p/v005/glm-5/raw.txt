import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # always 1
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bays = int(next(it))
        edges.append((u, v, bays))
    
    # Build graph for max flow
    # Each edge becomes two directed edges in residual graph
    # Edge index: original edge i -> 2*i (forward), 2*i+1 (backward)
    N = n + 1
    adj = [[] for _ in range(N)]
    
    for i, (u, v, bays) in enumerate(edges):
        # Forward edge
        adj[u].append((v, 2*i, bays))
        # Backward edge
        adj[v].append((u, 2*i+1, 0))
    
    cap = [0] * (2 * m)
    for i, (_, _, bays) in enumerate(edges):
        cap[2*i] = bays
        cap[2*i+1] = 0
    
    flow = [0] * (2 * m)
    
    def bfs():
        level = [-1] * N
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for v, eid, _ in adj[u]:
                if level[v] == -1 and cap[eid] - flow[eid] > 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level
    
    def dfs(u, f, level, it):
        if u == t:
            return f
        for j in range(it[u], len(adj[u])):
            it[u] = j
            v, eid, _ = adj[u][j]
            if level[v] == level[u] + 1 and cap[eid] - flow[eid] > 0:
                ret = dfs(v, min(f, cap[eid] - flow[eid]), level, it)
                if ret > 0:
                    flow[eid] += ret
                    flow[eid ^ 1] -= ret
                    return ret
        return 0
    
    total = 0
    while True:
        level = bfs()
        if level[t] == -1:
            break
        it = [0] * N
        while True:
            pushed = dfs(s, float('inf'), level, it)
            if pushed == 0:
                break
            total += pushed
    
    if total >= d:
        # Scale down flows
        scale = d / total if total > 0 else 1
        result = []
        for i in range(m):
            f = flow[2*i] * scale
            # Round to nearest integer, handling floating point
            result.append(int(round(f)))
        print("YES")
        print(" ".join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
