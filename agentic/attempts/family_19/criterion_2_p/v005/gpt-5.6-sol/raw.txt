```python
import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
_ = next(it)

s = next(it) - 1
t = next(it) - 1
demand = next(it)

graph = [[] for _ in range(n)]
to = []
capacity = []
forward_edges = []
original_capacity = []

def add_edge(u, v, cap):
    edge_id = len(to)
    to.append(v)
    capacity.append(cap)
    to.append(u)
    capacity.append(0)
    graph[u].append(edge_id)
    graph[v].append(edge_id ^ 1)
    forward_edges.append(edge_id)
    original_capacity.append(cap)

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    cap = next(it)
    add_edge(u, v, cap)

if s == t or demand == 0:
    print("YES")
    print(" ".join(["0"] * m))
    sys.exit()

level = [-1] * n
ptr = [0] * n

def bfs():
    level[:] = [-1] * n
    level[s] = 0
    queue = deque([s])

    while queue:
        u = queue.popleft()
        for edge_id in graph[u]:
            if capacity[edge_id] > 0:
                v = to[edge_id]
                if level[v] == -1:
                    level[v] = level[u] + 1
                    if v == t:
                        return True
                    queue.append(v)
    return level[t] != -1

def dfs(u, pushed):
    if u == t:
        return pushed

    edges = graph[u]
    while ptr[u] < len(edges):
        edge_id = edges[ptr[u]]
        v = to[edge_id]

        if capacity[edge_id] > 0 and level[v] == level[u] + 1:
            sent = dfs(v, min(pushed, capacity[edge_id]))
            if sent:
                capacity[edge_id] -= sent
                capacity[edge_id ^ 1] += sent
                return sent

        ptr[u] += 1

    return 0

flow = 0
while flow < demand and bfs():
    ptr[:] = [0] * n
    while flow < demand:
        sent = dfs(s, demand - flow)
        if sent == 0:
            break
        flow += sent

if flow < demand:
    print("NO")
else:
    result = [
        str(original_capacity[i] - capacity[edge_id])
        for i, edge_id in enumerate(forward_edges)
    ]
    print("YES")
    print(" ".join(result))
```
