from collections import deque
import sys

# Increase recursion depth for deep DFS paths in Dinic's algorithm
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))

    s = int(next(it))
    t = int(next(it))
    d = int(next(it))

    SRC = 0
    TGT = t

    head = [-1] * (n + 1)
    to = []
    cap = []
    nxt = []

    def add_edge(u, v, c):
        idx = len(to)
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = idx

        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = idx + 1
        return idx

    # Connect super source to s with capacity d
    add_edge(SRC, s, d)

    orig_edge_indices = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        c = int(next(it))
        idx = add_edge(u, v, c)
        orig_edge_indices.append((idx, c))

    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)

    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[SRC] = 0
        q = deque([SRC])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[TGT] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == TGT:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e]
            if level[u] + 1 == level[v] and tr > 0:
                push = dfs(v, min(pushed, tr))
                if push > 0:
                    cap[e] -= push
                    cap[e ^ 1] += push
                    return push
            e = nxt[e]
            ptr[u] = e
        return 0

    total_flow = 0
    while bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while True:
            pushed = dfs(SRC, float("inf"))
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow == d:
        print("YES")
        ans = [str(initial_cap - cap[idx]) for idx, initial_cap in orig_edge_indices]
        print(" ".join(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
