import sys

def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); ptr += 1
    m = int(data[ptr]); ptr += 1
    ptr += 1  # k (always 1)
    s = int(data[ptr]) - 1; ptr += 1
    t = int(data[ptr]) - 1; ptr += 1
    d = int(data[ptr]); ptr += 1

    if d < 0:
        s, t = t, s
        d = -d

    edges = []
    bad = False
    for _ in range(m):
        u = int(data[ptr]) - 1; ptr += 1
        v = int(data[ptr]) - 1; ptr += 1
        L = int(data[ptr]); ptr += 1
        if L < 0:
            bad = True
        edges.append((u, v, L))

    if s == t:
        if d == 0 and not bad:
            sys.stdout.write("YES\n" + " ".join(["0"] * m) + "\n")
        else:
            sys.stdout.write("NO\n")
        return

    if bad:
        sys.stdout.write("NO\n")
        return

    N = n + 2
    S = n
    T = n + 1

    graph = [[] for _ in range(N)]
    to = []
    cap = []

    fwd = [0] * m
    orig = [0] * m

    for i in range(m):
        u, v, L = edges[i]
        fi = len(to)
        orig[i] = L
        fwd[i] = fi
        graph[u].append(fi)
        to.append(v)
        cap.append(L)
        graph[v].append(fi + 1)
        to.append(u)
        cap.append(0)

    graph[S].append(len(to)); to.append(s); cap.append(d)
    graph[s].append(len(to)); to.append(S); cap.append(0)
    graph[t].append(len(to)); to.append(T); cap.append(d)
    graph[T].append(len(to)); to.append(t); cap.append(0)

    flow = 0
    while True:
        level = [-1] * N
        level[S] = 0
        q = [S]
        head = 0
        while head < len(q):
            u = q[head]; head += 1
            lu = level[u] + 1
            for e in graph[u]:
                if cap[e] > 0:
                    v = to[e]
                    if level[v] == -1:
                        level[v] = lu
                        q.append(v)
        if level[T] == -1:
            break
        it = [0] * N
        path = []
        u = S
        while True:
            if u == T:
                f = min(cap[e] for e in path)
                for e in path:
                    cap[e] -= f
                    cap[e ^ 1] += f
                flow += f
                cut = 0
                for i2 in range(len(path)):
                    if cap[path[i2]] == 0:
                        cut = i2
                        break
                u = to[path[cut] ^ 1]
                del path[cut:]
                continue
            gu = graph[u]
            iu = it[u]
            lng = len(gu)
            lu1 = level[u] + 1
            moved = False
            while iu < lng:
                e = gu[iu]
                if cap[e] > 0 and level[to[e]] == lu1:
                    it[u] = iu
                    path.append(e)
                    u = to[e]
                    moved = True
                    break
                iu += 1
            if moved:
                continue
            it[u] = iu
            if u == S:
                break
            level[u] = -1
            e = path.pop()
            u = to[e ^ 1]
            it[u] += 1
        if flow >= d:
            break

    if flow >= d:
        res = [str(orig[i] - cap[fwd[i]]) for i in range(m)]
        sys.stdout.write("YES\n" + " ".join(res) + "\n")
    else:
        sys.stdout.write("NO\n")
