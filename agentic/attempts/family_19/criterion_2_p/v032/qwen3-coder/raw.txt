        visited = [False] * (n + 2)  # nodes 1..n, plus 0 (super source), n+1 (super sink)

                edge.flow += path_flow
                edge.reverse.flow -= path_flow
                v_node = parent[v_node]
            
            max_flow += path_flow
        
        return max_flow
