We need to find integer flows on each edge such that:

- Flow on each edge is between 0 and its capacity.
- At source `s`: out − in = d.
- At sink `t`: in − out = d.
- At all other nodes: in = out.

Since k=1, this is a standard max-flow feasibility problem with lower bound 0 and upper bound L.

We can solve it using max flow. If the max flow from s to t is at least d, we can find a feasible flow of value d.

We’ll use Dinic’s algorithm for max flow, then reconstruct the flow on each edge.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.graph[u].append([v, cap, len(self.graph[v])])
        self.graph[v].append([u, 0, len(self.graph[u]) - 1])
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, rev in self.graph[u]:
                if cap > 0 and self.level[v] < 0:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] >= 0
    
    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.graph[u])):
            self.it[u] = i
            v, cap, rev = self.graph[u][i]
            if cap > 0 and self.level[u] + 1 == self.level[v]:
                ret = self.dfs(v, t, min(f, cap))
                if ret > 0:
                    self.graph[u][i][1] -= ret
                    self.graph[v][rev][1] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    n, m, k = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1
    
    edges = []
    dinic = Dinic(n)
    for _ in range(m):
        u, v, L = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v, L))
        dinic.add_edge(u, v, L)
    
    flow = dinic.max_flow(s, t)
    if flow < d:
        print("NO")
        return
    
    # Now we need to find a flow of exactly d.
    # We can just take the flow we have and reduce if needed.
    # But since capacities are integers and we only need d, we can
    # run max flow again with capacity d as target.
    # Simpler: run Dinic with a super source to s with capacity d.
    
    dinic2 = Dinic(n + 2)
    super_s = n
    super_t = n + 1
    dinic2.add_edge(super_s, s, d)
    dinic2.add_edge(t, super_t, d)
    for u, v, L in edges:
        dinic2.add_edge(u, v, L)
    
    flow2 = dinic2.max_flow(super_s, super_t)
    if flow2 < d:
        print("NO")
        return
    
    # Extract flow on original edges
    result = []
    # In dinic2, edges are added in order: first super source edge, then super sink edge, then original edges.
    # Original edges start at index 2 in the graph adjacency of their source node.
    # We'll store mapping.
    edge_index = 0
    # We'll just iterate over original edges and find their reverse edge capacity
    # In dinic2, for each original edge (u,v,L), we added it after the two super edges.
    # The flow on it = L - residual capacity of forward edge.
    # We can access via graph structure.
    
    # Let's rebuild mapping: for each original edge, we know its position in dinic2's graph.
    # We'll store the forward edge reference.
    edge_refs = []
    for u, v, L in edges:
        # In dinic2, we added super_s->s, t->super_t, then original edges.
        # So original edges are added after those two.
        idx = len(dinic2.graph[u]) - 1
        edge_refs.append((u, idx))
    
    for u, idx in edge_refs:
        # forward edge capacity after flow = residual
        residual = dinic2.graph[u][idx][1]
        original_cap = None
        # find original cap from edges list
        for uu, vv, LL in edges:
            if uu == u and dinic2.graph[u][idx][0] == vv:
                original_cap = LL
                break
        flow_on_edge = original_cap - residual
        result.append(flow_on_edge)
    
    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
```
