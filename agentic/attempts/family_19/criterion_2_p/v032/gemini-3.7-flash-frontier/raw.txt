import sys
from collections import deque

# Increase recursion depth for deep DFS in Dinic's algorithm
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    if s == t:
        if d == 0:
            print("YES")
            print(*(0 for _ in range(m)))
        else:
            print("NO")
        return

    adj = [[] for _ in range(n + 1)]
    edges = []
    
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        
        # Forward edge
        fwd = [v, cap, 0, None, i]
        # Backward edge
        bwd = [u, 0, 0, None, -1]
        
        fwd[3] = len(adj[v])
        bwd[3] = len(adj[u])
        
        adj[u].append(fwd)
        adj[v].append(bwd)
        edges.append(fwd)

    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)

    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for edge in adj[u]:
                v, cap, flow, _, _ = edge
                if cap - flow > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] != -1

    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        for cid in range(ptr[u], len(adj[u])):
            ptr[u] = cid
            edge = adj[u][cid]
            v, cap, flow, rev, _ = edge
            tr = cap - flow
            if level[u] + 1 != level[v] or tr <= 0:
                continue
            push = dfs(v, min(pushed, tr))
            if push == 0:
                continue
            edge[2] += push
            adj[v][rev][2] -= push
            return push
        return 0

    tot_flow = 0
    while tot_flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = 0
        while tot_flow < d:
            pushed = dfs(s, d - tot_flow)
            if pushed == 0:
                break
            tot_flow += pushed

    if tot_flow == d:
        print("YES")
        ans = [0] * m
        for edge in edges:
            ans[edge[4]] = edge[2]
        print(*(ans[i] for i in range(m)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
