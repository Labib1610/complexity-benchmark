import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    _ = next(iterator) # unused third parameter on first line
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # Adjacency list and edge list
    head = [-1] * (n + 1)
    to = []
    cap = []
    nxt = []
    
    edge_indices = []
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        
        edge_idx = len(to)
        edge_indices.append(edge_idx)
        
        # Forward edge
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = edge_idx
        
        # Backward edge
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = edge_idx + 1

    # Dinic's Algorithm
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    from collections import deque

    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[t] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == t:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e]
            if level[u] + 1 == level[v] and tr > 0:
                push = dfs(v, min(pushed, tr))
                if push > 0:
                    cap[e] -= push
                    cap[e ^ 1] += push
                    return push
            e = nxt[e]
            ptr[u] = e
        return 0

    flow = 0
    while flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while flow < d:
            pushed = dfs(s, d - flow)
            if pushed == 0:
                break
            flow += pushed

    if flow == d:
        print("YES")
        ans = [cap[idx ^ 1] for idx in edge_indices]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
