import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # unused
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        c = int(next(it))
        edges.append((u, v, c))
        adj[u].append((v, i))
        adj[v].append((u, ~i))
    
    # Build flow network
    N = n + 2
    S = n + 1
    T = n
    cap = [[0] * (N + 1) for _ in range(N + 1)]
    flow = [[0] * (N + 1) for _ in range(N + 1)]
    graph = [[] for _ in range(N + 1)]
    
    # Add super source and super sink
    graph[S].append(s)
    graph[s].append(S)
    graph[t].append(T)
    graph[T].append(t)
    cap[S][s] = d
    cap[t][T] = d
    
    # Add original edges
    for i, (u, v, c) in enumerate(edges):
        graph[u].append(v)
        graph[v].append(u)
        cap[u][v] = c
    
    # Dinic's algorithm
    level = [0] * (N + 1)
    ptr = [0] * (N + 1)
    
    def bfs():
        for i in range(N + 1):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            for v in graph[u]:
                if level[v] == -1 and cap[u][v] - flow[u][v] > 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[T] != -1
    
    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == T:
            return pushed
        for i in range(ptr[u], len(graph[u])):
            v = graph[u][i]
            if level[v] != level[u] + 1:
                continue
            f = cap[u][v] - flow[u][v]
            if f <= 0:
                continue
            tr = dfs(v, min(pushed, f))
            if tr == 0:
                continue
            flow[u][v] += tr
            flow[v][u] -= tr
            return tr
            ptr[u] += 1
        return 0
    
    total_flow = 0
    while bfs():
        ptr = [0] * (N + 1)
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            total_flow += pushed
    
    if total_flow != d:
        print("NO")
        return
    
    print("YES")
    result = []
    for u, v, c in edges:
        result.append(str(flow[u][v]))
    print(' '.join(result))

if __name__ == "__main__":
    solve()
