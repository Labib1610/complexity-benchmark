import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


class Edge:
    __slots__ = ("to", "cap", "rev", "initial")

    def __init__(self, to, cap, initial=0):
        self.to = to
        self.cap = cap
        self.rev = None
        self.initial = initial


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    demand = next(it)

    graph = [[] for _ in range(n)]
    original_edges = []

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)

        forward = Edge(v, cap, cap)
        backward = Edge(u, 0)
        forward.rev = backward
        backward.rev = forward

        graph[u].append(forward)
        graph[v].append(backward)
        original_edges.append(forward)

    if s == t:
        if demand == 0:
            print("YES")
            print(*([0] * m))
        else:
            print("NO")
        return

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            v = queue.popleft()
            for edge in graph[v]:
                if edge.cap > 0 and level[edge.to] == -1:
                    level[edge.to] = level[v] + 1
                    queue.append(edge.to)

        return level[t] != -1

    def dfs(v, pushed):
        if v == t:
            return pushed

        while ptr[v] < len(graph[v]):
            edge = graph[v][ptr[v]]
            if edge.cap > 0 and level[edge.to] == level[v] + 1:
                sent = dfs(edge.to, min(pushed, edge.cap))
                if sent:
                    edge.cap -= sent
                    edge.rev.cap += sent
                    return sent
            ptr[v] += 1

        return 0

    flow = 0
    while flow < demand and bfs():
        ptr[:] = [0] * n
        while flow < demand:
            sent = dfs(s, demand - flow)
            if sent == 0:
                break
            flow += sent

    if flow < demand:
        print("NO")
    else:
        result = [edge.initial - edge.cap for edge in original_edges]
        print("YES")
        print(*result)


if __name__ == "__main__":
    main()
