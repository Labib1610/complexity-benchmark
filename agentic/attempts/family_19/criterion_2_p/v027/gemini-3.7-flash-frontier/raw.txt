```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    # input_data[2] is 1 (asset type)
    
    s = int(input_data[3])
    t = int(input_data[4])
    d = int(input_data[5])
    
    if s == t:
        if d == 0:
            print("YES")
            print(" ".join(["0"] * m))
        else:
            # If s == t and d > 0, conservation implies net outbound d, impossible if s=t
            print("NO")
        return

    # Super source S = 0, vertices 1..n
    # Add edge S -> s with capacity d
    S = 0
    T = t
    num_nodes = n + 1
    
    head = [-1] * num_nodes
    to = []
    cap = []
    nxt = []
    
    def add_edge(u, v, c):
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = len(to) - 1
        
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = len(to) - 1

    # Super source edge
    add_edge(S, s, d)
    
    orig_edge_indices = []
    idx = 6
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        c = int(input_data[idx+2])
        idx += 3
        orig_edge_indices.append((len(to), c))
        add_edge(u, v, c)
        
    level = [-1] * num_nodes
    
    def bfs():
        for i in range(num_nodes):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[T] != -1

    ptr = [0] * num_nodes
    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == T:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e]
            if level[u] + 1 == level[v] and tr > 0:
                tr_pushed = dfs(v, min(pushed, tr))
                if tr_pushed > 0:
                    cap[e] -= tr_pushed
                    cap[e ^ 1] += tr_pushed
                    return tr_pushed
            e = nxt[e]
            ptr[u] = e
        return 0

    max_flow = 0
    INF = 10**18
    while bfs():
        for i in range(num_nodes):
            ptr[i] = head[i]
        while True:
            pushed = dfs(S, INF)
            if pushed == 0:
                break
            max_flow += pushed

    if max_flow == d:
        print("YES")
        flows = []
        for e_idx, initial_c in orig_edge_indices:
            flows.append(str(initial_c - cap[e_idx]))
        print(" ".join(flows))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
