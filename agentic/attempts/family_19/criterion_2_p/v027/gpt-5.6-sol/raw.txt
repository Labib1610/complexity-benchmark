```python
import sys
from collections import deque

class Edge:
    __slots__ = ("to", "rev", "cap")

    def __init__(self, to, rev, cap):
        self.to = to
        self.rev = rev
        self.cap = cap


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    next(it)  # Number of asset types, always 1

    s = next(it) - 1
    t = next(it) - 1
    required = next(it)

    graph = [[] for _ in range(n)]
    original_edges = []

    def add_edge(u, v, cap):
        if u == v:
            index = len(graph[u])
            forward = Edge(v, index + 1, cap)
            reverse = Edge(u, index, 0)
        else:
            forward = Edge(v, len(graph[v]), cap)
            reverse = Edge(u, len(graph[u]), 0)

        graph[u].append(forward)
        graph[v].append(reverse)
        original_edges.append((forward, cap))

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        add_edge(u, v, cap)

    if s == t:
        if required != 0:
            print("NO")
            return
        print("YES")
        print(" ".join("0" for _ in range(m)))
        return

    sys.setrecursionlimit(max(1000000, n * 2 + 10))

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            u = queue.popleft()
            for edge in graph[u]:
                if edge.cap > 0 and level[edge.to] == -1:
                    level[edge.to] = level[u] + 1
                    queue.append(edge.to)

        return level[t] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == t:
            return pushed

        while ptr[u] < len(graph[u]):
            edge = graph[u][ptr[u]]
            if edge.cap > 0 and level[edge.to] == level[u] + 1:
                sent = dfs(edge.to, min(pushed, edge.cap))
                if sent:
                    edge.cap -= sent
                    graph[edge.to][edge.rev].cap += sent
                    return sent
            ptr[u] += 1

        return 0

    flow = 0
    while flow < required and bfs():
        ptr[:] = [0] * n
        while flow < required:
            sent = dfs(s, required - flow)
            if sent == 0:
                break
            flow += sent

    if flow < required:
        print("NO")
        return

    answer = [initial_cap - edge.cap for edge, initial_cap in original_edges]
    print("YES")
    print(" ".join(map(str, answer)))


if __name__ == "__main__":
    main()
```
