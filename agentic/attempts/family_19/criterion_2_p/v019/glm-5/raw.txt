import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if total capacity from s is at least d
    total_cap_s = sum(cap for u, v, cap in edges if u == s)
    if total_cap_s < d:
        print("NO")
        return
    
    # Binary search for minimum feasible flow
    # We need to find if there exists a flow of value d
    # Using max flow algorithm
    
    # Build flow network
    # We'll use Dinic's algorithm
    
    class Dinic:
        def __init__(self, n, s, t):
            self.n = n
            self.s = s
            self.t = t
            self.adj = [[] for _ in range(n + 1)]
            self.edges = []
            
        def add_edge(self, u, v, cap, idx):
            self.edges.append((u, v, cap, 0, idx))
            self.edges.append((v, u, 0, 0, idx))
            self.adj[u].append(len(self.edges) - 2)
            self.adj[v].append(len(self.edges) - 1)
            
        def bfs(self):
            self.level = [-1] * (self.n + 1)
            q = deque([self.s])
            self.level[self.s] = 0
            while q:
                u = q.popleft()
                for ei in self.adj[u]:
                    v, cap, flow = self.edges[ei][1], self.edges[ei][2], self.edges[ei][3]
                    if cap - flow > 0 and self.level[v] == -1:
                        self.level[v] = self.level[u] + 1
                        q.append(v)
            return self.level[self.t] != -1
        
        def dfs(self, u, pushed):
            if u == self.t:
                return pushed
            for i in range(self.it[u], len(self.adj[u])):
                self.it[u] = i
                ei = self.adj[u][i]
                v = self.edges[ei][1]
                cap = self.edges[ei][2]
                flow = self.edges[ei][3]
                if cap - flow > 0 and self.level[v] == self.level[u] + 1:
                    ret = self.dfs(v, min(pushed, cap - flow))
                    if ret > 0:
                        self.edges[ei] = (self.edges[ei][0], self.edges[ei][1], self.edges[ei][2], self.edges[ei][3] + ret)
                        self.edges[ei ^ 1] = (self.edges[ei ^ 1][0], self.edges[ei ^ 1][1], self.edges[ei ^ 1][2], self.edges[ei ^ 1][3] - ret)
                        return ret
            return 0
        
        def max_flow(self):
            flow = 0
            while self.bfs():
                self.it = [0] * (self.n + 1)
                while True:
                    pushed = self.dfs(self.s, float('inf'))
                    if pushed == 0:
                        break
                    flow += pushed
            return flow
    
    dinic = Dinic(n, s, t)
    for i, (u, v, cap) in enumerate(edges):
        dinic.add_edge(u, v, cap, i)
    
    flow = dinic.max_flow()
    
    if flow < d:
        print("NO")
        return
    
    print("YES")
    result = [0] * m
    for i in range(0, len(dinic.edges), 2):
        u, v, cap, f, idx = dinic.edges[i]
        if f > 0:
            result[idx] = f
    
    # Scale down if flow > d
    if flow > d:
        # Find paths and reduce flow
        # Build residual graph
        res_adj = [[] for _ in range(n + 1)]
        for i, (u, v, cap) in enumerate(edges):
            if result[i] > 0:
                res_adj[u].append((v, i, result[i]))
        
        # Find augmenting paths from t to s (to reduce flow)
        excess = flow - d
        while excess > 0:
            # BFS to find path from t to s
            parent = [-1] * (n + 1)
            parent_edge = [-1] * (n + 1)
            q = deque([t])
            parent[t] = t
            found = False
            while q and not found:
                u = q.popleft()
                for v, idx, f in res_adj[u]:
                    if parent[v] == -1 and f > 0:
                        parent[v] = u
                        parent_edge[v] = idx
                        if v == s:
                            found = True
                            break
                        q.append(v)
            
            if not found:
                break
            
            # Find minimum flow on path
            path_flow = excess
            v = s
            while v != t:
                u = parent[v]
                idx = parent_edge[v]
                path_flow = min(path_flow, result[idx])
                v = u
            
            # Reduce flow
            v = s
            while v != t:
                u = parent[v]
                idx = parent_edge[v]
                result[idx] -= path_flow
                v = u
            
            excess -= path_flow
            
            # Rebuild residual graph
            res_adj = [[] for _ in range(n + 1)]
            for i, (u, v, cap) in enumerate(edges):
                if result[i] > 0:
                    res_adj[u].append((v, i, result[i]))
    
    print(' '.join(map(str, result)))

solve()
