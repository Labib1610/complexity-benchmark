import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    __slots__ = ('n','g','level','it')
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]
    def add_edge(self, fr, to, cap, idx=-1):
        fwd = [to, cap, None, idx]   # to, cap, rev, idx
        rev = [fr, 0, fwd, -1]
        fwd[2] = rev
        self.g[fr].append(fwd)
        self.g[to].append(rev)
        return fwd
    def bfs(self, s, t):
        level = [-1]*self.n
        q = [s]
        level[s]=0
        for v in q:
            for to, cap, rev, _ in self.g[v]:
                if cap and level[to]==-1:
                    level[to]=level[v]+1
                    q.append(to)
        self.level=level
        return level[t]!=-1
    def dfs(self, v, t, f):
        if v==t:
            return f
        for i in range(self.it[v], len(self.g[v])):
            self.it[v]=i
            to, cap, rev, _ = self.g[v][i]
            if cap and self.level[v]+1==self.level[to]:
                ret=self.dfs(to, t, min(f, cap))
                if ret:
                    self.g[v][i][1]-=ret
                    rev[1]+=ret
                    return ret
        return 0
    def max_flow(self, s, t, limit=None):
        flow=0
        INF=10**18
        while self.bfs(s,t):
            self.it=[0]*self.n
            while True:
                f=self.dfs(s,t, INF if limit is None else min(INF, limit-flow))
                if not f: break
                flow+=f
                if limit is not None and flow>=limit:
                    return flow
        return flow

def solve():
    data=list(map(int,sys.stdin.read().split()))
    if not data: 
        return
    it=iter(data)
    n=next(it); m=next(it); _=next(it)  # the 1 is ignored
    s=next(it)-1; t=next(it)-1; d=next(it)
    edges=[]
    dinic=Dinic(n)
    edge_objs=[]
    for i in range(m):
        u=next(it)-1; v=next(it)-1; cap=next(it)
        edges.append((u,v,cap))
        obj=dinic.add_edge(u,v,cap,i)
        edge_objs.append(obj)
    flow=dinic.max_flow(s,t,d)
    if flow<d:
        print("NO")
        return
    # retrieve flows on original edges
    res=[0]*m
    for obj in edge_objs:
        idx=obj[3]
        used = edges[idx][2] - obj[1]   # original cap - remaining cap
        res[idx]=used
    print("YES")
    print(' '.join(map(str,res)))

if __name__=="__main__":
    solve()
