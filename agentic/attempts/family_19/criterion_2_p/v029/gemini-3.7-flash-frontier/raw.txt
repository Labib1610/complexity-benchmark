import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    _ = int(next(iterator)) # the '1'
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # Adjacency list and edge list
    # Edge: [to, capacity, flow, rev_index]
    adj = [[] for _ in range(n + 1)]
    edges = []
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        
        idx_u = len(adj[u])
        idx_v = len(adj[v])
        if u == v:
            idx_v += 1
            
        adj[u].append([v, c, 0, idx_v])
        adj[v].append([u, 0, 0, idx_u])
        edges.append((u, idx_u))
        
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for to, cap, flow, rev in adj[u]:
                if cap - flow > 0 and level[to] == -1:
                    level[to] = level[u] + 1
                    q.append(to)
        return level[t] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == t:
            return pushed
        for cid in range(ptr[u], len(adj[u])):
            ptr[u] = cid
            to, cap, flow, rev = adj[u][cid]
            if level[u] + 1 != level[to] or cap - flow == 0:
                continue
            tr = dfs(to, min(pushed, cap - flow))
            if tr == 0:
                continue
            adj[u][cid][2] += tr
            adj[to][rev][2] -= tr
            return tr
        return 0

    total_flow = 0
    while total_flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = 0
        while total_flow < d:
            pushed = dfs(s, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow < d:
        print("NO")
    else:
        print("YES")
        res = []
        for u, idx in edges:
            res.append(str(adj[u][idx][2]))
        print(*(res))

if __name__ == '__main__':
    solve()
