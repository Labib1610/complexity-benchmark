import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
        self.level = [-1] * n
        self.it = [0] * n

    def add_edge(self, u, v, cap):
        forward = [v, len(self.graph[v]), cap]
        backward = [u, len(self.graph[u]), 0]
        self.graph[u].append(forward)
        self.graph[v].append(backward)
        return forward

    def bfs(self, source, sink):
        level = [-1] * self.n
        level[source] = 0
        queue = deque([source])

        while queue:
            u = queue.popleft()
            next_level = level[u] + 1
            for v, rev, cap in self.graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = next_level
                    queue.append(v)

        self.level = level
        return level[sink] >= 0

    def dfs(self, u, sink, pushed):
        if u == sink:
            return pushed

        graph_u = self.graph[u]
        while self.it[u] < len(graph_u):
            edge = graph_u[self.it[u]]
            v, rev, cap = edge

            if cap > 0 and self.level[v] == self.level[u] + 1:
                flow = self.dfs(v, sink, min(pushed, cap))
                if flow:
                    edge[2] -= flow
                    self.graph[v][rev][2] += flow
                    return flow

            self.it[u] += 1

        return 0

    def max_flow(self, source, sink, limit):
        total = 0
        while total < limit and self.bfs(source, sink):
            self.it = [0] * self.n
            while total < limit:
                pushed = self.dfs(source, sink, limit - total)
                if not pushed:
                    break
                total += pushed
        return total


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    iterator = iter(data)
    n = next(iterator)
    m = next(iterator)
    _ = next(iterator)

    s = next(iterator) - 1
    t = next(iterator) - 1
    d = next(iterator)

    edges = []
    for _ in range(m):
        u = next(iterator) - 1
        v = next(iterator) - 1
        cap = next(iterator)
        edges.append((u, v, cap))

    if d == 0:
        print("YES")
        print(*([0] * m))
        return

    if s == t:
        print("NO")
        return

    if d < 0:
        s, t = t, s
        d = -d

    dinic = Dinic(n)
    references = []

    for u, v, cap in edges:
        if u == v:
            references.append(None)
        else:
            edge_ref = dinic.add_edge(u, v, cap)
            references.append(edge_ref)

    if dinic.max_flow(s, t, d) != d:
        print("NO")
        return

    answer = []
    for (u, v, cap), ref in zip(edges, references):
        answer.append(0 if ref is None else cap - ref[2])

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
