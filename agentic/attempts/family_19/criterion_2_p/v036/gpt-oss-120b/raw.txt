import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = list(map(int, sys.stdin.read().strip().split()))
    if not data:
        return
    it = iter(data)
    n, m, _ = next(it), next(it), next(it)
    s, t, d = next(it)-1, next(it)-1, next(it)

    class Edge:
        __slots__ = ('to', 'rev', 'cap', 'orig')
        def __init__(self, to, rev, cap, orig=False):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.orig = orig

    graph = [[] for _ in range(n)]
    orig_edges = []

    def add_edge(fr, to, cap, idx):
        fwd = Edge(to, len(graph[to]), cap, True)
        rev = Edge(fr, len(graph[fr]), 0, False)
        graph[fr].append(fwd)
        graph[to].append(rev)
        orig_edges.append((fr, len(graph[fr]) - 1, idx))  # store position for output

    for idx in range(m):
        u, v, cap = next(it)-1, next(it)-1, next(it)
        add_edge(u, v, cap, idx)

    level = [0]*n
    it_ptr = [0]*n

    from collections import deque

    def bfs():
        for i in range(n):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in graph[v]:
                if e.cap and level[e.to] < 0:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] >= 0

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(graph[v])):
            it_ptr[v] = i
            e = graph[v][i]
            if e.cap and level[v] < level[e.to]:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    graph[e.to][e.rev].cap += ret
                    return ret
        return 0

    flow = 0
    INF = 10**18
    while flow < d and bfs():
        it_ptr = [0]*n
        while flow < d:
            pushed = dfs(s, d - flow)
            if not pushed:
                break
            flow += pushed

    if flow < d:
        print("NO")
        return

    # retrieve flows for original edges
    res = [0]*m
    for fr, pos, idx in orig_edges:
        e = graph[fr][pos]
        used = graph[e.to][e.rev].cap  # flow sent = reverse edge capacity
        res[idx] = used

    print("YES")
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
