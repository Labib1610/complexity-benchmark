from collections import defaultdict, deque

def solve():
    # Read input
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    edges = []
    graph = defaultdict(list)
    
    for i in range(m):
        u, v, cap = map(int, input().split())
        edges.append((u, v, cap))
        graph[u].append((v, cap, i))
        graph[v].append((u, 0, i))  # residual edge
    
    # Create a flow network with demands
    # We need to find a flow of value d from s to t
    
    # Use Ford-Fulkerson with BFS (Edmonds-Karp)
    def bfs():
        parent = [-1] * (n + 1)
        visited = [False] * (n + 1)
        queue = deque([s])
        visited[s] = True
        
        while queue:
            u = queue.popleft()
            
            for v, cap, idx in graph[u]:
                if not visited[v] and cap > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    
                    if v == t:
                        return parent
        
        return None
    
    # Initialize flow
    flow = [0] * m
    
    max_flow = 0
    while max_flow < d:
        parent = bfs()
        if parent is None:
            break
            
        # Find minimum capacity along the path
        path_flow = float('inf')
        curr = t
        path_edges = []
        
        while curr != s:
            prev = parent[curr]
            # Find the edge from prev to curr
            edge_found = False
            for v, cap, idx in graph[prev]:
                if v == curr and cap > 0:
                    path_flow = min(path_flow, cap)
                    path_edges.append((prev, curr, idx))
                    edge_found = True
                    break
            if not edge_found:
                break
            curr = prev
        
        if curr != s:
            break
            
        path_flow = min(path_flow, d - max_flow)
        
        # Update residual capacities and flow
        for u, v, idx in path_edges:
            # Update forward edge
            for i, (vv, cap, eidx) in enumerate(graph[u]):
                if vv == v and eidx == idx:
                    graph[u][i] = (vv, cap - path_flow, eidx)
                    if u == edges[idx][0] and v == edges[idx][1]:  # original edge
                        flow[idx] += path_flow
                    break
            
            # Update backward edge
            for i, (uu, cap, eidx) in enumerate(graph[v]):
                if uu == u and eidx == idx:
                    graph[v][i] = (uu, cap + path_flow, eidx)
                    if v == edges[idx][0] and u == edges[idx][1]:  # reverse edge
                        flow[idx] -= path_flow
                    break
        
        max_flow += path_flow
    
    if max_flow < d:
        print("NO")
        return
    
    print("YES")
    print(*flow)

solve()
