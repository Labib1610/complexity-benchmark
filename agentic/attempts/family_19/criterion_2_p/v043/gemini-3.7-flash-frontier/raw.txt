```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    # third number is 1, ignore
    
    s = int(input_data[3])
    t = int(input_data[4])
    d = int(input_data[5])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    
    idx = 6
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        cap = int(input_data[idx+2])
        idx += 3
        
        # Edge structure: [to, cap, rev_idx, orig_cap, edge_id]
        adj[u].append([v, cap, len(adj[v]), cap, i])
        adj[v].append([u, 0, len(adj[u]) - 1, 0, -1])
    
    if d == 0:
        print("YES")
        print(" ".join(["0"] * m))
        return
        
    if s == t:
        print("NO")
        return

    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)

    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            for edge in adj[v]:
                to, cap = edge[0], edge[1]
                if cap > 0 and level[to] == -1:
                    level[to] = level[v] + 1
                    q.append(to)
        return level[t] != -1

    def dfs(v, pushed):
        if pushed == 0 or v == t:
            return pushed
        for cid in range(ptr[v], len(adj[v])):
            ptr[v] = cid
            edge = adj[v][cid]
            to, cap, rev = edge[0], edge[1], edge[2]
            if level[v] + 1 != level[to] or cap == 0:
                continue
            tr = dfs(to, min(pushed, cap))
            if tr == 0:
                continue
            edge[1] -= tr
            adj[to][rev][1] += tr
            return tr
        return 0

    flow = 0
    while flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = 0
        while flow < d:
            pushed = dfs(s, d - flow)
            if pushed == 0:
                break
            flow += pushed

    if flow < d:
        print("NO")
    else:
        print("YES")
        ans = [0] * m
        for u in range(1, n + 1):
            for edge in adj[u]:
                eid = edge[4]
                if eid != -1:
                    # Original capacity - residual capacity
                    ans[eid] = edge[3] - edge[1]
        print(*(ans))

if __name__ == '__main__':
    solve()
```
