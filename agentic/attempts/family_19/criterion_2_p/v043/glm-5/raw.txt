from collections import deque

def solve():

    m = int(next(it))
    _ = int(next(it))
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]

        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, cap, i))
        adj[v].append((u, cap, i))
    
    # BFS to find augmenting paths
    def bfs():
        parent_edge = [-1] * (n + 1)
        visited = [False] * (n + 1)
        visited[s] = True
        q = deque([s])
        while q:
            u = q.popleft()
            if u == t:
                break
            for v, cap, idx in adj[u]:
                if not visited[v] and cap > 0:
                    visited[v] = True
                    parent_edge[v] = (u, idx)
                    q.append(v)
        if not visited[t]:
            return None
        path = []
        v = t
        while v != s:
            u, idx = parent_edge[v]
            path.append((u, v, idx))
            v = u
        path.reverse()
        return path
    
    flow = 0
    while flow < d:
        path = bfs()
        if path is None:
            break
        # find bottleneck
        bottleneck = float('inf')
        for u, v, idx in path:
            bottleneck = min(bottleneck, edges[idx][2])
        bottleneck = min(bottleneck, d - flow)
        # update residual capacities
        for u, v, idx in path:
            eu, ev, ecap = edges[idx]
            edges[idx] = (eu, ev, ecap - bottleneck)
            # update adjacency list
            for j in range(len(adj[eu])):
                nv, ncap, nidx = adj[eu][j]
                if nidx == idx and nv == ev:
                    adj[eu][j] = (nv, ncap - bottleneck, nidx)
                    break
            for j in range(len(adj[ev])):
                nu, ncap, nidx = adj[ev][j]
                if nidx == idx and nu == eu:
                    adj[ev][j] = (nu, ncap + bottleneck, nidx)
                    break
        flow += bottleneck
    
    if flow < d:
        print("NO")
        return
    
    # Reconstruct flow values
    # We need to track how much flow went through each edge

    # Reset
    edges = []
    adj = [[] for _ in range(n + 1)]

    # Re-read input

        adj[u].append([v, i, True])   # forward edge
        adj[v].append([u, i, False]) # backward edge
    
    flow = 0
    while flow < d2:
        # BFS
        parent = [-1] * (n2 + 1)
        parent_edge = [-1] * (n2 + 1)
        parent_dir = [False] * (n2 + 1)
        visited = [False] * (n2 + 1)
        visited[s2] = True
        q = deque([s2])
        found = False
        while q and not found:
            u = q.popleft()
            for v, idx, is_forward in adj[u]:
                if not visited[v]:
                    if is_forward and edges[idx][2] > 0:
                        visited[v] = True
                        parent[v] = u
                        parent_edge[v] = idx
                        parent_dir[v] = True
                        q.append(v)
                        if v == t2:

                    elif not is_forward and edges[idx][3] > 0:
                        visited[v] = True
                        parent[v] = u

                        parent_dir[v] = False
                        q.append(v)
                        if v == t2:

        # Find bottleneck
        bottleneck = float('inf')
        v = t2

                bottleneck = min(bottleneck, edges[idx][2])

                bottleneck = min(bottleneck, edges[idx][3])
            v = parent[v]

            else:
