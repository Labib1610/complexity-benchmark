Here’s a direct Python 3 solution for the problem:  

```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # the constant 1
    
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        edges.append((u, v, w))
        adj[u].append((v, idx))
    
    # BFS to find a path from s to t
    parent = [-1] * n
    parent_edge = [-1] * n
    visited = [False] * n
    q = deque([s])
    visited[s] = True
    
    while q:
        u = q.popleft()
        if u == t:
            break
        for v, idx in adj[u]:
            if not visited[v]:
                visited[v] = True
                parent[v] = u
                parent_edge[v] = idx
                q.append(v)
    
    if not visited[t]:
        print("NO")
        return
    
    # Find bottleneck capacity along the path
    path_edges = []
    cur = t
    min_cap = float('inf')
    while cur != s:
        idx = parent_edge[cur]
        u, v, w = edges[idx]
        min_cap = min(min_cap, w)
        path_edges.append(idx)
        cur = parent[cur]
    
    if min_cap < d:
        print("NO")
        return
    
    # Assign flow
    flow = [0] * m
    for idx in path_edges:
        flow[idx] = d
    
    print("YES")
    print(" ".join(str(x) for x in flow))

if __name__ == "__main__":
    solve()
```
