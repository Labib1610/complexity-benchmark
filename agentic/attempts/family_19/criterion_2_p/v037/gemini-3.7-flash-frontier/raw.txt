import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    _ = int(next(iterator))
    
    s = int(next(iterator)) - 1
    t = int(next(iterator)) - 1
    d = int(next(iterator))
    
    # Graph representation for Dinic
    adj = [[] for _ in range(n)]
    
    class Edge:
        __slots__ = ('to', 'cap', 'flow', 'rev')
        def __init__(self, to, cap, rev):
            self.to = to
            self.cap = cap
            self.flow = 0
            self.rev = rev

    forward_edges = []
    
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        w = int(next(iterator))
        
        e1 = Edge(v, w, None)
        e2 = Edge(u, 0, e1)
        e1.rev = e2
        
        adj[u].append(e1)
        adj[v].append(e2)
        forward_edges.append(e1)
        
    level = [-1] * n
    
    def bfs():
        for i in range(n):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for e in adj[u]:
                if e.cap - e.flow > 0 and level[e.to] == -1:
                    level[e.to] = level[u] + 1
                    q.append(e.to)
        return level[t] != -1

    ptr = [0] * n
    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        for cid in range(ptr[u], len(adj[u])):
            ptr[u] = cid
            e = adj[u][cid]
            tr = e.to
            if level[u] + 1 != level[tr] or e.cap - e.flow == 0:
                continue
            push = dfs(tr, min(pushed, e.cap - e.flow))
            if push == 0:
                continue
            e.flow += push
            e.rev.flow -= push
            return push
        return 0

    total_flow = 0
    while total_flow < d and bfs():
        ptr = [0] * n
        while total_flow < d:
            pushed = dfs(s, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed
            
    if total_flow < d:
        print("NO")
    else:
        print("YES")
        print(*(e.flow for e in forward_edges))

if __name__ == '__main__':
    solve()
