```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    # third number on line 1 (e.g. 1)
    _ = next(it)
    
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    if d == 0:
        print("YES")
        print(*(0 for _ in range(m)))
        return

    # Vertices could be 1-indexed or 0-indexed.
    # We will find the maximum vertex index to size our structures properly.
    # Also virtual source for capping total flow at d.
    
    edges_raw = []
    max_v = max(s, t)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges_raw.append((u, v, cap))
        if u > max_v: max_v = u
        if v > max_v: max_v = v
        
    src = max_v + 1
    sink = t
    
    # We'll use 0-based or 1-based indexing up to src+1
    V = src + 2
    
    head = [-1] * V
    to = []
    nxt = []
    cap = []
    
    def add_edge(u, v, c):
        idx = len(to)
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = idx
        
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = idx + 1
        return idx

    # Add edge from src to s with capacity d
    add_edge(src, s, d)
    
    orig_edge_indices = []
    for u, v, c in edges_raw:
        e_idx = add_edge(u, v, c)
        orig_edge_indices.append(e_idx)
        
    level = [-1] * V
    ptr = [-1] * V
    
    def bfs():
        for i in range(V):
            level[i] = -1
        level[src] = 0
        q = deque([src])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[sink] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == sink:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = level[v]
            if tr == level[u] + 1 and cap[e] > 0:
                push = cap[e] if pushed > cap[e] else pushed
                t_flow = dfs(v, push)
                if t_flow > 0:
                    cap[e] -= t_flow
                    cap[e ^ 1] += t_flow
                    return t_flow
            e = nxt[e]
            ptr[u] = e
        return 0

    total_flow = 0
    while bfs():
        for i in range(V):
            ptr[i] = head[i]
        while True:
            pushed = dfs(src, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed
            if total_flow == d:
                break
        if total_flow == d:
            break
            
    if total_flow == d:
        print("YES")
        flows = []
        for e_idx in orig_edge_indices:
            # Flow on forward edge is the capacity of the reverse edge
            flows.append(cap[e_idx ^ 1])
        print(*(flows))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
