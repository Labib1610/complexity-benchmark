import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    _ = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # If s == t:
    if s == t:
        if d == 0:
            print("YES")
            print(" ".join(["0"] * m))
        else:
            print("NO")
        return

    # Edge representation
    # We will store edges for Dinic's
    # Each directed edge from u to v has capacity `cap`, flow 0, and a reverse edge.
    adj = [[] for _ in range(n + 1)]
    edges = [] # list of [to, cap, flow, rev_idx, orig_edge_idx]
    
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        
        idx_forward = len(edges)
        idx_backward = idx_forward + 1
        
        edges.append([v, cap, 0, idx_backward, i])
        edges.append([u, 0, 0, idx_forward, -1])
        
        adj[u].append(idx_forward)
        adj[v].append(idx_backward)
        
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            for e_idx in adj[v]:
                to, cap, flow, _, _ = edges[e_idx]
                if cap - flow > 0 and level[to] == -1:
                    level[to] = level[v] + 1
                    q.append(to)
        return level[t] != -1

    def dfs(v, pushed):
        if pushed == 0:
            return 0
        if v == t:
            return pushed
        for cid in range(ptr[v], len(adj[v])):
            ptr[v] = cid
            e_idx = adj[v][cid]
            to, cap, flow, rev_idx, _ = edges[e_idx]
            tr = cap - flow
            if level[v] + 1 != level[to] or tr == 0:
                continue
            push = min(pushed, tr)
            pushed_flow = dfs(to, push)
            if pushed_flow == 0:
                continue
            edges[e_idx][2] += pushed_flow
            edges[rev_idx][2] -= pushed_flow
            return pushed_flow
        return 0

    total_flow = 0
    while total_flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = 0
        while total_flow < d:
            pushed = dfs(s, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed
            
    if total_flow == d:
        print("YES")
        ans = [0] * m
        for e in edges:
            orig_idx = e[4]
            if orig_idx != -1:
                ans[orig_idx] = e[2]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
