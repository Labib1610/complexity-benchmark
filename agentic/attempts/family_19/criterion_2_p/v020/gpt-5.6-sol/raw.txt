```python
import sys
from collections import deque

class Edge:
    __slots__ = ("to", "cap", "rev")

    def __init__(self, to, cap, rev):
        self.to = to
        self.cap = cap
        self.rev = rev


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    required = next(it)

    graph = [[] for _ in range(n)]
    original_edges = []

    def add_edge(u, v, cap):
        iu = len(graph[u])
        iv = len(graph[v])

        if u == v:
            forward = Edge(v, cap, iu + 1)
            backward = Edge(u, 0, iu)
        else:
            forward = Edge(v, cap, iv)
            backward = Edge(u, 0, iu)

        graph[u].append(forward)
        graph[v].append(backward)
        original_edges.append((u, iu, cap))

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        add_edge(u, v, cap)

    if required == 0:
        print("YES")
        print(*([0] * m))
        return

    if s == t:
        print("NO")
        return

    sys.setrecursionlimit(max(1_000_000, 2 * n + 100))

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            u = queue.popleft()
            next_level = level[u] + 1
            for edge in graph[u]:
                if edge.cap > 0 and level[edge.to] == -1:
                    level[edge.to] = next_level
                    queue.append(edge.to)

        return level[t] != -1

    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == t:
            return pushed

        while ptr[u] < len(graph[u]):
            edge = graph[u][ptr[u]]
            if edge.cap > 0 and level[edge.to] == level[u] + 1:
                sent = dfs(edge.to, min(pushed, edge.cap))
                if sent:
                    edge.cap -= sent
                    graph[edge.to][edge.rev].cap += sent
                    return sent
            ptr[u] += 1

        return 0

    flow = 0
    while flow < required and bfs():
        ptr[:] = [0] * n
        while flow < required:
            sent = dfs(s, required - flow)
            if sent == 0:
                break
            flow += sent

    if flow < required:
        print("NO")
        return

    answer = []
    for u, index, capacity in original_edges:
        answer.append(capacity - graph[u][index].cap)

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
```
