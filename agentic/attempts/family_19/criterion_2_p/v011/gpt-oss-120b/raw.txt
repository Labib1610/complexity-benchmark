import sys
sys.setrecursionlimit(1 << 25)

class Edge:
    __slots__ = ('to', 'rev', 'cap', 'orig_cap', 'idx')
    def __init__(self, to, rev, cap, idx=-1):
        self.to = to
        self.rev = rev
        self.cap = cap
        self.orig_cap = cap
        self.idx = idx  # index of original edge, -1 for added reverse edges

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = next(it)  # type count ignored
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    graph = [[] for _ in range(n)]
    edge_refs = [None] * m  # forward edge reference for each original edge

    def add_edge(u, v, cap, idx):
        fwd = Edge(v, len(graph[v]), cap, idx)
        rev = Edge(u, len(graph[u]), 0, -1)
        graph[u].append(fwd)
        graph[v].append(rev)

    for i in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        add_edge(u, v, cap, i)
        edge_refs[i] = graph[u][-1]  # the forward edge just added

    level = [0] * n
    it_ptr = [0] * n

    from collections import deque

    def bfs():
        for i in range(n):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in graph[v]:
                if e.cap and level[e.to] == -1:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] != -1

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(graph[v])):
            it_ptr[v] = i
            e = graph[v][i]
            if e.cap and level[e.to] == level[v] + 1:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    graph[e.to][e.rev].cap += ret
                    return ret
        return 0

    flow = 0
    INF = 10**18
    while flow < d and bfs():
        it_ptr = [0] * n
        while flow < d:
            pushed = dfs(s, d - flow)
            if not pushed:
                break
            flow += pushed

    if flow < d:
        print("NO")
        return

    # retrieve flows on original edges
    res = []
    for i in range(m):
        e = edge_refs[i]
        used = e.orig_cap - e.cap
        res.append(str(used))
    print("YES")
    print(' '.join(res))

if __name__ == "__main__":
    solve()
