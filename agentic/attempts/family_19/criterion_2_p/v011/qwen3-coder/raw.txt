from collections import defaultdict, deque

def solve():
    # Read input
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    
    # Build graph
    graph = [[] for _ in range(n + 1)]
    edges = []
    edge_index = {}
    
    for i in range(m):
        u, v, cap = map(int, input().split())
        edges.append((u, v, cap))
        edge_index[(u, v, cap)] = i
        graph[u].append((v, cap, len(graph[u])))
        
    # Ford-Fulkerson with BFS (Edmonds-Karp)
    def bfs():
        parent = [-1] * (n + 1)
        edge_used = [None] * (n + 1)
        queue = deque([s])
        parent[s] = -2  # Special value to indicate source
        
        while queue:
            node = queue.popleft()
            
            if node == t:
                break
                
            for neighbor, capacity, edge_id in graph[node]:
                if parent[neighbor] == -1 and capacity > 0:
                    parent[neighbor] = node
                    edge_used[neighbor] = (node, neighbor, capacity)
                    queue.append(neighbor)
                    
        return parent[t] != -1, parent, edge_used
    
    # Find max flow
    flow = [0] * m
    total_flow = 0
    
    while total_flow < d:
        found, parent, edge_used = bfs()
        if not found:
            print("NO")
            return
            
        # Find minimum capacity along path
        path_flow = float('inf')
        cur = t
        path_edges = []
        
        while cur != s:
            prev = parent[cur]
            u, v, cap = edge_used[cur]
            path_edges.append((u, v, cap))
            path_flow = min(path_flow, cap)
            cur = prev
            
        # Update flow and capacities
        path_flow = min(path_flow, d - total_flow)
        total_flow += path_flow
        
        for u, v, cap in path_edges:
            # Find the edge index
            for i, edge in enumerate(edges):
                if edge[0] == u and edge[1] == v:
                    flow[i] += path_flow
                    # Update residual capacity
                    edges[i] = (edge[0], edge[1], edge[2] - path_flow)
                    break
    
    if total_flow < d:
        print("NO")
        return
        
    print("YES")
    print(*flow)

solve()
