from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    one = int(data[idx])
    idx += 1
    
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    d = int(data[idx])
    idx += 1
    
    edges = []
    capacity = []
    graph = defaultdict(list)
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        cap = int(data[idx])
        idx += 1
        edges.append((u, v))
        capacity.append(cap)
        graph[u].append((v, i))
    
    # We will model this as a max-flow problem
    # Build residual graph
    # We'll use Ford-Fulkerson with BFS (Edmonds-Karp)
    
    def bfs(residual_graph, source, sink, parent_edge):
        visited = set()
        queue = deque([source])
        visited.add(source)
        parent_edge[source] = -1
        
        while queue:
            u = queue.popleft()
            for v, edge_idx in residual_graph[u]:
                if v not in visited and residual_capacity[edge_idx] > 0:
                    visited.add(v)
                    parent_edge[v] = edge_idx
                    if v == sink:
                        return True
                    queue.append(v)
        return False
    
    # Initialize residual capacity
    residual_capacity = capacity[:]
    # Build adjacency list for residual graph
    residual_graph = defaultdict(list)
    for i in range(m):
        u, v = edges[i]
        # Forward edge
        residual_graph[u].append((v, i))
        # Backward edge
        residual_graph[v].append((u, i))
    
    max_flow = 0
    parent_edge = [-1] * (n+1)
    
    while bfs(residual_graph, s, t, parent_edge):
        path_flow = float('inf')
        v = t
        while v != s:
            edge_idx = parent_edge[v]
            u_original, v_original = edges[edge_idx]
            if v == v_original:
                # Forward edge
                path_flow = min(path_flow, residual_capacity[edge_idx])
            else:
                # Backward edge
                path_flow = min(path_flow, residual_capacity[edge_idx])
            # Find the previous node in the path
            found = False
            for prev_node, e_idx in residual_graph[v]:
                if e_idx == edge_idx:
                    v = prev_node
                    found = True
                    break
            if not found:
                break
        
        v = t
        while v != s:
            edge_idx = parent_edge[v]
            u_original, v_original = edges[edge_idx]
            if v == v_original:
                # Forward edge
                residual_capacity[edge_idx] -= path_flow
            else:
                # Backward edge
                residual_capacity[edge_idx] += path_flow
            # Find the previous node in the path
            found = False
            for prev_node, e_idx in residual_graph[v]:
                if e_idx == edge_idx:
                    v = prev_node
                    found = True
                    break
            if not found:
                break
        
        max_flow += path_flow
    
    if max_flow < d:
        print("NO")
        return
    
    # Reconstruct the flow
    flow = [0] * m
    for i in range(m):
        # Original capacity - residual capacity = flow
        flow[i] = capacity[i] - residual_capacity[i]
    
    print("YES")
    print(' '.join(map(str, flow)))

solve()
