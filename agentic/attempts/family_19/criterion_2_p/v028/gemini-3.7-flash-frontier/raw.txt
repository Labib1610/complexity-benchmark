```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    n = int(next(iterator))
    m = int(next(iterator))
    _ = next(iterator)  # the '1' in line 1
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    if d == 0:
        print("YES")
        print(" ".join(["0"] * m))
        return

    # Super source S = 0
    # Nodes 1..n
    # Total nodes = n + 1
    S = 0
    T = t
    
    # Adjacency list for Dinic
    # Edge structure: [to, cap, flow, rev_idx]
    adj = [[] for _ in range(n + 1)]
    
    def add_edge(u, v, cap):
        idx_u = len(adj[u])
        idx_v = len(adj[v])
        adj[u].append([v, cap, 0, idx_v])
        adj[v].append([u, 0, 0, idx_u])
        return u, idx_u

    # Edge from S to s with capacity d
    add_edge(S, s, d)
    
    original_edges = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        u_node, idx = add_edge(u, v, cap)
        original_edges.append((u_node, idx))
        
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            for v, cap, flow, rev_idx in adj[u]:
                if cap - flow > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[T] != -1

    ptr = [0] * (n + 1)
    
    def dfs(u, pushed):
        if pushed == 0 or u == T:
            return pushed
        for cid in range(ptr[u], len(adj[u])):
            ptr[u] = cid
            v, cap, flow, rev_idx = adj[u][cid]
            tr = cap - flow
            if level[u] + 1 != level[v] or tr == 0:
                continue
            push = dfs(v, min(pushed, tr))
            if push == 0:
                continue
            adj[u][cid][2] += push
            adj[v][rev_idx][2] -= push
            return push
        return 0

    total_flow = 0
    INF = 10**18
    while bfs():
        ptr = [0] * (n + 1)
        while True:
            pushed = dfs(S, INF)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow == d:
        print("YES")
        ans = []
        for u, idx in original_edges:
            ans.append(str(adj[u][idx][2]))
        print(" ".join(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
