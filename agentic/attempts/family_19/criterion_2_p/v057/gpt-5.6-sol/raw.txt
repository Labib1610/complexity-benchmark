import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


class Edge:
    __slots__ = ("to", "rev", "cap")

    def __init__(self, to, rev, cap):
        self.to = to
        self.rev = rev
        self.cap = cap


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it)
    t = next(it)
    demand = next(it)

    graph = [[] for _ in range(n + 1)]
    original = []

    def add_edge(u, v, cap):
        forward = Edge(v, len(graph[v]), cap)
        backward = Edge(u, len(graph[u]), 0)
        graph[u].append(forward)
        graph[v].append(backward)
        return forward

    for _ in range(m):
        u = next(it)
        v = next(it)
        cap = next(it)
        if u == v:
            original.append(None)
        else:
            edge = add_edge(u, v, cap)
            original.append((edge, cap))

    if demand < 0 or (s == t and demand > 0):
        print("NO")
        return

    if demand == 0:
        print("YES")
        print(*([0] * m))
        return

    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)

    def bfs():
        level[:] = [-1] * (n + 1)
        level[s] = 0
        queue = deque([s])

        while queue:
            u = queue.popleft()
            for edge in graph[u]:
                if edge.cap > 0 and level[edge.to] == -1:
                    level[edge.to] = level[u] + 1
                    if edge.to == t:
                        return True
                    queue.append(edge.to)
        return level[t] != -1

    def dfs(u, pushed):
        if u == t:
            return pushed

        while ptr[u] < len(graph[u]):
            edge = graph[u][ptr[u]]
            if edge.cap > 0 and level[edge.to] == level[u] + 1:
                sent = dfs(edge.to, min(pushed, edge.cap))
                if sent:
                    edge.cap -= sent
                    graph[edge.to][edge.rev].cap += sent
                    return sent
            ptr[u] += 1
        return 0

    total_flow = 0
    while total_flow < demand and bfs():
        ptr[:] = [0] * (n + 1)
        while total_flow < demand:
            sent = dfs(s, demand - total_flow)
            if sent == 0:
                break
            total_flow += sent

    if total_flow < demand:
        print("NO")
        return

    flows = []
    for item in original:
        if item is None:
            flows.append(0)
        else:
            edge, initial_cap = item
            flows.append(initial_cap - edge.cap)

    print("YES")
    print(*flows)


if __name__ == "__main__":
    main()
