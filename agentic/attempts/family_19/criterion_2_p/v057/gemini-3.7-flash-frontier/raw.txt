import sys
from collections import deque

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        n_str = next(iterator)
    except StopIteration:
        return
        
    n = int(n_str)
    m = int(next(iterator))
    _ = int(next(iterator)) # the '1' indicating single pallet type
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    if d == 0:
        print("YES")
        print(" ".join(["0"] * m))
        return
        
    source = 0
    sink = t
    
    # We will map nodes 1..n to 1..n, and source to 0.
    # Total nodes: n + 1
    num_nodes = n + 1
    
    head = [-1] * num_nodes
    to = []
    nxt = []
    cap = []
    flow = []
    
    def add_edge(u, v, c):
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        head[u] = len(to) - 1
        
        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        head[v] = len(to) - 1

    # Edge from super-source to s
    add_edge(source, s, d)
    
    original_edge_indices = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        original_edge_indices.append(len(to))
        add_edge(u, v, c)
        
    level = [-1] * num_nodes
    ptr = [0] * num_nodes
    
    def bfs():
        for i in range(num_nodes):
            level[i] = -1
        level[source] = 0
        q = deque([source])
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                if cap[e] - flow[e] > 0 and level[to[e]] == -1:
                    level[to[e]] = level[v] + 1
                    q.append(to[e])
                e = nxt[e]
        return level[sink] != -1

    def dfs(v, pushed):
        if pushed == 0:
            return 0
        if v == sink:
            return pushed
        e = ptr[v]
        while e != -1:
            trg = to[e]
            if level[trg] == level[v] + 1 and cap[e] - flow[e] > 0:
                tr = dfs(trg, min(pushed, cap[e] - flow[e]))
                if tr > 0:
                    flow[e] += tr
                    flow[e ^ 1] -= tr
                    return tr
            e = nxt[e]
            ptr[v] = e
        return 0

    total_flow = 0
    while bfs():
        for i in range(num_nodes):
            ptr[i] = head[i]
        while True:
            pushed = dfs(source, float('inf'))
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow == d:
        print("YES")
        ans = [str(flow[idx]) for idx in original_edge_indices]
        print(" ".join(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
