```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # always 1
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, i))
        adj[v].append((u, i))  # for residual graph
    
    # Dinic's algorithm
    level = [0] * (n + 1)
    ptr = [0] * (n + 1)
    flow = [0] * m
    
    def bfs():
        for i in range(1, n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for v, idx in adj[u]:
                e = edges[idx]
                if e[0] == u:
                    # forward edge
                    if level[v] == -1 and flow[idx] < e[2]:
                        level[v] = level[u] + 1
                        q.append(v)
                else:
                    # backward edge
                    if level[v] == -1 and flow[idx] > 0:
                        level[v] = level[u] + 1
                        q.append(v)
        return level[t] != -1
    
    def dfs(u, pushed):
        if u == t:
            return pushed
        for i in range(ptr[u], len(adj[u])):
            ptr[u] = i
            v, idx = adj[u][i]
            e = edges[idx]
            if e[0] == u:
                # forward edge
                if level[v] != level[u] + 1 or flow[idx] >= e[2]:
                    continue
                tr = dfs(v, min(pushed, e[2] - flow[idx]))
                if tr:
                    flow[idx] += tr
                    return tr
            else:
                # backward edge
                if level[v] != level[u] + 1 or flow[idx] <= 0:
                    continue
                tr = dfs(v, min(pushed, flow[idx]))
                if tr:
                    flow[idx] -= tr
                    return tr
        return 0
    
    total = 0
    while bfs():
        ptr = [0] * (n + 1)
        while True:
            pushed = dfs(s, d - total)
            if not pushed:
                break
            total += pushed
            if total == d:
                break
        if total == d:
            break
    
    if total == d:
        print("YES")
        print(' '.join(map(str, flow)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
