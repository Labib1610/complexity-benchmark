```python
import sys

data = sys.stdin.buffer.read()
pos = 0
length = len(data)

def next_int():
    global pos
    while pos < length and data[pos] <= 32:
        pos += 1
    value = 0
    while pos < length and data[pos] > 32:
        value = value * 10 + data[pos] - 48
        pos += 1
    return value

n = next_int()
m = next_int()
_ = next_int()

s = next_int() - 1
t = next_int() - 1
demand = next_int()

adj = [[] for _ in range(n)]
to = []
capacity = []
original = []

for _ in range(m):
    u = next_int() - 1
    v = next_int() - 1
    cap = next_int()

    edge_id = len(to)
    to.append(v)
    capacity.append(cap)
    adj[u].append(edge_id)

    to.append(u)
    capacity.append(0)
    adj[v].append(edge_id + 1)

    original.append(cap)

if demand == 0 or s == t:
    sys.stdout.write("YES\n" + " ".join(["0"] * m) + "\n")
    sys.exit()

sys.setrecursionlimit(max(1_000_000, 2 * n + 10))

level = [-1] * n
ptr = [0] * n

def bfs():
    for i in range(n):
        level[i] = -1
    level[s] = 0
    queue = [s]
    head = 0

    while head < len(queue):
        v = queue[head]
        head += 1
        next_level = level[v] + 1

        for edge_id in adj[v]:
            u = to[edge_id]
            if capacity[edge_id] > 0 and level[u] == -1:
                level[u] = next_level
                queue.append(u)

    return level[t] != -1

def dfs(v, pushed):
    if v == t:
        return pushed

    edges = adj[v]
    while ptr[v] < len(edges):
        edge_id = edges[ptr[v]]
        u = to[edge_id]

        if capacity[edge_id] > 0 and level[u] == level[v] + 1:
            amount = dfs(u, min(pushed, capacity[edge_id]))
            if amount:
                capacity[edge_id] -= amount
                capacity[edge_id ^ 1] += amount
                return amount

        ptr[v] += 1

    return 0

total_flow = 0

while total_flow < demand and bfs():
    ptr = [0] * n
    while total_flow < demand:
        pushed = dfs(s, demand - total_flow)
        if pushed == 0:
            break
        total_flow += pushed

if total_flow < demand:
    sys.stdout.write("NO\n")
else:
    flows = [str(original[i] - capacity[2 * i]) for i in range(m)]
    sys.stdout.write("YES\n" + " ".join(flows) + "\n")
```
