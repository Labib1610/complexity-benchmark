import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    _k = int(next(iterator))
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # Graph representation for Dinic
    # Edge: [to, cap, flow, rev_idx, orig_edge_idx]
    adj = [[] for _ in range(n + 1)]
    
    orig_edges = []
    
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        
        idx_u = len(adj[u])
        idx_v = len(adj[v])
        if u == v:
            idx_v += 1
            
        edge_forward = [v, cap, 0, idx_v, i]
        edge_backward = [u, 0, 0, idx_u, -1]
        
        adj[u].append(edge_forward)
        adj[v].append(edge_backward)
        orig_edges.append(edge_forward)
        
    if s == t:
        if d == 0:
            print("YES")
            print(*(0 for _ in range(m)))
        else:
            print("NO")
        return

    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for edge in adj[u]:
                v, cap, flow, _, _ = edge
                if cap - flow > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] != -1

    ptr = [0] * (n + 1)
    
    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        for cid in range(ptr[u], len(adj[u])):
            ptr[u] = cid
            edge = adj[u][cid]
            v, cap, flow, rev, _ = edge
            tr = cap - flow
            if level[u] + 1 != level[v] or tr == 0:
                continue
            push = dfs(v, min(pushed, tr))
            if push == 0:
                continue
            edge[2] += push
            adj[v][rev][2] -= push
            return push
        return 0

    total_flow = 0
    while total_flow < d and bfs():
        ptr = [0] * (n + 1)
        while total_flow < d:
            pushed = dfs(s, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow == d:
        print("YES")
        ans = [e[2] for e in orig_edges]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
