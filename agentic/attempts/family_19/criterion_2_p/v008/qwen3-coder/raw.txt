from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    assert int(data[idx]) == 1
    idx += 1
    
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    d = int(data[idx])
    idx += 1
    
    edges = []
    graph = defaultdict(list)
    capacity = []
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        cap = int(data[idx])
        idx += 1
        edges.append((u, v, cap))
        graph[u].append((v, i))
        graph[v].append((u, i))
        capacity.append(cap)
    
    # Build residual graph for max flow
    # We'll use Ford-Fulkerson with BFS (Edmonds-Karp)
    
    def bfs(residual_graph, source, sink, parent):
        visited = set()
        queue = deque([source])
        visited.add(source)
        parent[source] = -1
        
        while queue:
            u = queue.popleft()
            
            for v, edge_idx in residual_graph[u]:
                if v not in visited and edge_idx >= 0:
                    queue.append(v)
                    visited.add(v)
                    parent[v] = u
                    if v == sink:
                        return True
        return False
    
    # Create residual graph structure
    # For each edge (u,v,cap), we store forward and backward edges
    res_graph = defaultdict(list)
    edge_flows = [0] * m  # To track actual flow on each edge
    
    # Initialize residual graph with forward edges
    for i in range(m):
        u, v, cap = edges[i]
        # Forward edge with capacity, backward edge with 0 capacity initially
        res_graph[u].append((v, i))      # forward edge index i
        res_graph[v].append((u, -(i+1))) # backward edge index -(i+1)
    
    def get_capacity(edge_idx):
        if edge_idx >= 0:
            return capacity[edge_idx] - edge_flows[edge_idx]
        else:
            original_idx = -edge_idx - 1
            return edge_flows[original_idx]
    
    # Max flow algorithm
    parent = {}
    max_flow = 0
    
    while bfs(res_graph, s, t, parent):
        path_flow = float('inf')
        v = t
        path_edges = []
        
        # Find minimum capacity along the path
        while v != s:
            u = parent[v]
            # Find the edge from u to v
            found_edge = None
            for next_v, edge_idx in res_graph[u]:
                if next_v == v:
                    found_edge = edge_idx
                    break
            
            if found_edge is None:
                break
                
            cap = get_capacity(found_edge)
            if cap < path_flow:
                path_flow = cap
            path_edges.append(found_edge)
            v = u
        
        # Update residual capacities
        v = t
        while v != s:
            u = parent[v]
            # Find the edge from u to v
            found_edge = None
            for next_v, edge_idx in res_graph[u]:
                if next_v == v:
                    found_edge = edge_idx
                    break
                    
            if found_edge is None:
                break
                
            if found_edge >= 0:
                edge_flows[found_edge] += path_flow
            else:
                original_idx = -found_edge - 1
                edge_flows[original_idx] -= path_flow
                
            v = u
        
        max_flow += path_flow
    
    if max_flow < d:
        print("NO")
        return
    
    # Check if we can send exactly 'd' units
    # We need to adjust flows to send exactly 'd'
    # Let's redo the flow computation to target exactly 'd'
    
    # Reset and compute flow for exactly 'd'
    edge_flows = [0] * m
    
    # Rebuild residual graph for exact flow 'd'
    def max_flow_exact(source, sink, target_flow):
        # Use scaling or just run until we reach target_flow
        total_flow = 0
        while total_flow < target_flow:
            parent = {}
            visited = set()
            queue = deque([(source, target_flow - total_flow)])
            visited.add(source)
            parent[source] = (None, None, 0)
            
            found_path = False
            while queue and not found_path:
                u, available_flow = queue.popleft()
                
                for v, edge_idx in res_graph[u]:
                    if v in visited:
                        continue
                        
                    cap = get_capacity(edge_idx)
                    if cap <= 0:
                        continue
                        
                    send_flow = min(available_flow, cap)
                    if send_flow <= 0:
                        continue
                        
                    visited.add(v)
                    parent[v] = (u, edge_idx, send_flow)
                    
                    if v == sink:
                        found_path = True
                        # Augment the path
                        path_flow = send_flow
                        current = v
                        while parent[current][0] is not None:
                            prev_node, edge_i, _ = parent[current]
                            if edge_i >= 0:
                                edge_flows[edge_i] += path_flow
                            else:
                                orig_idx = -edge_i - 1
                                edge_flows[orig_idx] -= path_flow
                            current = prev_node
                        total_flow += path_flow
                        break
                    else:
                        queue.append((v, send_flow))
                        
            if not found_path:
                break
                
        return total_flow >= target_flow
    
    success = max_flow_exact(s, t, d)
    
    if not success:
        print("NO")
        return
        
    print("YES")
    print(*edge_flows)

solve()
