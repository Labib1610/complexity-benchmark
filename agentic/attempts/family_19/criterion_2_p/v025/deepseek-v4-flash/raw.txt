We can model this as a **max-flow problem** with lower bounds of 0 and upper bounds equal to the channel capacities.  
We need to check if we can send exactly `d` units from `s` to `t`.

We’ll use Dinic’s algorithm for max flow.

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
    
    def add_edge(self, fr, to, cap):
        forward = [to, cap, None]
        backward = [fr, 0, forward]
        forward[2] = backward
        self.graph[fr].append(forward)
        self.graph[to].append(backward)
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            v = q.popleft()
            for to, cap, rev in self.graph[v]:
                if cap > 0 and self.level[to] < 0:
                    self.level[to] = self.level[v] + 1
                    q.append(to)
        return self.level[t] >= 0
    
    def dfs(self, v, t, f):
        if v == t:
            return f
        for i in range(self.it[v], len(self.graph[v])):
            self.it[v] = i
            to, cap, rev = self.graph[v][i]
            if cap > 0 and self.level[v] + 1 == self.level[to]:
                ret = self.dfs(to, t, min(f, cap))
                if ret > 0:
                    self.graph[v][i][1] -= ret
                    rev[1] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    n, m, _ = map(int, input().split())
    s, t, d = map(int, input().split())
    s -= 1
    t -= 1
    
    edges = []
    dinic = Dinic(n)
    for _ in range(m):
        u, v, cap = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v, cap))
        dinic.add_edge(u, v, cap)
    
    flow = dinic.max_flow(s, t)
    if flow < d:
        print("NO")
        return
    
    # We need to output a flow assignment of exactly d.
    # Since max flow >= d, we can just take any flow of value d.
    # We'll reconstruct from residual graph.
    # But easier: run flow and then read the forward edges' used capacity.
    # However, we need exactly d, not max flow. We can cap the flow at d.
    
    # Rebuild graph with capacity d as max possible from s.
    dinic2 = Dinic(n)
    for u, v, cap in edges:
        dinic2.add_edge(u, v, cap)
    # We'll just compute max flow and if >= d, we can output a feasible flow.
    # To get exactly d, we can add a dummy edge from t to s with capacity d
    # and find circulation, but simpler: run max flow, then reduce.
    
    # Actually, we can just run max flow and output the flow values.
    # If max flow > d, we can reduce some flows arbitrarily.
    # But for correctness, we can just output the max flow values; they satisfy
    # conservation and capacities, and total is >= d. But problem requires exactly d.
    # So we need to find a flow of value exactly d.
    
    # Approach: add super source connected to s with capacity d, and t to super sink with capacity d.
    # Then find max flow; if it equals d, we have a feasible flow of exactly d.
    
    S = n
    T = n + 1
    dinic3 = Dinic(n + 2)
    for u, v, cap in edges:
        dinic3.add_edge(u, v, cap)
    dinic3.add_edge(S, s, d)
    dinic3.add_edge(t, T, d)
    
    flow_val = dinic3.max_flow(S, T)
    if flow_val < d:
        print("NO")
        return
    
    # Extract flow on original edges
    result = [0] * m
    for idx, (u, v, cap) in enumerate(edges):
        # find the forward edge in dinic3 graph
        for e in dinic3.graph[u]:
            if e[0] == v and e[1] < cap:
                # used = cap - remaining
                result[idx] = cap - e[1]
                break
    
    print("YES")
    print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
```
