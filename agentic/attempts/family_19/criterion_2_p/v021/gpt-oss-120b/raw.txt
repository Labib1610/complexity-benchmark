```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = int(next(it))  # the constant 1
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    edges_input = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges_input.append((u, v, cap))

    # Dinic
    class Edge:
        __slots__ = ('to', 'rev', 'cap', 'orig_idx')
        def __init__(self, to, rev, cap, orig_idx=-1):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.orig_idx = orig_idx

    G = [[] for _ in range(n)]

    def add_edge(fr, to, cap, idx=-1):
        fwd = Edge(to, len(G[to]), cap, idx)
        rev = Edge(fr, len(G[fr]), 0, -1)
        G[fr].append(fwd)
        G[to].append(rev)

    for idx, (u, v, cap) in enumerate(edges_input):
        add_edge(u, v, cap, idx)

    level = [0]*n
    it_ptr = [0]*n

    from collections import deque
    def bfs():
        for i in range(n):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in G[v]:
                if e.cap and level[e.to] < 0:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] >= 0

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(G[v])):
            e = G[v][i]
            if e.cap and level[v] < level[e.to]:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    G[e.to][e.rev].cap += ret
                    return ret
            it_ptr[v] += 1
        return 0

    flow = 0
    while flow < d and bfs():
        it_ptr = [0]*n
        while flow < d:
            pushed = dfs(s, d - flow)
            if not pushed:
                break
            flow += pushed

    if flow < d:
        print("NO")
        return

    # retrieve flows for original edges
    result = [0]*m
    for u in range(n):
        for e in G[u]:
            if e.orig_idx != -1:
                # original capacity minus remaining capacity gives flow used
                used = edges_input[e.orig_idx][2] - e.cap
                result[e.orig_idx] = used

    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
```
