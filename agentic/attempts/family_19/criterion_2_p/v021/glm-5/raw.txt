import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # asset type, always 1
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[][] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, i))
    
    # Build flow network
    # Edge indices: 0 to 2*m-1
    # Original edge i -> 2*i, reverse edge 2*i + 1
    
    N = n + 1
    graph = [[] for _ in range(N)]
    
    for i, (u, v, cap) in enumerate(edges):
        idx = 2 * i
        graph[u].append([v, cap, idx])
        graph[v].append([u, 0, idx + 1])
    
    level = [0] * N
    ptr = [0] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for v, cap, _ in graph[u]:
                if cap > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] != -1
    
    def dfs(u, pushed):
        if u == t:
            return pushed
        for i in range(ptr[u], len(graph[u])):
            ptr[u] = i
            v, cap, idx = graph[u][i]
            if level[v] != level[u] + 1 or cap == 0:
                continue
            tr = dfs(v, min(pushed, cap))
            if tr:
                graph[u][i][1] -= tr
                # Find reverse edge
                for j, (nv, ncap, nidx) in enumerate(graph[v]):
                    if nidx == idx + 1:
                        graph[v][j][1] += tr
                        break
                return tr
        return 0
    
    flow = 0
    while bfs():
        ptr = [0] * N
        while True:
            pushed = dfs(s, d - flow)
            if not pushed:
                break
            flow += pushed
            if flow == d:
                break
        if flow == d:
            break
    
    if flow < d:
        print("NO")
        return
    
    print("YES")
    result = []
    for i in range(m):
        idx = 2 * i
        # Find the edge in graph to get remaining capacity
        u, v, cap = edges[i]
        for nv, ncap, nidx in graph[u]:
            if nidx == idx:
                used = cap - ncap
                result.append(str(used))
                break
    print(" ".join(result))

if __name__ == "__main__":
    solve()
