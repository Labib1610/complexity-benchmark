import sys
from collections import deque

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    # input_data[2] is 1
    s = int(input_data[3])
    t = int(input_data[4])
    d = int(input_data[5])

    if s == t:
        if d == 0:
            print("YES")
            print(" ".join(["0"] * m))
        else:
            print("NO")
        return

    # Dinic's Algorithm setup
    # Nodes: 0 to n + 1
    # Super source S = 0, Sink = t
    S = 0
    T = t
    num_nodes = n + 2

    head = [-1] * num_nodes
    to = []
    cap = []
    flow = []
    nxt = []

    def add_edge(u, v, c):
        idx = len(to)
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        head[u] = idx

        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        head[v] = idx + 1
        return idx

    # Add edge from S to s with capacity d
    add_edge(S, s, d)

    original_edge_indices = []
    idx = 6
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        e_idx = add_edge(u, v, c)
        original_edge_indices.append(e_idx)

    level = [-1] * num_nodes
    ptr = [0] * num_nodes

    def bfs():
        for i in range(num_nodes):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                tr = to[e]
                if cap[e] - flow[e] > 0 and level[tr] == -1:
                    level[tr] = level[v] + 1
                    q.append(tr)
                e = nxt[e]
        return level[T] != -1

    def dfs(v, pushed):
        if pushed == 0:
            return 0
        if v == T:
            return pushed
        e = ptr[v]
        while e != -1:
            tr = to[e]
            if level[tr] == level[v] + 1 and cap[e] - flow[e] > 0:
                tr_pushed = dfs(tr, min(pushed, cap[e] - flow[e]))
                if tr_pushed > 0:
                    flow[e] += tr_pushed
                    flow[e ^ 1] -= tr_pushed
                    return tr_pushed
            ptr[v] = nxt[e]
            e = nxt[e]
        return 0

    max_flow = 0
    INF = 10**18
    while bfs():
        for i in range(num_nodes):
            ptr[i] = head[i]
        while True:
            pushed = dfs(S, INF)
            if pushed == 0:
                break
            max_flow += pushed

    if max_flow == d:
        print("YES")
        ans = [str(flow[e_idx]) for e_idx in original_edge_indices]
        print(" ".join(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
