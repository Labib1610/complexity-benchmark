```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    n = int(next(iterator))
    m = int(next(iterator))
    _ = int(next(iterator)) # asset type count, always 1
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # 0 is super source
    # 1..n are ledgers
    # We add edge 0 -> s with capacity d
    # Target sink is t
    
    head = [-1] * (n + 1)
    to = []
    cap = []
    flow = []
    nxt = []
    
    def add_edge(u, v, c):
        idx = len(to)
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        head[u] = idx
        
        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        head[v] = idx ^ 1
        return idx

    # Super source edge
    add_edge(0, s, d)
    
    orig_edge_indices = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        idx = add_edge(u, v, c)
        orig_edge_indices.append(idx)
        
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[0] = 0
        q = deque([0])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] - flow[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[t] != -1

    ptr = [0] * (n + 1)
    
    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e] - flow[e]
            if level[v] == level[u] + 1 and tr > 0:
                push = dfs(v, min(pushed, tr))
                if push > 0:
                    flow[e] += push
                    flow[e ^ 1] -= push
                    return push
            e = nxt[e]
            ptr[u] = e
        return 0

    max_flow = 0
    while bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while True:
            pushed = dfs(0, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed
            
    if max_flow == d:
        print("YES")
        ans = [str(flow[idx]) for idx in orig_edge_indices]
        print(" ".join(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
