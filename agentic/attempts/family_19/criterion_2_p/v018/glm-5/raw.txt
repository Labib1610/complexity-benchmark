```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # product type, always 1
    
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        c = int(next(it))
        edges.append((u, v, c))
        adj[u].append(i)
        adj[v].append(i)
    
    # Build level graph using BFS
    level = [-1] * (n + 1)
    level[s] = 0
    q = deque([s])
    
    while q:
        u = q.popleft()
        for i in adj[u]:
            u_e, v_e, c_e = edges[i]
            if u == u_e:
                v = v_e
            else:
                v = u_e
            if level[v] == -1 and c_e > 0:
                level[v] = level[u] + 1
                q.append(v)
    
    if level[t] == -1:
        print("NO")
        return
    
    # Dinic's DFS
    flow = [0] * m
    it_ptr = [0] * (n + 1)
    
    def dfs(u, f):
        if u == t:
            return f
        for j in range(it_ptr[u], len(adj[u])):
            it_ptr[u] = j
            i = adj[u][j]
            u_e, v_e, c_e = edges[i]
            if u == u_e:
                v = v_e
                forward = True
            else:
                v = u_e
                forward = False
            
            if level[v] == level[u] + 1 and c_e > flow[i]:
                residual = c_e - flow[i]
                if not forward:
                    residual = flow[i]
                
                ret = dfs(v, min(f, residual))
                if ret > 0:
                    if forward:
                        flow[i] += ret
                    else:
                        flow[i] -= ret
                    return ret
        return 0
    
    total_flow = 0
    while True:
        ret = dfs(s, d - total_flow)
        if ret == 0:
            break
        total_flow += ret
        if total_flow >= d:
            break
    
    if total_flow >= d:
        print("YES")
        print(' '.join(map(str, flow)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
