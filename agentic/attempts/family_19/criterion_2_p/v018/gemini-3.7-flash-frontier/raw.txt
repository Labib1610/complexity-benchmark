import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    # input_data[2] is 1 (product type)
    s = int(input_data[3])
    t = int(input_data[4])
    d = int(input_data[5])

    if s == t:
        if d == 0:
            print("YES")
            print(" ".join(["0"] * m))
        else:
            print("NO")
        return

    # Dinic's Algorithm
    # Vertices: 0 is supersource S, 1..n are hubs
    # S = 0, T = t
    src = 0
    sink = t

    head = [-1] * (n + 1)
    to = []
    nxt = []
    cap = []
    flow = []

    def add_edge(u, v, c):
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        head[u] = len(to) - 1

        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        head[v] = len(to) - 1

    # Supersource edge: 0 -> s with capacity d
    add_edge(src, s, d)

    original_edge_indices = []
    idx = 6
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        c = int(input_data[idx + 2])
        idx += 3
        edge_idx = len(to)
        add_edge(u, v, c)
        original_edge_indices.append(edge_idx)

    # Dinic's implementation
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)

    from collections import deque

    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[src] = 0
        q = deque([src])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] - flow[e] > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[sink] != -1

    def dfs(u, pushed):
        if pushed == 0 or u == sink:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e] - flow[e]
            if level[u] + 1 == level[v] and tr > 0:
                tr = pushed if pushed < tr else tr
                push = dfs(v, tr)
                if push > 0:
                    flow[e] += push
                    flow[e ^ 1] -= push
                    return push
            ptr[u] = nxt[e]
            e = nxt[e]
        return 0

    max_flow = 0
    INF = 10**18
    while bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while True:
            pushed = dfs(src, INF)
            if pushed == 0:
                break
            max_flow += pushed

    if max_flow == d:
        print("YES")
        res = [str(flow[e_idx]) for e_idx in original_edge_indices]
        print(" ".join(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
