import sys
from collections import deque

def main():
    inp = sys.stdin.buffer

    n, m, _ = map(int, inp.readline().split())
    s, t, demand = map(int, inp.readline().split())

    size = n + 1
    head = [-1] * size
    to = []
    cap = []
    nxt = []

    def add_edge(u, v, capacity):
        idx = len(to)
        to.append(v)
        cap.append(capacity)
        nxt.append(head[u])
        head[u] = idx

        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = idx + 1

    for _ in range(m):
        u, v, c = map(int, inp.readline().split())
        add_edge(u, v, c)

    if demand == 0 or s == t:
        sys.stdout.write("YES\n" + " ".join(["0"] * m) + "\n")
        return

    level = [-1] * size
    ptr = [-1] * size
    total_flow = 0

    while total_flow < demand:
        for i in range(size):
            level[i] = -1

        level[s] = 0
        queue = deque([s])

        while queue:
            v = queue.popleft()
            edge = head[v]
            next_level = level[v] + 1
            while edge != -1:
                u = to[edge]
                if cap[edge] > 0 and level[u] == -1:
                    level[u] = next_level
                    queue.append(u)
                edge = nxt[edge]

        if level[t] == -1:
            break

        ptr[:] = head

        def send_one(limit):
            vertices = [s]
            edges = []
            minimum = [limit]

            while vertices:
                v = vertices[-1]

                if v == t:
                    pushed = minimum[-1]
                    for edge_idx in edges:
                        cap[edge_idx] -= pushed
                        cap[edge_idx ^ 1] += pushed
                    return pushed

                edge = ptr[v]
                while edge != -1:
                    u = to[edge]
                    if cap[edge] > 0 and level[u] == level[v] + 1:
                        break
                    edge = nxt[edge]
                ptr[v] = edge

                if edge == -1:
                    level[v] = -1
                    vertices.pop()
                    minimum.pop()

                    if not edges:
                        return 0

                    used_edge = edges.pop()
                    parent = vertices[-1]
                    ptr[parent] = nxt[used_edge]
                else:
                    edges.append(edge)
                    vertices.append(to[edge])
                    minimum.append(min(minimum[-1], cap[edge]))

            return 0

        while total_flow < demand:
            pushed = send_one(demand - total_flow)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow < demand:
        sys.stdout.write("NO\n")
    else:
        flows = (str(cap[2 * i + 1]) for i in range(m))
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(flows))
        sys.stdout.write("\n")

if __name__ == "__main__":
    main()
