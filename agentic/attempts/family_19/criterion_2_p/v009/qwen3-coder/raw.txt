from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    _ = int(data[idx])
    idx += 1
    
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    d = int(data[idx])
    idx += 1
    
    edges = []
    graph = defaultdict(list)
    capacity = []
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        cap = int(data[idx])
        idx += 1
        
        edges.append((u, v))
        graph[u].append((v, len(capacity)))
        graph[v].append((u, len(capacity)))  # Для обратного ребра
        capacity.append(cap)
    
    # Создаем новую сеть: исходная сеть + обратные ребра с нулевой пропускной способностью
    # Прямые ребра: 0..m-1, обратные: m..2m-1
    forward_edges = edges[:]
    forward_capacity = capacity[:]
    reverse_capacity = [0] * m
    total_edges = []
    
    adj = [[] for _ in range(n+1)]
    for i in range(m):
        u, v = forward_edges[i]
        # Прямое ребро
        adj[u].append((v, i, len(total_edges)))
        total_edges.append((u, v, forward_capacity[i], i))
        # Обратное ребро
        adj[v].append((u, i + m, len(total_edges)))
        total_edges.append((v, u, reverse_capacity[i], i + m))
    
    # Алгоритм Форда-Фалкерсона (Edmonds-Karp)
    def bfs():
        parent = [-1] * (n + 1)
        edge_index = [-1] * (n + 1)
        queue = deque([s])
        parent[s] = -2  # Посещен
        
        while queue:
            u = queue.popleft()
            for v, edge_id, idx_in_total in adj[u]:
                if parent[v] == -1:
                    # Определяем остаточную пропускную способность
                    residual_capacity = total_edges[idx_in_total][2]
                    if residual_capacity > 0:
                        parent[v] = u
                        edge_index[v] = idx_in_total
                        if v == t:
                            return parent, edge_index
                        queue.append(v)
        return None, None

    max_flow = 0
    flow = [0] * (2 * m)
    
    while True:
        parent, edge_index = bfs()
        if parent is None:
            break
            
        # Найти минимальную пропускную способность на пути
        path_flow = float('inf')
        cur = t
        path = []
        while cur != s:
            idx_in_total = edge_index[cur]
            path.append(idx_in_total)
            u = total_edges[idx_in_total][0]
            residual_capacity = total_edges[idx_in_total][2]
            path_flow = min(path_flow, residual_capacity)
            cur = u
            
        if path_flow == 0:
            break
            
        # Обновить остаточные пропускные способности и потоки
        for idx_in_total in path:
            total_edges[idx_in_total] = (
                total_edges[idx_in_total][0],
                total_edges[idx_in_total][1],
                total_edges[idx_in_total][2] - path_flow,
                total_edges[idx_in_total][3]
            )
            u, v, _, original_id = total_edges[idx_in_total]
            # Найти обратное ребро
            for next_node, edge_id, rev_idx in adj[v]:
                if next_node == u and (edge_id ^ original_id) >= m:
                    total_edges[rev_idx] = (
                        total_edges[rev_idx][0],
                        total_edges[rev_idx][1],
                        total_edges[rev_idx][2] + path_flow,
                        total_edges[rev_idx][3]
                    )
                    break
        
        max_flow += path_flow
        if max_flow >= d:
            break
    
    if max_flow < d:
        print("NO")
        return
    
    # Восстановление потоков на прямых ребрах
    result_flow = [0] * m
    for i in range(m):
        result_flow[i] = forward_capacity[i] - total_edges[2*i][2]
    
    print("YES")
    print(' '.join(map(str, result_flow)))

solve()
