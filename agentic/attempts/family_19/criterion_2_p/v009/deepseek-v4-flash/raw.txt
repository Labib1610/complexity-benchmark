We can solve this by modeling it as a maximum flow problem. Since all capacities are integers and we need a feasible integer flow, using the standard Ford–Fulkerson or Dinic algorithm works. Here is one implementation:  

```python
from collections import deque
import sys

def dinic(n, edges, s, t):
    graph = [[] for _ in range(n)]
    edge_index = []
    for u, v, cap in edges:
        graph[u].append([v, cap, len(graph[v])])
        graph[v].append([u, 0, len(graph[u]) - 1])
    flow = 0
    INF = 10**18
    while True:
        level = [-1] * n
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for v, cap, rev in graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        if level[t] < 0:
            break
        it = [0] * n
        def dfs(u, f):
            if u == t:
                return f
            while it[u] < len(graph[u]):
                v, cap, rev = graph[u][it[u]]
                if cap > 0 and level[u] + 1 == level[v]:
                    ret = dfs(v, min(f, cap))
                    if ret > 0:
                        graph[u][it[u]][1] -= ret
                        graph[v][rev][1] += ret
                        return ret
                it[u] += 1
            return 0
        while True:
            pushed = dfs(s, INF)
            if pushed == 0:
                break
            flow += pushed
    return flow, graph

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # constant 1
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    edges = []
    for i in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))
    
    maxflow, graph = dinic(n, edges, s, t)
    if maxflow < d:
        print("NO")
        return
    
    # Extract flow on original edges
    flow_on_edges = []
    for i, (u, v, cap_orig) in enumerate(edges):
        # find reverse edge in graph[u] to see flow
        found = False
        for vv, cap_res, rev in graph[u]:
            if vv == v and cap_orig - cap_res == cap_orig - cap_res:
                # This should match the forward edge; reverse edge holds flow used
                flow_on_edges.append(cap_orig - cap_res)
                found = True
                break
        if not found:
            # Fallback — this should not happen
            flow_on_edges.append(0)
    
    print("YES")
    print(" ".join(str(f) for f in flow_on_edges))

if __name__ == "__main__":
    main()
```
