import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]

    def add_edge(self, u, v, c):
        fwd = [v, c, None]   # to, cap, rev
        rev = [u, 0, fwd]
        fwd[2] = rev
        self.adj[u].append(fwd)
        self.adj[v].append(rev)
        return fwd

    def bfs(self, s, t, level):
        from collections import deque
        q = deque([s])
        level[:] = [-1]*self.n
        level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, rev in self.adj[u]:
                if cap and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] != -1

    def dfs(self, u, t, f, level, it):
        if u == t:
            return f
        for i in range(it[u], len(self.adj[u])):
            it[u] = i
            v, cap, rev = self.adj[u][i]
            if cap and level[v] == level[u] + 1:
                ret = self.dfs(v, t, min(f, cap), level, it)
                if ret:
                    self.adj[u][i][1] -= ret
                    rev[1] += ret
                    return ret
        return 0

    def max_flow(self, s, t, limit=None):
        flow = 0
        level = [0]*self.n
        INF = 10**18
        while self.bfs(s, t, level):
            it = [0]*self.n
            while True:
                pushed = self.dfs(s, t, INF if limit is None else limit - flow, level, it)
                if not pushed:
                    break
                flow += pushed
                if limit is not None and flow >= limit:
                    return flow
        return flow

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = next(it)  # constant 1 ignored
    s = int(next(it))-1; t = int(next(it))-1; d = int(next(it))
    dinic = Dinic(n)
    edge_refs = []
    caps = []
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1; cap = int(next(it))
        caps.append(cap)
        e = dinic.add_edge(u, v, cap)
        edge_refs.append(e)

    flow = dinic.max_flow(s, t)
    if flow < d:
        print("NO")
        return
    # need to reduce flow to exactly d
    excess = flow - d
    if excess > 0:
        # try to send excess from t to s in residual graph
        reduced = dinic.max_flow(t, s, excess)
        if reduced < excess:
            print("NO")
            return
    # output flows
    res = []
    for e, cap in zip(edge_refs, caps):
        used = cap - e[1]
        res.append(str(used))
    print("YES")
    print(" ".join(res))

if __name__ == "__main__":
    solve()
