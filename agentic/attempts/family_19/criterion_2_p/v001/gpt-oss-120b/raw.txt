import sys
sys.setrecursionlimit(1 << 25)

class Edge:
    __slots__ = ('to', 'rev', 'cap', 'orig', 'idx')
    def __init__(self, to, rev, cap, orig, idx):
        self.to = to
        self.rev = rev
        self.cap = cap          # residual capacity
        self.orig = orig        # original capacity
        self.idx = idx          # index in input order, -1 for auxiliary edges

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    edges_input = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges_input.append((u, v, cap))

    g = [[] for _ in range(n)]

    def add_edge(u, v, cap, idx):
        fwd = Edge(v, len(g[v]), cap, cap, idx)
        rev = Edge(u, len(g[u]), 0, 0, -1)
        g[u].append(fwd)
        g[v].append(rev)

    for idx, (u, v, cap) in enumerate(edges_input):
        add_edge(u, v, cap, idx)

    level = [0] * n
    it_ptr = [0] * n

    from collections import deque

    def bfs():
        for i in range(n):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in g[v]:
                if e.cap and level[e.to] == -1:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] != -1

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(g[v])):
            it_ptr[v] = i
            e = g[v][i]
            if e.cap and level[e.to] == level[v] + 1:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    g[e.to][e.rev].cap += ret
                    return ret
        return 0

    flow = 0
    INF = 10**18
    while flow < d and bfs():
        it_ptr = [0] * n
        while flow < d:
            pushed = dfs(s, d - flow)
            if not pushed:
                break
            flow += pushed

    if flow < d:
        print("NO")
        return

    # retrieve flows on original edges
    res = [0] * m
    for u in range(n):
        for e in g[u]:
            if e.idx != -1:
                used = e.orig - e.cap
                res[e.idx] = used

    print("YES")
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
