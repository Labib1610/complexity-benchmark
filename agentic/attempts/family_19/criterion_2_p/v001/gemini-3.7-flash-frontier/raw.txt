import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it)) # 1
    
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    # Graph representation for Dinic's algorithm
    head = [-1] * (n + 1)
    to = []
    cap = []
    flow = []
    nxt = []
    edge_idx = []
    
    def add_edge(u, v, c, idx):
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        edge_idx.append(idx)
        head[u] = len(to) - 1
        
        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        edge_idx.append(-1)
        head[v] = len(to) - 1

    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        c = int(next(it))
        add_edge(u, v, c, i)
        
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] - flow[e] > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[t] != -1

    ptr = [0] * (n + 1)
    
    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        e = ptr[u]
        while e != -1:
            v = to[e]
            tr = cap[e] - flow[e]
            if level[u] + 1 == level[v] and tr > 0:
                push = min(pushed, tr)
                res = dfs(v, push)
                if res > 0:
                    flow[e] += res
                    flow[e ^ 1] -= res
                    return res
            e = nxt[e]
            ptr[u] = e
        return 0

    total_flow = 0
    while total_flow < d and bfs():
        for i in range(n + 1):
            ptr[i] = head[i]
        while total_flow < d:
            pushed = dfs(s, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow < d:
        print("NO")
    else:
        print("YES")
        ans = [0] * m
        for e in range(0, len(to), 2):
            idx = edge_idx[e]
            if idx != -1:
                ans[idx] = flow[e]
        print(*(ans[i] for i in range(m)))

if __name__ == '__main__':
    solve()
