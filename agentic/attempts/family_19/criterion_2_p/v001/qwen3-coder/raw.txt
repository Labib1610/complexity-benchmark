from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    _ = int(data[idx])
    idx += 1
    
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    d = int(data[idx])
    idx += 1
    
    edges = []
    graph = defaultdict(list)
    capacity = []
    edge_index = {}
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        cap = int(data[idx])
        idx += 1
        
        edges.append((u, v))
        graph[u].append((v, len(capacity)))
        graph[v].append((u, len(capacity)))  # Add reverse edge
        capacity.append(cap)
        capacity.append(0)  # Reverse edge capacity
        edge_index[(u, v, i)] = len(capacity) - 2
    
    # Add reverse edges for all original edges
    # We already added them above
    
    # Ford-Fulkerson with BFS (Edmonds-Karp)
    def bfs():
        parent = [-1] * (n + 1)
        parent_edge = [-1] * (n + 1)
        visited = [False] * (n + 1)
        queue = deque([s])
        visited[s] = True
        
        while queue:
            u = queue.popleft()
            for v, edge_idx in graph[u]:
                if not visited[v] and capacity[edge_idx] > 0:
                    visited[v] = True
                    parent[v] = u
                    parent_edge[v] = edge_idx
                    queue.append(v)
                    if v == t:
                        return parent, parent_edge
        return None, None
    
    max_flow = 0
    flow = [0] * len(capacity)
    
    while True:
        parent, parent_edge = bfs()
        if parent is None:
            break
            
        # Find minimum capacity along the path
        path_flow = float('inf')
        s_node = t
        while s_node != s:
            edge_idx = parent_edge[s_node]
            path_flow = min(path_flow, capacity[edge_idx])
            s_node = parent[s_node]
        
        # Update capacities and flows
        v = t
        while v != s:
            u = parent[v]
            edge_idx = parent_edge[v]
            capacity[edge_idx] -= path_flow
            capacity[edge_idx ^ 1] += path_flow
            flow[edge_idx] += path_flow
            flow[edge_idx ^ 1] -= path_flow
            v = u
        
        max_flow += path_flow
        if max_flow >= d:
            break
    
    if max_flow < d:
        print("NO")
        return
    
    # Extract flow values for original edges
    result = []
    for i in range(m):
        u, v = edges[i]
        edge_idx = edge_index[(u, v, i)]
        result.append(str(flow[edge_idx]))
    
    print("YES")
    print(" ".join(result))

solve()
