import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    demand = next(it)

    adj = [[] for _ in range(n)]
    to = []
    residual = []
    original_caps = []

    for edge_index in range(m):
        u = next(it) - 1
        v = next(it) - 1
        capacity = next(it)

        forward = len(to)
        to.append(v)
        residual.append(capacity)
        adj[u].append(forward)

        backward = len(to)
        to.append(u)
        residual.append(0)
        adj[v].append(backward)

        original_caps.append(capacity)

    if demand == 0 or s == t:
        print("YES")
        print(*([0] * m))
        return

    sys.setrecursionlimit(max(1000000, n * 2 + 10))
    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            v = queue.popleft()
            for edge in adj[v]:
                nxt = to[edge]
                if residual[edge] > 0 and level[nxt] == -1:
                    level[nxt] = level[v] + 1
                    if nxt == t:
                        return True
                    queue.append(nxt)

        return level[t] != -1

    def dfs(v, pushed):
        if v == t:
            return pushed

        while ptr[v] < len(adj[v]):
            edge = adj[v][ptr[v]]
            nxt = to[edge]

            if residual[edge] > 0 and level[nxt] == level[v] + 1:
                amount = dfs(nxt, min(pushed, residual[edge]))
                if amount:
                    residual[edge] -= amount
                    residual[edge ^ 1] += amount
                    return amount

            ptr[v] += 1

        return 0

    total_flow = 0

    while total_flow < demand and bfs():
        ptr[:] = [0] * n
        while total_flow < demand:
            pushed = dfs(s, demand - total_flow)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow < demand:
        print("NO")
        return

    flows = [
        original_caps[i] - residual[2 * i]
        for i in range(m)
    ]

    print("YES")
    print(*flows)

if __name__ == "__main__":
    main()
