import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))
    
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    for j in range(m):
        u = int(next(it))
        v = int(next(it))
        C = int(next(it))
        edges.append((u, v, C))
        adj[u].append((v, j))
        adj[v].append((u, j))
    
    # Check connectivity
    visited = [False] * (n + 1)
    q = deque([s])
    visited[s] = True
    while q:
        u = q.popleft()
        for v, idx in adj[u]:
            if not visited[v]:
                visited[v] = True
                q.append(v)
    
    if not visited[t]:
        print("NO")
        return
    
    # BFS to find shortest path from s to t
    parent = [-1] * (n + 1)
    parent_edge = [-1] * (n + 1)
    q = deque([s])
    parent[s] = s
    
    while q:
        u = q.popleft()
        if u == t:
            break
        for v, idx in adj[u]:
            if parent[v] == -1:
                parent[v] = u
                parent_edge[v] = idx
                q.append(v)
    
    # Reconstruct path
    path_edges = []
    path_nodes = [t]
    curr = t
    while curr != s:
        idx = parent_edge[curr]
        prev = parent[curr]
        u, v, C = edges[idx]
        if u == prev:
            path_edges.append((idx, 1))
        else:
            path_edges.append((idx, -1))
        curr = prev
        path_nodes.append(curr)
    
    path_edges.reverse()
    path_nodes.reverse()
    
    # Find minimum capacity on path
    min_cap = float('inf')
    for idx, direction in path_edges:
        min_cap = min(min_cap, edges[idx][2])
    
    if min_cap < d:
        print("NO")
        return
    
    # Assign values
    x = [0] * m
    for idx, direction in path_edges:
        if direction == 1:
            x[idx] = d
        else:
            x[idx] = -d
    
    # Verify constraints
    P = [0] * (n + 1)
    Q = [0] * (n + 1)
    for j in range(m):
        u, v, C = edges[j]
        val = x[j]
        P[v] += val
        Q[u] += val
    
    valid = True
    for i in range(1, n + 1):
        expected = 0
        if i == s:
            expected = -d
        elif i == t:
            expected = d
        if P[i] - Q[i] != expected:
            valid = False
            break
    
    if not valid:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, x)))

if __name__ == "__main__":
    solve()
