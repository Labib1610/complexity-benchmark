import sys
from collections import defaultdict, deque

def read_input():
    lines = []
    for line in sys.stdin:
        lines.append(line.strip())
    return lines

def parse_input(lines):
    n, m, _ = map(int, lines[0].split())
    s, t, d = map(int, lines[1].split())
    variables = []
    for i in range(2, 2 + m):
        u, v, C = map(int, lines[i].split())
        variables.append((u, v, C))
    return n, m, s, t, d, variables

def build_graph(n, m, s, t, d, variables):
    # Create constraints
    P = defaultdict(list)  # P[i] contains variables where v == i
    Q = defaultdict(list)  # Q[i] contains variables where u == i
    
    for j in range(m):
        u, v, C = variables[j]
        P[v].append(j)
        Q[u].append(j)
    
    # Build the system: P[i] - Q[i] = E[i]
    # E[s] = -d, E[t] = d, others = 0
    E = [0] * (n + 1)
    E[s] = -d
    E[t] = d
    
    return P, Q, E

def solve_system(n, m, s, t, d, variables):

    #   +1 to constraint v (in P[v])

    # and sum_{j: v_j = t} x_j - sum_{j: u_j = t} x_j = d
    
    # A simple approach: find paths from s to t and push flow along them

        u, v, _ = variables[j]
        P_sum[v] += x[j]
        Q_sum[u] += x[j]
