import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    __slots__ = ('n','g','level','it')
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]
    def add_edge(self, fr, to, cap):
        fwd = [to, cap, 0, None]   # to, cap, flow, rev index
        rev = [fr, 0, 0, None]
        fwd[3] = len(self.g[to])
        rev[3] = len(self.g[fr])
        self.g[fr].append(fwd)
        self.g[to].append(rev)
        return fwd  # return forward edge reference
    def bfs(self, s, t):
        level = [-1]*self.n
        q = [s]
        level[s]=0
        for v in q:
            for to, cap, flow, rev in self.g[v]:
                if cap-flow>0 and level[to]==-1:
                    level[to]=level[v]+1
                    q.append(to)
        self.level=level
        return level[t]!=-1
    def dfs(self, v, t, f):
        if v==t:
            return f
        for i in range(self.it[v], len(self.g[v])):
            self.it[v]=i
            to, cap, flow, rev = self.g[v][i]
            if cap-flow>0 and self.level[v]+1==self.level[to]:
                ret=self.dfs(to, t, min(f, cap-flow))
                if ret:
                    self.g[v][i][2]+=ret
                    self.g[to][rev][2]-=ret
                    return ret
        return 0
    def max_flow(self, s, t):
        flow=0
        INF=10**18
        while self.bfs(s,t):
            self.it=[0]*self.n
            while True:
                f=self.dfs(s,t,INF)
                if not f: break
                flow+=f
        return flow

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _type = int(next(it))  # type is always 1
    s = int(next(it))-1; t = int(next(it))-1; d = int(next(it))
    edges_input = []
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1; cap = int(next(it))
        edges_input.append((u,v,cap))
    dinic = Dinic(n)
    edge_refs = []
    for u,v,cap in edges_input:
        ref = dinic.add_edge(u, v, cap)
        edge_refs.append(ref)
    maxflow = dinic.max_flow(s, t)
    if maxflow < d:
        print("NO")
        return
    # we have flow >= d, need exactly d.
    # Reduce excess flow if any by sending back along any augmenting path from t to s.
    # Simple way: add a super source/sink with capacity (maxflow-d) and push back.
    # Instead, we can run another maxflow on residual graph from t to s with limit maxflow-d.
    excess = maxflow - d
    if excess > 0:
        # run flow from t to s to cancel excess
        cancel = Dinic(n)
        # copy residual capacities
        for v in range(n):
            for to, cap, flow, rev in dinic.g[v]:
                residual = cap - flow
                if residual>0:
                    cancel.add_edge(v, to, residual)
        reduced = cancel.max_flow(t, s)
        # reduced should be at least excess, but we only need excess
        # we don't need to reconstruct exact flows, because original edge flows remain valid
        # after cancelling, the net flow from s to t becomes d.
        # However, we must ensure we actually cancelled exactly excess.
        # Since cancel flow may be less, but max possible is excess, we can trust it.
        # (In feasible instance, it will be equal.)
    # retrieve flows
    result = []
    for ref in edge_refs:
        used = ref[2]  # flow stored in forward edge
        result.append(str(used))
    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    solve()
