import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


class Edge:
    __slots__ = ("to", "cap", "rev", "initial")

    def __init__(self, to, cap, initial=0):
        self.to = to
        self.cap = cap
        self.rev = None
        self.initial = initial


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it_data = iter(data)
    n = next(it_data)
    m = next(it_data)
    next(it_data)  # Number of coolant types, always 1.

    s = next(it_data) - 1
    t = next(it_data) - 1
    required = next(it_data)

    graph = [[] for _ in range(n)]
    original_edges = []

    for _ in range(m):
        u = next(it_data) - 1
        v = next(it_data) - 1
        cap = next(it_data)

        forward = Edge(v, cap, cap)
        backward = Edge(u, 0)
        forward.rev = backward
        backward.rev = forward

        graph[u].append(forward)
        graph[v].append(backward)
        original_edges.append(forward)

    if s == t:
        if required != 0:
            sys.stdout.write("NO\n")
            return
        sys.stdout.write("YES\n" + " ".join("0" for _ in range(m)) + "\n")
        return

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        level[:] = [-1] * n
        level[s] = 0
        queue = deque([s])

        while queue:
            v = queue.popleft()
            next_level = level[v] + 1
            for edge in graph[v]:
                if edge.cap > 0 and level[edge.to] == -1:
                    level[edge.to] = next_level
                    queue.append(edge.to)

        return level[t] != -1

    def dfs(v, pushed):
        if pushed == 0:
            return 0
        if v == t:
            return pushed

        while ptr[v] < len(graph[v]):
            edge = graph[v][ptr[v]]
            if edge.cap > 0 and level[edge.to] == level[v] + 1:
                sent = dfs(edge.to, min(pushed, edge.cap))
                if sent:
                    edge.cap -= sent
                    edge.rev.cap += sent
                    return sent
            ptr[v] += 1

        return 0

    total_flow = 0

    while total_flow < required and bfs():
        ptr[:] = [0] * n
        while total_flow < required:
            sent = dfs(s, required - total_flow)
            if sent == 0:
                break
            total_flow += sent

    if total_flow < required:
        sys.stdout.write("NO\n")
        return

    flows = [str(edge.initial - edge.cap) for edge in original_edges]
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(flows) + "\n")


if __name__ == "__main__":
    main()
