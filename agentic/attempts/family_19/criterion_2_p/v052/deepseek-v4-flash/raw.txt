We can model this as a max-flow problem.  
We need to send exactly `d` units from `s` to `t` through the given directed graph with capacities.  
If the max flow is at least `d`, we can output the flow on each edge.

Here’s the implementation:

```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().strip().split()
    n, m, _ = map(int, input_data[:3])
    s, t, d = map(int, input_data[3:6])
    edges = []
    idx = 6
    graph = [[] for _ in range(n + 1)]
    edge_index = 0

    for _ in range(m):
        u, v, cap = map(int, input_data[idx:idx+3])
        idx += 3
        graph[u].append([v, cap, edge_index])
        graph[v].append([u, 0, edge_index])
        edges.append([u, v, cap, 0])  # u, v, cap, flow
        edge_index += 1

    # Dinic's algorithm
    level = [-1] * (n + 1)
    it = [0] * (n + 1)

    def bfs():
        for i in range(1, n + 1):
            level[i] = -1
        q = deque()
        q.append(s)
        level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, eid in graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] >= 0

    def dfs(u, f):
        if u == t:
            return f
        while it[u] < len(graph[u]):
            v, cap, eid = graph[u][it[u]]
            if cap > 0 and level[v] == level[u] + 1:
                ret = dfs(v, min(f, cap))
                if ret > 0:
                    # update residual
                    graph[u][it[u]][1] -= ret
                    # find reverse edge
                    for j in range(len(graph[v])):
                        if graph[v][j][2] == eid and graph[v][j][0] == u:
                            graph[v][j][1] += ret
                            break
                    edges[eid][3] += ret
                    return ret
            it[u] += 1
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        for i in range(1, n + 1):
            it[i] = 0
        while True:
            f = dfs(s, INF)
            if f == 0:
                break
            flow += f

    if flow >= d:
        print("YES")
        print(" ".join(str(edges[i][3]) for i in range(m)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
