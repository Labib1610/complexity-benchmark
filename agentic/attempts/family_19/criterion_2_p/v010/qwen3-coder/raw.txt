from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx+1])
    k = int(data[idx+2])
    idx += 3
    
    s = int(data[idx]) - 1  # Convert to 0-indexed
    t = int(data[idx+1]) - 1
    d = int(data[idx+2])
    idx += 3
    
    edges = []
    graph = [[] for _ in range(n)]
    capacity = []
    
    for i in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        cap = int(data[idx+2])
        idx += 3
        edges.append((u, v, cap))
        graph[u].append((v, i, cap))  # (to_node, edge_index, capacity)
        graph[v].append((u, i, 0))    # Reverse edge with 0 capacity initially
        capacity.append(cap)
    
    # We'll model this as a min-cost max-flow problem
    # But since we don't have costs, we can use a standard max-flow algorithm
    # However, we need to ensure exact flow values at source and sink
    
    # Let's create a super-source and super-sink
    # Connect super-source to 's' with demand d
    # Connect 't' to super-sink with demand d
    # For all other nodes, balance flow (in = out)
    
    # Actually, let's think of this differently.
    # This is a circulation problem with demands.
    # But we can convert it to a max-flow problem:
    # 1. Add a super-source SS and super-sink TT
    # 2. Connect SS to s with capacity d
    # 3. Connect t to TT with capacity d
    # 4. For each edge, keep its capacity
    # 5. Run max-flow from SS to TT, should get exactly d flow
    
    # But we also need to handle the fact that we want EXACTLY d out of s and exactly d into t
    # So we can do:
    # 1. Connect SS to s with capacity d
    # 2. Connect t to TT with capacity d
    # 3. All original edges have their capacities
    # 4. Run max flow SS to TT, check if we get exactly d
    
    # Actually, simpler approach:
    # Model as finding a feasible flow where:
    # - net flow out of s is d
    # - net flow into t is d
    # - conservation at all other nodes
    
    # We can solve this using a standard reduction to max-flow:
    # Create new source S and sink T
    # Connect S to s with capacity d
    # Connect t to T with capacity d
    # Original edges keep their capacities
    # Find max flow S to T. If it's < d, impossible.
    # If it's = d, we have a solution.
    
    # But we also need to extract the actual flow values on edges.
    
    # Let's implement Ford-Fulkerson with BFS (Edmonds-Karp)
    
    # Number of nodes: n + 2 (original nodes + S + T)
    SOURCE = n
    SINK = n + 1
    
    # Build adjacency list for residual graph
    # We'll store for each edge: (to, reverse_edge_index, capacity, flow_index)
    adj = [[] for _ in range(n + 2)]
    flow_edges = []  # Store (from, to, capacity, reverse) for each original edge pair
    
    # Add edges from original graph
    edge_map = {}  # Map from edge index to flow_edge index
    for i in range(m):
        u, v, cap = edges[i]
        # Forward edge
        fwd_idx = len(flow_edges)
        flow_edges.append((u, v, cap, len(adj[v])))  # to, rev_idx in v's adj
        adj[u].append(len(flow_edges)-1)
        # Backward edge
        back_idx = len(flow_edges)
        flow_edges.append((v, u, 0, len(adj[u])-1))  # to, rev_idx in u's adj
        adj[v].append(len(flow_edges)-1)
        edge_map[i] = fwd_idx
    
    # Connect SOURCE to s with capacity d
    fwd_idx = len(flow_edges)
    flow_edges.append((s, SOURCE, d, len(adj[SOURCE])))
    adj[SOURCE].append(len(flow_edges)-1)
    back_idx = len(flow_edges)
    flow_edges.append((SOURCE, s, 0, len(adj[s])-1))
    adj[s].append(len(flow_edges)-1)
    
    # Connect t to SINK with capacity d
    fwd_idx = len(flow_edges)
    flow_edges.append((SINK, t, d, len(adj[t])))
    adj[SINK].append(len(flow_edges)-1)
    back_idx = len(flow_edges)
    flow_edges.append((t, SINK, 0, len(adj[SINK])-1))
    adj[t].append(len(flow_edges)-1)
    
    def bfs():
        parent = [-1] * (n + 2)
        parent_edge = [-1] * (n + 2)
        q = deque([SOURCE])
        parent[SOURCE] = -2  # Special marker for source
        
        while q:
            u = q.popleft()
            for edge_idx in adj[u]:
                to, rev_idx, cap, _ = flow_edges[edge_idx]
                if parent[to] == -1 and cap > 0:
                    parent[to] = u
                    parent_edge[to] = edge_idx
                    if to == SINK:
                        return parent, parent_edge
                    q.append(to)
        return parent, parent_edge
    
    def edmonds_karp():
        total_flow = 0
        while True:
            parent, parent_edge = bfs()
            if parent[SINK] == -1:
                break
            
            # Find minimum capacity along the path
            path_flow = float('inf')
            cur = SINK
            while cur != SOURCE:
                edge_idx = parent_edge[cur]
                path_flow = min(path_flow, flow_edges[edge_idx][2])
                cur = parent[cur]
            
            # Update residual capacities
            cur = SINK
            while cur != SOURCE:
                edge_idx = parent_edge[cur]
                flow_edges[edge_idx] = (
                    flow_edges[edge_idx][0],
                    flow_edges[edge_idx][1],
                    flow_edges[edge_idx][2] - path_flow,
                    flow_edges[edge_idx][3]
                )
                
                # Update reverse edge
                to, _, _, rev_idx = flow_edges[edge_idx]
                rev_edge_idx = adj[to][rev_idx]
                flow_edges[rev_edge_idx] = (
                    flow_edges[rev_edge_idx][0],
                    flow_edges[rev_edge_idx][1],
                    flow_edges[rev_edge_idx][2] + path_flow,
                    flow_edges[rev_edge_idx][3]
                )
                cur = parent[cur]
            
            total_flow += path_flow
        
        return total_flow
    
    max_flow = edmonds_karp()
    
    if max_flow < d:
        print("NO")
        return
    
    # Extract flows on original edges
    result = [0] * m
    for i in range(m):
        fwd_idx = edge_map[i]
        u, v, _, _ = flow_edges[fwd_idx]
        # The actual flow is the original capacity minus current residual capacity
        original_cap = edges[i][2]
        current_cap = flow_edges[fwd_idx][2]
        flow = original_cap - current_cap
        result[i] = flow
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
