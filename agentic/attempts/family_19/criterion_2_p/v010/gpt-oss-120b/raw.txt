import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    __slots__ = ('n', 'g', 'level', 'it')
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, fr, to, cap, idx=-1):
        # forward edge
        f = [to, cap, None, idx]
        # backward edge
        b = [fr, 0, f, -1]
        f[2] = b
        self.g[fr].append(f)
        self.g[to].append(b)

    def bfs(self, s, t):
        level = [-1] * self.n
        q = [s]
        level[s] = 0
        for v in q:
            for to, cap, rev, _ in self.g[v]:
                if cap and level[to] < 0:
                    level[to] = level[v] + 1
                    q.append(to)
        self.level = level
        return level[t] >= 0

    def dfs(self, v, t, f):
        if v == t:
            return f
        for i in range(self.it[v], len(self.g[v])):
            self.it[v] = i
            to, cap, rev, _ = self.g[v][i]
            if cap and self.level[v] < self.level[to]:
                ret = self.dfs(to, t, min(f, cap))
                if ret:
                    self.g[v][i][1] -= ret
                    rev[1] += ret
                    return ret
        return 0

    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if not f:
                    break
                flow += f
        return flow

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = int(next(it))  # k is always 1
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))

    dinic = Dinic(n)
    # store reference to each forward edge for later flow extraction
    edge_refs = []
    for idx, (u, v, cap) in enumerate(edges):
        dinic.add_edge(u, v, cap, idx)
        edge_refs.append(dinic.g[u][-1])  # the forward edge just added

    maxflow = dinic.max_flow(s, t)
    if maxflow < d:
        print("NO")
        return

    # we have flow >= d, need to reduce to exactly d if larger.
    # Reduce excess flow by sending it back from t to s via residual edges.
    excess = maxflow - d
    if excess:
        # add a temporary edge t->s with capacity excess to cancel flow
        dinic.add_edge(t, s, excess)
        # run flow from t to s to push back excess
        dinic.max_flow(t, s)
        # remove the temporary edge (ignore, its flow will be exactly excess)
    # Now the flow from s to t is exactly d.
    # Extract flows on original edges
    result = [0] * m
    for idx, edge in enumerate(edge_refs):
        # original capacity minus remaining capacity = flow used
        used = edges[idx][2] - edge[1]
        result[idx] = used
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
