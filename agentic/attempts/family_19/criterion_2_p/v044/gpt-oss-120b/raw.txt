```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); _ = int(next(it))
    S = int(next(it)) - 1
    T = int(next(it)) - 1
    D = int(next(it))
    edges_input = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        c = int(next(it))
        edges_input.append((u, v, c))

    # Dinic
    N = n
    g = [[] for _ in range(N)]

    class Edge:
        __slots__ = ('to','rev','cap','orig')
        def __init__(self, to, rev, cap, orig=False):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.orig = orig

    def add_edge(fr, to, cap, orig=False):
        fwd = Edge(to, len(g[to]), cap, orig)
        rev = Edge(fr, len(g[fr]), 0, False)
        g[fr].append(fwd)
        g[to].append(rev)
        return fwd

    # store references to original edges to read flow later
    orig_refs = []
    for (u, v, c) in edges_input:
        e = add_edge(u, v, c, True)
        orig_refs.append(e)

    level = [0]*N
    itp = [0]*N

    def bfs(s, t):
        for i in range(N):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            v = q.popleft()
            for e in g[v]:
                if e.cap > 0 and level[e.to] < 0:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] >= 0

    def dfs(v, t, f):
        if v == t:
            return f
        for i in range(itp[v], len(g[v])):
            e = g[v][i]
            if e.cap > 0 and level[v] < level[e.to]:
                ret = dfs(e.to, t, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    g[e.to][e.rev].cap += ret
                    return ret
            itp[v] += 1
        return 0

    flow = 0
    INF = 10**18
    while bfs(S, T):
        itp = [0]*N
        while True:
            f = dfs(S, T, INF)
            if not f:
                break
            flow += f
            if flow >= D:
                break
        if flow >= D:
            break

    if flow < D:
        print("NO")
        return

    # we may have sent more than D, need to reduce excess if any
    excess = flow - D
    if excess > 0:
        # try to push back excess from T to S using residual graph
        # add a reverse edge T->S with infinite capacity and then send excess flow
        # but simpler: run another maxflow from T to S on residual to cancel excess
        # We'll just run bfs/dfs sending excess from T to S
        def bfs2(s, t):
            for i in range(N):
                level[i] = -1
            q = deque([s])
            level[s] = 0
            while q:
                v = q.popleft()
                for e in g[v]:
                    if e.cap > 0 and level[e.to] < 0:
                        level[e.to] = level[v] + 1
                        q.append(e.to)
            return level[t] >= 0

        def dfs2(v, t, f):
            if v == t:
                return f
            for i in range(itp[v], len(g[v])):
                e = g[v][i]
                if e.cap > 0 and level[v] < level[e.to]:
                    ret = dfs2(e.to, t, min(f, e.cap))
                    if ret:
                        e.cap -= ret
                        g[e.to][e.rev].cap += ret
                        return ret
                itp[v] += 1
            return 0

        while excess > 0 and bfs2(T, S):
            itp = [0]*N
            while excess > 0:
                f = dfs2(T, S, excess)
                if not f:
                    break
                excess -= f

    # retrieve flows on original edges
    res = []
    for e in orig_refs:
        # original capacity was stored in reverse edge's cap after flow
        flow_used = g[e.to][e.rev].cap  # this is the flow sent
        res.append(str(flow_used))
    print("YES")
    print(" ".join(res))

if __name__ == "__main__":
    solve()
```
