import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    _ = next(it)

    s = next(it) - 1
    t = next(it) - 1
    demand = next(it)

    adj = [[] for _ in range(n)]
    to = []
    cap = []
    original_capacity = []
    edge_ids = []

    def add_edge(u, v, c):
        edge_id = len(to)
        to.append(v)
        cap.append(c)
        original_capacity.append(c)
        adj[u].append(edge_id)

        to.append(u)
        cap.append(0)
        original_capacity.append(0)
        adj[v].append(edge_id + 1)
        return edge_id

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        c = next(it)
        edge_ids.append(add_edge(u, v, c))

    if demand < 0 or (s == t and demand > 0):
        print("NO")
        return

    if demand == 0:
        print("YES")
        print(*([0] * m))
        return

    level = [-1] * n
    ptr = [0] * n

    def bfs():
        for i in range(n):
            level[i] = -1
        level[s] = 0
        queue = deque([s])

        while queue:
            v = queue.popleft()
            next_level = level[v] + 1
            for e in adj[v]:
                w = to[e]
                if cap[e] > 0 and level[w] == -1:
                    level[w] = next_level
                    queue.append(w)

        return level[t] != -1

    def send_one(limit):
        vertices = [s]
        path_edges = []
        bottlenecks = [limit]

        while vertices:
            v = vertices[-1]

            if v == t:
                pushed = bottlenecks[-1]
                for e in path_edges:
                    cap[e] -= pushed
                    cap[e ^ 1] += pushed
                return pushed

            while ptr[v] < len(adj[v]):
                e = adj[v][ptr[v]]
                if cap[e] > 0 and level[to[e]] == level[v] + 1:
                    break
                ptr[v] += 1

            if ptr[v] == len(adj[v]):
                level[v] = -1
                vertices.pop()
                bottlenecks.pop()

                if path_edges:
                    path_edges.pop()
                    if vertices:
                        ptr[vertices[-1]] += 1
                continue

            e = adj[v][ptr[v]]
            path_edges.append(e)
            vertices.append(to[e])
            bottlenecks.append(min(bottlenecks[-1], cap[e]))

        return 0

    total_flow = 0

    while total_flow < demand and bfs():
        ptr[:] = [0] * n

        while total_flow < demand:
            pushed = send_one(demand - total_flow)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow < demand:
        print("NO")
        return

    flows = [original_capacity[e] - cap[e] for e in edge_ids]
    print("YES")
    print(*flows)

if __name__ == "__main__":
    main()
