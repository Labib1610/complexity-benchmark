import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = next(it)  # product varieties, always 1, ignore
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))
    edges_input = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        c = int(next(it))
        edges_input.append((u, v, c))

    # Dinic implementation
    N = n
    adj = [[] for _ in range(N)]

    class Edge:
        __slots__ = ('to', 'rev', 'cap', 'orig')
        def __init__(self, to, rev, cap, orig=False):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.orig = orig  # marks original edge for output

    def add_edge(fr, to, cap, orig=False):
        fwd = Edge(to, len(adj[to]), cap, orig)
        rev = Edge(fr, len(adj[fr]), 0, False)
        adj[fr].append(fwd)
        adj[to].append(rev)

    # store reference to original edges for later flow extraction
    orig_refs = []
    for u, v, c in edges_input:
        add_edge(u, v, c, True)
        orig_refs.append((u, len(adj[u]) - 1))  # (node, index of edge in adj[node])

    level = [0] * N
    it_ptr = [0] * N

    def bfs():
        for i in range(N):
            level[i] = -1
        q = deque()
        level[s] = 0
        q.append(s)
        while q:
            v = q.popleft()
            for e in adj[v]:
                if e.cap > 0 and level[e.to] < 0:
                    level[e.to] = level[v] + 1
                    q.append(e.to)
        return level[t] >= 0

    def dfs(v, f):
        if v == t:
            return f
        for i in range(it_ptr[v], len(adj[v])):
            it_ptr[v] = i
            e = adj[v][i]
            if e.cap > 0 and level[v] < level[e.to]:
                ret = dfs(e.to, min(f, e.cap))
                if ret:
                    e.cap -= ret
                    adj[e.to][e.rev].cap += ret
                    return ret
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        it_ptr = [0] * N
        while True:
            f = dfs(s, INF)
            if not f:
                break
            flow += f
            if flow >= d:
                break
        if flow >= d:
            break

    if flow < d:
        print("NO")
        return

    # We have at least d flow, but may have sent more. Need to adjust to exactly d.
    # Since we stopped early when flow >= d, the current flow may be > d.
    # Reduce excess flow by sending back along residual graph from t to s.
    excess = flow - d
    if excess > 0:
        # add a reverse edge from t to s with infinite capacity to allow canceling flow
        # Actually we can just run a DFS to push back excess.
        # We'll treat it as a max flow from t to s on the residual network.
        # Temporarily set source=t, sink=s.
        src, sink = t, s
        # reuse bfs/dfs with new source/sink
        def bfs_back():
            for i in range(N):
                level[i] = -1
            q = deque()
            level[src] = 0
            q.append(src)
            while q:
                v = q.popleft()
                for e in adj[v]:
                    if e.cap > 0 and level[e.to] < 0:
                        level[e.to] = level[v] + 1
                        q.append(e.to)
            return level[sink] >= 0

        def dfs_back(v, f):
            if v == sink:
                return f
            for i in range(it_ptr[v], len(adj[v])):
                it_ptr[v] = i
                e = adj[v][i]
                if e.cap > 0 and level[v] < level[e.to]:
                    ret = dfs_back(e.to, min(f, e.cap))
                    if ret:
                        e.cap -= ret
                        adj[e.to][e.rev].cap += ret
                        return ret
            return 0

        while excess > 0 and bfs_back():
            it_ptr = [0] * N
            while excess > 0:
                pushed = dfs_back(src, excess)
                if not pushed:
                    break
                excess -= pushed

    # Extract flows on original edges
    result = []
    for (u, idx) in orig_refs:
        e = adj[u][idx]
        # original capacity was stored in reverse edge's cap after flow sent
        # flow = original_cap - remaining_cap
        # original capacity is sum of forward cap + reverse cap (since reverse started 0)
        rev_edge = adj[e.to][e.rev]
        flow_used = rev_edge.cap  # amount that has been sent forward
        result.append(str(flow_used))

    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    solve()
