import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    n = int(next(iterator))
    m = int(next(iterator))
    _ = int(next(iterator))  # product varieties, always 1
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # Graph representation for Dinic's algorithm
    # Nodes: 0 is super source, 1..n are facilities
    # If s == t:
    if s == t:
        if d == 0:
            print("YES")
            print(*(0 for _ in range(m)))
        else:
            print("NO")
        return

    src = 0
    sink = t
    
    head = [-1] * (n + 1)
    to = []
    cap = []
    flow = []
    nxt = []
    
    def add_edge(u, v, c):
        idx = len(to)
        to.append(v)
        cap.append(c)
        flow.append(0)
        nxt.append(head[u])
        head[u] = idx
        
        to.append(u)
        cap.append(0)
        flow.append(0)
        nxt.append(head[v])
        head[v] = idx + 1
        return idx

    # Super source to s with capacity d
    add_edge(src, s, d)
    
    original_edge_indices = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        idx = add_edge(u, v, c)
        original_edge_indices.append(idx)
        
    # Dinic's Algorithm
    num_nodes = n + 1
    level = [-1] * num_nodes
    ptr = [0] * num_nodes
    
    from collections import deque
    
    def bfs():
        for i in range(num_nodes):
            level[i] = -1
        level[src] = 0
        q = deque([src])
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                u = to[e]
                if cap[e] - flow[e] > 0 and level[u] < 0:
                    level[u] = level[v] + 1
                    q.append(u)
                e = nxt[e]
        return level[sink] != -1

    def dfs(v, pushed):
        if pushed == 0:
            return 0
        if v == sink:
            return pushed
        e = ptr[v]
        while e != -1:
            u = to[e]
            tr = cap[e] - flow[e]
            if level[v] + 1 == level[u] and tr > 0:
                push = dfs(u, min(pushed, tr))
                if push > 0:
                    flow[e] += push
                    flow[e ^ 1] -= push
                    return push
            ptr[v] = nxt[e]
            e = nxt[e]
        return 0

    total_flow = 0
    while bfs():
        for i in range(num_nodes):
            ptr[i] = head[i]
        while True:
            pushed = dfs(src, d - total_flow)
            if pushed == 0:
                break
            total_flow += pushed
            if total_flow == d:
                break
        if total_flow == d:
            break

    if total_flow == d:
        print("YES")
        ans = [flow[idx] for idx in original_edge_indices]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
