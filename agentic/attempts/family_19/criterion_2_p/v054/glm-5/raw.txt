import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # container type, not used
    s = int(next(it))
    t = int(next(it))
    d = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Build flow network
    N = n + 2
    S = 0
    T = n + 1
    flow_adj = [[] for _ in range(N)]
    flow_edges = []
    
    def add_flow_edge(u, v, cap):
        flow_adj[u].append(len(flow_edges))
        flow_edges.append([v, cap, len(flow_edges) + 1])
        flow_adj[v].append(len(flow_edges))
        flow_edges.append([u, 0, len(flow_edges) - 1])
    
    # Edge mapping: original edge index -> (from_edge_idx, to_edge_idx) in flow graph
    edge_map = {}
    
    for i, (u, v, cap) in enumerate(edges):
        add_flow_edge(u, v, cap)
        edge_map[i] = (len(flow_edges) - 2, len(flow_edges) - 1)
    
    add_flow_edge(S, s, d)
    add_flow_edge(t, T, d)
    
    # Dinic's algorithm
    level = [0] * N
    ptr = [0] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            for idx in flow_adj[u]:
                v, cap, _ = flow_edges[idx]
                if cap > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[T] != -1
    
    def dfs(u, f):
        if u == T:
            return f
        for i in range(ptr[u], len(flow_adj[u])):
            ptr[u] = i
            idx = flow_adj[u][i]
            v, cap, rev = flow_edges[idx]
            if cap > 0 and level[v] == level[u] + 1:
                pushed = dfs(v, min(f, cap))
                if pushed > 0:
                    flow_edges[idx][1] -= pushed
                    flow_edges[rev][1] += pushed
                    return pushed
        return 0
    
    total_flow = 0
    while bfs():
        ptr = [0] * N
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            total_flow += pushed
    
    if total_flow != d:
        print("NO")
        return
    
    print("YES")
    result = []
    for i in range(m):
        from_idx, to_idx = edge_map[i]
        # The flow on edge i is the capacity minus remaining capacity
        original_cap = edges[i][2]
        remaining = flow_edges[from_idx][1]
        flow = original_cap - remaining
        result.append(str(flow))
    print(' '.join(result))

if __name__ == "__main__":
    solve()
