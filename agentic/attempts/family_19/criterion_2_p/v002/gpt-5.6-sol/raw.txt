```python
import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
_ = next(it)

s = next(it) - 1
t = next(it) - 1
demand = next(it)

adj = [[] for _ in range(n)]
to = []
capacity = []
original_edges = []

def add_edge(u, v, cap):
    edge_id = len(to)
    to.append(v)
    capacity.append(cap)
    adj[u].append(edge_id)

    to.append(u)
    capacity.append(0)
    adj[v].append(edge_id ^ 1)

    original_edges.append((edge_id, cap))

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    cap = next(it)
    add_edge(u, v, cap)

if demand == 0 or s == t:
    print("YES")
    print(*([0] * m))
    sys.exit()

level = [-1] * n
ptr = [0] * n

def bfs():
    level[:] = [-1] * n
    level[s] = 0
    q = deque([s])

    while q:
        u = q.popleft()
        for edge_id in adj[u]:
            v = to[edge_id]
            if capacity[edge_id] > 0 and level[v] == -1:
                level[v] = level[u] + 1
                if v == t:
                    return True
                q.append(v)
    return level[t] != -1

def dfs(u, pushed):
    if u == t:
        return pushed

    while ptr[u] < len(adj[u]):
        edge_id = adj[u][ptr[u]]
        v = to[edge_id]

        if capacity[edge_id] > 0 and level[v] == level[u] + 1:
            sent = dfs(v, min(pushed, capacity[edge_id]))
            if sent:
                capacity[edge_id] -= sent
                capacity[edge_id ^ 1] += sent
                return sent

        ptr[u] += 1

    return 0

flow = 0
while flow < demand and bfs():
    ptr[:] = [0] * n
    while flow < demand:
        sent = dfs(s, demand - flow)
        if sent == 0:
            break
        flow += sent

if flow < demand:
    print("NO")
else:
    answer = [initial_cap - capacity[edge_id]
              for edge_id, initial_cap in original_edges]
    print("YES")
    print(*answer)
```
