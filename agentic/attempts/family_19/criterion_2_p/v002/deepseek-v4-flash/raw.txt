We can solve this by finding a feasible flow of value `d` from `s` to `t` in a directed graph with capacities. Since it’s a single stream, we can use a simple max-flow algorithm (e.g., Dinic’s) and then output the flow on each edge.

```python
import sys
from collections import deque

def dinic(n, graph, source, sink):
    flow = 0
    INF = 10**18
    while True:
        level = [-1] * n
        level[source] = 0
        q = deque([source])
        while q:
            u = q.popleft()
            for v, cap, idx in graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        if level[sink] < 0:
            return flow
        it = [0] * n
        def dfs(u, f):
            if u == sink:
                return f
            while it[u] < len(graph[u]):
                v, cap, idx = graph[u][it[u]]
                if cap > 0 and level[u] + 1 == level[v]:
                    ret = dfs(v, min(f, cap))
                    if ret > 0:
                        graph[u][it[u]] = (v, cap - ret, idx)
                        rv, rcap, ridx = graph[v][idx]
                        graph[v][idx] = (rv, rcap + ret, ridx)
                        return ret
                it[u] += 1
            return 0
        while True:
            pushed = dfs(source, INF)
            if pushed == 0:
                break
            flow += pushed

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    _ = int(next(it))  # skip 1
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    d = int(next(it))

    edges = []
    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        idx = len(edges)
        edges.append([u, v, cap, 0])
        graph[u].append((v, cap, len(graph[v])))
        graph[v].append((u, 0, len(graph[u]) - 1))

    maxflow = dinic(n, graph, s, t)
    if maxflow < d:
        print("NO")
        return

    # Reconstruct flow on each original edge
    result = []
    for u, v, cap, _ in edges:
        # find residual capacity in graph[u] to v
        sent = 0
        for vv, rcap, idx in graph[u]:
            if vv == v:
                # original cap - residual cap = flow
                sent = cap - rcap
                break
        result.append(str(sent))
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
