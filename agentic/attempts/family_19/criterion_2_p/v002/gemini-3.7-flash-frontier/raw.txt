import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        n_str = next(iterator)
    except StopIteration:
        return
    
    n = int(n_str)
    m = int(next(iterator))
    _ = int(next(iterator))  # k = 1
    
    s = int(next(iterator))
    t = int(next(iterator))
    d = int(next(iterator))
    
    # Vertices are 1-based, from 1 to n.
    # Source for max flow: S = 0, Sink = t. Edge S -> s with capacity d.
    # Or simply run max flow from s to t up to d.
    
    S = 0
    adj = [[] for _ in range(n + 1)]
    edges = []
    
    def add_edge(u, v, cap, original_idx=-1):
        idx = len(edges)
        edges.append([u, v, cap, 0, original_idx])  # u, v, cap, flow, orig_idx
        adj[u].append(idx)
        edges.append([v, u, 0, 0, -1])
        adj[v].append(idx + 1)
        return idx
    
    add_edge(S, s, d)
    
    orig_edge_ptrs = []
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        e_idx = add_edge(u, v, cap, i)
        orig_edge_ptrs.append(e_idx)
        
    # Dinic's Algorithm from S to t
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            v = q.popleft()
            for edge_idx in adj[v]:
                u, to, cap, flow, _ = edges[edge_idx]
                if cap - flow > 0 and level[to] == -1:
                    level[to] = level[v] + 1
                    q.append(to)
        return level[t] != -1

    def dfs(v, pushed):
        if pushed == 0 or v == t:
            return pushed
        for cid in range(ptr[v], len(adj[v])):
            ptr[v] = cid
            edge_idx = adj[v][cid]
            u, to, cap, flow, _ = edges[edge_idx]
            tr = cap - flow
            if level[v] + 1 != level[to] or tr == 0:
                continue
            push = dfs(to, min(pushed, tr))
            if push == 0:
                continue
            edges[edge_idx][3] += push
            edges[edge_idx ^ 1][3] -= push
            return push
        return 0

    flow = 0
    while bfs():
        for i in range(n + 1):
            ptr[i] = 0
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            flow += pushed

    if flow == d:
        print("YES")
        ans = [edges[e_idx][3] for e_idx in orig_edge_ptrs]
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
