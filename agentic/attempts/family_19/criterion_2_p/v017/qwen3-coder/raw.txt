from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    _ = int(data[idx])
    idx += 1
    
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    d = int(data[idx])
    idx += 1
    
    edges = []
    graph = defaultdict(list)
    capacity = []
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        cap = int(data[idx])
        idx += 1
        edges.append((u, v))
        graph[u].append((v, i))
        graph[v].append((u, i))
        capacity.append(cap)
    
    # We model this as a max-flow problem:
    # - Source is s, sink is t
    # - Each edge has capacity as given
    # - We want to push 'd' units of flow from s to t
    
    # If max flow < d, then impossible
    
    # Implementing Ford-Fulkerson with BFS (Edmonds-Karp)
    def bfs():
        parent = [-1] * (n+1)
        visited = [False] * (n+1)
        q = deque()
        q.append(s)
        visited[s] = True
        while q:
            u = q.popleft()
            for v, edge_idx in graph[u]:
                if not visited[v] and flow[edge_idx] < capacity[edge_idx]:
                    visited[v] = True
                    parent[v] = (u, edge_idx)
                    q.append(v)
                    if v == t:
                        path = []
                        cur = v
                        while parent[cur] != -1:
                            prev_node, edge_idx = parent[cur]
                            path.append(edge_idx)
                            cur = prev_node
                        path.reverse()
                        return path
        return None
    
    # Initialize flow
    flow = [0] * m
    
    max_flow = 0
    while max_flow < d:
        path = bfs()
        if not path:
            break
        # Find bottleneck
        bottleneck = d - max_flow
        for edge_idx in path:
            bottleneck = min(bottleneck, capacity[edge_idx] - flow[edge_idx])
        if bottleneck == 0:
            break
        # Augment flow
        for edge_idx in path:
            flow[edge_idx] += bottleneck
        max_flow += bottleneck
    
    if max_flow >= d:
        print("YES")
        print(' '.join(map(str, flow)))
    else:
        print("NO")

solve()
