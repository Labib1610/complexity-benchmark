import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    _ = int(data[idx]); idx += 1
    s = int(data[idx]) - 1; idx += 1
    t = int(data[idx]) - 1; idx += 1
    d = int(data[idx]); idx += 1

    edges = []
    head = [-1] * n
    to = []
    cap = []
    nxt = []
    edge_index = []

    def add_edge(u, v, c, idx):
        head.append(0)
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = len(to) - 1
        edge_index.append(idx)

        head.append(0)
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = len(to) - 1
        edge_index.append(-1)

    for i in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        c = int(data[idx]); idx += 1
        edges.append((u, v, c))
        add_edge(u, v, c, i)

    N = len(to)
    level = [0] * n
    it = [0] * n

    def bfs():
        for i in range(n):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                if cap[e] > 0 and level[to[e]] < 0:
                    level[to[e]] = level[u] + 1
                    q.append(to[e])
                e = nxt[e]
        return level[t] >= 0

    def dfs(u, f):
        if u == t:
            return f
        while it[u] != -1:
            e = it[u]
            if cap[e] > 0 and level[to[e]] == level[u] + 1:
                ret = dfs(to[e], min(f, cap[e]))
                if ret > 0:
                    cap[e] -= ret
                    cap[e ^ 1] += ret
                    return ret
            it[u] = nxt[e]
        return 0

    flow = 0
    while flow < d:
        if not bfs():
            break
        for i in range(n):
            it[i] = head[i]
        while True:
            f = dfs(s, d - flow)
            if f == 0:
                break
            flow += f
            if flow >= d:
                break

    if flow < d:
        print("NO")
        return

    ans = [0] * m
    for e in range(0, N, 2):
        ei = edge_index[e]
        if ei != -1:
            ans[ei] = edges[ei][2] - cap[e]

    print("YES")
    print(' '.join(map(str, ans)))

solve()
