```python
import sys
import heapq
import time
import random
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    mandates = []
    for _ in range(k):
        s = int(next(iterator))
        t = int(next(iterator))
        d = int(next(iterator))
        mandates.append((s, t, d))
        
    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        edges.append((u, v, cap))
        adj[u].append((v, i))
        
    start_time = time.time()
    
    # Quick check for single mandate (k=1) using Max-Flow (Dinic)
    if k == 1:
        s_target, t_target, d_target = mandates[0]
        # Build flow network
        head = [-1] * (n + 1)
        to = []
        nxt = []
        cap_val = []
        flow_val = []
        edge_id = []
        
        def add_flow_edge(u, v, c, e_idx):
            to.append(v)
            cap_val.append(c)
            flow_val.append(0)
            edge_id.append(e_idx)
            nxt.append(head[u])
            head[u] = len(to) - 1
            
            to.append(u)
            cap_val.append(0)
            flow_val.append(0)
            edge_id.append(-1)
            nxt.append(head[v])
            head[v] = len(to) - 1
            
        for i, (u, v, c) in enumerate(edges):
            add_flow_edge(u, v, c, i)
            
        def bfs(s, t, level):
            for i in range(n + 1):
                level[i] = -1
            level[s] = 0
            q = deque([s])
            while q:
                u = q.popleft()
                e = head[u]
                while e != -1:
                    v = to[e]
                    if cap_val[e] - flow_val[e] > 0 and level[v] < 0:
                        level[v] = level[u] + 1
                        q.append(v)
                    e = nxt[e]
            return level[t] >= 0

        def dfs(u, t, pushed, level, ptr):
            if pushed == 0 or u == t:
                return pushed
            e = ptr[u]
            while e != -1:
                v = to[e]
                tr = cap_val[e] - flow_val[e]
                if level[v] == level[u] + 1 and tr > 0:
                    push = dfs(v, t, min(pushed, tr), level, ptr)
                    if push > 0:
                        flow_val[e] += push
                        flow_val[e ^ 1] -= push
                        return push
                e = nxt[e]
                ptr[u] = e
            return 0

        total_flow = 0
        level = [-1] * (n + 1)
        while total_flow < d_target and bfs(s_target, t_target, level):
            ptr = list(head)
            while True:
                pushed = dfs(s_target, t_target, d_target - total_flow, level, ptr)
                if pushed == 0:
                    break
                total_flow += pushed
                if total_flow >= d_target:
                    break
                    
        if total_flow < d_target:
            print("NO")
            return
            
        # Collect edge flows
        ans = [0] * m
        for e in range(0, len(to), 2):
            if edge_id[e] != -1:
                ans[edge_id[e]] = flow_val[e]
        print("YES")
        print(*(ans))
        return

    # For k > 1, use Negotiation-based Routing (PathFinder)
    unit_requests = []
    for m_idx, (s, t, d) in enumerate(mandates):
        for _ in range(d):
            unit_requests.append((s, t, m_idx))
            
    num_units = len(unit_requests)
    edge_caps = [e[2] for e in edges]
    usage = [0] * m
    h_cost = [0.0] * m
    unit_paths = [[] for _ in range(num_units)]
    
    def dijkstra(s, t):
        dist = [float('inf')] * (n + 1)
        parent_edge = [-1] * (n + 1)
        parent_node = [-1] * (n + 1)
        dist[s] = 0.0
        pq = [(0.0, s)]
        
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            if u == t:
                break
            for v, e_idx in adj[u]:
                cap = edge_caps[e_idx]
                occ = usage[e_idx]
                # Congestion cost formula
                pres = max(0, occ + 1 - cap)
                cost = 1.0 + h_cost[e_idx] + pres * pres * 10.0 + (occ / (cap + 0.1))
                if dist[u] + cost < dist[v]:
                    dist[v] = dist[u] + cost
                    parent_edge[v] = e_idx
                    parent_node[v] = u
                    heapq.heappush(pq, (dist[v], v))
                    
        if dist[t] == float('inf'):
            return None
            
        path = []
        curr = t
        while curr != s:
            e_idx = parent_edge[curr]
            path.append(e_idx)
            curr = parent_node[curr]
        path.reverse()
        return path

    # Initial routing
    for u_idx, (s, t, _) in enumerate(unit_requests):
        p = dijkstra(s, t)
        if p is None:
            print("NO")
            return
        unit_paths[u_idx] = p
        for e_idx in p:
            usage[e_idx] += 1
            
    def get_max_overflow():
        max_ov = 0
        for i in range(m):
            ov = usage[i] - edge_caps[i]
            if ov > max_ov:
                max_ov = ov
        return max_ov

    if get_max_overflow() == 0:
        # Success immediately
        output_solution(k, m, unit_requests, unit_paths)
        return

    # Negotiation loop
    iterations = 0
    indices = list(range(num_units))
    
    while time.time() - start_time < 4.5:
        iterations += 1
        # Update history costs for overused edges
        for i in range(m):
            if usage[i] > edge_caps[i]:
                h_cost[i] += (usage[i] - edge_caps[i]) * 1.5
                
        random.shuffle(indices)
        for u_idx in indices:
            s, t, _ = unit_requests[u_idx]
            cur_path = unit_paths[u_idx]
            
            # Check if this unit uses any congested edge
            uses_congested = any(usage[e] > edge_caps[e] for e in cur_path)
            if uses_congested or (random.random() < 0.1):
                # Rip up
                for e_idx in cur_path:
                    usage[e_idx] -= 1
                    
                # Reroute
                new_path = dijkstra(s, t)
                if new_path is None:
                    # Restore old path
                    new_path = cur_path
                    
                unit_paths[u_idx] = new_path
                for e_idx in new_path:
                    usage[e_idx] += 1
                    
                if get_max_overflow() == 0:
                    output_solution(k, m, unit_requests, unit_paths)
                    return

    print("NO")

def output_solution(k, m, unit_requests, unit_paths):
    ans = [[0] * m for _ in range(k)]
    for u_idx, (_, _, m_idx) in enumerate(unit_requests):
        for e_idx in unit_paths[u_idx]:
            ans[m_idx][e_idx] += 1
    print("YES")
    for row in ans:
        print(*(row))

if __name__ == '__main__':
    solve()
```
