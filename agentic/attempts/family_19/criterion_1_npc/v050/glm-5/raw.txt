import sys
import collections

def solve():
    input = sys.stdin.read().split()
    it = iter(input)
    
    try:
        n = int(next(it))
        m = int(next(it))
        k = int(next(it))
    except StopIteration:
        return

    mandates = []
    total_demand = 0
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        mandates.append((s, t, d))
        total_demand += d

    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, i))

    # Node indexing for Mandate i: base_i + u for u=1..n
    # base_i = (i-1) * n
    # Edge indexing for Mandate i: mand_edge_idx[i][j] = (i-1)*m + j
    
    base_node = [0] * k
    for i in range(k):
        base_node[i] = i * n

    # Global Sink
    sink_node = k * n
    source_node = k * n + 1
    n_nodes = k * n + 2
    
    # Build Dinic graph
    # Edge storage: to, cap, rev_idx
    graph = [[] for _ in range(n_nodes)]
    
    def add_edge(u, v, cap):
        # forward edge
        graph[u].append([v, cap, len(graph[v])])
        # backward edge
        graph[v].append([u, 0, len(graph[u]) - 1])

    # Edges for each mandate
    for i in range(k):
        s_i, t_i, d_i = mandates[i]
        b = base_node[i]
        
        for j in range(m):
            u, v, cap = edges[j]
            add_edge(b + u, b + v, cap)
        
        # Source/Sink connections to global source/sink
        add_edge(source_node, b + s_i, d_i)
        add_edge(b + t_i, sink_node, d_i)

    # Dinic
    level = [0] * n_nodes
    itr = [0] * n_nodes
    
    def bfs():
        for i in range(n_nodes):
            level[i] = -1
        queue = collections.deque()
        level[source_node] = 0
        queue.append(source_node)
        
        while queue:
            u = queue.popleft()
            for v, cap, rev in graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        return level[sink_node] >= 0

    def dfs(u, f):
        if u == sink_node:
            return f
        for idx in range(itr[u], len(graph[u])):
            itr[u] = idx
            v, cap, rev = graph[u][idx]
            if cap > 0 and level[u] + 1 == level[v]:
                ret = dfs(v, min(f, cap))
                if ret > 0:
                    graph[u][idx][1] -= ret
                    graph[v][rev][1] += ret
                    return ret
        return 0

    flow = 0
    while bfs():
        itr = [0] * n_nodes
        while True:
            f = dfs(source_node, float('inf'))
            if f == 0:
                break
            flow += f

    if flow != total_demand:
        print("NO")
        return

    print("YES")
    
    # Recover flow for each mandate
    for i in range(k):
        b = base_node[i]
        line = []
        for j in range(m):
            u, v, cap_orig = edges[j]
            # Find residual capacity on edge b+u -> b+v
            # The edge added was add_edge(b+u, b+v, cap_orig)
            # So we look for the edge in graph[b+u] that goes to b+v
            # And check its current capacity (which is residual)
            # Original cap - residual = flow
            # Or simply, look at reverse edge in graph[b+v] towards b+u
            # But we are not sure of indices. 
            # Better: look for edge b+u -> b+v in graph[b+u]
            
            # Find index of v in adj[u] to get edge id? 
            # Wait, the graph edges are stored differently.
            # In graph construction: add_edge(b+u, b+v, cap)
            # graph[b+u] has a list. We need to find the correct entry.
            # However, usually edges are added in order.
            # For mandate i, we added edges 0..m-1 sequentially.
            # So the edge for conduit j is the j-th edge added for this mandate.
            # Wait, we added other edges (source/sink) too.
            
            # Let's iterate through graph[b+u] to find the edge to b+v
            # This is safer.
            
            flow_val = 0
            for edge_data in graph[b + u]:
                # edge_data: [to, cap, rev]
                if edge_data[0] == b + v:
                    # original cap was edges[j][2]
                    # current cap is edge_data[1]
                    # flow = original - current
                    flow_val = edges[j][2] - edge_data[1]
                    break
            line.append(str(flow_val))
        print(" ".join(line))

if __name__ == '__main__':
    solve()
