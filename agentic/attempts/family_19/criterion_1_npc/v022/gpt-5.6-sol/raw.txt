import sys
import random

sys.setrecursionlimit(1_000_000)


class Dinic:
    __slots__ = ("n", "g", "level", "it")

    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]
        self.level = [-1] * n
        self.it = [0] * n

    def add_edge(self, u, v, cap):
        forward = [v, len(self.g[v]), cap]
        backward = [u, len(self.g[u]), 0]
        self.g[u].append(forward)
        self.g[v].append(backward)
        return forward

    def max_flow(self, s, t, limit=10**30):
        if s == t:
            return limit
        flow = 0
        n = self.n
        g = self.g

        while flow < limit:
            level = [-1] * n
            level[s] = 0
            q = [s]
            head = 0
            while head < len(q):
                v = q[head]
                head += 1
                for e in g[v]:
                    if e[2] > 0 and level[e[0]] < 0:
                        level[e[0]] = level[v] + 1
                        q.append(e[0])
            if level[t] < 0:
                break

            self.level = level
            self.it = [0] * n

            def dfs(v, pushed):
                if v == t:
                    return pushed
                while self.it[v] < len(g[v]):
                    e = g[v][self.it[v]]
                    if e[2] > 0 and level[e[0]] == level[v] + 1:
                        got = dfs(e[0], min(pushed, e[2]))
                        if got:
                            e[2] -= got
                            g[e[0]][e[1]][2] += got
                            return got
                    self.it[v] += 1
                return 0

            while flow < limit:
                pushed = dfs(s, limit - flow)
                if not pushed:
                    break
                flow += pushed

        return flow


def compute_flow(n, edges, residual, s, t, demand, need_usage=False):
    if demand == 0 or s == t:
        return demand, [0] * len(edges) if need_usage else None

    dinic = Dinic(n)
    refs = [None] * len(edges)
    initial = [0] * len(edges)

    for i, (u, v, _) in enumerate(edges):
        c = residual[i]
        initial[i] = c
        if c > 0:
            refs[i] = dinic.add_edge(u, v, c)

    value = dinic.max_flow(s, t, demand)
    if not need_usage:
        return value, None

    usage = [0] * len(edges)
    for i, ref in enumerate(refs):
        if ref is not None:
            usage[i] = initial[i] - ref[2]
    return value, usage


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    commodities = []
    for _ in range(k):
        s = next(it) - 1
        t = next(it) - 1
        d = next(it)
        commodities.append((s, t, d))

    edges = []
    capacities = []
    adjacency = [[] for _ in range(n)]
    for eid in range(m):
        u = next(it) - 1
        v = next(it) - 1
        c = next(it)
        edges.append((u, v, c))
        capacities.append(c)
        if c > 0:
            adjacency[u].append(eid)

    for u in range(n):
        adjacency[u].sort()

    base_stats = []
    for idx, (s, t, d) in enumerate(commodities):
        if d == 0 or s == t:
            base_stats.append((0, idx))
            continue
        value, _ = compute_flow(n, edges, capacities, s, t, d, False)
        if value < d:
            print("NO")
            return
        base_stats.append((value - d, idx))

    def try_order(order):
        residual = capacities[:]
        answer = [[0] * m for _ in range(k)]

        for ci in order:
            s, t, d = commodities[ci]
            if d == 0 or s == t:
                continue
            value, usage = compute_flow(
                n, edges, residual, s, t, d, True
            )
            if value < d:
                return None
            answer[ci] = usage
            for eid, amount in enumerate(usage):
                residual[eid] -= amount

        return answer

    candidate_orders = []
    seen_orders = set()

    def add_order(order):
        key = tuple(order)
        if key not in seen_orders:
            seen_orders.add(key)
            candidate_orders.append(list(order))

    add_order(range(k))
    add_order(reversed(range(k)))
    add_order([idx for _, idx in sorted(base_stats)])
    add_order([idx for _, idx in sorted(base_stats, reverse=True)])
    add_order(sorted(range(k), key=lambda i: -commodities[i][2]))
    add_order(sorted(range(k), key=lambda i: commodities[i][2]))

    rng = random.Random(712367821)
    random_trials = min(40, max(8, 3 * k))
    for _ in range(random_trials):
        order = list(range(k))
        rng.shuffle(order)
        add_order(order)

    for order in candidate_orders:
        answer = try_order(order)
        if answer is not None:
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in answer)
            sys.stdout.write("\n".join(out))
            return

    paths_by_commodity = [[] for _ in range(k)]

    def enumerate_paths(s, t):
        if s == t:
            return [()]
        result = []
        visited = [False] * n
        visited[s] = True
        path = []

        def dfs(v):
            if v == t:
                result.append(tuple(path))
                return
            for eid in adjacency[v]:
                _, to, c = edges[eid]
                if c <= 0 or visited[to]:
                    continue
                visited[to] = True
                path.append(eid)
                dfs(to)
                path.pop()
                visited[to] = False

        dfs(s)
        result.sort(key=lambda p: (len(p), p))
        return result

    for ci, (s, t, d) in enumerate(commodities):
        if d == 0 or s == t:
            paths_by_commodity[ci] = [()]
        else:
            paths = enumerate_paths(s, t)
            if not paths:
                print("NO")
                return
            paths_by_commodity[ci] = paths

    exact_order = list(range(k))
    exact_order.sort(
        key=lambda i: (
            0 if commodities[i][2] == 0 or commodities[i][0] == commodities[i][1] else 1,
            len(paths_by_commodity[i]),
            -commodities[i][2],
        )
    )

    residual = capacities[:]
    answer = [[0] * m for _ in range(k)]
    failed_states = set()
    max_cached_states = 250000

    def individually_feasible(from_pos):
        for pos in range(from_pos, k):
            ci = exact_order[pos]
            s, t, d = commodities[ci]
            if d == 0 or s == t:
                continue
            value, _ = compute_flow(n, edges, residual, s, t, d, False)
            if value < d:
                return False
        return True

    def solve_commodity(pos):
        if pos == k:
            return True

        state = (pos, tuple(residual))
        if state in failed_states:
            return False

        ci = exact_order[pos]
        s, t, demand = commodities[ci]

        if demand == 0 or s == t:
            if solve_commodity(pos + 1):
                return True
            if len(failed_states) < max_cached_states:
                failed_states.add(state)
            return False

        if not individually_feasible(pos):
            if len(failed_states) < max_cached_states:
                failed_states.add(state)
            return False

        paths = paths_by_commodity[ci]
        local_failed = set()

        def allocate(remaining, start_index):
            if remaining == 0:
                if individually_feasible(pos + 1) and solve_commodity(pos + 1):
                    return True
                return False

            key = (remaining, start_index, tuple(residual))
            if key in local_failed:
                return False

            for pi in range(start_index, len(paths)):
                path = paths[pi]
                bottleneck = remaining
                for eid in path:
                    if residual[eid] < bottleneck:
                        bottleneck = residual[eid]
                    if bottleneck == 0:
                        break
                if bottleneck == 0:
                    continue

                for amount in range(bottleneck, 0, -1):
                    for eid in path:
                        residual[eid] -= amount
                        answer[ci][eid] += amount

                    rem = remaining - amount
                    possible = True
                    if rem:
                        value, _ = compute_flow(
                            n, edges, residual, s, t, rem, False
                        )
                        possible = value >= rem

                    if possible and allocate(rem, pi + 1):
                        return True

                    for eid in path:
                        residual[eid] += amount
                        answer[ci][eid] -= amount

            if len(local_failed) < 100000:
                local_failed.add(key)
            return False

        if allocate(demand, 0):
            return True

        if len(failed_states) < max_cached_states:
            failed_states.add(state)
        return False

    if solve_commodity(0):
        out = ["YES"]
        out.extend(" ".join(map(str, row)) for row in answer)
        sys.stdout.write("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    main()
