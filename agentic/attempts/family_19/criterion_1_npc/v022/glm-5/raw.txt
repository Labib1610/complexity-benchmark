import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    streams = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        streams.append((s - 1, t - 1, d))
    
    edges = []
    adj = [[] for _ in range(n)]
    for i in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, i))
    
    # Build flow network
    # Nodes: 0 to n-1 are original switches
    # Super source: n, Super sink: n+1
    N = n + 2
    source = n
    sink = n + 1
    
    # Create adjacency list for flow network
    flow_adj = [[] for _ in range(N)]
    edge_list = []
    
    def add_edge(u, v, cap, rev_idx=-1):
        # edge: [to, cap, rev]
        # rev is index of reverse edge in adj[v]
        flow_adj[u].append([v, cap, len(flow_adj[v])])
        flow_adj[v].append([u, 0, len(flow_adj[u]) - 1])
    
    # Add edges from super source to each stream's ingress
    for i, (s, t, d) in enumerate(streams):
        add_edge(source, s, d)
    
    # Add edges from each stream's egress to super sink
    for i, (s, t, d) in enumerate(streams):
        add_edge(t, sink, d)
    
    # Add original network edges
    for u, v, cap in edges:
        add_edge(u, v, cap)
    
    # Dinic's algorithm
    level = [0] * N
    it_ptr = [0] * N
    
    def bfs():
        nonlocal level
        level = [-1] * N
        level[source] = 0
        q = deque([source])
        while q:
            u = q.popleft()
            for v, cap, rev in flow_adj[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[sink] >= 0
    
    def dfs(u, f):
        if u == sink:
            return f
        for i in range(it_ptr[u], len(flow_adj[u])):
            it_ptr[u] = i
            v, cap, rev = flow_adj[u][i]
            if cap > 0 and level[v] == level[u] + 1:
                ret = dfs(v, min(f, cap))
                if ret > 0:
                    flow_adj[u][i][1] -= ret
                    flow_adj[v][rev][1] += ret
                    return ret
        return 0
    
    total_demand = sum(d for _, _, d in streams)
    flow = 0
    while bfs():
        it_ptr = [0] * N
        while True:
            f = dfs(source, float('inf'))
            if f == 0:
                break
            flow += f
    
    if flow != total_demand:
        print("NO")
        return
    
    print("YES")
    
    # Extract flow on each original edge for each stream
    # The flow on edge (u,v) is original_cap - remaining_cap
    # We need to decompose this into per-stream flows
    
    # First, get total flow on each original edge
    edge_flow = [0] * m
    for i, (u, v, cap) in enumerate(edges):
        # Find the edge in flow_adj[u]
        for j, (to, rem_cap, rev) in enumerate(flow_adj[u]):
            if to == v:
                edge_flow[i] = cap - rem_cap
                break
    
    # Now we need to decompose flows
    # Build a residual graph for flow decomposition
    # For each stream, we'll find paths from s_i to t_i
    
    # Create adjacency list for residual graph
    # Each edge: (to, capacity, edge_index or -1)
    res_adj = [[] for _ in range(n)]
    for i, (u, v, cap) in enumerate(edges):
        f = edge_flow[i]
        if f > 0:
            res_adj[u].append((v, f, i))  # forward edge with remaining flow
        if cap - f > 0:
            res_adj[v].append((u, cap - f, -1))  # backward edge (residual)
    
    # For each stream, decompose flow
    stream_flows = [[0] * m for _ in range(k)]
    
    for stream_idx, (s, t, d) in enumerate(streams):
        remaining = d
        while remaining > 0:
            # BFS to find a path from s to t
            parent = [-1] * n
            parent_edge = [-1] * n
            q = deque([s])
            parent[s] = s
            while q and parent[t] == -1:
                u = q.popleft()
                for v, cap, eidx in res_adj[u]:
                    if cap > 0 and parent[v] == -1:
                        parent[v] = u
                        parent_edge[v] = (u, eidx)
                        q.append(v)
            
            if parent[t] == -1:
                # Should not happen if flow is valid
                print("NO")
                return
            
            # Find bottleneck
            path = []
            curr = t
            min_cap = float('inf')
            while curr != s:
                u, eidx = parent_edge[curr]
                path.append((u, curr, eidx))
                if eidx >= 0:
                    # Find capacity
                    for j, (to, cap, ei) in enumerate(res_adj[u]):
                        if to == curr and ei == eidx:
                            min_cap = min(min_cap, cap)
                            break
                curr = u
            
            path.reverse()
            flow_amount = min(min_cap, remaining)
            remaining -= flow_amount
            
            # Update residual graph and record flow
            for u, v, eidx in path:
                if eidx >= 0:
                    # This is an original edge
                    stream_flows[stream_idx][eidx] += flow_amount
                    # Update residual capacity
                    for j, (to, cap, ei) in enumerate(res_adj[u]):
                        if to == v and ei == eidx:
                            res_adj[u][j] = (to, cap - flow_amount, eidx)
                            break
                else:
                    # This is a backward edge (residual)
                    # We're pushing flow back, so we need to find the original edge
                    # and decrease its flow
                    # The backward edge corresponds to some original edge (v, u)
                    # Find the original edge index
                    for j, (to, cap, ei) in enumerate(res_adj[u]):
                        if to == v and ei == eidx:
                            res_adj[u][j] = (to, cap - flow_amount, eidx)
                            break
                    # Find the original edge (v, u) and update
                    for j, (to, cap, ei) in enumerate(res_adj[v]):
                        if to == u and ei >= 0:
                            # This is the original edge
                            stream_flows[stream_idx][ei] -= flow_amount
                            res_adj[v][j] = (to, cap + flow_amount, ei)
                            break
    
    for i in range(k):
        print(' '.join(map(str, stream_flows[i])))

if __name__ == "__main__":
    solve()
