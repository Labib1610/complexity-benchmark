import sys
import collections

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    streams = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        streams.append((s, t, d))
        
    edges = []
    for i in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))
        
    # Quick individual max-flow check for each commodity
    # to filter out impossible cases early
    def check_individual_max_flow(s, t, req):
        if s == t:
            return req == 0
        # Dinic
        head = [-1] * n
        to = []
        nxt = []
        cap = []
        
        def add_edge(u, v, c):
            to.append(v)
            cap.append(c)
            nxt.append(head[u])
            head[u] = len(to) - 1
            
            to.append(u)
            cap.append(0)
            nxt.append(head[v])
            head[v] = len(to) - 1

        for u, v, c in edges:
            add_edge(u, v, c)
            
        flow = 0
        while True:
            level = [-1] * n
            level[s] = 0
            q = collections.deque([s])
            while q:
                u = q.popleft()
                e = head[u]
                while e != -1:
                    v = to[e]
                    if cap[e] > 0 and level[v] == -1:
                        level[v] = level[u] + 1
                        q.append(v)
                    e = nxt[e]
            if level[t] == -1:
                break
            
            ptr = list(head)
            def dfs(u, pushed):
                if u == t or pushed == 0:
                    return pushed
                e = ptr[u]
                while e != -1:
                    v = to[e]
                    if level[v] == level[u] + 1 and cap[e] > 0:
                        tr = dfs(v, min(pushed, cap[e]))
                        if tr > 0:
                            cap[e] -= tr
                            cap[e ^ 1] += tr
                            return tr
                    e = nxt[e]
                    ptr[u] = e
                return 0
            
            while True:
                pushed = dfs(s, float('inf'))
                if pushed == 0:
                    break
                flow += pushed
                if flow >= req:
                    return True
        return flow >= req

    for s, t, d in streams:
        if not check_individual_max_flow(s, t, d):
            print("NO")
            return

    # Expand streams into unit demands if needed, or route path by path
    # Backtracking search with MRV and path finding
    adj = [[] for _ in range(n)]
    for e_idx, (u, v, c) in enumerate(edges):
        adj[u].append((v, c, e_idx))
        
    rem_cap = [e[2] for e in edges]
    ans = [[0] * m for _ in range(k)]
    
    # Flatten units
    units = []
    for s_idx, (s, t, d) in enumerate(streams):
        for _ in range(d):
            units.append((s_idx, s, t))
            
    # Sort units to prioritize ones with longer distance or fewer paths
    # Simple heuristic: sort by shortest distance descending
    def get_dist(s, t):
        dist = [-1] * n
        dist[s] = 0
        q = collections.deque([s])
        while q:
            u = q.popleft()
            if u == t:
                return dist[t]
            for v, _, _ in adj[u]:
                if dist[v] == -1:
                    dist[v] = dist[u] + 1
                    q.append(v)
        return float('inf')

    units.sort(key=lambda item: -get_dist(item[1], item[2]))
    
    # Backtracking with cycle-free path search
    def dfs_unit(unit_idx):
        if unit_idx == len(units):
            return True
        
        s_idx, s, t = units[unit_idx]
        
        # Find paths from s to t using DFS / BFS with current rem_cap
        # We can find simple paths
        visited = [False] * n
        path_edges = []
        
        def find_path(u):
            if u == t:
                return True
            visited[u] = True
            for v, _, e_idx in adj[u]:
                if not visited[v] and rem_cap[e_idx] > 0:
                    rem_cap[e_idx] -= 1
                    path_edges.append(e_idx)
                    if find_path(v):
                        return True
                    path_edges.pop()
                    rem_cap[e_idx] += 1
            visited[u] = False
            return False

        # Try to find all possible paths for this unit using a generator
        def get_all_paths(u, cur_path):
            if u == t:
                yield list(cur_path)
                return
            visited[u] = True
            for v, _, e_idx in adj[u]:
                if not visited[v] and rem_cap[e_idx] > 0:
                    cur_path.append(e_idx)
                    rem_cap[e_idx] -= 1
                    yield from get_all_paths(v, cur_path)
                    rem_cap[e_idx] += 1
                    cur_path.pop()
            visited[u] = False

        for path in get_all_paths(s, []):
            for e_idx in path:
                rem_cap[e_idx] -= 1
                ans[s_idx][e_idx] += 1
            
            if dfs_unit(unit_idx + 1):
                return True
            
            for e_idx in path:
                rem_cap[e_idx] += 1
                ans[s_idx][e_idx] -= 1
                
        return False

    if dfs_unit(0):
        print("YES")
        for row in ans:
            print(*(row))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
