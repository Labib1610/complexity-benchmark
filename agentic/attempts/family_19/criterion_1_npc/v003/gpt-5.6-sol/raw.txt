import sys
import random
import time

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

try:
    n = next(it)
    m = next(it)
    k = next(it)
except StopIteration:
    sys.exit(0)

commodities = []
for _ in range(k):
    s = next(it) - 1
    t = next(it) - 1
    d = next(it)
    commodities.append((s, t, d))

edges = []
out_edges = [[] for _ in range(n)]
for eid in range(m):
    u = next(it) - 1
    v = next(it) - 1
    c = next(it)
    edges.append((u, v, c))
    if c > 0 and u != v:
        out_edges[u].append(eid)

start_time = time.monotonic()
deadline = start_time + 4.75


class Dinic:
    __slots__ = ("n", "g", "level", "ptr")

    def __init__(self, size):
        self.n = size
        self.g = [[] for _ in range(size)]
        self.level = [-1] * size
        self.ptr = [0] * size

    def add_edge(self, u, v, cap):
        a = [v, len(self.g[v]), cap]
        b = [u, len(self.g[u]), 0]
        self.g[u].append(a)
        self.g[v].append(b)
        return a

    def max_flow(self, source, sink, limit):
        if source == sink:
            return limit

        total = 0
        g = self.g
        size = self.n

        while total < limit:
            level = [-1] * size
            level[source] = 0
            q = [source]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                nl = level[u] + 1
                for e in g[u]:
                    if e[2] > 0 and level[e[0]] < 0:
                        level[e[0]] = nl
                        q.append(e[0])

            if level[sink] < 0:
                break

            self.level = level
            ptr = [0] * size
            self.ptr = ptr

            def dfs(u, pushed):
                if u == sink:
                    return pushed
                gu = g[u]
                while ptr[u] < len(gu):
                    e = gu[ptr[u]]
                    if e[2] > 0 and level[e[0]] == level[u] + 1:
                        got = dfs(e[0], min(pushed, e[2]))
                        if got:
                            e[2] -= got
                            g[e[0]][e[1]][2] += got
                            return got
                    ptr[u] += 1
                return 0

            while total < limit:
                pushed = dfs(source, limit - total)
                if not pushed:
                    break
                total += pushed

        return total


def route_one(source, sink, demand, residual, edge_order=None, need_usage=True):
    if demand == 0 or source == sink:
        return demand, [0] * m if need_usage else None

    dinic = Dinic(n)
    refs = [None] * m

    order = edge_order if edge_order is not None else range(m)
    for eid in order:
        u, v, _ = edges[eid]
        cap = residual[eid]
        if cap > 0 and u != v:
            ref = dinic.add_edge(u, v, cap)
            refs[eid] = (ref, cap)

    value = dinic.max_flow(source, sink, demand)
    if not need_usage:
        return value, None

    usage = [0] * m
    for eid, item in enumerate(refs):
        if item is not None:
            ref, initial = item
            usage[eid] = initial - ref[2]
    return value, usage


initial_caps = [c for _, _, c in edges]

# Immediate individual feasibility checks.
for s, t, d in commodities:
    if d > 0 and s != t:
        value, _ = route_one(s, t, d, initial_caps, None, False)
        if value < d:
            print("NO")
            sys.exit(0)


def greedy_attempt(order, randomized_edges):
    residual = initial_caps[:]
    answer = [[0] * m for _ in range(k)]

    for ci in order:
        s, t, d = commodities[ci]
        if d == 0 or s == t:
            continue

        if randomized_edges:
            eorder = list(range(m))
            random.shuffle(eorder)
        else:
            eorder = None

        value, usage = route_one(s, t, d, residual, eorder, True)
        if value < d:
            return None

        answer[ci] = usage
        for eid, x in enumerate(usage):
            residual[eid] -= x

    return answer


active = [i for i, (_, _, d) in enumerate(commodities) if d > 0]
first_order = sorted(
    active,
    key=lambda i: (
        len(out_edges[commodities[i][0]]),
        -commodities[i][2],
        i
    )
)

answer = greedy_attempt(first_order, False)
if answer is None:
    answer = greedy_attempt(list(reversed(first_order)), False)

attempt = 0
while answer is None and attempt < 60 and time.monotonic() - start_time < 1.45:
    order = active[:]
    random.shuffle(order)
    answer = greedy_attempt(order, True)
    attempt += 1

if answer is not None:
    out = ["YES"]
    out.extend(" ".join(map(str, row)) for row in answer)
    sys.stdout.write("\n".join(out))
    sys.exit(0)

# Exact path-allocation search for instances small enough to enumerate.
class SearchTimeout(Exception):
    pass


paths = [[] for _ in range(k)]

try:
    for ci, (source, sink, demand) in enumerate(commodities):
        if demand == 0 or source == sink:
            continue

        visited = [False] * n
        visited[source] = True
        current = []

        def enumerate_paths(u):
            if time.monotonic() > deadline:
                raise SearchTimeout
            if u == sink:
                paths[ci].append(tuple(current))
                return

            candidates = out_edges[u]
            for eid in candidates:
                _, v, cap = edges[eid]
                if cap <= 0 or visited[v]:
                    continue
                visited[v] = True
                current.append(eid)
                enumerate_paths(v)
                current.pop()
                visited[v] = False

        enumerate_paths(source)

        if not paths[ci]:
            print("NO")
            sys.exit(0)

        def path_key(p):
            scarcity = 0.0
            bottleneck = 10**30
            for eid in p:
                c = initial_caps[eid]
                scarcity += 1.0 / c
                if c < bottleneck:
                    bottleneck = c
            return (scarcity, len(p), -bottleneck, p)

        paths[ci].sort(key=path_key)

except SearchTimeout:
    print("NO")
    sys.exit(0)

residual = initial_caps[:]
remaining = [d if commodities[i][0] != commodities[i][1] else 0
             for i, (_, _, d) in enumerate(commodities)]
path_start = [0] * k
allocation = [[0] * m for _ in range(k)]
failed = set()
flow_cache = {}


def feasible_individually(ci):
    if remaining[ci] == 0:
        return True
    key = (ci, tuple(residual))
    cached = flow_cache.get(key)
    if cached is not None:
        return cached >= remaining[ci]
    s, t, _ = commodities[ci]
    value, _ = route_one(s, t, remaining[ci], residual, None, False)
    flow_cache[key] = value
    return value >= remaining[ci]


def exact_search():
    if time.monotonic() > deadline:
        raise SearchTimeout

    unfinished = [i for i in active if remaining[i] > 0]
    if not unfinished:
        return True

    state = (tuple(residual), tuple(remaining), tuple(path_start))
    if state in failed:
        return False

    chosen = -1
    chosen_candidates = None

    for ci in unfinished:
        if not feasible_individually(ci):
            failed.add(state)
            return False

        candidates = []
        plist = paths[ci]
        for pi in range(path_start[ci], len(plist)):
            p = plist[pi]
            bottleneck = remaining[ci]
            for eid in p:
                if residual[eid] < bottleneck:
                    bottleneck = residual[eid]
                    if bottleneck == 0:
                        break
            if bottleneck > 0:
                candidates.append((pi, p, bottleneck))

        if not candidates:
            failed.add(state)
            return False

        if chosen == -1 or len(candidates) < len(chosen_candidates):
            chosen = ci
            chosen_candidates = candidates

    old_start = path_start[chosen]
    old_remaining = remaining[chosen]

    for pi, p, bottleneck in chosen_candidates:
        max_take = min(bottleneck, old_remaining)

        quantities = [max_take]
        if max_take != old_remaining and old_remaining <= bottleneck:
            quantities.insert(0, old_remaining)

        if max_take <= 64:
            quantities = list(range(max_take, 0, -1))
        else:
            special = {
                max_take, 1, old_remaining,
                max_take // 2, max_take // 3,
                (2 * max_take) // 3
            }
            quantities = sorted(
                (q for q in special if 1 <= q <= max_take),
                reverse=True
            )
            if len(quantities) < max_take:
                quantities.extend(
                    q for q in range(max_take - 1, 0, -1)
                    if q not in special
                )

        for take in quantities:
            path_start[chosen] = pi + 1
            remaining[chosen] = old_remaining - take
            for eid in p:
                residual[eid] -= take
                allocation[chosen][eid] += take

            if exact_search():
                return True

            for eid in p:
                residual[eid] += take
                allocation[chosen][eid] -= take
            remaining[chosen] = old_remaining
            path_start[chosen] = old_start

    failed.add(state)
    return False


try:
    found = exact_search()
except SearchTimeout:
    found = False

if found:
    out = ["YES"]
    out.extend(" ".join(map(str, row)) for row in allocation)
    sys.stdout.write("\n".join(out))
else:
    print("NO")
