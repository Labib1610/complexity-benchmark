import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

it = iter(data)
n = next(it)
m = next(it)
k = next(it)

commodities = []
for _ in range(k):
    s = next(it) - 1
    t = next(it) - 1
    d = next(it)
    commodities.append((s, t, d))

edges = []
out_edges = [[] for _ in range(n)]
for eid in range(m):
    u = next(it) - 1
    v = next(it) - 1
    c = next(it)
    edges.append((u, v, c))
    if u != v:
        out_edges[u].append(eid)

for adj in out_edges:
    adj.sort()


class Dinic:
    __slots__ = ("n", "g", "level", "pos")

    def __init__(self, size):
        self.n = size
        self.g = [[] for _ in range(size)]
        self.level = [-1] * size
        self.pos = [0] * size

    def add_edge(self, u, v, cap, eid):
        a = [v, len(self.g[v]), cap, eid, cap]
        b = [u, len(self.g[u]), 0, -1, 0]
        self.g[u].append(a)
        self.g[v].append(b)

    def bfs(self, s, t):
        level = self.level
        for i in range(self.n):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            nl = level[u] + 1
            for e in self.g[u]:
                if e[2] > 0 and level[e[0]] < 0:
                    level[e[0]] = nl
                    if e[0] == t:
                        return True
                    q.append(e[0])
        return level[t] >= 0

    def dfs(self, u, t, pushed):
        if u == t:
            return pushed
        g = self.g
        level = self.level
        while self.pos[u] < len(g[u]):
            idx = self.pos[u]
            e = g[u][idx]
            v = e[0]
            if e[2] > 0 and level[v] == level[u] + 1:
                got = self.dfs(v, t, min(pushed, e[2]))
                if got:
                    e[2] -= got
                    g[v][e[1]][2] += got
                    return got
            self.pos[u] += 1
        return 0

    def flow(self, s, t, limit):
        total = 0
        while total < limit and self.bfs(s, t):
            self.pos = [0] * self.n
            while total < limit:
                got = self.dfs(s, t, limit - total)
                if not got:
                    break
                total += got
        return total


normal_order = list(range(m))
reverse_order = normal_order[::-1]


def get_flow(s, t, need, capacities, mode=0):
    if need <= 0 or s == t:
        return need, []
    din = Dinic(n)
    refs = []
    order = normal_order if mode == 0 else reverse_order
    for eid in order:
        cap = capacities[eid]
        if cap <= 0:
            continue
        u, v, _ = edges[eid]
        if u == v:
            continue
        idx = len(din.g[u])
        din.add_edge(u, v, cap, eid)
        refs.append((u, idx, eid, cap))
    value = din.flow(s, t, need)
    allocation = []
    if value:
        for u, idx, eid, original_cap in refs:
            used = original_cap - din.g[u][idx][2]
            if used:
                allocation.append((eid, used))
    return value, allocation


initial_cap = [c for _, _, c in edges]
effective_demand = [
    0 if commodities[i][0] == commodities[i][1] else commodities[i][2]
    for i in range(k)
]

initial_values = [0] * k
for i, (s, t, _) in enumerate(commodities):
    d = effective_demand[i]
    if d == 0:
        initial_values[i] = 0
        continue
    value, _ = get_flow(s, t, d, initial_cap)
    if value < d:
        print("NO")
        sys.exit()
    initial_values[i] = value

active = [i for i in range(k) if effective_demand[i] > 0]

tight_order = sorted(
    active,
    key=lambda i: (
        initial_values[i] - effective_demand[i],
        initial_values[i],
        -effective_demand[i],
        i
    )
)

candidate_orders = [
    tuple(active),
    tuple(tight_order),
    tuple(reversed(tight_order)),
    tuple(sorted(active, key=lambda i: (-effective_demand[i], i))),
    tuple(sorted(active, key=lambda i: (effective_demand[i], i))),
]

seen_orders = set()
candidate_orders = [
    order for order in candidate_orders
    if not (order in seen_orders or seen_orders.add(order))
]


def greedy_attempt(order, mode):
    capacities = initial_cap[:]
    answer = [[0] * m for _ in range(k)]
    for ci in order:
        s, t, _ = commodities[ci]
        need = effective_demand[ci]
        value, allocation = get_flow(s, t, need, capacities, mode)
        if value < need:
            return None
        for eid, amount in allocation:
            capacities[eid] -= amount
            answer[ci][eid] += amount
    return answer


for order in candidate_orders:
    for mode in (0, 1):
        result = greedy_attempt(order, mode)
        if result is not None:
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in result)
            sys.stdout.write("\n".join(out))
            sys.exit()

order = tight_order
capacities = initial_cap[:]
answer = [[0] * m for _ in range(k)]


def future_individually_possible(pos):
    for p in range(pos, len(order)):
        ci = order[p]
        s, t, _ = commodities[ci]
        need = effective_demand[ci]
        value, _ = get_flow(s, t, need, capacities)
        if value < need:
            return False
    return True


def generate_paths(s, t, last_path):
    visited = [False] * n
    visited[s] = True
    path = []

    def dfs(u):
        if u == t:
            candidate = tuple(path)
            if last_path is None or candidate > last_path:
                yield candidate
            return

        for eid in out_edges[u]:
            if capacities[eid] <= 0:
                continue
            v = edges[eid][1]
            if visited[v]:
                continue
            visited[v] = True
            path.append(eid)
            yield from dfs(v)
            path.pop()
            visited[v] = False

    yield from dfs(s)


def solve_position(pos):
    if pos == len(order):
        return True

    if not future_individually_possible(pos):
        return False

    ci = order[pos]
    return fill_commodity(pos, ci, effective_demand[ci], None)


def fill_commodity(pos, ci, remaining, last_path):
    if remaining == 0:
        return solve_position(pos + 1)

    s, t, _ = commodities[ci]

    value, allocation = get_flow(s, t, remaining, capacities)
    if value < remaining:
        return False

    for eid, amount in allocation:
        capacities[eid] -= amount
        answer[ci][eid] += amount

    if solve_position(pos + 1):
        return True

    for eid, amount in allocation:
        capacities[eid] += amount
        answer[ci][eid] -= amount

    for path in generate_paths(s, t, last_path):
        bottleneck = remaining
        for eid in path:
            if capacities[eid] < bottleneck:
                bottleneck = capacities[eid]

        for amount in range(bottleneck, 0, -1):
            for eid in path:
                capacities[eid] -= amount
                answer[ci][eid] += amount

            if fill_commodity(pos, ci, remaining - amount, path):
                return True

            for eid in path:
                capacities[eid] += amount
                answer[ci][eid] -= amount

    return False


if solve_position(0):
    out = ["YES"]
    out.extend(" ".join(map(str, row)) for row in answer)
    sys.stdout.write("\n".join(out))
else:
    print("NO")
