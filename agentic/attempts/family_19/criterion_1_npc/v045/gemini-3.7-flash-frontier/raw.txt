```python
import sys
import heapq
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    consignments = []
    total_demand = 0
    for i in range(k):
        s = int(next(iterator)) - 1
        t = int(next(iterator)) - 1
        d = int(next(iterator))
        consignments.append((s, t, d))
        total_demand += d
        
    edges = []
    adj = [[] for _ in range(n)]
    for e_idx in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        c = int(next(iterator))
        edges.append((u, v, c))
        adj[u].append((v, e_idx))
        
    # Check simple feasibility: max total capacity vs demand
    # If k == 1, we can just run Dinic's Algorithm / Max Flow
    if k == 1:
        s, t, d = consignments[0]
        # Dinic's algorithm
        class Dinic:
            def __init__(self, num_nodes):
                self.n = num_nodes
                self.graph = [[] for _ in range(num_nodes)]
                self.edges = []
            def add_edge(self, u, v, cap, orig_idx):
                self.graph[u].append(len(self.edges))
                self.edges.append([u, v, cap, 0, orig_idx])
                self.graph[v].append(len(self.edges))
                self.edges.append([v, u, 0, 0, -1])
            def bfs(self, s, t, level):
                for i in range(self.n): level[i] = -1
                level[s] = 0
                q = [s]
                for u in q:
                    for e_idx in self.graph[u]:
                        u2, v2, cap, flow, _ = self.edges[e_idx]
                        if cap > flow and level[v2] == -1:
                            level[v2] = level[u] + 1
                            q.append(v2)
                return level[t] != -1
            def dfs(self, u, t, pushed, level, ptr):
                if pushed == 0 or u == t:
                    return pushed
                for i in range(ptr[u], len(self.graph[u])):
                    ptr[u] = i
                    e_idx = self.graph[u][i]
                    u2, v2, cap, flow, _ = self.edges[e_idx]
                    tr = cap - flow
                    if level[u2] + 1 != level[v2] or tr == 0:
                        continue
                    push = self.dfs(v2, t, min(pushed, tr), level, ptr)
                    if push == 0:
                        continue
                    self.edges[e_idx][3] += push
                    self.edges[e_idx ^ 1][3] -= push
                    return push
                return 0
            def max_flow(self, s, t, limit):
                flow = 0
                level = [-1] * self.n
                while flow < limit and self.bfs(s, t, level):
                    ptr = [0] * self.n
                    while True:
                        pushed = self.dfs(s, t, limit - flow, level, ptr)
                        if pushed == 0:
                            break
                        flow += pushed
                return flow

        dinic = Dinic(n)
        for e_idx, (u, v, c) in enumerate(edges):
            dinic.add_edge(u, v, c, e_idx)
        
        flow = dinic.max_flow(s, t, d)
        if flow < d:
            print("NO")
            return
        
        flow_on_edge = [0] * m
        for e in dinic.edges:
            orig = e[4]
            if orig != -1 and e[3] > 0:
                flow_on_edge[orig] = e[3]
        
        print("YES")
        print(*(flow_on_edge))
        return

    # Expand items
    items = []
    for c_idx, (s, t, d) in enumerate(consignments):
        for _ in range(d):
            items.append((s, t, c_idx))
            
    num_items = len(items)
    cap = [edges[e][2] for e in range(m)]
    usage = [0] * m
    history = [0] * m
    paths = [None] * num_items

    def dijkstra(s, t, item_idx):
        dist = {s: 0}
        parent = {}
        pq = [(0, s)]
        while pq:
            d_u, u = heapq.heappop(pq)
            if d_u > dist.get(u, float('inf')):
                continue
            if u == t:
                break
            for v, e_idx in adj[u]:
                c = cap[e_idx]
                u_cnt = usage[e_idx]
                cost = 1 + history[e_idx] * 2
                if u_cnt >= c:
                    cost += (u_cnt - c + 1) * 50
                d_v = d_u + cost
                if d_v < dist.get(v, float('inf')):
                    dist[v] = d_v
                    parent[v] = (u, e_idx)
                    heapq.heappush(pq, (d_v, v))
                    
        if t not in dist:
            return None
        path = []
        curr = t
        while curr != s:
            prev, e_idx = parent[curr]
            path.append(e_idx)
            curr = prev
        path.reverse()
        return path

    # Initial routing
    for i in range(num_items):
        s, t, c_idx = items[i]
        p = dijkstra(s, t, i)
        if p is None:
            print("NO")
            return
        paths[i] = p
        for e in p:
            usage[e] += 1

    # Pathfinder iterations
    max_iter = 120
    solved = False
    for it in range(max_iter):
        over_capacity = False
        for e in range(m):
            if usage[e] > cap[e]:
                over_capacity = True
                history[e] += (usage[e] - cap[e])
        
        if not over_capacity:
            solved = True
            break
            
        order = list(range(num_items))
        random.shuffle(order)
        for i in order:
            s, t, c_idx = items[i]
            # Rip up
            for e in paths[i]:
                usage[e] -= 1
            # Reroute
            p = dijkstra(s, t, i)
            if p is None:
                p = paths[i] # restore
            paths[i] = p
            for e in paths[i]:
                usage[e] += 1

    # Final check
    if not solved:
        for e in range(m):
            if usage[e] > cap[e]:
                print("NO")
                return

    ans = [[0] * m for _ in range(k)]
    for i in range(num_items):
        c_idx = items[i][2]
        for e in paths[i]:
            ans[c_idx][e] += 1

    print("YES")
    for i in range(k):
        print(*(ans[i]))

if __name__ == '__main__':
    solve()
```
