import sys
sys.setrecursionlimit(1_000_000)


class Dinic:
    class Edge:
        __slots__ = ("to", "rev", "cap", "initial")

        def __init__(self, to, rev, cap):
            self.to = to
            self.rev = rev
            self.cap = cap
            self.initial = cap

    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, u, v, cap):
        a = self.Edge(v, len(self.g[v]), cap)
        b = self.Edge(u, len(self.g[u]), 0)
        self.g[u].append(a)
        self.g[v].append(b)
        return a

    def max_flow(self, s, t, limit):
        flow = 0
        n = self.n
        while flow < limit:
            level = [-1] * n
            level[s] = 0
            q = [s]
            for u in q:
                for e in self.g[u]:
                    if e.cap > 0 and level[e.to] < 0:
                        level[e.to] = level[u] + 1
                        q.append(e.to)
            if level[t] < 0:
                break

            it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                while it[u] < len(self.g[u]):
                    e = self.g[u][it[u]]
                    if e.cap > 0 and level[e.to] == level[u] + 1:
                        got = dfs(e.to, min(pushed, e.cap))
                        if got:
                            e.cap -= got
                            self.g[e.to][e.rev].cap += got
                            return got
                    it[u] += 1
                return 0

            while flow < limit:
                pushed = dfs(s, limit - flow)
                if not pushed:
                    break
                flow += pushed
        return flow


def common_terminal_solution(n, m, orders, edges):
    active = [i for i, (_, _, d) in enumerate(orders) if d > 0]
    if not active:
        return [[0] * m for _ in orders]

    same_sink = all(orders[i][1] == orders[active[0]][1] for i in active)
    same_source = all(orders[i][0] == orders[active[0]][0] for i in active)

    if not same_sink and not same_source:
        return None

    total = sum(orders[i][2] for i in active)
    answer = [[0] * m for _ in orders]

    if same_sink:
        terminal = orders[active[0]][1]
        super_source = n
        dinic = Dinic(n + 1)
        refs = []
        for u, v, c in edges:
            refs.append(dinic.add_edge(u, v, c))
        for i in active:
            s, _, d = orders[i]
            dinic.add_edge(super_source, s, d)

        if dinic.max_flow(super_source, terminal, total) != total:
            return False
    else:
        source = orders[active[0]][0]
        super_sink = n
        dinic = Dinic(n + 1)
        refs = []
        for u, v, c in edges:
            refs.append(dinic.add_edge(u, v, c))
        for i in active:
            _, t, d = orders[i]
            dinic.add_edge(t, super_sink, d)

        if dinic.max_flow(source, super_sink, total) != total:
            return False

    remaining_flow = [ref.initial - ref.cap for ref in refs]
    out_edges = [[] for _ in range(n)]
    for j, (u, v, _) in enumerate(edges):
        out_edges[u].append(j)

    for idx in active:
        s, t, need = orders[idx]
        while need > 0:
            parent_edge = [-1] * n
            seen = [False] * n
            seen[s] = True
            queue = [s]

            for u in queue:
                if u == t:
                    break
                for eid in out_edges[u]:
                    if remaining_flow[eid] <= 0:
                        continue
                    v = edges[eid][1]
                    if not seen[v]:
                        seen[v] = True
                        parent_edge[v] = eid
                        queue.append(v)

            if not seen[t]:
                return None

            path = []
            cur = t
            amount = need
            while cur != s:
                eid = parent_edge[cur]
                if eid < 0:
                    return None
                path.append(eid)
                amount = min(amount, remaining_flow[eid])
                cur = edges[eid][0]

            for eid in path:
                remaining_flow[eid] -= amount
                answer[idx][eid] += amount
            need -= amount

    return answer


def milp_solution(n, m, k, orders, edges):
    try:
        import numpy as np
        from scipy.optimize import milp, LinearConstraint, Bounds
        from scipy.sparse import coo_matrix
    except Exception:
        return None

    variables = k * m
    flow_rows_per_commodity = max(0, n - 1)
    flow_rows = k * flow_rows_per_commodity
    rows_count = flow_rows + m

    row_idx = []
    col_idx = []
    values = []
    lower = np.full(rows_count, -np.inf, dtype=float)
    upper = np.empty(rows_count, dtype=float)

    omitted = n - 1

    for i, (s, t, d) in enumerate(orders):
        balances = {}
        balances[s] = balances.get(s, 0) + d
        balances[t] = balances.get(t, 0) - d

        base_row = i * flow_rows_per_commodity
        for v in range(n - 1):
            b = balances.get(v, 0)
            lower[base_row + v] = b
            upper[base_row + v] = b

        base_col = i * m
        for j, (u, v, _) in enumerate(edges):
            col = base_col + j
            if u != omitted:
                row_idx.append(base_row + u)
                col_idx.append(col)
                values.append(1.0)
            if v != omitted:
                row_idx.append(base_row + v)
                col_idx.append(col)
                values.append(-1.0)

    for j, (_, _, cap) in enumerate(edges):
        row = flow_rows + j
        upper[row] = cap
        for i in range(k):
            row_idx.append(row)
            col_idx.append(i * m + j)
            values.append(1.0)

    matrix = coo_matrix(
        (values, (row_idx, col_idx)),
        shape=(rows_count, variables),
        dtype=float
    ).tocsr()

    var_upper = np.tile(np.array([c for _, _, c in edges], dtype=float), k)

    try:
        result = milp(
            c=np.zeros(variables, dtype=float),
            integrality=np.ones(variables, dtype=np.uint8),
            bounds=Bounds(np.zeros(variables, dtype=float), var_upper),
            constraints=LinearConstraint(matrix, lower, upper),
            options={"presolve": True}
        )
    except Exception:
        return None

    if result.x is None or not result.success:
        return False

    rounded = np.rint(result.x).astype(object)
    answer = []
    for i in range(k):
        answer.append([int(rounded[i * m + j]) for j in range(m)])

    capacity_used = [0] * m
    for i, (s, t, d) in enumerate(orders):
        balance = [0] * n
        for j, (u, v, c) in enumerate(edges):
            x = answer[i][j]
            if x < 0 or x > c:
                return None
            balance[u] += x
            balance[v] -= x
            capacity_used[j] += x

        expected = [0] * n
        expected[s] += d
        expected[t] -= d
        if balance != expected:
            return None

    for j, (_, _, c) in enumerate(edges):
        if capacity_used[j] > c:
            return None

    return answer


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    orders = []
    for _ in range(k):
        s = next(it) - 1
        t = next(it) - 1
        d = next(it)
        orders.append((s, t, d))

    edges = []
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        c = next(it)
        edges.append((u, v, c))

    answer = common_terminal_solution(n, m, orders, edges)

    if answer is None:
        answer = milp_solution(n, m, k, orders, edges)

    if answer is False or answer is None:
        print("NO")
        return

    out = ["YES"]
    out.extend(" ".join(map(str, row)) for row in answer)
    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
