```python
import sys
import time
import random
from collections import deque

sys.setrecursionlimit(1_000_000)


class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, u, v, cap):
        a = [v, len(self.g[v]), cap]
        b = [u, len(self.g[u]), 0]
        self.g[u].append(a)
        self.g[v].append(b)
        return a

    def max_flow(self, s, t, limit):
        if s == t:
            return limit

        n = self.n
        g = self.g
        total = 0

        while total < limit:
            level = [-1] * n
            level[s] = 0
            q = deque([s])

            while q:
                u = q.popleft()
                nl = level[u] + 1
                for e in g[u]:
                    if e[2] > 0 and level[e[0]] < 0:
                        level[e[0]] = nl
                        q.append(e[0])

            if level[t] < 0:
                break

            it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                gu = g[u]
                while it[u] < len(gu):
                    e = gu[it[u]]
                    if e[2] > 0 and level[e[0]] == level[u] + 1:
                        got = dfs(e[0], min(pushed, e[2]))
                        if got:
                            e[2] -= got
                            g[e[0]][e[1]][2] += got
                            return got
                    it[u] += 1
                return 0

            while total < limit:
                pushed = dfs(s, limit - total)
                if not pushed:
                    break
                total += pushed

        return total


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    p = 0
    n, m, k = data[p], data[p + 1], data[p + 2]
    p += 3

    commodities = []
    for _ in range(k):
        s, t, d = data[p] - 1, data[p + 1] - 1, data[p + 2]
        p += 3
        commodities.append((s, t, d))

    edges = []
    capacities = []
    for _ in range(m):
        u, v, c = data[p] - 1, data[p + 1] - 1, data[p + 2]
        p += 3
        edges.append((u, v))
        capacities.append(c)

    if all(d == 0 or s == t for s, t, d in commodities):
        out = ["YES"]
        zero = " ".join(["0"] * m)
        out.extend(zero for _ in range(k))
        sys.stdout.write("\n".join(out))
        return

    rng = random.Random(0x5EED1234)
    start_time = time.perf_counter()
    deadline = start_time + 4.35

    base_orders = [
        list(range(k)),
        list(reversed(range(k))),
        sorted(range(k), key=lambda i: -commodities[i][2]),
        sorted(range(k), key=lambda i: commodities[i][2]),
    ]

    def attempt(order, reverse_graph, shuffle_edges):
        remaining = capacities[:]
        answer = [[0] * m for _ in range(k)]
        edge_order = list(range(m))
        if shuffle_edges:
            rng.shuffle(edge_order)

        for ci in order:
            s, t, demand = commodities[ci]
            if demand == 0 or s == t:
                continue

            dinic = Dinic(n)
            refs = [None] * m

            if reverse_graph:
                source, sink = t, s
                for eid in edge_order:
                    cap = remaining[eid]
                    if cap:
                        u, v = edges[eid]
                        ref = dinic.add_edge(v, u, cap)
                        refs[eid] = (ref, cap)
            else:
                source, sink = s, t
                for eid in edge_order:
                    cap = remaining[eid]
                    if cap:
                        u, v = edges[eid]
                        ref = dinic.add_edge(u, v, cap)
                        refs[eid] = (ref, cap)

            if dinic.max_flow(source, sink, demand) != demand:
                return None

            row = answer[ci]
            for eid in edge_order:
                item = refs[eid]
                if item is not None:
                    ref, initial = item
                    used = initial - ref[2]
                    if used:
                        row[eid] = used
                        remaining[eid] -= used

        return answer

    attempt_no = 0
    while attempt_no < 256 and time.perf_counter() < deadline:
        if attempt_no < len(base_orders):
            order = base_orders[attempt_no][:]
            reverse_graph = False
            shuffle_edges = attempt_no != 0
        elif attempt_no < 2 * len(base_orders):
            order = base_orders[attempt_no - len(base_orders)][:]
            reverse_graph = True
            shuffle_edges = True
        else:
            order = list(range(k))
            rng.shuffle(order)
            reverse_graph = bool(rng.getrandbits(1))
            shuffle_edges = True

        result = attempt(order, reverse_graph, shuffle_edges)
        if result is not None:
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in result)
            sys.stdout.write("\n".join(out))
            return

        attempt_no += 1

    sys.stdout.write("NO")


if __name__ == "__main__":
    main()
```
