from collections import defaultdict, deque
import sys

def solve():
    n, m, k = map(int, input().split())
    
    categories = []
    for i in range(k):
        s, t, d = map(int, input().split())
        categories.append((s, t, d))
    
    chutes = []
    chute_map = {}
    for i in range(m):
        u, v, cap = map(int, input().split())
        chutes.append((u, v, cap))
        chute_map[(u, v)] = i
    
    # Build graph
    graph = defaultdict(list)
    for u, v, cap in chutes:
        graph[u].append(v)
        graph[v].append(u)  # We need this for finding paths
    
    # Check if there's a path for each category
    def bfs_path(start, end):
        if start == end:
            return []
        queue = deque([(start, [])])
        visited = set([start])
        while queue:
            node, path = queue.popleft()
            for neighbor in graph[node]:
                if neighbor == end:
                    return path + [(node, neighbor)]
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, path + [(node, neighbor)]))
        return None
    
    # Find paths for each category
    paths = []
    for s, t, d in categories:
        path = bfs_path(s, t)
        if path is None:
            print("NO")
            return
        paths.append(path)
    
    # Now we model this as a flow problem
    # But since we have multiple commodities, we can't directly use max-flow
    # However, we can try to route each commodity separately and see if total usage is within capacity
    
    # Initialize flow on chutes to zero
    flow_on_chutes = [0] * m
    result_flows = []
    
    # For each category, find how much flow we need to send
    # We'll use a simple approach: for each category, find a path and send all flow along it
    # This might not work in general but let's try it
    
    category_flows = []
    for idx, (s, t, d) in enumerate(categories):
        # Try to find a path from s to t using BFS
        # And record the flow needed on each edge
        
        # Build adjacency list for this search
        adj = defaultdict(list)
        edge_index = {}  # Maps (u,v) to index in chutes
        
        for i, (u, v, cap) in enumerate(chutes):
            adj[u].append((v, i))
            adj[v].append((u, i))  # For traversal, we add both directions
            edge_index[(u,v)] = i
            
        # BFS to find path
        prev = [-1] * (n+1)
        prev_edge = [-1] * (n+1)
        queue = deque([s])
        visited = [False] * (n+1)
        visited[s] = True
        
        found = False
        while queue:
            u = queue.popleft()
            if u == t:
                found = True
                break
            for v, edge_idx in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    prev[v] = u
                    prev_edge[v] = edge_idx
                    queue.append(v)
        
        if not found:
            print("NO")
            return
            
        # Reconstruct path
        path_edges = []
        curr = t
        while prev[curr] != -1:
            edge_idx = prev_edge[curr]
            path_edges.append(edge_idx)
            curr = prev[curr]
            
        path_edges.reverse()
        
        # Record this path for this category
        category_flow = [0] * m
        for edge_idx in path_edges:
            category_flow[edge_idx] += d
            
        category_flows.append(category_flow)
        
    # Now check if total flow on any chute exceeds capacity
    total_flow = [0] * m
    for cat_flow in category_flows:
        for i in range(m):
            total_flow[i] += cat_flow[i]
            
    for i in range(m):
        if total_flow[i] > chutes[i][2]:  # capacity
            print("NO")
            return
            
    print("YES")
    for cat_flow in category_flows:
        print(' '.join(map(str, cat_flow)))

def main():
    solve()

if __name__ == "__main__":
    main()
