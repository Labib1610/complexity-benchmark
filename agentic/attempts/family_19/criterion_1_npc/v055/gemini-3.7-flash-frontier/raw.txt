import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    clients = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        clients.append((s, t, d))

    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))

    # Try solving using scipy.optimize.milp
    try:
        import numpy as np
        from scipy.optimize import Bounds, LinearConstraint, milp

        num_vars = k * m
        c = np.zeros(num_vars)
        integrality = np.ones(num_vars)

        lb = np.zeros(num_vars)
        ub = np.zeros(num_vars)
        for i in range(k):
            for e in range(m):
                ub[i * m + e] = edges[e][2]
        bounds = Bounds(lb, ub)

        rows = []
        lhs = []
        rhs = []

        # Capacity constraints
        for e in range(m):
            row = np.zeros(num_vars)
            for i in range(k):
                row[i * m + e] = 1.0
            rows.append(row)
            lhs.append(0.0)
            rhs.append(float(edges[e][2]))

        # Flow conservation constraints
        for i in range(k):
            s_i, t_i, d_i = clients[i]
            for v in range(1, n + 1):
                row = np.zeros(num_vars)
                for e, (u_e, v_e, _) in enumerate(edges):
                    if u_e == v:
                        row[i * m + e] += 1.0
                    if v_e == v:
                        row[i * m + e] -= 1.0

                supply = 0.0
                if v == s_i:
                    supply += d_i
                if v == t_i:
                    supply -= d_i

                rows.append(row)
                lhs.append(supply)
                rhs.append(supply)

        A = np.array(rows)
        constraints = LinearConstraint(A, lhs, rhs)

        res = milp(
            c=c, integrality=integrality, constraints=constraints, bounds=bounds
        )

        if res.success:
            x = np.round(res.x).astype(int)
            print("YES")
            for i in range(k):
                print(*(x[i * m + e] for e in range(m)))
            return
        else:
            print("NO")
            return

    except ImportError:
        pass

    # Fallback DFS / Backtracking for environments without scipy
    adj = [[] for _ in range(n + 1)]
    for e, (u, v, cap) in enumerate(edges):
        adj[u].append((v, e))

    rem_cap = [edges[e][2] for e in range(m)]
    flow = [[0] * m for _ in range(k)]

    tasks = []
    for i, (s, t, d) in enumerate(clients):
        for _ in range(d):
            tasks.append((i, s, t))

    def dfs_path(s, t, visited):
        if s == t:
            return []
        visited.add(s)
        for v, e in adj[s]:
            if rem_cap[e] > 0 and v not in visited:
                p = dfs_path(v, t, visited)
                if p is not None:
                    return [e] + p
        return None

    def backtrack(task_idx):
        if task_idx == len(tasks):
            return True
        c_idx, s, t = tasks[task_idx]

        # Find simple paths using DFS
        def find_all_paths(curr, dest, visited, path_edges):
            if curr == dest:
                yield list(path_edges)
                return
            for nxt, e in adj[curr]:
                if rem_cap[e] > 0 and nxt not in visited:
                    visited.add(nxt)
                    path_edges.append(e)
                    yield from find_all_paths(nxt, dest, visited, path_edges)
                    path_edges.pop()
                    visited.remove(nxt)

        for path in find_all_paths(s, t, {s}, []):
            for e in path:
                rem_cap[e] -= 1
                flow[c_idx][e] += 1

            if backtrack(task_idx + 1):
                return True

            for e in path:
                rem_cap[e] += 1
                flow[c_idx][e] -= 1

        return False

    if backtrack(0):
        print("YES")
        for i in range(k):
            print(*(flow[i][e] for e in range(m)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
