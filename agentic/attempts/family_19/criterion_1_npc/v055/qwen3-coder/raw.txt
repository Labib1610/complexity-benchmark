from collections import defaultdict, deque
import sys

def solve():
    n, m, k = map(int, input().split())
    
    clients = []
    for _ in range(k):
        s, t, d = map(int, input().split())
        clients.append((s, t, d))
    
    edges = []
    edge_index = {}
    adj = defaultdict(list)
    reverse_adj = defaultdict(list)
    
    for i in range(m):
        u, v, limit = map(int, input().split())
        edges.append((u, v, limit))
        edge_index[(u, v)] = i
        adj[u].append(v)
        reverse_adj[v].append(u)
    
    # Build super source and super sink
    SUPER_SOURCE = 0
    SUPER_SINK = n + 1
    
    # Add edges from super source to each client's factory
    # Add edges from each client's distribution to super sink
    client_demands = []
    for i in range(k):
        s, t, d = clients[i]
        client_demands.append(((SUPER_SOURCE, s), d, f"source_to_factory_{i}"))
        client_demands.append(((t, SUPER_SINK), d, f"distribution_to_sink_{i}"))
    
    # Create combined graph with all clients
    combined_edges = []
    combined_edge_limits = []
    
    # Original edges with shared capacity
    for u, v, limit in edges:
        combined_edges.append((u, v))
        combined_edge_limits.append(limit)
    
    # Client demand edges with infinite capacity
    for (u, v), demand, name in client_demands:
        combined_edges.append((u, v))
        combined_edge_limits.append(demand)  # Capacity equals demand
    
    # Check if flow is possible using max flow
    def bfs_max_flow(edges_list, capacities, source, sink, node_count):
        # Build adjacency list
        graph = defaultdict(lambda: defaultdict(int))
        for i, (u, v) in enumerate(edges_list):
            graph[u][v] += capacities[i]  # Multiple edges get summed
        
        # BFS to find augmenting paths
        def bfs():
            parent = {source: None}
            queue = deque([source])
            while queue:
                u = queue.popleft()
                if u == sink:
                    path = []
                    current = sink
                    while parent[current] is not None:
                        path.append((parent[current], current))
                        current = parent[current]
                    return path[::-1]
                
                for v in graph[u]:
                    if v not in parent and graph[u][v] > 0:
                        parent[v] = u
                        queue.append(v)
            return None
        
        max_flow = 0
        while True:
            path = bfs()
            if path is None:
                break
            
            # Find minimum capacity along the path
            min_capacity = float('inf')
            for u, v in path:
                min_capacity = min(min_capacity, graph[u][v])
            
            # Update residual capacities
            for u, v in path:
                graph[u][v] -= min_capacity
                graph[v][u] += min_capacity
            
            max_flow += min_capacity
        
        return max_flow
    
    # Total demand
    total_demand = sum(client[2] for client in clients)
    
    # Check if maximum flow can satisfy total demand
    # We need to create a proper flow network
    def max_flow_with_demands():
        # Nodes: 0 (super source), 1..n (junctions), n+1 (super sink)
        nodes = list(range(n + 2))
        
        # Build graph with capacities
        graph = defaultdict(lambda: defaultdict(int))
        
        # Add original track segments
        for u, v, limit in edges:
            graph[u][v] += limit
        
        # Add client demands as infinite capacity edges initially
        # But we'll check if we can route required flows
        
        # For max flow with vertex demands, we convert to edge flows
        # Each client needs to send d_i flow from s_i to t_i
        
        # Use successive shortest path or just check if total possible flow >= total_demand
        # But we need individual flows per client
        
        # Let's use a different approach - solve for each client separately with capacity constraints
        
        # Actually, let's model this properly as a multi-commodity flow problem
        # But since that's NP-hard, we'll try to find a feasible solution using flow networks
        
        # Better approach: Create a flow network where we try to route all demands
        # But respecting shared edge capacities
        
        # Model as min-cost flow would work but complex
        # Let's try iterative approach:
        # 1. Route each client optimally ignoring others
        # 2. Check if total usage exceeds limits
        # 3. Try to adjust
        
        # Even simpler: Check if the sum of individual max flows >= total demand
        # And that no edge exceeds capacity
        
        # Most straightforward: Use linear programming formulation
        # But we'll implement a heuristic approach
        
        # Let's try Ford-Fulkerson for each client, tracking global edge usage
        
        edge_usage = [0] * m  # Track how much each edge is used
        
        client_flows = []
        
        for client_idx in range(k):
            s, t, d = clients[client_idx]
            
            # Find max flow for this client given current edge usage
            def find_flow_for_client(source, sink, demand, current_usage):
                # Build residual graph considering current usage
                res_graph = defaultdict(lambda: defaultdict(int))
                
                # Add track segments with remaining capacity
                for i, (u, v, limit) in enumerate(edges):
                    remaining = limit - current_usage[i]
                    if remaining > 0:
                        res_graph[u][v] = remaining
                        res_graph[v][u] = 0  # Back edge starts at 0
                
                # Ford-Fulkerson
                def bfs_path():
                    parent = {source: None}
                    edge_flow = {}
                    queue = deque([source])
                    while queue:
                        u = queue.popleft()
                        if u == sink:
                            path = []
                            current = sink
                            while parent[current] is not None:
                                path.append((parent[current], current))
                                current = parent[current]
                            return path[::-1]
                        
                        for v in res_graph[u]:
                            if v not in parent and res_graph[u][v] > 0:
                                parent[v] = u
                                queue.append(v)
                    return None
                
                total_flow = 0
                flow_paths = []
                
                while total_flow < demand:
                    path = bfs_path()
                    if path is None:
                        break
                    
                    min_cap = min(res_graph[u][v] for u, v in path)
                    actual_flow = min(min_cap, demand - total_flow)
                    
                    for u, v in path:
                        res_graph[u][v] -= actual_flow
                        res_graph[v][u] += actual_flow
                    
                    total_flow += actual_flow
                    flow_paths.append((path, actual_flow))
                
                if total_flow < demand:
                    return None, None  # Cannot satisfy demand
                
                return flow_paths, total_flow
            
            flow_result, flow_value = find_flow_for_client(s, t, d, edge_usage)
            
            if flow_result is None:
                return False, [], []
            
            # Convert flow result to edge usage increments
            client_edge_flow = [0] * m
            for path, flow_amount in flow_result:
                for u, v in path:
                    if (u, v) in edge_index:
                        idx = edge_index[(u, v)]
                        client_edge_flow[idx] += flow_amount
            
            client_flows.append(client_edge_flow)
            
            # Update global edge usage
            for i in range(m):
                edge_usage[i] += client_edge_flow[i]
                if edge_usage[i] > edges[i][2]:  # Exceeds capacity
                    return False, [], []
        
        return True, client_flows, edge_usage
    
    # Run the flow algorithm
    success, flows, usage = max_flow_with_demands()
    
    if not success:
        print("NO")
        return
    
    print("YES")
    for flow in flows:
        print(' '.join(map(str, flow)))
