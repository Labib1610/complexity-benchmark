from collections import defaultdict, deque

    # We'll solve this as a multi-commodity flow problem using iterative approach
    # First check if it's possible by solving each product independently with 
    # ignoring other flows, then check if total flow on each edge is within capacity
    
    # But that won't work because flows are coupled
    
    # Let's model this as a single min-cost flow problem with k commodities
    # Actually, let's try a different approach - solve iteratively, adjusting capacities
    
    # Better approach: Use linear programming or iterative rounding
    # Since this is a coding competition, let's try a practical heuristic
    
    # Try to route each product one by one, reducing available capacity
    
    from collections import defaultdict, deque
    
    def bfs_path(graph, source, sink, capacity_left, edge_flow_map):
        parent = {source: None}
        queue = deque([source])
        
        while queue:
            u = queue.popleft()
            if u == sink:
                break
                
            for v, edge_id in graph[u]:
                if (u, v, edge_id) not in edge_flow_map:
                    edge_flow_map[(u, v, edge_id)] = 0
                    
                if edge_flow_map[(u, v, edge_id)] < capacity_left[edge_id] and v not in parent:
                    parent[v] = u
                    prev_edge_id = edge_id
                    queue.append(v)
                    
        if sink not in parent:
            return None
            
        path = []
        curr = sink
        while parent[curr] is not None:
            prev = parent[curr]
            edge_idx = edge_ids[(prev, curr)]
            path.append((prev, curr, edge_idx))
            curr = prev
            
        path.reverse()
        return path
    
    def find_flow(graph, source, sink, demand, capacity_left):
        # Find a path from source to sink
        # Use DFS/BFS to find augmenting paths
        
        # Build residual graph
        res_graph = defaultdict(list)
        forward_edges = {}
        backward_edges = {}
        
        for u in graph:
            for v, edge_id in graph[u]:
                # Forward edge
                res_graph[u].append((v, edge_id, 'forward'))
                forward_edges[(u, v, edge_id)] = capacity_left[edge_id]
                # Backward edge
                res_graph[v].append((u, edge_id, 'backward'))
                backward_edges[(v, u, edge_id)] = 0
                
        # Ford-Fulkerson with DFS
        def dfs(u, sink, flow, visited, path_edges):
            if u == sink:
                return flow
                
            visited.add(u)
            
            for v, edge_id, direction in res_graph[u]:
                if v in visited:
                    continue
                    
                if direction == 'forward':
                    available = forward_edges.get((u, v, edge_id), 0)
                else:  # backward
                    available = backward_edges.get((u, v, edge_id), 0)
                    
                if available > 0:
                    bottleneck = min(flow, available)
                    result = dfs(v, sink, bottleneck, visited, path_edges)
                    
                    if result > 0:
                        if direction == 'forward':
                            forward_edges[(u, v, edge_id)] -= result
                            backward_edges[(u, v, edge_id)] += result
                        else:
                            backward_edges[(u, v, edge_id)] -= result
                            forward_edges[(u, v, edge_id)] += result
                            
                        path_edges.append((u, v, edge_id, direction, result))
                        return result
                        
            visited.remove(u)
            return 0
            
        total_flow = 0
        edge_flows = defaultdict(int)
        
        while total_flow < demand:
            visited = set()
            path_edges = []
            flow_increment = dfs(source, sink, demand - total_flow, visited, path_edges)
            
            if flow_increment == 0:
                break
                
            total_flow += flow_increment
            
            # Update edge flows
            for u, v, edge_id, direction, amount in path_edges:
                if direction == 'forward':
                    edge_flows[(u, v, edge_id)] += amount
                else:  # This was a backward edge in residual graph
                    edge_flows[(v, u, edge_id)] -= amount
                    
        return total_flow, edge_flows
    
    # Try to solve the problem
    # We'll use a greedy approach: for each product, find a flow and check feasibility
    
    original_capacity = capacity[:]
    used_capacity = [0] * m
    product_flows = []

    # Check if there's a solution by trying to route each commodity
    success = True

    for prod_idx in range(k):
        s, t, d = products[prod_idx]
        
        # Create a local copy of residual capacities
        res_cap = [original_capacity[i] - temp_used_capacity[i] for i in range(m)]
        
        # Try to find a flow for this product
        # Use a simple path-finding algorithm
        
        def find_simple_path(graph, source, target, res_cap):
            # BFS to find a path
            parent = {}
            queue = deque([source])
            parent[source] = (None, None)  # (parent_node, edge_id)
            
            while queue:
                u = queue.popleft()
                
                if u == target:
                    # Reconstruct path
                    path = []
                    curr = u
                    while parent[curr][0] is not None:
                        prev, edge_id = parent[curr]
                        path.append(edge_id)
                        curr = prev
                    path.reverse()
                    return path
                    
                for v, edge_id in graph[u]:
                    if res_cap[edge_id] > 0 and v not in parent:
                        parent[v] = (u, edge_id)
                        queue.append(v)
                        
            return None
        
        # For simplicity, we'll assume we can decompose the flow into paths
        # Let's find d paths from s to t if possible
        
        flow_on_edges = [0] * m
        remaining_demand = d
        
        while remaining_demand > 0:
            path = find_simple_path(graph, s, t, res_cap)
            if not path:
                success = False
                break
                
            # Send one unit along this path
            for edge_id in path:
                flow_on_edges[edge_id] += 1
                res_cap[edge_id] -= 1
                
            remaining_demand -= 1
            
        if not success:
            break
            
        temp_product_flows.append(flow_on_edges[:])
        
        # Update temporary used capacity
        for i in range(m):
            temp_used_capacity[i] += flow_on_edges[i]
            if temp_used_capacity[i] > original_capacity[i]:
                success = False
                break
                
        if not success:
            break
    
    if not success:
        print("NO")
        return
        
    print("YES")
    for flow in temp_product_flows:
        print(' '.join(map(str, flow)))
