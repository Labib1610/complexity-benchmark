import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    products = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        products.append((s, t, d))
        
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))
        
    # Variables:
    # For each product p in 0..k-1 and edge e in 0..m-1:
    # var_idx(p, e) = p * m + e (total k * m flow variables)
    # Slacks for capacity: m variables
    # Total initial variables C_base = k * m + m
    
    # Base Constraints:
    # 1. Capacity constraints: for each edge e:
    #    sum_{p} x_{p, e} + slack_e = cap_e
    # 2. Conservation constraints: for each product p, for each node u:
    #    sum_{e out of u} x_{p, e} - sum_{e into u} x_{p, e} = b_{p, u}
    
    base_eqs = []
    
    # Capacity constraints
    for e in range(m):
        row = {}
        for p in range(k):
            row[p * m + e] = 1.0
        row[k * m + e] = 1.0  # slack variable
        base_eqs.append((row, float(edges[e][2])))
        
    # Conservation constraints
    for p in range(k):
        s_p, t_p, d_p = products[p]
        for u in range(n):
            row = {}
            for e in range(m):
                u_e, v_e, _ = edges[e]
                if u_e == u:
                    row[p * m + e] = row.get(p * m + e, 0.0) + 1.0
                if v_e == u:
                    row[p * m + e] = row.get(p * m + e, 0.0) - 1.0
            
            b = 0.0
            if u == s_p:
                b += d_p
            if u == t_p:
                b -= d_p
            base_eqs.append((row, b))
            
    num_orig_vars = k * m
    base_num_vars = k * m + m
    
    def solve_lp(extra_constraints):
        # extra_constraints: list of (var_idx, op, val) where op is '<=' or '>='
        all_eqs = []
        for row_dict, b in base_eqs:
            all_eqs.append((dict(row_dict), b))
            
        cur_num_vars = base_num_vars
        for var_idx, op, val in extra_constraints:
            row = {var_idx: 1.0}
            if op == '<=':
                row[cur_num_vars] = 1.0
                cur_num_vars += 1
                all_eqs.append((row, float(val)))
            elif op == '>=':
                row[cur_num_vars] = -1.0
                cur_num_vars += 1
                all_eqs.append((row, float(val)))
                
        R = len(all_eqs)
        C = cur_num_vars
        
        # Build Phase I Tableau
        # Size: (R + 1) x (C + R + 1)
        num_cols = C + R + 1
        num_rows = R + 1
        T = [[0.0] * num_cols for _ in range(num_rows)]
        basis = [0] * R
        
        for i in range(R):
            row_dict, b = all_eqs[i]
            if b < 0:
                for var, coef in row_dict.items():
                    T[i][var] = -coef
                T[i][C + i] = 1.0
                T[i][num_cols - 1] = -b
            else:
                for var, coef in row_dict.items():
                    T[i][var] = coef
                T[i][C + i] = 1.0
                T[i][num_cols - 1] = b
            basis[i] = C + i
            
        # Objective row (row R): - sum of all rows
        for j in range(num_cols):
            s = 0.0
            for i in range(R):
                s += T[i][j]
            T[R][j] = -s
            
        # Simplex loop
        max_iters = 2000 + 10 * (R + C)
        for _ in range(max_iters):
            # Find entering column
            q = -1
            min_val = -1e-8
            for j in range(num_cols - 1):
                if T[R][j] < min_val:
                    min_val = T[R][j]
                    q = j
            if q == -1:
                break
                
            # Find leaving row
            p = -1
            min_ratio = 1e18
            for i in range(R):
                if T[i][q] > 1e-8:
                    ratio = T[i][num_cols - 1] / T[i][q]
                    if ratio < min_ratio - 1e-11:
                        min_ratio = ratio
                        p = i
            if p == -1:
                break
                
            # Pivot at (p, q)
            p_row = T[p]
            inv = 1.0 / p_row[q]
            for j in range(num_cols):
                p_row[j] *= inv
                
            for i in range(num_rows):
                if i != p:
                    factor = T[i][q]
                    if abs(factor) > 1e-11:
                        row_i = T[i]
                        for j in range(num_cols):
                            row_i[j] -= factor * p_row[j]
                            
            basis[p] = q
            
        if -T[R][num_cols - 1] > 1e-5:
            return None  # Infeasible
            
        x = [0.0] * num_orig_vars
        for i in range(R):
            if basis[i] < num_orig_vars:
                x[basis[i]] = T[i][num_cols - 1]
                
        return x

    # Branch and Bound
    def branch_and_bound(extra_constraints):
        sol = solve_lp(extra_constraints)
        if sol is None:
            return None
            
        # Check integrality
        best_frac = 0.0
        best_var = -1
        for i in range(num_orig_vars):
            val = sol[i]
            frac = abs(val - round(val))
            if frac > 1e-4:
                if frac > best_frac:
                    best_frac = frac
                    best_var = i
                    
        if best_var == -1:
            return sol
            
        val = sol[best_var]
        floor_val = int(val)
        ceil_val = floor_val + 1
        
        # Branch order: closer integer first
        if val - floor_val >= 0.5:
            branches = [
                (best_var, '>=', ceil_val),
                (best_var, '<=', floor_val)
            ]
        else:
            branches = [
                (best_var, '<=', floor_val),
                (best_var, '>=', ceil_val)
            ]
            
        for branch in branches:
            extra_constraints.append(branch)
            res = branch_and_bound(extra_constraints)
            extra_constraints.pop()
            if res is not None:
                return res
                
        return None

    res = branch_and_bound([])
    if res is None:
        print("NO")
    else:
        print("YES")
        for p in range(k):
            row_out = []
            for e in range(m):
                val = int(round(res[p * m + e]))
                row_out.append(str(val))
            print(" ".join(row_out))

if __name__ == '__main__':
    solve()
