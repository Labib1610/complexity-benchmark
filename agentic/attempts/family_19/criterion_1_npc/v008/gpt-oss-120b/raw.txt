import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    prod = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        prod.append((s, t, d))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        c = int(next(it))
        edges.append([u, v, c])

    # build adjacency for Dinic
    class Dinic:
        def __init__(self, N):
            self.N = N
            self.adj = [[] for _ in range(N)]

        def add_edge(self, u, v, cap, idx=-1):
            a = [v, cap, None, idx]
            b = [u, 0, a, idx]
            a[2] = b
            self.adj[u].append(a)
            self.adj[v].append(b)

        def bfs(self, s, t, level):
            from collections import deque
            q = deque([s])
            level[:] = [-1]*self.N
            level[s] = 0
            while q:
                u = q.popleft()
                for v, cap, rev, _ in self.adj[u]:
                    if cap and level[v] < 0:
                        level[v] = level[u] + 1
                        q.append(v)
            return level[t] >= 0

        def dfs(self, u, t, f, level, it):
            if u == t:
                return f
            for i in range(it[u], len(self.adj[u])):
                it[u] = i
                v, cap, rev, _ = self.adj[u][i]
                if cap and level[v] == level[u] + 1:
                    ret = self.dfs(v, t, min(f, cap), level, it)
                    if ret:
                        self.adj[u][i][1] -= ret
                        rev[1] += ret
                        return ret
            return 0

        def max_flow(self, s, t):
            flow = 0
            level = [0]*self.N
            INF = 10**18
            while self.bfs(s, t, level):
                it = [0]*self.N
                while True:
                    f = self.dfs(s, t, INF, level, it)
                    if not f:
                        break
                    flow += f
            return flow

    # attempt greedy ordering
    residual = [c for _,_,c in edges]
    result = [[0]*m for _ in range(k)]

    for idx, (s, t, d) in enumerate(prod):
        din = Dinic(n)
        for e_idx, (u, v, _) in enumerate(edges):
            cap = residual[e_idx]
            if cap:
                din.add_edge(u, v, cap, e_idx)
        flow = din.max_flow(s, t)
        if flow < d:
            print("NO")
            return
        # retrieve used capacities from residual graph (edges with flow sent)
        # we run a DFS to push exactly d units and record usage
        used = [0]*m
        def send(u, need):
            if u == t:
                return need
            for v, cap, rev, e_idx in din.adj[u]:
                if e_idx==-1: continue
                used_cap = rev[1]  # flow sent on this edge = reverse capacity
                if used_cap:
                    take = min(need, used_cap)
                    used[e_idx] += take
                    rev[1] -= take
                    need -= take
                    if need==0:
                        break
            return need
        # simple approach: after max flow, the reverse edges hold the flow amount
        for e_idx, (u, v, _) in enumerate(edges):
            for to, cap, rev, idx2 in din.adj[u]:
                if idx2==e_idx:
                    used[e_idx] = rev[1]
        # scale down to d if excess (should not happen)
        # assign
        for e_idx in range(m):
            amt = used[e_idx]
            if amt>0:
                take = min(amt, d)
                result[idx][e_idx] = take
                residual[e_idx] -= take
                d -= take
        if d>0:
            # fallback: just reduce residual arbitrarily
            for e_idx in range(m):
                if residual[e_idx]>=d:
                    result[idx][e_idx] += d
                    residual[e_idx] -= d
                    d = 0
                    break
        if d>0:
            print("NO")
            return

    print("YES")
    out = sys.stdout
    for i in range(k):
        out.write(" ".join(str(x) for x in result[i]) + ("\n" if i+1<k else ""))

if __name__ == "__main__":
    solve()
