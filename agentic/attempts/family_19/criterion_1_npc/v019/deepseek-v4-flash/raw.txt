We can model this as a **multi-commodity flow** problem, but since the number of currencies and edges can be large, we need a more efficient approach.

A key observation:  
Each currency’s flow is independent except for the shared edge capacities.  
We can solve it using **linear programming / min-cost flow** style, but here we can use **max flow with lower bounds** or **feasible flow with multiple commodities** — but that’s NP-hard in general.

However, the constraints are small enough for a **flow decomposition** approach:  
We can treat each currency separately, find a feasible flow for it, and then check if the sum over all currencies respects edge capacities.

We can do this by:
1. For each currency, build a flow network with source = s_i, sink = t_i, demand = d_i.
2. Find any feasible flow (e.g., using max flow).
3. Accumulate flows on edges.
4. Check if any edge’s total flow exceeds its capacity.

If all pass, output the flows.

We’ll implement using **Dinic’s algorithm** for max flow.

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.graph[u].append([v, cap, len(self.graph[v])])
        self.graph[v].append([u, 0, len(self.graph[u]) - 1])
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque()
        q.append(s)
        self.level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, rev in self.graph[u]:
                if cap > 0 and self.level[v] < 0:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] >= 0
    
    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.graph[u])):
            self.it[u] = i
            v, cap, rev = self.graph[u][i]
            if cap > 0 and self.level[u] + 1 == self.level[v]:
                ret = self.dfs(v, t, min(f, cap))
                if ret > 0:
                    self.graph[u][i][1] -= ret
                    self.graph[v][rev][1] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def solve():
    n, m, k = map(int, input().split())
    currencies = []
    for _ in range(k):
        s, t, d = map(int, input().split())
        currencies.append((s-1, t-1, d))
    
    edges = []
    for _ in range(m):
        u, v, cap = map(int, input().split())
        edges.append((u-1, v-1, cap))
    
    # We'll store flow per currency per edge
    flows = [[0]*m for _ in range(k)]
    
    for idx, (s, t, d) in enumerate(currencies):
        # Build flow network for this currency
        dinic = Dinic(n)
        for j, (u, v, cap) in enumerate(edges):
            dinic.add_edge(u, v, cap)  # capacity is just a large number, we'll check later
        # Add super source/sink? Actually we just need max flow from s to t
        flow_val = dinic.max_flow(s, t)
        if flow_val < d:
            print("NO")
            return
        # Now we need to extract the flow on each edge
        # We can do this by checking residual graph
        for j, (u, v, cap) in enumerate(edges):
            # Find the forward edge in dinic graph
            # The edge added was at index j*2 in graph[u]
            # Actually we added edges in order, so we can track
            # But easier: we can store original capacities and subtract residual
            pass
        
        # Instead, we can do a simpler approach: run max flow with capacities = original caps,
        # then extract flow by looking at reverse edges.
        # But we need to know which edge index corresponds to which.
        # Let's rebuild with edge index mapping.
        
        dinic2 = Dinic(n)
        edge_idx = []
        for j, (u, v, cap) in enumerate(edges):
            dinic2.add_edge(u, v, cap)
            edge_idx.append((u, len(dinic2.graph[u])-1))
        
        flow_val = dinic2.max_flow(s, t)
        if flow_val < d:
            print("NO")
            return
        
        # Extract flow on each edge
        for j, (u, rev_idx) in enumerate(edge_idx):
            # The forward edge is at graph[u][rev_idx]
            # The reverse edge is at graph[v][rev]
            v, cap, rev = dinic2.graph[u][rev_idx]
            # flow = original_cap - remaining_cap
            original_cap = edges[j][2]
            used = original_cap - cap
            flows[idx][j] = used
    
    # Now check total capacity constraints
    for j in range(m):
        total = sum(flows[i][j] for i in range(k))
        if total > edges[j][2]:
            print("NO")
            return
    
    print("YES")
    for i in range(k):
        print(" ".join(map(str, flows[i])))

if __name__ == "__main__":
    solve()
```
