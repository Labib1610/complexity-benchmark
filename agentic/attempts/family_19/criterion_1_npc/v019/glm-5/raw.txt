import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    demands = []
    for _ in range(k):
        s = int(next(iterator))
        t = int(next(iterator))
        d = int(next(iterator))
        demands.append((s - 1, t - 1, d))
        
    edges = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        cap = int(next(iterator))
        edges.append((u - 1, v - 1, cap))
        
    # Total nodes in flow graph: 1 super source, 1 super sink, k currency sources, k currency sinks, n institutions
    # Indices:
    # 0: super source
    # 1: super sink
    # 2 to 2+k-1: currency sources
    # 2+k to 2+2k-1: currency sinks
    # 2+2k to 2+2k+n-1: institutions
    S = 0
    T = 1
    cs = 2
    ct = 2 + k
    ni = 2 + 2 * k
    N = 2 + 2 * k + n
    
    graph = [[] for _ in range(N)]
    
    def add_edge(u, v, cap):
        graph[u].append([v, cap, len(graph[v])])
        graph[v].append([u, 0, len(graph[u]) - 1])
        
    # Super source to currency sources
    for i in range(k):
        add_edge(S, cs + i, demands[i][2])
        
    # Currency sinks to super sink
    for i in range(k):
        add_edge(ct + i, T, demands[i][2])
        
    # Currency sources to their origin institutions
    for i in range(k):
        add_edge(cs + i, ni + demands[i][0], demands[i][2])
        
    # Destination institutions to currency sinks
    for i in range(k):
        add_edge(ni + demands[i][1], ct + i, demands[i][2])
        
    # Inter-institution edges
    for u, v, cap in edges:
        add_edge(ni + u, ni + v, cap)
        
    # Dinic's max flow
    level = [0] * N
    ptr = [0] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        queue = [S]
        level[S] = 0
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            for v, cap, _ in graph[u]:
                if level[v] == -1 and cap > 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        return level[T] != -1
        
    def dfs(u, flow):
        if u == T:
            return flow
        while ptr[u] < len(graph[u]):
            edge = graph[u][ptr[u]]
            v, cap, rev = edge[0], edge[1], edge[2]
            if level[v] == level[u] + 1 and cap > 0:
                pushed = dfs(v, min(flow, cap))
                if pushed > 0:
                    graph[u][ptr[u]][1] -= pushed
                    graph[v][rev][1] += pushed
                    return pushed
            ptr[u] += 1
        return 0
        
    total_flow = 0
    while bfs():
        for i in range(N):
            ptr[i] = 0
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            total_flow += pushed
            
    total_demand = sum(d for _, _, d in demands)
    if total_flow < total_demand:
        print("NO")
        return
        
    print("YES")
    ans = [[0] * m for _ in range(k)]
    
    # Reconstruct flows for each currency
    for i in range(k):
        # Find augmenting paths for currency i
        while True:
            for node in range(N):
                level[node] = -1
            queue = [cs + i]
            level[cs + i] = 0
            head = 0
            found = False
            while head < len(queue):
                u = queue[head]
                head += 1
                for edge in graph[u]:
                    v, cap, rev = edge[0], edge[1], edge[2]
                    if level[v] == -1 and cap > 0:
                        level[v] = level[u] + 1
                        if v == ct + i:
                            found = True
                            break
                        queue.append(v)
                if found:
                    break
                    
            if not found:
                break
                
            # DFS to find one path and push 1 unit of flow
            stack = [(cs + i, float('inf'), -1, -1)]
            path = []
            while stack:
                u, flow, pu, pi = stack.pop()
                if u == ct + i:
                    # Augment path
                    for pu, pi in path:
                        graph[pu][pi][1] -= 1
                        v, _, rev = graph[pu][pi]
                        graph[v][rev][1] += 1
                    # Update answer
                    for pu, pi in path:
                        if ni <= pu < ni + n:
                            for j in range(m):
                                if graph[pu][pi] is edges_ref[j]:
                                    ans[i][j] += 1
                                    break
                    break
                    
                found_child = False
                while ptr[u] < len(graph[u]):
                    edge = graph[u][ptr[u]]
                    v, cap, rev = edge[0], edge[1], edge[2]
                    if level[v] == level[u] + 1 and cap > 0:
                        path.append((u, ptr[u]))
                        stack.append((v, min(flow, cap), u, ptr[u]))
                        found_child = True
                        break
                    ptr[u] += 1
                
                if not found_child:
                    level[u] = -1 # Prune dead end
                    if path:
                        path.pop()
            
    # To correctly map edges, we need a reliable way to identify them.
    # Let's rewrite the path finding to just use the edge references.
    # Since we stored edges in `edges_ref`, we can check identity.
    
    # Actually, the above DFS logic is slightly flawed in stack management. Let's use a cleaner recursive DFS for path finding.
    # We will just find paths one by one.
    
    ans = [[0] * m for _ in range(k)]
    # Create a direct reference list for institution edges
    edges_ref = [None] * m
    for j in range(m):
        u, v, cap = edges[j]
        # The forward edge is the first one added for this pair
        edges_ref[j] = graph[ni + u][-2] # Since we added reverse edge right after forward, it's the second to last element
        
    for i in range(k):
        # Find paths from cs+i to ct+i
        def dfs_path(u):
            if u == ct + i:
                return True
            while ptr[u] < len(graph[u]):
                edge = graph[u][ptr[u]]
                v, cap, rev = edge[0], edge[1], edge[2]
                if cap > 0:
                    if dfs_path(v):
                        graph[u][ptr[u]][1] -= 1
                        graph[v][rev][1] += 1
                        # Check if this edge is one of the m institution edges
                        if ni <= u < ni + n:
                            for j in range(m):
                                if graph[u][ptr[u]] is edges_ref[j]:
                                    ans[i][j] += 1
                                    break
                        return True
                ptr[u] += 1
            return False
            
        while True:
            for node in range(N):
                ptr[node] = 0
            if not dfs_path(cs + i):
                break
                
    for i in range(k):
        print(" ".join(map(str, ans[i])))

if __name__ == '__main__':
    solve()
