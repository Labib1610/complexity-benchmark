from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    k = int(data[idx])
    idx += 1
    
    currencies = []
    for i in range(k):
        s = int(data[idx]) - 1  # Convert to 0-indexed
        idx += 1
        t = int(data[idx]) - 1
        idx += 1
        d = int(data[idx])
        idx += 1
        currencies.append((s, t, d))
    
    edges = []
    edge_map = {}
    capacity = []
    for i in range(m):
        u = int(data[idx]) - 1  # Convert to 0-indexed
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        cap = int(data[idx])
        idx += 1
        edges.append((u, v))
        edge_map[(u, v)] = i
        capacity.append(cap)
    
    # Build graph for max flow
    # For each currency, we need to find a flow from s to t with value d
    # We'll solve this as a multi-commodity flow problem with edge capacities shared
    
    # However, since this is a decision problem, we can model it as a single min-cost max-flow
    # But given the constraints, let's try to solve each currency separately and check capacity constraints
    
    # Approach:
    # 1. For each currency, compute a flow from s to t with demand d
    # 2. Sum up flows on each edge across all currencies
    # 3. Check if total flow on any edge exceeds its capacity
    
    # Since we need to route each currency individually, and the edge capacities are shared,
    # we can't just solve each independently. This becomes a multi-commodity flow problem.
    
    # Let's use the following approach:
    # Model this as a minimum cost flow problem where:
    # - Each currency is a commodity
    # - Edge capacities are shared among commodities
    # - We want to route exactly d_i units of commodity i from s_i to t_i
    
    # However, multi-commodity flow is NP-hard in general. 
    # Given small constraints, we might be able to solve it with integer programming or flow methods.
    
    # Let's think differently:
    # For each currency, we need to find a flow. 
    # The total flow on each edge across all currencies must not exceed its capacity.
    
    # This suggests modeling it as a single flow problem with multiple commodities.
    # But since commodities don't interact except through edge capacities, 
    # we can consider this as finding k flows such that their sum on each edge <= capacity.
    
    # A practical approach:
    # Try to find a flow for each currency independently using BFS/DFS,
    # respecting the remaining capacity on edges.
    # This greedy approach may not always work but is worth trying.
    
    # Better approach:
    # Use a flow network where we have a super source connected to each currency's source
    # with capacity d_i, and each currency's sink connected to a super sink with capacity d_i.
    # Edges have capacities as given.
    # This is a multi-source, multi-sink flow problem with shared edge capacities.
    
    # Even better, model it as a circulation problem or use integer programming.
    # But for implementation purposes, let's try a greedy approach first.
    
    # Let's implement a solution based on finding paths for each currency:
    
    from collections import defaultdict
    
    # Create adjacency list for the graph
    graph = [[] for _ in range(n)]
    for i in range(m):
        u, v = edges[i]
        graph[u].append((v, i))  # node, edge_index
    
    # For each currency, we need to send 'd' units from 's' to 't'
    # We'll try to find paths and allocate flow
    
    # Initialize flow on each edge to 0
    flow_on_edges = [0] * m
    result_flows = []  # Will store flow for each currency on each edge
    
    # For each currency, find how to route 'd' units from 's' to 't'
    for s, t, d in currencies:
        # Need to find a way to send 'd' units from s to t
        # We can use a flow algorithm or path finding
        
        # Let's build a residual graph for this specific currency
        # But we need to respect the global edge capacities
        
        # One approach: find 'd' paths from s to t (or use flow decomposition)
        # But we also need to ensure total usage of edges doesn't exceed capacity
        
        # Let's use a greedy approach:
        # Find one path at a time and update residual capacities
        
        local_flow = [0] * m  # Flow for this currency on each edge
        current_d = d
        
        while current_d > 0:
            # Find a path from s to t using BFS
            parent_edge = [-1] * n
            parent_node = [-1] * n
            queue = deque([s])
            visited = [False] * n
            visited[s] = True
            
            found_path = False
            while queue:
                node = queue.popleft()
                if node == t:
                    found_path = True
                    break
                
                for next_node, edge_idx in graph[node]:
                    if not visited[next_node] and (flow_on_edges[edge_idx] + local_flow[edge_idx]) < capacity[edge_idx]:
                        visited[next_node] = True
                        parent_node[next_node] = node
                        parent_edge[next_node] = edge_idx
                        queue.append(next_node)
            
            if not found_path:
                print("NO")
                return
            
            # Reconstruct path and find bottleneck capacity
            path_edges = []
            cur = t
            while cur != s:
                edge_idx = parent_edge[cur]
                path_edges.append(edge_idx)
                cur = parent_node[cur]
            
            # Send 1 unit of flow (since batches are indivisible)
            for edge_idx in path_edges:
                local_flow[edge_idx] += 1
            current_d -= 1
        
        # Add local flow to global flow
        for i in range(m):
            flow_on_edges[i] += local_flow[i]
        
        result_flows.append(local_flow[:])
    
    # Check if any edge exceeds capacity
    for i in range(m):
        if flow_on_edges[i] > capacity[i]:
            print("NO")
            return
    
    print("YES")
    for flow_list in result_flows:
        print(' '.join(map(str, flow_list)))

solve()
