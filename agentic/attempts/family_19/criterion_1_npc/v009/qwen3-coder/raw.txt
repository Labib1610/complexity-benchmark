from collections import defaultdict, deque

def max_flow(graph, source, sink, capacity, n):
    # Ford-Fulkerson with BFS (Edmonds-Karp)
    def bfs():
        parent = [-1] * n
        visited = [False] * n
        queue = deque([source])
        visited[source] = True
        
        while queue:
            u = queue.popleft()
            
            for v in range(n):
                if not visited[v] and capacity.get((u, v), 0) > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    
                    if v == sink:
                        return parent
        return None
    
    max_flow_value = 0
    flows = {}
    
    while True:
        parent = bfs()
        if parent is None:
            break
            
        path_flow = float('inf')
        s = sink
        while s != source:
            prev = parent[s]
            path_flow = min(path_flow, capacity.get((prev, s), 0))
            s = prev
            
        v = sink
        while v != source:
            u = parent[v]
            capacity[(u, v)] = capacity.get((u, v), 0) - path_flow
            capacity[(v, u)] = capacity.get((v, u), 0) + path_flow
            
            if (u, v) not in flows:
                flows[(u, v)] = 0
            if (v, u) not in flows:
                flows[(v, u)] = 0
                
            flows[(u, v)] += path_flow
            flows[(v, u)] -= path_flow
            v = u
            
        max_flow_value += path_flow
    
    return max_flow_value, flows
