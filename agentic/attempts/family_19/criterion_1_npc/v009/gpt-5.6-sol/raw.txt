import sys
import random
from collections import deque

sys.setrecursionlimit(1_000_000)


class Dinic:
    __slots__ = ("n", "g", "level", "it")

    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]
        self.level = [-1] * n
        self.it = [0] * n

    def add_edge(self, u, v, cap):
        ui = len(self.g[u])
        vi = len(self.g[v])
        self.g[u].append([v, cap, vi])
        self.g[v].append([u, 0, ui])
        return ui

    def bfs(self, s, t):
        level = self.level
        for i in range(self.n):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            nl = level[u] + 1
            for v, cap, _ in self.g[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = nl
                    if v == t:
                        return True
                    q.append(v)
        return level[t] >= 0

    def dfs(self, u, t, pushed):
        if u == t:
            return pushed
        g = self.g
        level = self.level
        it = self.it
        while it[u] < len(g[u]):
            idx = it[u]
            v, cap, rev = g[u][idx]
            if cap > 0 and level[v] == level[u] + 1:
                got = self.dfs(v, t, min(pushed, cap))
                if got:
                    g[u][idx][1] -= got
                    g[v][rev][1] += got
                    return got
            it[u] += 1
        return 0

    def max_flow(self, s, t, limit):
        if s == t or limit <= 0:
            return limit
        result = 0
        while result < limit and self.bfs(s, t):
            for i in range(self.n):
                self.it[i] = 0
            while result < limit:
                pushed = self.dfs(s, t, limit - result)
                if not pushed:
                    break
                result += pushed
        return result


def run_flow(n, edges, capacities, s, t, limit, edge_order=None, need_usage=False):
    if limit <= 0 or s == t:
        return limit, ([0] * len(edges) if need_usage else None)

    dinic = Dinic(n)
    refs = []
    order = edge_order if edge_order is not None else range(len(edges))

    for eid in order:
        cap = capacities[eid]
        if cap <= 0:
            continue
        u, v = edges[eid]
        if u == v:
            continue
        idx = dinic.add_edge(u, v, cap)
        if need_usage:
            refs.append((eid, u, idx, cap))

    value = dinic.max_flow(s, t, limit)

    if not need_usage:
        return value, None

    usage = [0] * len(edges)
    for eid, u, idx, original in refs:
        usage[eid] = original - dinic.g[u][idx][1]
    return value, usage


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    commodities = []
    for _ in range(k):
        s = next(it) - 1
        t = next(it) - 1
        d = next(it)
        commodities.append((s, t, d))

    edges = []
    initial_caps = []
    out_edges = [[] for _ in range(n)]
    rev_edges = [[] for _ in range(n)]

    for eid in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        edges.append((u, v))
        initial_caps.append(cap)
        if u != v:
            out_edges[u].append(eid)
            rev_edges[v].append(eid)

    active = [
        i for i, (s, t, d) in enumerate(commodities)
        if d > 0 and s != t
    ]

    base_values = [0] * k
    for i in active:
        s, t, d = commodities[i]
        value, _ = run_flow(n, edges, initial_caps, s, t, d)
        if value < d:
            print("NO")
            return
        base_values[i] = value

    def greedy_attempt(order, shuffled_order=None):
        residual = initial_caps[:]
        answer = [[0] * m for _ in range(k)]

        for cid in order:
            s, t, d = commodities[cid]
            value, usage = run_flow(
                n, edges, residual, s, t, d,
                shuffled_order, True
            )
            if value < d:
                return None
            answer[cid] = usage
            for eid, used in enumerate(usage):
                if used:
                    residual[eid] -= used
        return answer

    orders = []
    if active:
        orders.append(active[:])
        orders.append(list(reversed(active)))
        orders.append(sorted(active, key=lambda x: -commodities[x][2]))
        orders.append(sorted(active, key=lambda x: commodities[x][2]))
        orders.append(sorted(
            active,
            key=lambda x: (
                base_values[x] - commodities[x][2],
                -commodities[x][2]
            )
        ))

    seen_orders = set()
    unique_orders = []
    for order in orders:
        key = tuple(order)
        if key not in seen_orders:
            seen_orders.add(key)
            unique_orders.append(order)

    for order in unique_orders:
        answer = greedy_attempt(order)
        if answer is not None:
            output = ["YES"]
            output.extend(" ".join(map(str, row)) for row in answer)
            sys.stdout.write("\n".join(output))
            return

    rng = random.Random(712367821)
    random_trials = 14 if m * max(1, len(active)) <= 200000 else 5

    for _ in range(random_trials):
        order = active[:]
        rng.shuffle(order)
        edge_order = list(range(m))
        rng.shuffle(edge_order)
        answer = greedy_attempt(order, edge_order)
        if answer is not None:
            output = ["YES"]
            output.extend(" ".join(map(str, row)) for row in answer)
            sys.stdout.write("\n".join(output))
            return

    exact_order = sorted(
        active,
        key=lambda x: (
            base_values[x] - commodities[x][2],
            -commodities[x][2]
        )
    )

    residual = initial_caps[:]
    answer = [[0] * m for _ in range(k)]
    failed = set()

    def target_reachable(t):
        reachable = [False] * n
        reachable[t] = True
        q = deque([t])
        while q:
            v = q.popleft()
            for eid in rev_edges[v]:
                if residual[eid] <= 0:
                    continue
                u, _ = edges[eid]
                if not reachable[u]:
                    reachable[u] = True
                    q.append(u)
        return reachable

    def generate_paths(s, t, reachable, previous):
        visited = [False] * n
        visited[s] = True
        path = []

        def dfs(u):
            if u == t:
                key = tuple(path)
                if previous is None or key > previous:
                    yield key
                return

            for eid in out_edges[u]:
                if residual[eid] <= 0:
                    continue
                _, v = edges[eid]
                if visited[v] or not reachable[v]:
                    continue
                visited[v] = True
                path.append(eid)
                yield from dfs(v)
                path.pop()
                visited[v] = False

        yield from dfs(s)

    def individually_possible(pos, current_remaining):
        end = min(len(exact_order), pos + 6)
        for q in range(pos, end):
            cid = exact_order[q]
            s, t, d = commodities[cid]
            required = current_remaining if q == pos else d
            if required <= 0:
                continue
            value, _ = run_flow(n, edges, residual, s, t, required)
            if value < required:
                return False
        return True

    def solve(pos, remaining, previous):
        if pos == len(exact_order):
            return True

        cid = exact_order[pos]
        s, t, demand = commodities[cid]

        if remaining == 0:
            return solve(pos + 1, commodities[exact_order[pos + 1]][2], None) \
                if pos + 1 < len(exact_order) else True

        state = (pos, remaining, previous, tuple(residual))
        if state in failed:
            return False

        if not individually_possible(pos, remaining):
            if len(failed) < 120000:
                failed.add(state)
            return False

        reachable = target_reachable(t)
        if not reachable[s]:
            if len(failed) < 120000:
                failed.add(state)
            return False

        for path in generate_paths(s, t, reachable, previous):
            max_amount = remaining
            for eid in path:
                if residual[eid] < max_amount:
                    max_amount = residual[eid]

            for amount in range(max_amount, 0, -1):
                for eid in path:
                    residual[eid] -= amount
                    answer[cid][eid] += amount

                if solve(pos, remaining - amount, path):
                    return True

                for eid in path:
                    residual[eid] += amount
                    answer[cid][eid] -= amount

        if len(failed) < 120000:
            failed.add(state)
        return False

    if not exact_order:
        print("YES")
        for _ in range(k):
            print(" ".join(["0"] * m))
        return

    if solve(0, commodities[exact_order[0]][2], None):
        output = ["YES"]
        output.extend(" ".join(map(str, row)) for row in answer)
        sys.stdout.write("\n".join(output))
    else:
        print("NO")


if __name__ == "__main__":
    main()
