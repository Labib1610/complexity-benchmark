import sys
import random
from collections import deque

sys.setrecursionlimit(1_000_000)
input = sys.stdin.buffer.readline

n, m, k = map(int, input().split())
contracts = []
for _ in range(k):
    s, t, d = map(int, input().split())
    contracts.append((s - 1, t - 1, d))

edges = []
for _ in range(m):
    u, v, c = map(int, input().split())
    edges.append((u - 1, v - 1, c))


class Dinic:
    def __init__(self, size):
        self.n = size
        self.g = [[] for _ in range(size)]

    def add_edge(self, u, v, cap, edge_id):
        a = [v, cap, None, edge_id, cap]
        b = [u, 0, a, -1, 0]
        a[2] = b
        self.g[u].append(a)
        self.g[v].append(b)
        return a

    def max_flow(self, s, t, limit):
        if s == t:
            return limit
        total = 0
        n = self.n

        while total < limit:
            level = [-1] * n
            level[s] = 0
            q = deque([s])
            while q:
                u = q.popleft()
                for e in self.g[u]:
                    if e[1] > 0 and level[e[0]] < 0:
                        level[e[0]] = level[u] + 1
                        q.append(e[0])

            if level[t] < 0:
                break

            it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                gu = self.g[u]
                while it[u] < len(gu):
                    e = gu[it[u]]
                    if e[1] > 0 and level[e[0]] == level[u] + 1:
                        got = dfs(e[0], min(pushed, e[1]))
                        if got:
                            e[1] -= got
                            e[2][1] += got
                            return got
                    it[u] += 1
                return 0

            while total < limit:
                pushed = dfs(s, limit - total)
                if not pushed:
                    break
                total += pushed

        return total


def greedy_attempt(order, edge_order):
    remaining = [c for _, _, c in edges]
    answer = [[0] * m for _ in range(k)]

    for commodity in order:
        s, t, demand = contracts[commodity]
        if demand == 0 or s == t:
            continue

        dinic = Dinic(n)
        refs = [None] * m
        for eid in edge_order:
            u, v, _ = edges[eid]
            if remaining[eid] > 0:
                refs[eid] = dinic.add_edge(u, v, remaining[eid], eid)

        if dinic.max_flow(s, t, demand) != demand:
            return None

        for eid, ref in enumerate(refs):
            if ref is not None:
                used = ref[4] - ref[1]
                if used:
                    answer[commodity][eid] = used
                    remaining[eid] -= used

    return answer


def valid_answer(ans):
    if ans is None:
        return False

    loads = [0] * m
    for i, (s, t, demand) in enumerate(contracts):
        balance = [0] * n
        for eid, value in enumerate(ans[i]):
            if value < 0:
                return False
            u, v, _ = edges[eid]
            balance[u] += value
            balance[v] -= value
            loads[eid] += value

        for v in range(n):
            expected = 0
            if v == s:
                expected += demand
            if v == t:
                expected -= demand
            if balance[v] != expected:
                return False

    return all(loads[eid] <= edges[eid][2] for eid in range(m))


# Fast constructive attempts.
base_orders = [
    list(range(k)),
    sorted(range(k), key=lambda i: -contracts[i][2]),
    sorted(range(k), key=lambda i: contracts[i][2]),
]
rng = random.Random(712367821)
for _ in range(8):
    order = list(range(k))
    rng.shuffle(order)
    base_orders.append(order)

for attempt, order in enumerate(base_orders):
    edge_order = list(range(m))
    if attempt >= 3:
        rng.shuffle(edge_order)
    result = greedy_attempt(order, edge_order)
    if valid_answer(result):
        out = ["YES"]
        out.extend(" ".join(map(str, row)) for row in result)
        sys.stdout.write("\n".join(out))
        sys.exit(0)


# Determine which edge variables can belong to a simple source-destination path.
adj = [[] for _ in range(n)]
radj = [[] for _ in range(n)]
for eid, (u, v, _) in enumerate(edges):
    adj[u].append((v, eid))
    radj[v].append((u, eid))

variables = []
commodity_vars = [[] for _ in range(k)]
edge_vars = [[] for _ in range(m)]

for i, (s, t, demand) in enumerate(contracts):
    if demand == 0 or s == t:
        continue

    reachable = [False] * n
    reachable[s] = True
    stack = [s]
    while stack:
        u = stack.pop()
        for v, _ in adj[u]:
            if not reachable[v]:
                reachable[v] = True
                stack.append(v)

    can_reach_t = [False] * n
    can_reach_t[t] = True
    stack = [t]
    while stack:
        v = stack.pop()
        for u, _ in radj[v]:
            if not can_reach_t[u]:
                can_reach_t[u] = True
                stack.append(u)

    if not reachable[t]:
        print("NO")
        sys.exit(0)

    for eid, (u, v, cap) in enumerate(edges):
        if cap > 0 and reachable[u] and can_reach_t[v]:
            idx = len(variables)
            variables.append((i, eid))
            commodity_vars[i].append(idx)
            edge_vars[eid].append(idx)

qvars = len(variables)

if qvars == 0:
    zero_answer = [[0] * m for _ in range(k)]
    if valid_answer(zero_answer):
        out = ["YES"]
        out.extend(" ".join(map(str, row)) for row in zero_answer)
        sys.stdout.write("\n".join(out))
    else:
        print("NO")
    sys.exit(0)


# Sparse base constraints of the form sum(a_j*x_j) <= b.
base_constraints = []

for i, (s, t, demand) in enumerate(contracts):
    if demand == 0 or s == t:
        continue

    node_coeff = [dict() for _ in range(n)]
    for idx in commodity_vars[i]:
        _, eid = variables[idx]
        u, v, _ = edges[eid]
        node_coeff[u][idx] = node_coeff[u].get(idx, 0.0) + 1.0
        node_coeff[v][idx] = node_coeff[v].get(idx, 0.0) - 1.0

    # The sink equation is implied by all other conservation equations.
    for v in range(n):
        if v == t:
            continue
        rhs = float(demand if v == s else 0)
        coeff = node_coeff[v]
        if coeff or rhs != 0:
            base_constraints.append((coeff, rhs))
            base_constraints.append(({j: -a for j, a in coeff.items()}, -rhs))

for eid, indices in enumerate(edge_vars):
    if indices:
        base_constraints.append(({idx: 1.0 for idx in indices}, float(edges[eid][2])))


class LPSolver:
    EPS = 1e-9

    def __init__(self, num_vars, constraints):
        self.m = len(constraints)
        self.n = num_vars
        m, n = self.m, self.n
        self.B = [n + i for i in range(m)]
        self.N = list(range(n)) + [-1]
        self.D = [[0.0] * (n + 2) for _ in range(m + 2)]

        for i, (coeff, rhs) in enumerate(constraints):
            row = self.D[i]
            for j, value in coeff.items():
                row[j] = value
            row[n] = -1.0
            row[n + 1] = rhs

        self.D[m + 1][n] = 1.0

    def pivot(self, r, s):
        D = self.D
        m, n = self.m, self.n
        inv = 1.0 / D[r][s]

        for i in range(m + 2):
            if i == r:
                continue
            dis = D[i][s]
            if abs(dis) <= self.EPS:
                continue
            factor = dis * inv
            for j in range(n + 2):
                if j != s:
                    D[i][j] -= D[r][j] * factor

        for j in range(n + 2):
            if j != s:
                D[r][j] *= inv

        for i in range(m + 2):
            if i != r:
                D[i][s] *= -inv

        D[r][s] = inv
        self.B[r], self.N[s] = self.N[s], self.B[r]

    def simplex(self, phase):
        D = self.D
        m, n = self.m, self.n
        objective_row = m + 1 if phase == 1 else m

        while True:
            s = -1
            for j in range(n + 1):
                if phase == 2 and self.N[j] == -1:
                    continue
                if s == -1 or D[objective_row][j] < D[objective_row][s] - self.EPS or (
                    abs(D[objective_row][j] - D[objective_row][s]) <= self.EPS
                    and self.N[j] < self.N[s]
                ):
                    s = j

            if s == -1 or D[objective_row][s] >= -self.EPS:
                return True

            r = -1
            for i in range(m):
                if D[i][s] <= self.EPS:
                    continue
                if r == -1:
                    r = i
                else:
                    lhs = D[i][n + 1] / D[i][s]
                    rhs = D[r][n + 1] / D[r][s]
                    if lhs < rhs - self.EPS or (
                        abs(lhs - rhs) <= self.EPS and self.B[i] < self.B[r]
                    ):
                        r = i

            if r == -1:
                return False
            self.pivot(r, s)

    def solve(self):
        D = self.D
        m, n = self.m, self.n

        r = min(range(m), key=lambda i: D[i][n + 1]) if m else -1
        if r != -1 and D[r][n + 1] < -self.EPS:
            self.pivot(r, n)
            if not self.simplex(1) or D[m + 1][n + 1] < -self.EPS:
                return None
            if abs(D[m + 1][n + 1]) > 1e-7:
                return None

            for i in range(m):
                if self.B[i] == -1:
                    s = -1
                    for j in range(n + 1):
                        if s == -1 or D[i][j] < D[i][s] - self.EPS or (
                            abs(D[i][j] - D[i][s]) <= self.EPS
                            and self.N[j] < self.N[s]
                        ):
                            s = j
                    if s != -1 and abs(D[i][s]) > self.EPS:
                        self.pivot(i, s)

        if not self.simplex(2):
            return None

        x = [0.0] * n
        for i in range(m):
            if 0 <= self.B[i] < n:
                x[self.B[i]] = D[i][n + 1]
        return x


def build_integer_answer(values):
    ans = [[0] * m for _ in range(k)]
    for idx, value in enumerate(values):
        commodity, eid = variables[idx]
        ans[commodity][eid] = int(round(value))
    return ans


# Branch-and-bound over fractional LP variables.
# Each extra constraint is (variable, sign, rhs):
# sign=1 means x <= rhs, sign=-1 means -x <= rhs.
stack = [()]
while stack:
    branches = stack.pop()
    constraints = list(base_constraints)
    for var, sign, rhs in branches:
        constraints.append(({var: float(sign)}, float(rhs)))

    solution = LPSolver(qvars, constraints).solve()
    if solution is None:
        continue

    candidate = build_integer_answer(solution)
    if valid_answer(candidate):
        out = ["YES"]
        out.extend(" ".join(map(str, row)) for row in candidate)
        sys.stdout.write("\n".join(out))
        sys.exit(0)

    branch_var = -1
    best_fractionality = -1.0
    for idx, value in enumerate(solution):
        nearest = round(value)
        if abs(value - nearest) > 1e-7:
            frac = abs(value - int(value))
            frac = min(frac, 1.0 - frac)
            if frac > best_fractionality:
                best_fractionality = frac
                branch_var = idx

    if branch_var == -1:
        continue

    value = solution[branch_var]
    lower = int(value // 1)
    upper = lower + 1

    # Explore x <= floor(value) first; stack is LIFO.
    stack.append(branches + ((branch_var, -1, -upper),))
    stack.append(branches + ((branch_var, 1, lower),))

print("NO")
