from collections import defaultdict, deque
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    k = int(data[idx]); idx += 1
    
    # Read aid types
    aids = []
    for i in range(k):
        s = int(data[idx]) - 1; idx += 1  # Convert to 0-based
        t = int(data[idx]) - 1; idx += 1
        d = int(data[idx]); idx += 1
        aids.append((s, t, d))
    
    # Read corridors
    corridors = []
    graph = [[] for _ in range(n)]
    for i in range(m):
        u = int(data[idx]) - 1; idx += 1  # Convert to 0-based
        v = int(data[idx]) - 1; idx += 1
        cap = int(data[idx]); idx += 1
        corridors.append((u, v, cap))
        graph[u].append((v, i))
    
    # For each aid type, compute flow using max-flow
    flows = []
    total_flow_on_edge = [0] * m
    
    for s, t, d in aids:
        # Build residual graph for this aid type
        # We'll use Ford-Fulkerson with BFS (Edmonds-Karp)
        res_graph = [[0]*n for _ in range(n)]
        edge_index_map = defaultdict(list)  # Maps (u,v) to list of indices
        
        for i, (u, v, cap) in enumerate(corridors):
            available = cap - total_flow_on_edge[i]
            if available > 0:
                res_graph[u][v] += available
                edge_index_map[(u,v)].append(i)
        
        # Now run max-flow from s to t with demand d
        def bfs():
            parent = [-1] * n
            visited = [False] * n
            queue = deque()
            queue.append(s)
            visited[s] = True
            
            while queue:
                u = queue.popleft()
                for v in range(n):
                    if not visited[v] and res_graph[u][v] > 0:
                        visited[v] = True
                        parent[v] = u
                        queue.append(v)
                        if v == t:
                            return parent
            return None
        
        # Try to send 'd' units of flow
        flow_sent = 0
        path_flows = [0] * m  # Flow for this aid type on each corridor
        
        while flow_sent < d:
            parent = bfs()
            if parent is None:
                print("NO")
                return
            
            # Find minimum residual capacity along the path
            path_flow = d - flow_sent
            s_node = t
            while s_node != s:
                p = parent[s_node]
                path_flow = min(path_flow, res_graph[p][s_node])
                s_node = p
            
            # Update residual capacities and path_flows
            s_node = t
            while s_node != s:
                p = parent[s_node]
                res_graph[p][s_node] -= path_flow
                res_graph[s_node][p] += path_flow
                
                # Update path_flows: find which edge index corresponds to (p, s_node)
                # Since we may have multiple edges between same nodes, we pick one that still has capacity
                key = (p, s_node)
                if key in edge_index_map and edge_index_map[key]:
                    # Pick the first available edge index
                    edge_idx = edge_index_map[key][0]
                    path_flows[edge_idx] += path_flow
                    # If this edge is now saturated, remove it from map
                    if res_graph[p][s_node] == 0:
                        edge_index_map[key].pop(0)
                
                s_node = p
            
            flow_sent += path_flow
        
        flows.append(path_flows)
        # Update total_flow_on_edge
        for i in range(m):
            total_flow_on_edge[i] += path_flows[i]
    
    # If we reach here, all flows are feasible
    print("YES")
    for flow_list in flows:
        print(' '.join(map(str, flow_list)))

if __name__ == "__main__":
    main()
