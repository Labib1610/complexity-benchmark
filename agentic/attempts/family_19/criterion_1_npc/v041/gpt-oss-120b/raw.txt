import sys
from collections import deque

def readints():
    return list(map(int, sys.stdin.readline().split()))

def solve():
    n, m, k = map(int, sys.stdin.readline().split())
    s = []
    t = []
    d = []
    for _ in range(k):
        a, b, c = map(int, sys.stdin.readline().split())
        s.append(a-1)
        t.append(b-1)
        d.append(c)
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u, v, cap = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append([u, v, cap])
        adj[u].append((v, idx))

    total_d = sum(d)

    # Build super source and sink
    S = n
    T = n+1
    N = n+2
    cap = [[0]*N for _ in range(N)]
    for u, v, c in edges:
        cap[u][v] = c
    for i in range(k):
        cap[S][s[i]] += d[i]
        cap[t[i]][T] += d[i]

    # Edmonds-Karp
    flow = [[0]*N for _ in range(N)]
    def bfs():
        parent = [-1]*N
        parent_edge = [-1]*N
        q = deque([S])
        parent[S] = S
        while q:
            u = q.popleft()
            for v in range(N):
                if parent[v]==-1 and cap[u][v]-flow[u][v]>0:
                    parent[v]=u
                    q.append(v)
                    if v==T:
                        break
        if parent[T]==-1:
            return 0, parent
        # find bottleneck
        v=T
        bottleneck=10**18
        while v!=S:
            u=parent[v]
            bottleneck=min(bottleneck, cap[u][v]-flow[u][v])
            v=u
        v=T
        while v!=S:
            u=parent[v]
            flow[u][v]+=bottleneck
            flow[v][u]-=bottleneck
            v=u
        return bottleneck, parent

    maxflow=0
    while True:
        inc, _ = bfs()
        if inc==0:
            break
        maxflow+=inc

    if maxflow!=total_d:
        print("NO")
        return

    # Decompose flow into paths for each commodity
    # First, extract the total flow on original edges
    edge_flow = [0]*m
    for idx,(u,v,_) in enumerate(edges):
        edge_flow[idx]=flow[u][v]

    # Now we need to split this total flow into k commodities.
    # Since capacities are integral and all demands are integral,
    # we can greedily assign each commodity using residual capacities.
    res = [[0]*m for _ in range(k)]
    residual = edge_flow[:]

    for com in range(k):
        need = d[com]
        # run bfs on residual to send need units from s[com] to t[com]
        while need>0:
            parent = [-1]*n
            parent_edge = [-1]*n
            q = deque([s[com]])
            parent[s[com]]=s[com]
            while q:
                u = q.popleft()
                if u==t[com]:
                    break
                for v, idx in adj[u]:
                    if parent[v]==-1 and residual[idx]>0:
                        parent[v]=u
                        parent_edge[v]=idx
                        q.append(v)
            if parent[t[com]]==-1:
                print("NO")
                return
            # find bottleneck
            v=t[com]
            bottleneck=need
            while v!=s[com]:
                idx=parent_edge[v]
                bottleneck=min(bottleneck, residual[idx])
                v=parent[v]
            # augment
            v=t[com]
            while v!=s[com]:
                idx=parent_edge[v]
                residual[idx]-=bottleneck
                res[com][idx]+=bottleneck
                v=parent[v]
            need-=bottleneck

    print("YES")
    for i in range(k):
        print(' '.join(map(str,res[i])))

if __name__=="__main__":
    solve()
