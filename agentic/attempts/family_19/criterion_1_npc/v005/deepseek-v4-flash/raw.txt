import sys
from collections import deque

def max_flow(n, graph, s, t):
    flow = 0
    while True:
        parent = [-1]*n
        parent[s] = s
        q = deque([s])
        while q and parent[t] == -1:
            u = q.popleft()
            for v, cap in graph[u].items():
                if parent[v] == -1 and cap > 0:
                    parent[v] = u
                    q.append(v)
        if parent[t] == -1:
            break
        # find bottleneck
        v = t
        aug = float('inf')
        while v != s:
            u = parent[v]
            aug = min(aug, graph[u][v])
            v = u
        v = t
        while v != s:
            u = parent[v]
            graph[u][v] -= aug
            graph[v][u] = graph.get(v, {}).get(u, 0) + aug
            v = u
        flow += aug
    return flow
