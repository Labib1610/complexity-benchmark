import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    swarms = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        swarms.append((s, t, d))
    
    edges_input = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges_input.append((u, v, cap))
    
    # Build flow network
    # Nodes: 0 is super source, 1..n are waypoints, n+1 is super sink
    # For each swarm i, we need edges from source to s_i with capacity d_i
    # and from t_i to sink with capacity d_i
    # The corridors are shared with total capacity cap
    
    N = n + 2
    source = 0
    sink = n + 1
    
    # We'll use a more complex structure to track flows per swarm
    # First, solve the combined flow problem
    # Then decompose into individual swarm flows
    
    # Build adjacency list for the combined network
    # Each edge needs to track: to, rev, cap, flow, original_cap, edge_id
    # edge_id: -1 for source/sink edges, 0..m-1 for corridor edges
    
    graph = [[] for _ in range(N)]
    edge_list = []  # (from, to, edge_id) for each edge added
    
    def add_edge(fr, to, cap, edge_id):
        # forward edge
        graph[fr].append([to, len(graph[to]), cap, 0, edge_id])
        # backward edge
        graph[to].append([fr, len(graph[fr]) - 1, 0, 0, -1])
        edge_list.append((fr, len(graph[fr]) - 1))
    
    # Add edges from source to each swarm's start
    total_demand = 0
    for i, (s, t, d) in enumerate(swarms):
        add_edge(source, s, d, -1)  # -1 indicates source edge
        add_edge(t, sink, d, -1)   # -1 indicates sink edge
        total_demand += d
    
    # Add corridor edges
    for idx, (u, v, cap) in enumerate(edges_input):
        add_edge(u, v, cap, idx)
    
    # Dinic's algorithm
    def bfs():
        level = [-1] * N
        q = deque([source])
        level[source] = 0
        while q:
            v = q.popleft()
            for e in graph[v]:
                if e[2] > 0 and level[e[0]] < 0:
                    level[e[0]] = level[v] + 1
                    q.append(e[0])
        return level
    
    def dfs(v, t, f, level, it):
        if v == t:
            return f
        for i in range(it[v], len(graph[v])):
            e = graph[v][i]
            if e[2] > 0 and level[v] < level[e[0]]:
                ret = dfs(e[0], t, min(f, e[2]), level, it)
                if ret > 0:
                    e[2] -= ret
                    e[3] += ret
                    graph[e[0]][e[1]][2] += ret
                    graph[e[0]][e[1]][3] -= ret
                    return ret
            it[v] += 1
        return 0
    
    def max_flow():
        flow = 0
        while True:
            level = bfs()
            if level[sink] < 0:
                break
            it = [0] * N
            while True:
                f = dfs(source, sink, float('inf'), level, it)
                if f == 0:
                    break
                flow += f
        return flow
    
    flow = max_flow()
    
    if flow < total_demand:
        print("NO")
        return
    
    print("YES")
    
    # Now decompose the flow into individual swarm flows
    # For each swarm, we need to find paths from s_i to t_i carrying d_i units
    
    # Get the residual graph state
    # For each corridor edge, the flow is stored in e[3] (forward edge)
    
    # Build a working copy of the graph for decomposition
    # We need to track remaining capacity for each swarm
    
    # Result: flow_per_swarm[i][j] = flow of swarm i through corridor j
    result = [[0] * m for _ in range(k)]
    
    # For each swarm, find augmenting paths in the residual network
    for swarm_idx in range(k):
        s, t, d = swarms[swarm_idx]
        remaining = d
        
        while remaining > 0:
            # BFS to find a path from s to t with available capacity
            parent = [(-1, -1)] * N  # (node, edge_index in graph[node])
            q = deque([s])
            parent[s] = (s, -1)
            found = False
            
            while q and not found:
                v = q.popleft()
                for ei, e in enumerate(graph[v]):
                    # We can use this edge if it has remaining capacity
                    # and it's a forward edge (edge_id >= 0 or it's a backward edge we can push back)
                    if e[2] > 0 and parent[e[0]][0] == -1:
                        parent[e[0]] = (v, ei)
                        if e[0] == t:
                            found = True
                            break
                        q.append(e[0])
            
            if not found:
                # This shouldn't happen if max_flow succeeded
                break
            
            # Find the bottleneck capacity along the path
            path_cap = remaining
            v = t
            while v != s:
                pv, ei = parent[v]
                e = graph[pv][ei]
                path_cap = min(path_cap, e[2])
                v = pv
            
            # Update the residual graph and record flows
            v = t
            while v != s:
                pv, ei = parent[v]
                e = graph[pv][ei]
                e[2] -= path_cap
                e[3] += path_cap
                graph[v][e[1]][2] += path_cap
                graph[v][e[1]][3] -= path_cap
                
                # Record the flow if this is a corridor edge
                edge_id = e[4]
                if edge_id >= 0:
                    result[swarm_idx][edge_id] += path_cap
                
                v = pv
            
            remaining -= path_cap
    
    for row in result:
        print(' '.join(map(str, row)))

if __name__ == "__main__":
    solve()
