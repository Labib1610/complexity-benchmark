import sys
from collections import deque

EPS = 1e-8
INF = float("inf")


class LPSolver:
    def __init__(self, A, b, c):
        self.m = len(b)
        self.n = len(c)
        m, n = self.m, self.n

        self.B = [n + i for i in range(m)]
        self.N = list(range(n)) + [-1]
        self.D = [[0.0] * (n + 2) for _ in range(m + 2)]

        for i in range(m):
            self.D[i][:n] = A[i]
            self.D[i][n] = -1.0
            self.D[i][n + 1] = float(b[i])

        for j in range(n):
            self.D[m][j] = -float(c[j])

        self.D[m + 1][n] = 1.0

    def pivot(self, r, s):
        D = self.D
        m, n = self.m, self.n
        inv = 1.0 / D[r][s]
        row_r = D[r]

        for i in range(m + 2):
            if i == r:
                continue
            row_i = D[i]
            factor = row_i[s] * inv
            if abs(factor) <= EPS:
                row_i[s] = -factor
                continue
            for j in range(n + 2):
                if j != s:
                    row_i[j] -= row_r[j] * factor
            row_i[s] = -factor

        for j in range(n + 2):
            if j != s:
                row_r[j] *= inv
        row_r[s] = inv
        self.B[r], self.N[s] = self.N[s], self.B[r]

    def simplex(self, phase):
        m, n = self.m, self.n
        D, B, N = self.D, self.B, self.N
        x = m + 1 if phase == 1 else m

        while True:
            s = -1
            for j in range(n + 1):
                if phase == 2 and N[j] == -1:
                    continue
                if s == -1 or D[x][j] < D[x][s] - EPS or (
                    abs(D[x][j] - D[x][s]) <= EPS and N[j] < N[s]
                ):
                    s = j

            if D[x][s] >= -EPS:
                return True

            r = -1
            for i in range(m):
                if D[i][s] <= EPS:
                    continue
                if r == -1:
                    r = i
                else:
                    lhs = D[i][n + 1] / D[i][s]
                    rhs = D[r][n + 1] / D[r][s]
                    if lhs < rhs - EPS or (
                        abs(lhs - rhs) <= EPS and B[i] < B[r]
                    ):
                        r = i

            if r == -1:
                return False
            self.pivot(r, s)

    def solve(self):
        m, n = self.m, self.n
        D, B, N = self.D, self.B, self.N

        if m:
            r = min(range(m), key=lambda i: D[i][n + 1])
            if D[r][n + 1] < -EPS:
                self.pivot(r, n)
                if not self.simplex(1) or D[m + 1][n + 1] < -EPS:
                    return None
                if abs(D[m + 1][n + 1]) > 1e-7:
                    return None

                for i in range(m):
                    if B[i] == -1:
                        s = -1
                        for j in range(n + 1):
                            if s == -1 or D[i][j] < D[i][s] - EPS or (
                                abs(D[i][j] - D[i][s]) <= EPS and N[j] < N[s]
                            ):
                                s = j
                        if s != -1 and abs(D[i][s]) > EPS:
                            self.pivot(i, s)

        if not self.simplex(2):
            return None

        x = [0.0] * n
        for i in range(m):
            if 0 <= B[i] < n:
                x[B[i]] = D[i][n + 1]
        return x


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    commodities = [(next(it) - 1, next(it) - 1, next(it)) for _ in range(k)]

    edges = []
    adj = [[] for _ in range(n)]
    radj = [[] for _ in range(n)]

    for eid in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        edges.append((u, v, cap))
        if cap > 0:
            adj[u].append((v, eid))
            radj[v].append((u, eid))

    def reachable(start, graph):
        seen = [False] * n
        seen[start] = True
        q = deque([start])
        while q:
            u = q.popleft()
            for v, _ in graph[u]:
                if not seen[v]:
                    seen[v] = True
                    q.append(v)
        return seen

    var_info = []
    var_id = {}
    relevant_nodes = []

    for ci, (s, t, d) in enumerate(commodities):
        if d < 0:
            print("NO")
            return
        if d == 0 or s == t:
            relevant_nodes.append(set())
            continue

        from_s = reachable(s, adj)
        if not from_s[t]:
            print("NO")
            return
        to_t = reachable(t, radj)

        nodes = {v for v in range(n) if from_s[v] and to_t[v]}
        relevant_nodes.append(nodes)

        for eid, (u, v, cap) in enumerate(edges):
            if cap > 0 and from_s[u] and to_t[v]:
                var_id[(ci, eid)] = len(var_info)
                var_info.append((ci, eid))

    p = len(var_info)

    if p == 0:
        if any(d > 0 and s != t for s, t, d in commodities):
            print("NO")
            return
        print("YES")
        zero_line = " ".join(["0"] * m)
        for _ in range(k):
            print(zero_line)
        return

    base_A = []
    base_b = []

    edge_vars = [[] for _ in range(m)]
    commodity_vars = [[] for _ in range(k)]
    for vid, (ci, eid) in enumerate(var_info):
        edge_vars[eid].append(vid)
        commodity_vars[ci].append((eid, vid))

    for eid, (_, _, cap) in enumerate(edges):
        if edge_vars[eid]:
            row = [0.0] * p
            for vid in edge_vars[eid]:
                row[vid] = 1.0
            base_A.append(row)
            base_b.append(float(cap))

    for ci, (s, t, d) in enumerate(commodities):
        if d == 0 or s == t:
            continue

        nodes = relevant_nodes[ci]
        incident = {v: [] for v in nodes if v != t}

        for eid, vid in commodity_vars[ci]:
            u, v, _ = edges[eid]
            if u != t and u in incident:
                incident[u].append((vid, 1.0))
            if v != t and v in incident:
                incident[v].append((vid, -1.0))

        for v, terms in incident.items():
            rhs = float(d if v == s else 0)
            row = [0.0] * p
            for vid, coef in terms:
                row[vid] += coef

            if not terms:
                if abs(rhs) > EPS:
                    print("NO")
                    return
                continue

            base_A.append(row)
            base_b.append(rhs)
            base_A.append([-x for x in row])
            base_b.append(-rhs)

    objective = [-1.0] * p

    def solve_with_bounds(bounds):
        A = list(base_A)
        b = list(base_b)

        for vid, is_upper, value in bounds:
            row = [0.0] * p
            if is_upper:
                row[vid] = 1.0
                A.append(row)
                b.append(float(value))
            else:
                row[vid] = -1.0
                A.append(row)
                b.append(float(-value))

        return LPSolver(A, b, objective).solve()

    def validate(values):
        flows = [[0] * m for _ in range(k)]
        for vid, value in enumerate(values):
            ci, eid = var_info[vid]
            flows[ci][eid] = value

        for eid, (_, _, cap) in enumerate(edges):
            total = sum(flows[ci][eid] for ci in range(k))
            if total < 0 or total > cap:
                return None

        for ci, (s, t, d) in enumerate(commodities):
            balance = [0] * n
            for eid, (u, v, _) in enumerate(edges):
                f = flows[ci][eid]
                if f < 0:
                    return None
                balance[u] += f
                balance[v] -= f

            for v in range(n):
                wanted = d if v == s else (-d if v == t else 0)
                if s == t:
                    wanted = 0
                if balance[v] != wanted:
                    return None

        return flows

    stack = [[]]
    answer = None

    while stack:
        bounds = stack.pop()
        solution = solve_with_bounds(bounds)
        if solution is None:
            continue

        fractional = -1
        best_score = -1.0
        rounded = [0] * p

        for i, x in enumerate(solution):
            r = int(round(x))
            rounded[i] = r
            diff = abs(x - r)
            if diff > 1e-6:
                frac = x - int(x)
                score = min(frac, 1.0 - frac)
                if score > best_score:
                    best_score = score
                    fractional = i

        if fractional == -1:
            valid = validate(rounded)
            if valid is not None:
                answer = valid
                break
            continue

        x = solution[fractional]
        lo = int(x // 1)
        hi = lo + 1

        stack.append(bounds + [(fractional, False, hi)])
        stack.append(bounds + [(fractional, True, lo)])

    if answer is None:
        print("NO")
        return

    out = ["YES"]
    for ci in range(k):
        out.append(" ".join(map(str, answer[ci])))
    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
