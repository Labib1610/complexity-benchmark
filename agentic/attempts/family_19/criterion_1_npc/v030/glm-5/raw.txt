import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    k = int(input_data[idx]); idx += 1

    requests = []
    for _ in range(k):
        s = int(input_data[idx]); idx += 1
        t = int(input_data[idx]); idx += 1
        d = int(input_data[idx]); idx += 1
        requests.append((s, t, d))

    edges = []
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        c = int(input_data[idx]); idx += 1
        edges.append((u, v, c))

    # Build a super-source multi-commodity flow network.
    # Node indexing:
    # 0: super source
    # 1..n: commodity nodes (one per facility)
    # n+1..n+n: transport nodes (one per facility)
    # n+n+1..n+n+m: edge nodes (one per transport connection)
    # n+n+m+1: super sink

    N = 1 + n + n + m + 1
    S = 0
    T = N - 1
    commodity_offset = 1
    transport_offset = 1 + n
    edge_offset = 1 + n + n

    graph = [[] for _ in range(N)]

    def add_edge(u, v, cap):
        graph[u].append([v, cap, len(graph[v])])
        graph[v].append([u, 0, len(graph[u]) - 1])

    # Super source to commodity nodes
    for i in range(k):
        s, t, d = requests[i]
        add_edge(S, commodity_offset + s - 1, d)

    # Commodity nodes to transport nodes (infinite capacity)
    for u in range(1, n + 1):
        add_edge(commodity_offset + u - 1, transport_offset + u - 1, 10**18)

    # Transport nodes to edge nodes (infinite capacity)
    for i in range(m):
        u, v, c = edges[i]
        add_edge(transport_offset + u - 1, edge_offset + i, 10**18)

    # Edge nodes to transport nodes (capacity c)
    for i in range(m):
        u, v, c = edges[i]
        add_edge(edge_offset + i, transport_offset + v - 1, c)

    # Transport nodes to super sink (infinite capacity)
    for i in range(k):
        s, t, d = requests[i]
        add_edge(transport_offset + t - 1, T, d)

    # Dinic's max flow
    level = [0] * N
    it = [0] * N

    def bfs():
        for i in range(N):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            for v, cap, rev in graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[T] >= 0

    def dfs(u, f):
        if u == T:
            return f
        while it[u] < len(graph[u]):
            e = graph[u][it[u]]
            v, cap, rev = e
            if cap > 0 and level[v] == level[u] + 1:
                ret = dfs(v, min(f, cap))
                if ret > 0:
                    e[1] -= ret
                    graph[v][rev][1] += ret
                    return ret
            it[u] += 1
        return 0

    flow = 0
    total_demand = sum(d for _, _, d in requests)
    while bfs():
        it = [0] * N
        while True:
            f = dfs(S, 10**18)
            if f == 0:
                break
            flow += f

    if flow < total_demand:
        print("NO")
        return

    # Extract per-request, per-edge flow
    # For each edge node i, the flow on it comes from transport_offset + u - 1.
    # We need to split this flow among the k requests.
    # The flow from transport_offset + u - 1 to edge_offset + i is the sum over requests.
    # We can trace back the flow from each request by looking at the residual graph.
    
    # Actually, we can compute the flow per request by doing a flow decomposition.
    # But a simpler way: for each request, the flow is determined by the paths from s to t.
    # Since we have a super source, we can just find the flow on each edge for each request.
    
    # Let's use the fact that the flow from S to commodity_offset + s - 1 is d for request i.
    # We can trace the flow for each request through the network.
    
    # For each edge node i, the incoming edges are from transport_offset + u - 1.
    # The flow on this edge is the sum of flows from all requests.
    # We need to split this flow among the requests.
    
    # We can do this by greedily assigning flow to requests.
    # For each edge node i, we have a list of incoming edges from transport nodes.
    # Each transport node has incoming edges from commodity nodes (which correspond to requests).
    
    # Actually, let's just do a simple flow decomposition for each request.
    
    # Build a residual graph for the original flow
    # For each request i, we start at commodity_offset + s_i - 1 and find paths to transport_offset + t_i - 1
    
    # But we need to know how much flow from each request goes through each edge node.
    # The edge node i corresponds to the original edge i.
    
    # Let's think about the structure:
    # S -> commodity(s_i) -> transport(s_i) -> edge(j) -> transport(v_j) -> ... -> transport(t_i) -> T
    
    # The flow on edge(S, commodity(s_i)) is d_i.
    # The flow on edge(commodity(u), transport(u)) is the sum of flows of all requests starting at u.
    # The flow on edge(transport(u), edge(j)) is the flow on edge j from all requests.
    # The flow on edge(edge(j), transport(v)) is the flow on edge j.
    # The flow on edge(transport(t_i), T) is d_i.
    
    # To get per-request flow, we can do a DFS for each request.
    # For request i, we start at commodity(s_i) with flow d_i.
    # We follow edges with positive flow and subtract as we go.
    
    # But we need to be careful: the flow from commodity(u) to transport(u) is shared.
    # Actually, the flow from commodity(s_i) to transport(s_i) is exactly d_i (since it's the only outgoing edge).
    # Then from transport(s_i), the flow goes to various edge nodes.
    # We need to track how much of the flow at each transport node belongs to request i.
    
    # Let's do a flow decomposition for each request.
    # We'll maintain a "remaining flow" for each request at each node.
    
    # Actually, since the graph is a DAG (except for the residual edges), we can just do a DFS.
    
    # Let's build a simpler representation.
    # For each node, we have a list of outgoing edges with their current flow.
    
    # The flow on an edge (u, v) is original_cap - current_cap.
    # But we added edges with infinite capacity, so we need to track flow differently.
    
    # Let's just use the residual graph directly.
    # For request i, we start at commodity_offset + s_i - 1.
    # We need to send d_i units of flow to transport_offset + t_i - 1.
    # We can do this by finding paths in the residual graph.
    
    # But wait, the residual graph might have back-edges, so we need to be careful.
    # Actually, since we're decomposing the flow, we can just follow edges with positive flow.
    
    # Let's create a flow graph: for each edge (u, v) in the original graph, the flow is cap - residual.
    # But for infinite capacity edges, we need to track the flow.
    
    # Let's just use the residual graph and do a DFS for each request.
    
    # For request i:
    #   Start at commodity_offset + s_i - 1 with d_i units to send.
    #   At each node, follow outgoing edges with positive flow.
    #   The flow on edge (u, v) is the original capacity minus the current capacity.
    #   But for infinite capacity edges, we don't know the original capacity.
    
    # Hmm, this is tricky. Let's think differently.
    
    # We know the total flow on each edge node i is the flow on edge(edge_offset + i, transport_offset + v - 1).
    # This flow is the sum of flows from all requests.
    # We need to split this flow among the requests.
    
    # The key insight: the flow from commodity(s_i) to transport(s_i) is d_i.
    # Then the flow from transport(s_i) goes to various edge nodes.
    # The flow from transport(u) to edge(j) is part of the flow on edge j.
    # The flow from edge(j) to transport(v) is the total flow on edge j.
    # Then from transport(v), the flow goes to other edge nodes or to T.
    
    # So for each request, we can trace the flow by doing a DFS from transport(s_i).
    # At each transport node, we know how much flow belongs to this request.
    # We distribute this flow among the outgoing edges to edge nodes.
    
    # Let's implement this.
    
    # First, compute the flow on each edge in the network.
    # flow_on_edge[u][v] = original_cap - current_cap
    
    # But we need to know which edges correspond to which original edges.
    # The edge from transport_offset + u - 1 to edge_offset + i corresponds to the start of edge i.
    # The edge from edge_offset + i to transport_offset + v - 1 corresponds to the end of edge i.
    # The flow on both is the same (the flow on edge i).
    
    # Let's compute the flow on each edge in the network.
    # For each node u, for each edge (u, v, cap, rev) in graph[u]:
    #   The flow on this edge is the original capacity minus the current capacity.
    #   But we don't store the original capacity separately.
    #   However, for edges we added, the original capacity is either d_i, 10^18, c, or d_i.
    
    # Actually, we can compute the flow as follows:
    # For an edge (u, v) with capacity cap and reverse edge (v, u) with capacity rev_cap:
    #   flow = original_cap - cap
    #   But we don't know original_cap.
    
    # However, we know that flow = rev_cap (since the reverse edge starts at 0 and increases by the flow).
    # So the flow on edge (u, v) is graph[v][rev][1] (the capacity of the reverse edge).
    
    # Let's use this to compute flows.
    
    # For each request i, we'll do a DFS from transport_offset + s_i - 1.
    # We'll maintain a "flow to send" at each node.
    # At each transport node, we distribute the flow among outgoing edges to edge nodes.
    # At each edge node, all the flow goes to the outgoing edge to the next transport node.
    
    # Let's implement this.
    
    # flow_per_request[i][j] = flow of request i on edge j
    
    flow_per_request = [[0] * m for _ in range(k)]
    
    # For each request, do a DFS
    for i in range(k):
        s, t, d = requests[i]
        # Start at transport_offset + s - 1 with d units to send
        # We'll use a stack: (node, flow_to_send)
        # But we need to find paths to transport_offset + t - 1
        
        # Actually, let's just do a DFS that finds paths and subtracts flow.
        # We'll use the residual graph.
        
        # The flow on edge (u, v) is graph[v][rev][1]
        # We need to follow edges with positive flow.
        
        # But we need to be careful: we only want to follow edges that are part of the flow for this request.
        # Since we're decomposing the flow, we can just follow any edge with positive flow.
        # But we need to make sure we don't use flow from other requests.
        
        # Hmm, this is getting complicated. Let's think of a simpler approach.
        
        # Actually, since the flow is a valid flow, we can decompose it into paths.
        # Each path goes from S to T.
        # The paths for request i start at S -> commodity(s_i) -> transport(s_i) -> ... -> transport(t_i) -> T.
        # So we can just find paths from transport(s_i) to transport(t_i) in the flow graph.
        
        # Let's build a flow graph: for each edge (u, v) in the original network, the flow is graph[v][rev][1].
        # We'll only consider edges with positive flow.
        
        # For request i, we start at transport_offset + s - 1 with d units.
        # We find a path to transport_offset + t - 1, send as much flow as possible, and repeat.
        
        # But we need to track which edge node each flow goes through.
        # When we go from transport(u) to edge(j), we record that request i uses edge j.
        
        # Let's implement this.
        
        remaining = d
        start = transport_offset + s - 1
        end = transport_offset + t - 1
        
        while remaining > 0:
            # Find a path from start to end in the flow graph
            # BFS or DFS
            parent = [-1] * N
            parent_edge = [-1] * N  # index of edge in graph[u]
            visited = [False] * N
            stack = [start]
            visited[start] = True
            found = False
            while stack:
                u = stack.pop()
                if u == end:
                    found = True
                    break
                for j, e in enumerate(graph[u]):
                    v, cap, rev = e
                    flow_on_edge = graph[v][rev][1]
                    if flow_on_edge > 0 and not visited[v]:
                        visited[v] = True
                        parent[v] = u
                        parent_edge[v] = j
                        stack.append(v)
            
            if not found:
                # Should not happen if flow is valid
                break
            
            # Find the minimum flow on the path
            path = []
            u = end
            while u != start:
                p = parent[u]
                j = parent_edge[u]
                e = graph[p][j]
                v, cap, rev = e
                flow_on_edge = graph[v][rev][1]
                path.append((p, u, j, flow_on_edge))
                u = p
            path.reverse()
            
            min_flow = min(f for _, _, _, f in path)
            send = min(min_flow, remaining)
            
            # Subtract the flow and record edge usage
            for p, u, j, f in path:
                e = graph[p][j]
                v, cap, rev = e
                # Reduce the flow on this edge
                graph[v][rev][1] -= send
                # If this edge is from transport to edge node, record it
                if edge_offset <= u < edge_offset + m:
                    edge_idx = u - edge_offset
                    flow_per_request[i][edge_idx] += send
            
            remaining -= send
    
    print("YES")
    for i in range(k):
        print(' '.join(map(str, flow_per_request[i])))
