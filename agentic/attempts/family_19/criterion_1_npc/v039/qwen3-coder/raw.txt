from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx+1])
    k = int(data[idx+2])
    idx += 3
    
    tasks = []
    for _ in range(k):
        s = int(data[idx]) - 1  # Convert to 0-based indexing
        t = int(data[idx+1]) - 1
        d = int(data[idx+2])
        tasks.append((s, t, d))
        idx += 3
    
    links = []
    capacity = []
    graph = [[] for _ in range(n)]
    for i in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        cap = int(data[idx+2])
        links.append((u, v))
        capacity.append(cap)
        graph[u].append(i)
        idx += 3

    # For each task, we need to find a flow from source to destination
    # We'll model this as a min-cost max-flow problem, but since costs are uniform,
    # we just need max-flow with integer flows.
    # However, we have multiple tasks competing for the same edge capacities.
    # This becomes a multi-commodity flow problem which is NP-hard in general.
    # But since the constraints are small, we can try modeling it as an LP or using
    # integer programming approach via binary search or direct formulation.
    # Given time limit 5s and memory 256MB, let's attempt to solve it with network flow techniques.

    # Let's build a super-source and super-sink to handle all demands
    # But actually, we want to route each task independently within total capacity constraints.
    # This is more like a feasibility problem where we need to assign flows on edges for each task
    # such that:
    # 1. Flow conservation holds for each task at each node
    # 2. Total flow on each edge across all tasks <= capacity[edge]

    # Approach: Use Linear Programming relaxation + rounding might work, but since we need exact integer solutions,
    # and the problem asks for YES/NO + actual routing, let's formulate it properly.

    # Since this is essentially a multi-commodity flow problem with integer requirements,
    # we can model it as an integer linear program:
    # Variables: f[i][j] = amount of flow of task i on link j
    # Constraints:
    # 1. Flow conservation for each task at each node
    # 2. Capacity constraint: sum over all tasks i of f[i][j] <= capacity[j]
    # 3. Demand satisfaction: total outflow from source of task i = demand[i]
    # 4. Non-negativity and integrality

    # Given that k,n,m are small (<100), we can use a specialized solver or iterative approach.
    # But since we're coding from scratch, let's try to implement Ford-Fulkerson style algorithm
    # adapted for multiple commodities.

    # Actually, let's think differently. 
    # Model this as finding paths for each task such that the total usage of each link doesn't exceed capacity.
    # This is equivalent to packing paths under capacity constraints.

    # One way: Try all possible combinations? Infeasible.
    # Another way: Iteratively route tasks one by one, updating residual capacities.
    # But this greedy method may fail even if solution exists.

    # Best approach here: Formulate as Integer Linear Program and solve using simplex or other methods.
    # But implementing ILP solver is complex.

    # Let's consider using a known technique: Min-cost circulation or similar.
    # Or, since we don't care about cost, just feasibility, we can try to find a feasible flow.

    # Let's reduce this to a single commodity max-flow problem with additional constraints.
    # No, because flows of different tasks cannot be combined.

    # Final approach: Use a SAT/LP solver would be ideal, but not available.
    # So, let's implement a backtracking/branch-and-bound approach for small inputs.

    # Given the complexity, let's proceed with a practical heuristic:
    # 1. Build the graph
    # 2. For each task, compute shortest path (or any path) from source to destination
    # 3. Route the demand along that path, subtracting from edge capacities
    # 4. If at any point an edge exceeds capacity, backtrack or declare impossible.

    # But simple greedy won't work due to interdependencies.

    # Correct approach: This is a classic Multi-Commodity Flow problem.
    # We will model it as follows:
    # - For each task i and each edge e, variable x[i,e] represents how many packets of task i go through edge e.
    # - Constraints:
    #   * Flow conservation for each task at each node
    #   * Edge capacity constraints: sum_i x[i,e] <= c_e
    #   * Demand satisfaction: net outflow of task i from source = demand[i]
    #   * Integrality: x[i,e] >= 0 and integer

    # Since we must give an explicit solution, and the problem guarantees integer solutions exist when possible,
    # we can attempt to solve using a pseudo-polynomial method or enumeration.

    # Simplifying assumption: If we can decompose into paths, great. Otherwise, complex dependency.

    # Let's try a recursive backtracking approach with pruning.

    # First, precompute reachability for each task to check if routing is theoretically possible
    def bfs_reachable(start, end):
        visited = [False]*n
        queue = deque([start])
        visited[start] = True
        while queue:
            u = queue.popleft()
            if u == end:
                return True
            for edge_idx in graph[u]:
                v = links[edge_idx][1]
                if not visited[v]:
                    visited[v] = True
                    queue.append(v)
        return False

    for s,t,d in tasks:

            return

    # Number of variables: k*m (each task on each edge)

    # For small cases, we can brute-force or use heuristics.

    # Not guaranteed to work, but often sufficient for contest problems.

    # Initialize result matrix
    result = [[0]*m for _ in range(k)]
    used_capacity = [0]*m

    # Sort tasks by descending demand to place larger tasks first (heuristic)
    task_order = list(range(k))
    task_order.sort(key=lambda i: tasks[i][2], reverse=True)

    # Precompute adjacency list for easier traversal
    adj = [[] for _ in range(n)]
    rev_adj = [[] for _ in range(n)]
    for j in range(m):
        u,v = links[j]
        adj[u].append(j)
        rev_adj[v].append(j)

    # Function to find a path for a task using BFS
    def find_path_bfs(s, t):
        parent_edge = [-1]*n
        visited = [False]*n
        q = deque([s])
        visited[s] = True
        while q:
            u = q.popleft()
            if u == t:
                break
            for edge_idx in adj[u]:
                v = links[edge_idx][1]
                if not visited[v] and used_capacity[edge_idx] + 1 <= capacity[edge_idx]:
                    visited[v] = True
                    parent_edge[v] = edge_idx
                    q.append(v)
        if not visited[t]:
            return None
        path_edges = []
        curr = t
        while parent_edge[curr] != -1:
            e = parent_edge[curr]
            path_edges.append(e)
            curr = links[e][0]
        path_edges.reverse()
        return path_edges

    # Try to route each task
    success = True
    temp_used = used_capacity[:]
    temp_result = [row[:] for row in result]

    for task_id in task_order:
        s, t, d = tasks[task_id]
        # Try to push 'd' units from s to t without exceeding capacities
        # We'll do this by finding augmenting paths 'd' times
        for _ in range(d):
            path = find_path_bfs(s, t)
            if path is None:
                success = False
                break
            for e in path:
                temp_used[e] += 1
                temp_result[task_id][e] += 1
        if not success:
            break

    if success:
        print("YES")
        for i in range(k):
            print(' '.join(map(str, temp_result[i])))
        return

    # Greedy failed, try another approach: min-cost flow adaptation?
    # Or full integer programming simulation?

    # Let's try modeling the complete constraint matrix and solving via Gaussian elimination or similar?
    # Too complex manually.

    # Last resort: Assume that if individual paths exist and total demand < total capacity, then maybe solvable.
    # But this isn't reliable.

    # Re-examining example:
    # Routers: 1,2,3,4
    # Links: 1->2(2), 1->3(2), 2->3(1), 3->4(2), 2->4(2)
    # Tasks: 1->4 (2 pkts), 2->3 (1 pkt)
    # Solution shows routing Task1 on edges 0,3,4 and Task2 on edge 2.
    # That uses 1 unit on edges 0,2,3,4 and 1 unit on edge 1 (via Task1).

    # Let's re-implement with better pathfinding logic.

    # Reset
    result = [[0]*m for _ in range(k)]
    used_capacity = [0]*m

    # Try Edmonds-Karp style approach for each task sequentially
    def bfs_augmenting_path(s, t, res_graph):
        # res_graph[edge_idx] = remaining capacity
        parent_edge = [-1]*n
        visited = [False]*n
        q = deque([s])
        visited[s] = True
        while q:
            u = q.popleft()
            if u == t:
                break
            for edge_idx in adj[u]:
                v = links[edge_idx][1]
                if not visited[v] and res_graph[edge_idx] > 0:
                    visited[v] = True
                    parent_edge[v] = edge_idx
                    q.append(v)
        if not visited[t]:
            return None
        path_edges = []
        curr = t
        while parent_edge[curr] != -1:
            e = parent_edge[curr]
            path_edges.append(e)
            curr = links[e][0]
        path_edges.reverse()
        return path_edges

    def send_flow(task_id, s, t, amount):
        nonlocal used_capacity, result
        res_cap = [capacity[i] - used_capacity[i] for i in range(m)]
        for _ in range(amount):
            path = bfs_augmenting_path(s, t, res_cap)
            if path is None:
                return False
            for e in path:
                res_cap[e] -= 1
                used_capacity[e] += 1
                result[task_id][e] += 1
        return True

    success = True
    for i in range(k):
        s, t, d = tasks[i]
        if not send_flow(i, s, t, d):
            success = False
            break

    if success:
        print("YES")
        for i in range(k):
            print(' '.join(map(str, result[i])))
    else:
        print("NO")

solve()
