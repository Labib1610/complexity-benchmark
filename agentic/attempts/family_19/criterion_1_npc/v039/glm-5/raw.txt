import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    tasks = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        tasks.append((s, t, d))
    
    links = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        links.append((u, v, cap))
    
    # Build flow network
    # Nodes: 0 is super source, 1..n are routers, n+1..n+k are task destinations, n+k+1 is super sink
    # For each task i (0-indexed), its destination node is n+1+i
    # Total nodes: n + k + 2 (0 to n+k+1)
    
    N = n + k + 2
    S = 0
    T = n + k + 1
    
    # Build adjacency list for Dinic
    adj = [[] for _ in range(N)]
    edge_list = []  # To store edges for output
    
    # Edge structure: to, rev, cap, flow, link_id
    # link_id: -1 for virtual edges, 0..m-1 for actual links
    
    def add_edge(u, v, cap, link_id):
        # Forward edge
        adj[u].append([v, len(adj[v]), cap, 0, link_id])
        # Backward edge
        adj[v].append([u, len(adj[u]) - 1, 0, 0, -1])
    
    # Add edges from super source to each task's source
    for i, (s, t, d) in enumerate(tasks):
        add_edge(S, s, d, -1)
    
    # Add edges from each task's destination to super sink
    for i, (s, t, d) in enumerate(tasks):
        task_dest = n + 1 + i
        add_edge(task_dest, T, d, -1)
    
    # Add edges for each router link (can be used by any task)
    for link_id, (u, v, cap) in enumerate(links):
        add_edge(u, v, cap, link_id)
    
    # Add edges from each task's destination router to its task destination node
    for i, (s, t, d) in enumerate(tasks):
        task_dest = n + 1 + i
        add_edge(t, task_dest, d, -1)
    
    # Dinic's algorithm
    level = [0] * N
    ptr = [0] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            for e in adj[u]:
                v, rev, cap, flow, _ = e
                if level[v] == -1 and cap > flow:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[T] != -1
    
    def dfs(u, pushed):
        if pushed == 0 or u == T:
            return pushed
        for i in range(ptr[u], len(adj[u])):
            e = adj[u][i]
            v = e[0]
            if level[v] != level[u] + 1:
                continue
            remaining = e[2] - e[3]
            if remaining <= 0:
                continue
            tr = dfs(v, min(pushed, remaining))
            if tr > 0:
                e[3] += tr
                adj[v][e[1]][3] -= tr
                return tr
            ptr[u] += 1
        return 0
    
    def dinic():
        flow = 0
        while bfs():
            ptr = [0] * N
            while True:
                pushed = dfs(S, float('inf'))
                if pushed == 0:
                    break
                flow += pushed
        return flow
    
    total_demand = sum(d for _, _, d in tasks)
    max_flow = dinic()
    
    if max_flow != total_demand:
        print("NO")
        return
    
    print("YES")
    
    # Extract flow for each task on each link
    # We need to decompose the flow by task
    # For each task, we compute the flow on each link
    
    # Build a residual graph for each task
    # For task i, we need to find paths from s_i to t_i
    
    # Alternative: Use the flow decomposition approach
    # For each task, we compute how much of each link's flow belongs to it
    
    # We'll use a greedy approach: for each task, find augmenting paths in the residual graph
    # that was used for that task's flow
    
    # First, let's build a graph with only the actual links and their flows
    # For each link, we know the total flow through it
    # We need to decompose this flow by task
    
    # Build adjacency for each task's flow decomposition
    # For task i, we create a graph with edges (u, v) having capacity = flow of task i on that edge
    # Initially, we don't know the flow per task, so we need to compute it
    
    # Approach: For each task, compute the max flow from s_i to t_i in a graph where
    # each link has capacity = remaining flow (after previous tasks)
    
    # But this might not give the correct decomposition if tasks share links
    
    # Better approach: Use the flow network we built and decompose by task
    # For each task i, the flow from S to s_i is d_i, and from t_i to task_dest_i is d_i
    # We need to find the flow from s_i to t_i
    
    # Let's use a different approach: for each task, compute the flow in a graph
    # where each link has capacity = min(flow on that link, remaining demand)
    
    # Actually, let's use the following approach:
    # For each task, we compute the flow by finding paths in the residual graph
    
    # Build a graph for each task
    result = [[0] * m for _ in range(k)]
    
    # For each task, compute the flow decomposition
    for task_idx in range(k):
        s_i, t_i, d_i = tasks[task_idx]
        
        # Build a graph for this task
        # Nodes: 1..n (routers)
        # Edges: actual links with capacity = remaining flow for this task
        
        # We need to find how much of each link's flow belongs to this task
        # We'll use a greedy approach: find paths from s_i to t_i
        
        # Build adjacency list for this task
        task_adj = [[] for _ in range(n + 1)]
        edge_info = []  # (u, v, cap, link_id, rev_index)
        
        for link_id, (u, v, cap) in enumerate(links):
            # Find the edge in the original graph
            for e in adj[u]:
                if e[0] == v and e[4] == link_id:
                    flow_on_link = e[3]
                    if flow_on_link > 0:
                        # Add edge with capacity = flow_on_link
                        task_adj[u].append([v, len(edge_info), flow_on_link, link_id, len(task_adj[v])])
                        edge_info.append((u, v, flow_on_link, link_id))
                        task_adj[v].append([u, len(edge_info) - 1, 0, -1, len(task_adj[u]) - 1])
                    break
        
        # Find paths from s_i to t_i
        remaining = d_i
        
        while remaining > 0:
            # BFS to find a path
            parent = [-1] * (n + 1)
            parent_edge = [-1] * (n + 1)
            q = deque([s_i])
            parent[s_i] = s_i
            
            while q and parent[t_i] == -1:
                u = q.popleft()
                for i, e in enumerate(task_adj[u]):
                    v, ei, cap, link_id, _ = e
                    if cap > 0 and parent[v] == -1:
                        parent[v] = u
                        parent_edge[v] = i
                        q.append(v)
            
            if parent[t_i] == -1:
                # No path found, this shouldn't happen if flow is valid
                break
            
            # Find the minimum capacity on the path
            path_cap = float('inf')
            v = t_i
            while v != s_i:
                u = parent[v]
                ei = parent_edge[v]
                e = task_adj[u][ei]
                path_cap = min(path_cap, e[2])
                v = u
            
            # Update the flow
            path_cap = min(path_cap, remaining)
            v = t_i
            while v != s_i:
                u = parent[v]
                ei = parent_edge[v]
                e = task_adj[u][ei]
                e[2] -= path_cap
                # Update reverse edge
                rev_ei = e[4]
                task_adj[v][rev_ei][2] += path_cap
                # Update result if this is an actual link
                if e[3] >= 0:
                    result[task_idx][e[3]] += path_cap
                v = u
            
            remaining -= path_cap
    
    # Output the result
    for row in result:
        print(' '.join(map(str, row)))

solve()
