import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    contracts = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        contracts.append((s, t, d))

    corridors = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        corridors.append((u, v, cap))

    try:
        import numpy as np
        from scipy.optimize import milp, LinearConstraint, Bounds
        from scipy.sparse import dok_matrix

        num_vars = k * m
        c = np.zeros(num_vars)
        integrality = np.ones(num_vars)

        upper_bounds = np.zeros(num_vars)
        for i in range(k):
            for j in range(m):
                upper_bounds[i * m + j] = corridors[j][2]

        bounds = Bounds(lb=0, ub=upper_bounds)

        rows = []
        lhs = []
        rhs = []

        # Capacity constraints: for each corridor j, sum_i f_{i, j} <= cap_j
        for j in range(m):
            row_dict = {}
            for i in range(k):
                row_dict[i * m + j] = 1.0
            rows.append(row_dict)
            lhs.append(-np.inf)
            rhs.append(corridors[j][2])

        # Flow conservation constraints: for each contract i, for each vertex v
        for i in range(k):
            s_i, t_i, d_i = contracts[i]
            for v in range(1, n + 1):
                row_dict = {}
                for j in range(m):
                    u, w, _ = corridors[j]
                    if u == v:
                        row_dict[i * m + j] = row_dict.get(i * m + j, 0.0) + 1.0
                    if w == v:
                        row_dict[i * m + j] = row_dict.get(i * m + j, 0.0) - 1.0
                
                target = 0.0
                if v == s_i:
                    target += d_i
                if v == t_i:
                    target -= d_i
                
                rows.append(row_dict)
                lhs.append(target)
                rhs.append(target)

        num_constraints = len(rows)
        A = dok_matrix((num_constraints, num_vars), dtype=float)
        for r_idx, row_dict in enumerate(rows):
            for c_idx, val in row_dict.items():
                if val != 0:
                    A[r_idx, c_idx] = val

        constraints = LinearConstraint(A.tocsc(), lhs, rhs)
        res = milp(c=c, integrality=integrality, constraints=constraints, bounds=bounds)

        if res.success:
            print("YES")
            x = np.round(res.x).astype(int)
            for i in range(k):
                row_vals = [str(x[i * m + j]) for j in range(m)]
                print(*(row_vals))
        else:
            print("NO")

    except ImportError:
        # Fallback for environments without scipy (e.g. DFS for small instances)
        remaining_cap = [corridors[j][2] for j in range(m)]
        flow_sol = [[0] * m for _ in range(k)]

        adj = [[] for _ in range(n + 1)]
        for j, (u, v, cap) in enumerate(corridors):
            adj[u].append((v, j))

        def fulfill_contract(c_idx):
            if c_idx == k:
                return True
            s, t, req = contracts[c_idx]
            if req == 0:
                return fulfill_contract(c_idx + 1)

            # Find paths from s to t using DFS
            def send_flow(curr_node, current_req, visited_edges):
                if curr_node == t:
                    return True
                for nxt, edge_idx in adj[curr_node]:
                    if remaining_cap[edge_idx] >= current_req:
                        remaining_cap[edge_idx] -= current_req
                        flow_sol[c_idx][edge_idx] += current_req
                        if send_flow(nxt, current_req, visited_edges):
                            return True
                        remaining_cap[edge_idx] += current_req
                        flow_sol[c_idx][edge_idx] -= current_req
                return False

            if send_flow(s, req, set()):
                if fulfill_contract(c_idx + 1):
                    return True
                # Rollback
                for j in range(m):
                    remaining_cap[j] += flow_sol[c_idx][j]
                    flow_sol[c_idx][j] = 0
            return False

        if fulfill_contract(0):
            print("YES")
            for i in range(k):
                print(*(flow_sol[i]))
        else:
            print("NO")

if __name__ == '__main__':
    solve()
