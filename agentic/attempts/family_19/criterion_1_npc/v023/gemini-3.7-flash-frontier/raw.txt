import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    freqs = []
    total_demand = 0
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        freqs.append((s, t, d))
        total_demand += d

    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        c = int(next(it))
        edges.append((u, v, c))

    # Try solving with scipy if available, otherwise custom Branch-and-Bound
    try:
        from scipy.optimize import LinearConstraint, milp
        import numpy as np

        # Variables: x[f * m + e]
        num_vars = k * m
        c_obj = np.zeros(num_vars)

        # Bounds: 0 <= x[f, e] <= edge capacity
        lb = np.zeros(num_vars)
        ub = np.zeros(num_vars)
        for f in range(k):
            for e in range(m):
                ub[f * m + e] = edges[e][2]

        integrality = np.ones(num_vars)

        # Constraints
        # 1. Capacity constraints: sum_f x[f, e] <= c_e
        row_list = []
        lhs = []
        rhs = []

        for e in range(m):
            row = np.zeros(num_vars)
            for f in range(k):
                row[f * m + e] = 1.0
            row_list.append(row)
            lhs.append(0.0)
            rhs.append(float(edges[e][2]))

        # 2. Flow conservation for each f, node
        for f in range(k):
            s_f, t_f, d_f = freqs[f]
            for u in range(n):
                row = np.zeros(num_vars)
                for e in range(m):
                    u_e, v_e, _ = edges[e]
                    if u_e == u:
                        row[f * m + e] += 1.0
                    if v_e == u:
                        row[f * m + e] -= 1.0

                req = 0.0
                if u == s_f:
                    req += d_f
                if u == t_f:
                    req -= d_f

                row_list.append(row)
                lhs.append(req)
                rhs.append(req)

        A = np.array(row_list)
        constraints = LinearConstraint(A, lhs, rhs)

        res = milp(
            c=c_obj,
            integrality=integrality,
            bounds=(lb, ub),
            constraints=constraints,
        )

        if res.success:
            print("YES")
            x = np.round(res.x).astype(int)
            for f in range(k):
                row = [x[f * m + e] for e in range(m)]
                print(*(row))
            return
        else:
            print("NO")
            return

    except ImportError:
        pass

    # Fallback: Path-based DFS for integer flows (works for small total demand or instances)
    # Decompose into individual pulses of magic
    pulses = []
    for f in range(k):
        s, t, d = freqs[f]
        for _ in range(d):
            pulses.append((f, s, t))

    # Graph adjacency
    adj = [[] for _ in range(n)]
    for e_idx, (u, v, c) in enumerate(edges):
        adj[u].append((v, e_idx))

    cap = [e[2] for e in edges]
    pulse_paths = []

    def dfs(pulse_idx):
        if pulse_idx == len(pulses):
            return True
        f, s, t = pulses[pulse_idx]

        # Find simple paths from s to t using available capacity
        # BFS/DFS to find valid paths
        stack = [(s, [], {s})]
        found_paths = []
        while stack:
            curr, path, visited = stack.pop()
            if curr == t:
                found_paths.append(list(path))
                if len(found_paths) >= 10:  # limit branching
                    break
                continue
            for nxt, e_idx in adj[curr]:
                if cap[e_idx] > 0 and nxt not in visited:
                    stack.append((nxt, path + [e_idx], visited | {nxt}))

        for path in found_paths:
            for e_idx in path:
                cap[e_idx] -= 1
            pulse_paths.append((f, path))
            if dfs(pulse_idx + 1):
                return True
            pulse_paths.pop()
            for e_idx in path:
                cap[e_idx] += 1

        return False

    if dfs(0):
        print("YES")
        ans = [[0] * m for _ in range(k)]
        for f, path in pulse_paths:
            for e_idx in path:
                ans[f][e_idx] += 1
        for f in range(k):
            print(*(ans[f]))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
