                            if e.cap > 0 and level2[e.to] < 0:
                                level2[e.to] = level2[u] + 1
                                q2.append(e.to)
                    return level2[t_i] >= 0
                
                def dfs2(u, f):
                    if u == t_i:
                        return f
                    while it2[u] < len(graph[u]):
                        e = graph[u][it2[u]]
                        if e.cap > 0 and level2[u] < level2[e.to]:
                            ret = dfs2(e.to, min(f, e.cap))
                            if ret > 0:
                                e.cap -= ret
                                e.rev.cap += ret
                                return ret
                        it2[u] += 1
                    return 0
                
                flow_i = 0
                while bfs2():
                    while True:
                        f = dfs2(s_i, 10**18)
                        if f == 0:
                            break
                        flow_i += f
                
                if flow_i < d_i:
                    sys.stdout.write("NO\n")
                    return
                
                # Extract flow for this product
                for j in range(m):
                    e1, e2 = edge_list[j]
                    result[i][j] = e2.cap
                    # Reset for next product: remove this product's flow
                    e1.cap += e2.cap
                    e2.cap = 0
            
            break
    
    # Output
    out = ["YES"]
    for i in range(k):
        out.append(" ".join(map(str, result[i])))
    sys.stdout.write("\n".join(out) + "\n")
