import sys
import time
import random

sys.setrecursionlimit(1_000_000)

EPS = 1e-9
START_TIME = time.monotonic()
DEADLINE = START_TIME + 4.85


class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, u, v, cap, idx=-1):
        a = [v, len(self.g[v]), cap, idx]
        b = [u, len(self.g[u]), 0, -1]
        self.g[u].append(a)
        self.g[v].append(b)
        return a

    def max_flow(self, s, t, limit):
        if s == t:
            return limit

        n = self.n
        flow = 0

        while flow < limit:
            level = [-1] * n
            level[s] = 0
            q = [s]
            head = 0

            while head < len(q):
                u = q[head]
                head += 1
                for e in self.g[u]:
                    if e[2] > 0 and level[e[0]] < 0:
                        level[e[0]] = level[u] + 1
                        q.append(e[0])

            if level[t] < 0:
                break

            it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                while it[u] < len(self.g[u]):
                    e = self.g[u][it[u]]
                    if e[2] > 0 and level[e[0]] == level[u] + 1:
                        got = dfs(e[0], min(pushed, e[2]))
                        if got:
                            e[2] -= got
                            self.g[e[0]][e[1]][2] += got
                            return got
                    it[u] += 1
                return 0

            while flow < limit:
                pushed = dfs(s, limit - flow)
                if not pushed:
                    break
                flow += pushed

        return flow


def sequential_attempt(n, m, commodities, edges, order, edge_order):
    residual = [e[2] for e in edges]
    answer = [[0] * m for _ in commodities]

    for ci in order:
        s, t, demand = commodities[ci]
        if demand == 0 or s == t:
            continue

        dinic = Dinic(n)
        refs = [None] * m

        for j in edge_order:
            u, v, _ = edges[j]
            refs[j] = dinic.add_edge(u, v, residual[j], j)

        got = dinic.max_flow(s, t, demand)
        if got != demand:
            return None

        for j in range(m):
            used = residual[j] - refs[j][2]
            if used:
                answer[ci][j] = used
                residual[j] -= used

    return answer


class LPSolver:
    def __init__(self, A, b, c):
        self.m = len(b)
        self.n = len(c)
        m, n = self.m, self.n

        self.B = [n + i for i in range(m)]
        self.N = list(range(n)) + [-1]
        self.D = [[0.0] * (n + 2) for _ in range(m + 2)]

        for i in range(m):
            row = self.D[i]
            ai = A[i]
            for j in range(n):
                row[j] = float(ai[j])
            row[n] = -1.0
            row[n + 1] = float(b[i])

        for j in range(n):
            self.D[m][j] = -float(c[j])
        self.D[m + 1][n] = 1.0

    def pivot(self, r, s):
        D = self.D
        m, n = self.m, self.n
        inv = 1.0 / D[r][s]
        row_r = D[r]

        for i in range(m + 2):
            if i == r:
                continue
            row_i = D[i]
            factor = row_i[s] * inv
            if abs(factor) <= 1e-18:
                row_i[s] *= -inv
                continue
            for j in range(n + 2):
                if j != s:
                    row_i[j] -= row_r[j] * factor
            row_i[s] *= -inv

        for j in range(n + 2):
            if j != s:
                row_r[j] *= inv
        row_r[s] = inv

        self.B[r], self.N[s] = self.N[s], self.B[r]

    def simplex(self, phase):
        D = self.D
        m, n = self.m, self.n
        xrow = m + 1 if phase == 1 else m

        while True:
            s = -1
            for j in range(n + 1):
                if phase == 2 and self.N[j] == -1:
                    continue
                if s == -1 or D[xrow][j] < D[xrow][s] - EPS or (
                    abs(D[xrow][j] - D[xrow][s]) <= EPS
                    and self.N[j] < self.N[s]
                ):
                    s = j

            if s == -1 or D[xrow][s] >= -EPS:
                return True

            r = -1
            for i in range(m):
                if D[i][s] <= EPS:
                    continue
                if r == -1:
                    r = i
                else:
                    lhs = D[i][n + 1] / D[i][s]
                    rhs = D[r][n + 1] / D[r][s]
                    if lhs < rhs - EPS or (
                        abs(lhs - rhs) <= EPS and self.B[i] < self.B[r]
                    ):
                        r = i

            if r == -1:
                return False
            self.pivot(r, s)

    def solve(self):
        m, n = self.m, self.n
        D = self.D

        if m:
            r = min(range(m), key=lambda i: D[i][n + 1])
            if D[r][n + 1] < -EPS:
                self.pivot(r, n)
                if not self.simplex(1) or D[m + 1][n + 1] < -1e-7:
                    return None
                if abs(D[m + 1][n + 1]) > 1e-7:
                    return None

                for i in range(m):
                    if self.B[i] == -1:
                        s = -1
                        for j in range(n + 1):
                            if s == -1 or D[i][j] < D[i][s] - EPS or (
                                abs(D[i][j] - D[i][s]) <= EPS
                                and self.N[j] < self.N[s]
                            ):
                                s = j
                        if s != -1 and abs(D[i][s]) > EPS:
                            self.pivot(i, s)

        if not self.simplex(2):
            return None

        x = [0.0] * n
        for i in range(m):
            if 0 <= self.B[i] < n:
                x[self.B[i]] = D[i][n + 1]
        return x


def verify_solution(n, m, commodities, edges, values):
    k = len(commodities)

    for i in range(k):
        if len(values[i]) != m:
            return False
        for x in values[i]:
            if x < 0:
                return False

    for j, (_, _, cap) in enumerate(edges):
        if sum(values[i][j] for i in range(k)) > cap:
            return False

    for i, (s, t, demand) in enumerate(commodities):
        balance = [0] * n
        for j, (u, v, _) in enumerate(edges):
            x = values[i][j]
            balance[u] += x
            balance[v] -= x

        expected = [0] * n
        expected[s] += demand
        expected[t] -= demand
        if balance != expected:
            return False

    return True


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    commodities = []
    for _ in range(k):
        s = next(it) - 1
        t = next(it) - 1
        d = next(it)
        commodities.append((s, t, d))

    edges = []
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        edges.append((u, v, cap))

    zero_answer = [[0] * m for _ in range(k)]
    if all(d == 0 or s == t for s, t, d in commodities):
        print("YES")
        for row in zero_answer:
            print(*row)
        return

    base_edge_order = list(range(m))
    orders = []

    natural = list(range(k))
    orders.append(natural)
    orders.append(sorted(natural, key=lambda i: -commodities[i][2]))
    orders.append(sorted(natural, key=lambda i: commodities[i][2]))

    rng = random.Random(712367821)
    for _ in range(min(12, max(3, k * 2))):
        order = natural[:]
        rng.shuffle(order)
        orders.append(order)

    seen_orders = set()
    for attempt, order in enumerate(orders):
        if time.monotonic() > START_TIME + 0.9:
            break
        key = tuple(order)
        if key in seen_orders and attempt < 3:
            continue
        seen_orders.add(key)

        edge_order = base_edge_order[:]
        if attempt >= 3:
            rng.shuffle(edge_order)

        ans = sequential_attempt(
            n, m, commodities, edges, order, edge_order
        )
        if ans is not None and verify_solution(
            n, m, commodities, edges, ans
        ):
            print("YES")
            for row in ans:
                print(*row)
            return

    var_count = k * m
    base_A = []
    base_b = []

    # Edge capacity constraints.
    for j, (_, _, cap) in enumerate(edges):
        row = [0.0] * var_count
        for i in range(k):
            row[i * m + j] = 1.0
        base_A.append(row)
        base_b.append(float(cap))

    # Per-commodity flow conservation. One redundant node equation is omitted.
    for i, (s, t, demand) in enumerate(commodities):
        supply = [0] * n
        supply[s] += demand
        supply[t] -= demand

        omit = t if n > 0 else 0
        offset = i * m

        for v in range(n):
            if v == omit:
                continue
            row = [0.0] * var_count
            for j, (u, w, _) in enumerate(edges):
                if u == v:
                    row[offset + j] += 1.0
                if w == v:
                    row[offset + j] -= 1.0

            base_A.append(row)
            base_b.append(float(supply[v]))
            base_A.append([-x for x in row])
            base_b.append(float(-supply[v]))

    objective = [0.0] * var_count
    for i in range(k):
        for j in range(m):
            # Minimize total routed amount with deterministic tie-breaking.
            objective[i * m + j] = -(
                1.0 + ((i * 1000003 + j * 9176 + 37) % 1009) / 1009000.0
            )

    stack = [({}, {})]
    nodes = 0

    while stack and time.monotonic() < DEADLINE:
        lower, upper = stack.pop()
        nodes += 1

        A = list(base_A)
        b = list(base_b)

        infeasible_bounds = False
        for idx, lo in lower.items():
            hi = upper.get(idx)
            if hi is not None and lo > hi:
                infeasible_bounds = True
                break
            row = [0.0] * var_count
            row[idx] = -1.0
            A.append(row)
            b.append(float(-lo))

        if infeasible_bounds:
            continue

        for idx, hi in upper.items():
            row = [0.0] * var_count
            row[idx] = 1.0
            A.append(row)
            b.append(float(hi))

        solution = LPSolver(A, b, objective).solve()
        if solution is None:
            continue

        branch_var = -1
        branch_score = -1.0

        for idx, x in enumerate(solution):
            nearest = round(x)
            frac = abs(x - nearest)
            if frac > 1e-7:
                score = min(x - int(x), int(x) + 1 - x)
                if score > branch_score:
                    branch_score = score
                    branch_var = idx

        if branch_var == -1:
            ints = [max(0, int(round(x))) for x in solution]
            ans = [
                ints[i * m:(i + 1) * m]
                for i in range(k)
            ]
            if verify_solution(n, m, commodities, edges, ans):
                print("YES")
                for row in ans:
                    print(*row)
                return

            # Numerical fallback: force a variable with the largest deviation.
            best = 0.0
            for idx, x in enumerate(solution):
                frac = abs(x - round(x))
                if frac > best:
                    best = frac
                    branch_var = idx
            if branch_var == -1:
                continue

        x = solution[branch_var]
        floor_x = int(x // 1)
        ceil_x = floor_x + 1

        old_lo = lower.get(branch_var, 0)
        old_hi = upper.get(branch_var)

        if old_hi is None or ceil_x <= old_hi:
            lo2 = lower.copy()
            lo2[branch_var] = max(old_lo, ceil_x)
            stack.append((lo2, upper.copy()))

        if floor_x >= old_lo:
            hi2 = upper.copy()
            hi2[branch_var] = (
                floor_x if old_hi is None else min(old_hi, floor_x)
            )
            stack.append((lower.copy(), hi2))

    print("NO")


if __name__ == "__main__":
    main()
