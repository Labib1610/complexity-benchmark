from collections import defaultdict, deque
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    k = int(data[idx])
    idx += 1
    
    orders = []
    for _ in range(k):
        s = int(data[idx]) - 1
        idx += 1
        t = int(data[idx]) - 1
        idx += 1
        d = int(data[idx])
        idx += 1
        orders.append((s, t, d))
    
    edges = []
    edge_list = []
    capacity = []
    graph = [[] for _ in range(n)]
    
    for i in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        cap = int(data[idx])
        idx += 1
        edges.append((u, v))
        edge_list.append((u, v, cap))
        capacity.append(cap)
        graph[u].append(i)
        graph[v].append(i)
    
    # For each order, compute a flow from s to t
    flows = []
    total_flow_on_edge = [0] * m
    
    # Build a combined graph for max flow with all demands
    # We'll solve each order separately and check total capacity
    
    def bfs(s, t, adj, res_graph):
        parent = [-1] * len(adj)
        visited = [False] * len(adj)
        queue = deque()
        queue.append(s)
        visited[s] = True
        
        while queue:
            u = queue.popleft()
            for edge_idx in adj[u]:
                u_edge, v_edge, _ = edge_list[edge_idx]
                if u_edge == u:
                    v = v_edge
                else:
                    continue
                if not visited[v] and res_graph[edge_idx] > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    if v == t:
                        path = []
                        cur = t
                        while parent[cur] != -1:
                            path.append(cur)
                            cur = parent[cur]
                        path.append(s)
                        path.reverse()
                        return path
        return None
    
    def find_flow_path(path, res_graph):
        if not path:
            return [], 0
        flow = float('inf')
        edge_path = []
        for i in range(len(path)-1):
            u = path[i]
            v = path[i+1]
            found = False
            for edge_idx in graph[u]:
                u_edge, v_edge, _ = edge_list[edge_idx]
                if u_edge == u and v_edge == v:
                    if res_graph[edge_idx] > 0:
                        flow = min(flow, res_graph[edge_idx])
                        edge_path.append(edge_idx)
                        found = True
                        break
            if not found:
                return [], 0
        return edge_path, flow
    
    success = True
    individual_flows = []
    
    for order_idx in range(k):
        s, t, demand = orders[order_idx]
        # Create residual graph for this order
        res_graph = [0] * m
        for i in range(m):
            res_graph[i] = capacity[i] - total_flow_on_edge[i]
        
        # Find a flow of 'demand' from s to t
        flow_per_order = [0] * m
        sent = 0
        while sent < demand:
            path = bfs(s, t, graph, res_graph)
            if not path:
                success = False
                break
            edge_path, flow_val = find_flow_path(path, res_graph)
            if not edge_path:
                success = False
                break
            send = min(demand - sent, flow_val)
            if send <= 0:
                success = False
                break
            for edge_idx in edge_path:
                res_graph[edge_idx] -= send
                flow_per_order[edge_idx] += send
            sent += send
        
        if sent < demand:
            success = False
        
        if not success:
            break
            
        # Update total flow
        for i in range(m):
            total_flow_on_edge[i] += flow_per_order[i]
        individual_flows.append(flow_per_order)
    
    if not success:
        print("NO")
        return
    
    # Now we need to make sure that the total flow doesn't exceed capacity
    for i in range(m):
        if total_flow_on_edge[i] > capacity[i]:
            print("NO")
            return
    
    # Check flow conservation for each order
    # We'll recompute flows using a more precise method
    # Let's use a circulation approach or min-cost flow but since it's unweighted,
    # we can model as max-flow for each commodity separately ensuring capacity constraints
    
    # Rebuild solution properly using separate flows and checking integrality
    final_flows = []
    total_usage = [0] * m
    
    # We already have individual_flows which should satisfy:
    # Flow conservation at intermediate nodes
    # Demand satisfaction at source/sink
    # Capacity constraints
    
    for order_idx in range(k):
        flow = individual_flows[order_idx]
        final_flows.append(flow)
        for i in range(m):
            total_usage[i] += flow[i]
    
    # Final capacity check
    for i in range(m):
        if total_usage[i] > capacity[i]:
            print("NO")
            return
    
    print("YES")
    for flow in final_flows:
        print(' '.join(map(str, flow)))

if __name__ == "__main__":
    main()
