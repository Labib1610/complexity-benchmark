We need to model this as a multi-commodity flow problem, but since capacities are shared and we only need feasibility, we can combine all species into a single flow with lower bounds.

Here’s the approach:

- Each species has demand `d_i` from `s_i` to `t_i`.
- We can create a super source connected to each `s_i` with capacity `d_i`, and each `t_i` connected to a super sink with capacity `d_i`.
- Then we run a max flow from super source to super sink on the original graph with given capacities.
- If the max flow equals the sum of all `d_i`, then it’s feasible.
- To reconstruct the per-species flows, we can run separate flows for each species, but with residual capacities after previous species are assigned.

We’ll implement Dinic’s algorithm for max flow.

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.graph[u].append([v, cap, len(self.graph[v])])
        self.graph[v].append([u, 0, len(self.graph[u]) - 1])
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, rev in self.graph[u]:
                if cap > 0 and self.level[v] < 0:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] >= 0
    
    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.graph[u])):
            self.it[u] = i
            v, cap, rev = self.graph[u][i]
            if cap > 0 and self.level[u] + 1 == self.level[v]:
                ret = self.dfs(v, t, min(f, cap))
                if ret > 0:
                    self.graph[u][i][1] -= ret
                    self.graph[v][rev][1] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def main():
    n, m, k = map(int, input().split())
    species = []
    total_demand = 0
    for _ in range(k):
        s, t, d = map(int, input().split())
        s -= 1
        t -= 1
        species.append((s, t, d))
        total_demand += d
    
    edges = []
    for _ in range(m):
        u, v, cap = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v, cap))
    
    # First check feasibility with combined flow
    S = n
    T = n + 1
    dinic = Dinic(n + 2)
    for s, t, d in species:
        dinic.add_edge(S, s, d)
        dinic.add_edge(t, T, d)
    for u, v, cap in edges:
        dinic.add_edge(u, v, cap)
    
    if dinic.max_flow(S, T) != total_demand:
        print("NO")
        return
    
    # Now reconstruct per-species flows
    # We'll run each species separately, using remaining capacities
    # Build a residual graph from original capacities
    # We'll keep track of used flow on each edge
    
    used = [[0] * m for _ in range(k)]
    remaining_cap = [cap for _, _, cap in edges]
    
    for idx, (s, t, d) in enumerate(species):
        # Build flow network for this species only
        dinic2 = Dinic(n + 2)
        S2 = n
        T2 = n + 1
        dinic2.add_edge(S2, s, d)
        dinic2.add_edge(t, T2, d)
        # Add edges with remaining capacity
        for eid, (u, v, cap) in enumerate(edges):
            if remaining_cap[eid] > 0:
                dinic2.add_edge(u, v, remaining_cap[eid])
        flow_val = dinic2.max_flow(S2, T2)
        if flow_val != d:
            print("NO")
            return
        # Extract flow on each edge
        # We need to find the flow on original edges
        # The edges in dinic2 are: first k*2 edges are super source/sink, then m edges
        # Actually we added edges in order: S->s, t->T, then m edges
        # So edge index offset = 2 + eid
        for eid in range(m):
            # The forward edge in dinic2 for original edge eid is at index 2 + eid
            # Its reverse edge is at index 2 + eid + 1? No, we need to track.
            # Better: store edge references
            pass
    
    # Simpler: use flow decomposition by running each species and recording flows
    # We'll do it by building a new graph each time and reading flow values from residual
    # But easier: we can run max flow for each species and record the flow on each edge
    # by checking the reverse edge capacity.
    
    # Let's implement properly:
    used = [[0] * m for _ in range(k)]
    remaining_cap = [cap for _, _, cap in edges]
    
    for idx, (s, t, d) in enumerate(species):
        dinic2 = Dinic(n + 2)
        S2 = n
        T2 = n + 1
        dinic2.add_edge(S2, s, d)
        dinic2.add_edge(t, T2, d)
        edge_indices = []
        for eid, (u, v, cap) in enumerate(edges):
            if remaining_cap[eid] > 0:
                # add edge and remember its index in dinic2.graph[u]
                dinic2.add_edge(u, v, remaining_cap[eid])
                edge_indices.append((u, len(dinic2.graph[u]) - 1, eid))
        flow_val = dinic2.max_flow(S2, T2)
        if flow_val != d:
            print("NO")
            return
        # Extract flow on each original edge
        for u, edge_pos, eid in edge_indices:
            # The forward edge is at graph[u][edge_pos]
            # Its reverse edge is at graph[v][rev]
            v, cap_after, rev = dinic2.graph[u][edge_pos]
            # Flow = original_cap - cap_after
            flow_on_edge = remaining_cap[eid] - cap_after
            used[idx][eid] = flow_on_edge
            remaining_cap[eid] -= flow_on_edge
    
    print("YES")
    for row in used:
        print(" ".join(map(str, row)))

if __name__ == "__main__":
    main()
```
