import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    products = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        products.append((s, t, d))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    for idx in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, idx))
    
    # Build flow network
    # Nodes: 1..n are original hubs
    # Super source = n+1, super sink = n+2
    # For each product i: edge from super source to s_i with capacity d_i
    # For each product i: edge from t_i to super sink with capacity d_i
    # For each lane: edge with capacity cap
    
    N = n + 2
    source = n + 1
    sink = n + 2
    
    # Edge structure: to, capacity, reverse_edge_index, edge_id (for original edges)
    graph = [[] for _ in range(N + 1)]
    edge_counter = 0
    
    def add_edge(fr, to, cap, edge_id=-1):
        nonlocal edge_counter
        graph[fr].append([to, cap, len(graph[to]), edge_id])
        graph[to].append([fr, 0, len(graph[fr]) - 1, -1])
        edge_counter += 1
    
    # Add product edges
    for i, (s, t, d) in enumerate(products):
        add_edge(source, s, d)
        add_edge(t, sink, d)
    
    # Add lane edges
    for idx, (u, v, cap) in enumerate(edges):
        add_edge(u, v, cap, idx)
    
    # Dinic's algorithm
    level = [0] * (N + 1)
    ptr = [0] * (N + 1)
    
    def bfs():
        for i in range(N + 1):
            level[i] = -1
        queue = deque([source])
        level[source] = 0
        while queue:
            v = queue.popleft()
            for to, cap, rev, eid in graph[v]:
                if cap > 0 and level[to] == -1:
                    level[to] = level[v] + 1
                    queue.append(to)
        return level[sink] != -1
    
    def dfs(v, flow):
        if v == sink:
            return flow
        for i in range(ptr[v], len(graph[v])):
            ptr[v] = i
            to, cap, rev, eid = graph[v][i]
            if cap > 0 and level[to] == level[v] + 1:
                pushed = dfs(to, min(flow, cap))
                if pushed > 0:
                    graph[v][i][1] -= pushed
                    graph[to][rev][1] += pushed
                    return pushed
        return 0
    
    total_demand = sum(d for _, _, d in products)
    max_flow = 0
    
    while bfs():
        ptr = [0] * (N + 1)
        while True:
            pushed = dfs(source, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed
    
    if max_flow != total_demand:
        print("NO")
        return
    
    print("YES")
    
    # Extract flow on original edges
    # Original edges are those with edge_id >= 0
    flow_on_edges = [0] * m
    
    # Find which edges in graph correspond to original lanes
    # They were added with edge_id = idx
    for v in range(1, n + 1):
        for to, cap, rev, eid in graph[v]:
            if eid >= 0:
                # This is an original edge, flow = original_cap - current_cap
                original_cap = edges[eid][2]
                current_cap = cap
                flow_on_edges[eid] = original_cap - current_cap
    
    # Now we need to decompose the flow into k products
    # We'll use a greedy approach: for each product, find paths from s_i to t_i
    # and assign flow to them
    
    result = [[0] * m for _ in range(k)]
    
    # Build adjacency list for flow decomposition
    # Each entry: (to, edge_idx, remaining_flow)
    flow_adj = [[] for _ in range(n + 1)]
    for idx, (u, v, _) in enumerate(edges):
        if flow_on_edges[idx] > 0:
            flow_adj[u].append([v, idx, flow_on_edges[idx]])
    
    for prod_idx, (s, t, d) in enumerate(products):
        remaining = d
        while remaining > 0:
            # BFS to find a path from s to t with available flow
            parent = [-1] * (n + 1)
            parent_edge = [-1] * (n + 1)
            queue = deque([s])
            parent[s] = s
            
            while queue and parent[t] == -1:
                u = queue.popleft()
                for i, (v, eidx, flow) in enumerate(flow_adj[u]):
                    if flow > 0 and parent[v] == -1:
                        parent[v] = u
                        parent_edge[v] = eidx
                        queue.append(v)
            
            if parent[t] == -1:
                # Should not happen if flow is valid
                print("NO")
                return
            
            # Find bottleneck
            path_flow = remaining
            v = t
            while v != s:
                u = parent[v]
                eidx = parent_edge[v]
                # Find the flow on this edge
                for to, edge_idx, flow in flow_adj[u]:
                    if edge_idx == eidx and to == v:
                        path_flow = min(path_flow, flow)
                        break
                v = u
            
            # Update flow and result
            v = t
            while v != s:
                u = parent[v]
                eidx = parent_edge[v]
                result[prod_idx][eidx] += path_flow
                # Update flow_adj
                for i, (to, edge_idx, flow) in enumerate(flow_adj[u]):
                    if edge_idx == eidx and to == v:
                        flow_adj[u][i][2] -= path_flow
                        break
                v = u
            
            remaining -= path_flow
    
    for row in result:
        print(' '.join(map(str, row)))

if __name__ == "__main__":
    solve()
