```python
import sys
import itertools
import random
from collections import deque

sys.setrecursionlimit(1_000_000)


class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, u, v, cap):
        forward_index = len(self.g[u])
        reverse_index = len(self.g[v])
        self.g[u].append([v, reverse_index, cap])
        self.g[v].append([u, forward_index, 0])
        return forward_index

    def max_flow(self, s, t, limit):
        if s == t:
            return limit

        total = 0
        n = self.n
        g = self.g

        while total < limit:
            level = [-1] * n
            level[s] = 0
            q = deque([s])

            while q:
                u = q.popleft()
                for v, _, cap in g[u]:
                    if cap > 0 and level[v] < 0:
                        level[v] = level[u] + 1
                        q.append(v)

            if level[t] < 0:
                break

            it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                while it[u] < len(g[u]):
                    idx = it[u]
                    v, rev, cap = g[u][idx]
                    if cap > 0 and level[v] == level[u] + 1:
                        got = dfs(v, min(pushed, cap))
                        if got:
                            g[u][idx][2] -= got
                            g[v][rev][2] += got
                            return got
                    it[u] += 1
                return 0

            while total < limit:
                pushed = dfs(s, limit - total)
                if not pushed:
                    break
                total += pushed

        return total


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    sources = [0] * k
    targets = [0] * k
    demands = [0] * k

    for i in range(k):
        sources[i] = next(it) - 1
        targets[i] = next(it) - 1
        demands[i] = next(it)

    edges = []
    initial_caps = []
    adjacency = [[] for _ in range(n)]
    reverse_adjacency = [[] for _ in range(n)]

    for j in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        edges.append((u, v))
        initial_caps.append(cap)
        if cap > 0:
            adjacency[u].append(j)
            reverse_adjacency[v].append(j)

    for i in range(k):
        if sources[i] == targets[i]:
            demands[i] = 0

    def compute_flow(caps, s, t, required, edge_order=None, extract=False):
        if required == 0:
            return (0, [0] * m) if extract else 0

        dinic = Dinic(n)
        refs = [None] * m

        if edge_order is None:
            edge_order = range(m)

        for eid in edge_order:
            u, v = edges[eid]
            idx = dinic.add_edge(u, v, caps[eid])
            refs[eid] = (u, idx, caps[eid])

        value = dinic.max_flow(s, t, required)

        if not extract:
            return value

        used = [0] * m
        for eid in range(m):
            u, idx, old_cap = refs[eid]
            used[eid] = old_cap - dinic.g[u][idx][2]
        return value, used

    # Immediate individual feasibility check and ordering information.
    individual_slack = [0] * k
    for i in range(k):
        if demands[i] == 0:
            continue
        value = compute_flow(initial_caps, sources[i], targets[i], demands[i])
        if value < demands[i]:
            print("NO")
            return
        individual_slack[i] = value - demands[i]

    active = [i for i in range(k) if demands[i] > 0]
    base_order = sorted(
        active,
        key=lambda x: (individual_slack[x], -demands[x], x)
    )

    def greedy(order, edge_order):
        caps = initial_caps[:]
        answer = [[0] * m for _ in range(k)]

        for commodity in order:
            required = demands[commodity]
            value, used = compute_flow(
                caps,
                sources[commodity],
                targets[commodity],
                required,
                edge_order,
                True
            )
            if value < required:
                return None

            answer[commodity] = used
            for eid, amount in enumerate(used):
                caps[eid] -= amount

        return answer

    orders = []
    seen_orders = set()

    def add_order(order):
        key = tuple(order)
        if key not in seen_orders:
            seen_orders.add(key)
            orders.append(key)

    add_order(base_order)
    add_order(reversed(base_order))
    add_order(sorted(active, key=lambda x: (-demands[x], x)))
    add_order(sorted(active, key=lambda x: (demands[x], x)))

    if len(active) <= 6:
        for p in itertools.permutations(active):
            add_order(p)
    else:
        rng = random.Random(712367821)
        for _ in range(36):
            p = active[:]
            rng.shuffle(p)
            add_order(p)

    original_edge_order = tuple(range(m))
    reversed_edge_order = tuple(range(m - 1, -1, -1))

    for order in orders:
        result = greedy(order, original_edge_order)
        if result is not None:
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in result)
            sys.stdout.write("\n".join(out))
            return

        result = greedy(order, reversed_edge_order)
        if result is not None:
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in result)
            sys.stdout.write("\n".join(out))
            return

    # Exact exponential search. Integral flows can be decomposed into simple
    # source-target paths, so routing path amounts is complete.
    caps = initial_caps[:]
    remaining = demands[:]
    answer = [[0] * m for _ in range(k)]
    failed = set()

    def reachable_distances(target):
        dist = [-1] * n
        dist[target] = 0
        q = deque([target])
        while q:
            v = q.popleft()
            nd = dist[v] + 1
            for eid in reverse_adjacency[v]:
                if caps[eid] <= 0:
                    continue
                u, _ = edges[eid]
                if dist[u] < 0:
                    dist[u] = nd
                    q.append(u)
        return dist

    def path_generator(start, target, dist):
        visited = [False] * n
        path = []
        visited[start] = True

        def dfs(u):
            if u == target:
                yield tuple(path)
                return

            candidates = []
            for eid in adjacency[u]:
                if caps[eid] <= 0:
                    continue
                _, v = edges[eid]
                if not visited[v] and dist[v] >= 0:
                    candidates.append(eid)

            candidates.sort(
                key=lambda eid: (
                    dist[edges[eid][1]],
                    -caps[eid],
                    eid
                )
            )

            for eid in candidates:
                _, v = edges[eid]
                visited[v] = True
                path.append(eid)
                yield from dfs(v)
                path.pop()
                visited[v] = False

        yield from dfs(start)

    def search():
        if all(x == 0 for x in remaining):
            return True

        state = (tuple(remaining), tuple(caps))
        if state in failed:
            return False

        chosen = -1
        chosen_slack = None

        for commodity in active:
            need = remaining[commodity]
            if need == 0:
                continue

            possible = compute_flow(
                caps,
                sources[commodity],
                targets[commodity],
                need
            )
            if possible < need:
                failed.add(state)
                return False

            slack = possible - need
            score = (slack, -need, commodity)
            if chosen_slack is None or score < chosen_slack:
                chosen_slack = score
                chosen = commodity

        if chosen < 0:
            return True

        s = sources[chosen]
        t = targets[chosen]
        dist = reachable_distances(t)

        if dist[s] < 0:
            failed.add(state)
            return False

        for path in path_generator(s, t, dist):
            max_amount = remaining[chosen]
            for eid in path:
                if caps[eid] < max_amount:
                    max_amount = caps[eid]

            if max_amount <= 0:
                continue

            # Larger amounts first; all smaller amounts remain available for
            # completeness.
            for amount in range(max_amount, 0, -1):
                remaining[chosen] -= amount
                for eid in path:
                    caps[eid] -= amount
                    answer[chosen][eid] += amount

                if search():
                    return True

                for eid in path:
                    caps[eid] += amount
                    answer[chosen][eid] -= amount
                remaining[chosen] += amount

        failed.add(state)
        return False

    if search():
        out = ["YES"]
        out.extend(" ".join(map(str, row)) for row in answer)
        sys.stdout.write("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    main()
```
