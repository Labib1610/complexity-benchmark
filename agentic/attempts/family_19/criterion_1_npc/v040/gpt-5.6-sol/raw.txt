import sys
import random
from collections import deque

sys.setrecursionlimit(1_000_000)

EPS = 1e-8
INF = float("inf")


class Edge:
    __slots__ = ("to", "rev", "cap", "initial", "idx")

    def __init__(self, to, rev, cap, idx):
        self.to = to
        self.rev = rev
        self.cap = cap
        self.initial = cap
        self.idx = idx


class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, u, v, cap, idx=-1):
        a = Edge(v, len(self.g[v]), cap, idx)
        b = Edge(u, len(self.g[u]), 0, -1)
        self.g[u].append(a)
        self.g[v].append(b)

    def max_flow(self, s, t, limit):
        if s == t:
            return limit
        flow = 0
        n = self.n

        while flow < limit:
            level = [-1] * n
            level[s] = 0
            q = deque([s])
            while q:
                u = q.popleft()
                for e in self.g[u]:
                    if e.cap > 0 and level[e.to] < 0:
                        level[e.to] = level[u] + 1
                        q.append(e.to)

            if level[t] < 0:
                break

            it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                while it[u] < len(self.g[u]):
                    e = self.g[u][it[u]]
                    if e.cap > 0 and level[e.to] == level[u] + 1:
                        got = dfs(e.to, min(pushed, e.cap))
                        if got:
                            e.cap -= got
                            self.g[e.to][e.rev].cap += got
                            return got
                    it[u] += 1
                return 0

            while flow < limit:
                pushed = dfs(s, limit - flow)
                if not pushed:
                    break
                flow += pushed

        return flow


def route_one(n, edges, residual, s, t, demand):
    din = Dinic(n)
    refs = []
    for j, (u, v, _) in enumerate(edges):
        if residual[j] > 0 and u != v:
            pos = len(din.g[u])
            din.add_edge(u, v, residual[j], j)
            refs.append((u, pos, j))

    got = din.max_flow(s, t, demand)
    if got != demand:
        return None

    used = [0] * len(edges)
    for u, pos, j in refs:
        e = din.g[u][pos]
        f = e.initial - e.cap
        if f:
            used[j] += f
    return used


def sequential_attempt(n, edges, capacities, groups, order):
    residual = capacities[:]
    answer = [[0] * len(edges) for _ in groups]

    for gi in order:
        s, t, demand, _ = groups[gi]
        used = route_one(n, edges, residual, s, t, demand)
        if used is None:
            return None
        answer[gi] = used
        for j, x in enumerate(used):
            residual[j] -= x

    return answer


class LPSolver:
    def __init__(self, A, b, c):
        self.m = len(b)
        self.n = len(c)
        m, n = self.m, self.n

        self.B = [n + i for i in range(m)]
        self.N = list(range(n)) + [-1]
        self.D = [[0.0] * (n + 2) for _ in range(m + 2)]

        for i in range(m):
            row = self.D[i]
            ai = A[i]
            for j in range(n):
                row[j] = float(ai[j])
            row[n] = -1.0
            row[n + 1] = float(b[i])

        for j in range(n):
            self.D[m][j] = -float(c[j])
        self.D[m + 1][n] = 1.0

    def pivot(self, r, s):
        D = self.D
        m, n = self.m, self.n
        inv = 1.0 / D[r][s]

        for i in range(m + 2):
            if i == r:
                continue
            dis = D[i][s]
            if abs(dis) <= EPS:
                continue
            factor = dis * inv
            row_i = D[i]
            row_r = D[r]
            for j in range(n + 2):
                if j != s:
                    row_i[j] -= row_r[j] * factor
            row_i[s] = -dis * inv

        row_r = D[r]
        for j in range(n + 2):
            if j != s:
                row_r[j] *= inv
        row_r[s] = inv
        self.B[r], self.N[s] = self.N[s], self.B[r]

    def simplex(self, phase):
        D = self.D
        m, n = self.m, self.n
        objective_row = m + 1 if phase == 1 else m

        while True:
            s = -1
            for j in range(n + 1):
                if phase == 2 and self.N[j] == -1:
                    continue
                if s == -1 or D[objective_row][j] < D[objective_row][s] - EPS or (
                    abs(D[objective_row][j] - D[objective_row][s]) <= EPS
                    and self.N[j] < self.N[s]
                ):
                    s = j

            if D[objective_row][s] >= -EPS:
                return True

            r = -1
            for i in range(m):
                if D[i][s] <= EPS:
                    continue
                if r == -1:
                    r = i
                else:
                    lhs = D[i][n + 1] / D[i][s]
                    rhs = D[r][n + 1] / D[r][s]
                    if lhs < rhs - EPS or (
                        abs(lhs - rhs) <= EPS and self.B[i] < self.B[r]
                    ):
                        r = i

            if r == -1:
                return False
            self.pivot(r, s)

    def solve(self):
        D = self.D
        m, n = self.m, self.n

        if m:
            r = min(range(m), key=lambda i: D[i][n + 1])
            if D[r][n + 1] < -EPS:
                self.pivot(r, n)
                if not self.simplex(1) or D[m + 1][n + 1] < -EPS:
                    return None
                if abs(D[m + 1][n + 1]) > 1e-7:
                    return None

                for i in range(m):
                    if self.B[i] == -1:
                        s = -1
                        for j in range(n + 1):
                            if s == -1 or D[i][j] < D[i][s] - EPS or (
                                abs(D[i][j] - D[i][s]) <= EPS
                                and self.N[j] < self.N[s]
                            ):
                                s = j
                        if abs(D[i][s]) > EPS:
                            self.pivot(i, s)
                        break

        if not self.simplex(2):
            return None

        x = [0.0] * n
        for i in range(m):
            if 0 <= self.B[i] < n:
                x[self.B[i]] = D[i][n + 1]
        return x


def solve_ilp(base_A, base_b, objective, upper_initial):
    variables = len(objective)
    stack = [({}, {})]

    while stack:
        lower, upper = stack.pop()

        A = base_A[:]
        b = base_b[:]

        feasible_bounds = True
        for v, lo in lower.items():
            if lo > upper.get(v, upper_initial[v]):
                feasible_bounds = False
                break
            row = [0.0] * variables
            row[v] = -1.0
            A.append(row)
            b.append(-float(lo))

        if not feasible_bounds:
            continue

        for v, hi in upper.items():
            if hi < lower.get(v, 0):
                feasible_bounds = False
                break
            row = [0.0] * variables
            row[v] = 1.0
            A.append(row)
            b.append(float(hi))

        if not feasible_bounds:
            continue

        solution = LPSolver(A, b, objective).solve()
        if solution is None:
            continue

        branch_var = -1
        best_fractionality = 0.0
        for i, value in enumerate(solution):
            nearest = round(value)
            frac = abs(value - nearest)
            if frac > 1e-6 and frac > best_fractionality:
                best_fractionality = frac
                branch_var = i

        if branch_var == -1:
            rounded = [max(0, int(round(x))) for x in solution]
            return rounded

        value = solution[branch_var]
        floor_value = int(value // 1)
        ceil_value = floor_value + 1

        current_lo = lower.get(branch_var, 0)
        current_hi = upper.get(branch_var, upper_initial[branch_var])

        if ceil_value <= current_hi:
            new_lower = lower.copy()
            new_lower[branch_var] = max(current_lo, ceil_value)
            stack.append((new_lower, upper.copy()))

        if floor_value >= current_lo:
            new_upper = upper.copy()
            new_upper[branch_var] = min(current_hi, floor_value)
            stack.append((lower.copy(), new_upper))

    return None


def validate_group_flow(n, edges, capacities, groups, flow):
    m = len(edges)
    loads = [0] * m

    for gi, (s, t, demand, _) in enumerate(groups):
        balance = [0] * n
        for j, x in enumerate(flow[gi]):
            if x < 0:
                return False
            loads[j] += x
            u, v, _ = edges[j]
            balance[u] += x
            balance[v] -= x

        for v in range(n):
            wanted = demand if v == s else (-demand if v == t else 0)
            if balance[v] != wanted:
                return False

    return all(loads[j] <= capacities[j] for j in range(m))


def split_groups(n, edges, groups, group_flow, k):
    m = len(edges)
    result = [[0] * m for _ in range(k)]
    outgoing = [[] for _ in range(n)]
    for j, (u, v, _) in enumerate(edges):
        if u != v:
            outgoing[u].append((v, j))

    for gi, (s, t, _, members) in enumerate(groups):
        remaining_flow = group_flow[gi][:]
        member_pos = 0
        member_left = members[0][1] if members else 0

        while member_pos < len(members):
            while member_left == 0:
                member_pos += 1
                if member_pos == len(members):
                    break
                member_left = members[member_pos][1]
            if member_pos == len(members):
                break

            parent_node = [-1] * n
            parent_edge = [-1] * n
            parent_node[s] = s
            q = deque([s])

            while q and parent_node[t] == -1:
                u = q.popleft()
                for v, j in outgoing[u]:
                    if remaining_flow[j] > 0 and parent_node[v] == -1:
                        parent_node[v] = u
                        parent_edge[v] = j
                        q.append(v)
                        if v == t:
                            break

            if parent_node[t] == -1:
                return None

            path = []
            v = t
            bottleneck = member_left
            while v != s:
                j = parent_edge[v]
                path.append(j)
                bottleneck = min(bottleneck, remaining_flow[j])
                v = parent_node[v]

            order_idx = members[member_pos][0]
            for j in path:
                result[order_idx][j] += bottleneck
                remaining_flow[j] -= bottleneck
            member_left -= bottleneck

    return result


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    orders = []
    for _ in range(k):
        s = next(it) - 1
        t = next(it) - 1
        d = next(it)
        orders.append((s, t, d))

    edges = []
    capacities = []
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        edges.append((u, v, cap))
        capacities.append(cap)

    grouped = {}
    trivial = []
    for i, (s, t, d) in enumerate(orders):
        if d == 0 or s == t:
            trivial.append(i)
            continue
        grouped.setdefault((s, t), []).append((i, d))

    groups = []
    for (s, t), members in grouped.items():
        groups.append((s, t, sum(d for _, d in members), members))

    if not groups:
        print("YES")
        zero = " ".join(["0"] * m)
        for _ in range(k):
            print(zero)
        return

    gcount = len(groups)

    orders_to_try = []
    original_order = list(range(gcount))
    orders_to_try.append(original_order)
    orders_to_try.append(sorted(original_order, key=lambda x: -groups[x][2]))
    orders_to_try.append(sorted(original_order, key=lambda x: groups[x][2]))

    rng = random.Random(712367821)
    for _ in range(min(12, max(0, gcount * 2))):
        p = original_order[:]
        rng.shuffle(p)
        orders_to_try.append(p)

    group_flow = None
    seen_orders = set()
    for order in orders_to_try:
        key = tuple(order)
        if key in seen_orders:
            continue
        seen_orders.add(key)
        candidate = sequential_attempt(n, edges, capacities, groups, order)
        if candidate is not None:
            group_flow = candidate
            break

    active_edges = [
        j for j, (u, v, cap) in enumerate(edges)
        if cap > 0 and u != v
    ]

    if group_flow is None:
        ecount = len(active_edges)
        variables = gcount * ecount

        if variables == 0:
            print("NO")
            return

        base_A = []
        base_b = []

        # Shared edge capacities.
        for ep, j in enumerate(active_edges):
            row = [0.0] * variables
            for gi in range(gcount):
                row[gi * ecount + ep] = 1.0
            base_A.append(row)
            base_b.append(float(capacities[j]))

        # Flow conservation, with one redundant vertex omitted per group.
        for gi, (s, t, demand, _) in enumerate(groups):
            for v in range(n):
                if v == t:
                    continue

                row = [0.0] * variables
                offset = gi * ecount
                for ep, j in enumerate(active_edges):
                    u, w, _ = edges[j]
                    if u == v:
                        row[offset + ep] += 1.0
                    if w == v:
                        row[offset + ep] -= 1.0

                rhs = float(demand if v == s else 0)
                base_A.append(row)
                base_b.append(rhs)
                base_A.append([-x for x in row])
                base_b.append(-rhs)

        objective = [0.0] * variables
        upper_initial = []
        for _ in range(gcount):
            for j in active_edges:
                upper_initial.append(capacities[j])

        integer_solution = solve_ilp(
            base_A, base_b, objective, upper_initial
        )

        if integer_solution is None:
            print("NO")
            return

        group_flow = [[0] * m for _ in range(gcount)]
        for gi in range(gcount):
            offset = gi * ecount
            for ep, j in enumerate(active_edges):
                group_flow[gi][j] = integer_solution[offset + ep]

        if not validate_group_flow(
            n, edges, capacities, groups, group_flow
        ):
            print("NO")
            return

    result = split_groups(n, edges, groups, group_flow, k)
    if result is None:
        print("NO")
        return

    # Final exact verification.
    total_load = [0] * m
    for i, (s, t, demand) in enumerate(orders):
        balance = [0] * n
        for j, x in enumerate(result[i]):
            if x < 0:
                print("NO")
                return
            total_load[j] += x
            u, v, _ = edges[j]
            balance[u] += x
            balance[v] -= x

        if s != t:
            for v in range(n):
                expected = demand if v == s else (-demand if v == t else 0)
                if balance[v] != expected:
                    print("NO")
                    return

    for j in range(m):
        if total_load[j] > capacities[j]:
            print("NO")
            return

    out = ["YES"]
    out.extend(" ".join(map(str, row)) for row in result)
    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
