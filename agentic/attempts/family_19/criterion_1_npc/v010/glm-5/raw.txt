```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        k = int(next(iterator))
    except StopIteration:
        return

    contracts = []
    for _ in range(k):
        s = int(next(iterator))
        t = int(next(iterator))
        d = int(next(iterator))
        contracts.append((s, t, d))

    edges = []
    adj = [[] for _ in range(n + 1)]
    for j in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        edges.append((u, v, c))
        adj[u].append((v, j))

    # Build a single flow network
    # Nodes: 1..n are original warehouses
    # Super source: 0, Super sink: n+1
    N = n + 2
    source = 0
    sink = n + 1
    
    # Capacity matrix
    cap = [[0] * N for _ in range(N)]
    
    # Add edges for segments
    for u, v, c in edges:
        cap[u][v] += c
    
    # Add edges from super source to contract sources, and contract sinks to super sink
    total_demand = 0
    for s, t, d in contracts:
        cap[source][s] += d
        cap[t][sink] += d
        total_demand += d

    # Dinic's algorithm
    level = [0] * N
    ptr = [0] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        level[source] = 0
        q = deque([source])
        while q:
            u = q.popleft()
            for v in range(N):
                if level[v] == -1 and cap[u][v] > 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[sink] != -1

    def dfs(u, flow):
        if u == sink:
            return flow
        for v in range(ptr[u], N):
            ptr[u] = v
            if level[v] == level[u] + 1 and cap[u][v] > 0:
                pushed = dfs(v, min(flow, cap[u][v]))
                if pushed > 0:
                    cap[u][v] -= pushed
                    cap[v][u] += pushed
                    return pushed
        return 0

    max_flow = 0
    while bfs():
        ptr = [0] * N
        while True:
            pushed = dfs(source, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed

    if max_flow != total_demand:
        print("NO")
        return

    print("YES")
    
    # To find the flow for each contract, we decompose the flow
    # We use the residual graph to find paths from each contract source to sink
    
    # Build residual adjacency list
    # adj_res[u] contains (v, capacity, original_edge_index)
    # original_edge_index is -1 if not an original segment edge
    adj_res = [[] for _ in range(N)]
    for j, (u, v, _) in enumerate(edges):
        # Forward edge in residual graph has capacity cap[u][v]
        # This corresponds to unused capacity of the original edge
        # The flow on the original edge is original_capacity - residual_capacity
        # But we need to track flow, so we store original capacity
        # Actually, let's just use the cap matrix for residual capacities
        adj_res[u].append((v, j))
    
    # We also need to consider reverse edges for flow decomposition
    # But for finding augmenting paths for a specific commodity, 
    # we can just find any path in the residual graph that was used.
    # However, the cap matrix is modified. 
    # cap[u][v] is residual capacity. Flow is edges[j][2] - cap[u][v].
    
    # Let's try a greedy decomposition for each contract
    # For each contract, we find paths from s to t in the residual graph
    # where flow exists (i.e., original edge had flow).
    # Wait, the cap matrix now represents residual capacities.
    # Original capacity of edge j (u->v) is edges[j][2].
    # Flow on edge j is edges[j][2] - cap[u][v].
    
    # We need to assign this flow to contracts.
    # Let's build a flow graph where edges have flow = original_cap - residual_cap
    flow_on_edge = [0] * m
    for j, (u, v, c) in enumerate(edges):
        flow_on_edge[j] = c - cap[u][v]
    
    # Now we need to decompose this flow into k paths
    # We can do this by finding paths for each contract one by one
    # and subtracting the flow from the edges.
    
    # Let's create a mutable copy of flow_on_edge
    current_flow = list(flow_on_edge)
    
    # For each contract, find paths and subtract
    for s, t, d in contracts:
        contract_flow = [0] * m
        remaining = d
        while remaining > 0:
            # Find a path from s to t with available flow
            parent = [-1] * (n + 1)
            parent_edge = [-1] * (n + 1)
            q = deque([s])
            parent[s] = s
            found = False
            while q:
                u = q.popleft()
                if u == t:
                    found = True
                    break
                for v, edge_idx in adj[u]:
                    if parent[v] == -1 and current_flow[edge_idx] > 0:
                        parent[v] = u
                        parent_edge[v] = edge_idx
                        q.append(v)
            
            if not found:
                # Should not happen if max_flow == total_demand
                break
            
            # Find bottleneck
            path_flow = remaining
            curr = t
            while curr != s:
                edge_idx = parent_edge[curr]
                path_flow = min(path_flow, current_flow[edge_idx])
                curr = parent[curr]
            
            # Subtract flow
            curr = t
            while curr != s:
                edge_idx = parent_edge[curr]
                current_flow[edge_idx] -= path_flow
                contract_flow[edge_idx] += path_flow
                curr = parent[curr]
            
            remaining -= path_flow
        
        print(" ".join(map(str, contract_flow)))

if __name__ == '__main__':
    solve()
```
