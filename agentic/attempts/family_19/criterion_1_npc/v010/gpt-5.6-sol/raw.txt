import sys
import random
import itertools
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

it = iter(data)
n = next(it)
m = next(it)
k = next(it)

contracts = []
for _ in range(k):
    s = next(it) - 1
    t = next(it) - 1
    d = next(it)
    contracts.append((s, t, d))

edges = []
out_edges = [[] for _ in range(n)]
rev_edges = [[] for _ in range(n)]
initial_caps = []

for eid in range(m):
    u = next(it) - 1
    v = next(it) - 1
    c = next(it)
    edges.append((u, v))
    initial_caps.append(c)
    out_edges[u].append(eid)
    rev_edges[v].append(eid)


class Dinic:
    __slots__ = ("n", "g", "level", "pos")

    def __init__(self, size):
        self.n = size
        self.g = [[] for _ in range(size)]
        self.level = [-1] * size
        self.pos = [0] * size

    def add_edge(self, u, v, cap):
        a = [v, cap, None]
        b = [u, 0, a]
        a[2] = b
        self.g[u].append(a)
        self.g[v].append(b)
        return a

    def bfs(self, s, t):
        level = self.level
        for i in range(self.n):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            nl = level[u] + 1
            for e in self.g[u]:
                if e[1] > 0 and level[e[0]] < 0:
                    level[e[0]] = nl
                    if e[0] == t:
                        continue
                    q.append(e[0])
        return level[t] >= 0

    def dfs(self, u, t, pushed):
        if u == t:
            return pushed
        g = self.g
        level = self.level
        while self.pos[u] < len(g[u]):
            e = g[u][self.pos[u]]
            if e[1] > 0 and level[e[0]] == level[u] + 1:
                got = self.dfs(e[0], t, min(pushed, e[1]))
                if got:
                    e[1] -= got
                    e[2][1] += got
                    return got
            self.pos[u] += 1
        return 0

    def flow(self, s, t, limit):
        if s == t:
            return limit
        result = 0
        inf = 10**30
        while result < limit and self.bfs(s, t):
            for i in range(self.n):
                self.pos[i] = 0
            while result < limit:
                got = self.dfs(s, t, min(inf, limit - result))
                if not got:
                    break
                result += got
        return result


def single_flow(s, t, demand, caps, order=None):
    if demand == 0 or s == t:
        return demand, [0] * m

    dinic = Dinic(n)
    refs = [None] * m
    if order is None:
        order = range(m)

    for eid in order:
        u, v = edges[eid]
        refs[eid] = dinic.add_edge(u, v, caps[eid])

    value = dinic.flow(s, t, demand)
    used = [0] * m
    for eid in range(m):
        ref = refs[eid]
        used[eid] = caps[eid] - ref[1]
    return value, used


# Immediate individual feasibility checks and tightness estimates.
slacks = [0] * k
for i, (s, t, d) in enumerate(contracts):
    if d == 0 or s == t:
        slacks[i] = 10**30
        continue
    value, _ = single_flow(s, t, d, initial_caps)
    if value < d:
        print("NO")
        sys.exit()
    # Compute a capped estimate of available extra flow.
    extra_limit = d + sum(initial_caps)
    max_value, _ = single_flow(s, t, extra_limit, initial_caps)
    slacks[i] = max_value - d


def greedy_attempt(commodity_order, edge_order):
    caps = initial_caps[:]
    answer = [[0] * m for _ in range(k)]

    for ci in commodity_order:
        s, t, d = contracts[ci]
        if d == 0 or s == t:
            continue
        value, used = single_flow(s, t, d, caps, edge_order)
        if value < d:
            return None
        row = answer[ci]
        for eid, x in enumerate(used):
            if x:
                caps[eid] -= x
                row[eid] += x
    return answer


rng = random.Random(712367821)
base_edge_order = list(range(m))
tight_order = sorted(range(k), key=lambda x: (slacks[x], -contracts[x][2], x))

commodity_orders = []
seen_orders = set()


def add_order(order):
    order = tuple(order)
    if order not in seen_orders:
        seen_orders.add(order)
        commodity_orders.append(order)


add_order(range(k))
add_order(reversed(range(k)))
add_order(tight_order)
add_order(reversed(tight_order))

if k <= 7:
    for p in itertools.islice(itertools.permutations(range(k)), 120):

else:

        p = list(range(k))
        rng.shuffle(p)
        add_order(p)

edge_orders = [base_edge_order, list(reversed(base_edge_order))]
for _ in range(18):
    p = base_edge_order[:]
    rng.shuffle(p)
    edge_orders.append(p)

for idx, order in enumerate(commodity_orders):
    tries = 2 if len(commodity_orders) > 30 else min(6, len(edge_orders))
    for z in range(tries):
        edge_order = edge_orders[(idx + z) % len(edge_orders)]
        result = greedy_attempt(order, edge_order)
        if result is not None:
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in result)

# Exact branch-and-bound search by integral path decomposition.
caps = initial_caps[:]
remaining = [d if contracts[i][0] != contracts[i][1] else 0
             for i, (_, _, d) in enumerate(contracts)]
answer = [[0] * m for _ in range(k)]
memo = set()


def residual_distances(target):
    dist = [-1] * n
    dist[target] = 0
    q = deque([target])
    while q:
        v = q.popleft()
        nd = dist[v] + 1
        for eid in rev_edges[v]:
            if caps[eid] <= 0:
                continue
            u, _ = edges[eid]
            if dist[u] < 0:
                dist[u] = nd
                q.append(u)
    return dist


def path_generator(source, target, hint):
    dist = residual_distances(target)
    if dist[source] < 0:
        return

    visited = [False] * n
    visited[source] = True
    path = []

    def generate(u):
        if u == target:
            yield tuple(path)
            return

        candidates = []
        for eid in out_edges[u]:
            if caps[eid] <= 0:
                continue
            v = edges[eid][1]
            if visited[v] or dist[v] < 0:
                continue
            candidates.append(eid)

        candidates.sort(
            key=lambda eid: (
                0 if hint[eid] > 0 else 1,
                dist[edges[eid][1]],
                -hint[eid],
                -caps[eid],
                eid
            )
        )

        for eid in candidates:
            v = edges[eid][1]
            visited[v] = True
            path.append(eid)
            yield from generate(v)
            path.pop()
            visited[v] = False

    yield from generate(source)


def quantity_order(maximum):
    if maximum <= 0:
        return
    yield maximum
    if maximum > 1:
        yield 1
    for q in range(maximum - 1, 1, -1):
        yield q


def search():
    if all(x == 0 for x in remaining):
        return True

    state = (tuple(remaining), tuple(caps))
    if state in memo:
        return False

    chosen = -1
    chosen_hint = None
    best_key = None

    for ci in range(k):
        need = remaining[ci]
        if need <= 0:
            continue
        s, t, _ = contracts[ci]
        value, used = single_flow(s, t, need, caps)
        if value < need:
            memo.add(state)
            return False

        # Estimate slack only when useful; capped at need plus current capacity.
        probe_limit = need + sum(caps)
        available, _ = single_flow(s, t, probe_limit, caps)
        key = (available - need, need, ci)
        if best_key is None or key < best_key:
            best_key = key
            chosen = ci
            chosen_hint = used

    s, t, _ = contracts[chosen]
    need = remaining[chosen]

    for path in path_generator(s, t, chosen_hint):
        bottleneck = need
        for eid in path:
            if caps[eid] < bottleneck:
                bottleneck = caps[eid]

        for amount in quantity_order(bottleneck):
            remaining[chosen] -= amount
            row = answer[chosen]
            for eid in path:
                caps[eid] -= amount
                row[eid] += amount

            if search():
                return True

            for eid in path:
                caps[eid] += amount
                row[eid] -= amount
            remaining[chosen] += amount

    if len(memo) < 400000:
        memo.add(state)
    return False


if search():
    out = ["YES"]
    out.extend(" ".join(map(str, row)) for row in answer)
    sys.stdout.write("\n".join(out))
else:
    print("NO")
