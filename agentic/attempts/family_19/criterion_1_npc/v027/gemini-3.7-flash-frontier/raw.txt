import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    commodities = []
    for _ in range(k):
        s = int(next(iterator)) - 1
        t = int(next(iterator)) - 1
        d = int(next(iterator))
        commodities.append((s, t, d))
        
    edges = []
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        cap = int(next(iterator))
        edges.append((u, v, cap))

    # Try using scipy.optimize.milp if available
    try:
        from scipy.optimize import milp, LinearConstraint
        import numpy as np

        # Variables: f_{i, e} for i in [0..k-1], e in [0..m-1]
        # Total variables: num_vars = k * m
        num_vars = k * m
        
        # We don't have an objective to minimize/maximize, just find a feasible solution.
        c = np.zeros(num_vars)
        integrality = np.ones(num_vars)  # 1 for integer
        
        # Constraints:
        # 1. Edge capacity constraints: sum_i f_{i, e} <= cap(e) for each e
        # 2. Flow conservation for each commodity i and vertex v:
        #    sum_{e out of v} f_{i, e} - sum_{e into v} f_{i, e} = b_{i, v}
        
        row_list = []
        lhs_list = []
        rhs_list = []
        
        # Edge capacity
        for e, (u, v, cap) in enumerate(edges):
            row = np.zeros(num_vars)
            for i in range(k):
                row[i * m + e] = 1.0
            row_list.append(row)
            lhs_list.append(0.0)
            rhs_list.append(float(cap))
            
        # Flow conservation
        for i in range(k):
            s_i, t_i, d_i = commodities[i]
            for v in range(n):
                row = np.zeros(num_vars)
                for e, (eu, ev, cap) in enumerate(edges):
                    if eu == v:
                        row[i * m + e] += 1.0
                    if ev == v:
                        row[i * m + e] -= 1.0
                
                target = 0.0
                if v == s_i:
                    target += d_i
                if v == t_i:
                    target -= d_i
                    
                row_list.append(row)
                lhs_list.append(target)
                rhs_list.append(target)
                
        A = np.array(row_list)
        constraints = LinearConstraint(A, lhs_list, rhs_list)
        bounds = (np.zeros(num_vars), np.full(num_vars, np.inf))
        
        res = milp(c=c, integrality=integrality, constraints=constraints, bounds=bounds)
        
        if res.success:
            print("YES")
            x = np.round(res.x).astype(int)
            for i in range(k):
                row_vals = [str(x[i * m + e]) for e in range(m)]
                print(" ".join(row_vals))
            return
        else:
            print("NO")
            return
    except Exception:
        pass

    # Fallback to pure Python backtracking / path-finding if scipy fails/unavailable
    # Expand each commodity into individual unit-demand paths
    units = []
    for i, (s, t, d) in enumerate(commodities):
        for _ in range(d):
            units.append((i, s, t))

    # Adjacency list
    adj = [[] for _ in range(n)]
    for e_idx, (u, v, cap) in enumerate(edges):
        adj[u].append((v, e_idx))

    remaining_cap = [edges[e][2] for e in range(m)]
    flow = [[0] * m for _ in range(k)]

    def solve_units(idx):
        if idx == len(units):
            return True
        
        comm_idx, s, t = units[idx]
        
        # Simple DFS/BFS to find a path from s to t
        # We use DFS with visited array to avoid cycles in a single path
        def find_path(curr, target, visited, path):
            if curr == target:
                return True
            for nxt, e_idx in adj[curr]:
                if remaining_cap[e_idx] > 0 and nxt not in visited:
                    visited.add(nxt)
                    path.append(e_idx)
                    remaining_cap[e_idx] -= 1
                    flow[comm_idx][e_idx] += 1
                    
                    if find_path(nxt, target, visited, path):
                        if solve_units(idx + 1):
                            return True
                    
                    # Backtrack
                    flow[comm_idx][e_idx] -= 1
                    remaining_cap[e_idx] += 1
                    path.pop()
                    visited.remove(nxt)
            return False

        visited = {s}
        path = []
        return find_path(s, t, visited, path)

    if solve_units(0):
        print("YES")
        for i in range(k):
            print(" ".join(map(str, flow[i])))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
