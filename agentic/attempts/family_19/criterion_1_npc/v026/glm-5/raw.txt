        # Forward edge
        edge_list.append([v, u, cap, len(adj[v]), comp_id, tube_id])

        # Reverse edge
        edge_list.append([u, v, rev_cap, len(adj[u]) - 1, comp_id, tube_id])

        add_edge(compound_sink_start + i, sink, d)
