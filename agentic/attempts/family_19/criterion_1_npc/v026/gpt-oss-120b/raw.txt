import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    comps = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        comps.append((s, t, d))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append([u, v, cap])

    # adjacency for Dinic
    class Dinic:
        def __init__(self, N):
            self.N = N
            self.adj = [[] for _ in range(N)]

        def add_edge(self, u, v, c, idx=-1):
            self.adj[u].append([v, c, len(self.adj[v]), idx])
            self.adj[v].append([u, 0, len(self.adj[u]) - 1, -1])

        def bfs(self, s, t, level):
            q = [s]
            level[s] = 0
            for u in q:
                for v, cap, rev, _ in self.adj[u]:
                    if cap and level[v] < 0:
                        level[v] = level[u] + 1
                        q.append(v)
            return level[t] >= 0

        def dfs(self, u, t, f, level, it):
            if u == t:
                return f
            for i in range(it[u], len(self.adj[u])):
                it[u] = i
                v, cap, rev, _ = self.adj[u][i]
                if cap and level[u] + 1 == level[v]:
                    ret = self.dfs(v, t, min(f, cap), level, it)
                    if ret:
                        self.adj[u][i][1] -= ret
                        self.adj[v][rev][1] += ret
                        return ret
            return 0

        def max_flow(self, s, t):
            flow = 0
            INF = 10**18
            level = [-1]*self.N
            while self.bfs(s, t, level):
                it = [0]*self.N
                while True:
                    f = self.dfs(s, t, INF, level, it)
                    if not f:
                        break
                    flow += f
                level = [-1]*self.N
            return flow

        def get_edge_flow(self, u, idx):
            for v, cap, rev, i in self.adj[u]:
                if i == idx:
                    # original capacity - remaining capacity
                    return self.adj[v][rev][1]
            return 0

    # result matrix
    res = [[0]*m for _ in range(k)]
    caps = [e[2] for e in edges]

    for ci, (s, t, d) in enumerate(comps):
        din = Dinic(n)
        for ei, (u, v, _) in enumerate(edges):
            din.add_edge(u, v, caps[ei], ei)
        flow = din.max_flow(s, t)
        if flow < d:
            print("NO")
            return
        # retrieve flows on each edge
        for ei, (u, v, _) in enumerate(edges):
            f = din.get_edge_flow(u, ei)
            # we may have sent more than needed; limit to d
            used = min(f, d)
            res[ci][ei] = used
            caps[ei] -= used
            d -= used
        # if still demand left (should not happen)
        if d > 0:
            print("NO")
            return

    print("YES")
    out = sys.stdout
    for row in res:
        out.write(" ".join(map(str, row)) + "\n")

if __name__ == "__main__":
    solve()
