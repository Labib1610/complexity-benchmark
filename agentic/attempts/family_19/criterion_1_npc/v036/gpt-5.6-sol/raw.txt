import sys
import random
from collections import deque

sys.setrecursionlimit(1_000_000)


class Dinic:
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, u, v, cap):
        f = [v, len(self.g[v]), cap]
        r = [u, len(self.g[u]), 0]
        self.g[u].append(f)
        self.g[v].append(r)
        return f

    def max_flow(self, s, t, limit):
        if s == t:
            return limit
        total = 0
        n = self.n

        while total < limit:
            level = [-1] * n
            level[s] = 0
            q = deque([s])
            while q:
                u = q.popleft()
                for v, rev, cap in self.g[u]:
                    if cap > 0 and level[v] < 0:
                        level[v] = level[u] + 1
                        q.append(v)

            if level[t] < 0:
                break

            it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                gu = self.g[u]
                while it[u] < len(gu):
                    e = gu[it[u]]
                    v, rev, cap = e
                    if cap > 0 and level[v] == level[u] + 1:
                        got = dfs(v, min(pushed, cap))
                        if got:
                            e[2] -= got
                            self.g[v][rev][2] += got
                            return got
                    it[u] += 1
                return 0

            while total < limit:
                pushed = dfs(s, limit - total)
                if pushed == 0:
                    break
                total += pushed

        return total


def verify(flows, n, m, k, orders, edges):
    used = [0] * m
    for i in range(k):
        bal = [0] * n
        row = flows[i]
        if len(row) != m:
            return False
        for j, x in enumerate(row):
            if x < 0:
                return False
            u, v, cap = edges[j]
            used[j] += x
            bal[u] += x
            bal[v] -= x

        s, t, d = orders[i]
        wanted = [0] * n
        if s != t:
            wanted[s] += d
            wanted[t] -= d
        if bal != wanted:
            return False

    for j, (_, _, cap) in enumerate(edges):
        if used[j] > cap:
            return False
    return True


def sequential_attempt(n, m, k, orders, edges, order, shuffled_edges):
    residual = [e[2] for e in edges]
    answer = [[0] * m for _ in range(k)]

    for commodity in order:
        s, t, demand = orders[commodity]
        if demand == 0 or s == t:
            continue

        dinic = Dinic(n)
        refs = [None] * m
        before = [0] * m

        for j in shuffled_edges:
            u, v, _ = edges[j]
            before[j] = residual[j]
            refs[j] = dinic.add_edge(u, v, residual[j])

        got = dinic.max_flow(s, t, demand)
        if got != demand:
            return None

        for j in range(m):
            used = before[j] - refs[j][2]
            if used:
                answer[commodity][j] = used
                residual[j] -= used

    return answer


class LPSolver:
    EPS = 1e-9
    INF = float("inf")

    def __init__(self, A, b):
        self.m = len(b)
        self.n = len(A[0]) if A else 0
        m, n = self.m, self.n

        self.B = [n + i for i in range(m)]
        self.N = list(range(n)) + [-1]
        self.D = [[0.0] * (n + 2) for _ in range(m + 2)]

        for i in range(m):
            self.D[i][:n] = A[i]
            self.D[i][n] = -1.0
            self.D[i][n + 1] = float(b[i])

        self.D[m][:n] = [0.0] * n
        self.D[m + 1][n] = 1.0

    def pivot(self, r, s):
        D = self.D
        m, n = self.m, self.n
        inv = 1.0 / D[r][s]
        row_r = D[r]

        for i in range(m + 2):
            if i == r:
                continue
            row_i = D[i]
            factor = row_i[s] * inv
            if abs(factor) > self.EPS:
                for j in range(n + 2):
                    if j != s:
                        row_i[j] -= row_r[j] * factor

        for j in range(n + 2):
            if j != s:
                row_r[j] *= inv

        for i in range(m + 2):
            if i != r:
                D[i][s] *= -inv

        D[r][s] = inv
        self.B[r], self.N[s] = self.N[s], self.B[r]

    def simplex(self, phase):
        D = self.D
        m, n = self.m, self.n
        xrow = m + 1 if phase == 1 else m

        while True:
            s = -1
            for j in range(n + 1):
                if phase == 2 and self.N[j] == -1:
                    continue
                if (s == -1 or D[xrow][j] < D[xrow][s] - self.EPS or
                        (abs(D[xrow][j] - D[xrow][s]) <= self.EPS
                         and self.N[j] < self.N[s])):
                    s = j

            if s == -1 or D[xrow][s] >= -self.EPS:
                return True

            r = -1
            for i in range(m):
                if D[i][s] <= self.EPS:
                    continue
                ratio = D[i][n + 1] / D[i][s]
                if r == -1:
                    r = i
                else:
                    old_ratio = D[r][n + 1] / D[r][s]
                    if (ratio < old_ratio - self.EPS or
                            (abs(ratio - old_ratio) <= self.EPS
                             and self.B[i] < self.B[r])):
                        r = i

            if r == -1:
                return False
            self.pivot(r, s)

    def solve(self):
        m, n = self.m, self.n
        D = self.D

        if m == 0:
            return [0.0] * n

        r = min(range(m), key=lambda i: D[i][n + 1])
        if D[r][n + 1] < -self.EPS:
            self.pivot(r, n)
            if not self.simplex(1):
                return None
            if D[m + 1][n + 1] < -self.EPS:
                return None
            if abs(D[m + 1][n + 1]) > 1e-7:
                return None

            for i in range(m):
                if self.B[i] == -1:
                    s = -1
                    for j in range(n + 1):
                        if (s == -1 or D[i][j] < D[i][s] - self.EPS or
                                (abs(D[i][j] - D[i][s]) <= self.EPS
                                 and self.N[j] < self.N[s])):
                            s = j
                    if s != -1 and abs(D[i][s]) > self.EPS:
                        self.pivot(i, s)

        if not self.simplex(2):
            return None

        x = [0.0] * n
        for i in range(m):
            if 0 <= self.B[i] < n:
                x[self.B[i]] = D[i][n + 1]
        return x


def solve_ilp(n, m, k, orders, edges):
    variables = k * m
    A = []
    b = []

    # Shared conveyor capacities.
    for j, (_, _, cap) in enumerate(edges):
        row = [0.0] * variables
        for i in range(k):
            row[i * m + j] = 1.0
        A.append(row)
        b.append(cap)

    outgoing = [[] for _ in range(n)]
    incoming = [[] for _ in range(n)]
    for j, (u, v, _) in enumerate(edges):
        outgoing[u].append(j)
        incoming[v].append(j)

    # One conservation equation can be omitted for each commodity.
    omit = n - 1
    for i, (s, t, demand) in enumerate(orders):
        supply = [0] * n
        if s != t:
            supply[s] += demand
            supply[t] -= demand

        offset = i * m
        for v in range(n):
            if v == omit:
                continue
            row = [0.0] * variables
            for j in outgoing[v]:
                row[offset + j] += 1.0
            for j in incoming[v]:
                row[offset + j] -= 1.0

            A.append(row)
            b.append(supply[v])
            A.append([-x for x in row])
            b.append(-supply[v])

    stack = [()]
    while stack:
        branches = stack.pop()
        extra_a = []
        extra_b = []

        for var, direction, bound in branches:
            row = [0.0] * variables
            if direction == 0:       # x <= bound
                row[var] = 1.0
                extra_b.append(bound)
            else:                    # x >= bound
                row[var] = -1.0
                extra_b.append(-bound)
            extra_a.append(row)

        solver = LPSolver(A + extra_a, b + extra_b)
        solution = solver.solve()
        if solution is None:
            continue

        rounded = []
        fractional_var = -1
        best_fractionality = -1.0

        for idx, value in enumerate(solution):
            if -1e-7 < value < 0:
                value = 0.0
                solution[idx] = 0.0
            nearest = round(value)
            rounded.append(int(nearest))
            frac = abs(value - nearest)
            if frac > best_fractionality:
                best_fractionality = frac
                fractional_var = idx

        candidate = [
            rounded[i * m:(i + 1) * m]
            for i in range(k)
        ]
        if best_fractionality <= 1e-6 and verify(
                candidate, n, m, k, orders, edges):
            return candidate

        if fractional_var < 0 or best_fractionality <= 1e-10:
            continue

        value = solution[fractional_var]
        low = int(value // 1)
        high = low + 1

        # Explore both integer halves.
        stack.append(branches + ((fractional_var, 1, high),))
        stack.append(branches + ((fractional_var, 0, low),))

    return None


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    orders = []
    for _ in range(k):
        s = next(it) - 1
        t = next(it) - 1
        d = next(it)
        orders.append((s, t, d))

    edges = []
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        edges.append((u, v, cap))

    rng = random.Random(0xC0FFEE)
    base_edges = list(range(m))

    order_candidates = [
        list(range(k)),
        sorted(range(k), key=lambda i: -orders[i][2]),
        sorted(range(k), key=lambda i: orders[i][2]),
    ]

    attempts = max(8, min(40, 120 // max(1, k)))
    for attempt in range(attempts):
        if attempt < len(order_candidates):
            order = order_candidates[attempt][:]
        else:
            order = list(range(k))
            rng.shuffle(order)

        edge_order = base_edges[:]
        if attempt:
            rng.shuffle(edge_order)

        answer = sequential_attempt(
            n, m, k, orders, edges, order, edge_order
        )
        if answer is not None and verify(answer, n, m, k, orders, edges):
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in answer)
            sys.stdout.write("\n".join(out))
            return

    answer = solve_ilp(n, m, k, orders, edges)
    if answer is None:
        sys.stdout.write("NO")
    else:
        out = ["YES"]
        out.extend(" ".join(map(str, row)) for row in answer)
        sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
