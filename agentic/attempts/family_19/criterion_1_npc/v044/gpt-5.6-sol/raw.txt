import sys
from collections import deque
from itertools import permutations

sys.setrecursionlimit(1_000_000)


class Dinic:
    __slots__ = ("n", "g", "level", "it")

    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]
        self.level = [-1] * n
        self.it = [0] * n

    def add_edge(self, u, v, cap):
        a = [v, cap, None]
        b = [u, 0, a]
        a[2] = b
        self.g[u].append(a)
        self.g[v].append(b)
        return a

    def max_flow(self, s, t, limit):
        if s == t:
            return limit

        total = 0
        n = self.n
        while total < limit:
            level = [-1] * n
            level[s] = 0
            q = deque([s])

            while q:
                u = q.popleft()
                nl = level[u] + 1
                for e in self.g[u]:
                    if e[1] > 0 and level[e[0]] < 0:
                        level[e[0]] = nl
                        q.append(e[0])

            if level[t] < 0:
                break

            self.level = level
            self.it = [0] * n

            def dfs(u, pushed):
                if u == t:
                    return pushed
                gu = self.g[u]
                i = self.it[u]
                while i < len(gu):
                    self.it[u] = i
                    e = gu[i]
                    if e[1] > 0 and level[e[0]] == level[u] + 1:
                        got = dfs(e[0], min(pushed, e[1]))
                        if got:
                            e[1] -= got
                            e[2][1] += got
                            return got
                    i += 1
                    self.it[u] = i
                return 0

            while total < limit:
                pushed = dfs(s, limit - total)
                if not pushed:
                    break
                total += pushed

        return total


def route_in_order(n, edges, commodities, order):
    m = len(edges)
    residual = [e[2] for e in edges]
    answer = [[0] * m for _ in commodities]

    for ci in order:
        s, t, demand = commodities[ci]
        if demand == 0 or s == t:
            continue

        dinic = Dinic(n)
        refs = [None] * m
        before = [0] * m

        for j, (u, v, _) in enumerate(edges):
            c = residual[j]
            before[j] = c
            refs[j] = dinic.add_edge(u, v, c)

        if dinic.max_flow(s, t, demand) != demand:
            return None

        row = answer[ci]
        for j in range(m):
            used = before[j] - refs[j][1]
            row[j] = used
            residual[j] -= used

    return answer


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    commodities = []
    for _ in range(k):
        s = next(it) - 1
        t = next(it) - 1
        d = next(it)
        commodities.append((s, t, d))

    edges = []
    graph = [[] for _ in range(n)]
    reverse_graph = [[] for _ in range(n)]

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        cap = next(it)
        edges.append((u, v, cap))
        if cap > 0:
            graph[u].append(v)
            reverse_graph[v].append(u)

    # Necessary independent feasibility checks.
    for s, t, demand in commodities:
        if demand == 0 or s == t:
            continue
        dinic = Dinic(n)
        for u, v, cap in edges:
            dinic.add_edge(u, v, cap)
        if dinic.max_flow(s, t, demand) != demand:
            print("NO")
            return

    # Fast attempts using sequential maximum flows in several orders.
    orders = []
    seen_orders = set()

    def add_order(order):
        order = tuple(order)
        if order not in seen_orders:
            seen_orders.add(order)
            orders.append(order)

    add_order(range(k))
    add_order(range(k - 1, -1, -1))
    add_order(sorted(range(k), key=lambda i: -commodities[i][2]))
    add_order(sorted(range(k), key=lambda i: commodities[i][2]))

    if k <= 7:
        for order in permutations(range(k)):
            add_order(order)
    elif k <= 20:
        base = list(range(k))
        for shift in range(k):
            add_order(base[shift:] + base[:shift])

    for order in orders:
        answer = route_in_order(n, edges, commodities, order)
        if answer is not None:
            out = ["YES"]
            out.extend(" ".join(map(str, row)) for row in answer)
            sys.stdout.write("\n".join(out))
            return

    # Exact finite-domain constraint search.
    var_count = k * m
    lo = [0] * var_count
    hi = [0] * var_count

    equation_vars = [[] for _ in range(k * n)]
    equation_rhs = [0] * (k * n)
    capacity_vars = [[] for _ in range(m)]
    constraint_count = k * n + m
    var_constraints = [[] for _ in range(var_count)]

    for ci, (s, t, demand) in enumerate(commodities):
        if s != t:
            equation_rhs[ci * n + s] = demand
            equation_rhs[ci * n + t] = -demand

        reachable_from_s = [False] * n
        can_reach_t = [False] * n

        q = deque([s])
        reachable_from_s[s] = True
        while q:
            u = q.popleft()
            for v in graph[u]:
                if not reachable_from_s[v]:
                    reachable_from_s[v] = True
                    q.append(v)

        q = deque([t])
        can_reach_t[t] = True
        while q:
            u = q.popleft()
            for v in reverse_graph[u]:
                if not can_reach_t[v]:
                    can_reach_t[v] = True
                    q.append(v)

        for ej, (u, v, cap) in enumerate(edges):
            vid = ci * m + ej
            if (
                demand > 0
                and s != t
                and u != v
                and cap > 0
                and reachable_from_s[u]
                and can_reach_t[v]
            ):
                hi[vid] = min(cap, demand)

            capacity_vars[ej].append(vid)
            var_constraints[vid].append(k * n + ej)

            if u != v:
                equation_vars[ci * n + u].append((vid, 1))
                equation_vars[ci * n + v].append((vid, -1))
                var_constraints[vid].append(ci * n + u)
                var_constraints[vid].append(ci * n + v)

    trail = []

    def set_bounds(v, nl, nh):
        if nl < lo[v]:
            nl = lo[v]
        if nh > hi[v]:
            nh = hi[v]
        if nl > nh:
            return False
        if nl != lo[v] or nh != hi[v]:
            trail.append((v, lo[v], hi[v]))
            lo[v] = nl
            hi[v] = nh
        return True

    def propagate(initial_constraints):
        queue = deque()
        queued = set()

        for cid in initial_constraints:
            if cid not in queued:
                queued.add(cid)
                queue.append(cid)

        def update(v, nl, nh):
            if nl < lo[v]:
                nl = lo[v]
            if nh > hi[v]:
                nh = hi[v]
            if nl > nh:
                return False
            if nl == lo[v] and nh == hi[v]:
                return True

            trail.append((v, lo[v], hi[v]))
            lo[v] = nl
            hi[v] = nh

            for cid2 in var_constraints[v]:
                if cid2 not in queued:
                    queued.add(cid2)
                    queue.append(cid2)
            return True

        while queue:
            cid = queue.popleft()
            queued.remove(cid)

            if cid < k * n:
                items = equation_vars[cid]
                rhs = equation_rhs[cid]
                minimum = 0
                maximum = 0

                for v, sign in items:
                    if sign == 1:
                        minimum += lo[v]
                        maximum += hi[v]
                    else:
                        minimum -= hi[v]
                        maximum -= lo[v]

                if rhs < minimum or rhs > maximum:
                    return False

                changes = []
                for v, sign in items:
                    if sign == 1:
                        other_min = minimum - lo[v]
                        other_max = maximum - hi[v]
                        nl = max(lo[v], rhs - other_max)
                        nh = min(hi[v], rhs - other_min)
                    else:
                        other_min = minimum + hi[v]
                        other_max = maximum + lo[v]
                        nl = max(lo[v], other_min - rhs)
                        nh = min(hi[v], other_max - rhs)

                    if nl > nh:
                        return False
                    if nl != lo[v] or nh != hi[v]:
                        changes.append((v, nl, nh))

                for v, nl, nh in changes:
                    if not update(v, nl, nh):
                        return False

            else:
                edge_id = cid - k * n
                items = capacity_vars[edge_id]
                cap = edges[edge_id][2]
                minimum = sum(lo[v] for v in items)

                if minimum > cap:
                    return False

                changes = []
                for v in items:
                    nh = min(hi[v], cap - (minimum - lo[v]))
                    if nh < lo[v]:
                        return False
                    if nh != hi[v]:
                        changes.append((v, lo[v], nh))

                for v, nl, nh in changes:
                    if not update(v, nl, nh):
                        return False

        return True

    all_constraints = range(constraint_count)
    if not propagate(all_constraints):
        print("NO")
        return

    def rollback(marker):
        while len(trail) > marker:
            v, old_lo, old_hi = trail.pop()
            lo[v] = old_lo
            hi[v] = old_hi

    def search():
        chosen = -1
        best_width = None

        for v in range(var_count):
            if lo[v] < hi[v]:
                width = hi[v] - lo[v]
                if best_width is None or width < best_width:
                    best_width = width
                    chosen = v
                    if width == 1:
                        break

        if chosen < 0:
            return True

        v = chosen
        low = lo[v]
        high = hi[v]

        if low == 0:
            branches = ((0, 0), (1, high))
        else:
            mid = (low + high) // 2
            branches = ((low, mid), (mid + 1, high))

        for nl, nh in branches:
            if nl > nh:
                continue
            marker = len(trail)

            if set_bounds(v, nl, nh) and propagate(var_constraints[v]):
                if search():
                    return True

            rollback(marker)

        return False

    if not search():
        print("NO")
        return

    answer = []
    for ci in range(k):
        answer.append(lo[ci * m:(ci + 1) * m])

    out = ["YES"]
    out.extend(" ".join(map(str, row)) for row in answer)
    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
