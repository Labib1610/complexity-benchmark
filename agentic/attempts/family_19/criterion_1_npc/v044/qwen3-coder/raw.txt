from collections import defaultdict, deque

def max_flow(graph, source, sink, capacity, flow):
    n = len(graph)
    while True:
        # BFS to find augmenting path
        parent = [-1] * n
        queue = deque([source])
        parent[source] = source
        
        while queue and parent[sink] == -1:
            u = queue.popleft()
            for v in graph[u]:
                if parent[v] == -1 and capacity[(u, v)] > flow[(u, v)]:
                    parent[v] = u
                    queue.append(v)
        
        if parent[sink] == -1:
            break
            
        # Find minimum residual capacity along the path
        path_flow = float('inf')
        s = sink
        while s != source:
            path_flow = min(path_flow, capacity[(parent[s], s)] - flow[(parent[s], s)])
            s = parent[s]
            
        # Update flow
        s = sink
        while s != source:
            flow[(parent[s], s)] += path_flow
            flow[(s, parent[s])] -= path_flow
            s = parent[s]
    
    # Calculate total flow
    total_flow = 0
    for v in graph[source]:
        total_flow += flow[(source, v)]
    return total_flow

def max_flow(graph, source, sink, capacity, flow):
    n = max(max(nodes) for nodes in graph.values()) + 1 if graph else 0
    n = max(n, source+1, sink+1)
    
    while True:
        parent = [-1] * n
        visited = [False] * n
        queue = deque([source])
        visited[source] = True
        parent[source] = source
        
        while queue and not visited[sink]:
            u = queue.popleft()
            for v in graph[u]:
                if not visited[v] and capacity[(u, v)] > flow[(u, v)]:
                    parent[v] = u
                    visited[v] = True
                    queue.append(v)
        
        if not visited[sink]:
            break

        s = sink
        while s != source:
            path_flow = min(path_flow, capacity[(parent[s], s)] - flow[(parent[s], s)])
            s = parent[s]
            
        s = sink
        while s != source:
            flow[(parent[s], s)] += path_flow
            flow[(s, parent[s])] -= path_flow
            s = parent[s]
