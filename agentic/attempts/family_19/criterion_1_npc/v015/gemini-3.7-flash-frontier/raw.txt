import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    contracts = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        contracts.append((s, t, d))

    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        c = int(next(it))
        edges.append((u, v, c))

    # Fast path: DFS / Backtracking on paths for unit flows if total flow is small,
    # or general ILP Branch and Bound with Simplex.

    # Formulate ILP:
    # Variables: x_{i, j} for i in range(k), j in range(m) -> index i * m + j
    num_vars = k * m

    # Equality constraints:
    # For each contract i and terminal v:
    # sum_{j: u_j = v} x_{i, j} - sum_{j: v_j = v} x_{i, j} = b
    # where b = d_i if v == s_i, -d_i if v == t_i, 0 otherwise.

    # Inequality constraints:
    # sum_{i} x_{i, j} <= c_j

    # We can use a Branch-and-Bound / Backtracking solver.
    # First, let's try a DFS path-finding / residual graph approach if capacities/demands allow,
    # or Simplex-based B&B.

    # Let's implement Two-Phase Simplex:
    EPS = 1e-7

    def simplex(A, b, c):
        # Solves min c^T x s.t. A_eq x = b, x >= 0
        # A is M x N, b is M, c is N
        # Returns (status, val, x) where status is True (optimal) or False (infeasible)
        M = len(A)
        N = len(c)

        # Make b >= 0
        for i in range(M):
            if b[i] < -EPS:
                b[i] = -b[i]
                for j in range(N):
                    A[i][j] = -A[i][j]

        # Phase 1: Add artificial variables
        # Tableau size: (M + 2) x (N + M + 1)
        # rows: 0..M-1 constraints, row M: Phase 2 obj, row M+1: Phase 1 obj
        T = [[0.0] * (N + M + 1) for _ in range(M + 2)]
        basis = [0] * M

        for i in range(M):
            for j in range(N):
                T[i][j] = float(A[i][j])
            T[i][N + i] = 1.0
            T[i][-1] = float(b[i])
            basis[i] = N + i

        for j in range(N):
            T[M][j] = float(c[j])

        # Phase 1 objective: minimize sum of artificial variables
        # row M+1 = sum of all constraint rows
        for j in range(N + M + 1):
            s = 0.0
            for i in range(M):
                s += T[i][j]
            T[M + 1][j] = -s

        def pivot(r, c_p, num_rows):
            inv = 1.0 / T[r][c_p]
            for j in range(len(T[0])):
                T[r][j] *= inv
            for i in range(num_rows):
                if i != r:
                    factor = T[i][c_p]
                    if abs(factor) > 1e-12:
                        for j in range(len(T[0])):
                            T[i][j] -= factor * T[r][j]
            basis[r] = c_p

        # Solve Phase 1
        while True:
            # Find entering variable
            c_p = -1
            min_val = -EPS
            for j in range(N + M):
                if T[M + 1][j] < min_val:
                    min_val = T[M + 1][j]
                    c_p = j
            if c_p == -1:
                break

            # Find leaving variable
            r = -1
            min_ratio = 1e18
            for i in range(M):
                if T[i][c_p] > EPS:
                    ratio = T[i][-1] / T[i][c_p]
                    if ratio < min_ratio - EPS:
                        min_ratio = ratio
                        r = i
            if r == -1:
                break
            pivot(r, c_p, M + 2)

        if T[M + 1][-1] < -EPS:
            return False, None, None  # Infeasible

        # Check if artificial variables are in basis
        for i in range(M):
            if basis[i] >= N:
                # Pivot it out
                for j in range(N):
                    if abs(T[i][j]) > EPS:
                        pivot(i, j, M + 1)
                        break

        # Phase 2
        # Update Phase 2 objective to be consistent with basis
        for i in range(M):
            col = basis[i]
            if col < N:
                factor = T[M][col]
                if abs(factor) > 1e-12:
                    for j in range(len(T[0])):
                        T[M][j] -= factor * T[i][j]

        while True:
            c_p = -1
            min_val = -EPS
            for j in range(N):
                if T[M][j] < min_val:
                    min_val = T[M][j]
                    c_p = j
            if c_p == -1:
                break

            r = -1
            min_ratio = 1e18
            for i in range(M):
                if T[i][c_p] > EPS:
                    ratio = T[i][-1] / T[i][c_p]
                    if ratio < min_ratio - EPS:
                        min_ratio = ratio
                        r = i
            if r == -1:
                # Unbounded, but feasible
                break
            pivot(r, c_p, M + 1)

        sol = [0.0] * N
        for i in range(M):
            if basis[i] < N:
                sol[basis[i]] = T[i][-1]
        return True, -T[M][-1], sol

    # Build base equations and inequalities
    # Standard form: A_eq x = b, x >= 0
    # Slack variables for inequalities: sum_i x_{i, j} + s_j = c_j (m slack variables)

    total_vars = num_vars + m
    eq_rows = []
    eq_rhs = []

    # 1. Capacity constraints with slacks:
    for j in range(m):
        row = [0.0] * total_vars
        for i in range(k):
            row[i * m + j] = 1.0
        row[num_vars + j] = 1.0  # slack
        eq_rows.append(row)
        eq_rhs.append(float(edges[j][2]))

    # 2. Flow conservation for each contract i:
    for i in range(k):
        s_i, t_i, d_i = contracts[i]
        for v in range(n):
            if v == t_i:
                continue  # Redundant equation
            row = [0.0] * total_vars
            for j in range(m):
                u_j, v_j, _ = edges[j]
                if u_j == v:
                    row[i * m + j] += 1.0
                if v_j == v:
                    row[i * m + j] -= 1.0
            req = 0.0
            if v == s_i:
                req = float(d_i)
            eq_rows.append(row)
            eq_rhs.append(req)

    # Cost vector (all 0 since it's a feasibility problem)
    cost = [0.0] * total_vars

    # Branch and Bound
    best_sol = None

    stack = [([], [])]  # list of (var_index, op, val) bounds: op <= or >=

    # Instead of full general B&B, add upper/lower bounds as extra slack/surplus equations
    def solve_lp(extra_bounds):
        # extra_bounds is list of (var_idx, '<=', val) or (var_idx, '>=', val)
        curr_A = [r[:] for r in eq_rows]
        curr_b = eq_rhs[:]
        cur_vars = total_vars

        for var_idx, op, val in extra_bounds:
            row = [0.0] * cur_vars
            row[var_idx] = 1.0
            if op == "<=":
                # x + s = val
                for r in curr_A:
                    r.append(0.0)
                row.append(1.0)
                cur_vars += 1
                curr_A.append(row)
                curr_b.append(float(val))
            else:
                # x - s = val
                for r in curr_A:
                    r.append(0.0)
                row.append(-1.0)
                cur_vars += 1
                curr_A.append(row)
                curr_b.append(float(val))

        cur_cost = [0.0] * cur_vars
        ok, _, sol = simplex(curr_A, curr_b, cur_cost)
        if not ok:
            return None
        return sol[:num_vars]

    # Use DFS for B&B
    stack = [[]]
    while stack:
        bounds = stack.pop()
        sol = solve_lp(bounds)
        if sol is None:
            continue

        # Check integrality
        frac_var = -1
        frac_val = 0.0
        max_dist = 0.0
        for i in range(num_vars):
            val = sol[i]
            dist = abs(val - round(val))
            if dist > 1e-4:
                if dist > max_dist:
                    max_dist = dist
                    frac_var = i
                    frac_val = val

        if frac_var == -1:
            best_sol = [int(round(x)) for x in sol]
            break

        import math

        floor_val = math.floor(frac_val)
        ceil_val = math.ceil(frac_val)

        # Branch
        stack.append(bounds + [(frac_var, ">=", ceil_val)])
        stack.append(bounds + [(frac_var, "<=", floor_val)])

    if best_sol is None:
        print("NO")
    else:
        print("YES")
        for i in range(k):
            print(*(best_sol[i * m + j] for j in range(m)))


if __name__ == "__main__":
    solve()
