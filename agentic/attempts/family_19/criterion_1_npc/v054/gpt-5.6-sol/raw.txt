import sys

def main():
    import numpy as np
    from scipy.optimize import milp, LinearConstraint, Bounds
    from scipy.sparse import coo_matrix

    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)
    k = next(it)

    streams = [(next(it) - 1, next(it) - 1, next(it)) for _ in range(k)]
    edges = [(next(it) - 1, next(it) - 1, next(it)) for _ in range(m)]

    if k == 0:
        print("YES")
        return

    if m == 0:
        feasible = all(d == 0 or s == t for s, t, d in streams)
        if not feasible:
            print("NO")
            return
        sys.stdout.write("YES\n" + "\n" * k)
        return

    variables = k * m
    rows = []
    cols = []
    values = []
    lower = []
    upper = []
    row = 0

    # Flow conservation constraints.
    for commodity, (s, t, demand) in enumerate(streams):
        for v in range(n):
            if v == t:
                continue

            for e, (u, w, _) in enumerate(edges):
                idx = commodity * m + e
                if u == v:
                    rows.append(row)
                    cols.append(idx)
                    values.append(1.0)
                if w == v:
                    rows.append(row)
                    cols.append(idx)
                    values.append(-1.0)

            required = float(demand if v == s and s != t else 0)
            lower.append(required)
            upper.append(required)
            row += 1

    # Shared link capacity constraints.
    for e, (_, _, capacity) in enumerate(edges):
        for commodity in range(k):
            rows.append(row)
            cols.append(commodity * m + e)
            values.append(1.0)
        lower.append(-np.inf)
        upper.append(float(capacity))
        row += 1

    matrix = coo_matrix(
        (values, (rows, cols)),
        shape=(row, variables),
        dtype=np.float64
    ).tocsc()

    capacities = np.array([cap for _, _, cap in edges], dtype=np.float64)
    variable_upper = np.tile(capacities, k)

    result = milp(
        c=np.zeros(variables, dtype=np.float64),
        integrality=np.ones(variables, dtype=np.uint8),
        bounds=Bounds(np.zeros(variables), variable_upper),
        constraints=LinearConstraint(
            matrix,
            np.array(lower, dtype=np.float64),
            np.array(upper, dtype=np.float64)
        ),
        options={"time_limit": 4.5, "presolve": True}
    )

    if result.x is None:
        print("NO")
        return

    solution = np.rint(result.x).astype(object)
    answer = [[int(solution[i * m + e]) for e in range(m)] for i in range(k)]

    # Verify the rounded integer solution exactly.
    edge_totals = [0] * m
    for i, (s, t, demand) in enumerate(streams):
        balance = [0] * n
        for e, (u, v, capacity) in enumerate(edges):
            flow = answer[i][e]
            if flow < 0:
                print("NO")
                return
            edge_totals[e] += flow
            balance[u] += flow
            balance[v] -= flow

        for v in range(n):
            expected = 0
            if s != t:
                if v == s:
                    expected = demand
                elif v == t:
                    expected = -demand
            if balance[v] != expected:
                print("NO")
                return

    for e, (_, _, capacity) in enumerate(edges):
        if edge_totals[e] > capacity:
            print("NO")
            return

    out = ["YES"]
    out.extend(" ".join(map(str, flows)) for flows in answer)
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
