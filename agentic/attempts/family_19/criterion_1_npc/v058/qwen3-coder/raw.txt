def solve():

        u, v, cap = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        edges.append((u, v, cap))
        if (u, v) not in edge_map:
            edge_map[(u, v)] = []
        edge_map[(u, v)].append(i)

    # We'll try to solve this using linear programming formulation
    # But since this is a coding competition, let's try a more direct approach
    
    # Let's model this as a min-cost flow problem for each commodity
    # But actually, we just need feasibility, so we can use a circulation approach
    
    # For simplicity, let's try to solve using a greedy approach with flow decomposition
    # Actually, let's formulate this properly
    
    # Variables: f[i][j] = flow of order i on edge j
    # Constraints:
    # 1. Flow conservation at intermediate nodes for each order
    # 2. Edge capacity constraints (sum over all orders)
    # 3. Supply/demand constraints at source/sink for each order
    
    # This is a multi-commodity flow problem. Let's solve it using LP approach conceptually
    # but implement a practical solution
    
    from collections import deque
    
    def bfs_flow(graph, source, sink, capacity, flow):
        """Find augmenting path using BFS"""
        parent = [-1] * len(graph)
        visited = [False] * len(graph)
        queue = deque([source])
        visited[source] = True
        
        while queue:
            u = queue.popleft()
            if u == sink:
                break
            for v in range(len(graph)):
                if not visited[v] and capacity[u][v] > flow[u][v]:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
        
        path = []
        if visited[sink]:
            v = sink
            while v != source:
                path.append((parent[v], v))
                v = parent[v]
            path.reverse()
        return path
    
    def max_flow(graph, source, sink, capacity):
        """Compute maximum flow from source to sink"""
        n_nodes = len(graph)
        flow = [[0] * n_nodes for _ in range(n_nodes)]
        
        while True:
            path = bfs_flow(graph, source, sink, capacity, flow)
            if not path:
                break
            
            # Find bottleneck capacity
            bottleneck = float('inf')
            for u, v in path:
                bottleneck = min(bottleneck, capacity[u][v] - flow[u][v])
            
            # Augment flow
            for u, v in path:
                flow[u][v] += bottleneck
                flow[v][u] -= bottleneck
        
        return flow
    
    # Try to solve for each order independently first, then check capacity constraints
    # But this won't work because of shared capacity constraints
    
    # Let's build a comprehensive flow model
    # For each order, create a flow network and decompose into paths
    
    def find_paths(source, target, graph, capacity, demand):
        """Find paths to route 'demand' units from source to target"""
        # Make a copy of capacity
        res_capacity = [row[:] for row in capacity]
        paths = []
        amounts = []
        
        while demand > 0:
            # Find a path using BFS
            parent = [-1] * len(graph)
            visited = [False] * len(graph)
            queue = deque([source])
            visited[source] = True
            found = False
            
            while queue:
                u = queue.popleft()
                if u == target:
                    found = True
                    break
                for v in range(len(graph)):
                    if not visited[v] and res_capacity[u][v] > 0:
                        visited[v] = True
                        parent[v] = u
                        queue.append(v)
            
            if not found:
                break
                
            # Reconstruct path
            path = []
            v = target
            while v != source:
                path.append(v)
                v = parent[v]
            path.append(source)
            path.reverse()
            
            # Find bottleneck
            bottleneck = demand
            for i in range(len(path) - 1):
                u, v = path[i], path[i+1]
                bottleneck = min(bottleneck, res_capacity[u][v])
            
            if bottleneck == 0:
                break
                
            # Update residual capacities
            for i in range(len(path) - 1):
                u, v = path[i], path[i+1]
                res_capacity[u][v] -= bottleneck
                res_capacity[v][u] += bottleneck  # Reverse edge
            
            paths.append(path)
            amounts.append(bottleneck)
            demand -= bottleneck
        
        return paths, amounts, demand == 0
    
    # Build adjacency matrix for the graph
    graph_capacity = [[0] * n for _ in range(n)]
    for u, v, cap in edges:
        graph_capacity[u][v] += cap  # Handle parallel edges by summing

    order_flows = []  # order_flows[i][j] = flow of order i on edge j

        res_graph = [[0] * n for _ in range(n)]

        order_edge_flow = [0] * m

        while remaining_demand > 0:
            # Find shortest path using BFS
            parent = [-1] * n

                u = queue.popleft()

                for v in range(n):
                    if not visited[v] and res_graph[u][v] > 0:
                        visited[v] = True
                        parent[v] = u
                        queue.append(v)
            
            if not path_found:

            # Reconstruct path
            path_nodes = []
            curr = target
            while curr != -1:
                path_nodes.append(curr)
                curr = parent[curr]
            path_nodes.reverse()
            
            # Convert to edge path and find minimum capacity
            path_edges = []

            for i in range(len(path_nodes) - 1):
                u, v = path_nodes[i], path_nodes[i+1]
                path_edges.append((u, v))
                min_cap = min(min_cap, res_graph[u][v])
            
            if min_cap <= 0:
                print("NO")
                return
            
            # Update flows and residual graph
            actual_flow = min(min_cap, remaining_demand)
            for u, v in path_edges:
                res_graph[u][v] -= actual_flow
                res_graph[v][u] += actual_flow  # Reverse edge
                
                # Update the actual edge usage
                for edge_idx in edge_to_indices[(u, v)]:
                    order_edge_flow[edge_idx] += actual_flow
                    current_edge_usage[edge_idx] += actual_flow
                    edge_remaining[edge_idx] -= actual_flow
                    break  # Take the first available edge
            
            remaining_demand -= actual_flow
        
        temp_flows.append(order_edge_flow)
    
    # Check if all capacity constraints are satisfied
    for i in range(m):
        if current_edge_usage[i] > edges[i][2]:  # capacity exceeded
            print("NO")
            return
    
    # Verify flow conservation for each order at intermediate nodes
    valid = True
    final_flows = []
    
    for order_idx, (source, target, demand) in enumerate(orders):
        flow = temp_flows[order_idx]
        final_flows.append(flow)
        
        # Check flow conservation
        node_in = [0] * n
        node_out = [0] * n
        
        for edge_idx, (u, v, _) in enumerate(edges):
            node_out[u] += flow[edge_idx]
            node_in[v] += flow[edge_idx]
        
        # Check source
        if node_out[source] - node_in[source] != demand:
            valid = False
            break
            
        # Check target
        if node_in[target] - node_out[target] != demand:
            valid = False
            break
            
        # Check intermediate nodes
        for node in range(n):
            if node != source and node != target:
                if node_in[node] != node_out[node]:
                    valid = False
                    break
    
    if not valid:
        print("NO")
        return
        
    print("YES")
    for flow in final_flows:
        print(' '.join(map(str, flow)))
