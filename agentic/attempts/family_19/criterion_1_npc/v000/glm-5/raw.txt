import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); m = int(data[idx+1]); k = int(data[idx+2]); idx += 3
    commodities = []
    for _ in range(k):
        s = int(data[idx]); t = int(data[idx+1]); d = int(data[idx+2]); idx += 3
        commodities.append((s, t, d))
    edges = []
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); cap = int(data[idx+2]); idx += 3
        edges.append((u, v, cap))

    # Build node list: source/sink per commodity, then edges
    N = k * n
    node = lambda c, x: c * n + (x - 1)
    M = k * m
    S = N
    T = N + 1
    total_nodes = N + 2

    head = [-1] * total_nodes
    to = [0] * (4 * M)
    cap = [0] * (4 * M)
    nxt = [-1] * (4 * M)
    edge_cnt = 0

    def add_edge(u, v, c):
        nonlocal edge_cnt
        to[edge_cnt] = v; cap[edge_cnt] = c; nxt[edge_cnt] = head[u]; head[u] = edge_cnt; edge_cnt += 1
        to[edge_cnt] = u; cap[edge_cnt] = 0; nxt[edge_cnt] = head[v]; head[v] = edge_cnt; edge_cnt += 1

    # commodity edges (parallel per commodity)
    for c in range(k):
        for j, (u, v, ccap) in enumerate(edges):
            add_edge(node(c, u), node(c, v), ccap)

    # source/sink
    total_demand = 0
    for c, (s, t, d) in enumerate(commodities):
        add_edge(S, node(c, s), d)
        add_edge(node(c, t), T, d)
        total_demand += d

    # Dinic
    level = [0] * total_nodes
    ptr = [0] * total_nodes

    def bfs():
        for i in range(total_nodes):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[T] >= 0

    def dfs(u, f):
        if u == T:
            return f
        while ptr[u] != -1:
            e = ptr[u]
            v = to[e]
            if cap[e] > 0 and level[v] == level[u] + 1:
                pushed = dfs(v, f if f < cap[e] else cap[e])
                if pushed > 0:
                    cap[e] -= pushed
                    cap[e ^ 1] += pushed
                    return pushed
            ptr[u] = nxt[e]
        return 0

    flow = 0
    while bfs():
        for i in range(total_nodes):
            ptr[i] = head[i]
        while True:
            pushed = dfs(S, 10**18)
            if pushed == 0:
                break
            flow += pushed

    if flow < total_demand:
        print("NO")
        return

    # Extract per-commodity per-edge flow
    ans = [[0] * m for _ in range(k)]
    for c in range(k):
        for j in range(m):
            e = 2 * (c * m + j) + 1  # reverse edge index
            ans[c][j] = cap[e]

    out = ["YES"]
    for c in range(k):
        out.append(" ".join(map(str, ans[c])))
    sys.stdout.write("\n".join(out) + "\n")

solve()
