import sys
import numpy as np
from scipy.optimize import linprog

def solve():
    input_data = sys.stdin.read().strip().split()
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    products = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        products.append((s, t, d))

        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        routes.append((u, v, cap))

    num_vars = k * m

    c = np.zeros(num_vars)

    # 1. Route capacity: sum_i x_{i,j} <= cap_j for each route j
    # 2. Flow conservation at intermediate nodes for each product
    # 3. Source/sink balance for each product

    b_ub = []
    
    # Route capacity constraints
    for j in range(m):
        row = np.zeros(num_vars)
