from collections import defaultdict, deque
import sys

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    k = int(data[idx])
    idx += 1
    
    resources = []
    for _ in range(k):
        s = int(data[idx]) - 1
        idx += 1
        t = int(data[idx]) - 1
        idx += 1
        d = int(data[idx])
        idx += 1
        resources.append((s, t, d))
    
    edges = []
    edge_dict = {}
    graph = defaultdict(list)
    for i in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        cap = int(data[idx])
        idx += 1
        edges.append((u, v, cap))
        edge_dict[(u, v)] = i
        graph[u].append((v, i))
    
    # Total flow for each edge
    total_flow = [0] * m
    individual_flows = []
    
    # Check if the total capacity constraints are satisfiable
    # For each edge, we need to ensure that sum of all individual flows <= capacity
    
    # We will use a greedy approach: for each resource, compute a flow (using any max-flow algorithm)
    # and then check if the sum of all flows on each edge is within capacity
    
    # To do this properly, we need to solve a multi-commodity flow problem
    # However, since this is a general graph, and the problem is NP-hard in general,
    # we will try to find a feasible solution by computing individual flows and checking feasibility
    
    # Let's try to compute flows for each commodity separately and see if we can combine them
    
    # For each resource, we compute a flow (we can use a simple max-flow like Edmonds-Karp)
    # and store the flow on each edge
    
    def bfs_flow(source, sink, capacity, edge_list, n_nodes):
        # Build adjacency list with capacities
        adj = defaultdict(lambda: defaultdict(int))
        edge_index = {}
        for u, v, cap in edge_list:
            adj[u][v] = cap
            edge_index[(u,v)] = cap
        
        # Edmonds-Karp
        parent = [-1] * n_nodes
        parent_edge = [None] * n_nodes
        
        def bfs():
            visited = [False] * n_nodes
            queue = deque()
            queue.append(source)
            visited[source] = True
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if not visited[v] and adj[u][v] > 0:
                        visited[v] = True
                        parent[v] = u
                        parent_edge[v] = (u, v)
                        if v == sink:
                            return True
                        queue.append(v)
            return False
        
        max_flow_value = 0
        flow_edges = defaultdict(int)
        
        while bfs():
            path_flow = float('inf')
            s = sink
            while s != source:
                u = parent[s]
                path_flow = min(path_flow, adj[u][s])
                s = parent[s]
            
            max_flow_value += path_flow
            
            v = sink
            while v != source:
                u = parent[v]
                adj[u][v] -= path_flow
                adj[v][u] += path_flow
                flow_edges[(u, v)] += path_flow
                v = parent[v]
        
        return flow_edges, max_flow_value
    
    # For each resource, compute a flow
    all_flows = []
    for s, t, d in resources:
        # Create a copy of edges with remaining capacity
        # Initially, remaining capacity is the full capacity minus total_flow
        temp_edges = []
        for i, (u, v, cap) in enumerate(edges):
            temp_edges.append((u, v, cap - total_flow[i]))
        
        # Compute flow for this commodity
        flow_dict, flow_value = bfs_flow(s, t, d, temp_edges, n)
        
        if flow_value < d:
            print("NO")
            return
            
        # Check if we can add this flow to the total
        new_total_flow = total_flow[:]
        for (u, v), f in flow_dict.items():
            if (u, v) in edge_dict:
                i = edge_dict[(u, v)]
                new_total_flow[i] += f
        
        # Check if any edge exceeds capacity
        valid = True
        for i in range(m):
            if new_total_flow[i] > edges[i][2]:
                valid = False
                break
        
        if not valid:
            print("NO")
            return
            
        total_flow = new_total_flow
        all_flows.append(flow_dict)
    
    # Now we need to output the flows for each commodity
    print("YES")
    for flow_dict in all_flows:
        flow_list = [0] * m
        for (u, v), f in flow_dict.items():
            if (u, v) in edge_dict:
                i = edge_dict[(u, v)]
                flow_list[i] = f
        print(' '.join(map(str, flow_list)))

solve()
