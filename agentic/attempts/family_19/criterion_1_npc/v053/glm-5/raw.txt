import sys

def solve():
    input_data = sys.stdin.read().split()
    it = iter(input_data)
    
    try:
        n = int(next(it))
        m = int(next(it))
        k = int(next(it))
    except StopIteration:
        return
    
    compounds = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        compounds.append((s - 1, t - 1, d))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        cap = int(next(it))
        edges.append((u - 1, v - 1, cap))
    
    # Create adjacency list for the graph
    adj = [[] for _ in range(n)]
    for idx, (u, v, cap) in enumerate(edges):
        adj[u].append((v, idx))
    
    # Find paths using DFS
    def dfs(start, end, path_edges, visited_edges):
        if start == end:
            return True
        for v, idx in adj[start]:
            if idx not in visited_edges:
                visited_edges.add(idx)
                path_edges.append(idx)
                if dfs(v, end, path_edges, visited_edges):
                    return True
                path_edges.pop()
        return False
    
    # Precompute paths for each compound
    paths = []
    for i in range(k):
        s, t, d = compounds[i]
        path_edges = []
        visited_edges = set()
        if dfs(s, t, path_edges, visited_edges):
            paths.append(path_edges)
        else:
            paths.append([])
    
    # Flow assignment
    flow = [[0] * m for _ in range(k)]
    remaining_cap = [cap for u, v, cap in edges]
    remaining_demands = [d for s, t, d in compounds]
    
    # Iterate to assign flows
    changed = True
    max_iter = k * m + 100
    iter_count = 0
    
    while changed and iter_count < max_iter:
        changed = False
        iter_count += 1
        
        for compound_idx in range(k):
            if remaining_demands[compound_idx] <= 0:
                continue
            
            path = paths[compound_idx]
            if not path:
                continue
            
            # Find bottleneck capacity
            min_cap = remaining_demands[compound_idx]
            for edge_idx in path:
                if remaining_cap[edge_idx] < min_cap:
                    min_cap = remaining_cap[edge_idx]
            
            if min_cap > 0:
                # Assign flow
                for edge_idx in path:
                    flow[compound_idx][edge_idx] += min_cap
                    remaining_cap[edge_idx] -= min_cap
                remaining_demands[compound_idx] -= min_cap
                changed = True
    
    # Check if all demands are satisfied
    possible = True
    for compound_idx in range(k):
        if remaining_demands[compound_idx] > 0:
            possible = False
            break
    
    if possible:
        print("YES")
        for compound_idx in range(k):
            print(" ".join(map(str, flow[compound_idx])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
