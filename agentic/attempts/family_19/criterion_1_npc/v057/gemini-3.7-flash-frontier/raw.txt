import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        k = int(next(iterator))
    except StopIteration:
        return

    batches = []
    for _ in range(k):
        s = int(next(iterator)) - 1
        t = int(next(iterator)) - 1
        d = int(next(iterator))
        batches.append((s, t, d))

    edges = []
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        cap = int(next(iterator))
        edges.append((u, v, cap))

    # Number of variables = k * m
    num_vars = k * m

    # Build constraints:
    # A_eq * x = b_eq
    # A_ub * x <= b_ub
    # x >= 0
    #
    # Flow conservation constraints:
    # For each batch i, for each node v in 0..n-2 (one node is redundant, but keeping all is fine):
    # sum_{e out of v} x_{i, e} - sum_{e into v} x_{i, e} = demand
    
    rows = []
    rhs = []
    eq_type = [] # 'E' for ==, 'L' for <=

    for i in range(k):
        s_i, t_i, d_i = batches[i]
        for v in range(n):
            row = [0.0] * num_vars
            for e_idx, (u, v_dest, cap) in enumerate(edges):
                if u == v:
                    row[i * m + e_idx] += 1.0
                if v_dest == v:
                    row[i * m + e_idx] -= 1.0
            
            target = 0.0
            if v == s_i:
                target += d_i
            if v == t_i:
                target -= d_i
            
            rows.append(row)
            rhs.append(target)
            eq_type.append('E')

    # Capacity constraints:
    for e_idx in range(m):
        row = [0.0] * num_vars
        for i in range(k):
            row[i * m + e_idx] = 1.0
        rows.append(row)
        rhs.append(float(edges[e_idx][2]))
        eq_type.append('L')

    # Convert to standard form Ax = b, x >= 0
    # We will maintain bounds on variables during Branch and Bound: lb <= x <= ub
    # In B&B, variables have bounds: x_j >= 0, and upper bounds <= cap_e, plus dynamic bounds.

    # Let's use a Simplex Solver that handles bounded variables:
    # Or convert all inequalities and bounds to standard LP form.
    
    # Let's implement a Simplex solver for LP:
    # Minimize 0 subject to:
    # sum a_{ij} x_j = b_i (from 'E')
    # sum a_{ij} x_j + s_i = b_i (from 'L')
    # x_j >= 0

    # For B&B, we can add bound constraints:
    # x_j <= U  => x_j + s = U
    # x_j >= L  => x_j - s = L
    
    def solve_lp(extra_bounds):
        # extra_bounds: list of (var_idx, '<=' or '>=', val)
        # Returns (status, x_val)
        # status: True if feasible, False if infeasible
        
        # Build standard form:
        # We have base constraints (rows, rhs, eq_type)
        # plus extra bounds
        
        # Variables: x_0 .. x_{num_vars - 1}
        # Plus slack variables
        A = []
        b = []
        
        for r, val, tp in zip(rows, rhs, eq_type):
            A.append(list(r))
            b.append(val)
        
        for v_idx, tp, val in extra_bounds:
            r = [0.0] * num_vars
            r[v_idx] = 1.0
            A.append(r)
            b.append(val)
            
        num_constrs = len(A)
        # Determine slack variables needed
        # Base 'E' constraints: no slack
        # Base 'L' constraints: +1 slack
        # Extra '<=': +1 slack
        # Extra '>=': -1 slack
        
        # Let's count slacks
        col_types = [] # +1 for +slack, -1 for -slack, 0 for none
        for tp in eq_type:
            if tp == 'L':
                col_types.append(1)
            else:
                col_types.append(0)
        for _, tp, _ in extra_bounds:
            if tp == '<=':
                col_types.append(1)
            else:
                col_types.append(-1)
                
        total_cols = num_vars
        for ct in col_types:
            if ct != 0:
                total_cols += 1
                
        full_A = [[0.0] * total_cols for _ in range(num_constrs)]
        slack_idx = num_vars
        for i in range(num_constrs):
            for j in range(num_vars):
                full_A[i][j] = A[i][j]
            if col_types[i] == 1:
                full_A[i][slack_idx] = 1.0
                slack_idx += 1
            elif col_types[i] == -1:
                full_A[i][slack_idx] = -1.0
                slack_idx += 1
                
        # Make all b >= 0
        for i in range(num_constrs):
            if b[i] < -1e-9:
                b[i] = -b[i]
                for j in range(total_cols):
                    full_A[i][j] = -full_A[i][j]
            elif abs(b[i]) <= 1e-9:
                b[i] = 0.0

        # Phase 1: Add artificial variables
        # Artificial variable for each row unless it has a slack with coeff +1 and b >= 0
        art_vars = []
        art_count = 0
        slack_pos = num_vars
        basis = [-1] * num_constrs
        
        for i in range(num_constrs):
            if col_types[i] == 1 and full_A[i][slack_pos] == 1.0:
                basis[i] = slack_pos
                slack_pos += 1
            else:
                if col_types[i] != 0:
                    slack_pos += 1
                art_vars.append(i)
                art_count += 1

        p1_cols = total_cols + art_count
        tableau = [[0.0] * (p1_cols + 1) for _ in range(num_constrs + 1)]
        
        for i in range(num_constrs):
            for j in range(total_cols):
                tableau[i][j] = full_A[i][j]
            tableau[i][-1] = b[i]
            
        cur_art = total_cols
        for i, row_idx in enumerate(art_vars):
            tableau[row_idx][cur_art] = 1.0
            basis[row_idx] = cur_art
            cur_art += 1
            
        # Objective row for Phase 1: sum of artificials
        # Initially c = [0 ... 1 ... 1] for artificials
        # Eliminate basic artificials from objective row:
        for j in range(total_cols):
            tableau[-1][j] = 0.0
        for j in range(total_cols, p1_cols):
            tableau[-1][j] = 1.0
        tableau[-1][-1] = 0.0
        
        for i in art_vars:
            for j in range(p1_cols + 1):
                tableau[-1][j] -= tableau[i][j]
                
        # Phase 1 Simplex
        EPS = 1e-8
        
        def pivot(r, c, cols_count):
            inv_p = 1.0 / tableau[r][c]
            for j in range(cols_count + 1):
                tableau[r][j] *= inv_p
            tableau[r][c] = 1.0
            for i in range(num_constrs + 1):
                if i != r:
                    factor = tableau[i][c]
                    if abs(factor) > EPS:
                        for j in range(cols_count + 1):
                            tableau[i][j] -= factor * tableau[r][j]
                        tableau[i][c] = 0.0
            basis[r] = c

        while True:
            # Find entering variable (most negative in bottom row)
            min_val = -EPS
            pivot_col = -1
            for j in range(p1_cols):
                if tableau[-1][j] < min_val:
                    min_val = tableau[-1][j]
                    pivot_col = j
            if pivot_col == -1:
                break
                
            # Find leaving variable
            min_ratio = 1e18
            pivot_row = -1
            for i in range(num_constrs):
                if tableau[i][pivot_col] > EPS:
                    ratio = tableau[i][-1] / tableau[i][pivot_col]
                    if ratio < min_ratio - EPS:
                        min_ratio = ratio
                        pivot_row = i
            if pivot_row == -1:
                # Unbounded in Phase 1
                break
            pivot(pivot_row, pivot_col, p1_cols)
            
        if tableau[-1][-1] < -1e-5:
            return False, None
            
        # Drive any remaining artificial variables out of the basis
        for i in range(num_constrs):
            if basis[i] >= total_cols:
                # Need to pivot out
                pivoted = False
                for j in range(total_cols):
                    if abs(tableau[i][j]) > EPS:
                        pivot(i, j, total_cols)
                        pivoted = True
                        break
                if not pivoted:
                    # Redundant constraint, ignore
                    pass

        # Extract solution
        sol = [0.0] * num_vars
        for i in range(num_constrs):
            if basis[i] < num_vars:
                sol[basis[i]] = tableau[i][-1]
                
        return True, sol

    # Branch and Bound search
    solution_found = None

    def branch_and_bound(bounds):
        nonlocal solution_found
        if solution_found is not None:
            return
            
        feasible, sol = solve_lp(bounds)
        if not feasible:
            return
            
        # Check if integer
        frac_idx = -1
        max_dist = 0.0
        for j in range(num_vars):
            val = sol[j]
            dist = abs(val - round(val))
            if dist > 1e-4:
                if dist > max_dist:
                    max_dist = dist
                    frac_idx = j
                    
        if frac_idx == -1:
            solution_found = [int(round(x)) for x in sol]
            return
            
        # Branch on frac_idx
        v_val = sol[frac_idx]
        floor_val = int(v_val)
        ceil_val = floor_val + 1
        
        # Try branch closest to rounded value first
        if v_val - floor_val < 0.5:
            branch_and_bound(bounds + [(frac_idx, '<=', float(floor_val))])
            if solution_found is not None: return
            branch_and_bound(bounds + [(frac_idx, '>=', float(ceil_val))])
        else:
            branch_and_bound(bounds + [(frac_idx, '>=', float(ceil_val))])
            if solution_found is not None: return
            branch_and_bound(bounds + [(frac_idx, '<=', float(floor_val))])

    branch_and_bound([])
    
    if solution_found is None:
        print("NO")
    else:
        print("YES")
        for i in range(k):
            row_res = solution_found[i * m : (i + 1) * m]
            print(*(max(0, x) for x in row_res))

if __name__ == '__main__':
    solve()
