                        result[e[3]] = self.graph[e[0]][e[2]][1]
            return result
    
    best_result = None
    
    orderings = [list(range(k)), list(range(k))[::-1]]
    orderings.append(sorted(range(k), key=lambda i: -rituals[i][2]))
    orderings.append(sorted(range(k), key=lambda i: rituals[i][2]))
    
    for _ in range(200):
        perm = list(range(k))
        random.shuffle(perm)
        orderings.append(perm)
    
    for order in orderings:
        remaining_caps = [e[2] for e in edges]
        all_flows = [[0] * m for _ in range(k)]
        success = True
        
        for ri in order:
            s, t, d = rituals[ri]
            if d == 0:
                continue
            if s == t:
                success = False
                break
            
            dinic = Dinic(n)
            for i in range(m):
                u, v, _ = edges[i]
                if remaining_caps[i] > 0:
                    dinic.add_edge(u, v, remaining_caps[i], i)
            
            flow = dinic.max_flow(s, t, d)
            if flow < d:
                success = False
                break
            
            edge_flows = dinic.get_edge_flows()
            for i in range(m):
                f = edge_flows.get(i, 0)
                all_flows[ri][i] = f
                remaining_caps[i] -= f
        
        if success:
            best_result = all_flows
            break
    
    if best_result is not None:
        print("YES")
        for row in best_result:
            print(' '.join(map(str, row)))
    else:
        print("NO")
