```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    classes = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        classes.append((s, t, d))
    
    edges = []
    adj = [[] for _ in range(n)]
    for i in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Build the super source / super sink flow network
    # Nodes: 0..n-1 (original), n (super source), n+1 (super sink)
    N = n + 2
    S = n
    T = n + 1
    
    # Dinic's algorithm
    # Edge structure: [to, cap, flow, rev_index]
    graph = [[] for _ in range(N)]
    
    def add_edge(fr, to, cap):
        graph[fr].append([to, cap, 0, len(graph[to])])
        graph[to].append([fr, 0, 0, len(graph[fr]) - 1])
    
    # Add edges from super source to intake offices
    for i, (s, t, d) in enumerate(classes):
        add_edge(S, s, d)
    
    # Add edges from archive offices to super sink
    for i, (s, t, d) in enumerate(classes):
        add_edge(t, T, d)
    
    # Add original edges
    for i, (u, v, cap) in enumerate(edges):
        add_edge(u, v, cap)
    
    level = [0] * N
    it_ptr = [0] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        level[S] = 0
        q = deque([S])
        while q:
            v = q.popleft()
            for e in graph[v]:
                to, cap, flow, _ = e
                if cap - flow > 0 and level[to] < 0:
                    level[to] = level[v] + 1
                    q.append(to)
        return level[T] >= 0
    
    def dfs(v, f):
        if v == T:
            return f
        while it_ptr[v] < len(graph[v]):
            e = graph[v][it_ptr[v]]
            to, cap, flow, rev = e
            if cap - flow > 0 and level[v] < level[to]:
                d = dfs(to, min(f, cap - flow))
                if d > 0:
                    e[2] += d
                    graph[to][rev][2] -= d
                    return d
            it_ptr[v] += 1
        return 0
    
    def max_flow():
        flow = 0
        while bfs():
            for i in range(N):
                it_ptr[i] = 0
            while True:
                f = dfs(S, float('inf'))
                if f == 0:
                    break
                flow += f
        return flow
    
    total_demand = sum(d for s, t, d in classes)
    flow = max_flow()
    
    if flow < total_demand:
        print("NO")
        return
    
    # Now we need to decompose the flow for each class
    # We use the flow on edges from super source to intake to determine how much each class sends
    # But since multiple classes can share the same intake, we need to track flows per class
    
    # We'll do a flow decomposition for each class
    # First, let's find the flow on each original edge (total across all classes)
    # The original edges are added after the super source/sink edges
    # For each original edge i, it's at index 2*i and 2*i+1 in the graph (but we need to be careful)
    
    # Actually, let's just track the flow per class by doing a DFS for each class
    # We'll use the residual graph and for each class, push flow from its intake to its archive
    
    # But we need to know how much flow each class sends on each edge
    # Let's use a greedy approach: for each class, find paths from intake to archive in the residual graph
    
    # First, let's get the flow on each edge
    # The edges from S to intakes tell us how much each class sends (but we need to know which class)
    # Since we added edges in order, edge from S to classes[i].s is at index i in graph[S]
    # But multiple classes can have the same s, so we need to be careful
    
    # Let's instead do a flow decomposition per class
    # We'll track for each class, the amount of flow on each original edge
    
    # We'll use the following approach:
    # For each class, we know the demand d. We'll find d units of flow from s to t in the residual graph
    # But we need to be careful not to use flow that belongs to other classes
    
    # Actually, since we're decomposing the total flow, we can just find any path from s to t
    # and assign it to the class. The total flow is valid, so we can decompose it.
    
    # Let's build an adjacency list of the flow graph (only edges with flow > 0)
    # For original edges, the flow is in graph[u][2*i] (the forward edge)
    
    # Let's first compute the flow on each original edge
    edge_flow = [0] * m
    for i, (u, v, cap) in enumerate(edges):
        # The forward edge is at index 2*i + 2*k in graph[u] (since we added 2*k edges for super source/sink first)
        # Wait, let's recalculate
        # We added edges in this order:
        # 1. For each class: edge from S to s (k edges)
        # 2. For each class: edge from t to T (k edges)
        # 3. For each original edge: edge from u to v (m edges)
        # So in graph[u], the original edges start at index 2*k (if u is an intake or archive)
        # But actually, the edges are added to graph[u] in the order they appear
        # So for original edge i, it's the (2*k + i)-th edge added to graph[u] (if u is the source of that edge)
        # But we also added edges from S to s and from t to T, which affect graph[S] and graph[t]
        # So for a node u, the edges in graph[u] are:
        # - reverse edges of (S -> u) if u is an intake (but these are added to graph[u] when we add the edge S->u)
        # - reverse edges of (u -> T) if u is an archive
        # - forward edges of original edges starting at u
        # - reverse edges of original edges ending at u
        
        # This is getting complicated. Let's just search for the edge in graph[u] that goes to v
        for e in graph[u]:
            if e[0] == v and e[1] > 0:  # forward edge (cap > 0)
                edge_flow[i] = e[2]
                break
    
    # Now we need to decompose this flow per class
    # We'll use a greedy approach: for each class, find paths from s to t and assign flow
    
    # Build a flow graph (only edges with flow > 0)
    flow_adj = [[] for _ in range(n)]
    for i, (u, v, cap) in enumerate(edges):
        if edge_flow[i] > 0:
            flow_adj[u].append((v, i, edge_flow[i]))
    
    # For each class, we need to route d units from s to t
    # We'll use a DFS to find paths and assign flow
    class_edge_flow = [[0] * m for _ in range(k)]
    
    # We need to track remaining flow on each edge
    remaining_flow = edge_flow[:]
    
    # For each class, find paths from s to t
    for class_idx in range(k):
        s, t, d = classes[class_idx]
        remaining = d
        while remaining > 0:
            # Find a path from s to t using edges with remaining_flow > 0
            # We'll use DFS with path tracking
            parent = [-1] * n
            parent_edge = [-1] * n
            visited = [False] * n
            stack = [s]
            visited[s] = True
            found = False
            while stack:
                u = stack.pop()
                if u == t:
                    found = True
                    break
                for v, edge_idx, cap in flow_adj[u]:
                    if not visited[v] and remaining_flow[edge_idx] > 0:
                        visited[v] = True
                        parent[v] = u
                        parent_edge[v] = edge_idx
                        stack.append(v)
            if not found:
                print("NO")
                return
            # Reconstruct path and find min flow
            path_edges = []
            path_nodes = [t]
            curr = t
            while curr != s:
                edge_idx = parent_edge[curr]
                path_edges.append(edge_idx)
                curr = parent[curr]
                path_nodes.append(curr)
            path_edges.reverse()
            path_nodes.reverse()
            
            # Find min flow on path
            min_flow = min(remaining_flow[e] for e in path_edges)
            min_flow = min(min_flow, remaining)
            
            # Assign flow
            for e in path_edges:
                remaining_flow[e] -= min_flow
                class_edge_flow[class_idx][e] += min_flow
            remaining -= min_flow
    
    print("YES")
    for i in range(k):
        print(' '.join(map(str, class_edge_flow[i])))

solve()
```
