from collections import defaultdict, deque
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    k = int(data[idx]); idx += 1
    
    # Read classifications
    classifications = []
    for i in range(k):
        s = int(data[idx]) - 1; idx += 1  # Convert to 0-based
        t = int(data[idx]) - 1; idx += 1
        d = int(data[idx]); idx += 1
        classifications.append((s, t, d))

    edges = []
    edge_dict = {}
    graph = [[] for _ in range(n)]
    for i in range(m):
        u = int(data[idx]) - 1; idx += 1  # Convert to 0-based
        v = int(data[idx]) - 1; idx += 1

        edges.append((u, v, cap))
        edge_dict[(u, v)] = i

        # But we need to consider that previous classifications have already used some capacity

        # However, since we want to ensure that the SUM of all classifications on each edge <= cap,
        # we should instead model this as a multi-commodity flow problem, which is complex.
        # But the problem asks us to find a solution if exists, so we can try to solve each 
        # classification one by one, ensuring that we don't exceed edge capacities.
        
        # Let's think differently: we can model this as a min-cost max-flow problem or sequence of max-flows
        # but we need to make sure that the total flow on each edge doesn't exceed its capacity.
        # A simpler approach: for each commodity, find a flow of value d, and check if total <= cap.
        # But flows might interfere. Better way: solve all together.
        
        # Actually, let's solve each commodity separately, but keep track of how much flow is already assigned
        # to each edge. When solving a new commodity, the residual capacity of an edge is 
        # cap - total_flow_on_that_edge_so_far.
        
        # For this classification (s,t,d), build a flow network with current residual capacities
        # and find a flow of value d.
        
        # Let's do max-flow for this commodity with current residual capacities
        # Residual capacity on edge j is edges[j][2] - total_flow_per_edge[j]
        
        # But to reuse code, let's define a function for max-flow
        pass
    
    # Let's restart with a cleaner approach:
    # For each classification, we need to send 'd' units of flow from s to t
    # Edges have total capacity, shared among all classifications
    # This is a multi-source multi-sink max flow with edge capacity sharing
    # We can model this as a single min-cost max-flow problem or use iterative rounding
    # But perhaps easier: solve sequentially, updating residual capacities
    
    # But actually, we can solve this by building a larger graph:
    # Create a super-source connected to each s_i with capacity d_i
    # Connect each t_i to a super-sink with capacity d_i
    # Original edges have capacity cap_j
    # Then find a max-flow of value sum(d_i)
    # If achievable, decompose the flow into paths for each commodity
    
    # Let's implement this approach
    
    # Build extended graph
    # Nodes: 0..n-1 original, n super-source, n+1 super-sink
    super_source = n
    super_sink = n + 1
    N = n + 2
    
    # Edge list for the extended graph
    ext_edges = []
    ext_edge_index = {}  # map (u,v) to index in ext_edges
    ext_graph = [[] for _ in range(N)]
    
    # Add edges from super_source to each s_i with capacity d_i
    for i, (s, t, d) in enumerate(classifications):
        ext_edges.append((super_source, s, d))
        ext_edge_index[(super_source, s)] = len(ext_edges) - 1
        ext_graph[super_source].append(len(ext_edges) - 1)
    
    # Add edges from each t_i to super_sink with capacity d_i
    for i, (s, t, d) in enumerate(classifications):
        ext_edges.append((t, super_sink, d))
        ext_edge_index[(t, super_sink)] = len(ext_edges) - 1
        ext_graph[t].append(len(ext_edges) - 1)
    
    # Add original edges with their capacities
    for u, v, cap in edges:
        ext_edges.append((u, v, cap))
        ext_edge_index[(u, v)] = len(ext_edges) - 1
        ext_graph[u].append(len(ext_edges) - 1)
    
    E = len(ext_edges)
    
    # Now run max-flow from super_source to super_sink
    def bfs():
        parent = [-1] * N
        parent_edge = [-1] * E
        queue = deque()
        queue.append(super_source)
        parent[super_source] = -2  # Visited marker
        
        while queue:
            u = queue.popleft()
            for e_idx in ext_graph[u]:
                u_e, v_e, cap_e = ext_edges[e_idx]
                if u_e == u and cap_e > 0 and parent[v_e] == -1:
                    parent[v_e] = u
                    parent_edge[v_e] = e_idx
                    if v_e == super_sink:
                        # Found augmenting path
                        path_flow = cap_e
                        v = v_e
                        while v != super_source:
                            u_prev = parent[v]
                            e_prev = parent_edge[v]
                            path_flow = min(path_flow, ext_edges[e_prev][2])
                            v = u_prev
                        return path_flow, parent_edge
                    queue.append(v_e)
        return 0, None
    
    # Max flow algorithm
    max_flow_value = 0
    while True:
        path_flow, parent_edge = bfs()
        if path_flow == 0:
            break
        max_flow_value += path_flow
        # Update residual capacities
        v = super_sink
        while v != super_source:
            u = parent[v]
            e_idx = parent_edge[v]
            ext_edges[e_idx] = (ext_edges[e_idx][0], ext_edges[e_idx][1], ext_edges[e_idx][2] - path_flow)
            # Add reverse edge if not exists
            rev_key = (ext_edges[e_idx][1], ext_edges[e_idx][0])
            if rev_key in ext_edge_index:
                rev_idx = ext_edge_index[rev_key]
                ext_edges[rev_idx] = (ext_edges[rev_idx][0], ext_edges[rev_idx][1], ext_edges[rev_idx][2] + path_flow)
            else:
                # Create reverse edge
                ext_edges.append((ext_edges[e_idx][1], ext_edges[e_idx][0], path_flow))
                new_idx = len(ext_edges) - 1
                ext_edge_index[rev_key] = new_idx
                ext_graph[ext_edges[new_idx][0]].append(new_idx)
            v = u
    
    # Check if max flow equals total demand
    total_demand = sum(d for _, _, d in classifications)
    if max_flow_value < total_demand:
        print("NO")
        return
    
    # Flow decomposition: trace back the flow for each commodity
    # For each classification, we need to extract d units of flow from s to t
    
    # Reconstruct flows per commodity
    # We'll re-run flow for each commodity using remaining capacities
    
    # Reset total_flow_per_edge
    total_flow_per_edge = [0] * m
    individual_flows = []
    
    success = True
    
    for s, t, d in classifications:
        # Build flow network for this commodity with current residual capacities
        # Nodes 0..n-1
        # Edges with residual capacity cap_j - total_flow_per_edge[j]
        res_cap = [edges[j][2] - total_flow_per_edge[j] for j in range(m)]
        
        # Run max-flow from s to t with value d
        # Build local graph
        local_graph = [[] for _ in range(n)]
        local_edges = []  # (u, v, cap, orig_index)
        local_index_map = {}  # (u,v) -> index in local_edges
        
        for j, (u, v, _) in enumerate(edges):
            if res_cap[j] > 0:
                local_edges.append((u, v, res_cap[j], j))
                local_index_map[(u, v)] = len(local_edges) - 1
                local_graph[u].append(len(local_edges) - 1)
        
        L = len(local_edges)
        
        def local_bfs():
            p = [-1] * n
            pe = [-1] * L
            q = deque()
            q.append(s)
            p[s] = -2
            
            while q:
                u = q.popleft()
                for e_idx in local_graph[u]:
                    ue, ve, ce, _ = local_edges[e_idx]
                    if ue == u and ce > 0 and p[ve] == -1:
                        p[ve] = u
                        pe[ve] = e_idx
                        if ve == t:
                            pf = ce
                            v_temp = ve
                            while v_temp != s:
                                u_prev = p[v_temp]
                                e_prev = pe[v_temp]
                                pf = min(pf, local_edges[e_prev][2])
                                v_temp = u_prev
                            return pf, pe
                        q.append(ve)
            return 0, None
        
        # Find flow of value d
        flow_sent = 0
        flow_on_local_edges = [0] * L
        
        while flow_sent < d:
            pf, pe = local_bfs()
            if pf == 0:
                break
            pf = min(pf, d - flow_sent)
            flow_sent += pf
            # Update residual
            v_temp = t
            while v_temp != s:
                u_prev = p[v_temp]
                e_prev = pe[v_temp]
                local_edges[e_prev] = (local_edges[e_prev][0], local_edges[e_prev][1], local_edges[e_prev][2] - pf, local_edges[e_prev][3])
                flow_on_local_edges[e_prev] += pf
                # Handle reverse edge
                rev_key = (local_edges[e_prev][1], local_edges[e_prev][0])
                if rev_key in local_index_map:
                    rev_idx = local_index_map[rev_key]
                    local_edges[rev_idx] = (local_edges[rev_idx][0], local_edges[rev_idx][1], local_edges[rev_idx][2] + pf, local_edges[rev_idx][3])
                else:
                    local_edges.append((local_edges[e_prev][1], local_edges[e_prev][0], pf, local_edges[e_prev][3]))
                    new_idx = len(local_edges) - 1
                    local_index_map[rev_key] = new_idx
                    local_graph[local_edges[new_idx][0]].append(new_idx)
                v_temp = u_prev
        
        if flow_sent < d:
            success = False
            break
        
        # Map local flow to global
        flow_for_this_commodity = [0] * m
        for i in range(L):
            _, _, _, orig_idx = local_edges[i]
            flow_for_this_commodity[orig_idx] += flow_on_local_edges[i]
        
        individual_flows.append(flow_for_this_commodity)
        # Update total_flow_per_edge
        for j in range(m):
            total_flow_per_edge[j] += flow_for_this_commodity[j]
    
    if not success:
        print("NO")
        return
    
    print("YES")
    for flow_list in individual_flows:
        print(' '.join(map(str, flow_list)))

if __name__ == "__main__":
    main()
