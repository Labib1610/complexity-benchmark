```python
import sys
from fractions import Fraction

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    commodities = []
    for _ in range(k):
        s = int(next(iterator)) - 1
        t = int(next(iterator)) - 1
        d = int(next(iterator))
        commodities.append((s, t, d))
        
    edges = []
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        cap = int(next(iterator))
        edges.append((u, v, cap))
        
    # Variables: x_{i, j} for commodity i, edge j. Total = k * m variables.
    # We want to find an integer feasible solution satisfying:
    # 1. Flow conservation for each commodity i and node u:
    #    sum_{j: u_j = u} x_{i, j} - sum_{j: v_j = u} x_{i, j} = b_{i, u}
    # 2. Capacity for each edge j:
    #    sum_{i} x_{i, j} <= cap_j
    # 3. x_{i, j} >= 0 integer.

    num_vars = k * m
    def var_idx(c, e):
        return c * m + e

    # Build base equality constraints: A_eq * x = b_eq
    # and inequality constraints: A_ub * x <= b_ub
    A_eq = []
    b_eq = []
    
    for c in range(k):
        s_c, t_c, d_c = commodities[c]
        for u in range(n):
            # net out-flow
            req = 0
            if u == s_c:
                req += d_c
            if u == t_c:
                req -= d_c
            
            row = [0.0] * num_vars
            for e in range(m):
                u_e, v_e, _ = edges[e]
                if u_e == u:
                    row[var_idx(c, e)] += 1.0
                if v_e == u:
                    row[var_idx(c, e)] -= 1.0
            
            A_eq.append(row)
            b_eq.append(float(req))

    A_ub = []
    b_ub = []
    for e in range(m):
        row = [0.0] * num_vars
        for c in range(k):
            row[var_idx(c, e)] = 1.0
        A_ub.append(row)
        b_ub.append(float(edges[e][2]))

    # Simplex solver using standard 2-phase tableau
    # To handle equality and inequality constraints + bounds
    def solve_lp(extra_bounds):
        # extra_bounds: list of (var_idx, '<=' or '>=', val)
        cur_A_eq = [row[:] for row in A_eq]
        cur_b_eq = b_eq[:]
        
        cur_A_ub = [row[:] for row in A_ub]
        cur_b_ub = b_ub[:]
        
        for v_idx, op, val in extra_bounds:
            row = [0.0] * num_vars
            row[v_idx] = 1.0
            if op == '<=':
                cur_A_ub.append(row)
                cur_b_ub.append(float(val))
            else: # '>='
                row[v_idx] = -1.0
                cur_A_ub.append(row)
                cur_b_ub.append(-float(val))
                
        # Convert all to standard form: A x = b, x >= 0
        # For each inequality row in A_ub: add slack variable s_i >= 0
        # If RHS is negative, multiply row by -1 (for eq, RHS must be >= 0)
        
        num_slacks = len(cur_A_ub)
        total_cols = num_vars + num_slacks
        
        mat = []
        rhs = []
        
        # Add equality constraints
        for r in range(len(cur_A_eq)):
            row = cur_A_eq[r] + [0.0] * num_slacks
            b_val = cur_b_eq[r]
            if b_val < -1e-9:
                row = [-x for x in row]
                b_val = -b_val
            mat.append(row)
            rhs.append(b_val)
            
        # Add inequality constraints with slack
        for r in range(num_slacks):
            row = cur_A_ub[r] + [0.0] * num_slacks
            row[num_vars + r] = 1.0
            b_val = cur_b_ub[r]
            if b_val < -1e-9:
                row = [-x for x in row]
                b_val = -b_val
            mat.append(row)
            rhs.append(b_val)
            
        num_rows = len(mat)
        
        # Phase 1: minimize sum of artificial variables
        # Tableau:
        # Rows: 0 .. num_rows-1 (constraints)
        # Row num_rows: Phase 1 objective
        # Columns: total_cols (vars) + num_rows (artificials) + 1 (RHS)
        
        tableau = [[0.0] * (total_cols + num_rows + 1) for _ in range(num_rows + 1)]
        basis = [-1] * num_rows
        
        for i in range(num_rows):
            for j in range(total_cols):
                tableau[i][j] = mat[i][j]
            tableau[i][total_cols + i] = 1.0
            tableau[i][-1] = rhs[i]
            basis[i] = total_cols + i
            
        # Phase 1 objective: minimize sum of artificials
        # Initially, obj = sum of artificials = sum of (rhs[i] - sum_j mat[i][j]*x_j)
        # So in tableau row num_rows:
        for i in range(num_rows):
            for j in range(total_cols + 1):
                col_idx = j if j < total_cols else -1
                tableau[num_rows][col_idx] -= tableau[i][col_idx]

        # Simplex pivot loop
        def pivot(r, c):
            pivot_val = tableau[r][c]
            for j in range(len(tableau[0])):
                tableau[r][j] /= pivot_val
            for i in range(num_rows + 1):
                if i != r:
                    factor = tableau[i][c]
                    if abs(factor) > 1e-12:
                        for j in range(len(tableau[0])):
                            tableau[i][j] -= factor * tableau[r][j]
            basis[r] = c

        # Solve Phase 1
        while True:
            # find min negative in obj row (among all vars except RHS and Phase 1 artificials if we want)
            min_val = -1e-9
            pivot_col = -1
            for j in range(total_cols + num_rows):
                if tableau[num_rows][j] < min_val:
                    min_val = tableau[num_rows][j]
                    pivot_col = j
            if pivot_col == -1:
                break
                
            # Ratio test
            min_ratio = float('inf')
            pivot_row = -1
            for i in range(num_rows):
                if tableau[i][pivot_col] > 1e-9:
                    ratio = tableau[i][-1] / tableau[i][pivot_col]
                    if ratio < min_ratio - 1e-11:
                        min_ratio = ratio
                        pivot_row = i
            if pivot_row == -1:
                # Unbounded Phase 1 (shouldn't happen)
                break
            pivot(pivot_row, pivot_col)
            
        if abs(tableau[num_rows][-1]) > 1e-6:
            # Infeasible
            return None
            
        # Extract solution
        sol = [0.0] * num_vars
        for i in range(num_rows):
            if basis[i] < num_vars:
                sol[basis[i]] = max(0.0, tableau[i][-1])
                
        return sol

    # Branch and Bound DFS
    stack = [[]]
    
    while stack:
        bounds = stack.pop()
        sol = solve_lp(bounds)
        if sol is None:
            continue
            
        # Check if integer
        first_fractional = -1
        max_frac_dist = 0
        for i in range(num_vars):
            val = sol[i]
            nearest = round(val)
            dist = abs(val - nearest)
            if dist > 1e-4:
                if dist > max_frac_dist:
                    max_frac_dist = dist
                    first_fractional = i
                    
        if first_fractional == -1:
            # Found integer solution!
            print("YES")
            for c in range(k):
                row_res = []
                for e in range(m):
                    val = int(round(sol[var_idx(c, e)]))
                    row_res.append(str(val))
                print(*(row_res))
            return

        # Branch on first_fractional
        val = sol[first_fractional]
        floor_val = int(val)
        ceil_val = floor_val + 1
        
        # Heuristic: push the one closer to fractional part first to explore it second, or vice versa
        stack.append(bounds + [(first_fractional, '>=', ceil_val)])
        stack.append(bounds + [(first_fractional, '<=', floor_val)])
        
    print("NO")

if __name__ == '__main__':
    solve()
```
