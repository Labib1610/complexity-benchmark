```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    requests = []
    for _ in range(k):
        s = int(next(iterator)) - 1
        t = int(next(iterator)) - 1
        d = int(next(iterator))
        requests.append((s, t, d))
        
    edges = []
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        cap = int(next(iterator))
        edges.append((u, v, cap))

    # We need to find non-negative integers x_{i, j} for i in range(k), j in range(m)
    # Total variables: num_vars = k * m
    # Constraints:
    # 1. Flow conservation for each commodity i, node u:
    #    sum_{j: u->v} x_{i, j} - sum_{j: v->u} x_{i, j} = balance(i, u)
    #    where balance(i, s_i) = d_i, balance(i, t_i) = -d_i, else 0.
    #    (Note: we can drop one node per commodity since equations are redundant)
    # 2. Capacity constraints for each edge j:
    #    sum_{i} x_{i, j} <= cap_j
    # 3. x_{i, j} >= 0 and integer.

    num_vars = k * m

    # Index helper
    def var_idx(req_i, edge_j):
        return req_i * m + edge_j

    # A * x <= b (and A_eq * x == b_eq)
    # We will formulate for Simplex:
    # We can write an ILP solver using Branch and Bound with Simplex.

    A_eq = []
    b_eq = []
    
    for i in range(k):
        s_i, t_i, d_i = requests[i]
        for u in range(n - 1):  # drop last node for redundancy
            row = [0.0] * num_vars
            for j in range(m):
                eu, ev, _ = edges[j]
                if eu == u:
                    row[var_idx(i, j)] += 1.0
                if ev == u:
                    row[var_idx(i, j)] -= 1.0
            
            bal = 0.0
            if u == s_i:
                bal += d_i
            if u == t_i:
                bal -= d_i
            
            A_eq.append(row)
            b_eq.append(bal)

    A_ub = []
    b_ub = []
    for j in range(m):
        row = [0.0] * num_vars
        for i in range(k):
            row[var_idx(i, j)] = 1.0
        A_ub.append(row)
        b_ub.append(float(edges[j][2]))

    # Simplex solver
    # Solve LP: find x >= 0 satisfying A_ub x <= b_ub, A_eq x == b_eq, and bounds on x.
    # We can use Two-Phase Simplex in Tableau form.

    EPS = 1e-7

    def solve_lp(extra_bounds):
        # extra_bounds: list of (var_idx, '<=' or '>=', val)
        # Convert all to standard form: Ax = b, x >= 0
        # Variables: x (num_vars), slacks for <=, -surplus for >=
        
        # Collect inequality constraints
        rows = []
        rhs = []
        
        # A_eq
        for r, b in zip(A_eq, b_eq):
            if b < 0:
                rows.append([-val for val in r])
                rhs.append(-b)
            else:
                rows.append(list(r))
                rhs.append(b)
        num_eq = len(rows)

        # A_ub
        for r, b in zip(A_ub, b_ub):
            rows.append(list(r))
            rhs.append(b)
            
        # extra bounds
        for v_idx, op, val in extra_bounds:
            r = [0.0] * num_vars
            if op == '<=':
                r[v_idx] = 1.0
                rows.append(r)
                rhs.append(val)
            else: # '>='
                r[v_idx] = -1.0
                rows.append(r)
                rhs.append(-val)

        num_constraints = len(rows)
        num_slacks = num_constraints - num_eq

        # Create tableau for Phase I
        # Number of artificial variables: num_constraints
        # Actually, slacks for <= (where rhs >= 0) can serve as basic variables without artificials.
        # But if rhs < 0 (can happen for '>=' converted), we need artificials.
        
        # Let's standardize:
        # Each constraint: sum(a_j x_j) + s_i = rhs if <= and rhs >= 0
        # Otherwise if rhs < 0, multiply by -1 => >= with positive rhs => - s_i + a_i = rhs
        
        # Full Phase I formulation:
        # A_full * [x, slacks, artificials] = rhs
        
        final_rows = []
        final_rhs = []
        artificial_vars = []
        
        col_count = num_vars
        slack_map = []
        
        for i in range(num_constraints):
            r = rows[i][:]
            b = rhs[i]
            if i < num_eq:
                # Equality constraint: r * x + a_i = b (b >= 0)
                final_rows.append(r)
                final_rhs.append(b)
                artificial_vars.append(i)
            else:
                # Inequality: r * x <= b
                if b >= -EPS:
                    # r * x + s = b
                    # s is a slack variable
                    final_rows.append(r)
                    final_rhs.append(max(0.0, b))
                    slack_map.append((i, 1.0)) # +1 slack
                else:
                    # -r * x - s + a = -b
                    r = [-val for val in r]
                    final_rows.append(r)
                    final_rhs.append(-b)
                    slack_map.append((i, -1.0)) # -1 slack
                    artificial_vars.append(i)

        num_slacks = len(slack_map)
        num_art = len(artificial_vars)
        
        total_cols = num_vars + num_slacks + num_art
        tableau = [[0.0] * (total_cols + 1) for _ in range(num_constraints + 1)]
        
        slack_col_idx = num_vars
        art_col_idx = num_vars + num_slacks
        
        art_ptr = 0
        basis = [-1] * num_constraints
        
        for i in range(num_constraints):
            for j in range(num_vars):
                tableau[i][j] = final_rows[i][j]
            tableau[i][-1] = final_rhs[i]
            
            if i >= num_eq:
                s_idx = i - num_eq
                sign = slack_map[s_idx][1]
                tableau[i][slack_col_idx + s_idx] = sign
                if sign == 1.0:
                    basis[i] = slack_col_idx + s_idx
            
            if basis[i] == -1:
                tableau[i][art_col_idx + art_ptr] = 1.0
                basis[i] = art_col_idx + art_ptr
                art_ptr += 1

        # Phase I objective: minimize sum of artificial variables
        # obj = sum(art_vars)
        for i in range(num_constraints):
            if basis[i] >= art_col_idx:
                for j in range(total_cols + 1):
                    tableau[-1][j] -= tableau[i][j]

        def pivot(r, c):
            basis[r] = c
            piv = tableau[r][c]
            for j in range(total_cols + 1):
                tableau[r][j] /= piv
            for i in range(num_constraints + 1):
                if i != r and abs(tableau[i][c]) > 1e-11:
                    factor = tableau[i][c]
                    for j in range(total_cols + 1):
                        tableau[i][j] -= factor * tableau[r][j]

        # Simplex loop for Phase I
        while True:
            # Find entering variable
            c = -1
            min_val = -EPS
            for j in range(total_cols):
                if tableau[-1][j] < min_val:
                    min_val = tableau[-1][j]
                    c = j
            if c == -1:
                break
                
            # Find leaving variable
            r = -1
            min_ratio = float('inf')
            for i in range(num_constraints):
                if tableau[i][c] > EPS:
                    ratio = tableau[i][-1] / tableau[i][c]
                    if ratio < min_ratio - 1e-11:
                        min_ratio = ratio
                        r = i
            if r == -1:
                return None  # Unbounded Phase I (should not happen)
            pivot(r, c)

        # Check Phase I result
        if abs(tableau[-1][-1]) > EPS:
            return None # Infeasible

        # Phase I feasible, extract solution
        sol = [0.0] * num_vars
        for i in range(num_constraints):
            if basis[i] < num_vars:
                sol[basis[i]] = max(0.0, tableau[i][-1])
        return sol

    # Branch and Bound
    best_sol = None

    def branch_and_bound(bounds):
        nonlocal best_sol
        if best_sol is not None:
            return
        sol = solve_lp(bounds)
        if sol is None:
            return
        
        # Check if all integer
        first_frac = -1
        for i in range(num_vars):
            val = sol[i]
            if abs(val - round(val)) > 1e-4:
                first_frac = i
                break
        
        if first_frac == -1:
            best_sol = [int(round(x)) for x in sol]
            return
        
        val = sol[first_frac]
        import math
        floor_v = math.floor(val)
        ceil_v = math.ceil(val)
        
        # Branch on floor
        branch_and_bound(bounds + [(first_frac, '<=', float(floor_v))])
        if best_sol is not None:
            return
        # Branch on ceil
        branch_and_bound(bounds + [(first_frac, '>=', float(ceil_v))])

    branch_and_bound([])

    if best_sol is None:
        print("NO")
    else:
        print("YES")
        for i in range(k):
            row = [str(best_sol[var_idx(i, j)]) for j in range(m)]
            print(" ".join(row))

if __name__ == '__main__':
    solve()
```
