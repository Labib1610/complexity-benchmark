import sys

def solve():

    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    k = int(input_data[idx]); idx += 1

    requests = []
    for _ in range(k):
        s = int(input_data[idx]); idx += 1
        t = int(input_data[idx]); idx += 1
        d = int(input_data[idx]); idx += 1
        requests.append((s, t, d))

        v = int(input_data[idx]); idx += 1
        cap = int(input_data[idx]); idx += 1

    #   in-node:  i*n + (v-1)
    #   out-node: k*n + i*n + (v-1)
    # Super source: 2*k*n
    # Super sink: 2*k*n + 1
    base_in = 0
    base_out = k * n
    S = 2 * k * n
    T = 2 * k * n + 1
    total_nodes = 2 * k * n + 2

    graph = [[] for _ in range(total_nodes)]

    def add_edge(u, v, cap):
        graph[u].append([v, cap, len(graph[v])])
        graph[v].append([u, 0, len(graph[u]) - 1])

    # Request edges: in -> out with capacity d_i
    for i, (s, t, d) in enumerate(requests):
        for v in range(1, n + 1):
            add_edge(base_in + i * n + (v - 1), base_out + i * n + (v - 1), d)

    # Super source to request start in-nodes
    for i, (s, t, d) in enumerate(requests):
        add_edge(S, base_in + i * n + (s - 1), d)

    # Request end out-nodes to super sink
    for i, (s, t, d) in enumerate(requests):
        add_edge(base_out + i * n + (t - 1), T, d)

    # Tube segments: shared capacity
    # For each edge (u, v, cap), add cap parallel edges from out-nodes to in-nodes
    for u, v, cap in edges:
        for i in range(k):
            add_edge(base_out + i * n + (u - 1), base_in + i * n + (v - 1), cap)

    # Dinic's algorithm
    def bfs():
        nonlocal level
        level = [-1] * total_nodes
        level[S] = 0
        q = [S]
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            for v, cap, rev in graph[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[T] >= 0

    def dfs(u, f):
        if u == T:
            return f
        while ptr[u] < len(graph[u]):
            edge = graph[u][ptr[u]]
            v, cap, rev = edge
            if cap > 0 and level[v] == level[u] + 1:
                pushed = dfs(v, min(f, cap))
                if pushed > 0:
                    edge[1] -= pushed
                    graph[v][rev][1] += pushed
                    return pushed
            ptr[u] += 1
        return 0

    level = [-1] * total_nodes
    max_flow = 0
    while bfs():
        ptr = [0] * total_nodes
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed

    total_demand = sum(d for _, _, d in requests)
    if max_flow < total_demand:
        print("NO")
        return

    # For request i, edge j (u, v, cap):
    # The edge is from base_out + i*n + (u-1) to base_in + i*n + (v-1)
    # It was added as the (k + j*k + i)-th edge from the out-node (0-indexed)
    # Actually, let's track: for each out-node, the first k edges are to in-nodes of same station

    #   - 1 edge to in-node (i, u) [request edge]
    #   - for each tube segment (u, v, cap): 1 edge to in-node (i, v)

    #   in_node = base_in + i*n + (v-1) where v = edges[j][1]

    # Rebuild everything with tracking
