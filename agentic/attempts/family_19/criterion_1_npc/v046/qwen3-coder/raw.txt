from collections import defaultdict
import sys
import threading

def main():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    k = int(data[idx])
    idx += 1
    
    orders = []
    for i in range(k):
        s = int(data[idx]) - 1
        idx += 1
        t = int(data[idx]) - 1
        idx += 1
        d = int(data[idx])
        idx += 1
        orders.append((s, t, d))
    
    corridors = []
    graph = [[] for _ in range(n)]
    for j in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        c = int(data[idx])
        idx += 1
        corridors.append((u, v, c))
        graph[u].append((v, j))
    
    INF = float('inf')
    
    # For each order, compute the flow on each edge in a path from s to t
    # We'll use a max flow approach to find if a feasible solution exists
    
    # We'll model this as a multi-commodity flow problem with constraints
    # But since each commodity (order) has a single source and sink, and we only care about feasibility,
    # we can model it as a single min-cost max-flow problem with demands
    
    # However, since all costs are 0, we can just use a max-flow approach with demands
    
    # Let's instead model it as a system of linear equations and inequalities:
    # For each order i and corridor j: 0 <= f_ij <= some_upper_bound (but we don't know this yet)
    # For each order i and node v (not source/sink): in_flow = out_flow
    # For each order i and source s: out_flow - in_flow = demand
    # For each order i and sink t: in_flow - out_flow = demand
    # For each corridor j: sum over all orders i of f_ij <= capacity_j
    
    # This is a linear programming problem. We'll use a network flow approach.
    
    # Let's build a larger graph that includes all commodities
    # But since we only need feasibility, we can use the following approach:
    # For each corridor, we have a constraint: sum of all flows on it <= capacity
    # For each order, we need to find a flow from source to sink with value = demand
    # This is equivalent to finding a feasible multi-commodity flow
    
    # We'll solve this using the constraint method:
    # For each order, we'll find a path from source to sink and push flow
    # But we need to ensure that the total flow on each edge doesn't exceed capacity
    
    # Let's use a different approach: formulate as LP and solve with simplex or use network flow
    
    # Since this is a feasibility problem, we can model it as a circulation problem with demands
    # But it's a multi-commodity flow, which is NP-hard in general
    # However, for small instances, we can try to solve it
    
    # Let's try to solve it with a flow-based approach:
    # For each order, we need to route 'd' units of flow from 's' to 't'
    # The sum of flows from all orders on any corridor must not exceed its capacity
    
    # We'll use the following approach:
    # 1. For each order, find a path from s to t
    # 2. Push the required flow along that path
    # 3. Check if any corridor exceeds its capacity
    # 4. If yes, try to reroute flows to satisfy all constraints
    
    # This is complex. Let's instead model it as an integer linear program and solve with a solver
    # But since we can't use external libraries, we'll need to implement a solution
    
    # Let's try a greedy approach:
    # For each order, find a path and push flow, keeping track of corridor usage
    
    from collections import deque, defaultdict
    
    def bfs_path(graph, source, target, n):
        # BFS to find a path from source to target
        visited = [False] * n
        queue = deque()
        queue.append((source, []))
        visited[source] = True
        
        while queue:
            u, path = queue.popleft()
            if u == target:
                return path
            
            for v, edge_idx in graph[u]:
                if not visited[v]:
                    visited[v] = True
                    queue.append((v, path + [edge_idx]))
        
        return None  # No path found
    
    # Initialize flow on corridors for each order
    flow = [[0] * m for _ in range(k)]
    corridor_usage = [0] * m
    
    # For each order, find a path and push flow
    paths = []
    for i in range(k):
        s, t, d = orders[i]
        path = bfs_path(graph, s, t, n)
        if path is None:
            print("NO")
            return
        paths.append(path)
        
        # Push d units of flow on this path
        for edge_idx in path:
            flow[i][edge_idx] += d
            corridor_usage[edge_idx] += d
    
    # Check if any corridor exceeds its capacity
    valid = True
    for j in range(m):
        if corridor_usage[j] > corridors[j][2]:
            valid = False
            break
    
    if valid:
        print("YES")
        for i in range(k):
            print(' '.join(map(str, flow[i])))
        return
    
    # If not valid, we need to try a different approach
    # This greedy method doesn't work. Let's try to model it properly.
    
    # We'll model it as a system where we need to find flows that satisfy:
    # 1. Flow conservation for each commodity at each node
    # 2. Capacity constraints on edges
    # 3. Demand satisfaction for each commodity
    
    # This is a complex problem. For small inputs, we can try all possible combinations,

    # Let's try to solve it with a more sophisticated approach:
    # We'll use the fact that we can decompose flows into paths and cycles.
    # For each order, we can represent its flow as a sum of paths.
    
    # But this is getting too complex for a simple implementation.
    
    # Let's try a different greedy approach:
    # For each corridor, we'll keep track of how much flow we can still push
    # Then, for each order, we'll try to find a path that respects current capacities
    
    corridor_remaining = [corridors[j][2] for j in range(m)]
    
    # Reset flows
    flow = [[0] * m for _ in range(k)]
    
    def dfs_path_with_capacity(u, target, visited, path, min_cap, graph, corridor_remaining):
        if u == target:
            return path, min_cap
        visited[u] = True
        for v, edge_idx in graph[u]:
            if not visited[v] and corridor_remaining[edge_idx] > 0:
                new_path = path + [edge_idx]
                new_min = min(min_cap, corridor_remaining[edge_idx])
                result = dfs_path_with_capacity(v, target, visited, new_path, new_min, graph, corridor_remaining)
                if result is not None:
                    return result
        return None
    
    def find_path_with_capacity(source, target, n, graph, corridor_remaining):
        visited = [False] * n
        result = dfs_path_with_capacity(source, target, visited, [], INF, graph, corridor_remaining)
        return result
    
    success = True
    for i in range(k):
        s, t, d = orders[i]
        # Find a path from s to t that can carry at least d flow
        path_and_flow = find_path_with_capacity(s, t, n, graph, corridor_remaining)
        if path_and_flow is None:
            success = False
            break
        path, min_cap_on_path = path_and_flow
        # We can push at most min_cap_on_path, but we need exactly d
        if min_cap_on_path < d:
            success = False
            break
        # Push d units of flow
        for edge_idx in path:
            flow[i][edge_idx] += d
            corridor_remaining[edge_idx] -= d
            if corridor_remaining[edge_idx] < 0:
                success = False
                break
        if not success:
            break
    
    if success:
        print("YES")
        for i in range(k):
            print(' '.join(map(str, flow[i])))
    else:
        print("NO")

threading.Thread(target=main).start()
