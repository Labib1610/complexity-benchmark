import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    products = []
    for _ in range(k):
        s = int(next(it))
        t = int(next(it))
        d = int(next(it))
        products.append((s, t, d))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        c = int(next(it))
        edges.append((u, v, c))
    
    # Build a flow network for all products simultaneously
    # Node indexing: 0 = super source, 1..n = original nodes, n+1 = super sink
    # For each product i, we have edges from super source to s_i with capacity d_i
    # and from t_i to super sink with capacity d_i
    # All original edges have capacity c (shared across all products)
    
    N = n + 2
    source = 0
    sink = n + 1
    
    # Build adjacency list for Dinic
    adj = [[] for _ in range(N)]
    edge_list = []  # (to, cap, rev_idx)
    
    def add_edge(fr, to, cap):
        adj[fr].append(len(edge_list))
        edge_list.append([to, cap, len(adj[to])])
        adj[to].append(len(edge_list))
        edge_list.append([fr, 0, len(adj[fr]) - 1])
    
    # Add product source/sink edges
    total_demand = 0
    for s, t, d in products:
        add_edge(source, s, d)
        add_edge(t, sink, d)
        total_demand += d
    
    # Add original network edges
    edge_offset = len(edge_list)
    original_edge_indices = []
    for u, v, c in edges:
        idx = len(edge_list)
        original_edge_indices.append(idx)
        add_edge(u, v, c)
    
    # Dinic's algorithm
    level = [0] * N
    it_ptr = [0] * N
    
    def bfs():
        for i in range(N):
            level[i] = -1
        queue = deque([source])
        level[source] = 0
        while queue:
            v = queue.popleft()
            for ei in adj[v]:
                to, cap, _ = edge_list[ei]
                if cap > 0 and level[to] < 0:
                    level[to] = level[v] + 1
                    queue.append(to)
        return level[sink] >= 0
    
    def dfs(v, up):
        if v == sink:
            return up
        for i in range(it_ptr[v], len(adj[v])):
            it_ptr[v] = i
            ei = adj[v][i]
            to, cap, rev = edge_list[ei]
            if cap > 0 and level[v] < level[to]:
                ret = dfs(to, min(up, cap))
                if ret > 0:
                    edge_list[ei][1] -= ret
                    edge_list[rev][1] += ret
                    return ret
        return 0
    
    flow = 0
    while bfs():
        it_ptr = [0] * N
        while True:
            f = dfs(source, float('inf'))
            if f == 0:
                break
            flow += f
    
    if flow != total_demand:
        print("NO")
        return
    
    print("YES")
    
    # Now we need to decompose the flow into individual product flows
    # We use flow decomposition for each product
    
    # Get current flow on each edge
    def get_flow(ei):
        return edge_list[ei ^ 1][1]  # reverse edge's capacity = forward flow
    
    # For each product, find paths from s_i to t_i and extract flow
    result = [[0] * m for _ in range(k)]
    
    for prod_idx, (s, t, d) in enumerate(products):
        remaining = d
        while remaining > 0:
            # Find a path from s to t with positive flow
            parent = [-1] * N
            parent_edge = [-1] * N
            queue = deque([s])
            parent[s] = s
            while queue and parent[t] == -1:
                v = queue.popleft()
                for ei in adj[v]:
                    to, cap, _ = edge_list[ei]
                    if cap > 0 and parent[to] == -1:
                        parent[to] = v
                        parent_edge[to] = ei
                        queue.append(to)
            
            if parent[t] == -1:
                # No path found - shouldn't happen if flow was valid
                break
            
            # Find bottleneck
            path = []
            v = t
            min_cap = float('inf')
            while v != s:
                ei = parent_edge[v]
                min_cap = min(min_cap, edge_list[ei][1])
                path.append((v, ei))
                v = parent[v]
            
            # Reduce capacities
            for v, ei in path:
                edge_list[ei][1] -= min_cap
                edge_list[ei ^ 1][1] += min_cap
            
            # Record which original edges were used
            v = t
            while v != s:
                ei = parent_edge[v]
                # Check if this is an original edge
                if ei >= edge_offset and (ei - edge_offset) % 2 == 0:
                    orig_idx = (ei - edge_offset) // 2
                    result[prod_idx][orig_idx] += min_cap
                v = parent[v]
            
            remaining -= min_cap
    
    for row in result:
        print(' '.join(map(str, row)))

if __name__ == "__main__":
    solve()
