import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    k = int(next(iterator))
    
    contracts = []
    for _ in range(k):
        s = int(next(iterator))
        t = int(next(iterator))
        d = int(next(iterator))
        contracts.append((s, t, d))
        
    edges = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        c = int(next(iterator))
        edges.append((u, v, c))
        
    # Total nodes in flow graph: 
    # 1..n for original offices
    # n+1..n+k for contract sources
    # n+k+1..n+2k for contract sinks
    # n+2k+1 for super source S
    # n+2k+2 for super sink T
    S = n + 2 * k + 1
    T = n + 2 * k + 2
    N = T
    
    adj = [[] for _ in range(N + 1)]
    
    def add_edge(u, v, cap):
        adj[u].append([v, cap, len(adj[v])])
        adj[v].append([u, 0, len(adj[u]) - 1])
        
    # Super source to contract sources
    for i in range(k):
        s_i, t_i, d_i = contracts[i]
        cs = n + i + 1
        add_edge(S, cs, d_i)
        
    # Contract sources to drafting offices
    for i in range(k):
        s_i, t_i, d_i = contracts[i]
        cs = n + i + 1
        add_edge(cs, s_i, d_i)
        
    # Archival offices to contract sinks
    for i in range(k):
        s_i, t_i, d_i = contracts[i]
        ct = n + k + i + 1
        add_edge(t_i, ct, d_i)
        
    # Contract sinks to super sink
    for i in range(k):
        s_i, t_i, d_i = contracts[i]
        ct = n + k + i + 1
        add_edge(ct, T, d_i)
        
    # Original tubes
    for u, v, c in edges:
        add_edge(u, v, c)
        
    total_demand = sum(d for _, _, d in contracts)
    
    # Dinic's algorithm
    level = [0] * (N + 1)
    ptr = [0] * (N + 1)
    
    def bfs():
        for i in range(N + 1):
            level[i] = -1
        queue = [S]
        level[S] = 0
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            for v, cap, _ in adj[u]:
                if cap > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    if v == T:
                        return True
                    queue.append(v)
        return level[T] != -1
        
    def dfs(u, flow):
        if u == T:
            return flow
        while ptr[u] < len(adj[u]):
            e = adj[u][ptr[u]]
            v, cap, rev = e
            if cap > 0 and level[v] == level[u] + 1:
                pushed = dfs(v, min(flow, cap))
                if pushed > 0:
                    e[1] -= pushed
                    adj[v][rev][1] += pushed
                    return pushed
            ptr[u] += 1
        return 0
        
    max_flow = 0
    while bfs():
        ptr = [0] * (N + 1)
        while True:
            pushed = dfs(S, float('inf'))
            if pushed == 0:
                break
            max_flow += pushed
            
    if max_flow != total_demand:
        print("NO")
        return
        
    print("YES")
    # Reconstruct the flow on original edges
    for i in range(k):
        res = []
        for j in range(m):
            u, v, c = edges[j]
            # The edge from u to v is the k-th edge added from u.
            # Since we added exactly k edges from u before the tubes,
            # the tube edge is at index k + j in adj[u].
            flow_on_edge = adj[u][k + j][1]
            res.append(str(c - flow_on_edge))
        print(" ".join(res))

if __name__ == '__main__':
    solve()
