```python
from collections import defaultdict
import sys
import threading

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    k = int(data[idx])
    idx += 1
    
    contracts = []
    for _ in range(k):
        s = int(data[idx])
        idx += 1
        t = int(data[idx])
        idx += 1
        d = int(data[idx])
        idx += 1
        contracts.append((s, t, d))
    
    edges = []
    edge_dict = {}
    graph = [[] for _ in range(n+1)]
    reverse_graph = [[] for _ in range(n+1)]
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        c = int(data[idx])
        idx += 1
        edges.append((u, v, c))
        edge_dict[(u, v)] = i
        graph[u].append((v, i))
        reverse_graph[v].append((u, i))
    
    # For each contract, we need to find a flow from s to t of value d
    # This is a multi-commodity flow problem with integer flows and capacity constraints
    # We will model this as a linear programming feasibility problem and solve using network flow techniques
    
    # However, since we need integer flows and exact values, we'll use a different approach:
    # For each contract, compute a feasible flow (e.g., shortest path flow) and then adjust to satisfy capacities
    
    # But given the constraints and guarantees, let's try modeling it as a min-cost flow or just check feasibility
    
    # Let's build a super-source and super-sink
    # But actually, since we have multiple commodities, it's a bit tricky
    
    # Let's think differently: for each contract, we want to send d units from s to t
    # We can represent each contract separately and then check if the total usage of each edge <= capacity
    
    # Approach:
    # 1. For each contract, compute a feasible flow (we can use a simple path decomposition)
    # 2. Sum up the flows on each edge across all contracts
    # 3. Check if this sum <= capacity of the edge
    
    # Since we're not asked to minimize anything, just to find a feasible solution,
    # we can use any valid flow for each commodity
    
    # Let's implement a BFS-based flow assignment for each contract
    
    def bfs_flow(s, t, demand):
        # Find a simple path from s to t and push 'demand' units along it
        # If no path exists, return None
        from collections import deque
        queue = deque()
        queue.append(s)
        parent = [-1] * (n+1)
        visited = [False] * (n+1)
        visited[s] = True
        
        while queue:
            u = queue.popleft()
            if u == t:
                break
            for v, edge_idx in graph[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
        
        if not visited[t]:
            return None  # No path from s to t
            
        # Reconstruct path
        path = []
        curr = t
        while parent[curr] != -1:
            prev = parent[curr]
            if (prev, curr) in edge_dict:
                path.append(edge_dict[(prev, curr)])
            else:
                return None  # Should not happen
            curr = prev
        path.reverse()
        
        # Create flow vector for this path
        flow = [0] * m
        for e in path:
            flow[e] += demand
        return flow
    
    # Store flow for each contract
    contract_flows = []
    
    for s, t, d in contracts:
        flow = bfs_flow(s, t, d)
        if flow is None:
            print("NO")
            return
        contract_flows.append(flow)
    
    # Now check total usage of each edge
    total_usage = [0] * m
    for flow in contract_flows:
        for i in range(m):
            total_usage[i] += flow[i]
    
    # Check against capacities
    for i in range(m):
        if total_usage[i] > edges[i][2]:
            print("NO")
            return
    
    # If we reach here, a feasible solution exists
    print("YES")
    for flow in contract_flows:
        print(' '.join(map(str, flow)))

if __name__ == "__main__":
    threading.Thread(target=main).start()
```
