import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    commodities = []
    for _ in range(k):
        s = int(next(it)) - 1
        t = int(next(it)) - 1
        d = int(next(it))
        commodities.append((s, t, d))

    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        cap = int(next(it))
        edges.append((u, v, cap))

    # Fast path: DFS with randomized restarts for unit/small demands or general ILP solver
    # Let's use Branch and Bound with a Simplex-based LP solver

    num_vars = k * m

    # Build linear constraints:
    # 1. Capacity constraints: sum_{i} x_{i, e} <= cap_e
    # 2. Flow conservation: sum_{e out of v} x_{i, e} - sum_{e into v} x_{i, e} = b_{i, v}
    # where b_{i, v} = d_i if v == s_i, -d_i if v == t_i, 0 otherwise.

    # Simplex solver implementation
    class Simplex:

        def __init__(self, A, b, c):
            # Maximize c^T x subject to A x <= b, x >= 0
            self.m = len(A)
            self.n = len(c)
            self.tableau = [
                [0.0] * (self.n + self.m + 1) for _ in range(self.m + 1)
            ]
            self.basis = list(range(self.n, self.n + self.m))

            for i in range(self.m):
                for j in range(self.n):
                    self.tableau[i][j] = float(A[i][j])
                self.tableau[i][self.n + i] = 1.0
                self.tableau[i][-1] = float(b[i])

            for j in range(self.n):
                self.tableau[-1][j] = -float(c[j])

        def pivot(self, r, c):
            inv = 1.0 / self.tableau[r][c]
            for j in range(len(self.tableau[0])):
                self.tableau[r][j] *= inv
            for i in range(len(self.tableau)):
                if i != r:
                    factor = self.tableau[i][c]
                    if abs(factor) > 1e-12:
                        for j in range(len(self.tableau[0])):
                            self.tableau[i][j] -= factor * self.tableau[r][j]
            self.basis[r] = c

        def solve(self):
            # Phase 1: if any b_i < 0, make it feasible
            while True:
                min_b = 0.0
                r = -1
                for i in range(self.m):
                    if self.tableau[i][-1] < min_b - 1e-9:
                        min_b = self.tableau[i][-1]
                        r = i
                if r == -1:
                    break

                c = -1
                for j in range(self.n + self.m):
                    if self.tableau[r][j] < -1e-9:
                        c = j
                        break
                if c == -1:
                    return None  # Infeasible
                self.pivot(r, c)

            # Phase 2: Optimality
            while True:
                c = -1
                min_val = -1e-9
                for j in range(self.n + self.m):
                    if self.tableau[-1][j] < min_val:
                        min_val = self.tableau[-1][j]
                        c = j
                if c == -1:
                    break

                r = -1
                min_ratio = float("inf")
                for i in range(self.m):
                    if self.tableau[i][c] > 1e-9:
                        ratio = self.tableau[i][-1] / self.tableau[i][c]
                        if ratio < min_ratio - 1e-9:
                            min_ratio = ratio
                            r = i
                if r == -1:
                    return None  # Unbounded
                self.pivot(r, c)

            sol = [0.0] * self.n
            for i in range(self.m):
                if self.basis[i] < self.n:
                    sol[self.basis[i]] = self.tableau[i][-1]
            return sol

    # Build A_ub and b_ub
    A_ub = []
    b_ub = []

    # Capacities
    for e in range(m):
        row = [0.0] * num_vars
        for i in range(k):
            row[i * m + e] = 1.0
        A_ub.append(row)
        b_ub.append(float(edges[e][2]))

    # Conservation equalities converted to two inequalities
    for i in range(k):
        s_i, t_i, d_i = commodities[i]
        for v in range(n):
            row = [0.0] * num_vars
            for e in range(m):
                u, w, _ = edges[e]
                if u == v:
                    row[i * m + e] += 1.0
                if w == v:
                    row[i * m + e] -= 1.0
            target = 0.0
            if v == s_i:
                target = float(d_i)
            elif v == t_i:
                target = float(-d_i)

            A_ub.append(row)
            b_ub.append(target)
            A_ub.append([-x for x in row])
            b_ub.append(-target)

    # Branch and Bound
    c = [0.0] * num_vars

    stack = [(A_ub, b_ub)]
    solution = None

    while stack:
        cur_A, cur_b = stack.pop()
        solver = Simplex(cur_A, cur_b, c)
        sol = solver.solve()
        if sol is None:
            continue

        # Check integrality
        best_frac = 0.0
        split_var = -1
        is_int = True
        for j in range(num_vars):
            val = sol[j]
            frac = abs(val - round(val))
            if frac > 1e-4:
                is_int = False
                if frac > best_frac:
                    best_frac = frac
                    split_var = j

        if is_int:
            solution = [int(round(x)) for x in sol]
            break

        val = sol[split_var]
        floor_val = float(int(val))
        ceil_val = floor_val + 1.0

        # Branch 1: x <= floor_val
        row1 = [0.0] * num_vars
        row1[split_var] = 1.0
        stack.append((cur_A + [row1], cur_b + [floor_val]))

        # Branch 2: x >= ceil_val  =>  -x <= -ceil_val
        row2 = [0.0] * num_vars
        row2[split_var] = -1.0
        stack.append((cur_A + [row2], cur_b + [-ceil_val]))

    if solution is not None:
        print("YES")
        for i in range(k):
            stream_flow = solution[i * m : (i + 1) * m]
            print(*(max(0, x) for x in stream_flow))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
