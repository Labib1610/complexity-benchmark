from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    M = int(data[idx])
    idx += 1
    T = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    # Build graph
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(data[idx])
        idx += 1
        b = int(data[idx])
        idx += 1
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with day assignment
    # We'll use a greedy approach: assign the earliest possible day
    # We need to respect:
    # 1. Dependencies (topological order)
    # 2. At most M vials per day
    # 3. All within T days
    
    # Kahn's algorithm with day tracking
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    # day[vial] = the day it's assigned to
    day = [0] * (n + 1)
    # count[day] = number of vials assigned to that day
    count = [0] * (T + 1)
    
    # For each day, we maintain a list of nodes ready to be scheduled
    # We'll process nodes in topological order, and for each node,
    # we assign it to the earliest day where:
    # - count[day] < M
    # - day > max_day_of_predecessors
    
    # We'll keep a list of available days for each node
    # But a simpler approach: process in topological order and assign greedily
    
    # Let's track the maximum day of predecessors for each node
    max_pred_day = [0] * (n + 1)
    
    # List of nodes in topological order
    topo_order = []
    
    while queue:
        node = queue.popleft()
        topo_order.append(node)
        for neighbor in graph[node]:
            indegree[neighbor] -= 1
            max_pred_day[neighbor] = max(max_pred_day[neighbor], day[node])
            if indegree[neighbor] == 0:
                queue.append(neighbor)
    
    # Check if all nodes are included (no cycles)
    if len(topo_order) != n:
        print("NO")
        return
    
    # Now assign days greedily
    for node in topo_order:
        # Earliest day is max_pred_day[node] + 1
        start_day = max_pred_day[node] + 1
        assigned = False
        for d in range(start_day, T + 1):
            if count[d] < M:
                day[node] = d
                count[d] += 1
                assigned = True
                break
        if not assigned:
            print("NO")
            return
    
    # If we reached here, we have a valid assignment
    print("YES")
    print(' '.join(str(day[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
