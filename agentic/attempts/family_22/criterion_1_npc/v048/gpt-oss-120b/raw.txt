```python
import sys
import heapq

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); M = int(next(it)); T = int(next(it)); m = int(next(it))
    g = [[] for _ in range(n)]
    rg = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(m):
        a = int(next(it))-1
        b = int(next(it))-1
        g[a].append(b)
        rg[b].append(a)
        indeg[b] += 1

    # topological order
    from collections import deque
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    while q:
        v = q.popleft()
        topo.append(v)
        for to in g[v]:
            indeg[to] -= 1
            if indeg[to]==0:
                q.append(to)
    if len(topo)!=n:
        print("NO")
        return

    # earliest day
    earliest = [1]*n
    for v in topo:
        ev = earliest[v]
        for to in g[v]:
            if earliest[to] < ev + 1:
                earliest[to] = ev + 1

    # latest day
    latest = [T]*n
    for v in reversed(topo):
        lv = latest[v]
        for pre in rg[v]:
            if latest[pre] > lv - 1:
                latest[pre] = lv - 1

    for i in range(n):
        if earliest[i] > latest[i] or earliest[i] > T or latest[i] < 1:
            print("NO")
            return

    # bucket by earliest
    buckets = [[] for _ in range(T+2)]
    for i in range(n):
        if earliest[i] <= T:
            buckets[earliest[i]].append(i)

    heap = []  # (latest, node)
    assign = [0]*n
    for day in range(1, T+1):
        for node in buckets[day]:
            heapq.heappush(heap, (latest[node], node))
        cnt = 0
        while heap and cnt < M:
            lat, node = heapq.heappop(heap)
            if lat < day:   # missed deadline
                print("NO")
                return
            assign[node] = day
            cnt += 1

    if any(a==0 for a in assign):
        print("NO")
        return

    print("YES")
    print(' '.join(str(d) for d in assign))

if __name__ == "__main__":
    solve()
```
