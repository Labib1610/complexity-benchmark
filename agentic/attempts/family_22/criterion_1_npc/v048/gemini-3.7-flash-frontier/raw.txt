import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    if n > M * T:
        print("NO")
        return
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    out_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        out_degree[u] += 1
        
    # Check for cycles and compute topological sort
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo_order = []
    
    temp_in = list(in_degree)
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            temp_in[v] -= 1
            if temp_in[v] == 0:
                q.append(v)
                
    if len(topo_order) < n:
        print("NO")
        return
    
    # Compute E[u]: earliest day (1-based)
    E = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if E[u] + 1 > E[v]:
                E[v] = E[u] + 1
                
    # Compute L[u]: latest day (1-based)
    L = [T] * (n + 1)
    for u in reversed(topo_order):
        for p in rev_adj[u]:
            if L[u] - 1 < L[p]:
                L[p] = L[u] - 1
                
    for i in range(1, n + 1):
        if E[i] > L[i] or E[i] > T or L[i] < 1:
            print("NO")
            return

    # Backtracking / List scheduling search
    # Priority metric for tie-breaking: latest deadline L[u], then -height
    # Height from u to end
    height = [0] * (n + 1)
    for u in reversed(topo_order):
        for v in adj[u]:
            if height[v] + 1 > height[u]:
                height[u] = height[v] + 1

    current_in_degree = list(in_degree)
    assigned_day = [0] * (n + 1)
    
    # State for search
    # We will schedule day by day: day = 1 .. T
    # At day d, ready nodes are those with current_in_degree == 0 and assigned_day == 0
    # A node must be scheduled at or before L[u]. If L[u] < d, it's impossible.
    
    # To keep search fast, we use a recursive function with backtracking
    # We can use a greedy heuristic first. If greedy succeeds, return immediately.
    
    def try_greedy():
        cur_in = list(in_degree)
        ans = [0] * (n + 1)
        ready = [i for i in range(1, n + 1) if cur_in[i] == 0]
        
        for d in range(1, T + 1):
            # Check if any ready node has L[u] < d (impossible)
            for u in ready:
                if L[u] < d:
                    return None
            
            # Sort ready nodes by (L[u], -height[u], -len(adj[u]))
            ready.sort(key=lambda u: (L[u], -height[u], -len(adj[u])))
            
            # Choose up to M
            chosen = ready[:M]
            ready = ready[M:]
            
            for u in chosen:
                ans[u] = d
                for v in adj[u]:
                    cur_in[v] -= 1
                    if cur_in[v] == 0:
                        ready.append(v)
                        
        if all(ans[i] > 0 for i in range(1, n + 1)):
            return ans
        return None

    res = try_greedy()
    if res is not None:
        print("YES")
        print(*(res[1:]))
        return

    # If simple greedy didn't work, we can do branch and bound
    import itertools
    import time
    start_time = time.time()
    
    def backtrack(d, ready, cur_in, ans):
        if time.time() - start_time > 4.5:
            return None
        
        # Check validity
        for u in ready:
            if L[u] < d:
                return None
        
        # Remaining nodes to schedule
        rem = sum(1 for i in range(1, n + 1) if ans[i] == 0)
        if rem == 0:
            return ans
        if (T - d + 1) * M < rem:
            return None
        if d > T:
            return None

        # Mandatory nodes for today: L[u] == d
        must = [u for u in ready if L[u] == d]
        if len(must) > M:
            return None
        
        optional = [u for u in ready if L[u] > d]
        optional.sort(key=lambda u: (L[u], -height[u], -len(adj[u])))
        
        needed = min(M - len(must), len(optional))
        # Generate combinations of optional nodes of sizes from min(M-len(must), len(optional)) down to 0
        # Usually choosing the maximum possible M nodes is best.
        # We try larger subsets first.
        for take_k in range(needed, -1, -1):
            # Take top candidates greedily or combinations if small
            # Try the best combination first
            if take_k == needed:
                combo_list = [tuple(optional[:take_k])]
            else:
                # limited combinations
                combo_list = list(itertools.islice(itertools.combinations(optional, take_k), 5))
            
            for combo in combo_list:
                chosen = must + list(combo)
                new_ready = [u for u in ready if u not in chosen]
                new_in = list(cur_in)
                new_ans = list(ans)
                
                for u in chosen:
                    new_ans[u] = d
                    for v in adj[u]:
                        new_in[v] -= 1
                        if new_in[v] == 0:
                            new_ready.append(v)
                            
                sol = backtrack(d + 1, new_ready, new_in, new_ans)
                if sol is not None:
                    return sol
        return None

    ready_init = [i for i in range(1, n + 1) if in_degree[i] == 0]
    res = backtrack(1, ready_init, in_degree, [0] * (n + 1))
    
    if res is not None:
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
