from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    M = int(data[idx+1])
    T = int(data[idx+2])
    m = int(data[idx+3])
    idx += 4
    
    # Build graph and in-degree array
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx+1])
        graph[a].append(b)
        in_degree[b] += 1
        idx += 2
    
    # Topological sort with level tracking
    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    levels = [[] for _ in range(T + 1)]  # levels[term] = list of modules
    term = [0] * (n + 1)
    current_level = 1
    count = 0
    
    while current_level <= T:
        next_queue = deque()
        added_this_term = 0
        
        # Fill up to M modules in current term
        while queue and added_this_term < M:
            node = queue.popleft()
            levels[current_level].append(node)
            term[node] = current_level
            added_this_term += 1
            count += 1
            
            # Update in-degrees of neighbors
            for neighbor in graph[node]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    next_queue.append(neighbor)
        
        # Move to next level
        queue = next_queue
        current_level += 1
        
        # If no more nodes but we haven't placed all, try to continue
        if not queue and count < n:
            break
    
    # Check if all modules were scheduled
    if count != n:
        print("NO")
        return
    
    # Check if we exceeded T terms
    max_term = max(term[1:])
    if max_term > T:
        print("NO")
        return
        
    print("YES")
    print(' '.join(map(str, term[1:])))

solve()
