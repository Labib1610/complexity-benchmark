We can model this as a scheduling problem with precedence constraints and a capacity limit per time slot.  

We can solve it using a **topological sort with earliest start times**, then check if we can assign tasks to slots without exceeding the station limit.  

Here’s the Python solution:  

```python
from collections import deque
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1
    
    # earliest start time
    earliest = [1] * (n + 1)
    q = deque()
    for i in range(1, n + 1):
        if indeg[i] == 0:
            q.append(i)
    
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            earliest[v] = max(earliest[v], earliest[u] + 1)
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    # Check if any earliest exceeds T
    if max(earliest[1:]) > T:
        print("NO")
        return
    
    # Now assign slots greedily: place tasks in earliest possible slot
    # but respect capacity M per slot
    slots = [0] * (n + 1)
    count_in_slot = [0] * (T + 1)
    
    # Process in topological order
    for u in topo:
        # find earliest slot >= earliest[u] with capacity
        s = earliest[u]
        while s <= T and count_in_slot[s] >= M:
            s += 1
        if s > T:
            print("NO")
            return
        slots[u] = s
        count_in_slot[s] += 1
    
    print("YES")
    print(" ".join(str(slots[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
