import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1

    # earliest[v] = earliest round dish v can be served
    earliest = [1] * (n + 1)
    q = deque()
    for v in range(1, n + 1):
        if indeg[v] == 0:
            q.append(v)
    seen = 0
    while q:
        u = q.popleft()
        seen += 1
        for v in adj[u]:
            if earliest[u] + 1 > earliest[v]:
                earliest[v] = earliest[u] + 1
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if seen != n:
        print("NO")
        return

    # latest[v] = latest round dish v can be served
    latest = [T] * (n + 1)
    radj = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        for v in adj[u]:
            radj[v].append(u)
    q = deque()
    outdeg = [0] * (n + 1)
    for u in range(1, n + 1):
        outdeg[u] = len(adj[u])
        if outdeg[u] == 0:
            q.append(u)
    while q:
        v = q.popleft()
        for u in radj[v]:
            if latest[v] - 1 < latest[u]:
                latest[u] = latest[v] - 1
            outdeg[u] -= 1
            if outdeg[u] == 0:
                q.append(u)

    for v in range(1, n + 1):
        if earliest[v] > latest[v]:
            print("NO")
            return

    # Group dishes by earliest round
    by_earliest = [[] for _ in range(T + 2)]
    for v in range(1, n + 1):
        by_earliest[earliest[v]].append(v)

    # Greedy assignment: process rounds 1..T, maintain available dishes.
    # At each round, pick up to M dishes with smallest latest deadline.
    import heapq
    avail = []
    ans = [0] * (n + 1)
    ok = True
    for r in range(1, T + 1):
        for v in by_earliest[r]:
            heapq.heappush(avail, (latest[v], v))
        cnt = 0
        while avail and cnt < M:
            d, v = heapq.heappop(avail)
            if d < r:
                ok = False
                break
            ans[v] = r
            cnt += 1
        if not ok:
            break

    if not ok or avail:
        print("NO")
        return

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

solve()
