```python
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    M = int(data[1])
    T = int(data[2])
    m = int(data[3])
    
    edges = []
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx + 1])
        idx += 2
        edges.append((a, b))
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with round constraints
    # We need to assign rounds 1 to T, each round has at most M dishes
    # Dish a must be in a round < round of dish b for each (a,b) edge
    
    # First check if topological order exists (no cycles)
    queue = deque()
    indeg_copy = indegree[:]
    topo_order = []
    
    for i in range(1, n + 1):
        if indeg_copy[i] == 0:
            queue.append(i)
    
    while queue:
        node = queue.popleft()
        topo_order.append(node)
        for neighbor in graph[node]:
            indeg_copy[neighbor] -= 1
            if indeg_copy[neighbor] == 0:
                queue.append(neighbor)
    
    if len(topo_order) != n:
        print("NO")
        return
    
    # Now we try to assign rounds using a greedy approach with priority queue idea
    # We process nodes in topological order and assign earliest possible round
    # We track how many dishes are assigned to each round
    
    round_count = [0] * (T + 1)  # round_count[t] = number of dishes in round t
    assignment = [0] * (n + 1)   # assignment[i] = round of dish i
    
    # For each node in topo_order, assign it to the earliest round possible
    # respecting in-degree constraints (all predecessors assigned to earlier rounds)
    
    # We'll keep track of the latest round a predecessor was assigned to
    pred_max_round = [0] * (n + 1)
    
    for node in topo_order:
        # Find the earliest round > pred_max_round[node] with space
        min_round = pred_max_round[node] + 1
        assigned_round = -1
        for r in range(min_round, T + 1):
            if round_count[r] < M:
                assigned_round = r
                break
        
        if assigned_round == -1:
            print("NO")
            return
            
        assignment[node] = assigned_round
        round_count[assigned_round] += 1
        
        # Update successors' pred_max_round
        for succ in graph[node]:
            if pred_max_round[succ] < assigned_round:
                pred_max_round[succ] = assigned_round
    
    # Check if all nodes were assigned
    for i in range(1, n + 1):
        if assignment[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

if __name__ == "__main__":
    main()
```
