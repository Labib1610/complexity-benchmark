import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Compute earliest section using Kahn's algorithm
    earliest = [1] * n
    q = deque()
    for i in range(n):
        if indeg[i] == 0:
            q.append(i)
    
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    for i in range(n):
        if earliest[i] > T:
            print("NO")
            return
    
    # Compute latest section using reverse topological order
    latest = [T] * n
    for u in reversed(topo):
        for v in adj[u]:
            if latest[u] > latest[v] - 1:
                latest[u] = latest[v] - 1
    
    for i in range(n):
        if latest[i] < 1:
            print("NO")
            return
    
    # Greedy assignment: assign each document to its latest possible section
    # to leave room for predecessors
    sections = [0] * n
    for i in range(n):
        sections[i] = latest[i]
    
    # Verify capacity constraints
    cnt = [0] * (T + 2)
    for i in range(n):
        cnt[sections[i]] += 1
    
    for s in range(1, T + 1):
        if cnt[s] > M:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, sections)))

if __name__ == "__main__":
    solve()
