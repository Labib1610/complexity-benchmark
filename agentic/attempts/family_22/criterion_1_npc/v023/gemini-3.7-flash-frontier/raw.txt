import sys
from collections import deque
import random
import time

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    if n > M * T:
        print("NO")
        return
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    out_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        out_degree[u] += 1

    # Topological sort & cycle check
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo_order = []
    temp_in = list(in_degree)
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            temp_in[v] -= 1
            if temp_in[v] == 0:
                q.append(v)
                
    if len(topo_order) < n:
        print("NO")
        return
    
    # Compute EST (Earliest Start Time)
    est = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if est[u] + 1 > est[v]:
                est[v] = est[u] + 1
                
    # Compute LST (Latest Start Time)
    lst = [T] * (n + 1)
    for u in reversed(topo_order):
        for v in adj[u]:
            if lst[v] - 1 < lst[u]:
                lst[u] = lst[v] - 1
                
    for i in range(1, n + 1):
        if est[i] > lst[i] or lst[i] < 1 or est[i] > T:
            print("NO")
            return

    # Compute depth (longest path to sink)
    depth = [0] * (n + 1)
    for u in reversed(topo_order):
        d = 0
        for v in adj[u]:
            if depth[v] + 1 > d:
                d = depth[v] + 1
        depth[u] = d

    # Number of descendants
    desc_count = [0] * (n + 1)
    # Bitset / set for descendants if n is reasonable, else approximate
    if n <= 2000:
        desc_sets = [set() for _ in range(n + 1)]
        for u in reversed(topo_order):
            for v in adj[u]:
                desc_sets[u].add(v)
                desc_sets[u].update(desc_sets[v])
            desc_count[u] = len(desc_sets[u])
    else:
        for u in reversed(topo_order):
            c = len(adj[u])
            for v in adj[u]:
                c += desc_count[v]
            desc_count[u] = c

    # Coffman-Graham labels
    cg_label = [0] * (n + 1)
    # Sort successors by label descending
    for u in reversed(topo_order):
        # We can assign labels in a CG manner
        pass

    # Try list scheduling with a given priority key
    def run_list_scheduling(key_fn):
        cur_in = list(in_degree)
        ready = [i for i in range(1, n + 1) if cur_in[i] == 0]
        schedule = [0] * (n + 1)
        
        for phase in range(1, T + 1):
            if not ready:
                break
            ready.sort(key=key_fn, reverse=True)
            chosen = ready[:M]
            ready = ready[M:]
            
            for u in chosen:
                schedule[u] = phase
                for v in adj[u]:
                    cur_in[v] -= 1
                    if cur_in[v] == 0:
                        ready.append(v)
                        
        if all(schedule[i] > 0 for i in range(1, n + 1)):
            return schedule
        return None

    # Priority 1: LST asc, depth desc, desc_count desc
    sched = run_list_scheduling(lambda x: (-lst[x], depth[x], desc_count[x]))
    if sched:
        print("YES")
        print(*(sched[1:]))
        return

    # Priority 2: depth desc, LST asc, desc_count desc
    sched = run_list_scheduling(lambda x: (depth[x], -lst[x], desc_count[x]))
    if sched:
        print("YES")
        print(*(sched[1:]))
        return

    # Priority 3: desc_count desc, depth desc, -lst
    sched = run_list_scheduling(lambda x: (desc_count[x], depth[x], -lst[x]))
    if sched:
        print("YES")
        print(*(sched[1:]))
        return

    # Randomized list scheduling / local search
    start_time = time.time()
    while time.time() - start_time < 4.0:
        # Add random noise to priority
        w_lst = random.uniform(5.0, 15.0)
        w_dep = random.uniform(1.0, 5.0)
        w_des = random.uniform(0.1, 1.0)
        w_rnd = random.uniform(0.0, 2.0)
        
        sched = run_list_scheduling(lambda x: -w_lst * lst[x] + w_dep * depth[x] + w_des * desc_count[x] + random.random() * w_rnd)
        if sched:
            print("YES")
            print(*(sched[1:]))
            return
