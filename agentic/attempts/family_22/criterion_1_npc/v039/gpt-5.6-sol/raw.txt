```python
import sys
import heapq
import itertools
from collections import deque

sys.setrecursionlimit(1_000_000)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 4:
        return

    n, M, T, m = data[:4]

    if n == 0:
        print("YES")
        print()
        return

    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    succ_sets = [set() for _ in range(n)]
    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2
        if a != b:
            succ_sets[a].add(b)
        else:
            print("NO")
            return

    succ = [list(x) for x in succ_sets]
    pred = [[] for _ in range(n)]
    indeg = [0] * n
    for u in range(n):
        for v in succ[u]:
            pred[v].append(u)
            indeg[v] += 1

    q = deque(i for i in range(n) if indeg[i] == 0)
    topo = []
    deg = indeg[:]
    while q:
        u = q.popleft()
        topo.append(u)
        for v in succ[u]:
            deg[v] -= 1
            if deg[v] == 0:
                q.append(v)

    if len(topo) != n:
        print("NO")
        return

    def compute_bottom(adj, order):
        bottom = [1] * n
        for u in reversed(order):
            if adj[u]:
                bottom[u] = 1 + max(bottom[v] for v in adj[u])
        return bottom

    bottom_orig = compute_bottom(succ, topo)
    if max(bottom_orig) > T:
        print("NO")
        return

    if M == 1:
        if n > T:
            print("NO")
            return
        ans = [0] * n
        for t, u in enumerate(topo, 1):
            ans[u] = t
        print("YES")
        print(*ans)
        return

    if M >= n:
        ans = [1] * n
        for u in topo:
            if pred[u]:
                ans[u] = 1 + max(ans[v] for v in pred[u])
        if max(ans) <= T:
            print("YES")
            print(*ans)
        else:
            print("NO")
        return

    def list_schedule(adj, order, bottom):
        indegree = [0] * n
        for u in range(n):
            for v in adj[u]:
                indegree[v] += 1

        heap = []
        for u in range(n):
            if indegree[u] == 0:
                heapq.heappush(heap, (-bottom[u], -len(adj[u]), u))

        times = [0] * n
        completed = 0
        minute = 0

        while completed < n:
            minute += 1
            if minute > T or not heap:
                return None

            current = []
            for _ in range(min(M, len(heap))):
                _, _, u = heapq.heappop(heap)
                current.append(u)
                times[u] = minute

            completed += len(current)
            newly_available = []
            for u in current:
                for v in adj[u]:
                    indegree[v] -= 1
                    if indegree[v] == 0:
                        newly_available.append(v)

            for v in newly_available:
                heapq.heappush(heap, (-bottom[v], -len(adj[v]), v))

        return times

    ans = list_schedule(succ, topo, bottom_orig)
    if ans is not None:
        print("YES")
        print(*ans)
        return

    rev_succ = pred
    rev_topo = topo[::-1]
    bottom_rev = compute_bottom(rev_succ, rev_topo)

    rev_ans = list_schedule(rev_succ, rev_topo, bottom_rev)
    if rev_ans is not None:
        ans = [T + 1 - x for x in rev_ans]
        print("YES")
        print(*ans)
        return

    source_count = sum(1 for u in range(n) if not pred[u])
    sink_count = sum(1 for u in range(n) if not succ[u])

    if sink_count < source_count:
        adj = rev_succ
        order = rev_topo
        bottom = bottom_rev
        reversed_orientation = True
    else:
        adj = succ
        order = topo
        bottom = bottom_orig
        reversed_orientation = False

    pred_mask = [0] * n
    for u in range(n):
        bit = 1 << u
        for v in adj[u]:
            pred_mask[v] |= bit

    latest = [T - bottom[u] + 1 for u in range(n)]
    all_mask = (1 << n) - 1
    times = [0] * n
    failed = set()
    memo_limit = 500_000

    def dfs(done, minute):
        if done == all_mask:
            return True
        if minute > T:
            return False

        state = (done, minute)
        if state in failed:
            return False
        if len(failed) < memo_limit:
            failed.add(state)

        remaining_count = n - done.bit_count()
        if remaining_count > M * (T - minute + 1):
            return False

        rem = all_mask ^ done
        available = []
        forced = []
        deadline_counts = {}

        x = rem
        while x:
            b = x & -x
            u = b.bit_length() - 1
            x ^= b

            d = latest[u]
            if d < minute:
                return False
            deadline_counts[d] = deadline_counts.get(d, 0) + 1

            if pred_mask[u] & rem == 0:
                available.append(u)
                if d == minute:
                    forced.append(u)

        cumulative = 0
        for d in sorted(deadline_counts):
            cumulative += deadline_counts[d]
            if cumulative > M * (d - minute + 1):
                return False

        if not available:
            return False

        k = min(M, len(available))
        if len(forced) > k:
            return False

        forced_set = set(forced)
        optional = [u for u in available if u not in forced_set]
        optional.sort(key=lambda u: (latest[u], -len(adj[u]), u))
        need = k - len(forced)

        if need == len(optional):
            choices = (tuple(optional),)
        else:
            choices = itertools.combinations(optional, need)

        forced_mask = 0
        for u in forced:
            forced_mask |= 1 << u

        for extra in choices:
            chosen = list(forced)
            chosen.extend(extra)

            chosen_mask = forced_mask
            for u in extra:
                chosen_mask |= 1 << u

            for u in chosen:
                times[u] = minute

            if dfs(done | chosen_mask, minute + 1):
                return True

            for u in chosen:
                times[u] = 0

        return False

    if not dfs(0, 1):
        print("NO")
        return

    if reversed_orientation:
        ans = [T + 1 - x for x in times]
    else:
        ans = times

    print("YES")
    print(*ans)

if __name__ == "__main__":
    main()
```
