import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Binary search for minimum feasible max serving number
    lo, hi = 1, T
    best_T = -1
    best_assign = None
    
    while lo <= hi:
        mid = (lo + hi) // 2
        # Check if feasible with max serving = mid
        indeg_c = indeg[:]
        q = deque()
        for i in range(n):
            if indeg_c[i] == 0:
                q.append(i)
        
        assign = [0] * n
        valid = True
        serving = 1
        
        while q and serving <= mid:
            # Process all available nodes, but limit to M per serving
            batch = []
            while q and len(batch) < M:
                u = q.popleft()
                batch.append(u)
            
            for u in batch:
                assign[u] = serving
                for v in adj[u]:
                    indeg_c[v] -= 1
                    if indeg_c[v] == 0:
                        q.append(v)
            
            serving += 1
        
        # Check if all nodes assigned
        if all(assign[i] > 0 for i in range(n)):
            best_T = mid
            best_assign = assign[:]
            hi = mid - 1
        else:
            lo = mid + 1
    
    if best_T == -1:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, best_assign)))

if __name__ == "__main__":
    solve()
