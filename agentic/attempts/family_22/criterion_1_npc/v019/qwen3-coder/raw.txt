from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort with Kahn's algorithm
    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    topo_order = []
    while queue:
        node = queue.popleft()
        topo_order.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # Check if all nodes were visited (no cycles)
    if len(topo_order) != n:
        print("NO")
        return
    
    # Assign servings
    serving = [0] * (n + 1)
    current_serving = 1
    index = 0
    
    while index < n:
        # Count how many nodes we can place in current serving
        count = 0
        temp_index = index
        
        # Go through remaining nodes in topological order
        # Place as many as possible in current serving without violating dependencies
        while temp_index < n and count < M:
            node = topo_order[temp_index]
            # Check if all dependencies are already placed
            valid = True
            for dep in graph:
                if node in graph[dep]:
                    if serving[dep] == 0 or serving[dep] >= current_serving:
                        valid = False
                        break
            if valid:
                serving[node] = current_serving
                count += 1
                if count == M:
                    break
            temp_index += 1
        
        # If we couldn't place any node in this serving, it's impossible
        if count == 0:
            print("NO")
            return
            
        # Move to next serving
        current_serving += 1
        if current_serving > T:
            print("NO")
            return
            
        # Actually update the index by placing nodes
        placed = 0
        temp_idx = index
        while temp_idx < n and placed < M:
            node = topo_order[temp_idx]
            # Check if all dependencies are already placed in earlier servings
            valid = True
            for dep in graph:
                for neighbor in graph[dep]:
                    if neighbor == node and (serving[dep] == 0 or serving[dep] >= serving[node]):
                        valid = False
                        break
                if not valid:
                    break
            if not valid:
                temp_idx += 1
                continue
                
            # Also check direct dependencies in graph
            can_place = True
            for dep_node in range(1, n+1):
                if node in graph[dep_node]:  # node depends on dep_node
                    if serving[dep_node] == 0 or serving[dep_node] >= current_serving:
                        can_place = False
                        break
                        
            if can_place:
                serving[node] = current_serving
                placed += 1
            temp_idx += 1
        
        # Better approach: process nodes in order, place up to M in current serving
        placed = 0
        new_index = index
        for i in range(index, n):
            node = topo_order[i]
            if serving[node] == 0:  # Not yet assigned
                # Check if all predecessors are assigned to earlier servings
                can_assign = True
                for dep in graph:
                    if node in graph[dep]:
                        if serving[dep] == 0 or serving[dep] >= current_serving:
                            can_assign = False
                            break
                if can_assign:
                    serving[node] = current_serving
                    placed += 1
                    if placed == M:
                        # Find the next unassigned node
                        j = i + 1
                        while j < n and serving[topo_order[j]] != 0:
                            j += 1
                        new_index = j
                        break
                else:
                    new_index = i
                    break
            else:
                new_index = i + 1
                
        if placed == 0 and index < n:
            print("NO")
            return
            
        index = new_index if new_index < n else n
        
    # Check if we used too many servings
    max_serving = max(serving[1:])
    if max_serving > T:
        print("NO")
        return
        
    # Verify solution
    for a in range(1, n+1):
        for b in graph[a]:
            if serving[a] >= serving[b]:
                print("NO")
                return
    
    # Count nodes per serving
    serving_count = [0] * (max_serving + 1)
    for i in range(1, n+1):
        serving_count[serving[i]] += 1
        
    for count in serving_count[1:]:
        if count > M:
            print("NO")
            return
    
    print("YES")
    print(*serving[1:])

solve()
