import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    if n > T * M:
        print("NO")
        return
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        
    # Check for cycles and compute topological order
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)
                
    if len(topo) < n:
        print("NO")
        return
    
    # Compute L[u] (earliest possible day for u)
    L = [1] * (n + 1)
    for u in topo:
        for v in adj[u]:
            if L[u] + 1 > L[v]:
                L[v] = L[u] + 1
                
    # Compute R[u] (latest possible day for u)
    R = [T] * (n + 1)
    for u in reversed(topo):
        for p in rev_adj[u]:
            if R[u] - 1 < R[p]:
                R[p] = R[u] - 1
                
    for i in range(1, n + 1):
        if L[i] > R[i] or L[i] > T or R[i] < 1:
            print("NO")
            return

    # Height from u (longest path to a sink)
    height = [0] * (n + 1)
    for u in reversed(topo):
        for v in adj[u]:
            if height[v] + 1 > height[u]:
                height[u] = height[v] + 1

    # Backtracking / List Scheduling with EDF heuristic
    # Predecessor counts remaining
    rem_preds = [len(rev_adj[i]) for i in range(n + 1)]
    assigned_day = [0] * (n + 1)
    
    # Ready nodes
    ready = [i for i in range(1, n + 1) if rem_preds[i] == 0]
    
    def get_priority(u):
        return (R[u], -height[u], -L[u])
    
    ready.sort(key=get_priority)
    
    import heapq
    import itertools
    
    # State for search
    # We can schedule day by day
    def search(day, available, unscheduled_count):
        if unscheduled_count == 0:
            return True
        if day > T:
            return False
        
        # Check deadline capacity feasibility
        # Count remaining unscheduled with R[u] <= k
        # Must have count <= (k - day + 1) * M for all k in [day, T]
        # Quick check:
        r_counts = [0] * (T + 2)
        for u in range(1, n + 1):
            if assigned_day[u] == 0:
                r_counts[R[u]] += 1
        
        running_sum = 0
        for k in range(day, T + 1):
            running_sum += r_counts[k]
            if running_sum > (k - day + 1) * M:
                return False
        if r_counts[day - 1] > 0: # already missed deadline
            return False

        # Nodes that MUST be scheduled on or before `day` (actually R[u] == day)
        must_schedule = [u for u in available if R[u] <= day]
        if len(must_schedule) > M:
            return False
            
        other_available = [u for u in available if R[u] > day]
        
        # We need to pick a subset of size <= M containing all `must_schedule`
        needed = M - len(must_schedule)
        
        # Heuristic: try the best `needed` items from `other_available`
        candidates_pool = other_available[:max(needed + 2, 0)]
        
        # Try best choices first
        from itertools import combinations
        if len(other_available) <= needed:
            choice_subsets = [other_available]
        else:
            # Generate combinations of size up to needed from candidates_pool
            # Prefer largest size = needed first
            choice_subsets = []
            for sz in range(needed, -1, -1):
                for comb in combinations(candidates_pool, sz):
                    choice_subsets.append(list(comb))
                    if len(choice_subsets) > 10:
                        break
                if choice_subsets:
                    break
        
        for extra in choice_subsets:
            chosen = must_schedule + extra
            # Assign day
            for u in chosen:
                assigned_day[u] = day
            
            # Update next available
            next_available = [u for u in available if u not in chosen]
            newly_ready = []
            for u in chosen:
                for v in adj[u]:
                    rem_preds[v] -= 1
                    if rem_preds[v] == 0:
                        newly_ready.append(v)
            
            next_available.extend(newly_ready)
            next_available.sort(key=get_priority)
            
            if search(day + 1, next_available, unscheduled_count - len(chosen)):
                return True
                
            # Backtrack
            for u in chosen:
                assigned_day[u] = 0
                for v in adj[u]:
                    rem_preds[v] += 1
                    
        return False

    sys.setrecursionlimit(200000)
    if search(1, ready, n):
        print("YES")
        print(*(assigned_day[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
