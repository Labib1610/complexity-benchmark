import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    g = [[] for _ in range(n)]
    rg = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(m):
        a = int(next(it))-1
        b = int(next(it))-1
        g[a].append(b)
        rg[b].append(a)
        indeg[b] += 1

    # topological order
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    while q:
        v = q.popleft()
        topo.append(v)
        for to in g[v]:
            indeg[to] -= 1
            if indeg[to]==0:
                q.append(to)
    if len(topo)!=n:
        print("NO")
        return

    # earliest (longest path from sources) 1‑based
    earliest = [1]*n
    for v in topo:
        for to in g[v]:
            if earliest[to] < earliest[v] + 1:
                earliest[to] = earliest[v] + 1

    # longest distance to a sink
    dist_end = [0]*n
    for v in reversed(topo):
        for to in g[v]:
            if dist_end[v] < dist_end[to] + 1:
                dist_end[v] = dist_end[to] + 1

    latest = [T - dist_end[i] for i in range(n)]

    for i in range(n):
        if earliest[i] > latest[i] or earliest[i] > T or latest[i] < 1:
            print("NO")
            return

    # ---------- Dinic ----------
    class Edge:
        __slots__ = ('to','cap','rev')
        def __init__(self, to, cap, rev):
            self.to = to
            self.cap = cap
            self.rev = rev

    class Dinic:
        def __init__(self, N):
            self.N = N
            self.g = [[] for _ in range(N)]

        def add_edge(self, fr, to, cap):
            fwd = Edge(to, cap, len(self.g[to]))
            rev = Edge(fr, 0, len(self.g[fr]))
            self.g[fr].append(fwd)
            self.g[to].append(rev)

        def bfs_level(self, s, t):
            level = [-1]*self.N
            q = deque([s])
            level[s] = 0
            while q:
                v = q.popleft()
                for e in self.g[v]:
                    if e.cap and level[e.to] < 0:
                        level[e.to] = level[v] + 1
                        q.append(e.to)
            self.level = level
            return level[t] >= 0

        def dfs(self, v, t, f):
            if v == t:
                return f
            for i in range(self.it[v], len(self.g[v])):
                e = self.g[v][i]
                if e.cap and self.level[v] < self.level[e.to]:
                    ret = self.dfs(e.to, t, min(f, e.cap))
                    if ret:
                        e.cap -= ret
                        self.g[e.to][e.rev].cap += ret
                        return ret
                self.it[v] += 1
            return 0

        def max_flow(self, s, t):
            flow = 0
            INF = 10**9
            while self.bfs_level(s, t):
                self.it = [0]*self.N
                while True:
                    f = self.dfs(s, t, INF)
                    if not f:
                        break
                    flow += f
            return flow

    SRC = 0
    TASK_START = 1
    WEEK_START = TASK_START + n
    SNK = WEEK_START + T
    dinic = Dinic(SNK+1)

    for i in range(n):
        dinic.add_edge(SRC, TASK_START+i, 1)

    for w in range(1, T+1):
        dinic.add_edge(WEEK_START + w-1, SNK, M)

    for i in range(n):
        for w in range(earliest[i], latest[i]+1):
            dinic.add_edge(TASK_START+i, WEEK_START + w-1, 1)

    flow = dinic.max_flow(SRC, SNK)
    if flow != n:
        print("NO")
        return

    # retrieve assignment
    ans = [0]*n
    for i in range(n):
        v = TASK_START + i
        for e in dinic.g[v]:
            if WEEK_START <= e.to < WEEK_START+T and e.cap == 0:  # used edge
                week = e.to - WEEK_START + 1
                ans[i] = week
                break

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
