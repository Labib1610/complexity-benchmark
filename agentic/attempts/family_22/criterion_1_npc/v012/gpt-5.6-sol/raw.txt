```python
import sys
import heapq
import itertools

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

n, M, T, m = data[:4]
g = [[] for _ in range(n)]
rg = [[] for _ in range(n)]
pred_mask = [0] * n
indeg = [0] * n

p = 4
for _ in range(m):
    a = data[p] - 1
    b = data[p + 1] - 1
    p += 2
    if not ((pred_mask[b] >> a) & 1):
        g[a].append(b)
        rg[b].append(a)
        pred_mask[b] |= 1 << a
        indeg[b] += 1

if n == 0:
    print("YES")
    print()
    sys.exit()

if M <= 0 or T <= 0 or n > M * T:
    print("NO")
    sys.exit()

# Topological order
deg = indeg[:]
q = [i for i in range(n) if deg[i] == 0]
heapq.heapify(q)
topo = []

while q:
    v = heapq.heappop(q)
    topo.append(v)
    for u in g[v]:
        deg[u] -= 1
        if deg[u] == 0:
            heapq.heappush(q, u)

if len(topo) != n:
    print("NO")
    sys.exit()

# Longest path ending at / starting from each vertex.
top_depth = [1] * n
for v in topo:
    nv = top_depth[v] + 1
    for u in g[v]:
        if top_depth[u] < nv:
            top_depth[u] = nv

bottom_depth = [1] * n
for v in reversed(topo):
    best = 1
    for u in g[v]:
        cand = bottom_depth[u] + 1
        if cand > best:
            best = cand
    bottom_depth[v] = best

if max(top_depth) > T:
    print("NO")
    sys.exit()


def output_answer(ans):
    print("YES")
    print(" ".join(map(str, ans)))
    sys.exit()


# Easy complete cases.
if M == 1 or T >= n:
    ans = [0] * n
    for week, v in enumerate(topo, 1):
        ans[v] = week
    if week <= T:
        output_answer(ans)

if M >= n:
    output_answer(top_depth)


def forward_greedy(primary, tie):
    d = indeg[:]
    heap = []
    for v in range(n):
        if d[v] == 0:
            heapq.heappush(heap, (-primary[v], -tie[v], v))

    ans = [0] * n
    done = 0

    for week in range(1, T + 1):
        if not heap:
            break

        batch = []
        for _ in range(min(M, len(heap))):
            _, _, v = heapq.heappop(heap)
            batch.append(v)
            ans[v] = week

        done += len(batch)
        for v in batch:
            for u in g[v]:
                d[u] -= 1
                if d[u] == 0:
                    heapq.heappush(heap, (-primary[u], -tie[u], u))

        if done == n:
            return ans

    return None


def backward_greedy(primary, tie):
    outdeg = [len(g[v]) for v in range(n)]
    heap = []
    for v in range(n):
        if outdeg[v] == 0:
            heapq.heappush(heap, (-primary[v], -tie[v], v))

    ans = [0] * n
    done = 0

    for week in range(T, 0, -1):
        if not heap:
            break

        batch = []
        for _ in range(min(M, len(heap))):
            _, _, v = heapq.heappop(heap)
            batch.append(v)
            ans[v] = week

        done += len(batch)
        for v in batch:
            for u in rg[v]:
                outdeg[u] -= 1
                if outdeg[u] == 0:
                    heapq.heappush(heap, (-primary[u], -tie[u], u))

        if done == n:
            return ans

    return None


outdegree = [len(g[v]) for v in range(n)]
hash_priority = [
    ((i + 1) * 11995408973635179863) & ((1 << 64) - 1)
    for i in range(n)
]

attempts = [
    forward_greedy(bottom_depth, outdegree),
    forward_greedy(bottom_depth, hash_priority),
    forward_greedy(outdegree, bottom_depth),
    backward_greedy(top_depth, indeg),
    backward_greedy(top_depth, hash_priority),
    backward_greedy(indeg, top_depth),
]

for ans in attempts:
    if ans is not None:
        output_answer(ans)

# Exact search over non-delay schedules. It is reached only when the fast
# constructions fail.
full_mask = (1 << n) - 1
initial_available = 0
for v in range(n):
    if indeg[v] == 0:
        initial_available |= 1 << v

failed = set()
path = []


def dfs(mask, available, week):
    if mask == full_mask:
        return True
    if week > T or available == 0:
        return False

    remaining_count = n - mask.bit_count()
    remaining_weeks = T - week + 1
    if remaining_count > remaining_weeks * M:
        return False

    state = (mask, week)
    if state in failed:
        return False

    remaining = full_mask ^ mask
    deadlines = []
    x = remaining
    while x:
        bit = x & -x
        v = bit.bit_length() - 1
        if bottom_depth[v] > remaining_weeks:
            failed.add(state)
            return False
        deadlines.append(T - bottom_depth[v] + 1)
        x -= bit

    deadlines.sort()
    for k, deadline in enumerate(deadlines, 1):
        if deadline < week or k > M * (deadline - week + 1):
            failed.add(state)
            return False

    available_vertices = []
    x = available
    while x:
        bit = x & -x
        available_vertices.append(bit.bit_length() - 1)
        x -= bit

    available_vertices.sort(
        key=lambda v: (bottom_depth[v], outdegree[v], hash_priority[v]),
        reverse=True
    )

    take = min(M, len(available_vertices))

    if take == len(available_vertices):
        selections = (tuple(available_vertices),)
    else:
        selections = itertools.combinations(available_vertices, take)

    for chosen in selections:
        selected_mask = 0
        for v in chosen:
            selected_mask |= 1 << v

        new_mask = mask | selected_mask
        new_available = available & ~selected_mask

        for v in chosen:
            for u in g[v]:
                bit_u = 1 << u
                if not (new_mask & bit_u) and (pred_mask[u] & ~new_mask) == 0:
                    new_available |= bit_u

        path.append(selected_mask)
        if dfs(new_mask, new_available, week + 1):
            return True
        path.pop()

    failed.add(state)
    return False


if dfs(0, initial_available, 1):
    ans = [0] * n
    for week, selected in enumerate(path, 1):
        x = selected
        while x:
            bit = x & -x
            ans[bit.bit_length() - 1] = week
            x -= bit
    output_answer(ans)

print("NO")
```
