import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    radj = [[] for _ in range(n + 1)]
    in_deg = [0] * (n + 1)
    out_deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        radj[v].append(u)
        in_deg[v] += 1
        out_deg[u] += 1
        
    # Topological sort to compute ES and detect cycles
    q = deque([i for i in range(1, n + 1) if in_deg[i] == 0])
    topo = []
    es = [1] * (n + 1)
    
    in_d = list(in_deg)
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            if es[u] + 1 > es[v]:
                es[v] = es[u] + 1
            in_d[v] -= 1
            if in_d[v] == 0:
                q.append(v)
                
    if len(topo) < n:
        print("NO")
        return
        
    for i in range(1, n + 1):
        if es[i] > T:
            print("NO")
            return
            
    # Compute LS
    ls = [T] * (n + 1)
    for u in reversed(topo):
        for v in adj[u]:
            if ls[v] - 1 < ls[u]:
                ls[u] = ls[v] - 1
        if ls[u] < es[u] or ls[u] < 1:
            print("NO")
            return

    # Count ancestors / descendants to break ties
    anc_count = [0] * (n + 1)
    for u in topo:
        for v in adj[u]:
            anc_count[v] = max(anc_count[v], anc_count[u] + 1)

    # Check interval capacity bounds
    # For all 1 <= t1 <= t2 <= T, count of nodes with [es, ls] inside [t1, t2] <= (t2 - t1 + 1) * M
    for t1 in range(1, T + 1):
        for t2 in range(t1, T + 1):
            cnt = sum(1 for i in range(1, n + 1) if es[i] >= t1 and ls[i] <= t2)
            if cnt > (t2 - t1 + 1) * M:
                print("NO")
                return

    # Backtracking backward from week T down to 1
    # At week t, schedule ready tasks (all successors scheduled in weeks > t)
    assigned_week = [0] * (n + 1)
    curr_out_deg = [len(adj[i]) for i in range(n + 1)]
    
    # Priority for task: smaller ls, larger es, larger anc_count
    def task_priority(u):
        return (ls[u], -es[u], -anc_count[u], -len(radj[u]))

    import time
    start_time = time.time()
    
    def backtrack(t, ready_list):
        if time.time() - start_time > 4.5:
            return False
        if t == 0:
            return all(assigned_week[i] > 0 for i in range(1, n + 1))
        
        # Check mandatory tasks that must be scheduled at or after t
        # In backward scheduling at week t, any ready task with ls[u] >= t must be scheduled <= ls[u].
        # Specifically, if ls[u] == t, it MUST be scheduled at week t.
        mandatory = []
        optional = []
        for u in ready_list:
            if es[u] > t:
                # Cannot be scheduled at or before t
                return False
            if ls[u] == t:
                mandatory.append(u)
            else:
                optional.append(u)
                
        if len(mandatory) > M:
            return False
            
        # Optional tasks sorted by urgency
        optional.sort(key=task_priority)
        
        needed = M - len(mandatory)
        
        # Candidate subsets of optional tasks to pick
        # Heuristic: mostly pick the best `needed` tasks, but if that fails, try slight variations
        from itertools import combinations
        
        candidates_pool = optional[:needed + 3]
        
        chosen_combinations = []
        if len(optional) <= needed:
            chosen_combinations.append(tuple(optional))
        else:
            # First try the greedy choice
            chosen_combinations.append(tuple(optional[:needed]))
            # Then other combinations from candidate pool
            for comb in combinations(candidates_pool, needed):
                if comb != tuple(optional[:needed]):
                    chosen_combinations.append(comb)
                    if len(chosen_combinations) > 10:
                        break
                        
        for opt_chosen in chosen_combinations:
            chosen = mandatory + list(opt_chosen)
            
            # Apply choices
            new_ready = [u for u in ready_list if u not in chosen]
            for u in chosen:
                assigned_week[u] = t
                for p in radj[u]:
                    curr_out_deg[p] -= 1
                    if curr_out_deg[p] == 0:
                        new_ready.append(p)
                        
            # Check feasibility for new_ready
            feasible = True
            for u in new_ready:
                if es[u] > t - 1:
                    feasible = False
                    break
            
            if feasible and backtrack(t - 1, new_ready):
                return True
                
            # Revert
            for u in chosen:
                assigned_week[u] = 0
                for p in radj[u]:
                    curr_out_deg[p] += 1
                    
        return False

    initial_ready = [i for i in range(1, n + 1) if curr_out_deg[i] == 0]
    
    if backtrack(T, initial_ready):
        print("YES")
        print(*(assigned_week[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
