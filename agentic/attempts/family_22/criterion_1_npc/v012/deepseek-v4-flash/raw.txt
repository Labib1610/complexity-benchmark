We can model this as a scheduling problem on a DAG (directed acyclic graph) with resource constraints.  
We need to assign each node a week from 1 to T such that:

- For every edge (a → b), week[a] < week[b].
- No week has more than M nodes.

We can solve it by:

1. Topologically sorting the DAG.
2. Greedily assigning each node the earliest possible week, respecting both dependencies and the crew limit.

If any node cannot be assigned within T weeks, output `NO`.

```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Topological sort
    q = deque([i for i in range(n) if indeg[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    week = [0] * n
    # Count how many tasks assigned to each week
    week_count = [0] * (T + 1)
    
    for u in topo:
        # Earliest possible week = max(week[predecessor] + 1)
        earliest = 1
        for v in adj[u]:
            # Actually we need predecessors, so we need reverse edges
            pass
        # We'll compute using reverse adjacency
        # Let's build reverse adjacency first
        # But we can also compute by scanning all edges
        # Better: build rev_adj
        # We'll do it before loop
    
    # Build reverse adjacency
    rev_adj = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            rev_adj[v].append(u)
    
    for u in topo:
        earliest = 1
        for p in rev_adj[u]:
            earliest = max(earliest, week[p] + 1)
        # Find the first week >= earliest with count < M
        w = earliest
        while w <= T and week_count[w] >= M:
            w += 1
        if w > T:
            print("NO")
            return
        week[u] = w
        week_count[w] += 1
    
    print("YES")
    print(" ".join(str(week[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
