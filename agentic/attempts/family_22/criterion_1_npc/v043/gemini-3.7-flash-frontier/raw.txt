```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_deg = [0] * (n + 1)
    out_deg = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        rev_adj[v].append(u)
        out_deg[u] += 1
        in_deg[v] += 1

    # Topological sort & cycle check
    q = deque([i for i in range(1, n + 1) if in_deg[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            in_deg[v] -= 1
            if in_deg[v] == 0:
                q.append(v)
                
    if len(topo) < n:
        print("NO")
        return

    # Earliest shift (ASAP)
    E = [1] * (n + 1)
    for u in topo:
        for v in adj[u]:
            if E[u] + 1 > E[v]:
                E[v] = E[u] + 1

    # Latest shift (ALAP)
    L = [T] * (n + 1)
    for u in reversed(topo):
        for v in rev_adj[u]:
            if L[u] - 1 < L[v]:
                L[v] = L[u] - 1

    for i in range(1, n + 1):
        if E[i] > L[i] or E[i] > T or L[i] < 1:
            print("NO")
            return

    # Pre-calculate transitive successors count / height for heuristic
    height = [0] * (n + 1)
    for u in reversed(topo):
        h = 0
        for v in adj[u]:
            if height[v] + 1 > h:
                h = height[v] + 1
        height[u] = h

    # Forward search / Backtracking
    # We assign shifts from 1 to T.
    # At shift t, we pick at most M ready tasks.
    
    pred_count = [len(rev_adj[i]) for i in range(n + 1)]
    assignment = [0] * (n + 1)
    
    # Ready queue
    ready = [i for i in range(1, n + 1) if pred_count[i] == 0]
    
    def search(shift, ready_nodes, assigned_count):
        if assigned_count == n:
            return True
        if shift > T:
            return False
        
        # Pruning: remaining shifts * M must be >= remaining nodes
        rem_nodes = n - assigned_count
        if (T - shift + 1) * M < rem_nodes:
            return False
        
        # Check mandatory nodes that MUST be scheduled at or before this shift
        # i.e., L[u] <= shift. If their count > M, impossible.
        mandatory = [u for u in ready_nodes if L[u] <= shift]
        if len(mandatory) > M:
            return False
        
        # If not enough ready nodes to fill or we choose a subset
        # Sort ready nodes: prioritize those with smaller L[u], then larger height
        ready_sorted = sorted(ready_nodes, key=lambda x: (L[x], -height[x]))
        
        # Greedy choice: pick up to M best nodes
        # If we pick all mandatory nodes + best remaining up to M
        chosen = ready_sorted[:M]
        
        # Apply choice
        for u in chosen:
            assignment[u] = shift
            for v in adj[u]:
                pred_count[v] -= 1
                
        next_ready = [u for u in ready_nodes if u not in chosen]
        for u in chosen:
            for v in adj[u]:
                if pred_count[v] == 0:
                    next_ready.append(v)
                    
        if search(shift + 1, next_ready, assigned_count + len(chosen)):
            return True
            
        # Revert
        for u in chosen:
            assignment[u] = 0
            for v in adj[u]:
                pred_count[v] += 1
                
        # If greedy choice fails and we have choices, try subsets if M < len(ready_sorted)
        # But for most unit-task scheduling instances, the priority rule is sufficient.
        # If necessary, we can branch on other combinations.
        if len(ready_sorted) > M and len(mandatory) < M:
            import itertools
            mand_set = set(mandatory)
            other = [u for u in ready_sorted if u not in mand_set]
            needed = M - len(mandatory)
            
            # Try a few other top combinations
            count_comb = 0
            for extra in itertools.combinations(other, needed):
                cand = mandatory + list(extra)
                if cand == chosen:
                    continue
                count_comb += 1
                if count_comb > 10:
                    break
                    
                for u in cand:
                    assignment[u] = shift
                    for v in adj[u]:
                        pred_count[v] -= 1
                nr = [u for u in ready_nodes if u not in cand]
                for u in cand:
                    for v in adj[u]:
                        if pred_count[v] == 0:
                            nr.append(v)
                if search(shift + 1, nr, assigned_count + len(cand)):
                    return True
                for u in cand:
                    assignment[u] = 0
                    for v in adj[u]:
                        pred_count[v] += 1
                        
        return False

    if search(1, ready, 0):
        print("YES")
        print(*(assignment[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
