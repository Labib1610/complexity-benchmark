```python
import sys
import heapq
import itertools

sys.setrecursionlimit(1_000_000)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, M, T, m = data[:4]
    if n == 0:
        print("YES")
        print()
        return
    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    succ = [[] for _ in range(n)]
    pred = [[] for _ in range(n)]
    edges = set()

    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2
        if a == b:
            print("NO")
            return
        key = a * n + b
        if key not in edges:
            edges.add(key)
            succ[a].append(b)
            pred[b].append(a)

    indeg0 = [len(pred[v]) for v in range(n)]
    outdeg0 = [len(succ[v]) for v in range(n)]

    indeg = indeg0[:]
    heap = [v for v in range(n) if indeg[v] == 0]
    heapq.heapify(heap)
    topo = []

    while heap:
        v = heapq.heappop(heap)
        topo.append(v)
        for w in succ[v]:
            indeg[w] -= 1
            if indeg[w] == 0:
                heapq.heappush(heap, w)

    if len(topo) != n:
        print("NO")
        return

    top = [1] * n
    for v in topo:
        nv = top[v] + 1
        for w in succ[v]:
            if top[w] < nv:
                top[w] = nv

    down = [1] * n
    for v in reversed(topo):
        best = 1
        for w in succ[v]:
            val = down[w] + 1
            if val > best:
                best = val
        down[v] = best

    if max(down) > T:
        print("NO")
        return

    latest = [T - down[v] + 1 for v in range(n)]

    def forward_greedy(priority):
        indeg = indeg0[:]
        hp = []
        for v in range(n):
            if indeg[v] == 0:
                heapq.heappush(hp, (priority[v], v))

        answer = [0] * n
        done_count = 0
        slot = 1

        while hp and done_count < n and slot <= T:
            batch = []
            for _ in range(min(M, len(hp))):
                _, v = heapq.heappop(hp)
                batch.append(v)
                answer[v] = slot

            done_count += len(batch)
            newly = []
            for v in batch:
                for w in succ[v]:
                    indeg[w] -= 1
                    if indeg[w] == 0:
                        newly.append(w)
            for w in newly:
                heapq.heappush(hp, (priority[w], w))
            slot += 1

        return answer if done_count == n else None

    def backward_greedy(priority):
        outdeg = outdeg0[:]
        hp = []
        for v in range(n):
            if outdeg[v] == 0:
                heapq.heappush(hp, (priority[v], v))

        answer = [0] * n
        done_count = 0
        slot = T

        while hp and done_count < n and slot >= 1:
            batch = []
            for _ in range(min(M, len(hp))):
                _, v = heapq.heappop(hp)
                batch.append(v)
                answer[v] = slot

            done_count += len(batch)
            newly = []
            for v in batch:
                for u in pred[v]:
                    outdeg[u] -= 1
                    if outdeg[u] == 0:
                        newly.append(u)
            for u in newly:
                heapq.heappush(hp, (priority[u], u))
            slot -= 1

        return answer if done_count == n else None

    priorities_forward = [
        [(-down[v], -outdeg0[v], v) for v in range(n)],
        [(-outdeg0[v], -down[v], v) for v in range(n)],
        [(latest[v], -outdeg0[v], v) for v in range(n)],
        [((v * 11995408973635179863) & ((1 << 64) - 1),) for v in range(n)],
    ]

    for priority in priorities_forward:
        ans = forward_greedy(priority)
        if ans is not None:
            print("YES")
            print(*ans)
            return

    priorities_backward = [
        [(-top[v], -indeg0[v], v) for v in range(n)],
        [(-indeg0[v], -top[v], v) for v in range(n)],
        [(((v + 1) * 11400714819323198485) & ((1 << 64) - 1),) for v in range(n)],
    ]

    for priority in priorities_backward:
        ans = backward_greedy(priority)
        if ans is not None:
            print("YES")
            print(*ans)
            return

    pred_mask = [0] * n
    for v in range(n):
        mask = 0
        for u in pred[v]:
            mask |= 1 << u
        pred_mask[v] = mask

    all_mask = (1 << n) - 1
    answer = [0] * n
    failed = set()

    def dfs(done_mask, slot):
        if done_mask == all_mask:
            return True
        if slot > T:
            return False

        state = (done_mask, slot)
        if state in failed:
            return False

        remaining_count = n - done_mask.bit_count()
        if remaining_count > M * (T - slot + 1):
            failed.add(state)
            return False

        earliest = [0] * n
        available = []
        latest_values = []
        earliest_values = []

        for v in topo:
            if (done_mask >> v) & 1:
                continue

            e = slot
            for u in pred[v]:
                if not ((done_mask >> u) & 1):
                    ue = earliest[u] + 1
                    if ue > e:
                        e = ue

            if e > latest[v]:
                failed.add(state)
                return False

            earliest[v] = e
            earliest_values.append(e)
            latest_values.append(latest[v])

            if e == slot:
                available.append(v)

        if not available:
            failed.add(state)
            return False

        latest_values.sort()
        i = 0
        while i < len(latest_values):
            d = latest_values[i]
            while i < len(latest_values) and latest_values[i] == d:
                i += 1
            if i > M * (d - slot + 1):
                failed.add(state)
                return False

        earliest_values.sort(reverse=True)
        i = 0
        while i < len(earliest_values):
            d = earliest_values[i]
            while i < len(earliest_values) and earliest_values[i] == d:
                i += 1
            if i > M * (T - d + 1):
                failed.add(state)
                return False

        available.sort(key=lambda v: (latest[v], -down[v], -outdeg0[v], v))

        take = min(M, len(available))
        mandatory = [v for v in available if latest[v] == slot]
        if len(mandatory) > take:
            failed.add(state)
            return False

        mandatory_set = set(mandatory)
        optional = [v for v in available if v not in mandatory_set]
        need = take - len(mandatory)

        deadline_counts = {}
        for d in latest_values:
            deadline_counts[d] = deadline_counts.get(d, 0) + 1

        requirements = []
        cumulative = 0
        for d in sorted(deadline_counts):
            cumulative += deadline_counts[d]
            required_now = cumulative - M * (d - slot)
            if required_now > 0:
                requirements.append((d, required_now))

        if need < 0 or need > len(optional):
            failed.add(state)
            return False

        combinations = ((),) if need == 0 else itertools.combinations(optional, need)

        for chosen_optional in combinations:
            selected = mandatory + list(chosen_optional)

            valid = True
            for d, required_now in requirements:
                count = 0
                for v in selected:
                    if latest[v] <= d:
                        count += 1
                if count < required_now:
                    valid = False
                    break
            if not valid:
                continue

            new_mask = done_mask
            for v in selected:
                new_mask |= 1 << v
                answer[v] = slot

            if dfs(new_mask, slot + 1):
                return True

            for v in selected:
                answer[v] = 0

        failed.add(state)
        return False

    if dfs(0, 1):
        print("YES")
        print(*answer)
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
