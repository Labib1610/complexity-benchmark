```python
import sys
from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with room assignment
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    # room_assignment[i] = room number for relic i
    room_assignment = [0] * (n + 1)
    room_count = [0] * (T + 1)  # count of relics in each room
    
    # Process nodes in topological order
    while queue:
        current_relic = queue.popleft()
        
        # Determine the earliest room this relic can go in
        min_room = 1
        for prev in graph:
            for next_relic in graph[prev]:
                if next_relic == current_relic:
                    min_room = max(min_room, room_assignment[prev] + 1)
        
        # If we can't place it in any room, return NO
        placed = False
        for room in range(min_room, T + 1):
            if room_count[room] < M:
                room_assignment[current_relic] = room
                room_count[room] += 1
                placed = True
                break
        
        if not placed:
            print("NO")
            return
        
        # Update in-degrees of neighbors
        for neighbor in graph[current_relic]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                queue.append(neighbor)
    
    # Check if all relics were processed
    for i in range(1, n + 1):
        if room_assignment[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(*room_assignment[1:])

solve()
```
