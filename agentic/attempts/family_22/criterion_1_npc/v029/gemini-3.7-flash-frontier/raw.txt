import sys
from collections import deque
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        
    # Check for cycles and compute topological order
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo_order = []
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)
                
    if len(topo_order) < n:
        print("NO")
        return
        
    # Compute Earliest Start (ES) and Longest Path to Sink (depth)
    es = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if es[u] + 1 > es[v]:
                es[v] = es[u] + 1
                
    depth = [1] * (n + 1)
    ls = [T] * (n + 1)
    for u in reversed(topo_order):
        for v in adj[u]:
            if depth[v] + 1 > depth[u]:
                depth[u] = depth[v] + 1
        ls[u] = T - depth[u] + 1
        
    for u in range(1, n + 1):
        if es[u] > ls[u] or es[u] > T or ls[u] < 1:
            print("NO")
            return
            
    # Backtracking search to assign steps
    # We assign items step by step from 1 to T
    current_in_degree = [len(rev_adj[i]) for i in range(n + 1)]
    assigned_step = [0] * (n + 1)
    
    # Ready items initially
    ready = [i for i in range(1, n + 1) if current_in_degree[i] == 0]
    
    def check_feasibility(unscheduled, current_t):
        # Check volume bounds: for each t' from current_t to T, 
        # count items with ls <= t' must not exceed (t' - current_t + 1) * M
        counts = [0] * (T + 2)
        for u in unscheduled:
            if ls[u] < current_t:
                return False
            counts[ls[u]] += 1
        acc = 0
        for t_prime in range(current_t, T + 1):
            acc += counts[t_prime]
            if acc > (t_prime - current_t + 1) * M:
                return False
        return True

    def search(step, ready_list, unscheduled_set):
        if not unscheduled_set:
            return True
        if step > T:
            return False
        
        # Pruning check
        if not check_feasibility(unscheduled_set, step):
            return False
            
        must_schedule = [u for u in ready_list if ls[u] == step]
        if len(must_schedule) > M:
            return False
            
        # Any unscheduled item with ls <= step that is NOT in ready_list?
        for u in unscheduled_set:
            if ls[u] <= step and u not in ready_list:
                return False
                
        other_candidates = [u for u in ready_list if ls[u] > step]
        # Sort candidates by depth (descending), then ls (ascending)
        other_candidates.sort(key=lambda u: (-depth[u], ls[u]))
        
        needed = M - len(must_schedule)
        
        # Generate combinations of choices from other_candidates
        # Limit branching to prioritize greedy choice first
        max_to_pick = min(needed, len(other_candidates))
        
        # We can pick k items from other_candidates where k is up to max_to_pick
        # Usually, picking as many as possible (i.e. max_to_pick) is best
        # We iterate over possible counts k from max_to_pick down to 0
        for k in range(max_to_pick, -1, -1):
            # If total remaining capacity * remaining steps < remaining items, skip
            if (T - step) * M < len(unscheduled_set) - (len(must_schedule) + k):
                continue
                
            # Iterate combinations for this k
            # To avoid huge combination explosions, restrict combinations
            combs = itertools.combinations(other_candidates, k)
            count_comb = 0
            for comb in combs:
                count_comb += 1
                if count_comb > 50:  # Heuristic beam to prevent TLE
                    break
                chosen = list(must_schedule) + list(comb)
                
                # Apply choice
                for u in chosen:
                    assigned_step[u] = step
                new_unscheduled = unscheduled_set - set(chosen)
                
                newly_ready = []
                for u in chosen:
                    for v in adj[u]:
                        current_in_degree[v] -= 1
                        if current_in_degree[v] == 0:
                            newly_ready.append(v)
                            
                new_ready_list = [u for u in ready_list if u not in chosen] + newly_ready
                
                if search(step + 1, new_ready_list, new_unscheduled):
                    return True
                    
                # Revert
                for u in newly_ready:
                    for v_parent in rev_adj[u]:
                        pass # in degree will be restored below
                for u in chosen:
                    for v in adj[u]:
                        current_in_degree[v] += 1
                    assigned_step[u] = 0
                    
        return False

    unscheduled_all = set(range(1, n + 1))
    if search(1, ready, unscheduled_all):
        print("YES")
        print(*(assigned_step[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
