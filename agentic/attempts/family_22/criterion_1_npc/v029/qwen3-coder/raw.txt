```python
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    M = int(data[idx+1])
    T = int(data[idx+2])
    m = int(data[idx+3])
    idx += 4
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx+1])
        graph[a].append(b)
        in_degree[b] += 1
        idx += 2
    
    # Topological sort with level assignment
    queue = deque()
    level = [0] * (n + 1)
    
    # Find all nodes with in-degree 0
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
            level[i] = 1
    
    processed = 0
    while queue:
        current = queue.popleft()
        processed += 1
        current_level = level[current]
        
        if current_level > T:
            print("NO")
            return
            
        for neighbor in graph[current]:
            in_degree[neighbor] -= 1
            # Update level: neighbor must be at least one level after current
            if level[neighbor] < current_level + 1:
                level[neighbor] = current_level + 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    if processed != n:
        print("NO")
        return
    
    # Check if any level exceeds capacity
    level_count = [0] * (T + 1)
    for i in range(1, n + 1):
        level_count[level[i]] += 1
        
    for i in range(1, T + 1):
        if level_count[i] > M:
            print("NO")
            return
    
    # Output result
    print("YES")
    result = [str(level[i]) for i in range(1, n + 1)]
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
