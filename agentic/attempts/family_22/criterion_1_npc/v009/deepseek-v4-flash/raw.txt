We can model this as a scheduling problem with precedence constraints and a capacity limit per time slot.  
A straightforward approach is to use topological sorting with earliest start times, then check if we can fit all tasks within the deadline and capacity.

Here’s the Python solution:

```python
from collections import deque
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1
    
    # earliest start times
    earliest = [1] * (n + 1)
    q = deque()
    for i in range(1, n + 1):
        if indeg[i] == 0:
            q.append(i)
    
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            earliest[v] = max(earliest[v], earliest[u] + 1)
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    # Check if any earliest exceeds T
    if max(earliest[1:]) > T:
        print("NO")
        return
    
    # Now assign slots respecting capacity
    # We'll try to place tasks as early as possible, but if capacity is full, push later
    slot_count = [0] * (T + 1)
    slot = [0] * (n + 1)
    
    # Process in topological order
    for u in topo:
        s = earliest[u]
        while s <= T and slot_count[s] >= M:
            s += 1
        if s > T:
            print("NO")
            return
        slot[u] = s
        slot_count[s] += 1
    
    print("YES")
    print(" ".join(str(slot[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
