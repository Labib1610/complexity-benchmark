```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        in_degree[b] += 1

    # Topological sort
    q = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            q.append(i)

    topo_order = []
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)

    if len(topo_order) != n:
        print("NO")
        return

    # Find earliest possible cycle (1-indexed)
    min_cycle = [1] * (n + 1)
    cycle_at_minimal = [[] for _ in range(T + 2)] # Buckets for procedures at each cycle

    for u in topo_order:
        cu = min_cycle[u]
        if cu > T:
            print("NO")
            return
        cycle_at_minimal[cu].append(u)
        for v in adj[u]:
            if min_cycle[v] < cu + 1:
                min_cycle[v] = cu + 1

    # Fill slots greedily
    final_cycle = [0] * (n + 1)
    current_proc = 0
    procedures = topo_order
    num_procs = n

    for t in range(1, T + 1):
        capacity = M
        # First, schedule procedures that must run at this cycle (min_cycle == t)
        # We iterate through procedures not yet scheduled.
        # Since topo_order is ordered, we can scan from current_proc
        i = current_proc
        while i < num_procs and capacity > 0:
            u = procedures[i]
            if final_cycle[u] != 0:
                # Already scheduled (e.g. pulled earlier by a tighter min_cycle)
                # However, in the pure 'minimal' formulation, a node is only pulled
                # if its min_cycle is current time.
                # With the relaxation `min_cycle[v] = min(min_cycle[v], cu + 1)`,
                # a node u might become `min_cycle[u] == t` later.
                i += 1
                continue
            
            if min_cycle[u] == t:
                final_cycle[u] = t
                capacity -= 1
                # Update successors' min_cycle (relaxation)
                for v in adj[u]:
                    if min_cycle[v] > t + 1:
                        min_cycle[v] = t + 1
                # Optimization: swap to front to effectively remove from list
                # This helps skip over scheduled nodes.
                procedures[i], procedures[current_proc] = procedures[current_proc], procedures[i]
                current_proc += 1
                # Do not increment i, check the new value at position i
            else:
                i += 1
        
        if capacity == 0:
            continue
        
        # Second, pull procedures that are ready (min_cycle <= t)
        # Scan the remaining active procedures
        i = current_proc
        while i < num_procs and capacity > 0:
            u = procedures[i]
            if final_cycle[u] != 0:
                i += 1
                continue
            
            if min_cycle[u] <= t:
                final_cycle[u] = t
                capacity -= 1
                for v in adj[u]:
                    if min_cycle[v] > t + 1:
                        min_cycle[v] = t + 1
                procedures[i], procedures[current_proc] = procedures[current_proc], procedures[i]
                current_proc += 1
            else:
                i += 1

    if current_proc < num_procs:
        print("NO")
        return

    print("YES")
    output = []
    for i in range(1, n + 1):
        output.append(str(final_cycle[i]))
    print(' '.join(output))

if __name__ == '__main__':
    solve()
```
