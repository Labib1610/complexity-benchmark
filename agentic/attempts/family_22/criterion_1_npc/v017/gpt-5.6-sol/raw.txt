import sys
import heapq
from itertools import combinations

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 4:
        return

    n, M, T, m = data[:4]
    adj = [[] for _ in range(n)]
    radj = [[] for _ in range(n)]
    indeg = [0] * n
    outdeg = [0] * n

    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1
        outdeg[a] += 1

    if n == 0:
        print("YES")
        print()
        return

    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    # Topological ordering and cycle detection.
    deg = indeg[:]
    q = [i for i in range(n) if deg[i] == 0]
    topo = []
    head = 0
    while head < len(q):
        u = q[head]
        head += 1
        topo.append(u)
        for v in adj[u]:
            deg[v] -= 1
            if deg[v] == 0:
                q.append(v)

    if len(topo) != n:
        print("NO")
        return

    # Longest path lengths ending at / starting from each operation.
    top_depth = [1] * n
    for u in topo:
        du = top_depth[u] + 1
        for v in adj[u]:
            if top_depth[v] < du:
                top_depth[v] = du

    bottom_depth = [1] * n
    for u in reversed(topo):
        best = 1
        for v in adj[u]:
            cand = bottom_depth[v] + 1
            if cand > best:
                best = cand
        bottom_depth[u] = best

    if max(bottom_depth) > T:
        print("NO")
        return

    def tie_hash(v, seed):
        x = (v + 1) ^ (seed * 0x9E3779B1)
        x ^= x >> 16
        x = (x * 0x85EBCA6B) & 0xFFFFFFFF
        x ^= x >> 13
        return x

    def forward_schedule(mode, seed=0):
        d = indeg[:]
        heap = []

        def key(v):
            h = tie_hash(v, seed)
            if mode == 0:
                return (-bottom_depth[v], -outdeg[v], -top_depth[v], h, v)
            if mode == 1:
                return (-bottom_depth[v], -top_depth[v], -outdeg[v], h, v)
            if mode == 2:
                return (-outdeg[v], -bottom_depth[v], -top_depth[v], h, v)
            if mode == 3:
                return (-top_depth[v], -bottom_depth[v], -outdeg[v], h, v)
            return (-bottom_depth[v], h, -outdeg[v], v)

        for v in range(n):
            if d[v] == 0:
                heapq.heappush(heap, key(v))

        ans = [0] * n
        done = 0
        cycle = 1

        while done < n:
            if cycle > T or not heap:
                return None

            chosen = []
            for _ in range(min(M, len(heap))):
                chosen.append(heapq.heappop(heap)[-1])

            newly_available = []
            for u in chosen:
                ans[u] = cycle
                done += 1
                for v in adj[u]:
                    d[v] -= 1
                    if d[v] == 0:
                        newly_available.append(v)

            for v in newly_available:
                heapq.heappush(heap, key(v))
            cycle += 1

        return ans

    def backward_schedule(mode, seed=0):
        d = outdeg[:]
        heap = []

        def key(v):
            h = tie_hash(v, seed)
            if mode == 0:
                return (-top_depth[v], -indeg[v], -bottom_depth[v], h, v)
            if mode == 1:
                return (-top_depth[v], -bottom_depth[v], -indeg[v], h, v)
            if mode == 2:
                return (-indeg[v], -top_depth[v], -bottom_depth[v], h, v)
            if mode == 3:
                return (-bottom_depth[v], -top_depth[v], -indeg[v], h, v)
            return (-top_depth[v], h, -indeg[v], v)

        for v in range(n):
            if d[v] == 0:
                heapq.heappush(heap, key(v))

        ans = [0] * n
        done = 0
        cycle = T

        while done < n:
            if cycle < 1 or not heap:
                return None

            chosen = []
            for _ in range(min(M, len(heap))):
                chosen.append(heapq.heappop(heap)[-1])

            newly_available = []
            for u in chosen:
                ans[u] = cycle
                done += 1
                for v in radj[u]:
                    d[v] -= 1
                    if d[v] == 0:
                        newly_available.append(v)

            for v in newly_available:
                heapq.heappush(heap, key(v))
            cycle -= 1

        return ans

    work = n + m
    attempts = 4 if work > 300000 else 10

    for k in range(attempts):
        mode = k if k < 4 else 4
        seed = k * 1000003 + 17
        ans = forward_schedule(mode, seed)
        if ans is not None:
            print("YES")
            print(*ans)
            return

        ans = backward_schedule(mode, seed + 7919)
        if ans is not None:
            print("YES")
            print(*ans)
            return

    # Exact search for small hard instances.
    if n <= 22:
        pred_mask = [0] * n
        for v in range(n):
            mask = 0
            for u in radj[v]:
                mask |= 1 << u
            pred_mask[v] = mask

        full = (1 << n) - 1
        deadlines = [T - bottom_depth[v] + 1 for v in range(n)]
        failed = set()
        choice = {}

        def dfs(cycle, mask):
            if mask == full:
                return True
            if cycle > T:
                return False

            state = (cycle, mask)
            if state in failed:
                return False

            remaining_count = n - mask.bit_count()
            remaining_cycles = T - cycle + 1
            if remaining_count > M * remaining_cycles:
                failed.add(state)
                return False

            available = []
            forced = []
            for v in range(n):
                bit = 1 << v
                if mask & bit:
                    continue
                if deadlines[v] < cycle:
                    failed.add(state)
                    return False
                if pred_mask[v] & ~mask == 0:
                    available.append(v)
                    if deadlines[v] == cycle:
                        forced.append(v)

            if not available or len(forced) > M:
                failed.add(state)
                return False

            available.sort(
                key=lambda v: (bottom_depth[v], outdeg[v], top_depth[v]),
                reverse=True
            )
            forced_set = set(forced)
            optional = [v for v in available if v not in forced_set]

            take = min(M, len(available))
            need = take - len(forced)
            if need < 0:
                failed.add(state)
                return False

            if need == len(optional):
                candidates = (tuple(optional),)
            else:
                candidates = combinations(optional, need)

            forced_mask = 0
            for v in forced:
                forced_mask |= 1 << v

            for comb in candidates:
                selected_mask = forced_mask
                for v in comb:
                    selected_mask |= 1 << v

                new_mask = mask | selected_mask
                if dfs(cycle + 1, new_mask):
                    choice[state] = selected_mask
                    return True

            failed.add(state)
            return False

        if dfs(1, 0):
            ans = [0] * n
            cycle = 1
            mask = 0
            while mask != full:
                selected = choice[(cycle, mask)]
                bits = selected
                while bits:
                    bit = bits & -bits
                    v = bit.bit_length() - 1
                    ans[v] = cycle
                    bits -= bit
                mask |= selected
                cycle += 1

            print("YES")
            print(*ans)
            return

    print("NO")

if __name__ == "__main__":
    main()
