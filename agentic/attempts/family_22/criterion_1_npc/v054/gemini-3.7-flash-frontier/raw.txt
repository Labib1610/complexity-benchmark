```python
import sys
from collections import deque
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    out_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        out_degree[u] += 1
        idx += 2
        
    if n > M * T:
        print("NO")
        return
        
    # Check DAG and compute EST
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo = []
    est = [1] * (n + 1)
    
    in_deg_copy = list(in_degree)
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            if est[u] + 1 > est[v]:
                est[v] = est[u] + 1
            in_deg_copy[v] -= 1
            if in_deg_copy[v] == 0:
                q.append(v)
                
    if len(topo) < n:
        print("NO")
        return
        
    for i in range(1, n + 1):
        if est[i] > T:
            print("NO")
            return
            
    # Compute LST
    lst = [T] * (n + 1)
    for u in reversed(topo):
        for v in adj[u]:
            if lst[v] - 1 < lst[u]:
                lst[u] = lst[v] - 1
                
    for i in range(1, n + 1):
        if lst[i] < est[i] or lst[i] < 1:
            print("NO")
            return

    # Count descendants
    desc_count = [0] * (n + 1)
    for u in reversed(topo):
        desc = set()
        for v in adj[u]:
            desc.add(v)
            # small approximation or full set
        # accurate reachable count:
    # Full reachable sets
    reachable = [set() for _ in range(n + 1)]
    for u in reversed(topo):
        for v in adj[u]:
            reachable[u].add(v)
            reachable[u].update(reachable[v])
        desc_count[u] = len(reachable[u])

    # Try list scheduling heuristics
    # Direction 1: Forward (cycle 1 to T)
    # Direction 2: Backward (cycle T down to 1)

    def try_forward(priority_key):
        # priority_key(u) -> lower is higher priority
        # in_deg tracking
        in_deg = list(in_degree)
        ready = [u for u in range(1, n + 1) if in_deg[u] == 0]
        schedule = [0] * (n + 1)
        
        for t in range(1, T + 1):
            if not ready:
                break
            ready.sort(key=priority_key)
            chosen = []
            remaining = []
            
            # Tasks with LST == t MUST be scheduled now
            for u in ready:
                if lst[u] < t:
                    return None
            
            for u in ready:
                if len(chosen) < M:
                    chosen.append(u)
                else:
                    remaining.append(u)
            
            for u in remaining:
                if lst[u] <= t:
                    return None # couldn't schedule a critical task
            
            for u in chosen:
                schedule[u] = t
                for v in adj[u]:
                    in_deg[v] -= 1
                    if in_deg[v] == 0:
                        remaining.append(v)
            ready = remaining
            
        if all(schedule[i] > 0 for i in range(1, n + 1)):
            return schedule
        return None

    def try_backward(priority_key):
        out_deg = list(out_degree)
        ready = [u for u in range(1, n + 1) if out_deg[u] == 0]
        schedule = [0] * (n + 1)
        
        for t in range(T, 0, -1):
            if not ready:
                break
            ready.sort(key=priority_key)
            chosen = []
            remaining = []
            
            for u in ready:
                if est[u] > t:
                    return None
                    
            for u in ready:
                if len(chosen) < M:
                    chosen.append(u)
                else:
                    remaining.append(u)
                    
            for u in remaining:
                if est[u] >= t:
                    return None
                    
            for u in chosen:
                schedule[u] = t
                for v in rev_adj[u]:
                    out_deg[v] -= 1
                    if out_deg[v] == 0:
                        remaining.append(v)
            ready = remaining
            
        if all(schedule[i] > 0 for i in range(1, n + 1)):
            return schedule
        return None

    # Heuristics for forward:
    heuristics_fwd = [
        lambda u: (lst[u], -desc_count[u], est[u]),
        lambda u: (lst[u], est[u], -desc_count[u]),
        lambda u: (-desc_count[u], lst[u], est[u]),
        lambda u: (lst[u], -len(adj[u])),
    ]

    for h in heuristics_fwd:
        res = try_forward(h)
        if res:
            print("YES")
            print(*(res[1:]))
            return

    # Heuristics for backward:
    heuristics_bwd = [
        lambda u: (-est[u], -len(rev_adj[u]), -lst[u]),
        lambda u: (-est[u], lst[u]),
    ]

    for h in heuristics_bwd:
        res = try_backward(h)
        if res:
            print("YES")
            print(*(res[1:]))
            return

    # Randomized list scheduling
    random.seed(42)
    for _ in range(500):
        # Add random noise to lst
        noise = {u: random.random() for u in range(1, n + 1)}
        res = try_forward(lambda u: (lst[u], -desc_count[u] + noise[u], noise[u]))
        if res:
            print("YES")
            print(*(res[1:]))
            return
        res = try_backward(lambda u: (-est[u] + noise[u], -len(rev_adj[u]), noise[u]))
        if res:
            print("YES")
            print(*(res[1:]))
            return

    # Backtracking search if not solved yet
    # Forward backtracking
    def backtrack():
        in_deg = list(in_degree)
        ready = [u for u in range(1, n + 1) if in_deg[u] == 0]
        schedule = [0] * (n + 1)
        
        # We can implement a limited depth/states backtrack
        import itertools
        
        def dfs(t, cur_ready, cur_in_deg):
            if not cur_ready:
                return True
            if t > T:
                return False
                
            for u in cur_ready:
                if lst[u] < t:
                    return False
                    
            # Must pick all tasks with lst[u] == t
            must_pick = [u for u in cur_ready if lst[u] == t]
            if len(must_pick) > M:
                return False
                
            optional = [u for u in cur_ready if lst[u] > t]
            optional.sort(key=lambda u: (lst[u], -desc_count[u]))
            
            needed = min(M - len(must_pick), len(optional))
            
            # Generate combinations of optional tasks of size needed
            # To avoid huge branching, limit combinations
            from itertools import combinations
            # If many optional, just consider the top ones
            cand_optional = optional[:min(len(optional), needed + 4)]
            
            for extra in combinations(cand_optional, needed):
                chosen = must_pick + list(extra)
                new_ready = [u for u in cur_ready if u not in chosen]
                new_in_deg = list(cur_in_deg)
                
                for u in chosen:
                    schedule[u] = t
                    for v in adj[u]:
                        new_in_deg[v] -= 1
                        if new_in_deg[v] == 0:
                            new_ready.append(v)
                            
                if dfs(t + 1, new_ready, new_in_deg):
                    return True
                    
                for u in chosen:
                    schedule[u] = 0
                    
            return False

        if dfs(1, ready, in_deg):
            return schedule
        return None

    res = backtrack()
    if res:
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
