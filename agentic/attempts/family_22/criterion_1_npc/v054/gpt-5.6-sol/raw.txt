import sys
import heapq


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 4:
        return

    n, M, T, m = data[:4]
    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    graph = [[] for _ in range(n)]
    reverse_graph = [[] for _ in range(n)]
    indegree = [0] * n
    outdegree = [0] * n

    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2
        graph[a].append(b)
        reverse_graph[b].append(a)
        indegree[b] += 1
        outdegree[a] += 1

    degree = indegree[:]
    queue = [i for i in range(n) if degree[i] == 0]
    topo = []
    head = 0

    while head < len(queue):
        v = queue[head]
        head += 1
        topo.append(v)
        for u in graph[v]:
            degree[u] -= 1
            if degree[u] == 0:
                queue.append(u)

    if len(topo) != n:
        print("NO")
        return

    up = [1] * n
    for v in topo:
        nv = up[v] + 1
        for u in graph[v]:
            if up[u] < nv:
                up[u] = nv

    down = [1] * n
    for v in reversed(topo):
        best = 1
        for u in graph[v]:
            candidate = down[u] + 1
            if candidate > best:
                best = candidate
        down[v] = best

    if max(up, default=0) > T:
        print("NO")
        return

    if M == 1:
        if n > T:
            print("NO")
            return
        answer = [0] * n
        for cycle, v in enumerate(topo, 1):
            answer[v] = cycle
        print("YES")
        print(*answer)
        return

    def forward_schedule(mode):
        deg = indegree[:]
        heap = []

        def key(v):
            if mode == 0:
                return (-down[v], -outdegree[v], v)
            if mode == 1:
                return (-down[v], -indegree[v], v)
            if mode == 2:
                return (-outdegree[v], -down[v], v)
            if mode == 3:
                return (-down[v], outdegree[v], v)
            h = (v * 1103515245 + mode * 12345) & 0x7fffffff
            return (-down[v], h, v)

        for v in range(n):
            if deg[v] == 0:
                heapq.heappush(heap, key(v))

        answer = [0] * n
        done = 0
        cycle = 0

        while heap:
            cycle += 1
            if cycle > T:
                return None

            batch = []
            count = min(M, len(heap))
            for _ in range(count):
                item = heapq.heappop(heap)
                v = item[-1]
                answer[v] = cycle
                batch.append(v)
                done += 1

            for v in batch:
                for u in graph[v]:
                    deg[u] -= 1
                    if deg[u] == 0:
                        heapq.heappush(heap, key(u))

        return answer if done == n else None

    def backward_schedule(mode):
        deg = outdegree[:]
        heap = []

        def key(v):
            if mode == 0:
                return (-up[v], -indegree[v], v)
            if mode == 1:
                return (-up[v], -outdegree[v], v)
            if mode == 2:
                return (-indegree[v], -up[v], v)
            if mode == 3:
                return (-up[v], indegree[v], v)
            h = (v * 1664525 + mode * 1013904223) & 0x7fffffff
            return (-up[v], h, v)

        for v in range(n):
            if deg[v] == 0:
                heapq.heappush(heap, key(v))

        reverse_cycle = [0] * n
        done = 0
        cycle = 0

        while heap:
            cycle += 1
            if cycle > T:
                return None

            batch = []
            count = min(M, len(heap))
            for _ in range(count):
                item = heapq.heappop(heap)
                v = item[-1]
                reverse_cycle[v] = cycle
                batch.append(v)
                done += 1

            for v in batch:
                for u in reverse_graph[v]:
                    deg[u] -= 1
                    if deg[u] == 0:
                        heapq.heappush(heap, key(u))

        if done != n:
            return None

        return [T - reverse_cycle[v] + 1 for v in range(n)]

    for mode in range(6):
        answer = forward_schedule(mode)
        if answer is not None:
            print("YES")
            print(*answer)
            return

    for mode in range(6):
        answer = backward_schedule(mode)
        if answer is not None:
            print("YES")
            print(*answer)
            return

    print("NO")


if __name__ == "__main__":
    main()
