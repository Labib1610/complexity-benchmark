from collections import deque, defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    M = int(data[idx])
    idx += 1
    T = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    # Build graph
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(data[idx])
        idx += 1
        b = int(data[idx])
        idx += 1
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort with level assignment
    queue = deque()
    time_slot = [0] * (n + 1)
    
    # Find all nodes with in-degree 0
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    slot = 1
    while slot <= T and queue:
        # Process up to M tasks in current slot
        count = 0
        next_queue = deque()
        while queue and count < M:
            node = queue.popleft()
            time_slot[node] = slot
            count += 1
            # Reduce in-degree of neighbors
            for neighbor in graph[node]:
                in_degree[neighbor] -= 1
                if in_degree[neighbor] == 0:
                    next_queue.append(neighbor)
        # Add remaining nodes back to queue for next slot
        while queue:
            next_queue.append(queue.popleft())
        queue = next_queue
        slot += 1
    
    # Check if all tasks are scheduled
    for i in range(1, n + 1):
        if time_slot[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, time_slot[1:])))

if __name__ == "__main__":
    main()
