import sys
from collections import deque
import itertools

sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    if n > M * T:
        print("NO")
        return
        
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_deg = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        in_deg[v] += 1
        
    # Topological sort & cycle detection
    q = deque([i for i in range(1, n + 1) if in_deg[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            in_deg[v] -= 1
            if in_deg[v] == 0:
                q.append(v)
                
    if len(topo) < n:
        print("NO")
        return
        
    # Earliest stage E[u]
    E = [1] * (n + 1)
    for u in topo:
        for v in adj[u]:
            if E[u] + 1 > E[v]:
                E[v] = E[u] + 1
                
    # Latest stage L[u]
    L = [T] * (n + 1)
    for u in reversed(topo):
        for v in rev_adj[u]:
            if L[u] - 1 < L[v]:
                L[v] = L[u] - 1
                
    for i in range(1, n + 1):
        if E[i] > L[i] or E[i] > T or L[i] < 1:
            print("NO")
            return
            
    # Longest path to sink for heuristic tie-breaking
    path_to_sink = [0] * (n + 1)
    for u in reversed(topo):
        for v in adj[u]:
            path_to_sink[u] = max(path_to_sink[u], path_to_sink[v] + 1)
            
    # Interval capacity check
    for t1 in range(1, T + 1):
        for t2 in range(t1, T + 1):
            cnt = sum(1 for i in range(1, n + 1) if t1 <= E[i] and L[i] <= t2)
            if cnt > (t2 - t1 + 1) * M:
                print("NO")
                return

    curr_in_deg = [len(rev_adj[i]) for i in range(n + 1)]
    stage_assigned = [0] * (n + 1)
    
    def backtrack(stage, num_scheduled):
        if num_scheduled == n:
            return True
        if stage > T:
            return False
            
        ready = []
        must = []
        for i in range(1, n + 1):
            if stage_assigned[i] == 0:
                if curr_in_deg[i] == 0:
                    if L[i] == stage:
                        must.append(i)
                    else:
                        ready.append(i)
                else:
                    if L[i] <= stage:
                        return False
                        
        if len(must) > M:
            return False
            
        cand = ready
        cand.sort(key=lambda x: (L[x], -path_to_sink[x], -len(adj[x])))
        
        rem_slots = M - len(must)
        max_take = min(len(cand), rem_slots)
        best_choice = cand[:max_take]
        
        chosen_combinations = [best_choice]
        
        if len(cand) > max_take:
            pool = cand[:min(len(cand), max_take + 2)]
            for combo in itertools.combinations(pool, max_take):
                combo_list = list(combo)
                if combo_list != best_choice:
                    chosen_combinations.append(combo_list)
                    if len(chosen_combinations) >= 5:
                        break
                        
        for chosen_cand in chosen_combinations:
            this_stage = must + chosen_cand
            for u in this_stage:
                stage_assigned[u] = stage
                for v in adj[u]:
                    curr_in_deg[v] -= 1
                    
            if backtrack(stage + 1, num_scheduled + len(this_stage)):
                return True
                
            for u in this_stage:
                stage_assigned[u] = 0
                for v in adj[u]:
                    curr_in_deg[v] += 1
                    
        return False

    if backtrack(1, 0):
        print("YES")
        print(*(stage_assigned[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
