import sys
from collections import deque, defaultdict

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(m):
        a = int(next(it))-1
        b = int(next(it))-1
        adj[a].append(b)
        indeg[b] += 1

    # topological order
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    while q:
        v = q.popleft()
        topo.append(v)
        for to in adj[v]:
            indeg[to] -= 1
            if indeg[to]==0:
                q.append(to)
    if len(topo)!=n:
        print("NO")
        return

    # assign levels greedily
    level = [0]*n
    cap = [0]*(T+1)  # 1-indexed
    for v in topo:
        # find max level among predecessors
        min_allowed = 1
        for u in range(n):
            # we could store reverse edges
            pass
    # build reverse adjacency for fast predecessor lookup
    rev = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            rev[v].append(u)

    for v in topo:
        min_allowed = 1
        for pre in rev[v]:
            min_allowed = max(min_allowed, level[pre]+1)
        assigned = -1
        for l in range(min_allowed, T+1):
            if cap[l] < M:
                assigned = l
                break
        if assigned == -1:
            print("NO")
            return
        level[v] = assigned
        cap[assigned] += 1

    print("YES")
    print(' '.join(str(level[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
