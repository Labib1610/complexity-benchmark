We can model this as a topological scheduling problem with capacity constraints.  
We need to assign each component to a shift (1 to T) such that:

- All prerequisites are scheduled earlier.
- At most M components per shift.

We can solve it using a greedy topological order with a priority queue based on earliest possible shift.

Here’s the implementation:

```python
import sys
import heapq

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # earliest shift for each node
    earliest = [1] * n
    
    # min-heap: (earliest_shift, node)
    heap = []
    for i in range(n):
        if indeg[i] == 0:
            heapq.heappush(heap, (1, i))
    
    assigned = [0] * n
    # current load per shift
    load = [0] * (T + 1)
    
    processed = 0
    while heap:
        shift, u = heapq.heappop(heap)
        # find the earliest shift >= shift with capacity
        s = shift
        while s <= T and load[s] >= M:
            s += 1
        if s > T:
            print("NO")
            return
        assigned[u] = s
        load[s] += 1
        processed += 1
        
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                # earliest possible shift for v is at least s+1
                heapq.heappush(heap, (s + 1, v))
    
    if processed != n:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(x) for x in assigned))

if __name__ == "__main__":
    solve()
```
