import sys
from collections import deque

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    out_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        out_degree[u] += 1
        in_degree[v] += 1
        
    # Check for cycles and compute topological sort
    deg = list(in_degree)
    q = deque([i for i in range(1, n + 1) if deg[i] == 0])
    topo_order = []
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            deg[v] -= 1
            if deg[v] == 0:
                q.append(v)
                
    if len(topo_order) < n:
        print("NO")
        return

    # Compute EST (Earliest Start Time)
    est = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if est[u] + 1 > est[v]:
                est[v] = est[u] + 1

    # Compute LST (Latest Start Time)
    lst = [T] * (n + 1)
    for u in reversed(topo_order):
        for v in adj[u]:
            if lst[v] - 1 < lst[u]:
                lst[u] = lst[v] - 1

    # If any EST > LST, impossible
    for i in range(1, n + 1):
        if est[i] > lst[i]:
            print("NO")
            return

    # Height/Longest path to sink
    height = [0] * (n + 1)
    for u in reversed(topo_order):
        h = 0
        for v in adj[u]:
            if height[v] + 1 > h:
                h = height[v] + 1
        height[u] = h

    # Coffman-Graham-like priority
    # Score: lower LST is higher priority, higher height is higher priority
    # (lst, -height, -out_degree)
    priority = [0] * (n + 1)
    # We can just use a key tuple for sorting candidates: (lst[u], -height[u], -len(adj[u]))

    assigned_phase = [0] * (n + 1)
    rem_in_deg = list(in_degree)
    
    # State tracking
    def get_ready_tasks():
        return [i for i in range(1, n + 1) if rem_in_deg[i] == 0 and assigned_phase[i] == 0]

    # Combinations generator
    def get_subsets(candidates, k):
        # generate subsets of size k from candidates
        if k == 0:
            yield []
            return
        if not candidates:
            return
        # To avoid explosive branching, we can limit branching if candidates is large
        from itertools import combinations
        for comb in combinations(candidates, k):
            yield list(comb)

    found_solution = False
    solution = [0] * (n + 1)

    import time
    start_time = time.time()

    def search(phase, unassigned_count):
        nonlocal found_solution, solution
        if found_solution:
            return True
        if time.time() - start_time > 4.5:
            return False

        if unassigned_count == 0:
            found_solution = True
            solution = list(assigned_phase)
            return True

        if phase > T:
            return False

        # Pruning: bound check
        # Check capacity constraint for each future deadline
        # Count unassigned tasks with lst <= t_limit
        lst_counts = [0] * (T + 2)
        for i in range(1, n + 1):
            if assigned_phase[i] == 0:
                l = lst[i]
                if l < phase:
                    return False
                lst_counts[l] += 1
        
        cum = 0
        for t_limit in range(phase, T + 1):
            cum += lst_counts[t_limit]
            if cum > (t_limit - phase + 1) * M:
                return False

        # Ready tasks at this phase
        ready = [i for i in range(1, n + 1) if rem_in_deg[i] == 0 and assigned_phase[i] == 0]
        if not ready:
            return False

        # Must schedule tasks (lst == phase)
        must = [u for u in ready if lst[u] <= phase]
        if len(must) > M:
            return False

        other_ready = [u for u in ready if lst[u] > phase]
        # Sort other_ready by priority
        other_ready.sort(key=lambda u: (lst[u], -height[u], -out_degree[u]))

        needed = M - len(must)
        if len(other_ready) <= needed:
            # We must take all ready tasks
            chosen_batches = [must + other_ready]
        else:
            # We need to choose 'needed' tasks from other_ready
            # Best greedy choice first
            chosen_batches = []
            # We can explore best combinations
            # If needed is large or small, take combinations
            from itertools import combinations
            # Limit number of combinations to explore to avoid TLE
            count = 0
            for comb in combinations(other_ready, needed):
                chosen_batches.append(must + list(comb))
                count += 1
                if count >= 30: # heuristic search width limit per level
                    break

        for batch in chosen_batches:
            # Apply batch
            for u in batch:
                assigned_phase[u] = phase
                for v in adj[u]:
                    rem_in_deg[v] -= 1

            if search(phase + 1, unassigned_count - len(batch)):
                return True

            # Undo batch
            for u in batch:
                assigned_phase[u] = 0
                for v in adj[u]:
                    rem_in_deg[v] += 1

        return False

    if search(1, n):
        print("YES")
        print(*(solution[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
