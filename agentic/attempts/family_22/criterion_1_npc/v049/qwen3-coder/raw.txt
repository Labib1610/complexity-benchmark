from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort with cycle detection
    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    topological_order = []
    while queue:
        node = queue.popleft()
        topological_order.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # If we didn't visit all nodes, there's a cycle
    if len(topological_order) != n:
        print("NO")
        return
    
    # Assign cycles
    assignment = [0] * (n + 1)
    cycle_count = [0] * (T + 1)  # Count of modules per cycle
    
    for node in topological_order:
        # Find the earliest cycle where:
        # 1. All predecessors have finished (so node can start)
        # 2. Slot limit is not exceeded
        min_cycle = 1
        for predecessor in graph:
            if node in graph[predecessor]:
                min_cycle = max(min_cycle, assignment[predecessor] + 1)
        
        # Try to assign to the earliest valid cycle
        assigned = False
        for cycle in range(min_cycle, T + 1):
            if cycle_count[cycle] < M:
                assignment[node] = cycle
                cycle_count[cycle] += 1
                assigned = True
                break
        
        if not assigned:
            print("NO")
            return
    
    print("YES")
    print(*assignment[1:])

solve()
