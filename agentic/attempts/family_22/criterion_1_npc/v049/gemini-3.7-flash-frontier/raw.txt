```python
import sys
from collections import deque
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    if n > M * T:
        print("NO")
        return
    
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    out_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        out_degree[u] += 1
        
    # Check for cycles and compute topological order
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo_order = []
    in_deg_temp = list(in_degree)
    
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            in_deg_temp[v] -= 1
            if in_deg_temp[v] == 0:
                q.append(v)
                
    if len(topo_order) < n:
        print("NO")
        return
        
    # Compute EST (Earliest Start Time)
    est = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if est[u] + 1 > est[v]:
                est[v] = est[u] + 1
                
    for u in range(1, n + 1):
        if est[u] > T:
            print("NO")
            return
            
    # Compute LST (Latest Start Time) and heights
    lst = [T] * (n + 1)
    height = [1] * (n + 1)
    for u in reversed(topo_order):
        for v in adj[u]:
            if lst[v] - 1 < lst[u]:
                lst[u] = lst[v] - 1
            if height[v] + 1 > height[u]:
                height[u] = height[v] + 1
                
    for u in range(1, n + 1):
        if lst[u] < 1 or lst[u] < est[u]:
            print("NO")
            return

    # Count of descendants for tie-breaking
    desc_count = [0] * (n + 1)
    for u in reversed(topo_order):
        desc = set()
        for v in adj[u]:
            desc.add(v)
        desc_count[u] = len(desc)

    # Backtracking search
    # State: cycle t, scheduled tasks, remaining in-degrees
    assigned_cycle = [0] * (n + 1)
    curr_in_deg = list(in_degree)
    
    # Priority for sorting ready tasks:
    # (lst[x], -height[x], -desc_count[x], x)
    def task_key(x):
        return (lst[x], -height[x], -desc_count[x])
    
    # Backtrack function
    # To avoid deep recursion limits and speed up, we can use a stack or dfs
    sys.setrecursionlimit(max(2000, n + 50))
    
    # Ready set
    initial_ready = [i for i in range(1, n + 1) if curr_in_deg[i] == 0]
    
    def search(t, ready_list, scheduled_count):
        if scheduled_count == n:
            return True
        if t > T:
            return False
        if n - scheduled_count > (T - t + 1) * M:
            return False
            
        mandatory = []
        optional = []
        
        for u in ready_list:
            if lst[u] < t:
                return False
            elif lst[u] == t:
                mandatory.append(u)
            else:
                optional.append(u)
                
        if len(mandatory) > M:
            return False
            
        slots_left = M - len(mandatory)
        
        if len(optional) <= slots_left:
            chosen_combinations = [mandatory + optional]
        else:
            optional.sort(key=task_key)
            mandatory_set = mandatory
            # Choose slots_left items from optional
            # Prioritize prefix of sorted optional
            # Generate limited combinations to avoid huge search space if not needed
            combos = []
            # Best choice: take first slots_left
            combos.append(mandatory_set + optional[:slots_left])
            
            # If search might need alternatives, generate more combinations
            if len(optional) > slots_left:
                # Add a few more alternative combinations if slots_left > 0
                for extra in itertools.islice(itertools.combinations(optional, slots_left), 1, 30):
                    combos.append(mandatory_set + list(extra))
            chosen_combinations = combos

        for chosen in chosen_combinations:
            for u in chosen:
                assigned_cycle[u] = t
                for v in adj[u]:
                    curr_in_deg[v] -= 1
                    
            next_ready = [u for u in ready_list if assigned_cycle[u] == 0]
            for u in chosen:
                for v in adj[u]:
                    if curr_in_deg[v] == 0:
                        next_ready.append(v)
                        
            if search(t + 1, next_ready, scheduled_count + len(chosen)):
                return True
                
            # Backtrack
            for u in chosen:
                assigned_cycle[u] = 0
                for v in adj[u]:
                    curr_in_deg[v] += 1
                    
        return False

    if search(1, initial_ready, 0):
        print("YES")
        print(*(assigned_cycle[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
