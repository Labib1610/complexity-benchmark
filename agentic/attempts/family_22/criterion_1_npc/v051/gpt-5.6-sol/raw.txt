import sys
import heapq

sys.setrecursionlimit(1_000_000)


def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, M, T, m = data[:4]
    pred = [0] * n
    succ = [0] * n

    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2
        if not ((succ[a] >> b) & 1):
            succ[a] |= 1 << b
            pred[b] |= 1 << a

    if n == 0:
        print("YES")
        print()
        return

    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    indeg = [x.bit_count() for x in pred]
    queue = [i for i in range(n) if indeg[i] == 0]
    topo = []
    qpos = 0

    while qpos < len(queue):
        v = queue[qpos]
        qpos += 1
        topo.append(v)
        bits = succ[v]
        while bits:
            bit = bits & -bits
            u = bit.bit_length() - 1
            bits -= bit
            indeg[u] -= 1
            if indeg[u] == 0:
                queue.append(u)

    if len(topo) != n:
        print("NO")
        return

    head = [1] * n
    for v in topo:
        hv = head[v] + 1
        bits = succ[v]
        while bits:
            bit = bits & -bits
            u = bit.bit_length() - 1
            bits -= bit
            if head[u] < hv:
                head[u] = hv

    tail = [1] * n
    for v in reversed(topo):
        best = 1
        bits = succ[v]
        while bits:
            bit = bits & -bits
            u = bit.bit_length() - 1
            bits -= bit
            cand = tail[u] + 1
            if cand > best:
                best = cand
        tail[v] = best

    if max(tail) > T:
        print("NO")
        return

    H = min(T, n)
    outdeg = [x.bit_count() for x in succ]
    indeg0 = [x.bit_count() for x in pred]

    if M >= n:
        ans = head
        print("YES")
        print(*ans)
        return

    def forward_greedy(mode):
        deg = indeg0[:]
        heap = []

        def priority(v):
            if mode == 0:
                return (-tail[v], -outdeg[v], v)
            if mode == 1:
                return (-outdeg[v], -tail[v], v)
            if mode == 2:
                return (-(tail[v] * (n + 1) + outdeg[v]), -head[v], v)
            return (-tail[v], -head[v], -outdeg[v], v)

        for v in range(n):
            if deg[v] == 0:
                heapq.heappush(heap, priority(v))

        ans = [0] * n
        done = 0
        week = 1

        while heap and done < n and week <= H:
            batch = []
            for _ in range(min(M, len(heap))):
                item = heapq.heappop(heap)
                v = item[-1]
                batch.append(v)
                ans[v] = week

            done += len(batch)
            newly = []
            for v in batch:
                bits = succ[v]
                while bits:
                    bit = bits & -bits
                    u = bit.bit_length() - 1
                    bits -= bit
                    deg[u] -= 1
                    if deg[u] == 0:
                        newly.append(u)

            for u in newly:
                heapq.heappush(heap, priority(u))
            week += 1

        return ans if done == n else None

    for mode in range(4):
        ans = forward_greedy(mode)
        if ans is not None:
            print("YES")
            print(*ans)
            return

    def reverse_greedy(mode):
        deg = outdeg[:]
        heap = []

        def priority(v):
            if mode == 0:
                return (-head[v], -indeg0[v], v)
            return (-indeg0[v], -head[v], v)

        for v in range(n):
            if deg[v] == 0:
                heapq.heappush(heap, priority(v))

        ans = [0] * n
        done = 0
        week = H

        while heap and done < n and week >= 1:
            batch = []
            for _ in range(min(M, len(heap))):
                item = heapq.heappop(heap)
                v = item[-1]
                batch.append(v)
                ans[v] = week

            done += len(batch)
            newly = []
            for v in batch:
                bits = pred[v]
                while bits:
                    bit = bits & -bits
                    u = bit.bit_length() - 1
                    bits -= bit
                    deg[u] -= 1
                    if deg[u] == 0:
                        newly.append(u)

            for u in newly:
                heapq.heappush(heap, priority(u))
            week -= 1

        return ans if done == n else None

    for mode in range(2):
        ans = reverse_greedy(mode)
        if ans is not None:
            print("YES")
            print(*ans)
            return

    all_mask = (1 << n) - 1
    failed = set()
    choice = {}

    def dfs(mask, remaining_weeks):
        if mask == all_mask:
            return True
        if remaining_weeks == 0:
            return False

        key = (mask, remaining_weeks)
        if key in failed:
            return False

        remaining_mask = all_mask ^ mask
        remaining_count = remaining_mask.bit_count()
        if remaining_count > remaining_weeks * M:
            failed.add(key)
            return False

        available = []
        hist = [0] * (remaining_weeks + 1)
        bits = remaining_mask

        while bits:
            bit = bits & -bits
            v = bit.bit_length() - 1
            bits -= bit

            tv = tail[v]
            if tv > remaining_weeks:
                failed.add(key)
                return False
            hist[tv] += 1

            if pred[v] & remaining_mask == 0:
                available.append(v)

        cumulative = 0
        for k in range(remaining_weeks, 0, -1):
            cumulative += hist[k]
            if cumulative > M * (remaining_weeks - k + 1):
                failed.add(key)
                return False

        if not available:
            failed.add(key)
            return False

        if len(available) <= M:
            selected_mask = 0
            for v in available:
                selected_mask |= 1 << v
            if dfs(mask | selected_mask, remaining_weeks - 1):
                choice[key] = selected_mask
                return True
            failed.add(key)
            return False

        urgent = [v for v in available if tail[v] == remaining_weeks]
        if len(urgent) > M:
            failed.add(key)
            return False

        urgent_mask = 0
        urgent_set = set(urgent)
        for v in urgent:
            urgent_mask |= 1 << v

        need = M - len(urgent)
        candidates = [v for v in available if v not in urgent_set]

        groups_dict = {}
        for v in candidates:
            groups_dict.setdefault(succ[v], []).append(v)

        groups = list(groups_dict.values())
        groups.sort(
            key=lambda g: (
                -max(tail[v] for v in g),
                -max(outdeg[v] for v in g),
                g[0]
            )
        )

        suffix_capacity = [0] * (len(groups) + 1)
        for i in range(len(groups) - 1, -1, -1):
            suffix_capacity[i] = suffix_capacity[i + 1] + len(groups[i])

        def generate_group_choices(idx, left, selected_mask):
            if left == 0:
                yield selected_mask
                return
            if idx == len(groups) or suffix_capacity[idx] < left:
                return

            group = groups[idx]
            maximum = min(len(group), left)
            minimum = max(0, left - suffix_capacity[idx + 1])

            prefix_masks = [0]
            cur = 0
            for v in group:
                cur |= 1 << v
                prefix_masks.append(cur)

            for take in range(maximum, minimum - 1, -1):
                yield from generate_group_choices(
                    idx + 1,
                    left - take,
                    selected_mask | prefix_masks[take]
                )

        for extra_mask in generate_group_choices(0, need, 0):
            selected_mask = urgent_mask | extra_mask
            if dfs(mask | selected_mask, remaining_weeks - 1):
                choice[key] = selected_mask
                return True

        failed.add(key)
        return False

    if not dfs(0, H):
        print("NO")
        return

    ans = [0] * n
    mask = 0
    remaining_weeks = H
    week = 1

    while mask != all_mask:
        selected_mask = choice[(mask, remaining_weeks)]
        bits = selected_mask
        while bits:
            bit = bits & -bits
            v = bit.bit_length() - 1
            bits -= bit
            ans[v] = week
        mask |= selected_mask
        remaining_weeks -= 1
        week += 1

    print("YES")
    print(*ans)


if __name__ == "__main__":
    solve()
