```python
import sys
import heapq
import itertools

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 4:
        return

    n, M, T, m = data[:4]
    if n == 0:
        print("YES")
        print()
        return
    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    prereq = [0] * n
    adj = [[] for _ in range(n)]
    indeg = [0] * n

    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2
        bit = 1 << a
        if not (prereq[b] & bit):
            prereq[b] |= bit
            adj[a].append(b)
            indeg[b] += 1

    # Topological order and cycle detection.
    q = [i for i in range(n) if indeg[i] == 0]
    topo = []
    head = 0
    indeg_work = indeg[:]
    while head < len(q):
        u = q[head]
        head += 1
        topo.append(u)
        for v in adj[u]:
            indeg_work[v] -= 1
            if indeg_work[v] == 0:
                q.append(v)

    if len(topo) != n:
        print("NO")
        return

    # Longest path lengths to a sink and from a source.
    tail = [1] * n
    for u in reversed(topo):
        best = 0
        for v in adj[u]:
            if tail[v] > best:
                best = tail[v]
        tail[u] = best + 1

    if max(tail) > T:
        print("NO")
        return

    longest_head = [1] * n
    for u in topo:
        hu = longest_head[u] + 1
        for v in adj[u]:
            if hu > longest_head[v]:
                longest_head[v] = hu

    # Forward critical-path list scheduling.
    def forward_schedule():
        deg = indeg[:]
        heap = []
        for i in range(n):
            if deg[i] == 0:
                heapq.heappush(heap, (-tail[i], -len(adj[i]), i))

        answer = [0] * n
        done = 0
        cycle = 0

        while heap:
            cycle += 1
            batch = []
            for _ in range(min(M, len(heap))):
                _, _, u = heapq.heappop(heap)
                batch.append(u)
                answer[u] = cycle
                done += 1

            newly = []
            for u in batch:
                for v in adj[u]:
                    deg[v] -= 1
                    if deg[v] == 0:
                        newly.append(v)
            for v in newly:
                heapq.heappush(heap, (-tail[v], -len(adj[v]), v))

        if done == n and cycle <= T:
            return answer
        return None

    answer = forward_schedule()
    if answer is not None:
        print("YES")
        print(*answer)
        return

    # Reverse critical-path list scheduling.
    def backward_schedule():
        outdeg = [len(adj[i]) for i in range(n)]
        rev = [[] for _ in range(n)]
        for u in range(n):
            for v in adj[u]:
                rev[v].append(u)

        heap = []
        for i in range(n):
            if outdeg[i] == 0:
                heapq.heappush(heap, (-longest_head[i], -len(rev[i]), i))

        batches = []
        done = 0
        while heap:
            batch = []
            for _ in range(min(M, len(heap))):
                _, _, u = heapq.heappop(heap)
                batch.append(u)
                done += 1

            newly = []
            for u in batch:
                for v in rev[u]:
                    outdeg[v] -= 1
                    if outdeg[v] == 0:
                        newly.append(v)
            for v in newly:
                heapq.heappush(
                    heap, (-longest_head[v], -len(rev[v]), v)
                )
            batches.append(batch)

        if done != n or len(batches) > T:
            return None

        result = [0] * n
        total = len(batches)
        for i, batch in enumerate(batches):
            cycle = total - i
            for u in batch:
                result[u] = cycle
        return result

    answer = backward_schedule()
    if answer is not None:
        print("YES")
        print(*answer)
        return

    # Exact branch-and-bound search. The problem is NP-hard in general,
    # so this part may require exponential time on adversarial instances.
    sys.setrecursionlimit(max(1000000, n + T + 100))
    full = (1 << n) - 1
    assigned = [0] * n
    best_time = {}

    def dfs(mask, elapsed):
        if mask == full:
            return True

        old = best_time.get(mask)
        if old is not None and old <= elapsed:
            return False
        best_time[mask] = elapsed

        remaining_cycles = T - elapsed
        remaining_count = n - mask.bit_count()
        if remaining_cycles <= 0 or remaining_count > remaining_cycles * M:
            return False

        available = []
        remaining_tails = []

        for i in range(n):
            if not ((mask >> i) & 1):
                ti = tail[i]
                if ti > remaining_cycles:
                    return False
                remaining_tails.append(ti)
                if prereq[i] & ~mask == 0:
                    available.append(i)

        if not available:
            return False

        # Every task with tail L must be placed no later than
        # remaining_cycles-L+1 cycles from now.
        remaining_tails.sort(reverse=True)
        for count, length in enumerate(remaining_tails, 1):
            slots = remaining_cycles - length + 1
            if count > slots * M:
                return False

        if len(available) <= M:
            batch = available
            new_mask = mask
            cycle = elapsed + 1
            for u in batch:
                new_mask |= 1 << u
                assigned[u] = cycle
            if dfs(new_mask, cycle):
                return True
            for u in batch:
                assigned[u] = 0
            return False

        forced = []
        optional = []
        for u in available:
            if tail[u] == remaining_cycles:
                forced.append(u)
            else:
                optional.append(u)

        if len(forced) > M:
            return False

        optional.sort(
            key=lambda u: (-tail[u], -len(adj[u]), -longest_head[u], u)
        )
        need = M - len(forced)
        if need > len(optional):
            return False

        forced_mask = 0
        for u in forced:
            forced_mask |= 1 << u

        cycle = elapsed + 1
        for chosen in itertools.combinations(optional, need):
            new_mask = mask | forced_mask
            for u in chosen:
                new_mask |= 1 << u

            for u in forced:
                assigned[u] = cycle
            for u in chosen:
                assigned[u] = cycle

            if dfs(new_mask, cycle):
                return True

            for u in forced:
                assigned[u] = 0
            for u in chosen:
                assigned[u] = 0

        return False

    if dfs(0, 0):
        print("YES")
        print(*assigned)
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
