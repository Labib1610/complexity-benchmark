We can model this as a scheduling problem on a DAG, where we need to assign each node a time slot from 1 to T, such that:

1. For every edge (a → b), time[a] < time[b].
2. For each time t, the number of nodes with that time is ≤ M.

If no such assignment exists, print `NO`.

We can approach this using **topological sorting** combined with a **greedy earliest-time assignment** method:  
- We first compute the earliest possible time for each node based on dependencies.
- Then, we try to place all nodes within the deadline, packing them into cycles without exceeding M per cycle.

If at any point a node’s earliest time exceeds T, or we cannot fit enough nodes into available slots before the deadline, we output `NO`.

Here’s the implementation:

```python
from collections import deque
import sys

def solve():
    input = sys.stdin.readline
    n, M, T, m = map(int, input().split())
    graph = [[] for _ in range(n)]
    indeg = [0] * n
    
    for _ in range(m):
        a, b = map(int, input().split())
        a -= 1
        b -= 1
        graph[a].append(b)
        indeg[b] += 1
    
    # Step 1: Topological sort with earliest time assignment
    earliest = [1] * n
    q = deque()
    for i in range(n):
        if indeg[i] == 0:
            q.append(i)
    
    processed = 0
    topo_order = []
    while q:
        u = q.popleft()
        topo_order.append(u)
        processed += 1
        for v in graph[u]:
            earliest[v] = max(earliest[v], earliest[u] + 1)
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if processed != n:
        print("NO")
        return
    
    # Step 2: Check earliest deadline feasibility
    if max(earliest) > T:
        print("NO")
        return
    
    # Step 3: Greedy assignment inside capacity constraints
    # Sort nodes by earliest time, then assign them to the earliest possible cycle
    order = sorted(range(n), key=lambda x: earliest[x])
    assigned = [0] * n
    # slots[t] = number used so far in cycle t+1
    slots = [0] * (T + 1)
    
    for u in order:
        # Place at earliest possible slot that satisfies time and capacity
        placed = False
        for t in range(earliest[u], T + 1):
            if slots[t] < M:
                assigned[u] = t
                slots[t] += 1
                placed = True
                break
        if not placed:
            print("NO")
            return
    
    # Verify dependencies (strictly earlier)
    for u in range(n):
        for v in graph[u]:
            if assigned[u] >= assigned[v]:
                print("NO")
                return
    
    print("YES")
    print(" ".join(map(str, assigned)))

if __name__ == "__main__":
    solve()
```
