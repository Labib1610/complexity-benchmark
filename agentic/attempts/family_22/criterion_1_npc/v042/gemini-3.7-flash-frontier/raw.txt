import sys
from collections import deque
import itertools

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])
    
    if n > M * T:
        print("NO")
        return
        
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        
    # Check for cycles and get topological order
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)
                
    if len(topo) < n:
        print("NO")
        return

    # Compute longest path to sink (dist_to_sink) and L[u]
    dist_to_sink = [0] * (n + 1)
    for u in reversed(topo):
        for v in adj[u]:
            if dist_to_sink[v] + 1 > dist_to_sink[u]:
                dist_to_sink[u] = dist_to_sink[v] + 1
                
    L = [T - dist_to_sink[u] for u in range(n + 1)]
    
    # Compute dist_from_source and E[u]
    dist_from_source = [0] * (n + 1)
    for u in topo:
        for v in adj[u]:
            if dist_from_source[u] + 1 > dist_from_source[v]:
                dist_from_source[v] = dist_from_source[u] + 1
                
    E = [1 + dist_from_source[u] for u in range(n + 1)]
    
    for u in range(1, n + 1):
        if E[u] > L[u] or L[u] < 1 or E[u] > T:
            print("NO")
            return

    # Reset in_degree
    in_deg = [0] * (n + 1)
    for u in range(1, n + 1):
        for v in adj[u]:
            in_deg[v] += 1
            
    assigned_cycle = [0] * (n + 1)
    
    # Ready nodes have in_deg == 0
    ready = [u for u in range(1, n + 1) if in_deg[u] == 0]
    
    # Backtracking search
    def backtrack(t, ready_list, scheduled_count):
        if scheduled_count == n:
            return True
        if t > T:
            return False
        
        # Check if any unscheduled node has L[u] < t
        for u in ready_list:
            if L[u] < t:
                return False
        
        mandatory = []
        optional = []
        for u in ready_list:
            if L[u] == t:
                mandatory.append(u)
            else:
                optional.append(u)
                
        if len(mandatory) > M:
            return False
            
        cap = M - len(mandatory)
        # Sort optional by priority: smallest L[u], then largest dist_to_sink, then largest out-degree
        optional.sort(key=lambda u: (L[u], -dist_to_sink[u], -len(adj[u])))
        
        # We try to take as many optional as possible, starting from cap down to 0
        max_opt = min(cap, len(optional))
        
        # Generator for candidate combinations of optional items
        def get_combos():
            for k in range(max_opt, -1, -1):
                # First try the best prefix of size k
                best_prefix = tuple(optional[:k])
                yield best_prefix
                # Then other combinations of size k if needed
                for combo in itertools.combinations(optional, k):
                    if combo != best_prefix:
                        yield combo

        for chosen_opt in get_combos():
            chosen = mandatory + list(chosen_opt)
            for u in chosen:
                assigned_cycle[u] = t
                
            new_ready = [u for u in optional if u not in chosen_opt]
            # Unlock successors
            unlocked = []
            for u in chosen:
                for v in adj[u]:
                    in_deg[v] -= 1
                    if in_deg[v] == 0:
                        new_ready.append(v)
                        unlocked.append(v)
                        
            if backtrack(t + 1, new_ready, scheduled_count + len(chosen)):
                return True
                
            # Rollback
            for v in unlocked:
                pass
            for u in chosen:
                assigned_cycle[u] = 0
                for v in adj[u]:
                    in_deg[v] += 1
                    
        return False

    if backtrack(1, ready, 0):
        print("YES")
        print(*(assigned_cycle[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
