import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m_dep = int(next(it))

    adj = [[] for _ in range(n)]
    in_degree = [0] * n
    for _ in range(m_dep):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        in_degree[b] += 1

    # Earliest phases using BFS (Kahn's algorithm)
    phase = [1] * n
    q = deque()
    for i in range(n):
        if in_degree[i] == 0:
            q.append(i)

    topological_order = []
    while q:
        u = q.popleft()
        topological_order.append(u)
        for v in adj[u]:
            if phase[v] < phase[u] + 1:
                phase[v] = phase[u] + 1
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)

    if len(topological_order) != n:
        print("NO")
        return

    # Check feasibility of earliest phases
    for i in range(n):
        if phase[i] > T:
            print("NO")
            return

    # Shift phases to evenly distribute and minimize max used slots per phase
    # Greedy distribution similar to scheduling with capacity constraints
    
    # Sort varieties by their earliest phase
    order = list(range(n))
    order.sort(key=lambda x: phase[x])
    
    # Current phase load
    current_slots = [0] * (T + 1)
    final_phase = [0] * n
    
    # Process in order of earliest phase, assign to latest available slot >= earliest
    # This minimizes maximum load on earlier phases
    for u in order:
        earliest = phase[u]
        # Find the earliest phase >= 'earliest' with available capacity
        # Or push to the latest possible to balance load
        # Actually, to minimize max load, we should spread them out.
        # Let's try to assign to the phase that minimizes maximum current load
        best_p = -1
        min_load = float('inf')
        
        # Try phases from earliest to T, pick one with smallest current load
        for p in range(earliest, T + 1):
            if current_slots[p] < M:
                if current_slots[p] < min_load:
                    min_load = current_slots[p]
                    best_p = p
                if min_load == 0:
                    break
        
        if best_p == -1:
            print("NO")
            return
        
        final_phase[u] = best_p
        current_slots[best_p] += 1

    print("YES")
    print(' '.join(map(str, final_phase)))

if __name__ == "__main__":
    solve()
