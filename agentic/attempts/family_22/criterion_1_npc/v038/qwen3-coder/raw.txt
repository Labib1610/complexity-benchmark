from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort
    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    order = []
    while queue:
        node = queue.popleft()
        order.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # Check if all nodes are included (no cycles)
    if len(order) != n:
        print("NO")
        return
    
    # Assign phases
    phase = [0] * (n + 1)
    current_phase = 1
    index = 0
    
    while index < n:
        # Count how many can be assigned in current phase
        count = 0
        temp_index = index
        
        # Go through nodes in topological order and assign to current phase
        # if all dependencies are satisfied
        while temp_index < n and count < M:
            node = order[temp_index]
            # Check if all dependencies are in previous phases
            can_assign = True
            for dep in graph[node]:
                if phase[dep] == 0 or phase[dep] >= current_phase:
                    can_assign = False
                    break
            
            if can_assign and all(phase[dep] < current_phase for dep in range(1, n+1) if dep in graph[node] and phase[dep] != 0):
                # Actually, we need to be more careful
                # In topological order, if we're at this point, all dependencies 
                # should already be handled because of the order
                temp_index += 1
                count += 1
            else:
                break
        
        # Assign the counted nodes to current phase
        for i in range(index, index + count):
            phase[order[i]] = current_phase
            
        index += count
        current_phase += 1
        
        # If we exceed T phases, it's impossible
        if current_phase > T and index < n:
            print("NO")
            return
    
    # Check constraint: absolute difference between any two phases <= T-1
    # Since phases are between 1 and T, this is automatically satisfied
    
    # Check constraint: total unused slots = M*T - n
    # This means exactly n slots are used, which is satisfied since we assigned all n varieties
    
    print("YES")
    print(*[phase[i] for i in range(1, n + 1)])
