from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with level tracking
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    # Assign phases
    phases = [0] * (n + 1)
    current_phase = 1
    modules_in_current_phase = 0
    
    while queue:
        next_queue = deque()
        
        # Process up to M modules in current phase
        modules_in_current_phase = 0
        while queue and modules_in_current_phase < M:
            node = queue.popleft()
            phases[node] = current_phase
            modules_in_current_phase += 1
            
            # Update neighbors
            for neighbor in graph[node]:
                indegree[neighbor] -= 1
                if indegree[neighbor] == 0:
                    next_queue.append(neighbor)
        
        # Move to next phase
        current_phase += 1
        if current_phase > T:
            if queue or next_queue:
                print("NO")
                return
        
        # Add remaining nodes back to queue for next phase
        while queue:
            next_queue.append(queue.popleft())
        
        queue = next_queue
    
    # Check if all modules were assigned
    for i in range(1, n + 1):
        if phases[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(*phases[1:])

solve()
