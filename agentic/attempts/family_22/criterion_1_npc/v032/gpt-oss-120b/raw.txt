```python
import sys
import threading
def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); M = int(next(it)); T = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    radj = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(m):
        a = int(next(it))-1
        b = int(next(it))-1
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1

    # topological order
    from collections import deque
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v]==0:
                q.append(v)
    if len(topo)!=n:
        print("NO")
        return

    # earliest times
    earliest = [1]*n
    for u in topo:
        eu = earliest[u]
        for v in adj[u]:
            if earliest[v] < eu + 1:
                earliest[v] = eu + 1

    # latest times
    latest = [T]*n
    for u in reversed(topo):
        lu = latest[u]
        for p in radj[u]:
            if latest[p] > lu - 1:
                latest[p] = lu - 1

    for i in range(n):
        if earliest[i] > latest[i] or earliest[i] > T or latest[i] < 1:
            print("NO")
            return

    # assign using greedy with capacity M per slot
    slots = [[] for _ in range(T+1)]  # not needed, we store result directly
    assign = [0]*n
    import heapq
    heap = []  # (r_i, node)
    ptr = 0
    nodes_by_l = sorted(range(n), key=lambda x: earliest[x])
    for t in range(1, T+1):
        # add all nodes whose earliest == t
        while ptr < n and earliest[nodes_by_l[ptr]] == t:
            v = nodes_by_l[ptr]
            heapq.heappush(heap, (latest[v], v))
            ptr += 1
        # assign up to M nodes
        cnt = 0
        while cnt < M and heap:
            r, v = heapq.heappop(heap)
            if r < t:  # missed deadline
                print("NO")
                return
            assign[v] = t
            cnt += 1
    if any(a==0 for a in assign):
        # some nodes not assigned (their earliest > T)
        print("NO")
        return
    print("YES")
    print(' '.join(map(str, assign)))

if __name__ == "__main__":
    threading.Thread(target=solve).start()
```
