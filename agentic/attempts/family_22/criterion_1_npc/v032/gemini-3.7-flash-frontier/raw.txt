import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    out_degree = [0] * (n + 1)

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        out_degree[u] += 1

    # Topological sort & cycle detection
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo_order = []
    in_deg_copy = list(in_degree)
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            in_deg_copy[v] -= 1
            if in_deg_copy[v] == 0:
                q.append(v)

    if len(topo_order) < n:
        print("NO")
        return

    # Compute Earliest Phase E[u]
    E = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if E[u] + 1 > E[v]:
                E[v] = E[u] + 1

    # Compute Latest Phase L[u]
    L = [T] * (n + 1)
    for u in reversed(topo_order):
        for v in rev_adj[u]:
            if L[u] - 1 < L[v]:
                L[v] = L[u] - 1

    for i in range(1, n + 1):
        if E[i] > L[i]:
            print("NO")
            return

    # Try Forward Greedy List Scheduling
    def try_forward_greedy():
        cur_in_degree = list(in_degree)
        assigned_phase = [0] * (n + 1)
        available = [i for i in range(1, n + 1) if cur_in_degree[i] == 0]
        
        for t in range(1, T + 1):
            # Sort available by L[i] ascending, then out_degree descending
            available.sort(key=lambda x: (L[x], -out_degree[x]))
            
            # Pick up to M
            chosen = available[:M]
            available = available[M:]
            
            new_available = []
            for u in chosen:
                assigned_phase[u] = t
                for v in adj[u]:
                    cur_in_degree[v] -= 1
                    if cur_in_degree[v] == 0:
                        new_available.append(v)
            available.extend(new_available)

        if all(assigned_phase[i] > 0 for i in range(1, n + 1)):
            return assigned_phase
        return None

    # Try Backward Greedy List Scheduling
    def try_backward_greedy():
        cur_out_degree = list(out_degree)
        assigned_phase = [0] * (n + 1)
        available = [i for i in range(1, n + 1) if cur_out_degree[i] == 0]

        for t in range(T, 0, -1):
            available.sort(key=lambda x: (-E[x], -in_degree[x]))
            chosen = available[:M]
            available = available[M:]

            new_available = []
            for u in chosen:
                assigned_phase[u] = t
                for v in rev_adj[u]:
                    cur_out_degree[v] -= 1
                    if cur_out_degree[v] == 0:
                        new_available.append(v)
            available.extend(new_available)

        if all(assigned_phase[i] > 0 for i in range(1, n + 1)):
            # Validate precedence
            for u in range(1, n + 1):
                for v in adj[u]:
                    if assigned_phase[u] >= assigned_phase[v]:
                        return None
            return assigned_phase
        return None

    # Backtracking solver as fallback
    def solve_backtrack():
        assigned_phase = [0] * (n + 1)
        cur_in_degree = list(in_degree)
        available = [i for i in range(1, n + 1) if cur_in_degree[i] == 0]

        import itertools

        steps = [0]
        max_steps = 200000

        def dfs(t, avail):
            steps[0] += 1
            if steps[0] > max_steps:
                return False
            if t > T:
                return len(avail) == 0

            # Mandatory jobs that MUST be done at phase t
            mandatory = [x for x in avail if L[x] == t]
            if len(mandatory) > M:
                return False

            optional = [x for x in avail if L[x] > t]
            optional.sort(key=lambda x: (L[x], -out_degree[x]))

            needed = min(M - len(mandatory), len(optional))
            
            # If all available jobs fit
            if len(avail) <= M:
                combos = [list(avail)]
            else:
                # Try combinations taking all mandatory + subset of optional
                combos = []
                # Greedy choice first
                best_opt = optional[:needed]
                combos.append(mandatory + best_opt)
                # If needed, branch a bit
                if len(optional) > needed:
                    for extra in itertools.combinations(optional[:min(len(optional), needed + 2)], needed):
                        c = mandatory + list(extra)
                        if c not in combos:
                            combos.append(c)

            for chosen in combos:
                chosen_set = set(chosen)
                new_avail = [x for x in avail if x not in chosen_set]
                for u in chosen:
                    assigned_phase[u] = t
                    for v in adj[u]:
                        cur_in_degree[v] -= 1
                        if cur_in_degree[v] == 0:
                            new_avail.append(v)

                if dfs(t + 1, new_avail):
                    return True

                # Backtrack
                for u in chosen:
                    assigned_phase[u] = 0
                    for v in adj[u]:
                        cur_in_degree[v] += 1

            return False

        if dfs(1, available):
            return assigned_phase
        return None

    res = try_forward_greedy()
    if res is None:
        res = try_backward_greedy()
    if res is None:
        res = solve_backtrack()

    if res is not None:
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
