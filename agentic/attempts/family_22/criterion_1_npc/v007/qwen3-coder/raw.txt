from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort with priority to earliest slot
    # We use Kahn's algorithm with a twist: among available nodes, we prioritize by slot
    
    # Initialize queue with nodes having in-degree 0
    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    # slot_count[t] = number of tasks scheduled in slot t (1-based)
    slot_count = [0] * (T + 1)
    result = [0] * (n + 1)  # result[i] = slot of task i
    
    # For each task, track the earliest slot it can start based on dependencies
    earliest_slot = [1] * (n + 1)
    
    # Process tasks in topological order
    processed = 0
    
    while queue:
        # Sort queue to process tasks in a consistent order
        temp_queue = list(queue)
        temp_queue.sort()
        queue = deque(temp_queue)
        
        # Try to schedule as many tasks as possible in current slots
        next_queue = []
        
        while queue:
            node = queue.popleft()
            scheduled = False
            
            # Try to schedule this task in its earliest possible slot
            slot = earliest_slot[node]
            if slot <= T and slot_count[slot] < M:
                result[node] = slot
                slot_count[slot] += 1
                scheduled = True
                processed += 1
                
                # Update neighbors' earliest slot
                for neighbor in graph[node]:
                    in_degree[neighbor] -= 1
                    if earliest_slot[neighbor] < slot + 1:
                        earliest_slot[neighbor] = slot + 1
                    if in_degree[neighbor] == 0:
                        next_queue.append(neighbor)
            elif slot < T:  # Try next slot
                earliest_slot[node] = slot + 1
                queue.appendleft(node)  # Put it back to try again
            else:
                # Cannot schedule this task within deadline
                print("NO")
                return
        
        # Add newly available tasks to queue
        next_queue.sort()
        queue.extend(next_queue)
    
    if processed != n:
        print("NO")
        return
    
    print("YES")
    print(*result[1:])

solve()
