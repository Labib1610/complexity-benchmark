```python
import sys
import heapq
import itertools

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 4:
        return

    n, M, T, m = data[:4]

    if n == 0:
        print("YES")
        print()
        return

    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    successors = [[] for _ in range(n)]
    predecessors = [0] * n
    indegree = [0] * n

    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2

        if not (0 <= a < n and 0 <= b < n):
            print("NO")
            return

        bit = 1 << a
        if not (predecessors[b] & bit):
            predecessors[b] |= bit
            successors[a].append(b)
            indegree[b] += 1

    # Topological order and cycle detection.
    queue = [v for v in range(n) if indegree[v] == 0]
    topo = []
    head = 0
    indeg_work = indegree[:]

    while head < len(queue):
        v = queue[head]
        head += 1
        topo.append(v)
        for w in successors[v]:
            indeg_work[w] -= 1
            if indeg_work[w] == 0:
                queue.append(w)

    if len(topo) != n:
        print("NO")
        return

    # Longest path beginning at each job, including the job itself.
    bottom = [1] * n
    for v in reversed(topo):
        best = 0
        for w in successors[v]:
            if bottom[w] > best:
                best = bottom[w]
        bottom[v] = best + 1

    if max(bottom) > T:
        print("NO")
        return

    # Fast critical-path-priority list scheduling.
    heap = []
    indeg_work = indegree[:]
    for v in range(n):
        if indeg_work[v] == 0:
            heapq.heappush(heap, (-bottom[v], -len(successors[v]), v))

    greedy_slots = [0] * n
    current_slot = 0
    completed = 0

    while heap:
        current_slot += 1
        selected = []
        for _ in range(min(M, len(heap))):
            _, _, v = heapq.heappop(heap)
            selected.append(v)
            greedy_slots[v] = current_slot
            completed += 1

        newly_available = []
        for v in selected:
            for w in successors[v]:
                indeg_work[w] -= 1
                if indeg_work[w] == 0:
                    newly_available.append(w)

        for w in newly_available:
            heapq.heappush(heap, (-bottom[w], -len(successors[w]), w))

    if completed == n and current_slot <= T:
        print("YES")
        print(*greedy_slots)
        return

    # Exact search. It is only reached when the fast schedule misses the
    # deadline even though the elementary lower bounds permit feasibility.
    sys.setrecursionlimit(max(10000, n + 100))
    full_mask = (1 << n) - 1
    answer = [0] * n
    best_seen_slot = {}

    def dfs(done, slot):
        if done == full_mask:
            return True
        if slot > T:
            return False

        previous = best_seen_slot.get(done)
        if previous is not None and previous <= slot:
            return False
        best_seen_slot[done] = slot

        remaining_mask = full_mask ^ done
        remaining_jobs = remaining_mask.bit_count()
        remaining_slots = T - slot + 1

        if remaining_jobs > M * remaining_slots:
            return False

        available = []
        frequencies = {}

        bits = remaining_mask
        while bits:
            low = bits & -bits
            v = low.bit_length() - 1
            bits ^= low

            b = bottom[v]
            if b > remaining_slots:
                return False
            frequencies[b] = frequencies.get(b, 0) + 1

            if predecessors[v] & remaining_mask == 0:
                available.append(v)

        if not available:
            return False

        # Jobs with bottom value b have to be scheduled no later than
        # relative slot remaining_slots-b+1.
        cumulative = 0
        saturation_threshold = 0
        for b in sorted(frequencies, reverse=True):
            cumulative += frequencies[b]
            capacity = M * (remaining_slots - b + 1)
            if cumulative > capacity:
                return False
            if cumulative == capacity and b > saturation_threshold:
                saturation_threshold = b

        k = min(M, len(available))

        mandatory = [v for v in available if bottom[v] == remaining_slots]
        if len(mandatory) > k:
            return False

        mandatory_set = set(mandatory)

        if saturation_threshold:
            eligible = [v for v in available if bottom[v] >= saturation_threshold]
            if len(eligible) < k:
                return False
        else:
            eligible = available

        if any(v not in eligible for v in mandatory):
            return False

        optional = [v for v in eligible if v not in mandatory_set]
        optional.sort(key=lambda v: (-bottom[v], -len(successors[v]), v))
        mandatory.sort(key=lambda v: (-bottom[v], -len(successors[v]), v))

        need = k - len(mandatory)
        if need < 0 or need > len(optional):
            return False

        combinations = ((),) if need == 0 else itertools.combinations(optional, need)

        mandatory_mask = 0
        for v in mandatory:
            mandatory_mask |= 1 << v

        for combination in combinations:
            selected = mandatory + list(combination)
            selected_mask = mandatory_mask
            for v in combination:
                selected_mask |= 1 << v

            for v in selected:
                answer[v] = slot

            if dfs(done | selected_mask, slot + 1):
                return True

            for v in selected:
                answer[v] = 0

        return False

    if dfs(0, 1):
        print("YES")
        print(*answer)
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
