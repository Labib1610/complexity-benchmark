We can model this as a scheduling problem with precedence constraints and a capacity limit per time slot.  

We can solve it using a topological sort with a greedy assignment of the earliest possible slot, while respecting both precedence and station capacity.  

Here’s the implementation:  

```python
from collections import deque
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    deadline = int(next(it))
    p = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(p):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Kahn's algorithm with slot assignment
    indeg_copy = indeg[:]
    q = deque()
    for i in range(n):
        if indeg_copy[i] == 0:
            q.append(i)
    
    slot = [0] * n
    # capacity per slot
    cap = [0] * (deadline + 1)
    
    processed = 0
    while q:
        u = q.popleft()
        # assign earliest possible slot
        s = 1
        while s <= deadline and cap[s] >= m:
            s += 1
        if s > deadline:
            print("NO")
            return
        slot[u] = s
        cap[s] += 1
        processed += 1
        for v in adj[u]:
            indeg_copy[v] -= 1
            if indeg_copy[v] == 0:
                q.append(v)
    
    if processed != n:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(x) for x in slot))

if __name__ == "__main__":
    solve()
```
