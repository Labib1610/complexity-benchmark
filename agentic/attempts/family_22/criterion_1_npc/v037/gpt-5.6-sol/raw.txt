```python
import sys
import heapq
import itertools

def print_yes(slots):
    print("YES")
    print(*slots)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, deadline, e = data[:4]
    succ = [[] for _ in range(n)]
    pred = [[] for _ in range(n)]
    indeg = [0] * n

    pos = 4
    for _ in range(e):
        a = data[pos] - 1
        b = data[pos + 1] - 1
        pos += 2
        succ[a].append(b)
        pred[b].append(a)
        indeg[b] += 1

    if m <= 0 or deadline <= 0 or n > m * deadline:
        print("NO")
        return

    # Topological order and cycle detection.
    heap = [v for v in range(n) if indeg[v] == 0]
    heapq.heapify(heap)
    deg = indeg[:]
    topo = []

    while heap:
        v = heapq.heappop(heap)
        topo.append(v)
        for w in succ[v]:
            deg[w] -= 1
            if deg[w] == 0:
                heapq.heappush(heap, w)

    if len(topo) != n:
        print("NO")
        return

    # Longest path lengths ending at / starting from each vertex.
    top_len = [1] * n
    for v in topo:
        if pred[v]:
            top_len[v] = 1 + max(top_len[p] for p in pred[v])

    bottom_len = [1] * n
    for v in reversed(topo):
        if succ[v]:
            bottom_len[v] = 1 + max(bottom_len[w] for w in succ[v])

    critical_path = max(bottom_len, default=0)
    if critical_path > deadline:
        print("NO")
        return

    if n == 0:
        print("YES")
        print()
        return

    if m == 1:
        if n > deadline:
            print("NO")
        else:
            slots = [0] * n
            for t, v in enumerate(topo, 1):
                slots[v] = t
            print_yes(slots)
        return

    if deadline >= n:
        slots = [0] * n
        for t, v in enumerate(topo, 1):
            slots[v] = t
        print_yes(slots)
        return

    if m >= n:
        if critical_path <= deadline:
            print_yes(top_len)
        else:
            print("NO")
        return

    def coffman_graham_schedule():
        # Coffman-Graham lexicographic labeling.
        remaining_successors = [len(succ[v]) for v in range(n)]
        labels = [0] * n
        ready = []

        for v in range(n):
            if remaining_successors[v] == 0:
                heapq.heappush(ready, ((), v))

        for label in range(1, n + 1):
            if not ready:
                return None
            _, v = heapq.heappop(ready)
            labels[v] = label
            for p in pred[v]:
                remaining_successors[p] -= 1
                if remaining_successors[p] == 0:
                    key = tuple(sorted((labels[w] for w in succ[p]), reverse=True))
                    heapq.heappush(ready, (key, p))

        order = sorted(range(n), key=lambda x: labels[x], reverse=True)

        capacity = [0] * (n + 2)
        parent = list(range(n + 2))

        def find(x):
            y = x
            while parent[y] != y:
                y = parent[y]
            while parent[x] != x:
                nx = parent[x]
                parent[x] = y
                x = nx
            return y

        slots = [0] * n
        maximum = 0

        for v in order:
            earliest = 1
            for p in pred[v]:
                pv = slots[p] + 1
                if pv > earliest:
                    earliest = pv

            t = find(earliest)
            slots[v] = t
            capacity[t] += 1
            if capacity[t] == m:
                parent[t] = find(t + 1)
            if t > maximum:
                maximum = t

        return slots, maximum

    cg_result = coffman_graham_schedule()
    if cg_result is not None:
        cg_slots, cg_length = cg_result
        if cg_length <= deadline:
            print_yes(cg_slots)
            return
        # Coffman-Graham is optimal for two stations.
        if m == 2:
            print("NO")
            return

    def priority_list_schedule(priority):
        remaining = indeg[:]
        ready = []
        for v in range(n):
            if remaining[v] == 0:
                key = priority(v)
                heapq.heappush(ready, tuple(-x for x in key) + (v,))

        slots = [0] * n
        done = 0
        t = 0

        while ready:
            t += 1
            selected = []
            for _ in range(min(m, len(ready))):
                item = heapq.heappop(ready)
                selected.append(item[-1])

            done += len(selected)
            newly_ready = []
            for v in selected:
                slots[v] = t
                for w in succ[v]:
                    remaining[w] -= 1
                    if remaining[w] == 0:
                        newly_ready.append(w)

            for w in newly_ready:
                key = priority(w)
                heapq.heappush(ready, tuple(-x for x in key) + (w,))

        if done != n:
            return None, n + 1
        return slots, t

    priorities = [
        lambda v: (bottom_len[v], len(succ[v]), -v),
        lambda v: (bottom_len[v], -top_len[v], len(succ[v]), -v),
        lambda v: (len(succ[v]), bottom_len[v], -v),
    ]

    for priority in priorities:
        slots, length = priority_list_schedule(priority)
        if length <= deadline:
            print_yes(slots)
            return

    # Exact branch-and-bound. It may run exponentially, as the general problem
    # for three or more stations is NP-hard.
    source_count = sum(1 for v in range(n) if not pred[v])
    sink_count = sum(1 for v in range(n) if not succ[v])

    reversed_orientation = sink_count < source_count
    if reversed_orientation:
        oriented_pred = succ
        oriented_succ = pred
        tail = top_len
    else:
        oriented_pred = pred
        oriented_succ = succ
        tail = bottom_len

    pred_mask = [0] * n
    for v in range(n):
        mask = 0
        for p in oriented_pred[v]:
            mask |= 1 << p
        pred_mask[v] = mask

    initial_available = 0
    for v in range(n):
        if not oriented_pred[v]:
            initial_available |= 1 << v

    all_mask = (1 << n) - 1
    failed_with_remaining = {}
    path = []

    sys.setrecursionlimit(max(10000, n + 100))

    def necessary_bound(done, remaining_slots):
        remaining_mask = all_mask ^ done
        count = remaining_mask.bit_count()

        if count > m * remaining_slots:
            return False

        frequencies = [0] * (remaining_slots + 1)
        x = remaining_mask
        while x:
            bit = x & -x
            v = bit.bit_length() - 1
            b = tail[v]
            if b > remaining_slots:
                return False
            frequencies[b] += 1
            x ^= bit

        cumulative = 0
        for b in range(remaining_slots, 0, -1):
            cumulative += frequencies[b]
            available_early_slots = remaining_slots - b + 1
            if cumulative > m * available_early_slots:
                return False
        return True

    def add_newly_available(done, available, selected):
        new_available = available & ~selected
        x = selected
        remaining_mask = all_mask ^ done
        while x:
            bit = x & -x
            v = bit.bit_length() - 1
            for w in oriented_succ[v]:
                wb = 1 << w
                if (remaining_mask & wb) and not (new_available & wb):
                    if pred_mask[w] & remaining_mask == 0:
                        new_available |= wb
            x ^= bit
        return new_available

    def dfs(done, available, used_slots):
        if done == all_mask:
            return True

        remaining_slots = deadline - used_slots
        if remaining_slots <= 0:
            return False

        previous_failed = failed_with_remaining.get(done)
        if previous_failed is not None and previous_failed >= remaining_slots:
            return False

        if not necessary_bound(done, remaining_slots):
            failed_with_remaining[done] = max(
                failed_with_remaining.get(done, -1), remaining_slots
            )
            return False

        available_vertices = []
        x = available
        while x:
            bit = x & -x
            available_vertices.append(bit.bit_length() - 1)
            x ^= bit

        if not available_vertices:
            failed_with_remaining[done] = max(
                failed_with_remaining.get(done, -1), remaining_slots
            )
            return False

        available_vertices.sort(
            key=lambda v: (tail[v], len(oriented_succ[v]), -v),
            reverse=True
        )

        take = min(m, len(available_vertices))

        mandatory = [v for v in available_vertices if tail[v] == remaining_slots]
        if len(mandatory) > take:
            failed_with_remaining[done] = max(
                failed_with_remaining.get(done, -1), remaining_slots
            )
            return False

        mandatory_set = set(mandatory)
        optional = [v for v in available_vertices if v not in mandatory_set]
        optional_needed = take - len(mandatory)

        mandatory_mask = 0
        for v in mandatory:
            mandatory_mask |= 1 << v

        if optional_needed == 0:
            combinations = [()]
        elif optional_needed == len(optional):
            combinations = [tuple(optional)]
        else:
            combinations = itertools.combinations(optional, optional_needed)

        for combination in combinations:
            selected = mandatory_mask
            for v in combination:
                selected |= 1 << v

            new_done = done | selected
            new_available = add_newly_available(new_done, available, selected)

            if not necessary_bound(new_done, remaining_slots - 1):
                continue

            path.append(selected)
            if dfs(new_done, new_available, used_slots + 1):
                return True
            path.pop()

        failed_with_remaining[done] = max(
            failed_with_remaining.get(done, -1), remaining_slots
        )
        return False

    if not dfs(0, initial_available, 0):
        print("NO")
        return

    slots = [0] * n
    for t, selected in enumerate(path, 1):
        x = selected
        while x:
            bit = x & -x
            v = bit.bit_length() - 1
            slots[v] = deadline + 1 - t if reversed_orientation else t
            x ^= bit

    print_yes(slots)

if __name__ == "__main__":
    main()
```
