```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    D = int(next(it))
    k = int(next(it))

    adj = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(k):
        a = int(next(it))-1
        b = int(next(it))-1
        adj[a].append(b)
        indeg[b] += 1

    # topological order
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    while q:
        v = q.popleft()
        topo.append(v)
        for nb in adj[v]:
            indeg[nb] -= 1
            if indeg[nb]==0:
                q.append(nb)
    if len(topo)!=n:
        print("NO")
        return

    # compute earliest slot L_i (1‑based)
    L = [1]*n
    for v in topo:
        for nb in adj[v]:
            if L[nb] < L[v] + 1:
                L[nb] = L[v] + 1

    # if any earliest exceeds deadline -> impossible
    if any(l > D for l in L):
        print("NO")
        return

    # sort jobs by earliest slot
    jobs_by_L = sorted(range(n), key=lambda i: L[i])
    ptr = 0
    from collections import deque
    ready = deque()   # simple queue works because all jobs have same upper bound D
    slot_of = [0]*n
    for t in range(1, D+1):
        # add newly available jobs
        while ptr < n and L[jobs_by_L[ptr]] <= t:
            ready.append(jobs_by_L[ptr])
            ptr += 1
        # assign up to m jobs this slot
        for _ in range(m):
            if not ready:
                break
            job = ready.popleft()
            slot_of[job] = t
    # after all slots, check if all assigned
    if any(s==0 for s in slot_of):
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, slot_of)))

if __name__ == "__main__":
    solve()
```
