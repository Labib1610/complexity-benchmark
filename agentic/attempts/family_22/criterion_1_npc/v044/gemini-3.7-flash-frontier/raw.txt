import itertools
from collections import deque
import sys


def solve():
    sys.setrecursionlimit(200000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    M = int(input_data[1])
    T = int(input_data[2])
    m = int(input_data[3])

    if n > M * T:
        print("NO")
        return

    adj = [[] for _ in range(n + 1)]
    radj = [[] for _ in range(n + 1)]
    in_deg = [0] * (n + 1)
    idx = 4
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        radj[v].append(u)
        in_deg[v] += 1

    # Topological sort & cycle detection
    q = deque([i for i in range(1, n + 1) if in_deg[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            in_deg[v] -= 1
            if in_deg[v] == 0:
                q.append(v)

    if len(topo) < n:
        print("NO")
        return

    ES = [1] * (n + 1)
    for u in topo:
        for v in adj[u]:
            if ES[u] + 1 > ES[v]:
                ES[v] = ES[u] + 1

    LS = [T] * (n + 1)
    for u in reversed(topo):
        for v in adj[u]:
            if LS[v] - 1 < LS[u]:
                LS[u] = LS[v] - 1

    for u in range(1, n + 1):
        if ES[u] > LS[u] or ES[u] > T or LS[u] < 1:
            print("NO")
            return

    dist_to_sink = [0] * (n + 1)
    for u in reversed(topo):
        d = 0
        for v in adj[u]:
            if dist_to_sink[v] + 1 > d:
                d = dist_to_sink[v] + 1
        dist_to_sink[u] = d

    in_degree_curr = [len(radj[i]) for i in range(n + 1)]
    assigned_time = [0] * (n + 1)
    ready = [i for i in range(1, n + 1) if in_degree_curr[i] == 0]

    def get_priority(u):
        return (LS[u], -dist_to_sink[u], -len(adj[u]))

    def backtrack(t, ready_list, count_assigned):
        if count_assigned == n:
            return True
        if t > T:
            return False

        mandatory = []
        optional = []
        for u in ready_list:
            if LS[u] < t:
                return False
            elif LS[u] == t:
                mandatory.append(u)
            else:
                optional.append(u)

        if len(mandatory) > M:
            return False

        slots_left = M - len(mandatory)
        if len(optional) <= slots_left:
            chosen_subsets = [optional]
        else:
            optional.sort(key=get_priority)
            chosen_subsets = itertools.combinations(optional, slots_left)

        for opt_chosen in chosen_subsets:
            chosen = mandatory + list(opt_chosen)
            for u in chosen:
                assigned_time[u] = t

            next_ready = [u for u in ready_list if assigned_time[u] == 0]
            for u in chosen:
                for v in adj[u]:
                    in_degree_curr[v] -= 1
                    if in_degree_curr[v] == 0:
                        next_ready.append(v)

            if backtrack(t + 1, next_ready, count_assigned + len(chosen)):
                return True

            for u in chosen:
                for v in adj[u]:
                    in_degree_curr[v] += 1
                assigned_time[u] = 0

        return False

    if backtrack(1, ready, 0):
        print("YES")
        print(*(assigned_time[i] for i in range(1, n + 1)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
