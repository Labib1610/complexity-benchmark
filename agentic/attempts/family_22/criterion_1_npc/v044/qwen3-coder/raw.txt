from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    M = int(data[idx+1])
    T = int(data[idx+2])
    m = int(data[idx+3])
    idx += 4
    
    # Build graph
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx+1])
        idx += 2
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with priority to minimize time slots
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    # time[i] will store the time slot assigned to operation i
    time = [0] * (n + 1)
    count = 0
    
    # For each time slot, we can schedule up to M operations
    current_time = 1
    while count < n:
        if not queue:
            # No operation can be scheduled but not all are done => cycle
            print("NO")
            return
        
        # Schedule up to M operations in current_time
        scheduled_in_this_slot = 0
        next_queue = deque()
        while queue and scheduled_in_this_slot < M:
            op = queue.popleft()
            time[op] = current_time
            count += 1
            scheduled_in_this_slot += 1
            
            # Reduce indegree of neighbors
            for neighbor in graph[op]:
                indegree[neighbor] -= 1
                if indegree[neighbor] == 0:
                    next_queue.append(neighbor)
        
        # Add remaining operations back to queue for next time slot
        while queue:
            next_queue.append(queue.popleft())
        
        queue = next_queue
        current_time += 1
        
        if current_time > T:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, time[1:])))

if __name__ == "__main__":
    main()
