import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1

    # earliest feasible hour respecting dependencies
    earliest = [1] * (n + 1)
    q = deque(i for i in range(1, n + 1) if indeg[i] == 0)
    seen = 0
    while q:
        u = q.popleft()
        seen += 1
        for v in adj[u]:
            if earliest[u] + 1 > earliest[v]:
                earliest[v] = earliest[u] + 1
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if seen != n:
        print("NO")
        return

    if any(earliest[i] > T for i in range(1, n + 1)):
        print("NO")
        return

    # latest feasible hour respecting dependencies
    latest = [T] * (n + 1)
    radj = [[] for _ in range(n + 1)]
    outdeg = [0] * (n + 1)
    for u in range(1, n + 1):
        for v in adj[u]:
            radj[v].append(u)
            outdeg[u] += 1

    q = deque(i for i in range(1, n + 1) if outdeg[i] == 0)
    while q:
        v = q.popleft()
        for u in radj[v]:
            if latest[v] - 1 < latest[u]:
                latest[u] = latest[v] - 1
            outdeg[u] -= 1
            if outdeg[u] == 0:
                q.append(u)

    if any(latest[i] < earliest[i] for i in range(1, n + 1)):
        print("NO")
        return

    # group jobs by earliest hour
    by_earliest = [[] for _ in range(T + 2)]
    for i in range(1, n + 1):
        by_earliest[earliest[i]].append(i)

    hour = [0] * (n + 1)
    ready = []
    cap = [0] * (T + 2)

    for t in range(1, T + 1):
        for x in by_earliest[t]:
            ready.append((latest[x], x))
        ready.sort()

        # jobs that must be scheduled now or earlier
        must = [x for (lt, x) in ready if lt == t]
        if len(must) > M:
            print("NO")
            return

        for x in must:
            hour[x] = t
            cap[t] += 1

        ready = [(lt, x) for (lt, x) in ready if lt != t]

        # fill remaining capacity greedily by latest deadline
        idx = 0
        while cap[t] < M and idx < len(ready):
            lt, x = ready[idx]
            if lt < t:
                print("NO")
                return
            hour[x] = t
            cap[t] += 1
            idx += 1
        ready = ready[idx:]

    if any(hour[i] == 0 for i in range(1, n + 1)):
        print("NO")
        return

    print("YES")
    print(" ".join(str(hour[i]) for i in range(1, n + 1)))

solve()
