```python
import sys
import heapq
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

n, M, T, m = data[:4]
adj = [[] for _ in range(n)]
indeg = [0] * n

p = 4
for _ in range(m):
    a = data[p] - 1
    b = data[p + 1] - 1
    p += 2
    adj[a].append(b)
    indeg[b] += 1

if n == 0:
    print("YES")
    print()
    sys.exit()

if M <= 0 or T <= 0 or n > M * T:
    print("NO")
    sys.exit()

# Topological sort and cycle detection.
q = deque(i for i in range(n) if indeg[i] == 0)
topo = []
deg = indeg[:]

while q:
    u = q.popleft()
    topo.append(u)
    for v in adj[u]:
        deg[v] -= 1
        if deg[v] == 0:
            q.append(v)

if len(topo) != n:
    print("NO")
    sys.exit()

# bottom[u] = length of a longest dependency chain starting at u.
bottom = [1] * n
for u in reversed(topo):
    best = 0
    for v in adj[u]:
        if bottom[v] > best:
            best = bottom[v]
    bottom[u] = best + 1

if max(bottom) > T:
    print("NO")
    sys.exit()

# Necessary suffix-capacity conditions.
hist = [0] * (max(bottom) + 1)
for b in bottom:
    hist[b] += 1

count = 0
for k in range(len(hist) - 1, 0, -1):
    count += hist[k]
    if count > M * (T - k + 1):
        print("NO")
        sys.exit()


def greedy_schedule(mode):
    d = indeg[:]
    heap = []

    def key(u):
        if mode == 0:
            return (-bottom[u], -len(adj[u]), u)
        if mode == 1:
            return (-len(adj[u]), -bottom[u], u)
        return (-bottom[u], len(adj[u]), -u)

    for i in range(n):
        if d[i] == 0:
            heapq.heappush(heap, key(i))

    ans = [0] * n
    completed = 0
    hour = 1

    while heap:
        chosen = []
        for _ in range(min(M, len(heap))):
            item = heapq.heappop(heap)
            u = item[2] if mode != 2 else -item[2]
            chosen.append(u)
            ans[u] = hour

        completed += len(chosen)
        for u in chosen:
            for v in adj[u]:
                d[v] -= 1
                if d[v] == 0:
                    heapq.heappush(heap, key(v))

        hour += 1

    if completed == n and hour - 1 <= T:
        return ans
    return None


# Fast attempts that solve most instances.
for mode in range(3):
    answer = greedy_schedule(mode)
    if answer is not None:
        print("YES")
        print(*answer)
        sys.exit()

# Exact search for instances where list scheduling did not suffice.
pred_mask = [0] * n
for u in range(n):
    bit = 1 << u
    for v in adj[u]:
        pred_mask[v] |= bit

succ_key = []
for u in range(n):
    if len(adj[u]) <= 1:
        succ_key.append(tuple(adj[u]))
    else:
        succ_key.append(tuple(sorted(set(adj[u]))))

all_mask = (1 << n) - 1
answer = [0] * n
memo = {}


def dfs(done, hour):
    if done == all_mask:
        return True
    if hour > T:
        return False

    previous = memo.get(done)
    if previous is not None and previous <= hour:
        return False
    memo[done] = hour

    remaining_slots = T - hour + 1
    remaining_count = n - done.bit_count()
    if remaining_count > M * remaining_slots:
        return False

    available = []
    level_hist = [0] * (min(n, remaining_slots + 1) + 1)
    max_level = 0

    not_done = all_mask ^ done
    bits = not_done
    while bits:
        low = bits & -bits
        u = low.bit_length() - 1
        bits -= low

        b = bottom[u]
        if b > remaining_slots:
            return False
        if b > max_level:
            max_level = b
        if b >= len(level_hist):
            return False
        level_hist[b] += 1

        if pred_mask[u] & not_done == 0:
            available.append(u)

    if not available:
        return False

    # Deadline-capacity tests and a possible tight prefix threshold.
    due = 0
    tight_threshold = 0
    for k in range(max_level, 0, -1):
        due += level_hist[k]
        capacity = M * (remaining_slots - k + 1)
        if due > capacity:
            return False
        if due == capacity and tight_threshold == 0:
            tight_threshold = k

    available.sort(key=lambda u: (-bottom[u], -len(adj[u]), u))

    if len(available) <= M:
        chosen = available
        chosen_mask = 0
        for u in chosen:
            chosen_mask |= 1 << u
            answer[u] = hour

        if dfs(done | chosen_mask, hour + 1):
            return True

        for u in chosen:
            answer[u] = 0
        return False

    forced = [u for u in available if bottom[u] == remaining_slots]
    if len(forced) > M:
        return False

    forced_set = set(forced)
    candidates = [
        u for u in available
        if u not in forced_set and
        (tight_threshold == 0 or bottom[u] >= tight_threshold)
    ]

    need = M - len(forced)
    if len(candidates) < need:
        return False

    # Jobs with identical successor sets are interchangeable once available.
    groups_dict = {}
    for u in candidates:
        groups_dict.setdefault(succ_key[u], []).append(u)
    groups = list(groups_dict.values())
    groups.sort(key=lambda g: (-bottom[g[0]], -len(adj[g[0]]), g[0]))

    suffix_sizes = [0] * (len(groups) + 1)
    for i in range(len(groups) - 1, -1, -1):
        suffix_sizes[i] = suffix_sizes[i + 1] + len(groups[i])

    base_mask = 0
    for u in forced:
        base_mask |= 1 << u
        answer[u] = hour

    selected = []

    def enumerate_groups(index, left, selected_mask):
        if left == 0:
            for u in selected:
                answer[u] = hour

            if dfs(done | base_mask | selected_mask, hour + 1):
                return True

            for u in selected:
                answer[u] = 0
            return False

        if index == len(groups) or suffix_sizes[index] < left:
            return False

        group = groups[index]
        max_take = min(len(group), left)
        min_take = max(0, left - suffix_sizes[index + 1])

        for take in range(max_take, min_take - 1, -1):
            added_mask = 0
            if take:
                for j in range(take):
                    u = group[j]
                    selected.append(u)
                    added_mask |= 1 << u

            if enumerate_groups(index + 1, left - take,
                                selected_mask | added_mask):
                return True

            if take:
                del selected[-take:]

        return False

    if enumerate_groups(0, need, 0):
        return True

    for u in forced:
        answer[u] = 0
    return False


if dfs(0, 1):
    print("YES")
    print(*answer)
else:
    print("NO")
```
