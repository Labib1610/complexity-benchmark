import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    radj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1

    # earliest feasible slot respecting precedence
    earliest = [1] * (n + 1)
    q = deque(i for i in range(1, n + 1) if indeg[i] == 0)
    indeg_work = indeg[:]
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
            indeg_work[v] -= 1
            if indeg_work[v] == 0:
                q.append(v)

    if len(topo) != n:
        print("NO")
        return

    # latest feasible slot respecting precedence
    latest = [T] * (n + 1)
    for u in reversed(topo):
        for v in adj[u]:
            if latest[u] > latest[v] - 1:
                latest[u] = latest[v] - 1

    for i in range(1, n + 1):
        if earliest[i] > latest[i]:
            print("NO")
            return

    # group by earliest slot
    by_earliest = [[] for _ in range(T + 2)]
    for i in range(1, n + 1):
        by_earliest[earliest[i]].append(i)

    slot = [0] * (n + 1)
    ready = []
    cap = [0] * (T + 2)

    for t in range(1, T + 1):
        for v in by_earliest[t]:
            heapq.heappush(ready, (latest[v], v))
        while ready and ready[0][0] < t:
            heapq.heappop(ready)
        cnt = 0
        while cnt < M and ready:
            _, v = heapq.heappop(ready)
            slot[v] = t
            cnt += 1
        cap[t] = cnt

    if any(slot[i] == 0 for i in range(1, n + 1)):
        print("NO")
        return

    # verify precedence
    ok = True
    for u in range(1, n + 1):
        for v in adj[u]:
            if slot[u] >= slot[v]:
                ok = False
                break
        if not ok:
            break

    if not ok:
        print("NO")
        return

    print("YES")
    print(" ".join(str(slot[i]) for i in range(1, n + 1)))

import heapq
solve()
