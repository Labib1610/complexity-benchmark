from collections import defaultdict, deque

def solve():
    n, M, T, m = map(int, input().split())
    
    # Build graph and in-degree count
    graph = defaultdict(list)
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        a, b = map(int, input().split())
        graph[a].append(b)
        in_degree[b] += 1
    
    # Topological sort with slot assignment
    # We'll use a greedy approach: assign each task to the earliest possible slot
    # respecting both precedence and station capacity constraints
    
    # Find all nodes with in-degree 0
    queue = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    # slot_tasks[slot] = number of tasks assigned to slot
    slot_tasks = [0] * (T + 1)
    # task_slot[i] = slot assigned to task i
    task_slot = [0] * (n + 1)
    
    # Process nodes in topological order
    while queue:
        current = queue.popleft()
        
        # Find the earliest slot where we can place this task
        # It must be after all its predecessors
        earliest_slot = 1
        for pred in graph:
            if current in graph[pred]:
                earliest_slot = max(earliest_slot, task_slot[pred] + 1)
        
        # Place in the earliest available slot that has capacity
        placed = False
        for slot in range(earliest_slot, T + 1):
            if slot_tasks[slot] < M:
                slot_tasks[slot] += 1
                task_slot[current] = slot
                placed = True
                break
        
        if not placed:
            print("NO")
            return
        
        # Update in-degrees of neighbors
        for neighbor in graph[current]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # Check if all tasks were scheduled
    for i in range(1, n + 1):
        if task_slot[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(*task_slot[1:])

solve()
