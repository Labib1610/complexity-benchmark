We can model this as a scheduling problem with precedence constraints and capacity limits.

A common approach is:

1. Build a directed graph for the dependencies (a → b means a must be scheduled before b).
2. If there’s a cycle, no valid schedule exists.
3. Use topological sorting to determine earliest possible time for each operation.
4. Then adjust the schedule so that in each time slot, no more than M operations are scheduled.  
   This can be done by sorting operations by their earliest start time and assigning them to the earliest available slot respecting capacity.

Below is a Python implementation.  

```python
from collections import deque, defaultdict
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1

    # Topological sort to detect cycles and compute earliest times
    earliest = [1] * n
    q = deque([i for i in range(n) if indeg[i] == 0])
    count = 0
    while q:
        u = q.popleft()
        count += 1
        for v in adj[u]:
            earliest[v] = max(earliest[v], earliest[u] + 1)
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if count != n:
        print("NO")
        return

    # If any earliest time > T, impossible
    if max(earliest) > T:
        print("NO")
        return

    # Now assign times respecting capacity
    # Sort by earliest, then assign greedily to the earliest slot with capacity
    jobs = list(range(n))
    jobs.sort(key=lambda x: earliest[x])

    schedule = [0] * n
    # slots usage
    usage = [0] * (T + 1)
    for job in jobs:
        start = earliest[job]
        # find first slot >= start with capacity
        t = start
        while t <= T and usage[t] >= M:
            t += 1
        if t > T:
            print("NO")
            return
        schedule[job] = t
        usage[t] += 1

    print("YES")
    print(" ".join(map(str, schedule)))

if __name__ == "__main__":
    solve()
```
