import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    M = int(next(iterator))
    T = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n + 1)]
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(next(iterator))
        b = int(next(iterator))
        adj[a].append(b)
        indegree[b] += 1
    
    # Kahn's algorithm for topological sort
    in_degree = list(indegree)
    queue = deque()
    order = []
    
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            queue.append(i)
    
    while queue:
        u = queue.popleft()
        order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
    
    if len(order) != n:
        print("NO")
        return
    
    earliest = [1] * (n + 1)
    for u in order:
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
    
    # Check basic feasibility
    for i in range(1, n + 1):
        if earliest[i] > T:
            print("NO")
            return
    
    # Sort nodes by their earliest deadline (non-decreasing)
    nodes = list(range(1, n + 1))
    nodes.sort(key=lambda x: (earliest[x], x))
    
    schedule = [0] * (n + 1)
    bay_count = [0] * (T + 2)
    
    for u in nodes:
        t = earliest[u]
        while t <= T and bay_count[t] >= M:
            t += 1
        if t > T:
            print("NO")
            return
        schedule[u] = t
        bay_count[t] += 1
        for v in adj[u]:
            if earliest[v] < t + 1:
                earliest[v] = t + 1
                if earliest[v] > T:
                    print("NO")
                    return
    
    # Verify dependencies
    for u in range(1, n + 1):
        for v in adj[u]:
            if schedule[v] <= schedule[u]:
                print("NO")
                return
    
    print("YES")
    result_str = ' '.join(map(str, schedule[1:]))
    print(result_str)

if __name__ == "__main__":
    solve()
