from collections import defaultdict, deque

    idx += 1

        idx += 1

        idx += 1
        graph[a].append(b)

    time_slots = [0] * (T + 1)
    
    # For each operation, store its assigned time
    assign_time = [0] * (n + 1)
    
    # Priority queue: (time, operation)
    heap = []
    
    # Initialize heap with operations having in-degree 0
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            heappush(heap, (1, i))
    
    # Process operations in topological order
    while heap:
        current_time, op = heappop(heap)
        
        # Find the earliest time slot where we can schedule this operation
        # The operation must be scheduled after all its dependencies
        earliest = current_time
        for dep in graph[op]:
            if assign_time[op] != 0:
                earliest = max(earliest, assign_time[op] + 1)
            else:
                earliest = max(earliest, 1)
        
        # But also, we need to respect dependencies from incoming edges
        # Actually, we should track the max time of predecessors
        # Let's re-think the approach
        
    # Let's do a proper topological sort with tracking of earliest start time
    in_degree_copy = in_degree[:]
    assign_time = [0] * (n + 1)
    time_slots = [0] * (T + 1)
    queue = deque()
    earliest_start = [1] * (n + 1)
    
    for i in range(1, n + 1):
        if in_degree_copy[i] == 0:
            queue.append(i)
    
    processed = 0
    order = []
    
    while queue:
        op = queue.popleft()
        order.append(op)
        processed += 1
        for neighbor in graph[op]:
            in_degree_copy[neighbor] -= 1
            # Update earliest start time for neighbor
            earliest_start[neighbor] = max(earliest_start[neighbor], assign_time[op] + 1)
            if in_degree_copy[neighbor] == 0:
                queue.append(neighbor)
    
    if processed != n:
        print("NO")
        return
    
    # Now assign times respecting capacity M
    # We'll use a greedy approach: for each operation in topological order,
    # assign it to the earliest time slot that is not full and satisfies dependencies
    
    time_slots = [0] * (T + 1)
    
    for op in order:
        # Find earliest time >= earliest_start[op] where slot is not full
        t = earliest_start[op]
        while t <= T and time_slots[t] >= M:
            t += 1
        if t > T:
            print("NO")
            return
        assign_time[op] = t
        time_slots[t] += 1
    
    print("YES")
    print(' '.join(map(str, assign_time[1:])))
