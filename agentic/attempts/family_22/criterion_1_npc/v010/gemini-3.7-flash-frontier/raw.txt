import sys
from collections import deque

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    M = int(next(iterator))
    T = int(next(iterator))
    m = int(next(iterator))
    
    if n > M * T:
        print("NO")
        return
        
    adj = [[] for _ in range(n + 1)]
    rev_adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    out_degree = [0] * (n + 1)
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        rev_adj[v].append(u)
        in_degree[v] += 1
        out_degree[u] += 1
        
    # Topological sort & cycle detection
    q = deque([i for i in range(1, n + 1) if in_degree[i] == 0])
    topo_order = []
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)
                
    if len(topo_order) < n:
        print("NO")
        return
        
    # Earliest Start (ES)
    ES = [1] * (n + 1)
    for u in topo_order:
        for v in adj[u]:
            if ES[u] + 1 > ES[v]:
                ES[v] = ES[u] + 1
                
    # Latest Start (LS)
    LS = [T] * (n + 1)
    for u in reversed(topo_order):
        for v in adj[u]:
            if LS[v] - 1 < LS[u]:
                LS[u] = LS[v] - 1
                
    for i in range(1, n + 1):
        if ES[i] > LS[i] or ES[i] > T or LS[i] < 1:
            print("NO")
            return

    # Count of successors (depth / longest path to end)
    depth = [0] * (n + 1)
    for u in reversed(topo_order):
        d = 0
        for v in adj[u]:
            if depth[v] + 1 > d:
                d = depth[v] + 1
        depth[u] = d

    # Predecessor counts for dynamic tracking
    pred_count = [len(rev_adj[i]) for i in range(n + 1)]
    
    # Priority for choosing: smaller LS first, then larger depth, then larger out_degree
    # We can pre-calculate priority
    def priority(u):
        return (LS[u], -depth[u], -len(adj[u]))

    assigned_cycle = [0] * (n + 1)
    
    # Backtracking search to assign cycles from 1 to T
    # State: current cycle t, available tasks ready to be scheduled at <= t
    ready = [u for u in range(1, n + 1) if pred_count[u] == 0]
    
    import itertools

    found = False
    
    # Memoization of failed states: (t, tuple(sorted(ready)))
    # Since ready can be large, we can limit cache or use a set of visited states
    visited = set()
    
    def search(t, current_ready):
        nonlocal found
        if found:
            return True
        if t > T:
            return len(current_ready) == 0

        # Pruning: all tasks must finish within remaining capacity
        rem_tasks = sum(1 for i in range(1, n + 1) if assigned_cycle[i] == 0)
        if rem_tasks > (T - t + 1) * M:
            return False

        # Any task with LS < t cannot be completed in time
        for u in current_ready:
            if LS[u] < t:
                return False

        state_key = (t, tuple(sorted(current_ready)))
        if state_key in visited:
            return False

        # Critical tasks that MUST be scheduled at cycle t (because LS[u] == t)
        must_schedule = [u for u in current_ready if LS[u] == t]
        if len(must_schedule) > M:
            visited.add(state_key)
            return False

        other_ready = [u for u in current_ready if LS[u] > t]
        other_ready.sort(key=priority)
        
        slots_left = M - len(must_schedule)
        
        # If other_ready has fewer or equal elements than slots_left, we can take all of them
        if len(other_ready) <= slots_left:
            choices = [must_schedule + other_ready]
        else:
            # We must take must_schedule, and choose `slots_left` from `other_ready`.
            # Usually taking the highest priority elements is optimal or near-optimal.
            # We can try greedy first, then other combinations if needed.
            # To avoid exponential explosion, take combinations of the top candidates.
            top_candidates = other_ready[:min(len(other_ready), slots_left + 4)]
            choices = []
            for comb in itertools.combinations(top_candidates, slots_left):
                choices.append(must_schedule + list(comb))
        
        for chosen in choices:
            # Apply chosen
            chosen_set = set(chosen)
            for u in chosen:
                assigned_cycle[u] = t
            
            new_ready = [u for u in current_ready if u not in chosen_set]
            
            # Unlock new operations
            unlocked = []
            for u in chosen:
                for v in adj[u]:
                    pred_count[v] -= 1
                    if pred_count[v] == 0:
                        unlocked.append(v)
                        new_ready.append(v)
            
            if search(t + 1, new_ready):
                found = True
                return True
                
            # Rollback
            for v in unlocked:
                pred_count[v] += 1
            for u in chosen:
                assigned_cycle[u] = 0

        visited.add(state_key)
        return False

    if search(1, ready):
        print("YES")
        print(*(assigned_cycle[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
