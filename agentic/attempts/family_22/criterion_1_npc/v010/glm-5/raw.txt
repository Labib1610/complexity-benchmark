import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    radj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1

    # earliest[v]: longest chain ending at v (1-indexed cycle)
    earliest = [1] * (n + 1)
    q = deque()
    for v in range(1, n + 1):
        if indeg[v] == 0:
            q.append(v)

    cnt = 0
    while q:
        u = q.popleft()
        cnt += 1
        for v in adj[u]:
            if earliest[u] + 1 > earliest[v]:
                earliest[v] = earliest[u] + 1
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if cnt != n:
        print("NO")
        return

    for v in range(1, n + 1):
        if earliest[v] > T:
            print("NO")
            return

    # latest[v]: latest cycle so that successors can still fit within T
    latest = [T] * (n + 1)
    q = deque()
    outdeg = [0] * (n + 1)
    for v in range(1, n + 1):
        outdeg[v] = len(adj[v])
        if outdeg[v] == 0:
            q.append(v)

    while q:
        v = q.popleft()
        for u in radj[v]:
            cand = latest[v] - 1
            if cand < latest[u]:
                latest[u] = cand
            outdeg[u] -= 1
            if outdeg[u] == 0:
                q.append(u)

    for v in range(1, n + 1):
        if earliest[v] > latest[v]:
            print("NO")
            return

    # Greedy: process cycles 1..T, at each cycle pick available nodes
    # with smallest latest deadline (EDF), up to M per cycle.
    avail = [True] * (n + 1)
    rem_indeg = indeg[:]  # indeg was zeroed, rebuild
    rem_indeg = [0] * (n + 1)
    for v in range(1, n + 1):
        rem_indeg[v] = len(radj[v])
        if rem_indeg[v] != 0:
            avail[v] = False

    # ready queue keyed by latest[v]; we'll just scan since n can be large
    # Use a list of ready nodes and sort by latest each cycle.
    ready = []
    for v in range(1, n + 1):
        if rem_indeg[v] == 0:
            ready.append(v)

    cycle_of = [0] * (n + 1)
    assigned = 0

    import heapq
    heap = []
    for v in ready:
        heapq.heappush(heap, (latest[v], v))

    for c in range(1, T + 1):
        # remove nodes whose latest < c (they can't be scheduled)
        # Actually we only push nodes with latest >= their earliest;
        # but if latest < c, impossible.
        batch = []
        while heap and heap[0][0] < c:
            lv, v = heapq.heappop(heap)
            # cannot schedule v in time
            print("NO")
            return

        take = min(M, len(heap))
        for _ in range(take):
            lv, v = heapq.heappop(heap)
            cycle_of[v] = c
            assigned += 1
            for w in adj[v]:
                rem_indeg[w] -= 1
                if rem_indeg[w] == 0:
                    heapq.heappush(heap, (latest[w], w))

    if assigned != n:
        print("NO")
        return

    out = ["YES"]
    out.append(" ".join(str(cycle_of[v]) for v in range(1, n + 1)))
    sys.stdout.write("\n".join(out) + "\n")

solve()
