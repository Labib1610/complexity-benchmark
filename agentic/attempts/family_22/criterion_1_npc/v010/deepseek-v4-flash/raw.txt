We can model this as a scheduling problem with precedence constraints and capacity limits.  
A simple approach is to use topological sorting with cycle assignment while respecting the per-cycle capacity.

```python
from collections import deque
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Kahn's algorithm with capacity constraint
    indeg_copy = indeg[:]
    q = deque([i for i in range(n) if indeg_copy[i] == 0])
    cycle = [0] * n
    current_cycle = 1
    assigned = 0
    
    while q:
        # We can take up to M nodes for this cycle
        batch = []
        for _ in range(min(M, len(q))):
            u = q.popleft()
            batch.append(u)
        if not batch:
            break
        if current_cycle > T:
            print("NO")
            return
        for u in batch:
            cycle[u] = current_cycle
            assigned += 1
            for v in adj[u]:
                indeg_copy[v] -= 1
                if indeg_copy[v] == 0:
                    q.append(v)
        current_cycle += 1
    
    if assigned != n:
        print("NO")
        return
    
    # Check if any cycle exceeds T
    if max(cycle) > T:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(cycle[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
