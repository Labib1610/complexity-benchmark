```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    M = int(data[idx+1])
    T = int(data[idx+2])
    m = int(data[idx+3])
    idx += 4
    
    if m == 0:
        # No dependencies, just fit them into rooms
        rooms = []
        for i in range(n):
            room_num = (i // M) + 1
            if room_num > T:
                print("NO")
                return
            rooms.append(str(room_num))
        print("YES")
        print(' '.join(rooms))
        return
    
    edges = []
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx+1])
        idx += 2
        edges.append((a, b))
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort to get a valid order
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    topo_order = []
    while queue:
        node = queue.popleft()
        topo_order.append(node)
        for neighbor in graph[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                queue.append(neighbor)
    
    if len(topo_order) != n:
        print("NO")
        return
    
    # Now assign rooms
    room = [0] * (n + 1)
    
    # We process nodes in topological order
    # For each node, we place it in the earliest room such that:
    # 1. It's after all its predecessors
    # 2. The room isn't full (has < M artifacts)
    
    # For each room, keep track of how many artifacts are placed
    room_count = [0] * (T + 1)
    # For each node, keep track of the latest room of its predecessors
    latest_pred_room = [0] * (n + 1)
    
    # Compute latest_pred_room
    for u in topo_order:
        for v in graph[u]:
            latest_pred_room[v] = max(latest_pred_room[v], room[u])
    
    # Try to assign rooms
    for u in topo_order:
        # Earliest possible room is latest_pred_room[u] + 1
        start_room = latest_pred_room[u] + 1
        placed = False
        for r in range(start_room, T + 1):
            if room_count[r] < M:
                room[u] = r
                room_count[r] += 1
                placed = True
                break
        if not placed:
            print("NO")
            return
            
        # Update successors
        for v in graph[u]:
            latest_pred_room[v] = max(latest_pred_room[v], room[u])
    
    print("YES")
    print(' '.join(str(room[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
```
