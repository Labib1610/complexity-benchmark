```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    radj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1

    INF = T + 1
    earliest = [1] * (n + 1)
    latest = [T] * (n + 1)

    q = deque(i for i in range(1, n + 1) if indeg[i] == 0)
    seen = 0
    while q:
        u = q.popleft()
        seen += 1
        for v in adj[u]:
            if earliest[u] + 1 > earliest[v]:
                earliest[v] = earliest[u] + 1
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if seen != n:
        print("NO")
        return

    for v in range(1, n + 1):
        if earliest[v] > T:
            print("NO")
            return

    outdeg = [0] * (n + 1)
    for u in range(1, n + 1):
        outdeg[u] = len(adj[u])

    q = deque(i for i in range(1, n + 1) if outdeg[i] == 0)
    while q:
        u = q.popleft()
        for v in radj[u]:
            if latest[u] - 1 < latest[v]:
                latest[v] = latest[u] - 1
            outdeg[v] -= 1
            if outdeg[v] == 0:
                q.append(v)

    for v in range(1, n + 1):
        if earliest[v] > latest[v]:
            print("NO")
            return

    cnt = [0] * (T + 2)
    for v in range(1, n + 1):
        cnt[earliest[v]] += 1

    for t in range(1, T + 1):
        if cnt[t] > M:
            print("NO")
            return

    room = [0] * (n + 1)
    for v in range(1, n + 1):
        room[v] = earliest[v]

    for t in range(1, T + 1):
        if cnt[t] < M:
            extra = M - cnt[t]
            for v in range(1, n + 1):
                if extra == 0:
                    break
                if room[v] == t and latest[v] > t:
                    room[v] = t + 1
                    extra -= 1
            cnt[t] = M
            cnt[t + 1] += M - extra

    print("YES")
    print(" ".join(map(str, room[1:])))

solve()
```
