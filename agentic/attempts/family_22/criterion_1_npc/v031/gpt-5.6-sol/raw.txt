```python
import sys
import heapq
import itertools

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

it = iter(data)
n = next(it)
M = next(it)
T = next(it)
m = next(it)

succ = [[] for _ in range(n)]
pred_mask = [0] * n
indeg = [0] * n

for _ in range(m):
    a = next(it) - 1
    b = next(it) - 1
    bit = 1 << a
    if pred_mask[b] & bit:
        continue
    pred_mask[b] |= bit
    succ[a].append(b)
    indeg[b] += 1

if M <= 0 or T <= 0 or n > M * T:
    print("NO")
    sys.exit()

# Topological order and cycle detection.
deg = indeg[:]
queue = [i for i in range(n) if deg[i] == 0]
topo = []
head = 0

while head < len(queue):
    u = queue[head]
    head += 1
    topo.append(u)
    for v in succ[u]:
        deg[v] -= 1
        if deg[v] == 0:
            queue.append(v)

if len(topo) != n:
    print("NO")
    sys.exit()

# Longest remaining chain starting at each vertex.
bottom = [1] * n
succ_mask = [0] * n
for u in reversed(topo):
    best = 0
    sm = 0
    for v in succ[u]:
        sm |= 1 << v
        if bottom[v] > best:
            best = bottom[v]
    succ_mask[u] = sm
    bottom[u] = best + 1

if max(bottom, default=0) > T:
    print("NO")
    sys.exit()

# Fast list scheduling. If it meets the limit, it is already a valid answer.
deg = indeg[:]
heap = []
for u in range(n):
    if deg[u] == 0:
        heapq.heappush(heap, (-bottom[u], -len(succ[u]), u))

greedy_room = [0] * n
room_no = 0
scheduled = 0

while heap:
    room_no += 1
    batch = []
    for _ in range(min(M, len(heap))):
        _, _, u = heapq.heappop(heap)
        batch.append(u)

    for u in batch:
        greedy_room[u] = room_no
        scheduled += 1

    for u in batch:
        for v in succ[u]:
            deg[v] -= 1
            if deg[v] == 0:
                heapq.heappush(heap, (-bottom[v], -len(succ[v]), v))

if scheduled == n and room_no <= T:
    print("YES")
    print(*greedy_room)
    sys.exit()

# Special easy cases.
if M == 1:
    if n <= T:
        ans = [0] * n
        for r, u in enumerate(topo, 1):
            ans[u] = r
        print("YES")
        print(*ans)
    else:
        print("NO")
    sys.exit()

all_mask = (1 << n) - 1
initial_available = 0
for u in range(n):
    if indeg[u] == 0:
        initial_available |= 1 << u

# Masks grouped by longest-chain length, used for necessary deadline checks.
level_masks_dict = {}
for u, b in enumerate(bottom):
    level_masks_dict[b] = level_masks_dict.get(b, 0) | (1 << u)
level_items = sorted(level_masks_dict.items(), reverse=True)

failed = set()
path = []

def iter_vertices(mask):
    while mask:
        bit = mask & -mask
        yield bit.bit_length() - 1
        mask ^= bit

def deadline_check(done, slots):
    remaining_mask = all_mask ^ done
    cumulative = 0
    for b, mask in level_items:
        count = (mask & remaining_mask).bit_count()
        if not count:
            continue
        if b > slots:
            return False
        cumulative += count
        available_early_rooms = slots - b + 1
        if cumulative > M * available_early_rooms:
            return False
    return True

def equivalent_selection_generator(groups, need):
    g = len(groups)
    suffix = [0] * (g + 1)
    for i in range(g - 1, -1, -1):
        suffix[i] = suffix[i + 1] + len(groups[i])

    selected = []

    def generate(i, left):
        if left == 0:
            yield tuple(selected)
            return
        if i == g or suffix[i] < left:
            return

        group = groups[i]
        low = max(0, left - suffix[i + 1])
        high = min(len(group), left)

        for take in range(high, low - 1, -1):
            if take:
                selected.extend(group[:take])
            yield from generate(i + 1, left - take)
            if take:
                del selected[-take:]

    yield from generate(0, need)

def dfs(done, available, slots):
    if done == all_mask:
        return True
    if slots == 0 or available == 0:
        return False

    remaining = n - done.bit_count()
    if remaining > slots * M:
        return False
    if not deadline_check(done, slots):
        return False

    key = (slots << n) | done
    if key in failed:
        return False

    available_count = available.bit_count()

    if available_count <= M:
        selections = [tuple(iter_vertices(available))]
    else:
        mandatory = []
        optional = []

        for u in iter_vertices(available):
            if bottom[u] == slots:
                mandatory.append(u)
            else:
                optional.append(u)

        if len(mandatory) > M:
            failed.add(key)
            return False

        mandatory.sort(key=lambda x: (-bottom[x], -len(succ[x]), x))
        optional.sort(key=lambda x: (-bottom[x], -len(succ[x]), x))
        need = M - len(mandatory)

        if need == 0:
            selections = [tuple(mandatory)]
        else:
            # Vertices with identical outgoing sets are interchangeable once
            # they are available, so only canonical counts from each class
            # need to be considered.
            grouped = {}
            order = []
            for u in optional:
                signature = succ_mask[u]
                if signature not in grouped:
                    grouped[signature] = []
                    order.append(signature)
                grouped[signature].append(u)

            groups = [grouped[sig] for sig in order]
            optional_selections = equivalent_selection_generator(groups, need)
            selections = (
                tuple(mandatory) + chosen
                for chosen in optional_selections
            )

    for chosen in selections:
        chosen_mask = 0
        for u in chosen:
            chosen_mask |= 1 << u

        new_done = done | chosen_mask
        new_available = available & ~chosen_mask

        for u in chosen:
            for v in succ[u]:
                vbit = 1 << v
                if not (new_done & vbit) and not (pred_mask[v] & ~new_done):
                    new_available |= vbit

        path.append(chosen)
        if dfs(new_done, new_available, slots - 1):
            return True
        path.pop()

    if len(failed) >= 500_000:
        failed.clear()
    failed.add(key)
    return False

if not dfs(0, initial_available, T):
    print("NO")
else:
    answer = [0] * n
    for r, chosen in enumerate(path, 1):
        for u in chosen:
            answer[u] = r
    print("YES")
    print(*answer)
```
