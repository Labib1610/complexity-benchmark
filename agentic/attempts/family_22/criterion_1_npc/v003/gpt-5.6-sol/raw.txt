import sys
import heapq
import itertools
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if len(data) < 4:
    sys.exit()

n, M, T, m = data[:4]
edges = data[4:]

if n == 0:
    print("YES")
    print()
    sys.exit()

if M <= 0 or T <= 0 or n > M * T:
    print("NO")
    sys.exit()

succ = [[] for _ in range(n)]
pred = [[] for _ in range(n)]
pred_mask = [0] * n
indeg = [0] * n

pos = 0
seen_edges = set()
for _ in range(m):
    a = edges[pos] - 1
    b = edges[pos + 1] - 1
    pos += 2
    if not (0 <= a < n and 0 <= b < n):
        print("NO")
        sys.exit()
    key = (a, b)
    if key in seen_edges:
        continue
    seen_edges.add(key)
    succ[a].append(b)
    pred[b].append(a)
    pred_mask[b] |= 1 << a
    indeg[b] += 1

# Topological order and cycle detection.
dq = deque(i for i in range(n) if indeg[i] == 0)
topo = []
indeg_work = indeg[:]

while dq:
    u = dq.popleft()
    topo.append(u)
    for v in succ[u]:
        indeg_work[v] -= 1
        if indeg_work[v] == 0:
            dq.append(v)

if len(topo) != n:
    print("NO")
    sys.exit()

# Earliest and latest levels under unlimited station capacity.
earliest = [1] * n
for u in topo:
    eu = earliest[u] + 1
    for v in succ[u]:
        if earliest[v] < eu:
            earliest[v] = eu

bottom = [1] * n
for u in reversed(topo):
    best = 1
    for v in succ[u]:
        cand = bottom[v] + 1
        if cand > best:
            best = cand
    bottom[u] = best

if max(earliest) > T:
    print("NO")
    sys.exit()

# Easy polynomial cases.
if M == 1 or T >= n:
    if n > T:
        print("NO")
    else:
        answer = [0] * n
        for slot, u in enumerate(topo, 1):
            answer[u] = slot
        print("YES")
        print(*answer)
    sys.exit()

if M >= n:
    print("YES")
    print(*earliest)
    sys.exit()

if not seen_edges:
    answer = [i // M + 1 for i in range(n)]
    if answer[-1] <= T:
        print("YES")
        print(*answer)
    else:
        print("NO")
    sys.exit()

latest = [T - bottom[i] + 1 for i in range(n)]
for i in range(n):
    if earliest[i] > latest[i]:
        print("NO")
        sys.exit()

all_mask = (1 << n) - 1

deadline_masks = [0] * (T + 1)
by_deadline = [[] for _ in range(T + 1)]
for v in range(n):
    by_deadline[latest[v]].append(v)

cur_mask = 0
for d in range(1, T + 1):
    for v in by_deadline[d]:
        cur_mask |= 1 << v
    deadline_masks[d] = cur_mask

unique_deadlines = sorted(set(latest))

initial_avail = 0
for v in range(n):
    if pred_mask[v] == 0:
        initial_avail |= 1 << v

answer = [0] * n
failed = set()
FAILED_LIMIT = 400_000


def advance(done, avail, chosen):
    done2 = done | chosen
    avail2 = avail & ~chosen

    x = chosen
    while x:
        bit = x & -x
        u = bit.bit_length() - 1
        x ^= bit
        for v in succ[u]:
            vb = 1 << v
            if not (done2 & vb) and not (avail2 & vb):
                if pred_mask[v] & ~done2 == 0:
                    avail2 |= vb

    return done2, avail2


def relaxed_window_feasible(done, current_time):
    """Necessary test: ignore precedence after deriving unlimited-machine releases."""
    remaining = all_mask ^ done
    release = [0] * n
    buckets = [[] for _ in range(T + 2)]
    remaining_count = remaining.bit_count()

    for v in topo:
        vb = 1 << v
        if not (remaining & vb):
            continue

        r = current_time
        for p in pred[v]:
            if remaining & (1 << p):
                rp = release[p] + 1
                if rp > r:
                    r = rp

        if r > latest[v] or r > T:
            return False

        release[v] = r
        buckets[r].append(latest[v])

    heap = []
    scheduled = 0

    for slot in range(current_time, T + 1):
        for deadline in buckets[slot]:
            heapq.heappush(heap, deadline)

        if heap and heap[0] < slot:
            return False

        take = min(M, len(heap))
        for _ in range(take):
            d = heapq.heappop(heap)
            if d < slot:
                return False
            scheduled += 1

    return scheduled == remaining_count and not heap


def masks_from_combinations(items, choose_count):
    for comb in itertools.combinations(items, choose_count):
        mask = 0
        for v in comb:
            mask |= 1 << v
        yield mask


def dfs(done, avail, current_time):
    changed = []

    # Consume slots whose contents are forced because at most M tasks are available.
    while True:
        if done == all_mask:
            return True

        if current_time > T:
            for v in changed:
                answer[v] = 0
            return False

        remaining = all_mask ^ done
        if remaining.bit_count() > M * (T - current_time + 1):
            for v in changed:
                answer[v] = 0
            return False

        if remaining & deadline_masks[current_time - 1]:
            for v in changed:
                answer[v] = 0
            return False

        available_count = avail.bit_count()
        if available_count == 0:
            for v in changed:
                answer[v] = 0
            return False

        if available_count > M:
            break

        chosen = avail
        x = chosen
        while x:
            bit = x & -x
            v = bit.bit_length() - 1
            x ^= bit
            answer[v] = current_time
            changed.append(v)

        done, avail = advance(done, avail, chosen)
        current_time += 1

    state = (done, current_time)
    if state in failed:
        for v in changed:
            answer[v] = 0
        return False

    if not relaxed_window_feasible(done, current_time):
        if len(failed) >= FAILED_LIMIT:
            failed.clear()
        failed.add(state)
        for v in changed:
            answer[v] = 0
        return False

    remaining = all_mask ^ done
    mandatory = avail & deadline_masks[current_time]
    mandatory_count = mandatory.bit_count()

    if mandatory_count > M:
        if len(failed) >= FAILED_LIMIT:
            failed.clear()
        failed.add(state)
        for v in changed:
            answer[v] = 0
        return False

    optional_mask = avail & ~mandatory
    optional = []
    x = optional_mask
    while x:
        bit = x & -x
        v = bit.bit_length() - 1
        x ^= bit
        optional.append(v)

    optional.sort(key=lambda v: (latest[v], -len(succ[v]), v))

    need_optional = M - mandatory_count

    def quota_ok(chosen):
        for d in unique_deadlines:
            if d < current_time:
                continue
            count_due = (remaining & deadline_masks[d]).bit_count()
            required_now = count_due - M * (d - current_time)
            if required_now > 0:
                if (chosen & deadline_masks[d]).bit_count() < required_now:
                    return False
        return True

    optional_count = len(optional)

    if need_optional <= optional_count - need_optional:
        choice_iter = (
            mandatory | extra
            for extra in masks_from_combinations(optional, need_optional)
        )
    else:
        exclude_count = optional_count - need_optional
        exclusion_order = list(reversed(optional))
        choice_iter = (
            mandatory | (optional_mask & ~excluded)
            for excluded in masks_from_combinations(exclusion_order, exclude_count)
        )

    for chosen in choice_iter:
        if not quota_ok(chosen):
            continue

        x = chosen
        chosen_vertices = []
        while x:
            bit = x & -x
            v = bit.bit_length() - 1
            x ^= bit
            answer[v] = current_time
            chosen_vertices.append(v)

        done2, avail2 = advance(done, avail, chosen)

        if dfs(done2, avail2, current_time + 1):
            return True

        for v in chosen_vertices:
            answer[v] = 0

    if len(failed) >= FAILED_LIMIT:
        failed.clear()
    failed.add(state)

    for v in changed:
        answer[v] = 0
    return False


if dfs(0, initial_avail, 1):
    print("YES")
    print(*answer)
else:
    print("NO")
