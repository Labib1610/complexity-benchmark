import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    __slots__ = ('n', 'g', 'level', 'it')
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]

    def add_edge(self, fr, to, cap):
        self.g[fr].append([to, cap, len(self.g[to])])
        self.g[to].append([fr, 0, len(self.g[fr]) - 1])

    def bfs(self, s, t):
        level = [-1] * self.n
        q = [s]
        level[s] = 0
        for v in q:
            for to, cap, rev in self.g[v]:
                if cap and level[to] < 0:
                    level[to] = level[v] + 1
                    q.append(to)
        self.level = level
        return level[t] >= 0

    def dfs(self, v, t, f):
        if v == t:
            return f
        for i in range(self.it[v], len(self.g[v])):
            self.it[v] = i
            to, cap, rev = self.g[v][i]
            if cap and self.level[v] + 1 == self.level[to]:
                ret = self.dfs(to, t, min(f, cap))
                if ret:
                    self.g[v][i][1] -= ret
                    self.g[to][rev][1] += ret
                    return ret
        return 0

    def max_flow(self, s, t):
        flow = 0
        INF = 10 ** 18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if not f:
                    break
                flow += f
        return flow

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); M = int(next(it)); T = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    radj = [[] for _ in range(n)]
    indeg = [0]*n
    for _ in range(m):
        a = int(next(it))-1; b = int(next(it))-1
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1

    # topological order
    from collections import deque
    q = deque([i for i in range(n) if indeg[i]==0])
    topo = []
    while q:
        v = q.popleft()
        topo.append(v)
        for to in adj[v]:
            indeg[to]-=1
            if indeg[to]==0:
                q.append(to)
    if len(topo)!=n:
        print("NO")
        return

    # compute earliest (L)
    L = [1]*n
    for v in topo:
        for to in adj[v]:
            if L[to] < L[v] + 1:
                L[to] = L[v] + 1

    # compute latest (U) using reverse topo
    revLen = [1]*n
    for v in reversed(topo):
        for pre in radj[v]:
            if revLen[pre] < revLen[v] + 1:
                revLen[pre] = revLen[v] + 1
    U = [T - revLen[i] + 1 for i in range(n)]

    for i in range(n):
        if L[i] > U[i] or L[i] > T or U[i] < 1:
            print("NO")
            return

    # flow construction
    # nodes: source, operation nodes (n), segment tree nodes, sink
    # build segment tree for interval [1,T]
    seg_offset = n
    # segment tree size
    size = 1
    while size < T:
        size <<= 1
    seg_n = size << 1
    total_nodes = 1 + n + seg_n + 1
    SRC = 0
    OP_START = 1
    SEG_START = OP_START + n
    SNK = total_nodes - 1

    dinic = Dinic(total_nodes)

    # source to operations
    for i in range(n):
        dinic.add_edge(SRC, OP_START + i, 1)

    # segment tree edges
    INF = 10**9
    # leaves correspond to slots 1..T at positions size..size+T-1
    leaf_id = [0]*(T+1)
    for pos in range(1, T+1):
        node = SEG_START + size -1 + (pos-1)
        leaf_id[pos] = node
        dinic.add_edge(node, SNK, M)

    # internal nodes to children
    for node in range(SEG_START, SEG_START + size):
        left = node*2
        right = node*2+1
        if left < SEG_START + seg_n:
            dinic.add_edge(node, left, INF)
            dinic.add_edge(node, right, INF)

    # function to add edges from operation to interval
    def add_interval(op_idx, l, r):
        l += size -1
        r += size -1
        while l <= r:
            if l%2==1:
                dinic.add_edge(OP_START+op_idx, SEG_START + (l - (size)), INF)
                l+=1
            if r%2==0:
                dinic.add_edge(OP_START+op_idx, SEG_START + (r - (size)), INF)
                r-=1
            l//=2; r//=2

    for i in range(n):
        add_interval(i, L[i], U[i])

    flow = dinic.max_flow(SRC, SNK)
    if flow != n:
        print("NO")
        return

    # retrieve assignments
    assign = [0]*n
    for i in range(n):
        v = OP_START + i
        for to, cap, rev in dinic.g[v]:
            # edges to segment tree nodes have original INF capacity, now if flow used, cap==0
            if to >= SEG_START and to < SEG_START + seg_n and cap == 0:
                # need to find leaf descendant
                node = to
                while node < SEG_START + size:
                    # go down to leaf with used flow
                    left = node*2
                    right = node*2+1
                    # check which child has flow sent (edge capacity reduced)
                    # look at edge from node to child
                    used_child = None
                    for to2, cap2, rev2 in dinic.g[node]:
                        if (to2 == left or to2 == right) and cap2 < INF:
                            used_child = to2
                            break
                    if used_child is None:
                        break
                    node = used_child
                # now node is leaf
                slot = node - (SEG_START + size) + 1
                assign[i] = slot
                break

    print("YES")
    print(' '.join(map(str, assign)))

if __name__ == "__main__":
    solve()
