We can model this as a scheduling problem on a DAG (directed acyclic graph) where each node is a reaction, and edges represent dependencies.  
We need to assign each reaction a cycle from 1 to T, respecting:

- If there is an edge a → b, then cycle(a) < cycle(b).
- At most M reactions per cycle.

We can solve this by:

1. Topologically sorting the reactions.
2. Assigning each reaction the earliest possible cycle, but also ensuring we don’t exceed M per cycle.

A greedy earliest-start assignment works if we process in topological order, because dependencies only constrain earlier nodes.

Here’s the implementation:

```python
from collections import deque
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        indeg[b] += 1
    
    # Topological sort
    q = deque([i for i in range(n) if indeg[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    # Assign cycles
    cycle = [0] * n
    count_in_cycle = [0] * (T + 1)
    
    for u in topo:
        # Earliest possible cycle = 1 + max(cycle of predecessors)
        earliest = 1
        # We need to find max cycle among predecessors
        # Since we process in topo order, all predecessors are already assigned
        # But we don't have direct predecessor list; we can compute by scanning edges
        # Instead, we can store preds or compute on the fly.
        # Let's build preds list.
        pass

    # Better: build predecessors
    preds = [[] for _ in range(n)]
    for a in range(n):
        for b in adj[a]:
            preds[b].append(a)
    
    for u in topo:
        earliest = 1
        for p in preds[u]:
            earliest = max(earliest, cycle[p] + 1)
        # Find first cycle >= earliest with count < M
        c = earliest
        while c <= T and count_in_cycle[c] >= M:
            c += 1
        if c > T:
            print("NO")
            return
        cycle[u] = c
        count_in_cycle[c] += 1
    
    print("YES")
    print(" ".join(str(cycle[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
