```python
import sys
import heapq
import itertools
import math

sys.setrecursionlimit(1_000_000)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, M, T, m = data[:4]
    if n == 0:
        print("YES")
        print()
        return
    if M <= 0 or T <= 0:
        print("NO")
        return

    succ_sets = [set() for _ in range(n)]
    pred_sets = [set() for _ in range(n)]
    pos = 4
    for _ in range(m):
        a = data[pos] - 1
        b = data[pos + 1] - 1
        pos += 2
        if b not in succ_sets[a]:
            succ_sets[a].add(b)
            pred_sets[b].add(a)

    succ = [list(x) for x in succ_sets]
    pred = [list(x) for x in pred_sets]

    indeg = [len(pred[v]) for v in range(n)]
    q = [v for v in range(n) if indeg[v] == 0]
    topo = []
    head = 0
    while head < len(q):
        v = q[head]
        head += 1
        topo.append(v)
        for w in succ[v]:
            indeg[w] -= 1
            if indeg[w] == 0:
                q.append(w)

    if len(topo) != n:
        print("NO")
        return

    top_level = [1] * n
    for v in topo:
        best = 1
        for p in pred[v]:
            value = top_level[p] + 1
            if value > best:
                best = value
        top_level[v] = best

    bottom_level = [1] * n
    for v in reversed(topo):
        best = 1
        for w in succ[v]:
            value = bottom_level[w] + 1
            if value > best:
                best = value
        bottom_level[v] = best

    longest = max(top_level)
    if n > M * T or longest > T:
        print("NO")
        return

    if T >= n:
        answer = [0] * n
        for cycle, v in enumerate(topo, 1):
            answer[v] = cycle
        print("YES")
        print(*answer)
        return

    if M >= n:
        print("YES")
        print(*top_level)
        return

    def forward_heuristic():
        remaining_pred = [len(pred[v]) for v in range(n)]
        heap = []
        for v in range(n):
            if remaining_pred[v] == 0:
                heapq.heappush(heap, (-bottom_level[v], v))

        answer = [0] * n
        done = 0
        cycle = 1
        while heap and cycle <= T:
            batch = []
            for _ in range(min(M, len(heap))):
                _, v = heapq.heappop(heap)
                batch.append(v)
                answer[v] = cycle
            done += len(batch)
            for v in batch:
                for w in succ[v]:
                    remaining_pred[w] -= 1
                    if remaining_pred[w] == 0:
                        heapq.heappush(heap, (-bottom_level[w], w))
            cycle += 1
        return answer if done == n else None

    answer = forward_heuristic()
    if answer is not None:
        print("YES")
        print(*answer)
        return

    def backward_heuristic():
        remaining_succ = [len(succ[v]) for v in range(n)]
        heap = []
        for v in range(n):
            if remaining_succ[v] == 0:
                heapq.heappush(heap, (-top_level[v], v))

        answer = [0] * n
        done = 0
        cycle = T
        while heap and cycle >= 1:
            batch = []
            for _ in range(min(M, len(heap))):
                _, v = heapq.heappop(heap)
                batch.append(v)
                answer[v] = cycle
            done += len(batch)
            for v in batch:
                for p in pred[v]:
                    remaining_succ[p] -= 1
                    if remaining_succ[p] == 0:
                        heapq.heappush(heap, (-top_level[p], p))
            cycle -= 1
        return answer if done == n else None

    answer = backward_heuristic()
    if answer is not None:
        print("YES")
        print(*answer)
        return

    full_mask = (1 << n) - 1
    answer = [0] * n
    failed = set()

    def bounded_comb(nv, kv, limit=10**18):
        if kv < 0 or kv > nv:
            return limit
        kv = min(kv, nv - kv)
        result = 1
        for i in range(1, kv + 1):
            result = result * (nv - kv + i) // i
            if result >= limit:
                return limit
        return result

    def dfs(mask, left, right):
        if mask == 0:
            return True

        R = right - left + 1
        count = mask.bit_count()

        if R <= 0 or count > M * R:
            return False

        if count <= R:
            cycle = left
            for v in topo:
                if (mask >> v) & 1:
                    answer[v] = cycle
                    cycle += 1
            return True

        key = (mask, R)
        if key in failed:
            return False

        top = [0] * n
        bottom = [0] * n
        sources = []

        for v in topo:
            if not ((mask >> v) & 1):
                continue
            best = 1
            for p in pred[v]:
                if (mask >> p) & 1:
                    value = top[p] + 1
                    if value > best:
                        best = value
            top[v] = best
            if best == 1:
                sources.append(v)

        sinks = []
        jobs = []
        for v in reversed(topo):
            if not ((mask >> v) & 1):
                continue
            best = 1
            for w in succ[v]:
                if (mask >> w) & 1:
                    value = bottom[w] + 1
                    if value > best:
                        best = value
            bottom[v] = best
            if best == 1:
                sinks.append(v)

        for v in topo:
            if not ((mask >> v) & 1):
                continue
            deadline = R - bottom[v] + 1
            if top[v] > deadline:
                failed.add(key)
                return False
            jobs.append((top[v], deadline))

        jobs.sort()
        heap = []
        idx = 0
        total_jobs = len(jobs)
        for tm in range(1, R + 1):
            while idx < total_jobs and jobs[idx][0] <= tm:
                heapq.heappush(heap, jobs[idx][1])
                idx += 1
            if heap and heap[0] < tm:
                failed.add(key)
                return False
            for _ in range(min(M, len(heap))):
                heapq.heappop(heap)
            if heap and heap[0] <= tm:
                failed.add(key)
                return False

        if heap or idx < total_jobs:
            failed.add(key)
            return False

        forced_source = [v for v in sources if bottom[v] == R]
        forced_sink = [v for v in sinks if top[v] == R]

        if len(forced_source) > M or len(forced_sink) > M:
            failed.add(key)
            return False

        def branch_count(items, forced):
            if len(items) <= M:
                return 1
            slots = M - len(forced)
            optional_count = len(items) - len(forced)
            return bounded_comb(optional_count, slots)

        source_branches = branch_count(sources, forced_source)
        sink_branches = branch_count(sinks, forced_sink)
        use_forward = source_branches <= sink_branches

        if use_forward:
            items = sources
            forced = forced_source
            forced_set = set(forced)
            optional = [v for v in items if v not in forced_set]
            optional.sort(key=lambda v: (-bottom[v], v))

            if len(items) <= M:
                batches = (tuple(items),)
            else:
                slots = M - len(forced)
                batches = (
                    tuple(forced) + choice
                    for choice in itertools.combinations(optional, slots)
                )

            for batch in batches:
                removed = 0
                for v in batch:
                    answer[v] = left
                    removed |= 1 << v
                if dfs(mask ^ removed, left + 1, right):
                    return True
                for v in batch:
                    answer[v] = 0
        else:
            items = sinks
            forced = forced_sink
            forced_set = set(forced)
            optional = [v for v in items if v not in forced_set]
            optional.sort(key=lambda v: (-top[v], v))

            if len(items) <= M:
                batches = (tuple(items),)
            else:
                slots = M - len(forced)
                batches = (
                    tuple(forced) + choice
                    for choice in itertools.combinations(optional, slots)
                )

            for batch in batches:
                removed = 0
                for v in batch:
                    answer[v] = right
                    removed |= 1 << v
                if dfs(mask ^ removed, left, right - 1):
                    return True
                for v in batch:
                    answer[v] = 0

        failed.add(key)
        return False

    if dfs(full_mask, 1, T):
        print("YES")
        print(*answer)
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
