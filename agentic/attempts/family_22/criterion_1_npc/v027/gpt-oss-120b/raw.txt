```python
import sys
import heapq

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n)]
    radj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a = int(next(it)) - 1
        b = int(next(it)) - 1
        adj[a].append(b)
        radj[b].append(a)
        indeg[b] += 1

    # topological sort
    from collections import deque
    q = deque([i for i in range(n) if indeg[i] == 0])
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    if len(topo) != n:
        print("NO")
        return

    # earliest days
    earliest = [1] * n
    for u in topo:
        eu = earliest[u]
        for v in adj[u]:
            if earliest[v] < eu + 1:
                earliest[v] = eu + 1

    # latest days
    latest = [T] * n
    for u in reversed(topo):
        lu = latest[u]
        for v in adj[u]:
            # u must be before v
            if latest[u] > latest[v] - 1:
                latest[u] = latest[v] - 1

    for i in range(n):
        if earliest[i] > latest[i] or earliest[i] > T or latest[i] < 1:
            print("NO")
            return

    tasks = [(earliest[i], latest[i], i) for i in range(n)]
    tasks.sort(key=lambda x: x[0])  # sort by release time

    heap = []  # (latest, id)
    ans = [0] * n
    ptr = 0
    day = 1
    assigned = 0

    while assigned < n:
        # add all tasks whose earliest <= current day
        while ptr < n and tasks[ptr][0] <= day:
            _, l, idx = tasks[ptr]
            heapq.heappush(heap, (l, idx))
            ptr += 1

        if not heap:
            # jump to next release day
            if ptr < n:
                day = tasks[ptr][0]
                continue
            else:
                break  # should not happen
        # schedule up to M tasks today
        for _ in range(M):
            if not heap:
                break
            l, idx = heap[0]
            if l < day:          # missed deadline
                print("NO")
                return
            heapq.heappop(heap)
            ans[idx] = day
            assigned += 1
        day += 1
        if day > T and assigned < n:
            # no more days left
            print("NO")
            return

    print("YES")
    print(' '.join(str(x) for x in ans))

if __name__ == "__main__":
    solve()
```
