from collections import defaultdict, deque
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    M = int(data[1])
    T = int(data[2])
    m = int(data[3])
    
    edges = []
    graph = defaultdict(list)
    indegree = [0] * (n + 1)
    
    idx = 4
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx+1])
        idx += 2
        edges.append((a, b))
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort to check if DAG
    queue = deque()
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
    
    topo_order = []
    while queue:
        node = queue.popleft()
        topo_order.append(node)
        for neighbor in graph[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                queue.append(neighbor)
    
    if len(topo_order) != n:
        print("NO")
        return
    
    # Assign days using a greedy approach
    day_assigned = [0] * (n + 1)
    day_count = [0] * (T + 1)
    
    # We process nodes in topological order
    for node in topo_order:
        # Find the earliest day where we can place this node
        # It must be after all its dependencies
        earliest_day = 1
        for prev in graph:
            if node in graph[prev]:
                if day_assigned[prev] >= earliest_day:
                    earliest_day = day_assigned[prev] + 1
        
        # Now find the first day from 'earliest_day' to T where we can fit it
        placed = False
        for d in range(earliest_day, T + 1):
            if day_count[d] < M:
                day_assigned[node] = d
                day_count[d] += 1
                placed = True
                break
        
        if not placed:
            print("NO")
            return
    
    # If we reach here, we have a valid assignment
    print("YES")
    print(' '.join(map(str, day_assigned[1:])))

if __name__ == "__main__":
    main()
