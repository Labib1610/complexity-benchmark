import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 4:
        return

    n, M, T, m = data[:4]

    if n == 0:
        print("YES")
        print()
        return

    if M <= 0 or T <= 0 or n > M * T:
        print("NO")
        return

    succ = [0] * n
    pred = [0] * n

    p = 4
    for _ in range(m):
        a = data[p] - 1
        b = data[p + 1] - 1
        p += 2
        if ((succ[a] >> b) & 1) == 0:
            succ[a] |= 1 << b
            pred[b] |= 1 << a

    indeg = [pred[i].bit_count() for i in range(n)]
    queue = deque(i for i in range(n) if indeg[i] == 0)
    topo = []

    while queue:
        v = queue.popleft()
        topo.append(v)
        bits = succ[v]
        while bits:
            bit = bits & -bits
            bits -= bit
            u = bit.bit_length() - 1
            indeg[u] -= 1
            if indeg[u] == 0:
                queue.append(u)

    if len(topo) != n:
        print("NO")
        return

    if T >= n:
        answer = [0] * n
        for day, v in enumerate(topo, 1):
            answer[v] = day
        print("YES")
        print(*answer)
        return

    suffix_len = [1] * n
    descendants = [0] * n

    for v in reversed(topo):
        bits = succ[v]
        best = 0
        closure = 0
        while bits:
            bit = bits & -bits
            bits -= bit
            u = bit.bit_length() - 1
            if suffix_len[u] > best:
                best = suffix_len[u]
            closure |= bit | descendants[u]
        suffix_len[v] = best + 1
        descendants[v] = closure

    max_height = max(suffix_len)
    if max_height > T:
        print("NO")
        return

    all_mask = (1 << n) - 1
    masks_ge = [0] * (max_height + 2)
    by_len = [0] * (max_height + 1)

    for v in range(n):
        by_len[suffix_len[v]] |= 1 << v

    running = 0
    for length in range(max_height, 0, -1):
        running |= by_len[length]
        masks_ge[length] = running

    rank = [
        (-suffix_len[v], -descendants[v].bit_count(), v)
        for v in range(n)
    ]

    failed = set()
    chosen_days = []

    def search(day, done):
        if done == all_mask:
            return True
        if day > T:
            return False

        state = (day, done)
        if state in failed:
            return False

        remaining_mask = all_mask ^ done
        remaining_count = remaining_mask.bit_count()
        days_left = T - day + 1

        if remaining_count > days_left * M:
            failed.add(state)
            return False

        quotas = []
        upper = min(max_height, days_left)
        for length in range(1, upper + 1):
            count = (remaining_mask & masks_ge[length]).bit_count()
            slots = days_left - length + 1
            if count > slots * M:
                failed.add(state)
                return False
            required_today = count - (slots - 1) * M
            if required_today > 0:
                quotas.append((masks_ge[length], required_today))

        if max_height > days_left:
            if remaining_mask & masks_ge[days_left + 1]:
                failed.add(state)
                return False

        available = 0
        bits = remaining_mask
        while bits:
            bit = bits & -bits
            bits -= bit
            v = bit.bit_length() - 1
            if pred[v] & remaining_mask == 0:
                available |= bit

        if available == 0:
            failed.add(state)
            return False

        available_count = available.bit_count()
        take = min(M, available_count)

        for mask, required in quotas:
            if (available & mask).bit_count() < required:
                failed.add(state)
                return False

        if available_count <= M:
            selected = available
            for mask, required in quotas:
                if (selected & mask).bit_count() < required:
                    failed.add(state)
                    return False
            chosen_days.append(selected)
            if search(day + 1, done | selected):
                return True
            chosen_days.pop()
            failed.add(state)
            return False

        mandatory = available & masks_ge[days_left]
        mandatory_count = mandatory.bit_count()
        if mandatory_count > take:
            failed.add(state)
            return False

        need = take - mandatory_count
        optional = available ^ mandatory

        optional_vertices = []
        bits = optional
        while bits:
            bit = bits & -bits
            bits -= bit
            v = bit.bit_length() - 1
            optional_vertices.append(v)
        optional_vertices.sort(key=lambda v: rank[v])

        groups = []
        group_index = {}
        for v in optional_vertices:
            key = succ[v]
            idx = group_index.get(key)
            if idx is None:
                group_index[key] = len(groups)
                groups.append([v])
            else:
                groups[idx].append(v)

        group_masks = []
        prefix_masks = []
        for group in groups:
            gm = 0
            prefixes = [0]
            for v in group:
                gm |= 1 << v
                prefixes.append(gm)
            group_masks.append(gm)
            prefix_masks.append(prefixes)

        group_count = len(groups)
        suffix_sizes = [0] * (group_count + 1)
        for i in range(group_count - 1, -1, -1):
            suffix_sizes[i] = suffix_sizes[i + 1] + len(groups[i])

        def generate(i, left, selected_optional):
            if left < 0 or left > suffix_sizes[i]:
                return False
            if i == group_count:
                if left != 0:
                    return False
                selected = mandatory | selected_optional
                for qmask, required in quotas:
                    if (selected & qmask).bit_count() < required:
                        return False
                chosen_days.append(selected)
                if search(day + 1, done | selected):
                    return True
                chosen_days.pop()
                return False

            size = len(groups[i])
            high = min(size, left)
            low = max(0, left - suffix_sizes[i + 1])

            for count in range(high, low - 1, -1):
                new_selected = selected_optional | prefix_masks[i][count]

                possible = mandatory | new_selected
                future_groups_mask = 0
                for j in range(i + 1, group_count):
                    future_groups_mask |= group_masks[j]

                impossible = False
                for qmask, required in quotas:
                    have = (possible & qmask).bit_count()
                    can_add = min(
                        left - count,
                        (future_groups_mask & qmask).bit_count()
                    )
                    if have + can_add < required:
                        impossible = True
                        break

                if not impossible and generate(
                    i + 1, left - count, new_selected
                ):
                    return True
            return False

        if generate(0, need, 0):
            return True

        failed.add(state)
        return False

    if not search(1, 0):
        print("NO")
        return

    answer = [0] * n
    for day, mask in enumerate(chosen_days, 1):
        bits = mask
        while bits:
            bit = bits & -bits
            bits -= bit
            answer[bit.bit_length() - 1] = day

    print("YES")
    print(*answer)

if __name__ == "__main__":
    main()
