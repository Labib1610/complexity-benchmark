We can model this as a scheduling problem with precedence constraints and capacity limits.  
We need to assign each step a cycle from 1 to T, respecting:

- If step a must come before step b, then cycle[a] < cycle[b].
- At most M steps per cycle.

We can solve it using topological sorting with cycle assignment, or by checking feasibility with a greedy earliest-time assignment.

Here’s the Python solution:

```python
from collections import deque, defaultdict
import sys

def solve():
    input = sys.stdin.readline
    n, M, T, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    indeg = [0] * n
    for _ in range(m):
        a, b = map(int, input().split())
        a -= 1
        b -= 1
        adj[a].append(b)
        indeg[b] += 1

    # Kahn's algorithm with cycle assignment
    indeg_copy = indeg[:]
    q = deque()
    for i in range(n):
        if indeg_copy[i] == 0:
            q.append(i)

    cycle = [0] * n
    # We'll assign cycles greedily: earliest possible cycle respecting dependencies
    # and capacity
    # We'll process in topological order, but need to track capacity per cycle
    # Use a min-heap to pick the earliest cycle with free capacity for each node
    import heapq
    # For each node, we need to know the max cycle of its predecessors
    pred_max = [0] * n
    # We'll process nodes in topological order
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            indeg_copy[v] -= 1
            if indeg_copy[v] == 0:
                q.append(v)

    if len(topo) != n:
        print("NO")
        return

    # Now assign cycles
    # We'll use a priority queue of (cycle, node) but we need to respect capacity
    # Better: process in topological order, assign earliest cycle >= pred_max+1
    # with capacity < M
    capacity = [0] * (T + 1)  # 1-indexed cycles
    for u in topo:
        # earliest cycle = max(pred_max[u] + 1, 1)
        c = max(pred_max[u] + 1, 1)
        # find first cycle >= c with capacity < M
        while c <= T and capacity[c] >= M:
            c += 1
        if c > T:
            print("NO")
            return
        cycle[u] = c
        capacity[c] += 1
        # update pred_max for successors
        for v in adj[u]:
            if cycle[u] > pred_max[v]:
                pred_max[v] = cycle[u]

    print("YES")
    print(" ".join(str(cycle[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
