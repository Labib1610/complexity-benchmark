from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    M = int(data[1])
    T = int(data[2])
    m = int(data[3])
    
    edges = []
    indegree = [0] * (n + 1)
    graph = defaultdict(list)
    
    idx = 4
    for _ in range(m):
        a = int(data[idx])
        b = int(data[idx+1])
        idx += 2
        edges.append((a, b))
        graph[a].append(b)
        indegree[b] += 1
    
    # Topological sort with level assignment
    queue = deque()
    level = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if indegree[i] == 0:
            queue.append(i)
            level[i] = 1
    
    max_level = 0
    while queue:
        u = queue.popleft()
        current_level = level[u]
        max_level = max(max_level, current_level)
        
        for v in graph[u]:
            indegree[v] -= 1
            if level[v] < current_level + 1:
                level[v] = current_level + 1
            if indegree[v] == 0:
                queue.append(v)
    
    # Check if any node still has indegree > 0 (cycle exists)
    if any(indegree[i] > 0 for i in range(1, n + 1)):
        print("NO")
        return
    
    # Group nodes by levels
    levels = [[] for _ in range(max_level + 1)]
    for i in range(1, n + 1):
        levels[level[i]].append(i)
    
    # Assign cycles ensuring at most M steps per cycle
    cycle_assignment = [0] * (n + 1)
    current_cycle = 1
    count_in_current_cycle = 0
    
    # Process levels in order
    for lvl in range(1, max_level + 1):
        nodes_in_level = levels[lvl]
        for node in nodes_in_level:
            if count_in_current_cycle == M:
                current_cycle += 1
                count_in_current_cycle = 0
                if current_cycle > T:
                    print("NO")
                    return
            cycle_assignment[node] = current_cycle
            count_in_current_cycle += 1
    
    # Final check: ensure all cycles are within [1, T]
    for i in range(1, n + 1):
        if cycle_assignment[i] < 1 or cycle_assignment[i] > T:
            print("NO")
            return
            
    print("YES")
    print(' '.join(map(str, cycle_assignment[1:])))

if __name__ == "__main__":
    main()
