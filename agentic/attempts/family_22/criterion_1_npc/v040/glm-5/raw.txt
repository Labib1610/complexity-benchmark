import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    M = int(next(it))
    T = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1
    
    # earliest[i] = earliest cycle step i can be scheduled (1-indexed)
    earliest = [1] * (n + 1)
    
    # Topological order using Kahn's algorithm
    q = deque()
    for i in range(1, n + 1):
        if indeg[i] == 0:
            q.append(i)
    
    topo = []
    while q:
        u = q.popleft()
        topo.append(u)
        for v in adj[u]:
            if earliest[v] < earliest[u] + 1:
                earliest[v] = earliest[u] + 1
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(topo) != n:
        print("NO")
        return
    
    # Check if any step requires cycle > T
    for i in range(1, n + 1):
        if earliest[i] > T:
            print("NO")
            return
    
    # Group steps by their earliest cycle
    groups = [[] for _ in range(T + 1)]
    for i in range(1, n + 1):
        groups[earliest[i]].append(i)
    
    # Greedy scheduling: schedule each group as early as possible
    cycle_count = [0] * (T + 2)  # 1-indexed, extra slot for safety
    result = [0] * (n + 1)
    
    for c in range(1, T + 1):
        # Process groups from earliest to current
        for g in range(1, c + 1):
            remaining = []
            for step in groups[g]:
                if result[step] == 0:
                    remaining.append(step)
            groups[g] = remaining
            
            available = M - cycle_count[c]
            if available <= 0:
                break
            
            to_schedule = groups[g][:available]
            groups[g] = groups[g][available:]
            
            for step in to_schedule:
                result[step] = c
                cycle_count[c] += 1
    
    # Check if all steps are scheduled
    for i in range(1, n + 1):
        if result[i] == 0:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, result[1:])))

if __name__ == "__main__":
    solve()
