We can solve this using **topological sorting** (Kahn’s algorithm) on a directed graph where each edge `a -> b` means `a` must come before `b`.  
If we can process all nodes, we output `YES` and the order; otherwise, we output `NO`.

```python
from collections import deque
import sys

def main():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1

    q = deque([i for i in range(1, n + 1) if indeg[i] == 0])
    order = []

    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if len(order) == n:
        print("YES")
        print(" ".join(map(str, order)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
