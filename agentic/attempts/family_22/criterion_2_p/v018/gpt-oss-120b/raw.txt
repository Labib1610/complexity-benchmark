```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    indeg = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        a = int(next(it)); b = int(next(it))
        adj[a].append(b)
        indeg[b] += 1

    q = deque(i for i in range(1, n + 1) if indeg[i] == 0)
    order = []
    while q:
        v = q.popleft()
        order.append(v)
        for to in adj[v]:
            indeg[to] -= 1
            if indeg[to] == 0:
                q.append(to)

    out = []
    if len(order) == n:
        out.append("YES")
        out.append(" ".join(map(str, order)))
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
