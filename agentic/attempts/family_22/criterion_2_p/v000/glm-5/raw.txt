```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    g = [[] for _ in range(n+1)]
    indeg = [0]*(n+1)
    idx = 2
    for _ in range(m):
        a = int(data[idx]); b = int(data[idx+1]); idx += 2
        g[a].append(b)
        indeg[b] += 1
    from collections import deque
    q = deque()
    for i in range(1, n+1):
        if indeg[i] == 0:
            q.append(i)
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in g[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    if len(order) != n:
        sys.stdout.write("NO")
    else:
        sys.stdout.write("YES\n" + " ".join(map(str, order)))

solve()
```
