We can solve this using **topological sorting** (Kahn’s algorithm) on a directed graph where each rule `x y` means an edge from `x` to `y`. If the graph has a cycle, it’s impossible to order them.

```python
import sys
from collections import deque

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        x = int(data[idx])
        y = int(data[idx + 1])
        idx += 2
        adj[x].append(y)
        indeg[y] += 1
    
    q = deque()
    for i in range(1, n + 1):
        if indeg[i] == 0:
            q.append(i)
    
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    
    if len(order) != n:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, order)))

if __name__ == "__main__":
    main()
```
