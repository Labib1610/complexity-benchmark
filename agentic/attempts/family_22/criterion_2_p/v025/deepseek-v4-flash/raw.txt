We can solve this using **topological sorting** (Kahn’s algorithm) on a directed graph.  
If the graph has a cycle, it’s impossible → print `NO`.  
Otherwise, print `YES` followed by any valid topological order.

```python
import sys
from collections import deque

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    indeg = [0] * (n + 1)

    for _ in range(m):
        a, b = map(int, input().split())
        adj[a].append(b)
        indeg[b] += 1

    q = deque([i for i in range(1, n + 1) if indeg[i] == 0])
    order = []

    while q:
        u = q.popleft()
        order.append(u)
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)

    if len(order) != n:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, order)))

if __name__ == "__main__":
    main()
```
