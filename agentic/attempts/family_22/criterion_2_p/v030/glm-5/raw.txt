```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return
    
    adj = [[] for _ in range(n + 1)]
    in_degree = [0] * (n + 1)
    
    for _ in range(m):
        try:
            a = int(next(iterator))
            b = int(next(iterator))
            adj[a].append(b)
            in_degree[b] += 1
        except StopIteration:
            break
            
    dq = deque()
    for i in range(1, n + 1):
        if in_degree[i] == 0:
            dq.append(i)
            
    result = []
    
    while dq:
        u = dq.popleft()
        result.append(u)
        
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                dq.append(v)
                
    if len(result) == n:
        print("YES")
        print(*(result))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
