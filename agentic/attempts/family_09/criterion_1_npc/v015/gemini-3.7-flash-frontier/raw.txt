import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k > n:
        print("NO")
        return

    # 0-indexed vertices
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Complement graph adjacency (including self-loop exclusion)
    # non_adj[u] is the bitmask of vertices NOT adjacent to u in G (and != u)
    all_mask = (1 << n) - 1
    non_adj = [all_mask ^ adj[u] ^ (1 << u) for u in range(n)]

    def bron_kerbosch(R_mask, P_mask, X_mask, R_count):
        if P_mask == 0 and X_mask == 0:
            if R_count >= k:
                # Extract k vertices from R_mask
                res = []
                for i in range(n):
                    if (R_mask >> i) & 1:
                        res.append(i + 1)
                        if len(res) == k:
                            break
                return res
            return None

        if R_count + P_mask.bit_count() < k:
            return None

        # Choose pivot u in P | X that maximizes |P & non_adj[u]|
        PX = P_mask | X_mask
        best_u = -1
        max_size = -1
        temp = PX
        while temp:
            lsb = temp & -temp
            u = lsb.bit_length() - 1
            size = (P_mask & non_adj[u]).bit_count()
            if size > max_size:
                max_size = size
                best_u = u
            temp ^= lsb

        # Branch on vertices in P \ non_adj[best_u]
        branch_mask = P_mask & (~non_adj[best_u])
        while branch_mask:
            lsb = branch_mask & -branch_mask
            v = lsb.bit_length() - 1

            v_bit = 1 << v
            res = bron_kerbosch(
                R_mask | v_bit,
                P_mask & non_adj[v],
                X_mask & non_adj[v],
                R_count + 1,
            )
            if res is not None:
                return res

            P_mask ^= v_bit
            X_mask |= v_bit
            branch_mask ^= lsb

            if R_count + P_mask.bit_count() < k:
                break

        return None

    res = bron_kerbosch(0, all_mask, 0, 0)
    if res is not None:
        print("YES")
        print(*(res[:k]))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
