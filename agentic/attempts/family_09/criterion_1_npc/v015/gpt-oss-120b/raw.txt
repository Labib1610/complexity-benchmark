```python
import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_vertices = (1 << n) - 1
    solution = []

    # order vertices by degree descending for better branching
    order = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency according to new order
    new_adj = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for u in range(n):
            if adj[v] >> u & 1:
                mask |= 1 << pos[u]
        new_adj[i] = mask

    # recursive search
    found = False

    def dfs(cur_mask: int, cand_mask: int, cur_list):
        nonlocal found, solution
        if found:
            return
        cur_size = cur_mask.bit_count()
        if cur_size == k:
            solution = cur_list[:]
            found = True
            return
        # pruning: not enough candidates left
        if cur_size + cand_mask.bit_count() < k:
            return
        # choose a vertex: pick the one with highest degree in candidates
        # simple: take first set bit
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        new_cand = cand_mask & ~new_adj[v] & ~(1 << v)
        dfs(cur_mask | (1 << v), new_cand, cur_list + [order[v] + 1])
        if found:
            return
        # branch: exclude v
        dfs(cur_mask, cand_mask & ~(1 << v), cur_list)

    dfs(0, all_vertices, [])

    if found:
        print("YES")
        print(" ".join(map(str, solution)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
