import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    adj = [0] * n
    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    if m == 0:
        print("YES")
        print(*range(1, k + 1))
        return

    degrees = [adj[i].bit_count() for i in range(n)]

    # Quick lower bound: greedily construct an independent set.
    for order in (
        sorted(range(n), key=lambda x: (degrees[x], x)),
        sorted(range(n), key=lambda x: (-degrees[x], x)),
        list(range(n))
    ):
        chosen = []
        chosen_mask = 0
        for v in order:
            if adj[v] & chosen_mask == 0:
                chosen.append(v)
                chosen_mask |= 1 << v
                if len(chosen) == k:
                    print("YES")
                    print(*(x + 1 for x in chosen))
                    return

    # Quick upper bound by coloring the compatibility graph.
    # Each color class is a clique in the conflict graph.
    for order in (
        sorted(range(n), key=lambda x: (-degrees[x], x)),
        sorted(range(n), key=lambda x: (degrees[x], x))
    ):
        classes = []
        for v in order:
            for i, mask in enumerate(classes):
                if (adj[v] & mask) == mask:
                    classes[i] = mask | (1 << v)
                    break
            else:
                classes.append(1 << v)
        if len(classes) < k:
            print("NO")
            return

    # Put high-degree vertices in the enumerated half to reduce the number
    # of independent subsets generated there.
    order = sorted(range(n), key=lambda x: (-degrees[x], x))
    b_size = n // 2
    a_size = n - b_size
    a_vertices = order[:a_size]
    b_vertices = order[a_size:]

    full_b = (1 << b_size) - 1

    adj_b = [0] * b_size
    for i, u in enumerate(b_vertices):
        mask = 0
        au = adj[u]
        for j, v in enumerate(b_vertices):
            if (au >> v) & 1:
                mask |= 1 << j
        adj_b[i] = mask

    free_b = [full_b ^ x for x in adj_b]

    # dp[mask] = maximum independent-set size in B induced by mask.
    size_b = 1 << b_size
    dp = bytearray(size_b)
    for mask in range(1, size_b):
        bit = mask & -mask
        i = bit.bit_length() - 1
        rest = mask ^ bit
        without = dp[rest]
        with_i = 1 + dp[rest & free_b[i]]
        dp[mask] = with_i if with_i > without else without

    def reconstruct_b(mask, need):
        result = []
        while need:
            bit = mask & -mask
            i = bit.bit_length() - 1
            rest = mask ^ bit
            if dp[rest] == dp[mask]:
                mask = rest
            else:
                result.append(b_vertices[i])
                need -= 1
                mask = rest & free_b[i]
        return result

    if dp[full_b] >= k:
        answer = reconstruct_b(full_b, k)
        print("YES")
        print(*(x + 1 for x in answer))
        return

    adj_a = [0] * a_size
    compatible_b = [full_b] * a_size

    for i, u in enumerate(a_vertices):
        au = adj[u]

        mask_a = 0
        for j, v in enumerate(a_vertices):
            if (au >> v) & 1:
                mask_a |= 1 << j
        adj_a[i] = mask_a

        blocked_b = 0
        for j, v in enumerate(b_vertices):
            if (au >> v) & 1:
                blocked_b |= 1 << j
        compatible_b[i] = full_b ^ blocked_b

    result_a = 0
    result_allowed_b = 0
    result_count_a = 0

    sys.setrecursionlimit(1000)

    def search(candidates, selected, allowed_b, count_a):
        nonlocal result_a, result_allowed_b, result_count_a

        need = k - count_a
        best_b = dp[allowed_b]
        if best_b >= need:
            result_a = selected
            result_allowed_b = allowed_b
            result_count_a = count_a
            return True

        if count_a + candidates.bit_count() + best_b < k:
            return False

        while candidates:
            if count_a + candidates.bit_count() + best_b < k:
                return False

            bit = candidates & -candidates
            candidates ^= bit
            i = bit.bit_length() - 1

            next_candidates = candidates & ~adj_a[i]
            next_allowed = allowed_b & compatible_b[i]

            if search(
                next_candidates,
                selected | bit,
                next_allowed,
                count_a + 1
            ):
                return True

        return False

    if not search((1 << a_size) - 1, 0, full_b, 0):
        print("NO")
        return

    answer = []
    mask = result_a
    while mask:
        bit = mask & -mask
        i = bit.bit_length() - 1
        answer.append(a_vertices[i])
        mask ^= bit

    answer.extend(reconstruct_b(result_allowed_b, k - result_count_a))

    print("YES")
    print(*(x + 1 for x in answer))

if __name__ == "__main__":
    main()
