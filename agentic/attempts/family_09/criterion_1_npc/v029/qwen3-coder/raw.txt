from itertools import combinations
import sys

def main():
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Read interference links
    links = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        # Normalize the link by sorting
        links.add(tuple(sorted((u, v))))
    
    # Generate all possible combinations of k sensors
    sensors = list(range(1, n + 1))
    
    # For large n, we need a more efficient approach
    # But given n <= 46, and we're looking for k sensors,
    # if k is small, we can check all combinations
    # However, for very large k, this might be too slow
    
    # Let's use a meet-in-the-middle approach for better efficiency
    
    if k <= n // 2:
        # Check all combinations directly for small k
        for combo in combinations(sensors, k):
            valid = True
            # Check all pairs in this combination
            for i in range(len(combo)):
                for j in range(i + 1, len(combo)):
                    pair = tuple(sorted((combo[i], combo[j])))
                    if pair in links:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                print("YES")
                print(' '.join(map(str, combo)))
                return
    else:
        # For large k, it might be better to find a maximal independent set
        # But for simplicity and correctness, let's try a different approach
        # Since n <= 46, and we might have to check C(46,23) which is large,
        # we need a smarter approach
        
        # Let's try a greedy approach or branch and bound, but for simplicity
        # let's implement a more efficient backtracking with pruning
        
        # Build adjacency list
        adj = [[] for _ in range(n + 1)]
        for u, v in links:
            adj[u].append(v)
            adj[v].append(u)
        
        def is_valid(current_set, new_node):
            for node in current_set:
                if tuple(sorted((node, new_node))) in links:
                    return False
            return True
        
        def backtrack(nodes, current_set, target_size):
            if len(current_set) == target_size:
                return current_set[:]
            
            if not nodes:
                return None
                
            # Try including nodes in sorted order for consistency
            for i, node in enumerate(nodes):
                if is_valid(current_set, node):
                    current_set.append(node)
                    result = backtrack(nodes[i+1:], current_set, target_size)
                    if result is not None:
                        return result
                    current_set.pop()
            
            return None
        
        result = backtrack(sensors, [], k)
        if result:
            print("YES")
            print(' '.join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
