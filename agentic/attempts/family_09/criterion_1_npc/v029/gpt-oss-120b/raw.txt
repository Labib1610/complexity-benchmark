import sys

sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1
    result = []

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i
    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for j in range(n):
            if adj[v] >> j & 1:
                mask |= 1 << pos[j]
        adj2[i] = mask

    target = k
    found = False
    ans_mask = 0

    from functools import lru_cache

    @lru_cache(maxsize=None)
    def popcnt(x: int) -> int:
        return x.bit_count()

    def dfs(cur_mask: int, cand_mask: int, size: int):
        nonlocal found, ans_mask
        if found:
            return
        if size == target:
            found = True
            ans_mask = cur_mask
            return
        # prune if not enough candidates left
        if size + popcnt(cand_mask) < target:
            return
        # choose a vertex with maximum degree among candidates (heuristic)
        # simple: pick first bit
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        dfs(cur_mask | (1 << v), new_cand, size + 1)
        if found:
            return
        # branch: exclude v
        dfs(cur_mask, cand_mask & ~(1 << v), size)

    dfs(0, (1 << n) - 1, 0)

    if not found:
        print("NO")
        return

    # translate mask back to original vertex numbers
    res = []
    for i in range(n):
        if ans_mask >> i & 1:
            res.append(order[i] + 1)
    print("YES")
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
