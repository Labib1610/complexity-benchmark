We can solve this by checking for an independent set of size exactly k in a graph with up to 46 vertices.  
A simple backtracking with pruning will work because n is small.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find independent set of size k
    # Use bitmask for chosen set
    chosen = []
    # We'll do backtracking with pruning
    # We can also use maximum independent set approach but n=46 is small enough for branch and bound

    # Precompute possible candidates
    # We'll use recursion with ordering by degree

    # Simple backtracking
    def backtrack(start, chosen_set, count):
        if count == k:
            return chosen_set
        # If not enough remaining vertices
        if n - start < k - count:
            return None
        for v in range(start, n):
            # Check if v can be added
            if (chosen_set & adj[v]) == 0:
                res = backtrack(v + 1, chosen_set | (1 << v), count + 1)
                if res is not None:
                    return res
        return None

    result = backtrack(0, 0, 0)
    if result is None:
        print("NO")
    else:
        print("YES")
        ans = [str(i + 1) for i in range(n) if (result >> i) & 1]
        print(" ".join(ans))

if __name__ == "__main__":
    solve()
```
