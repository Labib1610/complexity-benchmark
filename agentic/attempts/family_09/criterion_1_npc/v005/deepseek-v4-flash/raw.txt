We can solve this by trying to find an independent set of size exactly `k` in the graph.  
Since `n ≤ 46`, we can use a backtracking approach with pruning, or a meet-in-the-middle technique.

Here’s a possible implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find independent set of size k
    # Use recursive backtracking with pruning
    # We'll store result in a list
    result = []

    def backtrack(start, chosen, mask):
        nonlocal result
        if len(chosen) == k:
            result = chosen[:]
            return True
        if start >= n:
            return False
        # Prune: not enough remaining vertices
        if n - start < k - len(chosen):
            return False
        # Try including start if possible
        if (mask & adj[start]) == 0:
            if backtrack(start + 1, chosen + [start], mask | (1 << start)):
                return True
        # Try excluding start
        return backtrack(start + 1, chosen, mask)

    if backtrack(0, [], 0):
        print("YES")
        print(" ".join(str(x + 1) for x in result))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
