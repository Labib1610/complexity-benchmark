import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [0] * n
    cross = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    n1 = n // 2
    n2 = n - n1
    # split vertices: first half [0..n1-1], second half [n1..n-1]
    # build cross adjacency from first to second
    cross_first = [0] * n1
    for i in range(n1):
        mask = 0
        for j in range(n2):
            if adj[i] >> (n1 + j) & 1:
                mask |= 1 << j
        cross_first[i] = mask

    # generate independent subsets for first half
    first_subsets = []
    def dfs_first(idx, cur_mask, cur_adj):
        if idx == n1:
            first_subsets.append((cur_mask, cur_adj, bin(cur_mask).count('1')))
            return
        # skip idx
        dfs_first(idx + 1, cur_mask, cur_adj)
        # take idx if possible
        if not (cur_adj >> idx) & 1:
            dfs_first(idx + 1,
                      cur_mask | (1 << idx),
                      cur_adj | adj[idx] & ((1 << n1) - 1))
    dfs_first(0, 0, 0)

    # generate independent subsets for second half, store by size
    second_by_size = [[] for _ in range(n2 + 1)]
    def dfs_second(idx, cur_mask, cur_adj):
        if idx == n2:
            sz = bin(cur_mask).count('1')
            second_by_size[sz].append(cur_mask)
            return
        dfs_second(idx + 1, cur_mask, cur_adj)
        if not (cur_adj >> idx) & 1:
            dfs_second(idx + 1,
                       cur_mask | (1 << idx),
                       cur_adj | (adj[n1 + idx] >> n1) & ((1 << n2) - 1))
    dfs_second(0, 0, 0)

    # try combine
    for mask1, adj1, sz1 in first_subsets:
        need = k - sz1
        if need < 0 or need > n2:
            continue
        # compute forbidden vertices in second half due to cross edges
        forbid = 0
        tmp = mask1
        while tmp:
            lsb = tmp & -tmp
            i = (lsb.bit_length() - 1)
            forbid |= cross_first[i]
            tmp ^= lsb
        for mask2 in second_by_size[need]:
            if mask2 & forbid == 0:
                # construct answer
                ans = []
                for i in range(n1):
                    if mask1 >> i & 1:
                        ans.append(i + 1)
                for j in range(n2):
                    if mask2 >> j & 1:
                        ans.append(n1 + j + 1)
                print("YES")
                print(' '.join(map(str, ans)))
                return
    print("NO")

if __name__ == "__main__":
    solve()
