import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_mask = (1 << n) - 1
    result = []

    # simple recursive search with pruning
    def dfs(chosen_mask: int, cand_mask: int, chosen_cnt: int) -> bool:
        nonlocal result
        # enough vertices left?
        if chosen_cnt + cand_mask.bit_count() < k:
            return False
        if chosen_cnt == k:
            # extract vertices
            res = []
            cm = chosen_mask
            while cm:
                v = (cm & -cm).bit_length() - 1
                res.append(v + 1)
                cm &= cm - 1
            result = res
            return True
        if cand_mask == 0:
            return False

        # choose a vertex (heuristic: vertex with max degree in candidates)
        # simple pick first
        v = (cand_mask & -cand_mask).bit_length() - 1

        # branch: include v
        new_chosen = chosen_mask | (1 << v)
        new_cand = cand_mask & ~adj[v] & ~(1 << v)
        if dfs(new_chosen, new_cand, chosen_cnt + 1):
            return True

        # branch: exclude v
        cand_mask &= ~(1 << v)
        if dfs(chosen_mask, cand_mask, chosen_cnt):
            return True

        return False

    if dfs(0, all_mask, 0):
        print("YES")
        print(" ".join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
