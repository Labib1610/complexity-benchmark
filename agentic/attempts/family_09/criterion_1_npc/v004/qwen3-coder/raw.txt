def solve():
    import sys
    from itertools import combinations
    
    # Read input
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Store friendships as a set of frozensets for fast lookup
    friendships = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        friendships.add(frozenset([u, v]))
    
    # Generate all possible combinations of k users
    users = list(range(1, n+1))
    
    # For small k, we can check all combinations
    # But with n up to 46, we need a smarter approach
    
    # Use meet-in-the-middle approach for larger n
    if n <= 30:
        # Brute force for smaller n
        for combo in combinations(users, k):
            valid = True
            # Check all pairs in this combination
            for u, v in combinations(combo, 2):
                if frozenset([u, v]) in friendships:
                    valid = False
                    break
            if valid:
                print("YES")
                print(' '.join(map(str, combo)))
                return
        
        print("NO")
        return
    else:
        # For larger n, use a more optimized approach
        # Split the users into two halves
        mid = n // 2
        left_users = users[:mid]
        right_users = users[mid:]
        
        # Generate all valid subsets for left half
        left_valid = {}
        for i in range(min(k+1, 1 << mid)):
            subset = []
            for j in range(mid):
                if i & (1 << j):
                    subset.append(left_users[j])
            
            # Check if this subset is valid (no internal friendships)
            valid = True
            for u, v in combinations(subset, 2):
                if frozenset([u, v]) in friendships:
                    valid = False
                    break
            
            if valid:
                size = len(subset)
                if size not in left_valid:
                    left_valid[size] = []
                left_valid[size].append(set(subset))
        
        # Generate all valid subsets for right half
        right_valid = {}
        for i in range(1 << (n - mid)):
            subset = []
            for j in range(n - mid):
                if i & (1 << j):
                    subset.append(right_users[j])
            
            # Check if this subset is valid (no internal friendships)
            valid = True
            for u, v in combinations(subset, 2):
                if frozenset([u, v]) in friendships:
                    valid = False
                    break
            
            if valid:
                size = len(subset)
                if size not in right_valid:
                    right_valid[size] = []
                right_valid[size].append(set(subset))
        
        # Now check for combinations
        for left_size in range(k+1):
            if left_size not in left_valid:
                continue
            right_size = k - left_size
            if right_size < 0 or right_size not in right_valid:
                continue
            
            # Check each pair of left and right subsets
            for left_set in left_valid[left_size]:
                for right_set in right_valid[right_size]:
                    # Check if there are any cross friendships
                    valid = True
                    for u in left_set:
                        for v in right_set:
                            if frozenset([u, v]) in friendships:
                                valid = False
                                break
                        if not valid:
                            break
                    
                    if valid:
                        result = sorted(list(left_set) + list(right_set))
                        print("YES")
                        print(' '.join(map(str, result)))
                        return
        
        print("NO")
        return

def solve_simple():
    import sys
    from itertools import combinations
    
    # Read input
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Store friendships as a set of frozensets for fast lookup
    friendships = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        friendships.add(frozenset([u, v]))
    
    # Try all combinations of k users
    users = list(range(1, n+1))
    
    # Since n can be up to 46, but we only need to find one solution,
    # and k is at most n, we can try a randomized approach or 
    # just iterate through combinations efficiently
    
    # For k=1, always possible
    if k == 1:
        print("YES")
        print("1")
        return
    
    # For small enough search space, try direct enumeration
    # But with C(46, k) this could be huge
    
    # Let's try a different approach: backtracking with pruning
    
    def is_valid_with(current_set, new_user):
        """Check if adding new_user to current_set maintains validity"""
        for user in current_set:
            if frozenset([user, new_user]) in friendships:
                return False
        return True
    
    def backtrack(current_set, start_index):
        if len(current_set) == k:
            return current_set[:]
        
        if len(current_set) + (n - start_index) < k:
            return None
            
        for i in range(start_index, n):
            user = users[i]
            if is_valid_with(current_set, user):
                current_set.append(user)
                result = backtrack(current_set, i + 1)
                if result is not None:
                    return result
                current_set.pop()
        
        return None
    
    # Try to build a solution using backtracking
    result = backtrack([], 0)
    if result is not None:
        print("YES")
        print(' '.join(map(str, result)))
    else:
        print("NO")

def solve_final():
    import sys
    from itertools import combinations
    import random
    
    # Read input
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Store friendships as a set of frozensets for fast lookup
    friendships = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        friendships.add(frozenset([u, v]))
    
    users = list(range(1, n+1))
    
    # Special case: k=1
    if k == 1:
        print("YES")
        print("1")
        return
    
    # For very small n, try all combinations
    if n <= 25:  # This gives at most C(25,10) ~ 3 million combinations
        for combo in combinations(users, k):
            valid = True
            for u, v in combinations(combo, 2):
                if frozenset([u, v]) in friendships:
                    valid = False
                    break
            if valid:
                print("YES")
                print(' '.join(map(str, combo)))
                return
        print("NO")
        return
    
    # For larger n, use heuristic/randomized approach
    # First try some random samples
    for _ in range(min(10000, 100000 // max(1, k * (k-1) // 2))):
        combo = random.sample(users, k)
        valid = True
        for u, v in combinations(combo, 2):
            if frozenset([u, v]) in friendships:
                valid = False
                break
        if valid:
            print("YES")
            print(' '.join(map(str, sorted(combo))))
            return
    
    # If that doesn't work, try greedy construction
    def build_greedy():
        # Try building a set greedily
        current = []
        remaining = users[:]
        
        for _ in range(k):
            found = False
            random.shuffle(remaining)
            for user in remaining[:]:  # Copy to avoid modification during iteration
                # Check if we can add this user
                can_add = True
                for existing in current:
                    if frozenset([existing, user]) in friendships:
                        can_add = False
                        break
                
                if can_add:
                    current.append(user)
                    remaining.remove(user)
                    found = True
                    break
            
            if not found:
                return None
        
        return current
    
    # Try greedy construction several times
    for _ in range(100):
        result = build_greedy()
        if result is not None:
            print("YES")
            print(' '.join(map(str, sorted(result))))
            return
    
    print("NO")
