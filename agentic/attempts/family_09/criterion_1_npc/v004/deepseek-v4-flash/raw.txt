 We need to find an independent set of size exactly k in an undirected graph with up to 46 vertices.  
A simple brute force over all subsets is too slow (2^46).  
We can use a meet-in-the-middle approach: split vertices into two halves, enumerate all independent sets in each half, then combine.

Here’s the implementation:

```python
import sys
from bisect import bisect_left

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Split vertices into two halves
    left_n = n // 2
    right_n = n - left_n

    # Enumerate all independent sets in left half
    left_sets = []  # list of (mask, size)
    for mask in range(1 << left_n):
        ok = True
        for i in range(left_n):
            if mask & (1 << i):
                if mask & adj[i]:
                    ok = False
                    break
        if ok:
            left_sets.append((mask, bin(mask).count('1')))
    # Sort by size for combination
    left_sets.sort(key=lambda x: x[1])

    # Enumerate all independent sets in right half
    right_sets = []  # list of (mask, size)
    for mask in range(1 << right_n):
        ok = True
        for i in range(right_n):
            if mask & (1 << i):
                if mask & (adj[i + left_n] >> left_n):
                    ok = False
                    break
        if ok:
            right_sets.append((mask, bin(mask).count('1')))
    right_sets.sort(key=lambda x: x[1])

    # For each left set, find a right set that is compatible and sizes sum to k
    # Precompute for each right set the maximum size for each possible size? Actually we need exact sum.
    # We'll group right sets by size.
    from collections import defaultdict
    right_by_size = defaultdict(list)
    for mask, sz in right_sets:
        right_by_size[sz].append(mask)

    # For each left set, we need a right set of size k - left_size that is compatible
    for lmask, lsz in left_sets:
        need = k - lsz
        if need < 0 or need > right_n:
            continue
        # Build forbidden mask for right side: any vertex adjacent to any vertex in lmask
        forbidden = 0
        for i in range(left_n):
            if lmask & (1 << i):
                forbidden |= (adj[i] >> left_n)
        # Now we need a right mask of size 'need' that has no intersection with forbidden
        for rmask in right_by_size.get(need, []):
            if rmask & forbidden == 0:
                # Found
                res = []
                for i in range(left_n):
                    if lmask & (1 << i):
                        res.append(i + 1)
                for j in range(right_n):
                    if rmask & (1 << j):
                        res.append(left_n + j + 1)
                print("YES")
                print(" ".join(map(str, res)))
                return

    print("NO")

if __name__ == "__main__":
    solve()
```
