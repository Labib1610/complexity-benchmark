import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    conflicts = [0] * n

    pos = 3
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        conflicts[u] |= 1 << v
        conflicts[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    full = (1 << n) - 1
    original_adj = [
        full & ~(conflicts[v] | (1 << v))
        for v in range(n)
    ]

    permutation = sorted(range(n), key=lambda v: original_adj[v].bit_count())
    new_index = [0] * n
    for i, v in enumerate(permutation):
        new_index[v] = i

    adj = [0] * n
    for i, old_v in enumerate(permutation):
        mask = original_adj[old_v]
        translated = 0
        while mask:
            bit = mask & -mask
            old_u = bit.bit_length() - 1
            translated |= 1 << new_index[old_u]
            mask ^= bit
        adj[i] = translated

    def reduce_candidates(mask, need):
        if need <= 1:
            return mask
        minimum_degree = need - 1
        while mask.bit_count() >= need:
            remove = 0
            q = mask
            while q:
                bit = q & -q
                v = bit.bit_length() - 1
                if (adj[v] & mask).bit_count() < minimum_degree:
                    remove |= bit
                q ^= bit
            if not remove:
                return mask
            mask &= ~remove
        return 0

    candidates = reduce_candidates(full, k)
    if candidates.bit_count() < k:
        print("NO")
        return

    def greedy(forced=-1):
        if forced >= 0:
            clique = [forced]
            available = candidates & adj[forced]
        else:
            clique = []
            available = candidates

        while len(clique) < k and available:
            q = available
            best_v = -1
            best_degree = -1
            while q:
                bit = q & -q
                v = bit.bit_length() - 1
                degree = (adj[v] & available).bit_count()
                if degree > best_degree:
                    best_degree = degree
                    best_v = v
                q ^= bit

            clique.append(best_v)
            available &= adj[best_v]

        return clique if len(clique) == k else None

    answer = greedy()
    if answer is None:
        q = candidates
        while q and answer is None:
            bit = q & -q
            v = bit.bit_length() - 1
            answer = greedy(v)
            q ^= bit

    if answer is not None:
        result = [permutation[v] + 1 for v in answer]
        print("YES")
        print(*result)
        return

    path = []
    solution = None

    def expand(mask, depth):
        nonlocal solution

        need = k - depth
        if need == 0:
            solution = path.copy()
            return True

        if mask.bit_count() < need:
            return False

        mask = reduce_candidates(mask, need)
        count = mask.bit_count()
        if count < need:
            return False

        if count == need:
            added = []
            q = mask
            while q:
                bit = q & -q
                v = bit.bit_length() - 1
                added.append(v)
                q ^= bit
            solution = path + added
            return True

        order = []
        bounds = []
        uncolored = mask
        color = 0

        while uncolored:
            color += 1
            available = uncolored
            while available:
                bit = available & -available
                v = bit.bit_length() - 1
                order.append(v)
                bounds.append(color)
                uncolored ^= bit
                available ^= bit
                available &= ~adj[v]

        for i in range(len(order) - 1, -1, -1):
            if depth + bounds[i] < k:
                return False

            v = order[i]
            bit = 1 << v
            path.append(v)

            if expand(mask & adj[v], depth + 1):
                return True

            path.pop()
            mask &= ~bit

            if mask.bit_count() < need:
                return False

        return False

    if expand(candidates, 0):
        result = [permutation[v] + 1 for v in solution]
        print("YES")
        print(*result)
    else:
        print("NO")

if __name__ == "__main__":
    main()
