import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Greedy heuristic to find a large independent set
    # Sort vertices by degree (ascending) to prioritize low-degree nodes
    order = sorted(range(n), key=lambda x: len(adj[x]))
    
    best_set = []
    # Try different orderings to find a set of size k
    # We try the sorted order and its reverse
    for _ in range(2):
        current_set = []
        used = [False] * n
        for u in order:
            if not used[u]:
                can_add = True
                for v in adj[u]:
                    if used[v]:
                        can_add = False
                        break
                if can_add:
                    current_set.append(u)
                    used[u] = True
                    if len(current_set) == k:
                        break
        if len(current_set) > len(best_set):
            best_set = current_set
            if len(best_set) == k:
                break
        order.reverse()
    
    if len(best_set) >= k:
        print("YES")
        print(' '.join(str(x + 1) for x in best_set[:k]))
        return
    
    # If greedy fails, use meet-in-the-middle
    # Split vertices into two halves
    mid = n // 2
    left = list(range(mid))
    right = list(range(mid, n))
    
    # Precompute conflict masks for each half
    left_conflict = [0] * len(left)
    for i, u in enumerate(left):
        mask = 0
        for v in adj[u]:
            if v < mid:
                mask |= (1 << (left.index(v) if v in left else -1))
        # More efficient: direct bit position
        mask = 0
        for v in adj[u]:
            if v < mid:
                mask |= (1 << v)
        left_conflict[i] = mask
    
    right_conflict = [0] * len(right)
    for i, u in enumerate(right):
        mask = 0
        for v in adj[u]:
            if v >= mid:
                mask |= (1 << (v - mid))
        right_conflict[i] = mask
    
    # Generate all valid subsets for left half
    left_subsets = {}
    left_size = len(left)
    for mask in range(1 << left_size):
        valid = True
        size = 0
        for i in range(left_size):
            if mask & (1 << i):
                size += 1
                if mask & left_conflict[i]:
                    valid = False
                    break
        if valid:
            if size not in left_subsets:
                left_subsets[size] = []
            if len(left_subsets[size]) < 1:  # Store only one per size
                left_subsets[size].append(mask)
    
    # Generate all valid subsets for right half
    right_subsets = {}
    right_size = len(right)
    for mask in range(1 << right_size):
        valid = True
        size = 0
        for i in range(right_size):
            if mask & (1 << i):
                size += 1
                if mask & right_conflict[i]:
                    valid = False
                    break
        if valid:
            if size not in right_subsets:
                right_subsets[size] = []
            if len(right_subsets[size]) < 1:
                right_subsets[size].append(mask)
    
    # Check for valid combination
    # Precompute cross-conflict: which left vertices conflict with which right vertices
    cross_conflict_left = [0] * left_size  # For each left vertex, which right vertices it conflicts with
    cross_conflict_right = [0] * right_size  # For each right vertex, which left vertices it conflicts with
    
    for i, u in enumerate(left):
        for v in adj[u]:
            if v >= mid:
                j = v - mid
                cross_conflict_left[i] |= (1 << j)
                cross_conflict_right[j] |= (1 << i)
    
    for ls in left_subsets:
        rs = k - ls
        if rs in right_subsets:
            for lm in left_subsets[ls]:
                for rm in right_subsets[rs]:
                    # Check cross-conflict
                    valid = True
                    # For each selected left vertex, check if it conflicts with any selected right vertex
                    for i in range(left_size):
                        if lm & (1 << i):
                            if lm & (cross_conflict_left[i] << 0):  # This is wrong
                                pass
                            # Check if any selected right vertex conflicts
                            if rm & cross_conflict_left[i]:
                                valid = False
                                break
                    if valid:
                        # Found valid combination
                        result = []
                        for i in range(left_size):
                            if lm & (1 << i):
                                result.append(left[i] + 1)
                        for i in range(right_size):
                            if rm & (1 << i):
                                result.append(right[i] + 1)
                        print("YES")
                        print(' '.join(map(str, result)))
                        return
    
    print("NO")

if __name__ == "__main__":
    solve()
