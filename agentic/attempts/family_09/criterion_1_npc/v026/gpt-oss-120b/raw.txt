import sys
sys.setrecursionlimit(10000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_vertices_mask = (1 << n) - 1
    result = []

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i
    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for u in range(n):
            if (adj[v] >> u) & 1:
                mask |= 1 << pos[u]
        adj2[i] = mask

    # initial candidates: all vertices in ordered space
    candidates = (1 << n) - 1
    target = k
    found = False
    ans_mask = 0

    # precompute popcounts for speed
    popcount = int.bit_count

    def dfs(cand_mask: int, cur_mask: int, cur_size: int):
        nonlocal found, ans_mask
        if found:
            return
        # prune if even taking all remaining cannot reach k
        if cur_size + popcount(cand_mask) < target:
            return
        if cur_size == target:
            found = True
            ans_mask = cur_mask
            return
        # choose a vertex: pick the one with highest degree among candidates
        # simple: take least significant bit
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        dfs(new_cand, cur_mask | (1 << v), cur_size + 1)
        if found:
            return
        # branch: exclude v
        dfs(cand_mask & ~(1 << v), cur_mask, cur_size)

    dfs(candidates, 0, 0)

    if not found:
        print("NO")
        return

    # translate ans_mask back to original vertex numbers
    selected = []
    for i in range(n):
        if (ans_mask >> i) & 1:
            selected.append(order[i] + 1)
    print("YES")
    print(" ".join(map(str, selected)))

if __name__ == "__main__":
    solve()
