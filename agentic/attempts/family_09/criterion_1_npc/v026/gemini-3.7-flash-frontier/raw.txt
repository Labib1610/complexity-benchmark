import sys


def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])
    k = int(data[2])

    if k > n:
        print("NO")
        return

    if k == 0:
        print("YES\n")
        return

    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Reorder vertices by degree ascending to improve pruning
    deg = [adj[i].bit_count() for i in range(n)]
    order = sorted(range(n), key=lambda x: deg[x])
    inv_order = [0] * n
    for i, v in enumerate(order):
        inv_order[v] = i

    new_adj = [0] * n
    for i in range(n):
        orig_u = order[i]
        mask = 0
        for j in range(n):
            orig_v = order[j]
            if (adj[orig_u] >> orig_v) & 1:
                mask |= 1 << j
        new_adj[i] = mask

    adj = new_adj

    # Max clique / independent set search
    ans = None

    def get_upper_bound(candidates):
        colors = 0
        rem = candidates
        while rem:
            colors += 1
            v_mask = rem
            while v_mask:
                v = (v_mask & -v_mask).bit_length() - 1
                rem &= ~(1 << v)
                v_mask &= ~adj[v]
                v_mask &= rem
        return colors

    def search(candidates, current_set):
        nonlocal ans
        if len(current_set) == k:
            ans = [order[x] + 1 for x in current_set]
            return True

        if len(current_set) + candidates.bit_count() < k:
            return False

        if len(current_set) + get_upper_bound(candidates) < k:
            return False

        c = candidates
        while c:
            if len(current_set) + c.bit_count() < k:
                break
            v = (c & -c).bit_length() - 1
            c &= c - 1
            current_set.append(v)
            if search(c & ~adj[v], current_set):
                return True
            current_set.pop()

        return False

    initial_candidates = (1 << n) - 1
    if search(initial_candidates, []):
        print("YES")
        print(*(ans[:k]))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
