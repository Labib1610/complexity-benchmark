import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    neigh = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        neigh[u] |= 1 << v
        neigh[v] |= 1 << u

    ALL = (1 << n) - 1
    result = []

    # recursive search for independent set of size k
    def dfs(cur_mask: int, cand_mask: int) -> bool:
        nonlocal result
        cur_size = cur_mask.bit_count()
        if cur_size == k:
            # extract vertices
            res = []
            for i in range(n):
                if cur_mask >> i & 1:
                    res.append(i + 1)
            result = res
            return True
        # prune if not enough candidates left
        if cur_size + cand_mask.bit_count() < k:
            return False
        # choose a vertex: heuristic - smallest degree in candidates
        # compute degrees within candidates
        # pick first vertex
        # simple heuristic: pick vertex with minimal (neigh & cand).bit_count()
        v = -1
        min_deg = n + 1
        cm = cand_mask
        while cm:
            i = (cm & -cm).bit_length() - 1
            deg = (neigh[i] & cand_mask).bit_count()
            if deg < min_deg:
                min_deg = deg
                v = i
                if deg == 0:
                    break
            cm &= cm - 1
        if v == -1:
            return False

        # branch 1: include v
        new_cur = cur_mask | (1 << v)
        new_cand = cand_mask & ~ (neigh[v] | (1 << v))
        if dfs(new_cur, new_cand):
            return True
        # branch 2: exclude v
        cand_mask &= ~(1 << v)
        return dfs(cur_mask, cand_mask)

    if dfs(0, ALL):
        print("YES")
        print(" ".join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
