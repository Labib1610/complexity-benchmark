import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    full = (1 << n) - 1
    adj = [full ^ (1 << i) for i in range(n)]

    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u] &= ~(1 << v)
        adj[v] &= ~(1 << u)

    if k == 1:
        print("YES")
        print(1)
        return

    # Fast greedy attempts.
    starts = sorted(range(n), key=lambda v: adj[v].bit_count(), reverse=True)
    for start in starts:
        chosen = [start]
        candidates = adj[start]
        while candidates and len(chosen) < k:
            q = candidates
            best = -1
            best_score = -1
            while q:
                bit = q & -q
                v = bit.bit_length() - 1
                score = (adj[v] & candidates).bit_count()
                if score > best_score:
                    best_score = score
                    best = v
                q ^= bit
            chosen.append(best)
            candidates &= adj[best]
        if len(chosen) == k:
            print("YES")
            print(*[v + 1 for v in chosen])
            return

    # Every vertex of a k-clique belongs to the (k-1)-core.
    active = full
    while True:
        remove = 0
        q = active
        while q:
            bit = q & -q
            v = bit.bit_length() - 1
            if (adj[v] & active).bit_count() < k - 1:
                remove |= bit
            q ^= bit
        if not remove:
            break
        active &= ~remove

    if active.bit_count() < k:
        print("NO")
        return

    vertices = []
    q = active
    while q:
        bit = q & -q
        vertices.append(bit.bit_length() - 1)
        q ^= bit

    # Low-degree vertices get lower indices; reverse branching then tends
    # to consider high-degree vertices first.
    vertices.sort(key=lambda v: (adj[v] & active).bit_count())
    size = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}

    new_adj = [0] * size
    for i, old_v in enumerate(vertices):
        q = adj[old_v] & active
        mask = 0
        while q:
            bit = q & -q
            old_u = bit.bit_length() - 1
            mask |= 1 << index[old_u]
            q ^= bit
        new_adj[i] = mask

    solution = None
    stack = []

    def expand(candidates):
        nonlocal solution

        current_size = len(stack)
        if current_size + candidates.bit_count() < k:
            return False

        order = []
        colors = []
        uncolored = candidates
        color = 0

        # Greedy coloring gives an upper bound on any clique in candidates.
        while uncolored:
            color += 1
            available = uncolored
            while available:
                bit = available & -available
                v = bit.bit_length() - 1
                order.append(v)
                colors.append(color)
                uncolored ^= bit
                available ^= bit
                available &= ~new_adj[v]

        for i in range(len(order) - 1, -1, -1):
            if current_size + colors[i] < k:
                return False

            v = order[i]
            bit = 1 << v
            if not (candidates & bit):
                continue

            stack.append(v)
            if len(stack) == k:
                solution = stack.copy()
                return True

            next_candidates = candidates & new_adj[v]
            if len(stack) + next_candidates.bit_count() >= k:
                if expand(next_candidates):
                    return True

            stack.pop()
            candidates ^= bit

        return False

    expand((1 << size) - 1)

    if solution is None:
        print("NO")
    else:
        answer = [vertices[v] + 1 for v in solution]
        print("YES")
        print(*answer)

if __name__ == "__main__":
    main()
