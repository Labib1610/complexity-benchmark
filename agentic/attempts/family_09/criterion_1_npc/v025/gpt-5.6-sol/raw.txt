import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    adj = [0] * n

    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    if m == 0:
        print("YES")
        print(*range(1, k + 1))
        return

    if k == n:
        print("NO")
        return

    # Quickly try a minimum-current-degree greedy independent set.
    candidates = (1 << n) - 1
    greedy = []
    while candidates:
        x = candidates
        best_v = -1
        best_deg = n + 1
        while x:
            bit = x & -x
            v = bit.bit_length() - 1
            deg = (adj[v] & candidates).bit_count()
            if deg < best_deg:
                best_deg = deg
                best_v = v
            x ^= bit

        greedy.append(best_v)
        if len(greedy) == k:
            print("YES")
            print(*(v + 1 for v in greedy))
            return
        candidates &= ~(adj[best_v] | (1 << best_v))

    # Put higher-degree vertices in the searched half.
    order = sorted(range(n), key=lambda v: adj[v].bit_count(), reverse=True)
    b_size = n // 2
    a_size = n - b_size
    a_vertices = order[:a_size]
    b_vertices = order[a_size:]

    pos_a = [-1] * n
    pos_b = [-1] * n
    for i, v in enumerate(a_vertices):
        pos_a[v] = i
    for i, v in enumerate(b_vertices):
        pos_b[v] = i

    adj_a = [0] * a_size
    cross = [0] * a_size
    for i, u in enumerate(a_vertices):
        mask_a = 0
        mask_b = 0
        neighbors = adj[u]
        for j, v in enumerate(a_vertices):
            if (neighbors >> v) & 1:
                mask_a |= 1 << j
        for j, v in enumerate(b_vertices):
            if (neighbors >> v) & 1:
                mask_b |= 1 << j
        adj_a[i] = mask_a
        cross[i] = mask_b

    adj_b = [0] * b_size
    for i, u in enumerate(b_vertices):
        mask = 0
        neighbors = adj[u]
        for j, v in enumerate(b_vertices):
            if (neighbors >> v) & 1:
                mask |= 1 << j
        adj_b[i] = mask

    full_b = (1 << b_size) - 1

    # dp[mask] is the maximum independent-set size in B induced by mask.
    dp = bytearray(1 << b_size)
    for i in range(b_size):
        bit = 1 << i
        keep = (bit - 1) & ~adj_b[i]
        for lower in range(bit):
            without = dp[lower]
            with_vertex = dp[lower & keep] + 1
            dp[bit + lower] = with_vertex if with_vertex > without else without

    def reconstruct_b(mask, target):
        chosen = []
        while target:
            i = mask.bit_length() - 1
            bit = 1 << i
            lower = mask ^ bit
            keep = (bit - 1) & ~adj_b[i]
            reduced = lower & keep
            if dp[reduced] + 1 >= target:
                chosen.append(i)
                target -= 1
                mask = reduced
            else:
                mask = lower
        return chosen

    full_a = (1 << a_size) - 1
    allowed_after = [full_b ^ cross[i] for i in range(a_size)]

    sys.setrecursionlimit(10000)

    def search(remaining, allowed_b, count, selected):
        best_b = dp[allowed_b]
        need = k - count

        if best_b >= need:
            return selected, allowed_b, need

        if not remaining:
            return None

        if count + remaining.bit_count() + best_b < k:
            return None

        bit = remaining & -remaining
        v = bit.bit_length() - 1
        rest = remaining ^ bit

        result = search(
            rest & ~adj_a[v],
            allowed_b & allowed_after[v],
            count + 1,
            selected | bit
        )
        if result is not None:
            return result

        return search(rest, allowed_b, count, selected)

    result = search(full_a, full_b, 0, 0)

    if result is None:
        print("NO")
        return

    selected_a, allowed_b, need_b = result
    answer = []

    x = selected_a
    while x:
        bit = x & -x
        i = bit.bit_length() - 1
        answer.append(a_vertices[i] + 1)
        x ^= bit

    for i in reconstruct_b(allowed_b, need_b):
        answer.append(b_vertices[i] + 1)

    print("YES")
    print(*answer[:k])

if __name__ == "__main__":
    main()
