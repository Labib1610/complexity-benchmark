import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        idx += 2

    def solve_subgraph_deg1(p_mask):
        res = []
        rem = p_mask
        while rem:
            v = (rem & -rem).bit_length() - 1
            res.append(v)
            rem &= ~(adj[v] | (1 << v))
        return res

    def search(p_mask, current):
        if len(current) == k:
            return current
        if len(current) + p_mask.bit_count() < k:
            return None

        # Find vertex with maximum degree in the candidate subgraph
        max_deg = -1
        best_v = -1
        temp = p_mask
        while temp:
            v = (temp & -temp).bit_length() - 1
            temp &= temp - 1
            deg = (adj[v] & p_mask).bit_count()
            if deg > max_deg:
                max_deg = deg
                best_v = v

        if max_deg <= 1:
            indep = solve_subgraph_deg1(p_mask)
            if len(current) + len(indep) >= k:
                return current + indep[: k - len(current)]
            return None

        # Branch 1: include best_v
        current.append(best_v)
        res = search(p_mask & ~(adj[best_v] | (1 << best_v)), current)
        if res is not None:
            return res
        current.pop()

        # Branch 2: exclude best_v
        res = search(p_mask & ~(1 << best_v), current)
        if res is not None:
            return res

        return None

    initial_mask = (1 << n) - 1
    result = search(initial_mask, [])

    if result is not None:
        print("YES")
        print(*(v + 1 for v in result))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
