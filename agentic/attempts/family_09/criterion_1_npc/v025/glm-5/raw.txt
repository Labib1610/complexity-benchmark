import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [set() for _ in range(n)]
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    complement_adj = [set() for _ in range(n)]
    for i in range(n):
        for j in range(n):
            if i != j and j not in adj[i]:
                complement_adj[i].add(j)
    
    def is_clique(nodes):
        nodes_list = list(nodes)
        for i in range(len(nodes_list)):
            for j in range(i+1, len(nodes_list)):
                u, v = nodes_list[i], nodes_list[j]
                if v not in complement_adj[u]:
                    return False
        return True
    
    def bron_kerbosch_pivot(R, P, X):
        if len(R) + len(P) < k:
            return None
        
        if len(R) >= k:
            return list(R)[:k]
        
        if not P and not X:
            if len(R) >= k:
                return list(R)[:k]
            return None
        
        if len(P) == 0:
            return None
        
        max_clique_in_component = len(R) + len(P)
        if max_clique_in_component < k:
            return None
        
        pivot = max(P | X, key=lambda v: len(P & complement_adj[v]))
        candidates_for_iteration = P - complement_adj[pivot]
        
        for v in sorted(list(candidates_for_iteration), key=lambda x: -len(P & complement_adj[x])):
            new_R = R | {v}
            new_P = P & complement_adj[v]
            new_X = X & complement_adj[v]
            
            result = bron_kerbosch_pivot(new_R, new_P, new_X)
            if result is not None:
                return result
            
            P.remove(v)
            X.add(v)
        
        return None
    
    result = bron_kerbosch_pivot(set(), set(range(n)), set())
    
    if result is not None:
        print("YES")
        print(" ".join(str(x + 1) for x in result))
    else:
        print("NO")

solve()
