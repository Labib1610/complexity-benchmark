We can model this as finding an independent set of size exactly `k` in a graph with up to 46 vertices.  
The constraints are small enough for a backtracking approach with pruning.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # If k > n, impossible
    if k > n:
        print("NO")
        return
    
    # Try to find an independent set of size k
    # Use backtracking with pruning
    chosen = []
    
    def backtrack(start, chosen_set, chosen_list):
        if len(chosen_list) == k:
            return True
        # Prune: not enough remaining vertices
        if n - start < k - len(chosen_list):
            return False
        for v in range(start, n):
            # Check if v conflicts with chosen_set
            if (adj[v] & chosen_set) == 0:
                chosen_set |= (1 << v)
                chosen_list.append(v)
                if backtrack(v + 1, chosen_set, chosen_list):
                    return True
                chosen_list.pop()
                chosen_set &= ~(1 << v)
        return False
    
    if backtrack(0, 0, chosen):
        print("YES")
        print(" ".join(str(x + 1) for x in chosen))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
