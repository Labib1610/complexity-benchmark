import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]

    if k == 1:
        print("YES")
        print(1)
        return

    forbidden = [0] * n
    pos = 3
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        forbidden[u] |= 1 << v
        forbidden[v] |= 1 << u

    full = (1 << n) - 1
    original_adj = [
        full & ~(forbidden[v] | (1 << v))
        for v in range(n)
    ]

    # High-degree vertices first usually improve both greedy construction
    # and the coloring-based branch-and-bound search.
    permutation = sorted(
        range(n),
        key=lambda v: original_adj[v].bit_count(),
        reverse=True
    )
    inverse = [0] * n
    for new, old in enumerate(permutation):
        inverse[old] = new

    adj = [0] * n
    for new_v, old_v in enumerate(permutation):
        mask = original_adj[old_v]
        converted = 0
        while mask:
            bit = mask & -mask
            old_u = bit.bit_length() - 1
            converted |= 1 << inverse[old_u]
            mask ^= bit
        adj[new_v] = converted

    all_vertices = (1 << n) - 1

    # Fast greedy attempts, one from every possible starting vertex.
    for start in range(n):
        clique = [start]
        candidates = adj[start]
        while len(clique) < k and candidates:
            scan = candidates
            best_v = -1
            best_degree = -1
            while scan:
                bit = scan & -scan
                v = bit.bit_length() - 1
                degree = (adj[v] & candidates).bit_count()
                if degree > best_degree:
                    best_degree = degree
                    best_v = v
                scan ^= bit
            clique.append(best_v)
            candidates &= adj[best_v]

        if len(clique) == k:
            answer = [permutation[v] + 1 for v in clique]
            print("YES")
            print(*answer)
            return

    path = []

    def expand(candidates):
        need = k - len(path)
        count = candidates.bit_count()
        if count < need:
            return False

        if need == 1:
            bit = candidates & -candidates
            path.append(bit.bit_length() - 1)
            return True

        if need == 2:
            scan = candidates
            while scan:
                bit = scan & -scan
                v = bit.bit_length() - 1
                neighbors = adj[v] & candidates
                if neighbors:
                    other_bit = neighbors & -neighbors
                    path.append(v)
                    path.append(other_bit.bit_length() - 1)
                    return True
                scan ^= bit
            return False

        # Greedy coloring gives an upper bound on any clique contained
        # in each prefix of the produced ordering.
        order = []
        colors = []
        remaining = candidates
        color = 0

        while remaining:
            color += 1
            available = remaining
            while available:
                bit = available & -available
                v = bit.bit_length() - 1
                order.append(v)
                colors.append(color)
                remaining ^= bit
                available ^= bit
                available &= ~adj[v]

        depth = len(path)
        for i in range(len(order) - 1, -1, -1):
            if depth + colors[i] < k:
                return False

            v = order[i]
            bit = 1 << v
            path.append(v)

            next_candidates = candidates & adj[v]
            if expand(next_candidates):
                return True

            path.pop()
            candidates ^= bit

        return False

    if expand(all_vertices):
        answer = [permutation[v] + 1 for v in path[:k]]
        print("YES")
        print(*answer)
    else:
        print("NO")

if __name__ == "__main__":
    main()
