from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    # Build adjacency list to represent the graph
    adj = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # If k == 1, we can always choose any single participant
    if k == 1:
        print("YES")
        print(1)
        return
    
    # Generate all combinations of k participants
    # Since n <= 46, and we need to check combinations, we should optimize
    # However, in worst case, this could be too slow, but given constraints, let's proceed
    
    # We'll try all combinations of k nodes and check if they form an independent set
    # To optimize a bit, we can iterate through combinations in a more efficient way
    # But since n is small enough, and we might prune early, it's acceptable
    
    # Let's try a different approach: since n<=46, 2^46 is too large, but we can use meet-in-the-middle
    # Or just try combinations if k is small enough
    
    # Actually, let's just try combinations for small k, and for larger k, it's likely impossible due to density
    # But let's implement a direct approach first
    
    # Try all combinations of k participants
    for combo in combinations(range(1, n + 1), k):
        valid = True
        # Check if all pairs in combo are NOT in the adjacency list
        for i in range(k):
            for j in range(i + 1, k):
                u = combo[i]
                v = combo[j]
                if v in adj[u]:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            print("YES")
            print(' '.join(map(str, combo)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
