def solve():
    import sys
    from itertools import combinations

    # Read input
    line = sys.stdin.readline().strip()
    n, m, k = map(int, line.split())
    
    # Read edges
    edges = set()
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        if u > v:
            u, v = v, u
        edges.add((u, v))
    
    # If k is 1, we can always pick any node
    if k == 1:
        print("YES")
        print(1)
        return
    
    # Generate all possible combinations of k nodes
    nodes = list(range(1, n+1))
    for combo in combinations(nodes, k):
        # Check if this combination forms an independent set
        valid = True
        for i in range(len(combo)):
            for j in range(i+1, len(combo)):
                u = combo[i]
                v = combo[j]
                if u > v:
                    u, v = v, u
                if (u, v) in edges:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            print("YES")
            print(' '.join(map(str, combo)))
            return
    
    print("NO")

solve()
