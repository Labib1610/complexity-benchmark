import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Try to find an independent set of size k
    # Using branch and bound with bitmask representation
    
    ALL = (1 << (n + 1)) - 1
    
    def search(chosen, candidates, depth):
        if depth == k:
            return chosen
        # Pruning: check if enough candidates remain
        cnt = bin(candidates).count('1')
        if cnt < k - depth:
            return 0
        
        # Pick the vertex with maximum degree among candidates (heuristic)
        # to reduce branching factor
        best = -1
        best_deg = -1
        temp = candidates
        while temp:
            v = (temp & -temp).bit_length() - 1
            deg = bin(adj[v] & candidates).count('1')
            if deg > best_deg:
                best_deg = deg
                best = v
            temp &= temp - 1
        
        v = best
        
        # Branch 1: include v
        new_chosen = chosen | (1 << v)
        new_candidates = candidates & ~(1 << v) & ~(adj[v])
        result = search(new_chosen, new_candidates, depth + 1)
        if result:
            return result
        
        # Branch 2: exclude v
        new_candidates = candidates & ~(1 << v)
        return search(chosen, new_candidates, depth)
    
    candidates = 0
    for i in range(1, n + 1):
        candidates |= 1 << i
    
    result = search(0, candidates, 0)
    
    if result:
        print("YES")
        members = []
        for i in range(1, n + 1):
            if result & (1 << i):
                members.append(str(i))
        print(' '.join(members))
    else:
        print("NO")

solve()
