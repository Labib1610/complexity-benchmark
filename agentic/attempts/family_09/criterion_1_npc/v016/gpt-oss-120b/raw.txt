```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full_mask = (1 << n) - 1
    solution = None

    # depth‑first search with pruning
    def dfs(cur_mask: int, rem_mask: int):
        nonlocal solution
        if solution is not None:
            return True
        cur_sz = cur_mask.bit_count()
        if cur_sz == k:
            solution = cur_mask
            return True
        # not enough vertices left
        if cur_sz + rem_mask.bit_count() < k:
            return False
        # choose a vertex (heuristic: vertex with max degree in rem)
        # simple: take lowest set bit
        v = (rem_mask & -rem_mask).bit_length() - 1

        # branch: include v
        new_rem = rem_mask & ~adj[v] & ~(1 << v)
        if dfs(cur_mask | (1 << v), new_rem):
            return True

        # branch: exclude v
        if dfs(cur_mask, rem_mask & ~(1 << v)):
            return True

        return False

    dfs(0, full_mask)

    if solution is None:
        print("NO")
    else:
        res = [str(i + 1) for i in range(n) if (solution >> i) & 1]
        print("YES")
        print(" ".join(res))

if __name__ == "__main__":
    solve()
```
