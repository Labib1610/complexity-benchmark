import sys


def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])
    k = int(data[2])

    if k > n:
        print("NO")
        return

    if k == 1:
        print("YES")
        print(1)
        return

    # 0-indexed internally
    conflict = [0] * n
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        idx += 2
        conflict[u] |= 1 << v
        conflict[v] |= 1 << u

    # Complement graph: adj[u] has bit set if no conflict between u and v, and u != v
    all_mask = (1 << n) - 1
    adj = [(all_mask ^ conflict[i]) ^ (1 << i) for i in range(n)]

    # Sort vertices by degree in complement graph to speed up search
    deg = [adj[i].bit_count() for i in range(n)]
    order = sorted(range(n), key=lambda x: deg[x])
    inv_order = {order[i]: i for i in range(n)}

    new_adj = [0] * n
    for i in range(n):
        orig_u = order[i]
        mask = 0
        for j in range(n):
            orig_v = order[j]
            if (adj[orig_u] >> orig_v) & 1:
                mask |= 1 << j
        new_adj[i] = mask

    res = []

    def dfs(candidates, path):
        if len(path) == k:
            res.extend(path)
            return True
        if len(path) + candidates.bit_count() < k:
            return False

        c = candidates
        while c:
            if len(path) + c.bit_count() < k:
                return False
            v = (c & -c).bit_length() - 1
            c &= c - 1
            if dfs(c & new_adj[v], path + [v]):
                return True
        return False

    initial_candidates = (1 << n) - 1
    if dfs(initial_candidates, []):
        print("YES")
        ans = [order[v] + 1 for v in res]
        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    main()
