We can model this as finding an independent set of size exactly k in a graph with up to 46 vertices.  
A simple backtracking with pruning will work well for this size.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # If k is 0, but constraints say k >= 1, so skip.
    if k == 0:
        print("YES")
        print()
        return

    # Try to find an independent set of size k
    # Use backtracking with pruning: order vertices by degree (smaller first)
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"))
    chosen = []
    used_mask = 0

    def backtrack(start, depth):
        if depth == k:
            return True
        # Prune: not enough remaining vertices
        if n - start < k - depth:
            return False
        for i in range(start, n):
            v = order[i]
            if used_mask & (1 << v):
                continue
            if adj[v] & used_mask:
                continue
            # choose v
            chosen.append(v)
            old_mask = used_mask
            used_mask |= 1 << v
            if backtrack(i + 1, depth + 1):
                return True
            # backtrack
            used_mask = old_mask
            chosen.pop()
        return False

    if backtrack(0, 0):
        print("YES")
        print(" ".join(str(x + 1) for x in chosen))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
