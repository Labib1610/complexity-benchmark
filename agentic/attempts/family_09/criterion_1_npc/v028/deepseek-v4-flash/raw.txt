We can solve this using a backtracking approach with pruning, since **n ≤ 46** is small enough for a meet-in-the-middle or branch-and-bound search.

Here’s the Python solution:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find an independent set of size k
    # Use backtracking with pruning
    chosen = []
    def backtrack(start, chosen_mask, chosen_list):
        if len(chosen_list) == k:
            return True
        if start >= n:
            return False
        # Prune: not enough remaining vertices
        if len(chosen_list) + (n - start) < k:
            return False
        # Try to include start
        if (chosen_mask & adj[start]) == 0:
            chosen_list.append(start + 1)
            if backtrack(start + 1, chosen_mask | (1 << start), chosen_list):
                return True
            chosen_list.pop()
        # Try to skip start
        if backtrack(start + 1, chosen_mask, chosen_list):
            return True
        return False

    chosen_list = []
    if backtrack(0, 0, chosen_list):
        print("YES")
        print(" ".join(map(str, chosen_list)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
