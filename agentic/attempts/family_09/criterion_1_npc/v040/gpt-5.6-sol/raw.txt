import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]

    original = [0] * n
    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        original[u] |= 1 << v
        original[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    full = (1 << n) - 1
    compatible = [0] * n
    for v in range(n):
        compatible[v] = full ^ original[v] ^ (1 << v)

    # Put high-degree vertices first to improve greedy coloring.
    vertices = sorted(
        range(n),
        key=lambda v: compatible[v].bit_count(),
        reverse=True
    )
    new_index = [0] * n
    for i, v in enumerate(vertices):
        new_index[v] = i

    adjacency = [0] * n
    for i, old_v in enumerate(vertices):
        mask = compatible[old_v]
        new_mask = 0
        while mask:
            bit = mask & -mask
            old_u = bit.bit_length() - 1
            new_mask |= 1 << new_index[old_u]
            mask ^= bit
        adjacency[i] = new_mask

    path = []

    def color_sort(candidates):
        order = []
        bounds = []
        remaining = candidates
        color = 0

        while remaining:
            color += 1
            available = remaining
            while available:
                bit = available & -available
                v = bit.bit_length() - 1
                order.append(v)
                bounds.append(color)
                remaining ^= bit
                available &= remaining & ~adjacency[v]

        return order, bounds

    def search(candidates):
        depth = len(path)
        if depth + candidates.bit_count() < k:
            return False

        order, bounds = color_sort(candidates)

        for i in range(len(order) - 1, -1, -1):
            if depth + bounds[i] < k:
                return False

            v = order[i]
            bit = 1 << v
            path.append(v)

            if len(path) == k:
                return True

            next_candidates = candidates & adjacency[v]
            if len(path) + next_candidates.bit_count() >= k:
                if search(next_candidates):
                    return True

            path.pop()
            candidates ^= bit

        return False

    if search(full):
        answer = [vertices[v] + 1 for v in path]
        print("YES")
        print(*answer)
    else:
        print("NO")

if __name__ == "__main__":
    main()
