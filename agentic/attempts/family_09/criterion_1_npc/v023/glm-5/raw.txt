import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    try:
        n = int(next(it))
        m = int(next(it))
        k = int(next(it))
    except StopIteration:
        return

    if k == 0:
        print("NO")
        return

    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)

    # Check if the set of candidate vertices can support a clique of size k in G_bar.
    # Independent set in G = Clique in complement graph G_bar.
    # G_bar has edge (u,v) iff u and v are NOT interfering.
    
    # We only need to find 1 clique.
    # Optimization: Degree in G_bar.
    # Filtering isolated vertices in G_bar (degree(n-1) in G) is possible but copying list is O(N).
    # Iterating and skipping is sufficient.

    best_clique = []

    # Since we need to find exactly 1 clique of size k, 
    # and N is small (46 max licensed bands for the multiplexer? No, input n is small for practical purposes.
    # Wait, problem says "regulatory body limits the maximum number of bands an operator can license to exactly 46".
    # This implies n <= 46.
    # If n can be large but we only care about the connected component or small degree?
    # Given problem type and limits, O(N * 2^(N/2)) or O(N^3) is acceptable for N<=46.
    # With brute force O(N^k), we can pass.
    # Let's use the Bron-Kerbosch algorithm with pivot and pruning for finding one maximum clique.
    
    sys.setrecursionlimit(2000)

    def bronk_pivot(R, P, X):
        nonlocal best_clique
        
        # Pruning: if even with all P added, we can't beat or match k
        if len(R) + len(P) < k:
            return
        if len(best_clique) >= k:
            return

        if not P and not X:
            if len(R) >= k:
                best_clique = list(R)
            return

        # Pivot selection
        # Choose u from P U X to maximize |P intersection N(u)|
        # Here N(u) in G_bar means non-adjacent in G.
        # We iterate P^c to implement pivot optimization:
        # For each node in P, if it is not neighbor of pivot, recurse.
        
        # Union P and X
        # pivot = next(iter(P)) if P else next(iter(X)) # Simple pivot
        # Let's use degree heuristic for potentially better pruning
        
        union = P | X
        pivot = -1
        max_neighbors = -1
        if union:
            # Pick pivot maximizing intersection with P
            # Iterating union is better than iterating all nodes if union is sparse
            for u_node in union:
                # Count neighbors in G_bar which are in P
                # Neighbors in G_bar are those NOT in adj[u_node] and != u_node
                # P is already potential candidates
                
                # Optimizing: counting manually
                # adj[u_node] contains non-neighbors in G_bar
                # Neighbors in G_bar in P = P - adj[u_node]
                cnt = 0
                # iterating set difference
                for v_node in P:
                    if v_node not in adj[u_node]:
                        cnt += 1
                
                if cnt > max_neighbors:
                    max_neighbors = cnt
                    pivot = u_node
            
            # Candidates for recursion are P - N_bar(pivot)
            # P - (Nodes not adjacent to pivot in G)
            # Which is P - (All nodes except pivot - adj[pivot])
            # = P AND (adj[pivot] U {pivot})
            # Candidates to check = P - (P AND (adj[pivot] U {pivot}))
            # Candidates = P - (adj[pivot]) because pivot not in P (pivot in P|X)
            
            candidates_to_check = P - adj[pivot]
        else:
            # Should be covered by not P and not X check, but safety
            candidates_to_check = P

        for v in candidates_to_check:
            new_R = R | {v}
            
            # New P = P intersection N_bar(v)
            # N_bar(v) = all nodes - adj[v] - {v}
            new_P = set()
            for u_node in P:
                if u_node != v and u_node not in adj[v]:
                    new_P.add(u_node)
            
            # New X = X intersection N_bar(v)
            new_X = set()
            for u_node in X:
                if u_node != v and u_node not in adj[v]:
                    new_X.add(u_node)
            
            bronk_pivot(new_R, new_P, new_X)
            if len(best_clique) >= k:
                return
            
            P.remove(v)
            X.add(v)

    # Initial candidates P = all vertices in G_bar
    # Vertices with degree < k-1 in G_bar cannot be in a clique of size k.
    # Degree in G_bar = n - 1 - degree in G.
    # Threshold: n - 1 - deg_G(v) >= k - 1 => n - k >= deg_G(v)
    
    MAX_DEG_G_ALLOWED = n - k
    
    initial_P = set()
    for v in range(1, n + 1):
        if len(adj[v]) <= MAX_DEG_G_ALLOWED:
            initial_P.add(v)

    bronk_pivot(set(), initial_P, set())

    if len(best_clique) >= k:
        print("YES")
        # Just print k of them
        out_res = list(best_clique)[:k]
        print(*(out_res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
