```python
import sys
from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    # Build adjacency list for interference graph
    adj = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
    
    # If k == 1, just pick any band
    if k == 1:
        print("YES")
        print(1)
        return
    
    # If k > n, impossible
    if k > n:
        print("NO")
        return
    
    # Try all combinations of k bands
    # This is feasible because n <= 46 and we can optimize
    # But given constraints, let's try a greedy approach or limited search if needed
    # However, since k can be up to n, and n up to 46, combinations can be too much
    # Let's try a different approach: build complement graph and find a clique of size k
    # Or, since m can be up to n*(n-1)/2, it might be small
    
    # Let's try greedy or backtracking for small k, else combinations
    
    # For larger inputs, we might need a more sophisticated algorithm
    # But let's proceed with combinations for now, with early termination
    
    # Actually, let's think differently: model as finding an independent set of size k
    # Which is equivalent to finding a clique of size k in the complement graph
    
    # Let's create the complement graph
    complement_adj = [set() for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(i + 1, n + 1):
            if j not in adj[i]:
                complement_adj[i].add(j)
                complement_adj[j].add(i)
    
    # Now we need to find a clique of size k in this complement graph
    # We'll use a recursive backtracking approach
    
    def find_clique(nodes, k, current_clique):
        if k == 0:
            return current_clique[:]
        if len(nodes) < k:
            return None
        
        # Pick a node and try including/excluding it
        # Optimization: pick node with highest degree in remaining nodes
        best_node = max(nodes, key=lambda x: len(complement_adj[x] & nodes))
        candidates = nodes & complement_adj[best_node]
        
        # Include best_node
        current_clique.append(best_node)
        result = find_clique(candidates, k - 1, current_clique)
        if result is not None:
            return result
        current_clique.pop()
        
        # Exclude best_node
        remaining_nodes = nodes - {best_node}
        return find_clique(remaining_nodes, k, current_clique)
    
    # Start with all nodes
    all_nodes = set(range(1, n + 1))
    clique = find_clique(all_nodes, k, [])
    
    if clique is not None:
        print("YES")
        print(' '.join(map(str, sorted(clique))))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
