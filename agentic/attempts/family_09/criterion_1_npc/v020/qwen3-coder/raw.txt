from itertools import combinations
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    # Read dangerous pairs
    dangerous = set()
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        if u > v:
            u, v = v, u
        dangerous.add((u, v))
        idx += 2
    
    # Generate all combinations of k materials
    materials = list(range(1, n+1))
    for combo in combinations(materials, k):
        # Check if any pair in this combination is dangerous
        valid = True
        for i in range(k):
            for j in range(i+1, k):
                u = combo[i]
                v = combo[j]
                if u > v:
                    u, v = v, u
                if (u, v) in dangerous:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            print("YES")
            print(' '.join(map(str, combo)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
