import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k == 1:
        print("YES")
        print(1)
        return

    if k > n:
        print("NO")
        return

    # 0-indexed vertices
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Complement graph adjacency
    all_mask = (1 << n) - 1
    comp_adj = [0] * n
    for i in range(n):
        comp_adj[i] = (all_mask ^ adj[i]) ^ (1 << i)

    # Order vertices by degree in complement graph (degeneracy-like ordering)
    degrees = [comp_adj[i].bit_count() for i in range(n)]
    order = sorted(range(n), key=lambda x: degrees[x])
    pos = [0] * n
    for p, v in enumerate(order):
        pos[v] = p

    # Reorder vertices so that order[i] becomes vertex i
    new_comp_adj = [0] * n
    for i in range(n):
        orig_u = order[i]
        mask = 0
        for j in range(n):
            orig_v = order[j]
            if (comp_adj[orig_u] >> orig_v) & 1:
                mask |= 1 << j
        new_comp_adj[i] = mask

    # Maximum clique in complement graph using Ostergard's approach
    found = None
    c = [0] * n  # c[i] stores max clique size in subgraph {i, i+1, ..., n-1}

    def search(r_mask, p_mask, size):
        nonlocal found
        if found is not None:
            return
        if size == k:
            found = r_mask
            return
        if size + p_mask.bit_count() < k:
            return

        while p_mask:
            if size + p_mask.bit_count() < k:
                return
            # Pick the lowest set bit
            v = (p_mask & -p_mask).bit_length() - 1
            if size + c[v] < k:
                return
            p_mask &= ~(1 << v)
            search(r_mask | (1 << v), p_mask & new_comp_adj[v], size + 1)
            if found is not None:
                return

    for i in range(n - 1, -1, -1):
        if found is not None:
            break
        # Candidates for vertex i are vertices j > i that are connected to i
        candidates = new_comp_adj[i] & ~((1 << (i + 1)) - 1)
        search(1 << i, candidates, 1)
        # Update c[i]
        c[i] = c[i + 1] if i + 1 < n else 0
        if found is not None:
            break
        # To accurately set c[i], we would need to know the max clique found starting at i.
        # But for decision problem (finding size k), c[i] = (c[i+1] + 1) is a safe upper bound:
        c[i] = min(n - i, (c[i + 1] + 1) if i + 1 < n else 1)

    if found is not None:
        result = []
        for i in range(n):
            if (found >> i) & 1:
                result.append(order[i] + 1)
        print("YES")
        print(*(result[:k]))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
