We can solve this by trying to find an independent set of size exactly `k` in the graph.  
Since `n` is at most 46, we can use a backtracking approach with pruning.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # If k is 1, always possible
    if k == 1:
        print("YES")
        print(1)
        return
    
    # Try to find independent set of size k
    # Use bitmask to represent candidate set
    # We'll do backtracking with pruning
    
    # Precompute for each vertex, the set of vertices that are not neighbors (including itself)
    non_neighbors = [0] * n
    for i in range(n):
        non_neighbors[i] = ((1 << n) - 1) ^ adj[i]
    
    # We'll try to build the set
    result = []
    
    def backtrack(start, chosen, mask):
        if len(chosen) == k:
            result.extend(chosen)
            return True
        # Prune: if not enough remaining vertices
        remaining = n - start
        if len(chosen) + remaining < k:
            return False
        for i in range(start, n):
            # If i is not adjacent to any chosen
            if (mask >> i) & 1 == 0:
                # Choose i
                chosen.append(i + 1)
                new_mask = mask | adj[i]
                if backtrack(i + 1, chosen, new_mask):
                    return True
                chosen.pop()
        return False
    
    if backtrack(0, [], 0):
        print("YES")
        print(" ".join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
