import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)
    
    # Greedy heuristic to find a large independent set
    def greedy_is():
        deg = [(bin(adj[i]).count('1'), i) for i in range(1, n + 1)]
        deg.sort()
        order = [x[1] for x in deg]
        forbidden = 0
        result = []
        for u in order:
            mask = 1 << (u - 1)
            if not (forbidden & mask):
                result.append(u)
                forbidden |= adj[u] | mask
        return result
    
    greedy = greedy_is()
    if len(greedy) >= k:
        print("YES")
        print(" ".join(map(str, greedy[:k])))
        return
    
    # Split and list algorithm
    half = n // 2
    size1 = half
    size2 = n - half
    
    # Precompute adjacency masks for each part
    adj1 = [0] * (1 << size1)
    adj2 = [0] * (1 << size2)
    
    for i in range(size1):
        adj1[1 << i] = adj[i + 1]
    for i in range(1, 1 << size1):
        lsb = i & -i
        adj1[i] = adj1[i ^ lsb] | adj1[lsb]
    
    for i in range(size2):
        adj2[1 << i] = adj[half + i + 1]
    for i in range(1, 1 << size2):
        lsb = i & -i
        adj2[i] = adj2[i ^ lsb] | adj2[lsb]
    
    # Precompute internal validity for each subset
    valid1 = [True] * (1 << size1)
    for i in range(1, 1 << size1):
        lsb = i & -i
        j = (lsb.bit_length() - 1)
        prev = i ^ lsb
        if not valid1[prev]:
            valid1[i] = False
        else:
            # Check if vertex j is adjacent to any vertex in prev
            if adj1[prev] & (1 << j):
                valid1[i] = False
    
    valid2 = [True] * (1 << size2)
    for i in range(1, 1 << size2):
        lsb = i & -i
        j = (lsb.bit_length() - 1)
        prev = i ^ lsb
        if not valid2[prev]:
            valid2[i] = False
        else:
            if adj2[prev] & (1 << j):
                valid2[i] = False
    
    # Group valid subsets of second part by size
    subsets_by_size = [[] for _ in range(size2 + 1)]
    for i in range(1 << size2):
        if valid2[i]:
            subsets_by_size[bin(i).count('1')].append(i)
    
    # For each valid subset of first part, try to find compatible subset in second part
    for mask1 in range(1 << size1):
        if not valid1[mask1]:
            continue
        cnt1 = bin(mask1).count('1')
        if cnt1 > k:
            continue
        need = k - cnt1
        if need > size2:
            continue
        
        # Forbidden vertices in part 2: neighbors of vertices in mask1
        # adj1[mask1] contains neighbors of all vertices in mask1
        # We need to extract which part 2 vertices are forbidden
        forbidden2 = 0
        neighbors = adj1[mask1]
        # Vertices in part 2 are numbered half+1 to n (indices half to n-1)
        # Check which of these are neighbors
        for i in range(size2):
            if neighbors & (1 << (half + i)):
                forbidden2 |= (1 << i)
        
        # We need a valid subset of part 2 of size 'need' that doesn't intersect forbidden2
        for mask2 in subsets_by_size[need]:
            if mask2 & forbidden2:
                continue
            # Found a solution
            result = []
            for i in range(size1):
                if mask1 & (1 << i):
                    result.append(i + 1)
            for i in range(size2):
                if mask2 & (1 << i):
                    result.append(half + i + 1)
            print("YES")
            print(" ".join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
