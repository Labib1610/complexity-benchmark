```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # If k is 1, always possible
    if k == 1:
        print("YES")
        print(1)
        return

    # Try to find an independent set of size k
    # Use recursive backtracking with pruning
    # We'll try to pick vertices in order

    # Precompute possible candidates for each vertex
    # We'll use a recursive function

    # To speed up, we can use a simple branch and bound:
    # At each step, we have a set of candidates (vertices not adjacent to chosen ones)
    # We pick the vertex with the smallest degree among candidates (greedy)

    chosen = []
    candidates = (1 << n) - 1  # all vertices available

    # For pruning: if remaining candidates + chosen < k, fail

    def backtrack(chosen_mask, candidates_mask, depth):
        if depth == k:
            return True
        # Prune: not enough candidates
        if bin(candidates_mask).count("1") < k - depth:
            return False
        # Choose next vertex: pick one with smallest degree among candidates
        # We'll iterate over candidates
        c = candidates_mask
        best_v = -1
        best_deg = n + 1
        while c:
            lsb = c & -c
            v = (lsb.bit_length() - 1)
            deg = bin(adj[v] & candidates_mask).count("1")
            if deg < best_deg:
                best_deg = deg
                best_v = v
            c -= lsb
        # Try including best_v
        chosen.append(best_v + 1)
        new_candidates = candidates_mask & ~adj[best_v] & ~(1 << best_v)
        if backtrack(chosen_mask | (1 << best_v), new_candidates, depth + 1):
            return True
        chosen.pop()
        # Try excluding best_v
        candidates_mask &= ~(1 << best_v)
        if backtrack(chosen_mask, candidates_mask, depth):
            return True
        return False

    if backtrack(0, candidates, 0):
        print("YES")
        print(" ".join(map(str, chosen)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
