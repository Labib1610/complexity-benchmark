import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    interference = [0] * n

    pos = 3
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        interference[u] |= 1 << v
        interference[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    # Build the complement graph. An independent set in the input graph
    # is a clique in this graph.
    complement_degree = [n - 1 - interference[v].bit_count() for v in range(n)]
    permutation = sorted(range(n), key=lambda v: complement_degree[v])
    adj = [0] * n

    for i in range(n):
        oi = permutation[i]
        mask = 0
        for j in range(n):
            if i != j and not ((interference[oi] >> permutation[j]) & 1):
                mask |= 1 << j
        adj[i] = mask

    # Iteratively remove vertices that cannot belong to a clique of size k.
    active = (1 << n) - 1
    while True:
        remove = 0
        x = active
        while x:
            b = x & -x
            v = b.bit_length() - 1
            if (adj[v] & active).bit_count() < k - 1:
                remove |= b
            x ^= b
        if not remove:
            break
        active &= ~remove

    if active.bit_count() < k:
        print("NO")
        return

    # A clique is contained in one connected component.
    components = []
    unseen = active
    while unseen:
        seed = unseen & -unseen
        unseen ^= seed
        frontier = seed
        component = 0

        while frontier:
            b = frontier & -frontier
            frontier ^= b
            v = b.bit_length() - 1
            component |= b

            new_vertices = adj[v] & unseen
            unseen &= ~new_vertices
            frontier |= new_vertices

        if component.bit_count() >= k:
            components.append(component)

    components.sort(key=int.bit_count, reverse=True)

    # Fast greedy attempts before exact branch-and-bound.
    for component in components:
        seeds = component
        while seeds:
            sb = seeds & -seeds
            seeds ^= sb
            s = sb.bit_length() - 1

            clique = [s]
            candidates = component & adj[s]

            while candidates and len(clique) < k:
                x = candidates
                best_v = -1
                best_degree = -1
                while x:
                    b = x & -x
                    v = b.bit_length() - 1
                    degree = (adj[v] & candidates).bit_count()
                    if degree > best_degree:
                        best_degree = degree
                        best_v = v
                    x ^= b

                clique.append(best_v)
                candidates &= adj[best_v]

            if len(clique) >= k:
                answer = [permutation[v] + 1 for v in clique[:k]]
                print("YES")
                print(*answer)
                return

    sys.setrecursionlimit(10000)
    path = []
    answer_mask = 0

    def expand(candidates, depth):
        nonlocal answer_mask

        if depth + candidates.bit_count() < k:
            return False

        order = []
        colors = []
        remaining = candidates
        color = 0

        # Greedy coloring gives an upper bound on any clique extension.
        while remaining:
            color += 1
            available = remaining
            while available:
                b = available & -available
                v = b.bit_length() - 1

                order.append(v)
                colors.append(color)

                remaining ^= b
                available ^= b
                available &= ~adj[v]

        current = candidates

        for i in range(len(order) - 1, -1, -1):
            if depth + colors[i] < k:
                return False

            v = order[i]
            bit = 1 << v
            path.append(v)

            if depth + 1 == k:
                mask = 0
                for u in path:
                    mask |= 1 << u
                answer_mask = mask
                return True

            next_candidates = current & adj[v]
            if depth + 1 + next_candidates.bit_count() >= k:
                if expand(next_candidates, depth + 1):
                    return True

            path.pop()
            current &= ~bit

        return False

    for component in components:
        path.clear()
        if expand(component, 0):
            answer = []
            x = answer_mask
            while x:
                b = x & -x
                v = b.bit_length() - 1
                answer.append(permutation[v] + 1)
                x ^= b

            print("YES")
            print(*answer)
            return

    print("NO")

if __name__ == "__main__":
    main()
