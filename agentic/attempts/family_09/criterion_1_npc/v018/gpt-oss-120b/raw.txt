import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1

    # recursive search for independent set of size k
    solution = []

    from functools import lru_cache

    @lru_cache(maxsize=None)
    def popcnt(x: int) -> int:
        return x.bit_count()

    def dfs(cand: int, cur: int, cur_sz: int):
        nonlocal solution
        if cur_sz == k:
            # extract vertices from cur
            sol = []
            bits = cur
            while bits:
                v = (bits & -bits).bit_length() - 1
                sol.append(v + 1)
                bits &= bits - 1
            solution = sol
            return True
        if cur_sz + popcnt(cand) < k:
            return False
        # choose a vertex (heuristic: vertex with max degree in cand)
        # simple: take first bit
        v = (cand & -cand).bit_length() - 1

        # branch: include v
        new_cand = cand & ~adj[v] & ~(1 << v)
        if dfs(new_cand, cur | (1 << v), cur_sz + 1):
            return True
        # branch: exclude v
        cand_without_v = cand & ~(1 << v)
        if dfs(cand_without_v, cur, cur_sz):
            return True
        return False

    dfs(ALL, 0, 0)

    if solution:
        print("YES")
        print(" ".join(map(str, solution)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
