import sys

def build_dp(q, adjacency):
    dp = bytearray(1 << q)
    base = 1
    for v in range(q):
        av = adjacency[v]
        offset = base
        for rest in range(base):
            without_v = dp[rest]
            with_v = dp[rest & av] + 1
            dp[offset + rest] = with_v if with_v > without_v else without_v
        base <<= 1
    return dp

def reconstruct(mask, need, adjacency, dp):
    result = 0
    while need:
        v = mask.bit_length() - 1
        bit = 1 << v
        rest = mask ^ bit
        if dp[rest] >= need:
            mask = rest
        else:
            result |= bit
            need -= 1
            mask = rest & adjacency[v]
    return result

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    friends = [0] * n

    pos = 3
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        friends[u] |= 1 << v
        friends[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    all_vertices = (1 << n) - 1
    nonfriends = [
        all_vertices ^ (1 << i) ^ friends[i]
        for i in range(n)
    ]

    a = n // 2
    b = n - a

    full_a = (1 << a) - 1
    full_b = (1 << b) - 1

    adj_a = [nonfriends[i] & full_a for i in range(a)]
    cross = [(nonfriends[i] >> a) & full_b for i in range(a)]
    adj_b = [
        (nonfriends[a + i] >> a) & full_b
        for i in range(b)
    ]

    dp_b = build_dp(b, adj_b)

    if dp_b[full_b] >= k:
        chosen_b = reconstruct(full_b, k, adj_b, dp_b)
        answer = [a + i + 1 for i in range(b) if chosen_b >> i & 1]
        print("YES")
        print(*answer)
        return

    dp_a = build_dp(a, adj_a)

    if dp_a[full_a] >= k:
        chosen_a = reconstruct(full_a, k, adj_a, dp_a)
        answer = [i + 1 for i in range(a) if chosen_a >> i & 1]
        print("YES")
        print(*answer)
        return

    if dp_a[full_a] + dp_b[full_b] < k:
        print("NO")
        return

    sys.setrecursionlimit(1000)

    def search(candidates, compatible_b, chosen_a, count_a):
        need = k - count_a

        if dp_b[compatible_b] >= need:
            chosen_b = reconstruct(compatible_b, need, adj_b, dp_b)
            return chosen_a, chosen_b

        if dp_a[candidates] + dp_b[compatible_b] < need:
            return None

        while candidates:
            v = candidates.bit_length() - 1
            bit = 1 << v
            candidates ^= bit

            next_candidates = candidates & adj_a[v]
            next_compatible = compatible_b & cross[v]

            if 1 + dp_a[next_candidates] + dp_b[next_compatible] >= need:
                result = search(
                    next_candidates,
                    next_compatible,
                    chosen_a | bit,
                    count_a + 1
                )
                if result is not None:
                    return result

            if dp_a[candidates] + dp_b[compatible_b] < need:
                break

        return None

    result = search(full_a, full_b, 0, 0)

    if result is None:
        print("NO")
        return

    chosen_a, chosen_b = result
    answer = [i + 1 for i in range(a) if chosen_a >> i & 1]
    answer.extend(a + i + 1 for i in range(b) if chosen_b >> i & 1)

    print("YES")
    print(*answer)

if __name__ == "__main__":
    main()
