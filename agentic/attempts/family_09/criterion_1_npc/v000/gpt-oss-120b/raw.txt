```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_vertices = (1 << n) - 1
    result = []

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: adj[x].bit_count(), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency according to new order
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for j in range(n):
            if adj[v] >> j & 1:
                mask |= 1 << pos[j]
        adj2[i] = mask

    target_k = k
    found = False

    def dfs(cand: int, cur: list):
        nonlocal found, result
        if found:
            return
        cur_len = len(cur)
        if cur_len == target_k:
            result = [order[i] + 1 for i in cur]  # convert back to 1‑based
            found = True
            return
        # prune if even taking all remaining cannot reach k
        if cur_len + cand.bit_count() < target_k:
            return
        # choose a vertex: pick the one with highest degree in cand
        # simple: take the least significant bit
        v = (cand & -cand).bit_length() - 1
        # branch 1: include v
        dfs(cand & ~adj2[v] & ~(1 << v), cur + [v])
        if found:
            return
        # branch 2: exclude v
        dfs(cand & ~(1 << v), cur)

    dfs(all_vertices, [])
    out = sys.stdout
    if found:
        out.write("YES\n")
        out.write(" ".join(map(str, result)) + "\n")
    else:
        out.write("NO\n")

if __name__ == "__main__":
    solve()
```
