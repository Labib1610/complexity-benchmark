We can solve this by trying to find an independent set of size exactly k in the graph where edges represent dangerous reactions.  
Since n ≤ 46, we can use a meet-in-the-middle or backtracking with pruning.  

Here’s the implementation:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find independent set of size k
    # Use recursive backtracking with pruning
    def backtrack(start, chosen, mask):
        if len(chosen) == k:
            return chosen[:]
        if start >= n:
            return None
        # Prune: if remaining vertices + chosen < k, impossible
        if n - start + len(chosen) < k:
            return None
        # Try including start if possible
        if (mask & adj[start]) == 0:
            chosen.append(start)
            res = backtrack(start + 1, chosen, mask | (1 << start))
            if res:
                return res
            chosen.pop()
        # Try excluding start
        return backtrack(start + 1, chosen, mask)

    result = backtrack(0, [], 0)
    if result is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(str(x + 1) for x in result))

if __name__ == "__main__":
    solve()
```
