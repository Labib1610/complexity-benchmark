```python
import sys
from itertools import combinations

def output_yes(vertices):
    print("YES")
    print(*vertices)
    sys.exit(0)

data = list(map(int, sys.stdin.buffer.read().split()))
n, m, k = data[0], data[1], data[2]

adj = [0] * n
p = 3
for _ in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2
    adj[u] |= 1 << v
    adj[v] |= 1 << u

if k == 1:
    output_yes([1])

if k <= 3:
    for comb in combinations(range(n), k):
        ok = True
        for i in range(k):
            for j in range(i + 1, k):
                if (adj[comb[i]] >> comb[j]) & 1:
                    ok = False
                    break
            if not ok:
                break
        if ok:
            output_yes([v + 1 for v in comb])
    print("NO")
    sys.exit(0)

excluded_count = n - k
if excluded_count <= 3:
    full = (1 << n) - 1
    for excluded in combinations(range(n), excluded_count):
        selected = full
        for v in excluded:
            selected ^= 1 << v

        ok = True
        mask = selected
        while mask:
            bit = mask & -mask
            v = bit.bit_length() - 1
            if adj[v] & selected:
                ok = False
                break
            mask ^= bit

        if ok:
            output_yes([i + 1 for i in range(n) if (selected >> i) & 1])

    print("NO")
    sys.exit(0)

degrees = [a.bit_count() for a in adj]

# Fast greedy attempts.
orders = [
    sorted(range(n), key=lambda v: (degrees[v], v)),
    sorted(range(n), key=lambda v: (-degrees[v], v)),
    list(range(n))
]

for vertex_order in orders:
    chosen_mask = 0
    chosen = []
    for v in vertex_order:
        if not (adj[v] & chosen_mask):
            chosen_mask |= 1 << v
            chosen.append(v + 1)
            if len(chosen) == k:
                output_yes(chosen)

# Put high-degree vertices in the enumerated half, which usually reduces
# the number of independent subsets visited.
order = sorted(range(n), key=lambda v: (-degrees[v], v))
position = [0] * n
for i, v in enumerate(order):
    position[v] = i

new_adj = [0] * n
for i, original_v in enumerate(order):
    mask = adj[original_v]
    converted = 0
    while mask:
        bit = mask & -mask
        original_u = bit.bit_length() - 1
        converted |= 1 << position[original_u]
        mask ^= bit
    new_adj[i] = converted

left_n = n // 2
right_n = n - left_n
left_full = (1 << left_n) - 1
right_full = (1 << right_n) - 1

left_adj = [0] * left_n
cross_forbidden = [0] * left_n
for i in range(left_n):
    left_adj[i] = new_adj[i] & left_full
    cross_forbidden[i] = (new_adj[i] >> left_n) & right_full

right_adj = [0] * right_n
for i in range(right_n):
    right_adj[i] = (new_adj[left_n + i] >> left_n) & right_full

# best[mask] is the maximum independent-set size in the right-half mask.
limit = 1 << right_n
best = bytearray(limit)
br = best
ra = right_adj

for mask in range(1, limit):
    bit = mask & -mask
    rest = mask - bit
    i = bit.bit_length() - 1

    without = br[rest]
    with_vertex = br[rest & ~ra[i]] + 1
    br[mask] = with_vertex if with_vertex > without else without

sys.setrecursionlimit(1000000)
answer_state = None

def search(candidates, selected, selected_size, allowed_right):
    global answer_state

    required = k - selected_size
    if required <= 0:
        answer_state = (selected, allowed_right, 0)
        return True

    right_best = br[allowed_right]
    if right_best >= required:
        answer_state = (selected, allowed_right, required)
        return True

    if selected_size + candidates.bit_count() + right_best < k:
        return False

    remaining = candidates
    while remaining:
        bit = remaining & -remaining
        remaining ^= bit
        i = bit.bit_length() - 1

        next_candidates = remaining & ~left_adj[i]
        next_allowed = allowed_right & ~cross_forbidden[i]

        if search(
            next_candidates,
            selected | bit,
            selected_size + 1,
            next_allowed
        ):
            return True

    return False

if not search(left_full, 0, 0, right_full):
    print("NO")
    sys.exit(0)

selected_left, allowed_right, required_right = answer_state
result = []

mask = selected_left
while mask:
    bit = mask & -mask
    i = bit.bit_length() - 1
    result.append(order[i] + 1)
    mask ^= bit

mask = allowed_right
need = required_right
while need > 0:
    bit = mask & -mask
    i = bit.bit_length() - 1
    rest = mask ^ bit

    if br[rest] >= need:
        mask = rest
    else:
        result.append(order[left_n + i] + 1)
        need -= 1
        mask = rest & ~right_adj[i]

print("YES")
print(*result[:k])
```
