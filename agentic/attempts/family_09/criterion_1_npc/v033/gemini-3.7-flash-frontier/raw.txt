import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k > n:
        print("NO")
        return

    if k == 1:
        print("YES")
        print(1)
        return

    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        idx += 2

    all_mask = (1 << n) - 1
    non_adj = [all_mask ^ adj[i] ^ (1 << i) for i in range(n)]

    max_val = [0] * (n + 1)
    ans = None

    for i in range(n - 1, -1, -1):
        cand = non_adj[i] & ~((1 << (i + 1)) - 1)
        best_len = [1]

        def search(cand_mask, cur):
            nonlocal ans
            if len(cur) > best_len[0]:
                best_len[0] = len(cur)
                if len(cur) >= k:
                    ans = cur[:k]
                    return True

            m_mask = cand_mask
            while m_mask > 0:
                p = m_mask & -m_mask
                v = p.bit_length() - 1
                if len(cur) + max_val[v] <= best_len[0]:
                    return False
                m_mask ^= p
                cur.append(v)
                next_cand = cand_mask & non_adj[v] & ~((1 << (v + 1)) - 1)
                if search(next_cand, cur):
                    return True
                cur.pop()
            return False

        cur = [i]
        if search(cand, cur) or best_len[0] >= k:
            break
        max_val[i] = max(max_val[i + 1], best_len[0])

    if ans is not None:
        print("YES")
        print(*(x + 1 for x in ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
