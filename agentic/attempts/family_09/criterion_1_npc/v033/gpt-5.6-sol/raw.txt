import sys
from array import array

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    adj = [0] * n

    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    # Try a greedy independent set first.
    remaining = (1 << n) - 1
    greedy = []
    while remaining:
        x = remaining
        best_v = -1
        best_deg = n + 1
        while x:
            b = x & -x
            v = b.bit_length() - 1
            d = (adj[v] & remaining).bit_count()
            if d < best_deg:
                best_deg = d
                best_v = v
            x ^= b
        greedy.append(best_v)
        remaining &= ~(adj[best_v] | (1 << best_v))

    if len(greedy) >= k:
        print("YES")
        print(*[v + 1 for v in greedy[:k]])
        return

    # Put high-degree vertices in the branching half.
    order = sorted(range(n), key=lambda v: adj[v].bit_count(), reverse=True)
    left_size = n // 2
    left_vertices = order[:left_size]
    right_vertices = order[left_size:]
    right_size = len(right_vertices)

    left_pos = {v: i for i, v in enumerate(left_vertices)}
    right_pos = {v: i for i, v in enumerate(right_vertices)}

    left_adj = [0] * left_size
    cross_adj = [0] * left_size
    for i, v in enumerate(left_vertices):
        lm = 0
        rm = 0
        x = adj[v]
        while x:
            b = x & -x
            u = b.bit_length() - 1
            j = left_pos.get(u)
            if j is not None:
                lm |= 1 << j
            else:
                j = right_pos.get(u)
                if j is not None:
                    rm |= 1 << j
            x ^= b
        left_adj[i] = lm
        cross_adj[i] = rm

    right_adj = [0] * right_size
    for i, v in enumerate(right_vertices):
        rm = 0
        x = adj[v]
        while x:
            b = x & -x
            u = b.bit_length() - 1
            j = right_pos.get(u)
            if j is not None:
                rm |= 1 << j
            x ^= b
        right_adj[i] = rm

    # best[mask] packs:
    #   low right_size bits: a maximum independent subset of mask
    #   higher bits: its size
    unit = 1 << right_size
    witness_mask = unit - 1
    total_masks = 1 << right_size
    best = array('I', [0]) * total_masks

    arr = best
    for i in range(right_size):
        bit = 1 << i
        compatible_lower = (bit - 1) & ~right_adj[i]
        base = unit + bit
        for low in range(bit):
            without = arr[low]
            with_vertex = arr[low & compatible_lower] + base
            arr[bit + low] = with_vertex if with_vertex > without else without

    full_right = (1 << right_size) - 1
    full_left = (1 << left_size) - 1

    root_best = best[full_right]
    if (root_best >> right_size) >= k:
        wm = root_best & witness_mask
        answer = []
        while wm and len(answer) < k:
            b = wm & -wm
            answer.append(right_vertices[b.bit_length() - 1] + 1)
            wm ^= b
        print("YES")
        print(*answer)
        return

    def clique_cover_bound(vertices):
        """Upper bound on an independent set via a partition into friendship cliques."""
        rem = vertices
        groups = 0
        while rem:
            groups += 1
            candidates = rem
            while candidates:
                b = candidates & -candidates
                i = b.bit_length() - 1
                rem ^= b
                candidates &= left_adj[i] & rem
        return groups

    result = None

    def dfs(available, compatible_right, chosen, chosen_count):
        nonlocal result

        packed = best[compatible_right]
        right_count = packed >> right_size
        needed = k - chosen_count

        if right_count >= needed:
            result = (chosen, packed & witness_mask, needed)
            return True

        if not available:
            return False

        simple_upper = chosen_count + available.bit_count() + right_count
        if simple_upper < k:
            return False

        if chosen_count + clique_cover_bound(available) + right_count < k:
            return False

        bit = available & -available
        i = bit.bit_length() - 1
        rest = available ^ bit

        # Include this left vertex.
        if dfs(
            rest & ~left_adj[i],
            compatible_right & ~cross_adj[i],
            chosen | bit,
            chosen_count + 1
        ):
            return True

        # Exclude this left vertex.
        if dfs(rest, compatible_right, chosen, chosen_count):
            return True

        return False

    dfs(full_left, full_right, 0, 0)

    if result is None:
        print("NO")
        return

    chosen_left, chosen_right, right_needed = result
    answer = []

    x = chosen_left
    while x:
        b = x & -x
        answer.append(left_vertices[b.bit_length() - 1] + 1)
        x ^= b

    x = chosen_right
    while x and right_needed:
        b = x & -x
        answer.append(right_vertices[b.bit_length() - 1] + 1)
        x ^= b
        right_needed -= 1

    print("YES")
    print(*answer[:k])

if __name__ == "__main__":
    main()
