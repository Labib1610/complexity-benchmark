import sys
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [set() for _ in range(n)]
    
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Simple cases
    if k == 1:
        print("YES")
        print(1)
        return
    
    if k == 0:
        print("YES")
        print()
        return
    
    # Heuristic: try to find k nodes such that count of neighbors among them is <= k - 2
    # These nodes definitely have a valid subset inside
    
    candidates = []
    for v in range(n):
        # Count how many of current candidates are neighbors of v
        cnt = 0
        for u in candidates:
            if u in adj[v]:
                cnt += 1
        # If adding v keeps count <= k - 2, add it
        if cnt + len(candidates) >= k:
            # Check if current candidates have solution
            pass
        candidates.append(v)
    
    result = None
    
    # Check each prefix of candidates
    current_candidates = []
    for v in candidates:
        current_candidates.append(v)
        if len(current_candidates) >= k:
            # Check if solution exists in current_candidates
            c = current_candidates
            
            # Build induced subgraph
            idx_map = {u: i for i, u in enumerate(c)}
            deg = [0] * len(c)
            edges = []
            for i, u in enumerate(c):
                for w in adj[u]:
                    if w in idx_map and w > u:
                        j = idx_map[w]
                        deg[i] += 1
                        deg[j] += 1
                        edges.append((i, j))
            
            # Use branch and bound for MIS
            global found_result
            found_result = None
            
            def mis_search(picked, available_pos, best_size):
                global found_result
                if found_result is not None:
                    return
                if len(picked) >= k:
                    found_result = picked[:k]
                    return
                if len(picked) + len(available_pos) < k:
                    return
                
                # Simple vertex ordering
                # Find vertex with minimum degree in remaining graph
                min_deg = float('inf')
                best_v = -1
                for pos in available_pos:
                    d = 0
                    for other in available_pos:
                        if other != pos and (min(pos, other), max(pos, other)) in [(min(e[0], e[1]), max(e[0], e[1])) for e in edges]:
                            d += 1
                    if d < min_deg:
                        min_deg = d
                        best_v = pos
                
                # Try including best_v
                v = best_v
                new_available = [u for u in available_pos if u != v and (min(v, u), max(v, u)) not in [(min(e[0], e[1]), max(e[0], e[1])) for e in edges]]
                picked.append(c[v])
                mis_search(picked, new_available, best_size)
                picked.pop()
                
                if found_result is not None:
                    return
                
                # Try excluding best_v
                new_available2 = [u for u in available_pos if u != v]
                mis_search(picked, new_available2, best_size)
            
            # Simpler approach: just try all combinations with pruning
            found_result = None
            
            def find_kmis(idx, picked, available_count):
                global found_result
                if found_result is not None:
                    return
                sz = len(picked)
                if sz == k:
                    found_result = picked[:]
                    return
                if idx >= len(c):
                    return
                # Pruning: need k - sz more vertices
                # available_count counts how many from idx..end can still be used
                if available_count < k - sz:
                    return
                
                remaining = len(c) - idx
                if remaining < k - sz:
                    return
                
                # Try including c[idx]
                can_include = True
                for u in picked:
                    if u in adj[c[idx]]:
                        can_include = False
                        break
                if can_include:
                    picked.append(c[idx])
                    # Count potential remaining vertices more precisely
                    # But for simplicity, just recurse
                    find_kmis(idx + 1, picked, remaining - 1)
                    picked.pop()
                
                if found_result is not None:
                    return
                
                # Try excluding c[idx]
                find_kmis(idx + 1, picked, remaining - 1)
            
            find_kmis(0, [], len(c))
            
            if found_result is not None:
                result = found_result
                break
    
    if result is None:
        # Try all subsets
        found_result = None
        
        def find_all(idx, picked):
            global found_result
            if found_result is not None:
                return
            if len(picked) == k:
                found_result = picked[:]
                return
            if idx >= n:
                return
            # Pruning
            if n - idx < k - len(picked):
                return
            
            # Try including vertex idx
            can_include = True
            for u in picked:
                if u in adj[idx]:
                    can_include = False
                    break
            if can_include:
                picked.append(idx)
                find_all(idx + 1, picked)
                picked.pop()
            
            if found_result is not None:
                return
            
            # Try excluding vertex idx
            find_all(idx + 1, picked)
        
        find_all(0, [])
        result = found_result
    
    if result is not None:
        print("YES")
        print(' '.join(str(x + 1) for x in result))
    else:
        print("NO")

solve()
