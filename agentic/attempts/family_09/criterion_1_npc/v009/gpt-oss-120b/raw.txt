```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    orig_adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        orig_adj[u] |= 1 << v
        orig_adj[v] |= 1 << u

    all_mask = (1 << n) - 1
    # complement graph adjacency (edges where there is NO original edge, except self)
    comp_adj = [0] * n
    for i in range(n):
        comp_adj[i] = (all_mask ^ (1 << i)) & ~orig_adj[i]

    target = k
    answer = None

    # popcount
    def popcnt(x: int) -> int:
        return x.bit_count()

    # recursive search for clique of size target in complement graph
    def bronk(R_mask: int, P_mask: int, X_mask: int, depth: int):
        nonlocal answer
        if answer is not None:
            return
        rsize = depth
        if rsize == target:
            # found
            ans = []
            for i in range(n):
                if (R_mask >> i) & 1:
                    ans.append(i + 1)
            answer = ans
            return
        # prune if even taking all P cannot reach target
        if rsize + popcnt(P_mask) < target:
            return
        # choose pivot u from P ∪ X with max degree in P
        union = P_mask | X_mask
        # simple heuristic: pick first vertex in union
        # better: choose vertex with max |P ∩ N(u)|
        max_deg = -1
        u = -1
        temp = union
        while temp:
            v = (temp & -temp).bit_length() - 1
            deg = popcnt(P_mask & comp_adj[v])
            if deg > max_deg:
                max_deg = deg
                u = v
            temp &= temp - 1
        # vertices to iterate: P \ N(u)
        candidates = P_mask & ~comp_adj[u] if u != -1 else P_mask
        while candidates:
            v = (candidates & -candidates).bit_length() - 1
            candidates &= candidates - 1
            bronk(R_mask | (1 << v),
                  P_mask & comp_adj[v],
                  X_mask & comp_adj[v],
                  depth + 1)
            if answer is not None:
                return
            P_mask &= ~(1 << v)
            X_mask |= (1 << v)

    # start with empty R, P = all vertices, X = empty
    bronk(0, all_mask, 0, 0)

    if answer is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, answer)))

if __name__ == "__main__":
    solve()
```
