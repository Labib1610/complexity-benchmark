We can model this as finding an independent set of size exactly k in a graph with up to 46 vertices.  
A simple backtracking with pruning will work because 46 is small enough if we use bitmasks and recursion.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # If k is 1, always possible
    if k == 1:
        print("YES")
        print(1)
        return

    # Try to find independent set of size k
    # We'll use recursive backtracking with bitmasks
    # candidates: bitmask of vertices we can still pick
    # chosen: list of chosen vertices

    def backtrack(start, chosen, candidates):
        if len(chosen) == k:
            return chosen[:]
        # Prune: not enough candidates left
        if len(chosen) + bin(candidates).count("1") < k:
            return None
        # Try picking next vertex
        # We iterate over set bits from start to n-1
        for v in range(start, n):
            if (candidates >> v) & 1:
                # choose v
                chosen.append(v)
                # new candidates: remove v and its neighbors
                new_candidates = candidates & ~(1 << v) & ~adj[v]
                res = backtrack(v + 1, chosen, new_candidates)
                if res is not None:
                    return res
                chosen.pop()
        return None

    # Initial candidates: all vertices
    all_mask = (1 << n) - 1
    result = backtrack(0, [], all_mask)

    if result is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(str(x + 1) for x in result))

if __name__ == "__main__":
    solve()
```
