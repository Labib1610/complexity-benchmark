import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k == 1:
        print("YES")
        print(1)
        return

    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    result = None

    def bron_kerbosch(R, P, X):
        nonlocal result
        if result is not None:
            return
        if len(R) == k:
            result = list(R)
            return
        if len(R) + P.bit_count() < k:
            return

        # Choose pivot u in P | X that minimizes |P & (adj[u] | {u if u in P})|
        PX = P | X
        pivot = -1
        min_branches = n + 2

        # Iterate through vertices in P | X to find best pivot
        temp = PX
        while temp:
            lsb = temp & -temp
            u = lsb.bit_length() - 1
            temp ^= lsb

            # Branch count if u is pivot
            branches = (P & adj[u]).bit_count() + ((P >> u) & 1)
            if branches < min_branches:
                min_branches = branches
                pivot = u
                if branches == 0:
                    break

        # Branching set: ({pivot} if pivot in P else empty) | (P & adj[pivot])
        branch_set = P & adj[pivot]
        if (P >> pivot) & 1:
            branch_set |= 1 << pivot

        temp = branch_set
        while temp:
            lsb = temp & -temp
            v = lsb.bit_length() - 1
            temp ^= lsb

            v_mask = 1 << v
            R.append(v + 1)
            bron_kerbosch(R, P & ~adj[v] & ~v_mask, X & ~adj[v] & ~v_mask)
            R.pop()
            if result is not None:
                return

            P ^= v_mask
            X |= v_mask

    initial_P = (1 << n) - 1
    bron_kerbosch([], initial_P, 0)

    if result is not None:
        print("YES")
        print(*(result[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    main()
