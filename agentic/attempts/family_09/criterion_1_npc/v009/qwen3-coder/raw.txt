from itertools import combinations

def main():
    n, m, k = map(int, input().split())
    
    # Read edges and build adjacency set for quick lookup
    edges = set()
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-based indexing
        v -= 1
        edges.add((u, v))
        edges.add((v, u))  # Since friendship is bidirectional
    
    # Generate all combinations of k users
    users = list(range(n))
    for combo in combinations(users, k):
        # Check if any two users in this combination are friends
        valid = True
        for i in range(k):
            for j in range(i + 1, k):
                u, v = combo[i], combo[j]
                if (u, v) in edges:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            # Found a valid all-strangers group
            print("YES")
            print(" ".join(str(x + 1) for x in combo))  # Convert back to 1-based indexing
            return
    
    print("NO")

if __name__ == "__main__":
    main()
