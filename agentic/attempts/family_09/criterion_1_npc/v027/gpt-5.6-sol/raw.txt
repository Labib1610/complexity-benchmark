import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]
    conflict = [0] * n
    p = 3
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        conflict[u] |= 1 << v
        conflict[v] |= 1 << u

    # Fast greedy attempts for easy instances.
    orders = [
        range(n),
        sorted(range(n), key=lambda v: conflict[v].bit_count()),
        sorted(range(n), key=lambda v: conflict[v].bit_count(), reverse=True),
    ]
    for order in orders:
        chosen_mask = 0
        chosen = []
        for v in order:
            if conflict[v] & chosen_mask == 0:
                chosen_mask |= 1 << v
                chosen.append(v + 1)
                if len(chosen) == k:
                    print("YES")
                    print(*chosen)
                    return

    right_size = n // 2
    left_size = n - right_size

    full_right = (1 << right_size) - 1
    full_left = (1 << left_size) - 1

    # Compatibility masks inside the right half.
    right_compat = [0] * right_size
    for i in range(right_size):
        original = left_size + i
        right_conflicts = (conflict[original] >> left_size) & full_right
        right_compat[i] = full_right & ~(right_conflicts | (1 << i))

    # dp[mask] = maximum independent-set size contained in mask.
    dp = bytearray(1 << right_size)
    base = 1
    for i in range(right_size):
        comp = right_compat[i]
        end = base << 1
        for x in range(base):
            without_i = dp[x]
            with_i = dp[x & comp] + 1
            dp[base + x] = with_i if with_i > without_i else without_i
        base = end

    left_compat = [0] * left_size
    cross_compat = [0] * left_size

    for i in range(left_size):
        left_conflicts = conflict[i] & full_left
        left_compat[i] = full_left & ~(left_conflicts | (1 << i))
        right_conflicts = (conflict[i] >> left_size) & full_right
        cross_compat[i] = full_right & ~right_conflicts

    selected_left = []
    answer_left = None
    answer_right_mask = 0

    sys.setrecursionlimit(10000)

    def search(candidates, allowed_right, count):
        nonlocal answer_left, answer_right_mask

        right_best = dp[allowed_right]
        if count + right_best >= k:
            answer_left = selected_left.copy()
            answer_right_mask = allowed_right
            return True

        while candidates:
            if count + candidates.bit_count() + right_best < k:
                return False

            bit = candidates & -candidates
            candidates ^= bit
            i = bit.bit_length() - 1

            selected_left.append(i)
            next_candidates = candidates & left_compat[i]
            next_allowed = allowed_right & cross_compat[i]

            if search(next_candidates, next_allowed, count + 1):
                return True

            selected_left.pop()

        return False

    if not search(full_left, full_right, 0):
        print("NO")
        return

    need = k - len(answer_left)
    mask = answer_right_mask
    selected_right = []

    while need > 0:
        i = mask.bit_length() - 1
        bit = 1 << i
        without_i = mask ^ bit

        if dp[without_i] >= need:
            mask = without_i
        else:
            selected_right.append(left_size + i + 1)
            mask = without_i & right_compat[i]
            need -= 1

    answer = [i + 1 for i in answer_left] + selected_right
    print("YES")
    print(*answer)

if __name__ == "__main__":
    main()
