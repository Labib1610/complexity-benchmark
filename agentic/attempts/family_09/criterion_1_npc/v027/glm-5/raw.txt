import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    if k == 0:
        print("YES")
        print()
        return
    
    if k == 1:
        print("YES")
        print(1)
        return
    
    # Build complement graph (non-edges)
    comp = [0] * (n + 1)
    for i in range(1, n + 1):
        comp[i] = ((1 << (n + 1)) - 1) ^ adj[i] ^ (1 << i)
        # restrict to vertices 1..n
        comp[i] &= (1 << (n + 1)) - 2
    
    # Find a clique of size k in complement graph
    # Using Bron-Kerbosch with pivoting and early termination
    
    best_clique = []
    
    def bron_kerbosch(R, P, X):
        nonlocal best_clique
        if len(best_clique) >= k:
            return
        if bin(P).count('1') + bin(R).count('1') < k:
            return
        if P == 0 and X == 0:
            sz = bin(R).count('1')
            if sz > len(best_clique):
                best_clique = [i for i in range(1, n+1) if R & (1 << i)]
            return
        # Choose pivot
        combined = P | X
        pivot = 0
        max_deg = -1
        temp = combined
        while temp:
            v = temp & (-temp)
            temp ^= v
            v_idx = v.bit_length() - 1
            deg = bin(comp[v_idx] & P).count('1')
            if deg > max_deg:
                max_deg = deg
                pivot = v_idx
        # Candidates = P \ N(pivot)
        pivot_neighbors = comp[pivot] if pivot else 0
        candidates = P & ~pivot_neighbors
        temp = candidates
        while temp:
            v = temp & (-temp)
            temp ^= v
            v_idx = v.bit_length() - 1
            newR = R | v
            if bin(newR).count('1') >= k:
                best_clique = [i for i in range(1, n+1) if newR & (1 << i)]
                return
            bron_kerbosch(newR, P & comp[v_idx], X & comp[v_idx])
            if len(best_clique) >= k:
                return
            P ^= v
            X |= v
    
    P = 0
    for i in range(1, n+1):
        P |= 1 << i
    
    bron_kerbosch(0, P, 0)
    
    if len(best_clique) >= k:
        print("YES")
        print(' '.join(map(str, best_clique[:k])))
    else:
        print("NO")

solve()
