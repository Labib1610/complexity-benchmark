We can solve this by trying to build an independent set of size exactly `k` in the graph.  
Since `n` is at most 46, a brute-force over all subsets is too large, but we can use a backtracking approach with pruning.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find an independent set of size k
    # Use backtracking with pruning
    chosen = []
    def dfs(start, chosen_set, count):
        if count == k:
            return True
        # Try to add vertices from start to n-1
        for v in range(start, n):
            # Check if v is not adjacent to any chosen
            if (chosen_set & adj[v]) == 0:
                chosen.append(v)
                if dfs(v + 1, chosen_set | (1 << v), count + 1):
                    return True
                chosen.pop()
        return False

    if dfs(0, 0, 0):
        print("YES")
        print(" ".join(str(x + 1) for x in chosen))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
