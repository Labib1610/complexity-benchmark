import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]

    if k == 1:
        print("YES")
        print(1)
        return

    original = [0] * n
    pos = 3
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        original[u] |= 1 << v
        original[v] |= 1 << u

    full = (1 << n) - 1
    clique_adj = [
        full & ~(original[v] | (1 << v))
        for v in range(n)
    ]

    # Remove vertices that cannot belong to a clique of size k.
    active = full
    degree = [(clique_adj[v] & active).bit_count() for v in range(n)]
    threshold = k - 1
    queue = [v for v in range(n) if degree[v] < threshold]

    while queue:
        v = queue.pop()
        vb = 1 << v
        if not (active & vb) or degree[v] >= threshold:
            continue

        active ^= vb
        neighbors = clique_adj[v] & active
        while neighbors:
            ub = neighbors & -neighbors
            neighbors ^= ub
            u = ub.bit_length() - 1
            degree[u] -= 1
            if degree[u] == threshold - 1:
                queue.append(u)

    if active.bit_count() < k:
        print("NO")
        return

    vertices = [v for v in range(n) if active >> v & 1]
    vertices.sort(key=lambda v: (clique_adj[v] & active).bit_count())

    size = len(vertices)
    adj = [0] * size
    for i, old_v in enumerate(vertices):
        mask = 0
        old_neighbors = clique_adj[old_v]
        for j, old_u in enumerate(vertices):
            if old_neighbors >> old_u & 1:
                mask |= 1 << j
        adj[i] = mask

    all_candidates = (1 << size) - 1

    # Fast greedy attempts, one from each possible first vertex.
    starts = sorted(range(size), key=lambda v: adj[v].bit_count(), reverse=True)
    for start in starts:
        chosen = [start]
        candidates = all_candidates & adj[start]

        while candidates and len(chosen) < k:
            best = -1
            best_degree = -1
            scan = candidates
            while scan:
                vb = scan & -scan
                scan ^= vb
                v = vb.bit_length() - 1
                d = (adj[v] & candidates).bit_count()
                if d > best_degree:
                    best_degree = d
                    best = v

            chosen.append(best)
            candidates &= adj[best]

        if len(chosen) == k:
            print("YES")
            print(*[vertices[v] + 1 for v in chosen])
            return

    stack = []

    def search(candidates, depth):
        if depth + candidates.bit_count() < k:
            return False

        order = []
        colors = []
        remaining = candidates
        color = 0

        # Greedy coloring gives an upper bound on any clique.
        while remaining:
            color += 1
            available = remaining
            while available:
                vb = available & -available
                v = vb.bit_length() - 1

                remaining ^= vb
                order.append(v)
                colors.append(color)

                available &= remaining & ~adj[v]

        for i in range(len(order) - 1, -1, -1):
            if depth + colors[i] < k:
                return False

            v = order[i]
            vb = 1 << v
            stack.append(v)

            if depth + 1 == k:
                return True

            next_candidates = candidates & adj[v]
            if depth + 1 + next_candidates.bit_count() >= k:
                if search(next_candidates, depth + 1):
                    return True

            stack.pop()
            candidates ^= vb

        return False

    if search(all_candidates, 0):
        print("YES")
        print(*[vertices[v] + 1 for v in stack[:k]])
    else:
        print("NO")


if __name__ == "__main__":
    main()
