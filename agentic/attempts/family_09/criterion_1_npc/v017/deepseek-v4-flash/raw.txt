We can solve this by trying to find an independent set of size exactly `k` in the graph.  
Since `n ≤ 46`, we can use a backtracking approach with pruning, or a bitmask-based search.

Here’s the Python solution:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find an independent set of size k
    # Use backtracking with pruning
    chosen = []
    def backtrack(start, chosen_mask, chosen_count):
        if chosen_count == k:
            return True
        # Try candidates from start to n-1
        for i in range(start, n):
            # If i is not connected to any chosen
            if (adj[i] & chosen_mask) == 0:
                chosen.append(i + 1)
                if backtrack(i + 1, chosen_mask | (1 << i), chosen_count + 1):
                    return True
                chosen.pop()
        return False

    if backtrack(0, 0, 0):
        print("YES")
        print(" ".join(map(str, chosen)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
