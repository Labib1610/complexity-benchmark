import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_mask = (1 << n) - 1
    chosen = []

    from functools import lru_cache

    @lru_cache(maxsize=None)
    def popcnt(x):
        return x.bit_count()

    def dfs(mask: int, size: int) -> bool:
        # success if we already have k vertices
        if size == k:
            return True
        # not enough vertices left
        if size + popcnt(mask) < k:
            return False
        # pick a vertex (least significant bit)
        v = (mask & -mask).bit_length() - 1

        # Try to include v
        if not (adj[v] & chosen_mask):
            # update masks
            new_mask = mask & ~adj[v] & ~(1 << v)
            chosen.append(v)
            nonlocal chosen_mask
            chosen_mask |= 1 << v
            if dfs(new_mask, size + 1):
                return True
            # backtrack
            chosen_mask &= ~(1 << v)
            chosen.pop()

        # Exclude v
        if dfs(mask & ~(1 << v), size):
            return True

        return False

    # mask of currently chosen vertices for quick conflict check
    chosen_mask = 0
    if dfs(all_mask, 0):
        print("YES")
        # convert to 1‑based indices
        print(' '.join(str(v + 1) for v in chosen))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
