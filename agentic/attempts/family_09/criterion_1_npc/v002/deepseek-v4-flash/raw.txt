We can solve this by searching for an independent set of size exactly `k` in a graph with up to 46 vertices.  
A simple backtracking with pruning works well for this size.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # Try to find an independent set of size k
    # We'll use backtracking with bitmasks
    # candidates: vertices that can still be added
    # current: chosen vertices so far
    
    # Precompute possible candidates for each state
    # Use recursive search with pruning
    
    chosen = []
    
    def backtrack(start, candidates_mask, depth):
        if depth == k:
            return True
        # Prune: not enough candidates left
        if bin(candidates_mask).count("1") < k - depth:
            return False
        # Try vertices in increasing order
        v = start
        while v < n:
            if (candidates_mask >> v) & 1:
                chosen.append(v)
                # New candidates: remove v and its neighbors
                new_candidates = candidates_mask & ~(1 << v) & ~adj[v]
                if backtrack(v + 1, new_candidates, depth + 1):
                    return True
                chosen.pop()
            v += 1
        return False
    
    initial_candidates = (1 << n) - 1
    if backtrack(0, initial_candidates, 0):
        print("YES")
        print(" ".join(str(x + 1) for x in chosen))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
