import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k == 1:
        print("YES")
        print(1)
        return

    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        idx += 2

    all_mask = (1 << n) - 1
    non_adj = [all_mask ^ adj[i] ^ (1 << i) for i in range(n)]

    max_is = [0] * (n + 1)
    best_set = []

    target_size = 0

    def dfs(candidates, current):
        if len(current) == target_size:
            return current
        c = candidates
        while c:
            if len(current) + c.bit_count() < target_size:
                return None
            v = (c & -c).bit_length() - 1
            if len(current) + 1 + max_is[v] < target_size:
                return None
            c &= c - 1
            res = dfs(candidates & non_adj[v], current + [v])
            if res is not None:
                return res
        return None

    for i in range(n - 1, -1, -1):
        target_size = max_is[i + 1] + 1
        candidates = non_adj[i] & ~((1 << (i + 1)) - 1)
        res = dfs(candidates, [i])
        if res is not None:
            max_is[i] = target_size
            best_set = res
            if max_is[i] >= k:
                print("YES")
                print(*(v + 1 for v in best_set[:k]))
                return
        else:
            max_is[i] = max_is[i + 1]

    print("NO")


if __name__ == "__main__":
    solve()
