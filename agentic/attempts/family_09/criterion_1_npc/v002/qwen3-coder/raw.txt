from itertools import combinations
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    # Build adjacency list for conflicts
    conflicts = [set() for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        conflicts[u].add(v)
        conflicts[v].add(u)
        idx += 2
    
    # If k == 1, always possible
    if k == 1:
        print("YES")
        print(1)
        return
    
    # Generate all combinations of k substances
    substances = list(range(1, n + 1))
    
    # For efficiency, use meet-in-the-middle for larger n
    if n <= 25:
        # Brute-force with combinations
        for combo in combinations(substances, k):
            valid = True
            combo_list = list(combo)
            for i in range(len(combo_list)):
                for j in range(i + 1, len(combo_list)):
                    if combo_list[j] in conflicts[combo_list[i]]:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                print("YES")
                print(' '.join(map(str, combo)))
                return
    else:
        # Meet-in-the-middle approach
        mid = n // 2
        left_subs = list(range(1, mid + 1))
        right_subs = list(range(mid + 1, n + 1))
        
        # Generate all subsets of left and right
        from collections import defaultdict
        
        def get_subsets(lst):
            subsets = []
            for i in range(1 << len(lst)):
                subset = [lst[j] for j in range(len(lst)) if (i & (1 << j))]
                subsets.append(subset)
            return subsets
        
        left_subsets = get_subsets(left_subs)
        right_subsets = get_subsets(right_subs)
        
        # Map independent sets from left
        left_map = defaultdict(list)
        for s in left_subsets:
            if not s:
                continue
            valid = True
            for i in range(len(s)):
                for j in range(i+1, len(s)):
                    if s[j] in conflicts[s[i]]:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                left_map[len(s)].append(set(s))
        
        # Map independent sets from right
        right_map = defaultdict(list)
        for s in right_subsets:
            if not s:
                continue
            valid = True
            for i in range(len(s)):
                for j in range(i+1, len(s)):
                    if s[j] in conflicts[s[i]]:
                        valid = False
                        break
                if not valid:
                    break
            if valid:
                right_map[len(s)].append(set(s))
        
        # Check for combinations
        for left_size in range(1, k):
            right_size = k - left_size
            if left_size not in left_map or right_size not in right_map:
                continue
            for left_set in left_map[left_size]:
                for right_set in right_map[right_size]:
                    # Check if there's conflict between left and right
                    conflict_found = False
                    for u in left_set:
                        for v in right_set:
                            if v in conflicts[u]:
                                conflict_found = True
                                break
                        if conflict_found:
                            break
                    if not conflict_found:
                        result = sorted(list(left_set) + list(right_set))
                        print("YES")
                        print(' '.join(map(str, result)))
                        return
    
    print("NO")

if __name__ == "__main__":
    main()
