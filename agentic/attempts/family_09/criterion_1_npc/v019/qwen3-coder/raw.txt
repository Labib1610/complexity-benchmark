from itertools import combinations

def main():
    n, m, k = map(int, input().split())
    
    # Read the conflicting pairs
    conflicts = set()
    for _ in range(m):
        u, v = map(int, input().split())
        if u > v:
            u, v = v, u
        conflicts.add((u, v))
    
    # Generate all possible combinations of k modules
    modules = list(range(1, n + 1))
    
    for combo in combinations(modules, k):
        # Check if this combination has any conflicts
        valid = True
        for i in range(k):
            for j in range(i + 1, k):
                u, v = combo[i], combo[j]
                if u > v:
                    u, v = v, u
                if (u, v) in conflicts:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            print("YES")
            print(*combo)
            return
    
    print("NO")

if __name__ == "__main__":
    main()
