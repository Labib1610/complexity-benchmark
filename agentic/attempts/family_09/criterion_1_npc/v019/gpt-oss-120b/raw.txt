import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1

    # order vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency according to new order
    adj2 = [0] * n
    for i, v in enumerate(order):
        mask = 0
        for u in range(n):
            if (adj[v] >> u) & 1:
                mask |= 1 << pos[u]
        adj2[i] = mask

    # initial candidate set: all vertices (in ordered indices)
    candidates = (1 << n) - 1
    solution_mask = 0
    found = False

    # popcount shortcut
    popcnt = int.bit_count

    def dfs(cur_mask: int, cand_mask: int, size: int) -> bool:
        nonlocal solution_mask, found
        if size == k:
            solution_mask = cur_mask
            return True
        if size + popcnt(cand_mask) < k:
            return False
        # choose a vertex with maximum degree in cand_mask (heuristic)
        # simple: pick first vertex
        v = (cand_mask & -cand_mask).bit_length() - 1  # least set bit index
        # branch: include v
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        if dfs(cur_mask | (1 << v), new_cand, size + 1):
            return True
        # branch: exclude v
        cand_mask &= ~(1 << v)
        if dfs(cur_mask, cand_mask, size):
            return True
        return False

    if dfs(0, candidates, 0):
        # translate solution_mask back to original vertex numbers
        res = []
        for i in range(n):
            if (solution_mask >> i) & 1:
                res.append(order[i] + 1)
        print("YES")
        print(" ".join(map(str, res)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
