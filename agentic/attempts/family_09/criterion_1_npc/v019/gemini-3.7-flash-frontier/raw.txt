import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k == 1:
        print("YES")
        print(1)
        return

    conflict = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        conflict[u] |= 1 << v
        conflict[v] |= 1 << u

    full = (1 << n) - 1
    adj = [0] * n
    for i in range(n):
        adj[i] = full ^ conflict[i] ^ (1 << i)

    c = [0] * n
    max_so_far = 0

    def clique_search(cand, size):
        nonlocal max_so_far
        if size > max_so_far:
            max_so_far = size
        if max_so_far >= k:
            return True
        while cand:
            p = (cand & -cand).bit_length() - 1
            if size + c[p] <= max_so_far:
                return False
            cand &= cand - 1
            if clique_search(cand & adj[p], size + 1):
                return True
        return False

    for i in range(n - 1, -1, -1):
        cand = adj[i] & ~((1 << (i + 1)) - 1)
        clique_search(cand, 1)
        c[i] = max_so_far
        if max_so_far >= k:
            break

    if max_so_far < k:
        print("NO")
        return

    def get_clique(cand, cur):
        if len(cur) == k:
            return cur
        while cand:
            p = (cand & -cand).bit_length() - 1
            if len(cur) + c[p] < k:
                return None
            cand &= cand - 1
            cur.append(p + 1)
            res = get_clique(cand & adj[p], cur)
            if res is not None:
                return res
            cur.pop()
        return None

    solution = None
    for i in range(n):
        cand = adj[i] & ~((1 << (i + 1)) - 1)
        solution = get_clique(cand, [i + 1])
        if solution:
            break

    print("YES")
    print(*(solution))


if __name__ == "__main__":
    solve()
