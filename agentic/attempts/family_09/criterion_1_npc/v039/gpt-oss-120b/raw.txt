```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # vertices as bits 0..n-1
    all_vertices = (1 << n) - 1

    # order vertices by degree (small to large) to improve pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"))
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency in ordered space
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for j in range(n):
            if adj[v] >> order[j] & 1:
                mask |= 1 << j
        adj2[i] = mask

    # recursive search
    from functools import lru_cache

    @lru_cache(maxsize=None)
    def popcnt(x: int) -> int:
        return x.bit_count()

    def dfs(cur_mask: int, cand_mask: int, need: int):
        """return mask of independent set of size need extending cur_mask, or None"""
        if need == 0:
            return cur_mask
        if popcnt(cand_mask) < need:
            return None
        # choose a vertex (heuristic: smallest degree)
        # pick first vertex in cand_mask
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        new_cur = cur_mask | (1 << v)
        new_cand = cand_mask & ~adj2[v] & ~(1 << v)
        res = dfs(new_cur, new_cand, need - 1)
        if res is not None:
            return res
        # branch: exclude v
        cand_mask_without_v = cand_mask & ~(1 << v)
        return dfs(cur_mask, cand_mask_without_v, need)

    result_mask = dfs(0, all_vertices, k)
    if result_mask is None:
        print("NO")
    else:
        # translate back to original numbering
        ans = []
        for i in range(n):
            if result_mask >> i & 1:
                ans.append(str(order[i] + 1))
        print("YES")
        print(" ".join(ans))

if __name__ == "__main__":
    solve()
```
