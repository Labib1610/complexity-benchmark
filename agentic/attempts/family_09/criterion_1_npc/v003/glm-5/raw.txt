import sys
input = sys.stdin.readline

def solve():
    n, m, k = map(int, input().split())
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Greedy algorithm: repeatedly pick vertex with minimum degree in remaining graph
    # This is a standard heuristic for finding large independent sets
    # For n <= 46, this should be sufficient
    
    remaining = set(range(n))
    result = []
    
    while remaining and len(result) < k:
        # Find vertex with minimum degree in the remaining graph
        best = -1
        best_deg = n + 1
        for v in remaining:
            deg = len(adj[v] & remaining)
            if deg < best_deg:
                best_deg = deg
                best = v
                if deg == 0:
                    break
        
        # Add best to result
        result.append(best + 1)
        remaining.remove(best)
        
        # Remove all neighbors of best from remaining
        to_remove = adj[best] & remaining
        remaining -= to_remove
    
    if len(result) >= k:
        print("YES")
        print(' '.join(map(str, result[:k])))
    else:
        print("NO")

solve()
