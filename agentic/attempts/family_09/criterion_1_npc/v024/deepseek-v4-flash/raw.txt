We can solve this by trying to find an independent set of size exactly `k` in the conflict graph.  
Since `n ≤ 46`, we can use a backtracking approach with pruning, or a meet-in-the-middle / bitmask DP.

Here’s the Python solution:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to find an independent set of size k
    # Use backtracking with pruning
    ans = []

    def backtrack(start, chosen, mask):
        nonlocal ans
        if len(chosen) == k:
            ans = chosen[:]
            return True
        if start >= n:
            return False
        # Prune: not enough remaining vertices
        if len(chosen) + (n - start) < k:
            return False

        # Try to include start
        if (mask & (1 << start)) == 0:
            # Check if start conflicts with chosen
            conflict = False
            for v in chosen:
                if (adj[start] >> v) & 1:
                    conflict = True
                    break
            if not conflict:
                chosen.append(start)
                if backtrack(start + 1, chosen, mask | (1 << start)):
                    return True
                chosen.pop()

        # Try to exclude start
        if backtrack(start + 1, chosen, mask):
            return True
        return False

    if backtrack(0, [], 0):
        print("YES")
        print(" ".join(str(x + 1) for x in ans))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
