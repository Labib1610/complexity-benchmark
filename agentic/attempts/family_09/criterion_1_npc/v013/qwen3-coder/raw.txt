from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    edges = []
    index = 3
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Create adjacency matrix
    adj = [[False]*n for _ in range(n)]
    for u, v in edges:
        adj[u][v] = True
        adj[v][u] = True
    
    # Check all combinations of k nodes
    for comb in combinations(range(n), k):
        valid = True
        for i in range(k):
            for j in range(i+1, k):
                if adj[comb[i]][comb[j]]:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            print("YES")
            print(' '.join(str(x+1) for x in comb))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
