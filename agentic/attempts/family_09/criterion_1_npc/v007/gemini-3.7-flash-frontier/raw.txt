import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        idx += 2

    def solve_deg2(cand_mask):
        rem = cand_mask
        comp_nodes = []
        visited = 0
        nodes = []
        for i in range(n):
            if (rem >> i) & 1:
                nodes.append(i)

        indep_set = []

        for v in nodes:
            if (visited >> v) & 1:
                continue

            # BFS / DFS to get component
            comp = []
            queue = [v]
            visited |= 1 << v
            for u in queue:
                comp.append(u)
                nbrs = adj[u] & cand_mask
                for w in nodes:
                    if ((nbrs >> w) & 1) and not ((visited >> w) & 1):
                        visited |= 1 << w
                        queue.append(w)

            # Check degrees within component
            start_node = None
            for u in comp:
                d = (adj[u] & cand_mask).bit_count()
                if d <= 1:
                    start_node = u
                    break

            if start_node is not None:
                # It's a path (or isolated vertex)
                path = [start_node]
                curr = start_node
                prev = -1
                while True:
                    nbrs = adj[curr] & cand_mask
                    nxt = -1
                    for w in comp:
                        if ((nbrs >> w) & 1) and w != prev:
                            nxt = w
                            break
                    if nxt == -1:
                        break
                    path.append(nxt)
                    prev, curr = curr, nxt
                for i in range(0, len(path), 2):
                    indep_set.append(path[i])
            else:
                # It's a cycle
                cycle = [comp[0]]
                curr = comp[0]
                prev = -1
                while True:
                    nbrs = adj[curr] & cand_mask
                    nxt = -1
                    for w in comp:
                        if ((nbrs >> w) & 1) and w != prev:
                            nxt = w
                            break
                    if nxt == -1 or nxt == comp[0]:
                        break
                    cycle.append(nxt)
                    prev, curr = curr, nxt
                max_take = len(cycle) // 2
                for i in range(max_take):
                    indep_set.append(cycle[2 * i])

        return indep_set

    def backtrack(cand_mask, chosen):
        if len(chosen) == k:
            return chosen
        if cand_mask.bit_count() + len(chosen) < k:
            return None

        # Find vertex with max degree in cand_mask
        max_d = -1
        best_v = -1
        for i in range(n):
            if (cand_mask >> i) & 1:
                d = (cand_mask & adj[i]).bit_count()
                if d > max_d:
                    max_d = d
                    best_v = i

        if max_d <= 2:
            s = solve_deg2(cand_mask)
            if len(chosen) + len(s) >= k:
                needed = k - len(chosen)
                return chosen + s[:needed]
            else:
                return None

        # Branch 1: include best_v
        res = backtrack(
            cand_mask & ~adj[best_v] & ~(1 << best_v), chosen + [best_v]
        )
        if res is not None:
            return res

        # Branch 2: exclude best_v
        res = backtrack(cand_mask & ~(1 << best_v), chosen)
        if res is not None:
            return res

        return None

    initial_cand = (1 << n) - 1
    ans = backtrack(initial_cand, [])

    if ans is not None:
        print("YES")
        print(*(x + 1 for x in ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
