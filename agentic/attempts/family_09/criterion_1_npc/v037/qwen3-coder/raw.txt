def solve():
    n, m, k = map(int, input().split())
    
    # Create adjacency list
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # Try all subsets of size k using bit manipulation
    # But this is too slow for n=46, so we need a better approach
    
    # Use branch and bound / backtracking with pruning
    def is_valid(candidate_set, new_node):
        # Check if adding new_node to candidate_set maintains all-strangers property
        for node in candidate_set:
            if new_node in adj[node]:
                return False
        return True
    
    def backtrack(current_set, start_idx):
        if len(current_set) == k:
            return current_set[:]
        
        if len(current_set) + (n - start_idx + 1) < k:
            return None
            
        for i in range(start_idx, n + 1):
            if is_valid(current_set, i):
                current_set.append(i)
                result = backtrack(current_set, i + 1)
                if result is not None:
                    return result
                current_set.pop()
        
        return None
    
    # For large n, we might need a more optimized approach
    # But given the constraints and that we only need one solution,
    # let's try a different strategy: build complement graph and find clique
    
    # However, let's first try a more direct approach with optimization
    
    # Try a greedy approach with backtracking
    solution = backtrack([], 1)
    
    if solution:
        print("YES")
        print(*solution)
    else:
        print("NO")
