We can solve this by trying to find an independent set of size exactly `k` in the graph. Since `n` is at most 46, we can use a backtracking approach with pruning.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    n, m, k = map(int, data[:3])
    edges = []
    idx = 3
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
    
    # Try to find independent set of size k
    # Use backtracking with pruning
    candidates = list(range(n))
    chosen = []
    
    def backtrack(start, chosen):
        if len(chosen) == k:
            return True
        # Prune: not enough remaining nodes
        if len(chosen) + (n - start) < k:
            return False
        for i in range(start, n):
            # Check if i is compatible with all chosen
            ok = True
            for c in chosen:
                if i in adj[c]:
                    ok = False
                    break
            if ok:
                chosen.append(i)
                if backtrack(i + 1, chosen):
                    return True
                chosen.pop()
        return False
    
    if backtrack(0, chosen):
        print("YES")
        print(" ".join(str(x + 1) for x in chosen))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
