from itertools import combinations

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    # Read edges and build adjacency set for quick lookup
    edges = set()
    index = 3
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        index += 2
        edges.add((u, v))
        edges.add((v, u))  # Since interaction is bidirectional
    
    # Generate all possible combinations of k nodes
    nodes = list(range(n))
    
    # For larger n, we can optimize by checking only combinations that could possibly work
    # But given n <= 46 and k <= n, we proceed with brute-force checking
    
    for combo in combinations(nodes, k):
        valid = True
        # Check all pairs within this combination
        for i in range(k):
            for j in range(i+1, k):
                u = combo[i]
                v = combo[j]
                if (u, v) in edges:
                    valid = False
                    break
            if not valid:
                break
        if valid:
            # Found a valid combination
            print("YES")
            print(' '.join(str(x + 1) for x in combo))  # Convert back to 1-based indexing
            return
    
    print("NO")

if __name__ == "__main__":
    main()
