import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m, k = data[0], data[1], data[2]

    original = [0] * n
    pos = 3
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        original[u] |= 1 << v
        original[v] |= 1 << u

    if k == 1:
        print("YES")
        print(1)
        return

    full = (1 << n) - 1
    adj = [0] * n
    for v in range(n):
        adj[v] = full & ~(original[v] | (1 << v))

    # Only vertices in the (k-1)-core of the complement can belong to a k-clique.
    candidates = full
    while True:
        removed = 0
        bits = candidates
        while bits:
            b = bits & -bits
            bits ^= b
            v = b.bit_length() - 1
            if (adj[v] & candidates).bit_count() < k - 1:
                removed |= b
        if not removed:
            break
        candidates &= ~removed

    if candidates.bit_count() < k:
        print("NO")
        return

    # A clique of size at least two is contained in one connected component.
    components = []
    unseen = candidates
    while unseen:
        seed = unseen & -unseen
        unseen ^= seed
        component = seed
        frontier = seed

        while frontier:
            b = frontier & -frontier
            frontier ^= b
            v = b.bit_length() - 1
            new_vertices = adj[v] & unseen
            unseen &= ~new_vertices
            frontier |= new_vertices
            component |= new_vertices

        if component.bit_count() >= k:
            components.append(component)

    components.sort(key=int.bit_count, reverse=True)

    for component in components:
        vertices = []
        bits = component
        while bits:
            b = bits & -bits
            bits ^= b
            vertices.append(b.bit_length() - 1)

        vertices.sort(
            key=lambda v: (adj[v] & component).bit_count(),
            reverse=True
        )

        size = len(vertices)
        index = {v: i for i, v in enumerate(vertices)}
        local_adj = [0] * size

        for i, v in enumerate(vertices):
            neighbors = adj[v] & component
            mask = 0
            while neighbors:
                b = neighbors & -neighbors
                neighbors ^= b
                mask |= 1 << index[b.bit_length() - 1]
            local_adj[i] = mask

        all_local = (1 << size) - 1

        # Fast greedy attempt.
        greedy = []
        possible = all_local
        while possible and len(greedy) < k:
            scan = possible
            best_v = -1
            best_degree = -1
            while scan:
                b = scan & -scan
                scan ^= b
                v = b.bit_length() - 1
                degree = (local_adj[v] & possible).bit_count()
                if degree > best_degree:
                    best_degree = degree
                    best_v = v
            greedy.append(best_v)
            possible &= local_adj[best_v]

        if len(greedy) == k:
            answer = [vertices[v] + 1 for v in greedy]
            print("YES")
            print(*answer)
            return

        path = []
        answer = None

        def expand(possible_mask):
            nonlocal answer

            path_len = len(path)
            if path_len + possible_mask.bit_count() < k:
                return False

            order = []
            colors = []
            uncolored = possible_mask
            color = 0

            while uncolored:
                color += 1
                available = uncolored
                while available:
                    b = available & -available
                    available ^= b
                    v = b.bit_length() - 1
                    uncolored ^= b
                    order.append(v)
                    colors.append(color)
                    available &= ~local_adj[v]

            current = possible_mask
            for i in range(len(order) - 1, -1, -1):
                if path_len + colors[i] < k:
                    return False

                v = order[i]
                b = 1 << v

                path.append(v)
                if path_len + 1 == k:
                    answer = path.copy()
                    return True

                next_mask = current & local_adj[v]
                if next_mask and expand(next_mask):
                    return True

                path.pop()
                current ^= b

            return False

        if expand(all_local):
            result = [vertices[v] + 1 for v in answer]
            print("YES")
            print(*result)
            return

    print("NO")

if __name__ == "__main__":
    main()
