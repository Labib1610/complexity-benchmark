import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        
    if k == 0:
        print("YES")
        print()
        return
        
    # Check if there is an independent set of size k
    # If a vertex has degree >= n - k, it cannot be in the independent set
    # So we only need to consider vertices with degree < n - k
    candidates = []
    for i in range(n):
        if bin(adj[i]).count('1') < n - k:
            candidates.append(i)
            
    R = len(candidates)
    if R > 2 * k:
        print("NO")
        return
        
    # Map candidate index to bit position
    idx = {u: i for i, u in enumerate(candidates)}
    
    # Create adjacency bitmask restricted to candidates
    adj_R = [0] * R
    for i, u in enumerate(candidates):
        mask = 0
        for j, v in enumerate(candidates):
            if adj[u] & (1 << v):
                mask |= 1 << j
        adj_R[i] = mask
        
    # Split candidates into two halves
    half = R // 2
    L1 = half
    L2 = R - half
    
    # DP for the first half
    dp1 = [0] * (1 << L1)
    dp1[0] = (1 << L2) - 1
    for mask in range(1, 1 << L1):
        lsb = mask & -mask
        v = lsb.bit_length() - 1
        prev = mask ^ lsb
        dp1[mask] = dp1[prev] & ~adj_R[v]
        
    # DP for the second half
    dp2 = [0] * (1 << L2)
    dp2[0] = (1 << L1) - 1
    for mask in range(1, 1 << L2):
        lsb = mask & -mask
        v = (lsb.bit_length() - 1) + L1
        prev = mask ^ lsb
        dp2[mask] = dp2[prev] & ~adj_R[v]
        
    # Find independent set of size k
    for mask1 in range(1 << L1):
        c1 = bin(mask1).count('1')
        if c1 > k:
            continue
        c2_needed = k - c1
        if c2_needed < 0:
            continue
            
        avail2 = dp1[mask1]
        # Check if there is a subset of avail2 of size c2_needed that is independent in the second half
        # We can iterate over all subsets of avail2
        sub = avail2
        while True:
            if bin(sub).count('1') == c2_needed:
                if (dp2[sub] & mask1) == mask1:
                    # Found valid independent set
                    res = []
                    for i in range(L1):
                        if mask1 & (1 << i):
                            res.append(candidates[i] + 1)
                    for i in range(L2):
                        if sub & (1 << i):
                            res.append(candidates[i + L1] + 1)
                    print("YES")
                    print(" ".join(map(str, res)))
                    return
            if sub == 0:
                break
            sub = (sub - 1) & avail2
            
    print("NO")

if __name__ == '__main__':
    solve()
