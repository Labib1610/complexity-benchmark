from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    index = 3
    conflicts = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        conflicts[u].add(v)
        conflicts[v].add(u)
        index += 2
    
    # Generate all possible subsets of size k using bit manipulation
    # We'll use a recursive approach with pruning for efficiency
    
    def is_valid_panel(panel_members):
        # Check if the panel is valid according to the rules
        # For each member added, flag incompatible members
        flagged = set()
        for member in panel_members:
            if member in flagged:
                return False
            # Flag all conflicting members
            for conflict in conflicts[member]:
                flagged.add(conflict)
        return True
    
    # Since n <= 46, we can't check all combinations directly
    # But since k <= n, and we need to find one valid combination,
    # we can try a greedy approach or backtracking
    
    # Let's try a different approach: 
    # Try all subsets of size k and check validity
    
    # However, with n=46, C(46, k) could be too large
    # Let's optimize: use a recursive backtracking approach
    
    def solve():
        # Try to build a valid panel of size k
        # Use backtracking
        
        def backtrack(current_panel, candidates):
            if len(current_panel) == k:
                return current_panel[:]
            
            # Prune if remaining candidates + current panel < k
            if len(current_panel) + len(candidates) < k:
                return None
            
            # Sort candidates for consistent ordering
            candidates.sort()
            
            for i, member in enumerate(candidates):
                # Check if member is compatible with current panel
                is_compatible = True
                for p_member in current_panel:
                    if member in conflicts[p_member]:
                        is_compatible = False
                        break
                
                if not is_compatible:
                    continue
                
                # Add member to panel
                current_panel.append(member)
                
                # Create new candidates list without this member and its conflicts
                new_candidates = []
                member_conflicts = conflicts[member]
                for c in candidates[i+1:]:
                    if c != member and c not in member_conflicts:
                        new_candidates.append(c)
                
                result = backtrack(current_panel, new_candidates)
                if result is not None:
                    return result
                
                # Backtrack
                current_panel.pop()
            
            return None
        
        # Start with all members as candidates
        initial_candidates = list(range(n))
        panel = backtrack([], initial_candidates)
        
        if panel is not None:
            # Convert back to 1-based indexing
            panel = [x + 1 for x in panel]
            return "YES\n" + " ".join(map(str, panel))
        else:
            return "NO"
    
    # Optimization for small cases
    print(solve())

if __name__ == "__main__":
    main()
