import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # For independent set of size k:
    # We can search recursively.
    # Order vertices by degree (descending or ascending).
    # To prune efficiently, we can use vertex coloring / max clique bounds or simple branching.

    # Pre-include adjacency with self for convenience when clearing candidates
    adj_with_self = [adj[i] | (1 << i) for i in range(n)]

    # Degeneracy / degree ordering
    degrees = [bin(adj[i]).count("1") for i in range(n)]
    order = sorted(range(n), key=lambda x: degrees[x])

    # Reorder vertices according to degree ordering to speed up bit operations
    remap = {order[i]: i for i in range(n)}
    rev_remap = {i: order[i] for i in range(n)}

    new_adj_with_self = [0] * n
    for i in range(n):
        orig_u = rev_remap[i]
        mask = 0
        for j in range(n):
            orig_v = rev_remap[j]
            if (adj_with_self[orig_u] >> orig_v) & 1:
                mask |= 1 << j
        new_adj_with_self[i] = mask

    # Graph coloring bound
    def get_color_bound(candidates):
        # Greedy coloring bound on candidates
        colors = 0
        cand = candidates
        while cand:
            colors += 1
            # find an independent set greedily
            cur = cand
            while cur:
                v = (cur & -cur).bit_length() - 1
                cand ^= 1 << v
                cur &= ~new_adj_with_self[v]
        return colors

    ans = []

    def search(candidates, needed):
        if needed == 0:
            return True
        if candidates.bit_count() < needed:
            return False

        # If candidates is relatively small or we want to prune:
        if get_color_bound(candidates) < needed:
            return False

        # Pick vertex with highest degree among candidates
        best_v = -1
        max_deg = -1
        temp = candidates
        while temp:
            v = (temp & -temp).bit_length() - 1
            deg = (candidates & new_adj_with_self[v]).bit_count()
            if deg > max_deg:
                max_deg = deg
                best_v = v
            temp &= temp - 1

        v = best_v

        # Branch 1: Include v
        ans.append(rev_remap[v] + 1)
        if search(candidates & ~new_adj_with_self[v], needed - 1):
            return True
        ans.pop()

        # Branch 2: Exclude v
        if search(candidates ^ (1 << v), needed):
            return True

        return False

    all_candidates = (1 << n) - 1
    if search(all_candidates, k):
        print("YES")
        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
