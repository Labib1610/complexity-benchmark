import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    if k == 0:
        print("YES\n")
        return
    if k > n:
        print("NO")
        return
    
    adj = [0] * n
    idx = 3
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    # Vertices sorted by degree descending or ascending
    # For independent set, ordering by degree can help
    deg = [bin(adj[i]).count('1') for i in range(n)]
    order = sorted(range(n), key=lambda x: deg[x])
    
    # Re-index vertices
    old_to_new = {order[i]: i for i in range(n)}
    new_to_old = {i: order[i] for i in range(n)}
    
    new_adj = [0] * n
    for i in range(n):
        u = new_to_old[i]
        mask = 0
        for v in range(n):
            if (adj[u] >> v) & 1:
                mask |= (1 << old_to_new[v])
        new_adj[i] = mask

    ans = None

    def search(candidates, current_set):
        nonlocal ans
        if len(current_set) == k:
            ans = current_set[:]
            return True
        
        rem = k - len(current_set)
        if candidates.bit_length() < rem:
            return False
        if candidates.bit_count() < rem:
            return False
        
        # Clique cover bound on candidates for pruning
        # Greedily cover candidates with cliques in the original graph (edges present)
        # Each clique can contribute at most 1 vertex to an independent set.
        temp_cand = candidates
        clique_count = 0
        while temp_cand:
            clique_count += 1
            if clique_count + len(current_set) + temp_cand.bit_count() < k: # early exit impossible to prune
                pass
            # Pick the lowest bit
            v = (temp_cand & -temp_cand).bit_length() - 1
            # Form a greedy clique starting with v
            clique_mask = 1 << v
            temp_cand &= ~(1 << v)
            # Find common neighbors in temp_cand
            common = temp_cand & new_adj[v]
            while common:
                u = (common & -common).bit_length() - 1
                clique_mask |= (1 << u)
                temp_cand &= ~(1 << u)
                common &= new_adj[u] & temp_cand
                
        if len(current_set) + clique_count < k:
            return False

        # Branching
        cur_cand = candidates
        while cur_cand:
            if cur_cand.bit_count() + len(current_set) < k:
                break
            v = (cur_cand & -cur_cand).bit_length() - 1
            cur_cand &= ~(1 << v)
            
            # Choose v
            next_cand = cur_cand & ~new_adj[v]
            current_set.append(v)
            if search(next_cand, current_set):
                return True
            current_set.pop()
            
        return False

    initial_candidates = (1 << n) - 1
    if search(initial_candidates, []):
        print("YES")
        res = [new_to_old[v] + 1 for v in ans]
        print(*(sorted(res)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
