import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    # Build adjacency set for each node
    adj = [set() for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].add(v)
        adj[v].add(u)
    
    # Greedy approach: repeatedly pick vertex with minimum degree in remaining graph
    remaining = set(range(1, n + 1))
    
    while len(remaining) > k:
        # Find vertex with minimum degree in the remaining graph
        min_deg = n + 1
        min_v = -1
        for v in remaining:
            deg = len(adj[v] & remaining)
            if deg < min_deg:
                min_deg = deg
                min_v = v
                if min_deg == 0:
                    break
        
        if min_deg >= k:
            # All remaining vertices have degree >= k
            # This means we can't find an independent set of size k
            print("NO")
            return
        
        remaining.remove(min_v)
    
    # Check if remaining set is an independent set
    remaining_list = list(remaining)
    for i in range(len(remaining_list)):
        for j in range(i + 1, len(remaining_list)):
            u = remaining_list[i]
            v = remaining_list[j]
            if v in adj[u]:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, remaining_list)))

solve()
