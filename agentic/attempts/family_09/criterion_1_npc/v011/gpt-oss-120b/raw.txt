import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL = (1 << n) - 1
    answer = []

    # simple recursive search for independent set of size k
    found = False

    # ordering vertices by degree descending for better pruning
    order = sorted(range(n), key=lambda x: bin(adj[x]).count("1"), reverse=True)
    # map original index to position in ordered list
    pos = [0] * n
    for i, v in enumerate(order):
        pos[v] = i

    # rebuild adjacency according to new order
    adj2 = [0] * n
    for i in range(n):
        v = order[i]
        mask = 0
        for j in range(n):
            if adj[v] >> order[j] & 1:
                mask |= 1 << j
        adj2[i] = mask
    adj = adj2

    def popcnt(x: int) -> int:
        return x.bit_count()

    def dfs(cur_mask: int, cand_mask: int, chosen: list):
        nonlocal found, answer
        if found:
            return
        cur_size = len(chosen)
        if cur_size == k:
            answer = [order[i] + 1 for i in chosen]  # back to 1‑based original ids
            found = True
            return
        # prune
        if cur_size + popcnt(cand_mask) < k:
            return
        # choose a vertex: pick the one with smallest degree in candidates (heuristic)
        # simple: take first bit
        v = (cand_mask & -cand_mask).bit_length() - 1
        # branch: include v
        dfs(cur_mask | (1 << v), cand_mask & ~adj[v] & ~(1 << v), chosen + [v])
        if found:
            return
        # branch: exclude v
        dfs(cur_mask, cand_mask & ~(1 << v), chosen)

    dfs(0, ALL, [])
    if not found:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, answer)))

if __name__ == "__main__":
    solve()
