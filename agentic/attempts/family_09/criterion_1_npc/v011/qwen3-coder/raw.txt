from itertools import combinations

def solve():
    n, m, k = map(int, input().split())
    
    # Create adjacency list representation of the graph
    # where vertices are markers and edges represent interference
    adj = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # Check all combinations of k markers
    markers = list(range(1, n + 1))
    
    for combo in combinations(markers, k):
        # Check if this combination has no interfering pairs
        valid = True
        for i in range(k):
            for j in range(i + 1, k):
                if combo[j] in adj[combo[i]]:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            print("YES")
            print(*combo)
            return
    
    print("NO")

solve()
