We can model the problem as finding an independent set of size exactly `k` in an undirected graph with up to 46 vertices.  
A brute-force over all subsets is too large, but we can use a meet-in-the-middle or backtracking with pruning.  

Here’s a possible solution:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Split vertices into two halves
    half = n // 2
    left = half
    right = n - half

    # For each subset of left half, check if it's independent
    left_independent = []
    for mask in range(1 << left):
        ok = True
        for i in range(left):
            if mask & (1 << i):
                if adj[i] & mask:
                    ok = False
                    break
        if ok:
            left_independent.append(mask)

    # For each subset of right half, check if it's independent
    right_independent = []
    for mask in range(1 << right):
        ok = True
        for i in range(right):
            if mask & (1 << i):
                if adj[left + i] & (mask << left):
                    ok = False
                    break
        if ok:
            right_independent.append(mask)

    # For each left independent set, store max size and mask
    # We'll try to combine with right sets
    # Precompute for each right mask the max size and the mask itself
    # But we need to check cross edges

    # For each left mask, we need to know which right vertices are allowed
    allowed_right = []
    for mask in left_independent:
        forbidden = 0
        for i in range(left):
            if mask & (1 << i):
                forbidden |= (adj[i] >> left)
        allowed = ((1 << right) - 1) & ~forbidden
        allowed_right.append(allowed)

    # For each right mask, store its size
    right_size = [0] * (1 << right)
    for mask in range(1 << right):
        right_size[mask] = bin(mask).count("1")

    # Now try to find combination
    for idx, lmask in enumerate(left_independent):
        lsize = bin(lmask).count("1")
        if lsize > k:
            continue
        need = k - lsize
        if need < 0 or need > right:
            continue
        allowed = allowed_right[idx]
        # Search for right mask subset of allowed with size == need
        sub = allowed
        while True:
            if right_size[sub] == need:
                # Found
                chosen = []
                for i in range(left):
                    if lmask & (1 << i):
                        chosen.append(i + 1)
                for i in range(right):
                    if sub & (1 << i):
                        chosen.append(left + i + 1)
                print("YES")
                print(" ".join(map(str, chosen)))
                return
            if sub == 0:
                break
            sub = (sub - 1) & allowed

    print("NO")

if __name__ == "__main__":
    solve()
```
