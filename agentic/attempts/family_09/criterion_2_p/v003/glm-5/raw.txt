```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    k = int(input_data[idx]); idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    sides = [[], []]

    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            sides[0].append(start)
            queue = deque([start])
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        sides[color[v] - 1].append(v)
                        queue.append(v)

    if k == 1:
        sys.stdout.write("YES\n1\n")
        return

    if k > n:
        sys.stdout.write("NO\n")
        return

    if m == 0:
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(str(i) for i in range(1, k + 1)) + "\n")
        return

    min_deg = n + 1
    min_node = -1
    for i in range(1, n + 1):
        d = len(adj[i])
        if d < min_deg:
            min_deg = d
            min_node = i

    if min_deg >= k:
        side0 = sides[0]
        side1 = sides[1]
        if len(side0) >= k:
            sys.stdout.write("YES\n")
            sys.stdout.write(" ".join(str(x) for x in side0[:k]) + "\n")
        elif len(side1) >= k:
            sys.stdout.write("YES\n")
            sys.stdout.write(" ".join(str(x) for x in side1[:k]) + "\n")
        else:
            sys.stdout.write("NO\n")
        return

    if min_deg + 1 >= k:
        sys.stdout.write("YES\n")
        sys.stdout.write(str(min_node) + "\n")
        return

    neighbors = set(adj[min_node])
    result = [min_node]
    for i in range(1, n + 1):
        if i != min_node and i not in neighbors:
            result.append(i)
            if len(result) == k:
                break

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(x) for x in result) + "\n")

solve()
```
