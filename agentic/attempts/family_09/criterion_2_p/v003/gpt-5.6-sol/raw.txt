import sys
from collections import deque

def main():
    input = sys.stdin.buffer.readline
    n, m, k = map(int, input().split())

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    for start in range(n):
        if color[start] != -1:
            continue
        color[start] = 0
        stack = [start]
        while stack:
            u = stack.pop()
            next_color = color[u] ^ 1
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = next_color
                    stack.append(v)

    left = [u for u in range(n) if color[u] == 0]
    right = [u for u in range(n) if color[u] == 1]

    match = [-1] * n
    dist = [0] * n
    inf = n + 1

    while True:
        q = deque()
        for u in left:
            if match[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = inf

        shortest = inf
        while q:
            u = q.popleft()
            if dist[u] >= shortest:
                continue
            nd = dist[u] + 1
            for v in adj[u]:
                w = match[v]
                if w == -1:
                    shortest = nd
                elif dist[w] == inf:
                    dist[w] = nd
                    q.append(w)

        if shortest == inf:
            break

        edge_pos = [0] * n
        augmented = 0

        for root in left:
            if match[root] != -1 or dist[root] != 0:
                continue

            stack_u = [root]
            path_v = []
            found = False

            while stack_u:
                u = stack_u[-1]
                descended = False

                while edge_pos[u] < len(adj[u]):
                    v = adj[u][edge_pos[u]]
                    edge_pos[u] += 1
                    w = match[v]

                    if w == -1:
                        if dist[u] + 1 == shortest:
                            for i, pv in enumerate(path_v):
                                pu = stack_u[i]
                                match[pu] = pv
                                match[pv] = pu
                            match[u] = v
                            match[v] = u
                            found = True
                            break
                    elif dist[w] == dist[u] + 1:
                        path_v.append(v)
                        stack_u.append(w)
                        descended = True
                        break

                if found:
                    augmented += 1
                    break

                if descended:
                    continue

                dist[u] = inf
                stack_u.pop()
                if len(path_v) >= len(stack_u):
                    path_v.pop()

        if augmented == 0:
            break

    reachable_left = bytearray(n)
    reachable_right = bytearray(n)
    q = deque()

    for u in left:
        if match[u] == -1:
            reachable_left[u] = 1
            q.append(u)

    while q:
        u = q.popleft()
        matched_v = match[u]
        for v in adj[u]:
            if v == matched_v:
                continue
            if not reachable_right[v]:
                reachable_right[v] = 1
                w = match[v]
                if w != -1 and not reachable_left[w]:
                    reachable_left[w] = 1
                    q.append(w)

    independent = []
    for u in left:
        if reachable_left[u]:
            independent.append(u + 1)
    for v in right:
        if not reachable_right[v]:
            independent.append(v + 1)

    if len(independent) < k:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, independent[:k])) + "\n")

if __name__ == "__main__":
    main()
