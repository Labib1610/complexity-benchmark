```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # bipartition via BFS
    color = [-1] * n
    left_nodes = []
    right_nodes = []
    for start in range(n):
        if color[start] != -1:
            continue
        q = deque([start])
        color[start] = 0
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                # else: graph guaranteed bipartite
    for i, c in enumerate(color):
        if c == 0:
            left_nodes.append(i)
        else:
            right_nodes.append(i)

    L = len(left_nodes)
    R = len(right_nodes)
    # map original id -> index in left/right
    lid = {v: i for i, v in enumerate(left_nodes)}
    rid = {v: i for i, v in enumerate(right_nodes)}

    # build adjacency for HK (left index -> list of right indices)
    g = [[] for _ in range(L)]
    for u, v in edges:
        if color[u] == 0:
            g[lid[u]].append(rid[v])
        else:
            g[lid[v]].append(rid[u])

    # Hopcroft–Karp
    INF = 10**9
    pairU = [-1] * L
    pairV = [-1] * R
    dist = [0] * L

    def bfs():
        q = deque()
        for u in range(L):
            if pairU[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        d = INF
        while q:
            u = q.popleft()
            if dist[u] < d:
                for v in g[u]:
                    if pairV[v] == -1:
                        d = dist[u] + 1
                    elif dist[pairV[v]] == INF:
                        dist[pairV[v]] = dist[u] + 1
                        q.append(pairV[v])
        return d != INF

    def dfs(u):
        for v in g[u]:
            pu = pairV[v]
            if pairV[v] == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(L):
            if pairU[u] == -1 and dfs(u):
                matching += 1

    # derive minimum vertex cover
    # BFS on alternating graph from free left vertices
    visL = [False] * L
    visR = [False] * R
    q = deque()
    for u in range(L):
        if pairU[u] == -1:
            q.append(u)
            visL[u] = True
    while q:
        u = q.popleft()
        for v in g[u]:
            if not visR[v] and pairU[u] != v:  # edge not in matching
                visR[v] = True
                pu = pairV[v]
                if pu != -1 and not visL[pu]:
                    visL[pu] = True
                    q.append(pu)

    # min vertex cover = (L not visited) U (R visited)
    # independent set = (L visited) U (R not visited)
    indep = []
    for i, u in enumerate(left_nodes):
        if visL[i]:
            indep.append(u + 1)
    for i, v in enumerate(right_nodes):
        if not visR[i]:
            indep.append(v + 1)

    if len(indep) >= k:
        print("YES")
        print(' '.join(map(str, indep[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
