import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    k = int(data[idx])
    idx += 1
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Since the graph is bipartite, we can color it with two colors
    # 0: unvisited, 1: color A, -1: color B
    color = [0] * (n + 1)
    partition_a = []
    partition_b = []
    
    # Check if the graph is bipartite and get the partitions
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        partition_a.append(start)
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            for neighbor in adj[node]:
                if color[neighbor] == 0:
                    color[neighbor] = -current_color
                    if color[neighbor] == 1:
                        partition_a.append(neighbor)
                    else:
                        partition_b.append(neighbor)
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    # This should not happen as per problem statement
                    return False
        return True
    
    # Process all components
    visited = [False] * (n + 1)
    all_partitions = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            partition_a.clear()
            partition_b.clear()
            queue = deque()
            queue.append(i)
            color[i] = 1
            partition_a.append(i)
            
            bipartite = True
            while queue:
                node = queue.popleft()
                current_color = color[node]
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = -current_color
                        if color[neighbor] == 1:
                            partition_a.append(neighbor)
                        else:
                            partition_b.append(neighbor)
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        bipartite = False
                        break
                if not bipartite:
                    break
            if not bipartite:
                # Should not happen per problem statement
                print("NO")
                return
            all_partitions.append((list(partition_a), list(partition_b)))
    
    # Now, for each component, we can choose one side
    # We want to select exactly k nodes with no edges between them
    # So we can pick from one side of each bipartition
    
    # Let's think of it this way:
    # Each component is bipartite. We can pick nodes only from one side of each component.
    # So the problem becomes: for each component, pick either side A or side B,
    # such that the total number of nodes picked is exactly k.
    
    # This is a subset sum problem variant.
    # But since we can pick any subset from each side, we just need to know sizes.
    
    components = []
    for part_a, part_b in all_partitions:
        components.append((len(part_a), len(part_b)))
    
    # Now, we need to pick for each component either size_a or size_b
    # such that their sum is exactly k.
    
    # DP to find if we can make sum k
    # But we also need to track which side we pick to reconstruct the solution
    
    # Let's use a different approach: generate all possible sums
    # But that might be too slow.
    
    # Better: DP with reconstruction
    # dp[i][s] = whether we can achieve sum s using first i components
    # But to reconstruct, we need to track parent pointers or similar.
    
    # Let's do DP with tracking choices
    total_components = len(components)
    max_sum = sum(max(a, b) for a, b in components)
    if k > max_sum:
        print("NO")
        return
    
    # dp[s] = whether sum s is achievable
    # prev[s] = list of (component_index, picked_size, picked_side) that led to sum s
    dp = [-1] * (k + 1)
    dp[0] = 0  # 0 sum is achievable with 0 components
    parent = [-1] * (k + 1)  # parent[s] = (comp_index, picked_size, picked_side: 0 for A, 1 for B)
    
    for i in range(total_components):
        size_a, size_b = components[i]
        # Traverse backwards to avoid using same component multiple times
        new_dp = dp[:]
        new_parent = parent[:]
        for s in range(k + 1):
            if dp[s] == i:  # Achieved with first i components
                # Try picking side A
                if s + size_a <= k:
                    if new_dp[s + size_a] < i + 1:
                        new_dp[s + size_a] = i + 1
                        new_parent[s + size_a] = (i, size_a, 0)  # 0 means side A
                # Try picking side B
                if s + size_b <= k:
                    if new_dp[s + size_b] < i + 1:
                        new_dp[s + size_b] = i + 1
                        new_parent[s + size_b] = (i, size_b, 1)  # 1 means side B
        dp = new_dp
        parent = new_parent
    
    if dp[k] != total_components:
        print("NO")
        return
    
    # Reconstruct the solution
    chosen_nodes = []
    current_sum = k
    while current_sum > 0:
        comp_index, picked_size, picked_side = parent[current_sum]
        part_a, part_b = all_partitions[comp_index]
        if picked_side == 0:
            chosen_nodes.extend(part_a)
        else:
            chosen_nodes.extend(part_b)
        current_sum -= picked_size
    
    print("YES")
    print(' '.join(map(str, chosen_nodes)))

if __name__ == "__main__":
    main()
