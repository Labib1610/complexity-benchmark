```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    if k == 1:
        sys.stdout.write("YES\n1\n")
        return

    if m == 0:
        if k <= n:
            out = "YES\n" + " ".join(str(i) for i in range(1, k + 1)) + "\n"
            sys.stdout.write(out)
        else:
            sys.stdout.write("NO\n")
        return

    color = [0] * (n + 1)
    comp_sizes = []
    comp_nodes = []

    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            stack = [start]
            cnt = [0, 0]
            cnt[0] += 1
            nodes = [start]
            while stack:
                u = stack.pop()
                cu = color[u]
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - cu
                        cnt[color[v] - 1] += 1
                        nodes.append(v)
                        stack.append(v)
            comp_sizes.append((cnt[0], cnt[1]))
            comp_nodes.append(nodes)

    total0 = sum(c[0] for c in comp_sizes)
    total1 = sum(c[1] for c in comp_sizes)

    if k <= total0:
        chosen = []
        for nodes in comp_nodes:
            for u in nodes:
                if color[u] == 1:
                    chosen.append(u)
                    if len(chosen) == k:
                        break
            if len(chosen) == k:
                break
        sys.stdout.write("YES\n" + " ".join(str(x) for x in chosen) + "\n")
        return

    if k <= total1:
        chosen = []
        for nodes in comp_nodes:
            for u in nodes:
                if color[u] == 2:
                    chosen.append(u)
                    if len(chosen) == k:
                        break
            if len(chosen) == k:
                break
        sys.stdout.write("YES\n" + " ".join(str(x) for x in chosen) + "\n")
        return

    sys.stdout.write("NO\n")

solve()
```
