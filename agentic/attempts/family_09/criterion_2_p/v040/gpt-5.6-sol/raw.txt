```python
import sys
from collections import deque

sys.setrecursionlimit(300000)

data = sys.stdin.buffer.read()
tokens = data.split()

n = int(tokens[0])
m = int(tokens[1])
k = int(tokens[2])

graph = [[] for _ in range(n + 1)]
pos = 3
for _ in range(m):
    u = int(tokens[pos])
    v = int(tokens[pos + 1])
    pos += 2
    graph[u].append(v)
    graph[v].append(u)

del tokens, data

color = [-1] * (n + 1)
left = []
right = []

for start in range(1, n + 1):
    if color[start] != -1:
        continue
    color[start] = 0
    stack = [start]
    while stack:
        u = stack.pop()
        if color[u] == 0:
            left.append(u)
        else:
            right.append(u)
        next_color = color[u] ^ 1
        for v in graph[u]:
            if color[v] == -1:
                color[v] = next_color
                stack.append(v)

if len(left) >= k:
    answer = left[:k]
    sys.stdout.write("YES\n" + " ".join(map(str, answer)))
    sys.exit()

if len(right) >= k:
    answer = right[:k]
    sys.stdout.write("YES\n" + " ".join(map(str, answer)))
    sys.exit()

pair_left = [0] * (n + 1)
pair_right = [0] * (n + 1)
dist = [0] * (n + 1)
INF = n + 1

def bfs():
    queue = deque()
    for u in left:
        if pair_left[u] == 0:
            dist[u] = 0
            queue.append(u)
        else:
            dist[u] = INF

    dist[0] = INF

    while queue:
        u = queue.popleft()
        if dist[u] >= dist[0]:
            continue
        next_dist = dist[u] + 1
        for v in graph[u]:
            matched_left = pair_right[v]
            if dist[matched_left] == INF:
                dist[matched_left] = next_dist
                if matched_left != 0:
                    queue.append(matched_left)

    return dist[0] != INF

def dfs(u):
    if u == 0:
        return True

    next_dist = dist[u] + 1
    for v in graph[u]:
        matched_left = pair_right[v]
        if dist[matched_left] == next_dist and dfs(matched_left):
            pair_left[u] = v
            pair_right[v] = u
            return True

    dist[u] = INF
    return False

matching = 0
while bfs():
    for u in left:
        if pair_left[u] == 0 and dfs(u):
            matching += 1

if n - matching < k:
    sys.stdout.write("NO")
    sys.exit()

reachable_left = bytearray(n + 1)
reachable_right = bytearray(n + 1)
stack = []

for u in left:
    if pair_left[u] == 0:
        reachable_left[u] = 1
        stack.append(u)

while stack:
    u = stack.pop()
    matched_v = pair_left[u]
    for v in graph[u]:
        if v == matched_v or reachable_right[v]:
            continue
        reachable_right[v] = 1
        next_left = pair_right[v]
        if next_left != 0 and not reachable_left[next_left]:
            reachable_left[next_left] = 1
            stack.append(next_left)

answer = []
for u in left:
    if reachable_left[u]:
        answer.append(u)
        if len(answer) == k:
            break

if len(answer) < k:
    for v in right:
        if not reachable_right[v]:
            answer.append(v)
            if len(answer) == k:
                break

sys.stdout.write("YES\n" + " ".join(map(str, answer)))
```
