An optimal and fast solution in Python 3 using the **Hopcroft–Karp algorithm** to find the Maximum Independent Set in a bipartite graph via Konig's Theorem.

```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # 2-color the graph to find bipartite partition L and R
    color = [-1] * (n + 1)
    L = []
    R = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            L.append(i)
            queue = [i]
            for u in queue:
                c = color[u]
                nc = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = nc
                        if nc == 0:
                            L.append(v)
                        else:
                            R.append(v)
                        queue.append(v)

    # Hopcroft-Karp algorithm for Maximum Bipartite Matching
    # L vertices are matched with R vertices
    pair_u = [0] * (n + 1)
    pair_v = [0] * (n + 1)
    dist = [0] * (n + 1)
    INF = 10**9

    def bfs():
        q = []
        for u in L:
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching_size = 0
    while bfs():
        for u in L:
            if pair_u[u] == 0:
                if dfs(u):
                    matching_size += 1

    # Konig's theorem to construct the Maximum Independent Set:
    # 1. BFS/DFS from unmatched vertices in L along alternating paths:
    #    L -> R: non-matching edges
    #    R -> L: matching edges
    visited_L = [False] * (n + 1)
    visited_R = [False] * (n + 1)
    
    q = []
    for u in L:
        if pair_u[u] == 0:
            visited_L[u] = True
            q.append(u)
            
    head = 0
    while head < len(q):
        u = q[head]
        head += 1
        for v in adj[u]:
            if v != pair_u[u] and not visited_R[v]:
                visited_R[v] = True
                match_u = pair_v[v]
                if match_u != 0 and not visited_L[match_u]:
                    visited_L[match_u] = True
                    q.append(match_u)

    # Max Independent Set consists of:
    # - Vertices in L that are visited
    # - Vertices in R that are NOT visited
    mis = []
    for u in L:
        if visited_L[u]:
            mis.append(u)
    for v in R:
        if not visited_R[v]:
            mis.append(v)
            
    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
