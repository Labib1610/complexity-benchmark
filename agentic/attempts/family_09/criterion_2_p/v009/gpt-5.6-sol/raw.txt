import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


def main():
    data = iter(map(int, sys.stdin.buffer.read().split()))
    n = next(data)
    m = next(data)
    k = next(data)

    if k == 1:
        print("YES")
        print(1)
        return

    if m == 0:
        print("YES")
        print(*range(1, k + 1))
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = next(data) - 1
        v = next(data) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    left = []
    right = []
    simple_independent = []

    for start in range(n):
        if color[start] != -1:
            continue

        color[start] = 0
        queue = [start]
        qpos = 0
        side0 = []
        side1 = []

        while qpos < len(queue):
            u = queue[qpos]
            qpos += 1

            if color[u] == 0:
                side0.append(u)
                left.append(u)
            else:
                side1.append(u)
                right.append(u)

            next_color = color[u] ^ 1
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = next_color
                    queue.append(v)

        if len(side0) >= len(side1):
            simple_independent.extend(side0)
        else:
            simple_independent.extend(side1)

    if len(simple_independent) >= k:
        answer = simple_independent[:k]
        print("YES")
        print(*(x + 1 for x in answer))
        return

    pair = [-1] * n
    dist = [0] * n
    inf = n + 1
    phase_distance = inf

    def bfs():
        q = deque()
        for u in left:
            if pair[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = inf

        shortest = inf
        while q:
            u = q.popleft()
            if dist[u] >= shortest:
                continue
            nd = dist[u] + 1
            for v in adj[u]:
                matched_left = pair[v]
                if matched_left == -1:
                    shortest = nd
                elif dist[matched_left] == inf:
                    dist[matched_left] = nd
                    q.append(matched_left)
        return shortest

    ptr = [0] * n

    def dfs(u):
        while ptr[u] < len(adj[u]):
            v = adj[u][ptr[u]]
            ptr[u] += 1
            matched_left = pair[v]

            if matched_left == -1:
                if dist[u] + 1 != phase_distance:
                    continue
            elif dist[matched_left] != dist[u] + 1 or not dfs(matched_left):
                continue

            pair[u] = v
            pair[v] = u
            return True

        dist[u] = inf
        return False

    matching_size = 0
    while True:
        phase_distance = bfs()
        if phase_distance == inf:
            break

        ptr = [0] * n
        for u in left:
            if pair[u] == -1 and dfs(u):
                matching_size += 1

    if n - matching_size < k:
        print("NO")
        return

    visited_left = bytearray(n)
    visited_right = bytearray(n)
    q = deque()

    for u in left:
        if pair[u] == -1:
            visited_left[u] = 1
            q.append(u)

    while q:
        u = q.popleft()
        matched_v = pair[u]
        for v in adj[u]:
            if v == matched_v or visited_right[v]:
                continue
            visited_right[v] = 1
            matched_left = pair[v]
            if matched_left != -1 and not visited_left[matched_left]:
                visited_left[matched_left] = 1
                q.append(matched_left)

    answer = []
    for u in left:
        if visited_left[u]:
            answer.append(u)
            if len(answer) == k:
                break

    if len(answer) < k:
        for v in right:
            if not visited_right[v]:
                answer.append(v)
                if len(answer) == k:
                    break

    print("YES")
    print(*(x + 1 for x in answer))


if __name__ == "__main__":
    main()
