We can solve this by noticing the graph is bipartite.  
We can color the users into two camps (colors 0 and 1).  
Then, an all-strangers group must be entirely within one camp, because all edges go between camps.  

So we just need to check if either camp has at least `k` users.  
If yes, pick any `k` from that camp.  

```python
import sys

def main():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    camp0 = []
    camp1 = []

    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            stack = [i]
            camp0.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        if color[v] == 0:
                            camp0.append(v)
                        else:
                            camp1.append(v)
                        stack.append(v)

    if len(camp0) >= k:
        print("YES")
        print(" ".join(str(x + 1) for x in camp0[:k]))
    elif len(camp1) >= k:
        print("YES")
        print(" ".join(str(x + 1) for x in camp1[:k]))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
