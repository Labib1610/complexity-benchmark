import sys

# Increase recursion depth just in case
sys.setrecursionlimit(300000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k > n:
        print("NO")
        return

    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # 1. 2-color the graph to find bipartite partition
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            q = [i]
            for u in q:
                c = color[u]
                nc = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = nc
                        q.append(v)

    # Left: color == 0, Right: color == 1
    # Orient edges from Left to Right
    adj_L = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        if color[u] == 0:
            adj_L[u] = adj[u]

    # Hopcroft-Karp algorithm for Maximum Bipartite Matching
    pair_u = [0] * (n + 1)  # L -> R
    pair_v = [0] * (n + 1)  # R -> L
    dist = [0] * (n + 1)
    INF = 10**9

    def bfs():
        q = []
        for u in range(1, n + 1):
            if color[u] == 0:
                if pair_u[u] == 0:
                    dist[u] = 0
                    q.append(u)
                else:
                    dist[u] = INF
        dist[0] = INF
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            if dist[u] < dist[0]:
                for v in adj_L[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj_L[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching_size = 0
    while bfs():
        for u in range(1, n + 1):
            if color[u] == 0 and pair_u[u] == 0:
                if dfs(u):
                    matching_size += 1

    mis_size = n - matching_size
    if mis_size < k:
        print("NO")
        return

    # Find the Maximum Independent Set using Konig's construction
    # Reachable vertices from unmatched L vertices via alternating paths
    # Alternating path: L -> R along unmatched edges, R -> L along matched edges
    visited_L = [False] * (n + 1)
    visited_R = [False] * (n + 1)

    q = []
    for u in range(1, n + 1):
        if color[u] == 0 and pair_u[u] == 0:
            visited_L[u] = True
            q.append(u)

    head = 0
    while head < len(q):
        u = q[head]
        head += 1
        for v in adj_L[u]:
            if not visited_R[v] and pair_u[u] != v:
                visited_R[v] = True
                u2 = pair_v[v]
                if u2 != 0 and not visited_L[u2]:
                    visited_L[u2] = True
                    q.append(u2)

    # Max Independent Set is:
    # (L vertices that are visited) UNION (R vertices that are NOT visited)
    mis = []
    for u in range(1, n + 1):
        if color[u] == 0:
            if visited_L[u]:
                mis.append(u)
        else:
            if not visited_R[u]:
                mis.append(u)

    print("YES")
    print(*(mis[:k]))


if __name__ == "__main__":
    solve()
