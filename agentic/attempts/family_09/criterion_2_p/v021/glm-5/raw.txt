```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    k = int(data[idx]); idx += 1

    if m == 0:
        if k <= n:
            out = "YES\n" + " ".join(str(i) for i in range(1, k + 1))
            sys.stdout.write(out)
        else:
            sys.stdout.write("NO")
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    color[1] = 1
    stack = [1]
    while stack:
        u = stack.pop()
        for v in adj[u]:
            if color[v] == 0:
                color[v] = 3 - color[u]
                stack.append(v)

    cnt1 = 0
    cnt2 = 0
    for i in range(1, n + 1):
        if color[i] == 1:
            cnt1 += 1
        else:
            cnt2 += 1

    if cnt1 >= k:
        out = "YES\n" + " ".join(str(i) for i in range(1, n + 1) if color[i] == 1)[:k]
        sys.stdout.write(out)
    elif cnt2 >= k:
        out = "YES\n" + " ".join(str(i) for i in range(1, n + 1) if color[i] == 2)[:k]
        sys.stdout.write(out)
    else:
        sys.stdout.write("NO")

solve()
```
