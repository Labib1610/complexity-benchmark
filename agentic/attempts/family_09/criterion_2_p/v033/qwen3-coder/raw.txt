import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, k, edges

def solve(n, m, k, edges):
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # If no edges, we can pick any k nodes
    if m == 0:
        return "YES\n" + " ".join(map(str, range(1, k+1)))
    
    # BFS to color the graph (bipartite check and get components)
    color = {}
    components = []
    
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if visited[start]:
            continue
            
        # BFS to find component and color it
        queue = deque([start])
        visited[start] = True
        color[start] = 0
        comp_nodes = [[], []]  # comp_nodes[0] for color 0, comp_nodes[1] for color 1
        comp_nodes[0].append(start)
        
        is_bipartite = True
        
        while queue and is_bipartite:
            node = queue.popleft()
            node_color = color[node]
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = 1 - node_color
                    comp_nodes[1 - node_color].append(neighbor)
                    queue.append(neighbor)
                elif color[neighbor] == node_color:
                    is_bipartite = False
                    break
        
        if not is_bipartite:
            # This shouldn't happen according to problem statement
            continue
            
        components.append(comp_nodes)
    
    # For each component, we can pick at most all nodes from one side
    total_max_independent = 0
    best_selection = []
    
    for comp_nodes in components:
        size_0 = len(comp_nodes[0])
        size_1 = len(comp_nodes[1])
        
        if size_0 >= k:
            return "YES\n" + " ".join(map(str, comp_nodes[0][:k]))
            
        if size_1 >= k:
            return "YES\n" + " ".join(map(str, comp_nodes[1][:k]))
    
    # If no single component can provide k nodes,
    # we need to combine components
    
    # Collect all possible independent sets (one side from each component)
    # But since we want exactly k, and we can take from different components,
    # we just collect all nodes from one side of each component
    
    # Greedy approach: collect as many as possible
    selected = []
    
    for comp_nodes in components:
        # Pick the larger side
        if len(comp_nodes[0]) >= len(comp_nodes[1]):
            selected.extend(comp_nodes[0])
        else:
            selected.extend(comp_nodes[1])
            
        if len(selected) >= k:
            return "YES\n" + " ".join(map(str, selected[:k]))
    
    # If still not enough
    if len(selected) >= k:
        return "YES\n" + " ".join(map(str, selected[:k]))
    else:
        return "NO"

def main():
    n, m, k, edges = read_input()
    result = solve(n, m, k, edges)
    print(result)

if __name__ == "__main__":
    main()
