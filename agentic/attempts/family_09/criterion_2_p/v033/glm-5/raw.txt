```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    k = int(data[idx]); idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    # Find connected components and bipartition
    color = [0] * (n + 1)
    visited = [False] * (n + 1)
    components = []

    for start in range(1, n + 1):
        if not visited[start]:
            visited[start] = True
            color[start] = 1
            comp = [start]
            q = deque([start])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 3 - color[u]
                        comp.append(v)
                        q.append(v)
            components.append(comp)

    # Check if any component has a side with >= k users
    for comp in components:
        side1 = [u for u in comp if color[u] == 1]
        side2 = [u for u in comp if color[u] == 2]
        if len(side1) >= k:
            print("YES")
            print(" ".join(map(str, side1[:k])))
            return
        if len(side2) >= k:
            print("YES")
            print(" ".join(map(str, side2[:k])))
            return

    # Otherwise, greedily pick users from the smaller side of each component
    result = []
    for comp in components:
        side1 = [u for u in comp if color[u] == 1]
        side2 = [u for u in comp if color[u] == 2]
        if len(side1) <= len(side2):
            result.extend(side1)
        else:
            result.extend(side2)

    if len(result) >= k:
        print("YES")
        print(" ".join(map(str, result[:k])))
    else:
        print("NO")

solve()
```
