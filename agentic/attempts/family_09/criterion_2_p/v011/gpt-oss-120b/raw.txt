```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # 1. bipartition (color 0/1)
    color = [-1] * (n + 1)
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        color[start] = 0
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)

    left = [i for i in range(1, n + 1) if color[i] == 0]
    right = [i for i in range(1, n + 1) if color[i] == 1]

    # adjacency from left to right only
    L_adj = {u: [] for u in left}
    for u, v in edges:
        if color[u] == 0:
            L_adj[u].append(v)
        else:
            L_adj[v].append(u)

    # Hopcroft–Karp
    INF = 10 ** 9
    pair_u = {u: 0 for u in left}
    pair_v = {v: 0 for v in right}
    dist = {}

    def bfs():
        q = deque()
        for u in left:
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in L_adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return dist[0] != INF

    def dfs(u):
        if u == 0:
            return True
        for v in L_adj[u]:
            if dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v]):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in left:
            if pair_u[u] == 0 and dfs(u):
                matching += 1

    # 2. Minimum vertex cover via alternating BFS from free left vertices
    Lvis = set()
    Rvis = set()
    q = deque()
    for u in left:
        if pair_u[u] == 0:
            q.append(u)
            Lvis.add(u)
    while q:
        u = q.popleft()
        for v in L_adj[u]:
            if v not in Rvis and pair_u[u] != v:  # edge not in matching
                Rvis.add(v)
                if pair_v[v] and pair_v[v] not in Lvis:
                    Lvis.add(pair_v[v])
                    q.append(pair_v[v])

    # Minimum vertex cover = (Left - Lvis) ∪ (Right ∩ Rvis)
    # Independent set = complement
    indep = []
    for u in left:
        if u in Lvis:
            indep.append(u)
    for v in right:
        if v not in Rvis:
            indep.append(v)

    if len(indep) < k:
        print("NO")
        return
    print("YES")
    print(' '.join(map(str, indep[:k])))

if __name__ == "__main__":
    solve()
```
