import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Bipartition the graph
    color = [-1] * (n + 1)
    parts = [[], []]
    is_bipartite = True
    
    for i in range(1, n + 1):
        if color[i] == -1:
            queue = deque([i])
            color[i] = 0
            parts[0].append(i)
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        parts[color[v]].append(v)
                        queue.append(v)
                    elif color[v] == color[u]:
                        is_bipartite = False
    
    # Since the problem guarantees the two-family structure, the graph is bipartite.
    # We just need to pick the larger partition if k allows, or any valid subset.
    
    # Choose the larger partition to maximize chances
    if len(parts[0]) < len(parts[1]):
        parts[0], parts[1] = parts[1], parts[0]
    
    if len(parts[0]) >= k:
        print("YES")
        print(" ".join(map(str, parts[0][:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
