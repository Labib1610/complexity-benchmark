```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    components = []
    
    for start in range(1, n + 1):
        if color[start] == 0:
            comp = [[], []]
            color[start] = 1
            comp[0].append(start)
            q = deque([start])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp[color[v] - 1].append(v)
                        q.append(v)
            components.append(comp)
    
    total = n + 1
    dp = [False] * total
    dp[0] = True
    parent = [-1] * total
    choice = [-1] * total
    
    for idx, comp in enumerate(components):
        a = len(comp[0])
        b = len(comp[1])
        for s in range(total - 1, -1, -1):
            if dp[s]:
                if s + a < total and not dp[s + a]:
                    dp[s + a] = True
                    parent[s + a] = s
                    choice[s + a] = (idx, 0)
                if s + b < total and not dp[s + b]:
                    dp[s + b] = True
                    parent[s + b] = s
                    choice[s + b] = (idx, 1)
    
    if not dp[k]:
        sys.stdout.write("NO\n")
        return
    
    selected = [False] * len(components)
    cur = k
    while cur > 0:
        idx, side = choice[cur]
        selected[idx] = side
        cur = parent[cur]
    
    result = []
    for i in range(len(components)):
        side = selected[i]
        result.extend(components[i][side])
    
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, result)) + "\n")

solve()
```
