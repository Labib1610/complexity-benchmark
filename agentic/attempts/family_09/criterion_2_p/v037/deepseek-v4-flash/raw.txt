We can solve this by realizing the graph is bipartite.  
An all-strangers group means we can only pick users from the same camp (since friendships only exist between camps).  

So the largest possible all-strangers group is the size of the larger camp.  
If `k` is larger than that, answer is `NO`.  
Otherwise, we can pick any `k` users from the larger camp.

Here’s the implementation:

```python
import sys
from collections import deque

def main():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * (n + 1)
    camp0 = []
    camp1 = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            camp0.append(i)
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        if color[v] == 0:
                            camp0.append(v)
                        else:
                            camp1.append(v)
                        q.append(v)
    
    if len(camp0) >= k:
        print("YES")
        print(" ".join(map(str, camp0[:k])))
    elif len(camp1) >= k:
        print("YES")
        print(" ".join(map(str, camp1[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
