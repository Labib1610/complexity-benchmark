import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    # Bipartition the graph
    color = [0] * (n + 1)
    camp1 = []
    camp2 = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            queue = deque([i])
            while queue:
                u = queue.popleft()
                if color[u] == 1:
                    camp1.append(u)
                else:
                    camp2.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
    
    # Choose the larger camp if needed, but we need at least k
    if len(camp1) >= k:
        result = camp1[:k]
    elif len(camp2) >= k:
        result = camp2[:k]
    else:
        print("NO")
        return
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
