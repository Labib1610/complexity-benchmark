```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    # bipartite coloring
    color = [-1]*(n+1)
    for start in range(1, n+1):
        if color[start] != -1:
            continue
        color[start] = 0
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u]^1
                    q.append(v)

    # build left adjacency (only edges from left(0) to right(1))
    left_nodes = [i for i in range(1, n+1) if color[i]==0]
    right_nodes = [i for i in range(1, n+1) if color[i]==1]

    # Hopcroft–Karp
    INF = 10**9
    pair_u = [0]*(n+1)   # match for left (or any vertex)
    pair_v = [0]*(n+1)   # match for right (or any vertex)
    dist = [0]*(n+1)

    def bfs() -> bool:
        dq = deque()
        for u in left_nodes:
            if pair_u[u]==0:
                dist[u]=0
                dq.append(u)
            else:
                dist[u]=INF
        dist[0]=INF
        while dq:
            u = dq.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if color[v]!=1:  # ensure edge direction
                        continue
                    if dist[pair_v[v]]==INF:
                        dist[pair_v[v]] = dist[u]+1
                        dq.append(pair_v[v])
        return dist[0]!=INF

    def dfs(u: int) -> bool:
        if u==0:
            return True
        for v in adj[u]:
            if color[v]!=1:
                continue
            if dist[pair_v[v]]==dist[u]+1 and dfs(pair_v[v]):
                pair_u[u]=v
                pair_v[v]=u
                return True
        dist[u]=INF
        return False

    matching = 0
    while bfs():
        for u in left_nodes:
            if pair_u[u]==0 and dfs(u):
                matching+=1

    # Minimum vertex cover from König's theorem
    # BFS on alternating graph starting from free left vertices
    visL = [False]*(n+1)
    visR = [False]*(n+1)
    dq = deque()
    for u in left_nodes:
        if pair_u[u]==0:
            dq.append(u)
            visL[u]=True
    while dq:
        u = dq.popleft()
        if color[u]==0:  # left side, traverse non‑matched edges to right
            for v in adj[u]:
                if color[v]!=1: continue
                if pair_u[u]==v:  # matched edge, skip
                    continue
                if not visR[v]:
                    visR[v]=True
                    dq.append(v)
        else:  # right side, traverse matched edge back to left
            v = pair_v[u]
            if v and not visL[v]:
                visL[v]=True
                dq.append(v)

    min_vertex_cover = set()
    for u in left_nodes:
        if not visL[u]:
            min_vertex_cover.add(u)
    for v in right_nodes:
        if visR[v]:
            min_vertex_cover.add(v)

    independent = [i for i in range(1, n+1) if i not in min_vertex_cover]
    if len(independent) >= k:
        print("YES")
        print(' '.join(map(str, independent[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
