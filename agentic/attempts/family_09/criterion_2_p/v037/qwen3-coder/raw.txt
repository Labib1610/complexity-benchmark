```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    index = 3
    edges = []
    adj = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].add(v)
        adj[v].add(u)
    
    # If there are no edges, all nodes can be in the same group
    if m == 0:
        print("YES")
        print(' '.join(map(str, range(1, k+1))))
        return
    
    # Check if the graph is bipartite and find the two camps
    color = [-1] * (n + 1)
    camp0 = []
    camp1 = []
    
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 0
        camp0.append(start) if color[start] == 0 else camp1.append(start)
        
        while queue:
            node = queue.popleft()
            for neighbor in adj[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - color[node]
                    if color[neighbor] == 0:
                        camp0.append(neighbor)
                    else:
                        camp1.append(neighbor)
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    # This should not happen as per problem statement
                    return False
        return True
    
    # Process all components
    for i in range(1, n+1):
        if color[i] == -1:
            if not bfs(i):
                # Not bipartite, but problem guarantees it is
                pass
    
    # Now we have two camps
    # An all-strangers group is a subset of one camp
    # Check if either camp has at least k nodes
    
    if len(camp0) >= k:
        print("YES")
        print(' '.join(map(str, camp0[:k])))
    elif len(camp1) >= k:
        print("YES")
        print(' '.join(map(str, camp1[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
