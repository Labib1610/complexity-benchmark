import sys
from collections import deque

def int_stream(data):
    num = 0
    in_num = False
    for c in data:
        if 48 <= c <= 57:
            num = num * 10 + c - 48
            in_num = True
        elif in_num:
            yield num
            num = 0
            in_num = False
    if in_num:
        yield num

data = sys.stdin.buffer.read()
it = iter(int_stream(data))
n = next(it)
m = next(it)
k = next(it)

adj = [[] for _ in range(n + 1)]
for _ in range(m):
    u = next(it)
    v = next(it)
    adj[u].append(v)
    adj[v].append(u)

# Bipartition each connected component, orienting its smaller side as color 1.
color = bytearray(n + 1)
q = deque()

for start in range(1, n + 1):
    if color[start]:
        continue

    color[start] = 1
    q.append(start)
    component = []
    count1 = 0
    count2 = 0

    while q:
        u = q.popleft()
        component.append(u)
        if color[u] == 1:
            count1 += 1
            next_color = 2
        else:
            count2 += 1
            next_color = 1

        for v in adj[u]:
            if color[v] == 0:
                color[v] = next_color
                q.append(v)

    if count1 > count2:
        for u in component:
            color[u] = 3 - color[u]

left = [u for u in range(1, n + 1) if color[u] == 1]
right_size = n - len(left)

# Color 2 is an independent set.
if right_size >= k:
    answer = []
    for u in range(1, n + 1):
        if color[u] == 2:
            answer.append(u)
            if len(answer) == k:
                break
    sys.stdout.write("YES\n" + " ".join(map(str, answer)))
    sys.exit()

# Hopcroft-Karp maximum matching.
pair = [0] * (n + 1)
dist = [0] * (n + 1)
INF = n + 1
matching = 0

while True:
    q.clear()
    for u in left:
        if pair[u] == 0:
            dist[u] = 0
            q.append(u)
        else:
            dist[u] = INF

    nil_dist = INF

    while q:
        u = q.popleft()
        if dist[u] >= nil_dist:
            continue
        nd = dist[u] + 1
        for v in adj[u]:
            pu = pair[v]
            if pu == 0:
                if nd < nil_dist:
                    nil_dist = nd
            elif dist[pu] == INF:
                dist[pu] = nd
                q.append(pu)

    if nil_dist == INF:
        break

    edge_pos = [0] * (n + 1)
    augmented_this_phase = 0

    for root in left:
        if pair[root] != 0:
            continue

        stack = [root]
        via_right = []
        augmented = False

        while stack:
            u = stack[-1]
            descended = False

            while edge_pos[u] < len(adj[u]):
                v = adj[u][edge_pos[u]]
                edge_pos[u] += 1
                pu = pair[v]

                if pu == 0:
                    if dist[u] + 1 != nil_dist:
                        continue

                    pair[u] = v
                    pair[v] = u
                    for i in range(len(stack) - 2, -1, -1):
                        rv = via_right[i]
                        lu = stack[i]
                        pair[lu] = rv
                        pair[rv] = lu

                    augmented = True
                    augmented_this_phase += 1
                    break

                if dist[pu] == dist[u] + 1:
                    stack.append(pu)
                    via_right.append(v)
                    descended = True
                    break

            if augmented:
                break

            if descended:
                continue

            dist[u] = INF
            stack.pop()
            if via_right:
                via_right.pop()

    matching += augmented_this_phase

    if augmented_this_phase == 0:
        break

if n - matching < k:
    sys.stdout.write("NO")
    sys.exit()

# Alternating reachability from unmatched left vertices.
vis_left = bytearray(n + 1)
vis_right = bytearray(n + 1)
q.clear()

for u in left:
    if pair[u] == 0:
        vis_left[u] = 1
        q.append(u)

while q:
    u = q.popleft()
    matched_v = pair[u]

    for v in adj[u]:
        if v == matched_v:
            continue
        if vis_right[v]:
            continue

        vis_right[v] = 1
        pu = pair[v]
        if pu != 0 and not vis_left[pu]:
            vis_left[pu] = 1
            q.append(pu)

# Complement of the minimum vertex cover:
# (reachable left) union (unreachable right).
answer = []
for u in range(1, n + 1):
    if (color[u] == 1 and vis_left[u]) or (color[u] == 2 and not vis_right[u]):
        answer.append(u)
        if len(answer) == k:
            break

sys.stdout.write("YES\n" + " ".join(map(str, answer)))
