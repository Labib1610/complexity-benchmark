```python
import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * (n + 1)
    L = []
    R = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            L.append(i)
            queue = deque([i])
            while queue:
                u = queue.popleft()
                c = color[u]
                nc = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = nc
                        if nc == 0:
                            L.append(v)
                        else:
                            R.append(v)
                        queue.append(v)
    
    # Hopcroft-Karp algorithm
    pair_u = [0] * (n + 1)
    pair_v = [0] * (n + 1)
    dist = [0] * (n + 1)
    INF = 10**9
    
    def bfs():
        queue = deque()
        for u in L:
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        queue.append(pu)
        return dist[0] != INF

    sys.setrecursionlimit(max(300000, n + 500))
    ptr = [0] * (n + 1)
    
    def dfs(u):
        if u != 0:
            adj_u = adj[u]
            while ptr[u] < len(adj_u):
                v = adj_u[ptr[u]]
                ptr[u] += 1
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    while bfs():
        for u in L:
            ptr[u] = 0
        for u in L:
            if pair_u[u] == 0:
                dfs(u)
                
    # Find Maximum Independent Set
    visited_L = [False] * (n + 1)
    visited_R = [False] * (n + 1)
    queue = deque()
    
    for u in L:
        if pair_u[u] == 0:
            visited_L[u] = True
            queue.append(u)
            
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if v != pair_u[u] and not visited_R[v]:
                visited_R[v] = True
                match_u = pair_v[v]
                if match_u != 0 and not visited_L[match_u]:
                    visited_L[match_u] = True
                    queue.append(match_u)
                    
    mis = []
    for u in L:
        if visited_L[u]:
            mis.append(u)
    for v in R:
        if not visited_R[v]:
            mis.append(v)
            
    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
