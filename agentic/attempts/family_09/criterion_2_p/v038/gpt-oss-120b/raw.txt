```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # bipartition (graph is guaranteed bipartite)
    color = [-1] * n
    left, right = [], []
    for i in range(n):
        if color[i] == -1:
            q = deque([i])
            color[i] = 0
            while q:
                v = q.popleft()
                (left if color[v] == 0 else right).append(v)
                for nb in adj[v]:
                    if color[nb] == -1:
                        color[nb] = color[v] ^ 1
                        q.append(nb)
    # if any side already large enough, answer directly
    if len(left) >= k:
        print("YES")
        print(' '.join(str(v+1) for v in left[:k]))
        return
    if len(right) >= k:
        print("YES")
        print(' '.join(str(v+1) for v in right[:k]))
        return

    # Build bipartite graph for matching: left side indices 0..L-1, right 0..R-1
    L = len(left); R = len(right)
    idxL = {v:i for i,v in enumerate(left)}
    idxR = {v:i for i,v in enumerate(right)}
    g = [[] for _ in range(L)]
    for u,v in edges:
        if color[u]==0:
            g[idxL[u]].append(idxR[v])
        else:
            g[idxL[v]].append(idxR[u])

    # Hopcroft–Karp
    INF = 10**9
    pairU = [-1]*L
    pairV = [-1]*R
    dist = [0]*L

    def bfs():
        q = deque()
        for u in range(L):
            if pairU[u]==-1:
                dist[u]=0
                q.append(u)
            else:
                dist[u]=INF
        d = INF
        while q:
            u = q.popleft()
            if dist[u] < d:
                for v in g[u]:
                    if pairV[v]==-1:
                        d = dist[u]+1
                    elif dist[pairV[v]]==INF:
                        dist[pairV[v]] = dist[u]+1
                        q.append(pairV[v])
        return d!=INF

    def dfs(u):
        for v in g[u]:
            pu = pairV[v]
            if pairV[v]==-1 or (dist[pu]==dist[u]+1 and dfs(pu)):
                pairU[u]=v
                pairV[v]=u
                return True
        dist[u]=INF
        return False

    matching = 0
    while bfs():
        for u in range(L):
            if pairU[u]==-1 and dfs(u):
                matching+=1

    # Minimum vertex cover from König's theorem
    # BFS from free left vertices in the alternating graph
    visitedL = [False]*L
    visitedR = [False]*R
    q = deque()
    for u in range(L):
        if pairU[u]==-1:
            q.append(u)
            visitedL[u]=True
    while q:
        u = q.popleft()
        for v in g[u]:
            if not visitedR[v] and pairU[u]!=v:  # edge not in matching
                visitedR[v]=True
                if pairV[v]!=-1 and not visitedL[pairV[v]]:
                    visitedL[pairV[v]]=True
                    q.append(pairV[v])

    # Vertex cover = (Left not visited) U (Right visited)
    cover_set = set()
    for i in range(L):
        if not visitedL[i]:
            cover_set.add(left[i])
    for i in range(R):
        if visitedR[i]:
            cover_set.add(right[i])

    independent = [v for v in range(n) if v not in cover_set]
    if len(independent) >= k:
        print("YES")
        print(' '.join(str(v+1) for v in independent[:k]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
