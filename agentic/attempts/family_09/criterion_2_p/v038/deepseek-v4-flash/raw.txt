We can model the problem as finding an independent set of size exactly `k` in a graph that is guaranteed to be bipartite (even cycles, no triangles).  

Since the graph is bipartite, we can color it with two colors. Then, the larger color class is an independent set. If its size is at least `k`, we can pick any `k` vertices from it. Otherwise, no independent set of size `k` exists.

Here’s the implementation:

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    part = [[], []]
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            part[0].append(i)
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        part[color[v]].append(v)
                        q.append(v)
    
    # Choose the larger partition
    if len(part[0]) >= len(part[1]):
        big = part[0]
    else:
        big = part[1]
    
    if len(big) < k:
        print("NO")
        return
    
    print("YES")
    print(" ".join(str(x + 1) for x in big[:k]))

if __name__ == "__main__":
    solve()
```
