from collections import deque
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # 2-color the graph
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            q = [i]
            for u in q:
                c = color[u]
                nc = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = nc
                        q.append(v)

    # Hopcroft-Karp Algorithm
    # L: color == 0, R: color == 1
    # pair_u[u] = v (u in L, v in R)
    # pair_v[v] = u (v in R, u in L)
    pair_u = [0] * (n + 1)
    pair_v = [0] * (n + 1)
    dist = [0] * (n + 1)
    INF = 10**9

    L_nodes = [i for i in range(1, n + 1) if color[i] == 0]

    def bfs():
        q = deque()
        for u in L_nodes:
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF

        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        q.append(pu)
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching_size = 0
    while bfs():
        for u in L_nodes:
            if pair_u[u] == 0:
                if dfs(u):
                    matching_size += 1

    mis_size = n - matching_size
    if mis_size < k:
        sys.stdout.write("NO\n")
        return

    # Find reachable vertices in alternating paths from unmatched L vertices
    # Directed edges:
    # u in L -> v in R if v != pair_u[u] (unmatched edge)
    # v in R -> u in L if u == pair_v[v] (matched edge)
    visited = [False] * (n + 1)
    q = deque()
    for u in L_nodes:
        if pair_u[u] == 0:
            visited[u] = True
            q.append(u)

    while q:
        curr = q.popleft()
        if color[curr] == 0:
            # from L to R via non-matching edges
            for v in adj[curr]:
                if v != pair_u[curr] and not visited[v]:
                    visited[v] = True
                    q.append(v)
        else:
            # from R to L via matching edge
            u = pair_v[curr]
            if u != 0 and not visited[u]:
                visited[u] = True
                q.append(u)

    # MIS = (L with visited=True) + (R with visited=False)
    chosen = []
    for u in range(1, n + 1):
        if color[u] == 0:
            if visited[u]:
                chosen.append(u)
        else:
            if not visited[u]:
                chosen.append(u)
        if len(chosen) == k:
            break

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, chosen)) + "\n")


if __name__ == "__main__":
    solve()
