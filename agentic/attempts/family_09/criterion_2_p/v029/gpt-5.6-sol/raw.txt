import sys
from collections import deque

def main():
    input = sys.stdin.buffer.readline
    n, m, k = map(int, input().split())

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append(v)
        graph[v].append(u)

    color = [-1] * n
    for start in range(n):
        if color[start] != -1:
            continue
        color[start] = 0
        stack = [start]
        while stack:
            u = stack.pop()
            nc = color[u] ^ 1
            for v in graph[u]:
                if color[v] == -1:
                    color[v] = nc
                    stack.append(v)

    left = []
    right = []
    for i, c in enumerate(color):
        if c == 0:
            left.append(i)
        else:
            right.append(i)

    # Only adjacency lists of the left side are needed from now on.
    for v in right:
        graph[v] = []

    pair_left = [-1] * n
    pair_right = [-1] * n
    matching = 0
    INF = n + 1

    while True:
        dist = [-1] * n
        queue = deque()

        for u in left:
            if pair_left[u] == -1:
                dist[u] = 0
                queue.append(u)

        shortest = INF

        while queue:
            u = queue.popleft()
            du = dist[u]
            if du >= shortest:
                continue

            for v in graph[u]:
                w = pair_right[v]
                if w == -1:
                    shortest = du
                elif dist[w] == -1:
                    dist[w] = du + 1
                    queue.append(w)

        if shortest == INF:
            break

        ptr = [0] * n

        for root in left:
            if pair_left[root] != -1 or dist[root] != 0:
                continue

            stack_u = [root]
            stack_v = []
            augmented = False

            while stack_u:
                u = stack_u[-1]
                adj = graph[u]
                advanced = False

                while ptr[u] < len(adj):
                    v = adj[ptr[u]]
                    ptr[u] += 1
                    w = pair_right[v]

                    if w == -1:
                        if dist[u] == shortest:
                            path_v = stack_v + [v]
                            for path_u, path_r in zip(stack_u, path_v):
                                pair_left[path_u] = path_r
                                pair_right[path_r] = path_u
                            augmented = True
                            break
                    elif dist[u] < shortest and dist[w] == dist[u] + 1:
                        stack_v.append(v)
                        stack_u.append(w)
                        advanced = True
                        break

                if augmented:
                    break

                if not advanced:
                    dist[u] = -1
                    stack_u.pop()
                    if stack_v and len(stack_v) >= len(stack_u):
                        stack_v.pop()

            if augmented:
                matching += 1

    if n - matching < k:
        sys.stdout.write("NO\n")
        return

    # Alternating reachability from unmatched left-side vertices.
    reachable_left = bytearray(n)
    reachable_right = bytearray(n)
    queue = deque()

    for u in left:
        if pair_left[u] == -1:
            reachable_left[u] = 1
            queue.append(u)

    while queue:
        u = queue.popleft()
        matched_v = pair_left[u]

        for v in graph[u]:
            if v == matched_v:
                continue
            if not reachable_right[v]:
                reachable_right[v] = 1
                w = pair_right[v]
                if w != -1 and not reachable_left[w]:
                    reachable_left[w] = 1
                    queue.append(w)

    answer = []

    for u in left:
        if reachable_left[u]:
            answer.append(u + 1)
            if len(answer) == k:
                break

    if len(answer) < k:
        for v in right:
            if not reachable_right[v]:
                answer.append(v + 1)
                if len(answer) == k:
                    break

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, answer)) + "\n")

if __name__ == "__main__":
    main()
