```python
import sys
from collections import defaultdict, deque

def solve():
    n, m, k = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(set)
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # BFS to find bipartition
    color = {}
    for i in range(1, n+1):
        if i not in color:
            queue = deque([i])
            color[i] = 0
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if neighbor not in color:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # This shouldn't happen according to problem statement
                        pass
    
    # Separate nodes by color
    set0 = [i for i in range(1, n+1) if color.get(i, 0) == 0]
    set1 = [i for i in range(1, n+1) if color.get(i, 0) == 1]
    
    # Check which set can provide k nodes
    if len(set0) >= k:
        print("YES")
        print(*set0[:k])
    elif len(set1) >= k:
        print("YES")
        print(*set1[:k])
    else:
        print("NO")

solve()
```
