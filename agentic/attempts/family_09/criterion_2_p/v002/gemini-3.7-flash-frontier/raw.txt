An optimal and efficient solution in Python 3 using the Hopcroft-Karp algorithm to find the Maximum Independent Set in a bipartite graph.

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        
    # 2-color the graph into Left and Right sets
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            queue = deque([i])
            while queue:
                u = queue.popleft()
                c = color[u]
                nc = 2 if c == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = nc
                        queue.append(v)

    # Left vertices are color 1, Right vertices are color 2
    # Hopcroft-Karp Algorithm
    pair_u = [0] * (n + 1)
    pair_v = [0] * (n + 1)
    dist = [0] * (n + 1)
    
    left_nodes = [i for i in range(1, n + 1) if color[i] == 1]
    
    INF = 10**9

    def bfs():
        q = deque()
        for u in left_nodes:
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        q.append(pu)
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    while bfs():
        for u in left_nodes:
            if pair_u[u] == 0:
                dfs(u)

    # To find the Maximum Independent Set:
    # 1. Run BFS from all unmatched Left vertices in the alternating tree
    # 2. MIS = (Left vertices visited) U (Right vertices not visited)
    visited_left = [False] * (n + 1)
    visited_right = [False] * (n + 1)
    
    q = deque()
    for u in left_nodes:
        if pair_u[u] == 0:
            visited_left[u] = True
            q.append(u)
            
    while q:
        u = q.popleft()
        for v in adj[u]:
            if not visited_right[v] and pair_u[u] != v:
                visited_right[v] = True
                pu = pair_v[v]
                if pu != 0 and not visited_left[pu]:
                    visited_left[pu] = True
                    q.append(pu)
                    
    mis = []
    for i in range(1, n + 1):
        if color[i] == 1:
            if visited_left[i]:
                mis.append(i)
        else:
            if not visited_right[i]:
                mis.append(i)
                
    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
