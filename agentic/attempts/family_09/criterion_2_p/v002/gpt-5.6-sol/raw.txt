import sys
from collections import deque

def main():
    input = sys.stdin.buffer.readline
    n, m, k = map(int, input().split())

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        graph[u].append(v)
        graph[v].append(u)

    color = bytearray(b'\xff') * n

    for start in range(n):
        if color[start] != 255:
            continue
        color[start] = 0
        stack = [start]
        while stack:
            u = stack.pop()
            next_color = color[u] ^ 1
            for v in graph[u]:
                if color[v] == 255:
                    color[v] = next_color
                    stack.append(v)

    left = []
    right = []
    for v in range(n):
        if color[v] == 0:
            left.append(v)
        else:
            right.append(v)

    if len(left) >= k:
        answer = left[:k]
        print("YES")
        print(" ".join(str(v + 1) for v in answer))
        return

    if len(right) >= k:
        answer = right[:k]
        print("YES")
        print(" ".join(str(v + 1) for v in answer))
        return

    pair = [-1] * n
    dist = [-1] * n
    inf = n + 1

    while True:
        queue = []
        for u in left:
            if pair[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = -1

        head = 0
        shortest = inf

        while head < len(queue):
            u = queue[head]
            head += 1
            du = dist[u]

            if du + 1 > shortest:
                continue

            for v in graph[u]:
                matched_left = pair[v]
                if matched_left == -1:
                    if du + 1 < shortest:
                        shortest = du + 1
                elif du + 1 < shortest and dist[matched_left] == -1:
                    dist[matched_left] = du + 1
                    queue.append(matched_left)

        if shortest == inf:
            break

        ptr = [0] * n
        augmented = 0

        for root in left:
            if pair[root] != -1 or dist[root] != 0:
                continue

            stack = [root]
            path_edges = []
            success = False

            while stack:
                u = stack[-1]
                descended = False
                adjacency = graph[u]

                while ptr[u] < len(adjacency):
                    v = adjacency[ptr[u]]
                    ptr[u] += 1
                    matched_left = pair[v]

                    if matched_left == -1:
                        if dist[u] + 1 != shortest:
                            continue

                        pair[u] = v
                        pair[v] = u

                        for i in range(len(stack) - 2, -1, -1):
                            path_v = path_edges[i]
                            parent = stack[i]
                            pair[parent] = path_v
                            pair[path_v] = parent

                        success = True
                        break

                    if dist[matched_left] == dist[u] + 1:
                        path_edges.append(v)
                        stack.append(matched_left)
                        descended = True
                        break

                if success:
                    augmented += 1
                    break

                if descended:
                    continue

                dist[u] = -1
                stack.pop()
                if path_edges:
                    path_edges.pop()

        if augmented == 0:
            break

    reachable = bytearray(n)
    queue = deque()

    for u in left:
        if pair[u] == -1:
            reachable[u] = 1
            queue.append(u)

    while queue:
        u = queue.popleft()
        matched_v = pair[u]

        for v in graph[u]:
            if v == matched_v:
                continue
            if reachable[v]:
                continue

            reachable[v] = 1
            matched_left = pair[v]
            if matched_left != -1 and not reachable[matched_left]:
                reachable[matched_left] = 1
                queue.append(matched_left)

    independent = []
    for u in left:
        if reachable[u]:
            independent.append(u)
    for v in right:
        if not reachable[v]:
            independent.append(v)

    if len(independent) < k:
        print("NO")
    else:
        print("YES")
        print(" ".join(str(v + 1) for v in independent[:k]))

if __name__ == "__main__":
    main()
