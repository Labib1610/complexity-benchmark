```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # bipartite coloring
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        q.append(v)

    # map to left/right indices
    left_id = [-1] * n
    right_id = [-1] * n
    L = R = 0
    for i in range(n):
        if color[i] == 0:
            left_id[i] = L
            L += 1
        else:
            right_id[i] = R
            R += 1

    # build adjacency for left side
    g = [[] for _ in range(L)]
    for u, v in edges:
        if color[u] == 0:
            g[left_id[u]].append(right_id[v])
        else:
            g[left_id[v]].append(right_id[u])

    # Hopcroft-Karp
    INF = 10**9
    pairU = [-1] * L
    pairV = [-1] * R
    dist = [0] * L

    def bfs() -> bool:
        dq = deque()
        for u in range(L):
            if pairU[u] == -1:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        found = False
        while dq:
            u = dq.popleft()
            for v in g[u]:
                pu = pairV[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    dq.append(pu)
                if pu == -1:
                    found = True
        return found

    def dfs(u: int) -> bool:
        for v in g[u]:
            pu = pairV[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(L):
            if pairU[u] == -1 and dfs(u):
                matching += 1

    # find minimum vertex cover
    # BFS from free left vertices in the alternating graph
    visL = [False] * L
    visR = [False] * R
    dq = deque()
    for u in range(L):
        if pairU[u] == -1:
            dq.append(u)
            visL[u] = True
    while dq:
        u = dq.popleft()
        for v in g[u]:
            if not visR[v] and pairU[u] != v:  # edge not in matching
                visR[v] = True
                pu = pairV[v]
                if pu != -1 and not visL[pu]:
                    visL[pu] = True
                    dq.append(pu)

    # vertex cover = (left not visited) U (right visited)
    indep = []
    for i in range(n):
        if color[i] == 0:
            if not visL[left_id[i]]:
                indep.append(i + 1)
        else:
            if not visR[right_id[i]]:
                indep.append(i + 1)

    if len(indep) >= k:
        print("YES")
        print(" ".join(map(str, indep[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
