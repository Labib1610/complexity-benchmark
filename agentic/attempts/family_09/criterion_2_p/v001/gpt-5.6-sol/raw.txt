```python
import sys
from collections import deque

sys.setrecursionlimit(300000)

data = list(map(int, sys.stdin.buffer.read().split()))
n, m, k = data[0], data[1], data[2]

adj = [[] for _ in range(n + 1)]
p = 3
for _ in range(m):
    u, v = data[p], data[p + 1]
    p += 2
    adj[u].append(v)
    adj[v].append(u)
del data

color = [-1] * (n + 1)
left = []
right = []

for start in range(1, n + 1):
    if color[start] != -1:
        continue
    color[start] = 0
    q = deque([start])
    while q:
        u = q.popleft()
        if color[u] == 0:
            left.append(u)
        else:
            right.append(u)
        nc = color[u] ^ 1
        for v in adj[u]:
            if color[v] == -1:
                color[v] = nc
                q.append(v)

if len(left) >= k:
    print("YES")
    print(*left[:k])
    sys.exit()

if len(right) >= k:
    print("YES")
    print(*right[:k])
    sys.exit()

pair_left = [0] * (n + 1)
pair_right = [0] * (n + 1)
dist = [0] * (n + 1)
INF = n + 1

def bfs():
    q = deque()
    for u in left:
        if pair_left[u] == 0:
            dist[u] = 0
            q.append(u)
        else:
            dist[u] = INF

    shortest = INF
    while q:
        u = q.popleft()
        if dist[u] >= shortest:
            continue
        for v in adj[u]:
            matched_u = pair_right[v]
            if matched_u == 0:
                shortest = dist[u] + 1
            elif dist[matched_u] == INF:
                dist[matched_u] = dist[u] + 1
                q.append(matched_u)
    return shortest

def dfs(u, shortest):
    for v in adj[u]:
        matched_u = pair_right[v]
        if matched_u == 0:
            if dist[u] + 1 == shortest:
                pair_left[u] = v
                pair_right[v] = u
                return True
        elif dist[matched_u] == dist[u] + 1 and dfs(matched_u, shortest):
            pair_left[u] = v
            pair_right[v] = u
            return True
    dist[u] = INF
    return False

matching = 0
while True:
    shortest = bfs()
    if shortest == INF:
        break
    for u in left:
        if pair_left[u] == 0 and dfs(u, shortest):
            matching += 1

if n - matching < k:
    print("NO")
    sys.exit()

reachable_left = bytearray(n + 1)
reachable_right = bytearray(n + 1)
q = deque()

for u in left:
    if pair_left[u] == 0:
        reachable_left[u] = 1
        q.append(u)

while q:
    u = q.popleft()
    matched_v = pair_left[u]
    for v in adj[u]:
        if v == matched_v or reachable_right[v]:
            continue
        reachable_right[v] = 1
        matched_u = pair_right[v]
        if matched_u and not reachable_left[matched_u]:
            reachable_left[matched_u] = 1
            q.append(matched_u)

answer = []
for u in left:
    if reachable_left[u]:
        answer.append(u)
        if len(answer) == k:
            break

if len(answer) < k:
    for v in right:
        if not reachable_right[v]:
            answer.append(v)
            if len(answer) == k:
                break

print("YES")
print(*answer)
```
