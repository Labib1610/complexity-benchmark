```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    sizes = []
    reps = []

    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            q = deque([start])
            cnt = [0, 0]
            cnt[1] = 1
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        cnt[color[v]] += 1
                        q.append(v)
            sizes.append(cnt)
            reps.append(start)

    total = sum(max(s[0], s[1]) for s in sizes)
    if total < k:
        print("NO")
        return

    print("YES")
    remaining = k
    out = []
    for idx, (cnt, start) in enumerate(zip(sizes, reps)):
        if remaining <= 0:
            break
        take = min(remaining, max(cnt[0], cnt[1]))
        remaining -= take
        target_color = 1 if cnt[0] >= cnt[1] else 2
        q = deque([start])
        seen = [False] * (n + 1)
        seen[start] = True
        while q and take > 0:
            u = q.popleft()
            if color[u] == target_color:
                out.append(u)
                take -= 1
                if take == 0:
                    break
            for v in adj[u]:
                if not seen[v]:
                    seen[v] = True
                    q.append(v)

    print(' '.join(map(str, out)))

solve()
```
