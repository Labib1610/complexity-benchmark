```python
import sys
from collections import defaultdict, deque

def solve():
    n, m, k = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Bipartite graph coloring
    color = [-1] * (n + 1)  # -1: unvisited, 0: color A, 1: color B
    visited = [False] * (n + 1)
    
    # Check if graph is bipartite and get the two sets
    set_a = []
    set_b = []
    
    for start in range(1, n + 1):
        if not visited[start]:
            queue = deque([start])
            color[start] = 0
            visited[start] = True
            
            # Count nodes in each color class for this component
            count_a = 0
            count_b = 0
            
            while queue:
                node = queue.popleft()
                if color[node] == 0:
                    set_a.append(node)
                    count_a += 1
                else:
                    set_b.append(node)
                    count_b += 1
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # This shouldn't happen according to problem statement
                        # but let's handle it gracefully
                        pass
    
    # Choose the larger set if k <= size of that set
    if len(set_a) >= k:
        print("YES")
        print(*set_a[:k])
    elif len(set_b) >= k:
        print("YES")
        print(*set_b[:k])
    else:
        print("NO")

solve()
```
