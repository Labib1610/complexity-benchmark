import sys

def solve():
    sys.setrecursionlimit(300000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    # 2-color the graph to find L and R
    color = [-1] * (n + 1)
    L = []
    R = []
    
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            L.append(i)
            queue = [i]
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                c = color[u]
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        if color[v] == 0:
                            L.append(v)
                        else:
                            R.append(v)
                        queue.append(v)
    
    # Hopcroft-Karp algorithm
    pair_u = [0] * (n + 1)
    pair_v = [0] * (n + 1)
    dist = [0] * (n + 1)
    INF = 10**9
    
    # Pre-filter adjacency lists for L to only contain R vertices
    adj_L = [[] for _ in range(n + 1)]
    for u in L:
        adj_L[u] = adj[u]
        
    def bfs():
        q = []
        for u in L:
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            if dist[u] < dist[0]:
                for v in adj_L[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        q.append(pu)
        return dist[0] != INF

    # Pointer for DFS
    ptr = [0] * (n + 1)

    def dfs(u):
        if u != 0:
            adj_u = adj_L[u]
            while ptr[u] < len(adj_u):
                v = adj_u[ptr[u]]
                ptr[u] += 1
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in L:
            ptr[u] = 0
        for u in L:
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1

    max_indep_size = n - matching
    if max_indep_size < k:
        sys.stdout.write("NO\n")
        return

    # Find Maximum Independent Set via Konig's Theorem
    # Reachable vertices from unmatched L vertices via alternating paths
    visited_L = [False] * (n + 1)
    visited_R = [False] * (n + 1)
    
    q = []
    for u in L:
        if pair_u[u] == 0:
            visited_L[u] = True
            q.append(u)
            
    head = 0
    while head < len(q):
        u = q[head]
        head += 1
        for v in adj_L[u]:
            if not visited_R[v] and pair_u[u] != v:
                visited_R[v] = True
                pu = pair_v[v]
                if pu != 0 and not visited_L[pu]:
                    visited_L[pu] = True
                    q.append(pu)
                    
    # MIS = (L & visited_L) | (R & not visited_R)
    res = []
    for u in L:
        if visited_L[u]:
            res.append(u)
            if len(res) == k:
                break
    if len(res) < k:
        for v in R:
            if not visited_R[v]:
                res.append(v)
                if len(res) == k:
                    break
                    
    sys.stdout.write("YES\n" + " ".join(map(str, res)) + "\n")

if __name__ == '__main__':
    solve()
