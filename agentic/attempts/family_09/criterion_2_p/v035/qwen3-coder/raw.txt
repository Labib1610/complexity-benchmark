```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    edges = []
    index = 3
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        edges.append((u, v))
        index += 2
    
    # Build adjacency list
    adj = defaultdict(set)
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # If no edges, we can pick any k nodes
    if m == 0:
        print("YES")
        print(' '.join(map(str, range(1, k+1))))
        return
    
    # Bipartite check and get the two sets
    color = {}
    visited = set()
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0
        visited.add(start)
        set0 = [start]
        set1 = []
        while queue:
            node = queue.popleft()
            c = color[node]
            for nei in adj[node]:
                if nei not in visited:
                    visited.add(nei)
                    color[nei] = 1 - c
                    if color[nei] == 0:
                        set0.append(nei)
                    else:
                        set1.append(nei)
                    queue.append(nei)
                elif color[nei] == c:
                    # Should not happen as per problem statement
                    pass
        return set0, set1
    
    all_nodes = set(range(1, n+1))
    components = []
    
    for node in range(1, n+1):
        if node not in visited:
            s0, s1 = bfs(node)
            components.append((s0, s1))
    
    # For each component, we can choose to take nodes from one side or the other
    # But within a component, we must take nodes from the same side
    
    # We will try to build a set of size k by selecting nodes from independent sets
    # Since the graph is bipartite, we can take all nodes from one side of each component
    
    # Let's collect all possible sides
    sides = []
    for s0, s1 in components:
        sides.append(s0)
        sides.append(s1)
    
    # But that's not right either. We need to think differently.
    
    # The key insight: since the graph is bipartite, we can color it with two colors.
    # Then, in each connected component, we have two sets (color classes).
    # An all-stranger group can take any subset of nodes as long as no two are adjacent.
    # In a bipartite graph, this means we can take any subset of nodes that are not connected,
    # but more specifically, we can take any combination of nodes from different color classes 
    # across components, but within a component we can only take nodes from one side.
    
    # So our approach:
    # 1. Find all connected components
    # 2. For each component, determine its two color classes
    # 3. For each component, we can choose to take nodes from class 0 or class 1 (or none)
    # 4. We want to select a total of k nodes
    
    # This becomes a subset sum problem variant.
    # But let's simplify: just take the larger side from each component until we reach k
    
    # Actually, let's re-color properly:
    color = {}
    visited = [False] * (n + 1)
    
    def bfs_color(start):
        queue = deque([start])
        color[start] = 0
        visited[start] = True
        c0 = [start]
        c1 = []
        while queue:
            u = queue.popleft()
            cur_color = color[u]
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = 1 - cur_color
                    if color[v] == 0:
                        c0.append(v)
                    else:
                        c1.append(v)
                    queue.append(v)
        return c0, c1
    
    # Get all components
    components = []
    for i in range(1, n+1):
        if not visited[i]:
            c0, c1 = bfs_color(i)
            components.append((c0, c1))
    
    # Now, for each component, we can choose to take nodes from c0 or c1
    # We want to maximize the number of nodes we can take without conflict
    # Since taking from different components doesn't create conflicts
    
    # Greedy: for each component, take the larger side
    total = 0
    chosen_sides = []
    for c0, c1 in components:
        if len(c0) >= len(c1):
            total += len(c0)
            chosen_sides.append((c0, True))  # True means take c0
        else:
            total += len(c1)
            chosen_sides.append((c1, False)) # False means take c1
    
    # But we don't need maximum, we need exactly k
    # So we need to see if we can form a subset of size k
    
    # This is a subset selection problem:
    # For each component, we can take 0, |c0|, or |c1| nodes
    # Can we select subsets that sum to k?
    
    # Let's use DP on prefixes
    # dp[i] = can we make size i using first j components
    
    # But that might be overkill. Let's think greedy again.
    
    # Let's sort components by max(|c0|, |c1|) descending
    comp_sizes = []
    for idx, (c0, c1) in enumerate(components):
        m0, m1 = len(c0), len(c1)
        comp_sizes.append((max(m0, m1), idx, m0, m1))
        
    comp_sizes.sort(reverse=True)
    
    # Try to build up to k greedily
    selected_nodes = []
    current_count = 0
    
    for max_size, idx, m0, m1 in comp_sizes:
        c0, c1 = components[idx]
        # Decide which side to take
        if current_count + max(m0, m1) <= k:
            # Take the larger
            if m0 >= m1:
                selected_nodes.extend(c0)
                current_count += m0
            else:
                selected_nodes.extend(c1)
                current_count += m1
        else:
            # See if we can take part of it
            needed = k - current_count
            if needed == 0:
                break
            # We can only take full side from a component
            if m0 == needed:
                selected_nodes.extend(c0)
                current_count += m0
                break
            elif m1 == needed:
                selected_nodes.extend(c1)
                current_count += m1
                break
                
    if len(selected_nodes) == k:
        print("YES")
        print(' '.join(map(str, selected_nodes)))
        return
        
    # Let's do proper DP
    # For each component, we have a choice: take c0, take c1, or take nothing (0)
    # We want a subsequence that sums to k
    
    options_per_component = []
    for c0, c1 in components:
        options = [(len(c0), c0), (len(c1), c1)]
        options_per_component.append(options)
        
    # DP: reachable sizes
    # But we also need to track the actual nodes
    # Let's use a different approach: meet-in-the-middle or simple recursion with memoization
    
    # Since number of components is at most n, and usually much less, let's try recursive approach
    
    def find_combination(comp_idx, current_nodes, current_size):
        if current_size == k:
            return current_nodes[:]
        if comp_idx >= len(components) or current_size > k:
            return None
            
        c0, c1 = components[comp_idx]
        # Option 1: take c0
        if current_size + len(c0) <= k:
            current_nodes.extend(c0)
            res = find_combination(comp_idx + 1, current_nodes, current_size + len(c0))
            if res is not None:
                return res
            del current_nodes[-len(c0):]
                
        # Option 2: take c1
        if current_size + len(c1) <= k:
            current_nodes.extend(c1)
            res = find_combination(comp_idx + 1, current_nodes, current_size + len(c1))
            if res is not None:
                return res
            del current_nodes[-len(c1):]
            
        # Option 3: take neither
        res = find_combination(comp_idx + 1, current_nodes, current_size)
        return res
    
    result = find_combination(0, [], 0)
    if result is not None:
        print("YES")
        print(' '.join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
