 Here’s the Python solution for the problem:  

```python
import sys
from collections import deque

def main():
    input = sys.stdin.readline
    n, m, k = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            q = deque([i])
            comp = [[], []]
            comp[0].append(i)
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        comp[color[v]].append(v)
                        q.append(v)
            components.append(comp)

    # Check if we can pick k users with no edges between them.
    # Since graph is bipartite, an independent set can contain at most one side of each connected component.
    # We need to see if we can pick total k from the two sides of each component.
    # This is a DP over components: possible sizes of independent set.
    possible = [False] * (n + 1)
    possible[0] = True
    parent = [-1] * (n + 1)
    choice = [-1] * (n + 1)

    for idx, (left, right) in enumerate(components):
        new_possible = possible[:]
        for s in range(n, -1, -1):
            if possible[s]:
                if s + len(left) <= n and not new_possible[s + len(left)]:
                    new_possible[s + len(left)] = True
                    parent[s + len(left)] = s
                    choice[s + len(left)] = 0
                if s + len(right) <= n and not new_possible[s + len(right)]:
                    new_possible[s + len(right)] = True
                    parent[s + len(right)] = s
                    choice[s + len(right)] = 1
        possible = new_possible

    if not possible[k]:
        print("NO")
        return

    # Reconstruct
    selected = []
    s = k
    for idx in range(len(components) - 1, -1, -1):
        c = choice[s]
        comp = components[idx]
        if c == 0:
            selected.extend(comp[0])
            s = parent[s]
        else:
            selected.extend(comp[1])
            s = parent[s]

    print("YES")
    print(" ".join(map(str, selected)))

if __name__ == "__main__":
    main()
```
