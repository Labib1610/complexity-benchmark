import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, k, edges

def bipartite_partition(n, edges):
    # Create adjacency list
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Color nodes with BFS to check bipartiteness and partition
    color = [-1] * (n + 1)  # -1: unvisited, 0: color A, 1: color B
    partition = [[], []]  # partition[0] and partition[1]
    
    for start_node in range(1, n + 1):
        if color[start_node] == -1:  # Unvisited component
            queue = deque([start_node])
            color[start_node] = 0
            partition[0].append(start_node)
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                next_color = 1 - current_color
                
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = next_color
                        partition[next_color].append(v)
                        queue.append(v)
                    elif color[v] == current_color:
                        # Not bipartite - but problem guarantees bipartite graph
                        pass
    
    return partition[0], partition[1]

def solve():
    n, m, k, edges = read_input()
    
    # Partition nodes into two sets based on channels
    set_a, set_b = bipartite_partition(n, edges)
    
    # Check which set (or combination) can form an interference-free set of size k
    if len(set_a) >= k:
        print("YES")
        print(' '.join(map(str, set_a[:k])))
    elif len(set_b) >= k:
        print("YES")
        print(' '.join(map(str, set_b[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
