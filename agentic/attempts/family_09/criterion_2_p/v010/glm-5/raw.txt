import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    k = int(input_data[idx]); idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    deg = [0] * (n + 1)
    for i in range(1, n + 1):
        deg[i] = len(adj[i])

    if k == 1:
        print("YES")
        print(1)
        return

    removed = [False] * (n + 1)
    deg_cur = deg[:]
    q = deque()

    for i in range(1, n + 1):
        if deg_cur[i] < k - 1:
            q.append(i)
            removed[i] = True

    while q:
        u = q.popleft()
        for v in adj[u]:
            if not removed[v]:
                deg_cur[v] -= 1
                if deg_cur[v] < k - 1:
                    removed[v] = True
                    q.append(v)

    cand = [i for i in range(1, n + 1) if not removed[i]]

    if cand:
        print("YES")
        print(' '.join(map(str, cand[:k])))
        return

    if k > n:
        print("NO")
        return

    removed2 = [False] * (n + 1)
    deg2 = deg[:]
    q2 = deque()

    for i in range(1, n + 1):
        if deg2[i] < k - 1:
            q2.append(i)
            removed2[i] = True

    while q2:
        u = q2.popleft()
        for v in adj[u]:
            if not removed2[v]:
                deg2[v] -= 1
                if deg2[v] < k - 1:
                    removed2[v] = True
                    q2.append(v)

    cand2 = [i for i in range(1, n + 1) if not removed2[i]]
    if cand2:
        print("YES")
        print(' '.join(map(str, cand2[:k])))
        return

    removed3 = [False] * (n + 1)
    deg3 = deg[:]
    q3 = deque()

    for i in range(1, n + 1):
        if deg3[i] < k:
            q3.append(i)
            removed3[i] = True

    while q3:
        u = q3.popleft()
        for v in adj[u]:
            if not removed3[v]:
                deg3[v] -= 1
                if deg3[v] < k:
                    removed3[v] = True
                    q3.append(v)

    cand3 = [i for i in range(1, n + 1) if not removed3[i]]
    if cand3:
        print("YES")
        print(' '.join(map(str, cand3[:k])))
        return

    print("NO")

solve()
