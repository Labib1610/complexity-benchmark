```
def hopcroft_karp(nU, nV, adj):  # adj: list of lists for U side
    pairU = [-1] * nU
    pairV = [-1] * nV
    dist = [0] * nU
    matching = 0
    while True:
        # BFS
        queue = deque()
        for u in range(nU):
            if pairU[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        found = False
        while queue:
            u = queue.popleft()
            du = dist[u] + 1
            for v in adj[u]:
                pu = pairV[v]
                if pu == -1:
                    found = True  # reached free V vertex
                elif dist[pu] > du:  # not visited
                    dist[pu] = du
                    queue.append(pu)
        if not found:
            break
        # DFS
        for u in range(nU):
            if pairU[u] == -1:
                if dfs(u): matching += 1
    return matching, pairU, pairV
```
