Let's write a fast Hopcroft-Karp algorithm in Python:
Hopcroft-Karp algorithm finds max matching in bipartite graph in $O(E \sqrt{V})$.
After finding maximum matching, we construct MIS by:
1. Direct edges from Left to Right if not in matching, and Right to Left if in matching.
2. Find all reachable vertices from unmatched Left vertices via DFS/BFS.
3. Let $L_{reach}$ and $R_{reach}$ be reachable vertices.
4. The minimum vertex cover is $(L \setminus L_{reach}) \cup R_{reach}$.
5. The maximum independent set is $V \setminus MVC = L_{reach} \cup (R \setminus R_{reach})$.

Let's optimize Hopcroft-Karp for Python:
First, 2-color the graph to partition into Left and Right.
Let Left be 1-indexed up to $n$.
Hopcroft-Karp with BFS for levels, DFS for augmenting paths.
With BFS queue and iterative DFS (or recursion with sys.setrecursionlimit), Hopcroft-Karp runs extremely fast on typical graphs.

Let's structure the code efficiently.
```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # 2-color the graph
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            queue = [i]
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                c = color[u]
                nc = 2 if c == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = nc
                        queue.append(v)

    # Hopcroft-Karp algorithm
    # L: color == 1, R: color == 2
    # match_L[u] = v in R, match_R[v] = u in L
    match_L = [0] * (n + 1)
    match_R = [0] * (n + 1)
    dist = [0] * (n + 1)
    INF = 10**9
    
    L_nodes = [i for i in range(1, n + 1) if color[i] == 1]
    
    def bfs():
        queue = []
        for u in L_nodes:
            if match_L[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            if dist[u] < dist[0]:
                d_next = dist[u] + 1
                for v in adj[u]:
                    mv = match_R[v]
                    if dist[mv] == INF:
                        dist[mv] = d_next
                        queue.append(mv)
        return dist[0] != INF

    # Iterative DFS for Hopcroft-Karp to avoid recursion limits
    def dfs_all():
        ptrs = [0] * (n + 1)
        stack = []
        flow = 0
        
        for root in L_nodes:
            if match_L[root] != 0:
                continue
            
            # DFS from root
            stack.append(root)
            while stack:
                u = stack[-1]
                if u == 0:
                    # Augmenting path found, trace back
                    stack.pop() # pop 0
                    while stack:
                        cur_u = stack.pop()
                        # The edge we used from cur_u is adj[cur_u][ptrs[cur_u] - 1]
                        cur_v = adj[cur_u][ptrs[cur_u] - 1]
                        match_L[cur_u] = cur_v
                        match_R[cur_v] = cur_u
                    flow += 1
                    break
                
                advanced = False
                u_adj = adj[u]
                d_u = dist[u]
                while ptrs[u] < len(u_adj):
                    v = u_adj[ptrs[u]]
                    ptrs[u] += 1
                    mv = match_R[v]
                    if dist[mv] == d_u + 1:
                        stack.append(mv)
                        advanced = True
                        break
                
                if not advanced:
                    dist[u] = INF
                    stack.pop()
                    
        return flow

    while bfs():
        dfs_all()
        
    # Find reachable vertices from unmatched vertices in L
    # Directed graph: L -> R along non-matching edges, R -> L along matching edges
    visited_L = [False] * (n + 1)
    visited_R = [False] * (n + 1)
    
    queue = []
    for u in L_nodes:
        if match_L[u] == 0:
            visited_L[u] = True
            queue.append(u)
            
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        for v in adj[u]:
            if v != match_L[u]: # non-matching edge u -> v
                if not visited_R[v]:
                    visited_R[v] = True
                    mv = match_R[v]
                    if mv != 0 and not visited_L[mv]:
                        visited_L[mv] = True
                        queue.append(mv)
                        
    # Maximum Independent Set:
    # vertices in L that are visited (L_reach) + vertices in R that are NOT visited (R \ R_reach)
    mis = []
    for i in range(1, n + 1):
        if color[i] == 1:
            if visited_L[i]:
                mis.append(i)
        else:
            if not visited_R[i]:
                mis.append(i)
                
    if len(mis) >= k:
        sys.stdout.write("YES\n" + " ".join(map(str, mis[:k])) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
```
