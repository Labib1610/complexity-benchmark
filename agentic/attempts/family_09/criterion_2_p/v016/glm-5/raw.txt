import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    components = []
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            comp = []
            while stack:
                u = stack.pop()
                comp.append(u)
                cu = color[u]
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = -cu
                        stack.append(v)
            components.append(comp)
    
    C = len(components)
    comp_color = [0] * (n + 1)
    for idx, comp in enumerate(components):
        for v in comp:
            comp_color[v] = idx
    
    comp_size = [len(comp) for comp in components]
    comp_A = [0] * C
    comp_B = [0] * C
    for idx, comp in enumerate(components):
        cnt1 = 0
        cnt2 = 0
        for v in comp:
            if color[v] == 1:
                cnt1 += 1
            else:
                cnt2 += 1
        comp_A[idx] = cnt1
        comp_B[idx] = cnt2
    
    if k > n:
        print("NO")
        return
    
    if k == 0:
        print("YES")
        print()
        return
    
    if C == 0:
        print("NO")
        return
    
    total = sum(comp_size)
    if k > total:
        print("NO")
        return
    
    if k == total:
        print("YES")
        result = list(range(1, n + 1))
        print(' '.join(map(str, result)))
        return
    
    if k == 0:
        print("YES")
        print()
        return
    
    if C == 1:
        if comp_A[0] >= k:
            print("YES")
            result = [v for v in components[0] if color[v] == 1][:k]
            print(' '.join(map(str, result)))
        elif comp_B[0] >= k:
            print("YES")
            result = [v for v in components[0] if color[v] == -1][:k]
            print(' '.join(map(str, result)))
        else:
            print("NO")
        return
    
    S = k
    max_S = S
    for idx in range(C):
        max_S += min(comp_A[idx], comp_B[idx])
    
    if max_S < k:
        print("NO")
        return
    
    dp = [set() for _ in range(S + 1)]
    dp[0].add(0)
    
    possible = True
    for idx in range(C):
        a = comp_A[idx]
        b = comp_B[idx]
        new_dp = [set() for _ in range(S + 1)]
        for s in range(S + 1):
            for mask in dp[s]:
                if s + a <= S:
                    new_dp[s + a].add(mask | (1 << idx))
                if s + b <= S:
                    new_dp[s + b].add(mask)
        dp = new_dp
        if not any(dp[s] for s in range(S + 1)):
            possible = False
            break
    
    if not possible or not dp[S]:
        print("NO")
        return
    
    mask = next(iter(dp[S]))
    
    result = []
    for idx in range(C):
        if mask & (1 << idx):
            for v in components[idx]:
                if color[v] == 1:
                    result.append(v)
        else:
            for v in components[idx]:
                if color[v] == -1:
                    result.append(v)
    
    result = result[:k]
    print("YES")
    print(' '.join(map(str, result)))

solve()
