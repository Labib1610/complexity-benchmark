```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    # bipartition via BFS
    color = [-1] * n
    left = []
    right = []
    for i in range(n):
        if color[i] == -1:
            q = deque([i])
            color[i] = 0
            while q:
                v = q.popleft()
                for to in adj[v]:
                    if color[to] == -1:
                        color[to] = color[v] ^ 1
                        q.append(to)
    left = [i for i in range(n) if color[i] == 0]
    right = [i for i in range(n) if color[i] == 1]

    # build adjacency from left to right for matching
    L = len(left)
    R = len(right)
    idxL = {v: i for i, v in enumerate(left)}
    idxR = {v: i for i, v in enumerate(right)}
    g = [[] for _ in range(L)]
    for u, v in edges:
        if color[u] == 0:
            g[idxL[u]].append(idxR[v])
        else:
            g[idxL[v]].append(idxR[u])

    # Hopcroft-Karp
    INF = 10**9
    pairU = [-1] * L
    pairV = [-1] * R
    dist = [0] * L

    def bfs():
        q = deque()
        for u in range(L):
            if pairU[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        d = INF
        while q:
            u = q.popleft()
            if dist[u] < d:
                for v in g[u]:
                    if pairV[v] == -1:
                        d = dist[u] + 1
                    elif dist[pairV[v]] == INF:
                        dist[pairV[v]] = dist[u] + 1
                        q.append(pairV[v])
        return d != INF

    def dfs(u):
        for v in g[u]:
            pu = pairV[v]
            if pairV[v] == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pairU[u] = v
                pairV[v] = u
                return True
        dist[u] = INF
        return False

    while bfs():
        for u in range(L):
            if pairU[u] == -1:
                dfs(u)

    # find min vertex cover
    visitedL = [False] * L
    visitedR = [False] * R
    q = deque()
    for u in range(L):
        if pairU[u] == -1:
            q.append(u)
            visitedL[u] = True
    while q:
        u = q.popleft()
        for v in g[u]:
            if not visitedR[v] and pairU[u] != v:  # edge not in matching
                visitedR[v] = True
                pu = pairV[v]
                if pu != -1 and not visitedL[pu]:
                    visitedL[pu] = True
                    q.append(pu)

    independent = []
    for i, v in enumerate(left):
        if visitedL[i]:  # left ∩ Z
            independent.append(v + 1)
    for i, v in enumerate(right):
        if not visitedR[i]:  # right \ Z
            independent.append(v + 1)

    if len(independent) >= k:
        print("YES")
        print(' '.join(map(str, independent[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
