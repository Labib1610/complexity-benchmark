import sys

# Increase recursion depth for DFS in Hopcroft-Karp
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    if k > n:
        print("NO")
        return

    head = [-1] * (n + 1)
    to = [0] * (2 * m)
    nxt = [0] * (2 * m)
    edge_cnt = 0

    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2

        to[edge_cnt] = v
        nxt[edge_cnt] = head[u]
        head[u] = edge_cnt
        edge_cnt += 1

        to[edge_cnt] = u
        nxt[edge_cnt] = head[v]
        head[v] = edge_cnt
        edge_cnt += 1

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            q = [i]
            head_q = 0
            while head_q < len(q):
                curr = q[head_q]
                head_q += 1
                c = color[curr] ^ 1
                e = head[curr]
                while e != -1:
                    neighbor = to[e]
                    if color[neighbor] == -1:
                        color[neighbor] = c
                        q.append(neighbor)
                    e = nxt[e]

    pair_u = [0] * (n + 1)
    pair_v = [0] * (n + 1)
    dist = [0] * (n + 1)
    cur_edge = [0] * (n + 1)
    INF = 10**9

    def bfs():
        q = []
        for u in range(1, n + 1):
            if color[u] == 0:
                if pair_u[u] == 0:
                    dist[u] = 0
                    q.append(u)
                else:
                    dist[u] = INF

        dist[0] = INF
        head_q = 0
        while head_q < len(q):
            u = q[head_q]
            head_q += 1
            if dist[u] < dist[0]:
                e = head[u]
                while e != -1:
                    v = to[e]
                    u_next = pair_v[v]
                    if dist[u_next] == INF:
                        dist[u_next] = dist[u] + 1
                        q.append(u_next)
                    e = nxt[e]
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            e = cur_edge[u]
            while e != -1:
                v = to[e]
                u_next = pair_v[v]
                if dist[u_next] == dist[u] + 1:
                    if dfs(u_next):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
                e = nxt[e]
                cur_edge[u] = e
            dist[u] = INF
            return False
        return True

    while bfs():
        for u in range(1, n + 1):
            cur_edge[u] = head[u]
        for u in range(1, n + 1):
            if color[u] == 0 and pair_u[u] == 0:
                dfs(u)

    visited = [False] * (n + 1)
    q = []
    for u in range(1, n + 1):
        if color[u] == 0 and pair_u[u] == 0:
            visited[u] = True
            q.append(u)

    head_q = 0
    while head_q < len(q):
        curr = q[head_q]
        head_q += 1
        if color[curr] == 0:
            e = head[curr]
            while e != -1:
                v = to[e]
                if v != pair_u[curr] and not visited[v]:
                    visited[v] = True
                    q.append(v)
                e = nxt[e]
        else:
            u = pair_v[curr]
            if u != 0 and not visited[u]:
                visited[u] = True
                q.append(u)

    mis = []
    for i in range(1, n + 1):
        if color[i] == 0 and visited[i]:
            mis.append(i)
        elif color[i] == 1 and not visited[i]:
            mis.append(i)

    if len(mis) >= k:
        print("YES")
        print(" ".join(map(str, mis[:k])))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
