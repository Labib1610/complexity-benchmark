```python
import sys
from collections import defaultdict, deque

def solve():
    import sys
    sys.setrecursionlimit(1000000)
    
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(set)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # If there are no edges, we can pick any k nodes
    if m == 0:
        print("YES")
        print(' '.join(map(str, range(1, min(k+1, n+1)))))
        return
    
    # Find bipartition using BFS/DFS
    color = {}
    camp1 = []
    camp2 = []
    
    visited = [False] * (n + 1)
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0
        camp1.append(start) if color[start] == 0 else camp2.append(start)
        visited[start] = True
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = 1 - current_color
                    if color[neighbor] == 0:
                        camp1.append(neighbor)
                    else:
                        camp2.append(neighbor)
                    queue.append(neighbor)
                elif color[neighbor] == current_color:
                    # This should never happen with the given constraints
                    return False
        return True
    
    # Check all components
    for i in range(1, n + 1):
        if not visited[i]:
            if not bfs(i):
                # Not bipartite - but problem guarantees it is
                pass
    
    # Now we have two camps: camp1 and camp2
    # An all-strangers group can only contain nodes from the same camp
    # So we need to find if either camp has at least k nodes
    
    if len(camp1) >= k:
        print("YES")
        print(' '.join(map(str, camp1[:k])))
    elif len(camp2) >= k:
        print("YES")
        print(' '.join(map(str, camp2[:k])))
    else:
        print("NO")

solve()
```
