```python
import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

n = next(it)
m = next(it)
k = next(it)

adj = [[] for _ in range(n)]
for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    adj[u].append(v)
    adj[v].append(u)

del data, it

# Recover a bipartition.
color = bytearray([255]) * n
left = []

for start in range(n):
    if color[start] != 255:
        continue
    color[start] = 0
    stack = [start]

    while stack:
        u = stack.pop()
        if color[u] == 0:
            left.append(u)
        next_color = color[u] ^ 1
        for v in adj[u]:
            if color[v] == 255:
                color[v] = next_color
                stack.append(v)

# Either side of a bipartition is an independent set.
if len(left) >= k:
    answer = left[:k]
    sys.stdout.write("YES\n" + " ".join(str(x + 1) for x in answer))
    sys.exit()

if n - len(left) >= k:
    answer = []
    for i in range(n):
        if color[i] == 1:
            answer.append(i)
            if len(answer) == k:
                break
    sys.stdout.write("YES\n" + " ".join(str(x + 1) for x in answer))
    sys.exit()

# Hopcroft-Karp maximum matching.
match = [-1] * n
dist = [-1] * n
matching_size = 0

while True:
    q = deque()
    for u in left:
        if match[u] == -1:
            dist[u] = 0
            q.append(u)
        else:
            dist[u] = -1

    best = -1

    while q:
        u = q.popleft()
        du = dist[u]

        if best != -1 and du > best:
            continue

        for v in adj[u]:
            w = match[v]
            if w == -1:
                if best == -1:
                    best = du
            elif (best == -1 or du < best) and dist[w] == -1:
                dist[w] = du + 1
                q.append(w)

    if best == -1:
        break

    ptr = [0] * n

    def dfs(u):
        while ptr[u] < len(adj[u]):
            v = adj[u][ptr[u]]
            ptr[u] += 1
            w = match[v]

            if w == -1:
                if dist[u] == best:
                    match[u] = v
                    match[v] = u
                    return True
            elif dist[u] < best and dist[w] == dist[u] + 1 and dfs(w):
                match[u] = v
                match[v] = u
                return True

        dist[u] = -1
        return False

    augmented = 0
    for u in left:
        if match[u] == -1 and dist[u] == 0 and dfs(u):
            augmented += 1

    matching_size += augmented

if n - matching_size < k:
    sys.stdout.write("NO")
    sys.exit()

# From a maximum matching, find the maximum independent set:
# reachable left vertices plus unreachable right vertices.
visited = bytearray(n)
q = deque()

for u in left:
    if match[u] == -1:
        visited[u] = 1
        q.append(u)

while q:
    u = q.popleft()
    matched_v = match[u]

    for v in adj[u]:
        if v == matched_v:
            continue
        if not visited[v]:
            visited[v] = 1
            w = match[v]
            if w != -1 and not visited[w]:
                visited[w] = 1
                q.append(w)

answer = []
for i in range(n):
    if (color[i] == 0 and visited[i]) or (color[i] == 1 and not visited[i]):
        answer.append(i + 1)
        if len(answer) == k:
            break

sys.stdout.write("YES\n" + " ".join(map(str, answer)))
```
