```python
import sys

# Increase recursion depth just in case, though HK can be iterative/DFS with small depth
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # 2-color the graph into L (color 0) and R (color 1)
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            stack = [i]
            while stack:
                u = stack.pop()
                c = color[u]
                nc = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = nc
                        stack.append(v)

    # Hopcroft-Karp algorithm
    # L vertices: color == 0, R vertices: color == 1
    # match_l[u] = v (in R), match_r[v] = u (in L)
    match_l = [0] * (n + 1)
    match_r = [0] * (n + 1)
    dist = [0] * (n + 1)
    
    # We only need adj edges directed from L to R
    adj_l = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        if color[u] == 0:
            adj_l[u] = adj[u]
            
    INF = 10**9
    
    def bfs():
        from collections import deque
        queue = deque()
        for u in range(1, n + 1):
            if color[u] == 0:
                if match_l[u] == 0:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj_l[u]:
                    mv = match_r[v]
                    if dist[mv] == INF:
                        dist[mv] = dist[u] + 1
                        queue.append(mv)
        return dist[0] != INF

    # Pointer for DFS
    ptr = [0] * (n + 1)
    def dfs(u):
        if u != 0:
            adj_u = adj_l[u]
            while ptr[u] < len(adj_u):
                v = adj_u[ptr[u]]
                ptr[u] += 1
                mv = match_r[v]
                if dist[mv] == dist[u] + 1:
                    if dfs(mv):
                        match_l[u] = v
                        match_r[v] = u
                        return True
            dist[u] = INF
            return False
        return True

    while bfs():
        for i in range(1, n + 1):
            if color[i] == 0:
                ptr[i] = 0
        for u in range(1, n + 1):
            if color[u] == 0 and match_l[u] == 0:
                dfs(u)

    # Find reachable vertices in alternating graph starting from unmatched vertices in L
    # Reachable from unmatched L via:
    # L -> R along non-matching edges (all edges in adj_l except match_l[u])
    # R -> L along matching edges (match_r[v])
    visited_l = [False] * (n + 1)
    visited_r = [False] * (n + 1)
    
    stack = []
    for u in range(1, n + 1):
        if color[u] == 0 and match_l[u] == 0:
            visited_l[u] = True
            stack.append(u)
            
    while stack:
        u = stack.pop()
        for v in adj_l[u]:
            if v != match_l[u] and not visited_r[v]:
                visited_r[v] = True
                mu = match_r[v]
                if mu != 0 and not visited_l[mu]:
                    visited_l[mu] = True
                    stack.append(mu)

    # Max Independent Set:
    # (L \cap Z) \cup (R \setminus Z)
    mis = []
    for u in range(1, n + 1):
        if color[u] == 0:
            if visited_l[u]:
                mis.append(u)
        else:
            if not visited_r[u]:
                mis.append(u)

    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
