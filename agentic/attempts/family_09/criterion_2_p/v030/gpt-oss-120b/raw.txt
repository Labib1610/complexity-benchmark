```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it))
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    # bipartition via BFS coloring
    color = [-1]*(n+1)
    for start in range(1, n+1):
        if color[start]==-1:
            color[start]=0
            q=deque([start])
            while q:
                u=q.popleft()
                for v in adj[u]:
                    if color[v]==-1:
                        color[v]=color[u]^1
                        q.append(v)

    left_nodes = [i for i in range(1,n+1) if color[i]==0]
    right_nodes = [i for i in range(1,n+1) if color[i]==1]

    # map node id -> index in left/right arrays
    left_id = {v:i for i,v in enumerate(left_nodes)}
    right_id = {v:i for i,v in enumerate(right_nodes)}
    L = len(left_nodes)
    R = len(right_nodes)

    g = [[] for _ in range(L)]
    for u in left_nodes:
        for v in adj[u]:
            if color[v]==1:
                g[left_id[u]].append(right_id[v])

    # Hopcroft–Karp
    INF = 10**9
    pairU = [-1]*L
    pairV = [-1]*R
    dist = [0]*L

    def bfs():
        q = deque()
        for u in range(L):
            if pairU[u]==-1:
                dist[u]=0
                q.append(u)
            else:
                dist[u]=INF
        d = INF
        while q:
            u = q.popleft()
            if dist[u] < d:
                for v in g[u]:
                    if pairV[v]==-1:
                        d = dist[u]+1
                    elif dist[pairV[v]]==INF:
                        dist[pairV[v]] = dist[u]+1
                        q.append(pairV[v])
        return d!=INF

    def dfs(u):
        for v in g[u]:
            if pairV[v]==-1 or (dist[pairV[v]]==dist[u]+1 and dfs(pairV[v])):
                pairU[u]=v
                pairV[v]=u
                return True
        dist[u]=INF
        return False

    matching=0
    while bfs():
        for u in range(L):
            if pairU[u]==-1 and dfs(u):
                matching+=1

    # Minimum vertex cover from max matching
    # BFS on alternating graph from free left vertices
    visL = [False]*L
    visR = [False]*R
    q = deque()
    for u in range(L):
        if pairU[u]==-1:
            q.append(u)
            visL[u]=True
    while q:
        u = q.popleft()
        for v in g[u]:
            if not visR[v] and pairU[u]!=v:   # edge not in matching
                visR[v]=True
                if pairV[v]!=-1 and not visL[pairV[v]]:
                    visL[pairV[v]]=True
                    q.append(pairV[v])

    # min vertex cover = (L \ visL) ∪ (R ∩ visR)
    in_cover = [False]*(n+1)
    for idx,u in enumerate(left_nodes):
        if not visL[idx]:
            in_cover[u]=True
    for idx,v in enumerate(right_nodes):
        if visR[idx]:
            in_cover[v]=True

    independent = [i for i in range(1,n+1) if not in_cover[i]]
    if len(independent) >= k:
        print("YES")
        print(' '.join(map(str, independent[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
