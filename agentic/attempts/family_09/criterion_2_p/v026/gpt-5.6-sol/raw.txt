import sys

def main():
    input_buffer = sys.stdin.buffer
    n, m, k = map(int, input_buffer.readline().split())

    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input_buffer.readline().split())
        graph[u].append(v)
        graph[v].append(u)

    color = bytearray(b'\xff') * (n + 1)
    queue = []
    head = 0

    for start in range(1, n + 1):
        if color[start] != 255:
            continue
        color[start] = 0
        queue.append(start)

        while head < len(queue):
            u = queue[head]
            head += 1
            next_color = color[u] ^ 1
            for v in graph[u]:
                if color[v] == 255:
                    color[v] = next_color
                    queue.append(v)

    del queue

    left = [u for u in range(1, n + 1) if color[u] == 0]
    mate = [0] * (n + 1)
    dist = [0] * (n + 1)
    inf = n + 1

    while True:
        bfs_queue = []
        for u in left:
            if mate[u] == 0:
                dist[u] = 0
                bfs_queue.append(u)
            else:
                dist[u] = inf

        shortest = inf
        head = 0
        while head < len(bfs_queue):
            u = bfs_queue[head]
            head += 1
            du = dist[u]

            if du >= shortest:
                continue

            for v in graph[u]:
                paired_left = mate[v]
                if paired_left == 0:
                    shortest = du + 1
                elif dist[paired_left] == inf and du + 1 < shortest:
                    dist[paired_left] = du + 1
                    bfs_queue.append(paired_left)

        if shortest == inf:
            break

        ptr = [0] * (n + 1)

        for root in left:
            if mate[root] != 0 or dist[root] != 0:
                continue

            stack = [root]
            path_edges = []
            augmented = False

            while stack:
                u = stack[-1]
                adjacency = graph[u]
                advanced = False

                while ptr[u] < len(adjacency):
                    v = adjacency[ptr[u]]
                    ptr[u] += 1
                    paired_left = mate[v]

                    if paired_left == 0:
                        if dist[u] + 1 != shortest:
                            continue

                        mate[u] = v
                        mate[v] = u

                        for i in range(len(path_edges) - 1, -1, -1):
                            parent = stack[i]
                            edge_v = path_edges[i]
                            mate[parent] = edge_v
                            mate[edge_v] = parent

                        augmented = True
                        break

                    if dist[paired_left] == dist[u] + 1:
                        stack.append(paired_left)
                        path_edges.append(v)
                        advanced = True
                        break

                if augmented:
                    break

                if advanced:
                    continue

                dist[u] = inf
                stack.pop()
                if path_edges:
                    path_edges.pop()

    reachable_left = bytearray(n + 1)
    reachable_right = bytearray(n + 1)
    alternating_queue = []

    for u in left:
        if mate[u] == 0:
            reachable_left[u] = 1
            alternating_queue.append(u)

    head = 0
    while head < len(alternating_queue):
        u = alternating_queue[head]
        head += 1
        matched_v = mate[u]

        for v in graph[u]:
            if v == matched_v or reachable_right[v]:
                continue
            reachable_right[v] = 1
            paired_left = mate[v]
            if paired_left != 0 and not reachable_left[paired_left]:
                reachable_left[paired_left] = 1
                alternating_queue.append(paired_left)

    answer = []

    for u in left:
        if reachable_left[u]:
            answer.append(u)
            if len(answer) == k:
                break

    if len(answer) < k:
        for v in range(1, n + 1):
            if color[v] == 1 and not reachable_right[v]:
                answer.append(v)
                if len(answer) == k:
                    break

    if len(answer) < k:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, answer)) + "\n")

if __name__ == "__main__":
    main()
