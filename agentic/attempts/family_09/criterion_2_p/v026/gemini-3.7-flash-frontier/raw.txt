import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * (n + 1)
    
    # 2-color the graph
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            stack = [i]
            while stack:
                u = stack.pop()
                c = color[u]
                nc = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = nc
                        stack.append(v)

    # Prepare directed edges from Left (color 0) to Right (color 1)
    adj_L = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        if color[u] == 0:
            adj_L[u] = adj[u]

    # Hopcroft-Karp algorithm
    pair_u = [0] * (n + 1) # u in L -> matched v in R
    pair_v = [0] * (n + 1) # v in R -> matched u in L
    dist = [0] * (n + 1)
    INF = 10**9

    def bfs():
        queue = []
        for u in range(1, n + 1):
            if color[u] == 0:
                if pair_u[u] == 0:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = INF
        dist[0] = INF
        
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            if dist[u] < dist[0]:
                for v in adj_L[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        queue.append(pu)
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj_L[u]:
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, n + 1):
            if color[u] == 0 and pair_u[u] == 0:
                if dfs(u):
                    matching += 1

    # Find Minimum Vertex Cover, then Maximum Independent Set
    # Min Vertex Cover:
    # 1. Start DFS/BFS from all unmatched vertices in L, alternating paths: L to R (non-matching edges), R to L (matching edges).
    # 2. Vertices reachable: Z
    # 3. MVC = (L \ Z) U (R \cap Z)
    # 4. MIS = (L \cap Z) U (R \ Z)
    
    visited_L = [False] * (n + 1)
    visited_R = [False] * (n + 1)
    
    stack = []
    for u in range(1, n + 1):
        if color[u] == 0 and pair_u[u] == 0:
            visited_L[u] = True
            stack.append(u)
            
    while stack:
        u = stack.pop()
        for v in adj_L[u]:
            if pair_u[u] != v: # non-matching edge from L to R
                if not visited_R[v]:
                    visited_R[v] = True
                    pu = pair_v[v]
                    if pu != 0 and not visited_L[pu]:
                        visited_L[pu] = True
                        stack.append(pu)

    mis = []
    for u in range(1, n + 1):
        if color[u] == 0:
            if visited_L[u]:
                mis.append(u)
        else:
            if not visited_R[u]:
                mis.append(u)

    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
