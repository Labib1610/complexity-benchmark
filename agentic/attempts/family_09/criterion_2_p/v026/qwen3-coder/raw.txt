```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m, k = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    if m == 0:
        # No conflicts, pick any k nodes
        print("YES")
        print(' '.join(map(str, range(1, k+1))))
        return
    
    # Since the graph is bipartite (conflicts only between tiers),
    # we can try to 2-color it to find the two partitions
    color = {}
    visited = set()
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0
        visited.add(start)
        while queue:
            u = queue.popleft()
            for v in adj[u]:
                if v not in visited:
                    visited.add(v)
                    color[v] = 1 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    # This should not happen according to problem statement
                    return False
        return True
    
    # Find all components and color them
    for i in range(1, n+1):
        if i not in visited:
            if not bfs(i):
                # Not bipartite, but problem guarantees it is
                pass
    
    # Separate nodes into two sets based on color
    set0 = [node for node in color if color[node] == 0]
    set1 = [node for node in color if color[node] == 1]
    
    # Check if we can take k nodes from either set
    if len(set0) >= k:
        print("YES")
        print(' '.join(map(str, set0[:k])))
    elif len(set1) >= k:
        print("YES")
        print(' '.join(map(str, set1[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
