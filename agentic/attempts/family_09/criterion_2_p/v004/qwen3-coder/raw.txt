```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    edges = []
    adj = defaultdict(list)
    edge_index = 3
    for _ in range(m):
        u = int(data[edge_index])
        v = int(data[edge_index+1])
        edge_index += 2
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # If no edges, any k nodes will do
    if m == 0:
        print("YES")
        print(' '.join(map(str, range(1, min(k+1, n+1)))))
        return
    
    # Bipartite check and get partitions
    color = [-1] * (n+1)
    partition = [[], []]
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0
        partition[0].append(start)
        while queue:
            node = queue.popleft()
            for neighbor in adj[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - color[node]
                    partition[color[neighbor]].append(neighbor)
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    # Should not happen as per problem statement
                    return False
        return True
    
    for i in range(1, n+1):
        if color[i] == -1:
            if not bfs(i):
                # This should not happen according to problem statement
                return
    
    # Now we have two partitions
    part0 = partition[0]
    part1 = partition[1]
    
    # Check if we can take k from one side
    if len(part0) >= k:
        print("YES")
        print(' '.join(map(str, part0[:k])))
    elif len(part1) >= k:
        print("YES")
        print(' '.join(map(str, part1[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
