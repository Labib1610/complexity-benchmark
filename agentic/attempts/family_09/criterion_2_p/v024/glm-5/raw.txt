import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))

    if k == 0:
        sys.stdout.write("YES\n\n")
        return

    if m == 0:
        if k <= n:
            out = "YES\n" + " ".join(str(i) for i in range(1, k + 1))
            sys.stdout.write(out)
            if not out.endswith('\n'):
                sys.stdout.write('\n')
        else:
            sys.stdout.write("NO\n")
        return

    head = [-1] * (n + 1)
    to = [0] * (2 * m)
    nxt = [0] * (2 * m)
    idx = 0
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        to[idx] = v
        nxt[idx] = head[u]
        head[u] = idx
        idx += 1
        to[idx] = u
        nxt[idx] = head[v]
        head[v] = idx
        idx += 1

    color = [0] * (n + 1)
    comp_sizes = []
    comp_nodes = []

    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            stack = [start]
            cnt1 = 0
            cnt2 = 0
            nodes1 = []
            nodes2 = []
            while stack:
                u = stack.pop()
                if color[u] == 1:
                    cnt1 += 1
                    nodes1.append(u)
                else:
                    cnt2 += 1
                    nodes2.append(u)
                e = head[u]
                while e != -1:
                    v = to[e]
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    e = nxt[e]
            comp_sizes.append((cnt1, cnt2))
            comp_nodes.append((nodes1, nodes2))

    total = sum(max(a, b) for a, b in comp_sizes)
    if k > total:
        sys.stdout.write("NO\n")
        return

    dp = {0: 0}
    parent = {}
    for i, (a, b) in enumerate(comp_sizes):
        new_dp = {}
        for s, idx_val in dp.items():
            ns1 = s + a
            if ns1 not in new_dp or new_dp[ns1] > idx_val:
                new_dp[ns1] = idx_val
                parent[(i + 1, ns1)] = (idx_val, s, 1)
            ns2 = s + b
            if ns2 not in new_dp or new_dp[ns2] > idx_val:
                new_dp[ns2] = idx_val
                parent[(i + 1, ns2)] = (idx_val, s, 2)
        dp = new_dp

    if k not in dp:
        sys.stdout.write("NO\n")
        return

    choices = [0] * len(comp_sizes)
    cur = k
    for i in range(len(comp_sizes), 0, -1):
        _, prev, choice = parent[(i, cur)]
        choices[i - 1] = choice
        cur = prev

    result = []
    for i, choice in enumerate(choices):
        nodes = comp_nodes[i][choice - 1]
        result.extend(nodes)

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, result)))
    sys.stdout.write("\n")

solve()
