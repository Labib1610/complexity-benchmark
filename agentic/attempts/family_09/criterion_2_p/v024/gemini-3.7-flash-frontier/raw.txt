import sys

# Increase recursion depth just in case
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # 2-color the graph
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            queue = [i]
            for u in queue:
                c = color[u]
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        queue.append(v)

    # Hopcroft-Karp algorithm
    pair_u = [0] * (n + 1)
    pair_v = [0] * (n + 1)
    dist = [0] * (n + 1)
    
    INF = 10**9
    
    L = [i for i in range(1, n + 1) if color[i] == 1]
    
    def bfs():
        queue = []
        for u in L:
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    while bfs():
        for u in L:
            if pair_u[u] == 0:
                dfs(u)
                
    # Find reachable vertices Z from unmatched vertices in L
    # Reachable via alternating paths: L -> R along non-matching edges, R -> L along matching edges
    visited = [False] * (n + 1)
    queue = []
    for u in L:
        if pair_u[u] == 0:
            visited[u] = True
            queue.append(u)
            
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        if color[u] == 1:
            for v in adj[u]:
                if v != pair_u[u] and not visited[v]:
                    visited[v] = True
                    queue.append(v)
        else:
            v = pair_v[u]
            if v != 0 and not visited[v]:
                visited[v] = True
                queue.append(v)

    # Maximum Independent Set:
    # (L and visited) OR (R and not visited)
    mis = []
    for i in range(1, n + 1):
        if color[i] == 1:
            if visited[i]:
                mis.append(i)
        else:
            if not visited[i]:
                mis.append(i)
                
    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
