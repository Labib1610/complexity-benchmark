```python
import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    k = int(data[2])
    
    adj = [[] for _ in range(n + 1)]
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        
    # 2-color the graph
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            color[i] = 0
            q = [i]
            for u in q:
                c = color[u]
                nc = 1 - c
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = nc
                        q.append(v)

    # Left and Right partitions
    # Directed bipartite graph: left_adj for L -> R edges
    left_adj = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        if color[u] == 0:
            left_adj[u] = adj[u]

    # Hopcroft-Karp algorithm
    match_L = [0] * (n + 1)
    match_R = [0] * (n + 1)
    dist = [0] * (n + 1)
    
    INF = 10**9
    
    def bfs():
        q = deque()
        for u in range(1, n + 1):
            if color[u] == 0:
                if match_L[u] == 0:
                    dist[u] = 0
                    q.append(u)
                else:
                    dist[u] = INF
        dist[0] = INF
        
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in left_adj[u]:
                    mv = match_R[v]
                    if dist[mv] == INF:
                        dist[mv] = dist[u] + 1
                        q.append(mv)
        return dist[0] != INF

    def dfs(u):
        for v in left_adj[u]:
            mv = match_R[v]
            if dist[mv] == dist[u] + 1:
                if mv == 0 or dfs(mv):
                    match_L[u] = v
                    match_R[v] = u
                    return True
        dist[u] = INF
        return False

    while bfs():
        for u in range(1, n + 1):
            if color[u] == 0 and match_L[u] == 0:
                dfs(u)

    # Finding Maximum Independent Set from matching
    # BFS from all unmatched vertices in L along alternating paths:
    # L -> R via non-matching edges
    # R -> L via matching edges
    visited_L = [False] * (n + 1)
    visited_R = [False] * (n + 1)
    
    q = deque()
    for u in range(1, n + 1):
        if color[u] == 0 and match_L[u] == 0:
            visited_L[u] = True
            q.append(u)
            
    while q:
        u = q.popleft()
        for v in left_adj[u]:
            if not visited_R[v] and match_L[u] != v:
                visited_R[v] = True
                mu = match_R[v]
                if mu != 0 and not visited_L[mu]:
                    visited_L[mu] = True
                    q.append(mu)

    # MIS = (L & visited) | (R & not visited)
    mis = []
    for u in range(1, n + 1):
        if color[u] == 0:
            if visited_L[u]:
                mis.append(u)
        else:
            if not visited_R[u]:
                mis.append(u)
                
    if len(mis) >= k:
        print("YES")
        print(*(mis[:k]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
