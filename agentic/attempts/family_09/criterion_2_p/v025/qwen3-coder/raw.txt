import sys
from collections import defaultdict, deque

def read_input():
    n, m, k = map(int, sys.stdin.readline().split())
    edges = []
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    return n, m, k, edges, adj

def check_bipartite(n, adj):
    color = {}
    for i in range(1, n+1):
        if i not in color:
            queue = deque([i])
            color[i] = 0
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if neighbor not in color:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        return None, None
    left = [i for i in color if color[i] == 0]
    right = [i for i in color if color[i] == 1]
    return left, right

def main():
    n, m, k, edges, adj = read_input()
    
    # Check if the graph is bipartite
    left, right = check_bipartite(n, adj)
    
    if left is None:
        print("NO")
        return
    
    # Try to select k nodes from the larger side
    if len(left) >= k:
        print("YES")
        print(' '.join(map(str, left[:k])))
    elif len(right) >= k:
        print("YES")
        print(' '.join(map(str, right[:k])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
