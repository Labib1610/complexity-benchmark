import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])
    k = int(input_data[2])

    head = [-1] * (n + 1)
    to = [0] * (2 * m)
    nxt = [0] * (2 * m)
    edge_cnt = 0

    idx = 3
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2

        to[edge_cnt] = v
        nxt[edge_cnt] = head[u]
        head[u] = edge_cnt
        edge_cnt += 1

        to[edge_cnt] = u
        nxt[edge_cnt] = head[v]
        head[v] = edge_cnt
        edge_cnt += 1

    color = [-1] * (n + 1)
    q = [0] * (n + 1)

    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        color[i] = 0
        qh = 0
        qt = 0
        q[qt] = i
        qt += 1
        while qh < qt:
            u = q[qh]
            qh += 1
            c = color[u] ^ 1
            e = head[u]
            while e != -1:
                v = to[e]
                if color[v] == -1:
                    color[v] = c
                    q[qt] = v
                    qt += 1
                e = nxt[e]

    # Hopcroft-Karp algorithm
    pair_u = [0] * (n + 1)  # Left side matched with right
    pair_v = [0] * (n + 1)  # Right side matched with left
    dist = [0] * (n + 1)
    INF = 10**9

    def bfs():
        qh = 0
        qt = 0
        for u in range(1, n + 1):
            if color[u] == 0:
                if pair_u[u] == 0:
                    dist[u] = 0
                    q[qt] = u
                    qt += 1
                else:
                    dist[u] = INF
        dist[0] = INF

        while qh < qt:
            u = q[qh]
            qh += 1
            if dist[u] < dist[0]:
                e = head[u]
                while e != -1:
                    v = to[e]
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        q[qt] = pu
                        qt += 1
                    e = nxt[e]
        return dist[0] != INF

    # Iterative DFS using explicit stack
    cur_edge = [0] * (n + 1)

    def dfs(u):
        if u != 0:
            e = cur_edge[u]
            while e != -1:
                v = to[e]
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_v[v] = u
                        pair_u[u] = v
                        cur_edge[u] = e
                        return True
                e = nxt[e]
            cur_edge[u] = -1
            dist[u] = INF
            return False
        return True

    while bfs():
        for u in range(1, n + 1):
            if color[u] == 0:
                cur_edge[u] = head[u]
        for u in range(1, n + 1):
            if color[u] == 0 and pair_u[u] == 0:
                dfs(u)

    # Find reachable vertices in alternating graph from unmatched vertices in L (color == 0)
    visited = [False] * (n + 1)
    qh = 0
    qt = 0
    for u in range(1, n + 1):
        if color[u] == 0 and pair_u[u] == 0:
            visited[u] = True
            q[qt] = u
            qt += 1

    while qh < qt:
        u = q[qh]
        qh += 1
        if color[u] == 0:
            # From L to R: non-matching edges
            e = head[u]
            while e != -1:
                v = to[e]
                if v != pair_u[u] and not visited[v]:
                    visited[v] = True
                    q[qt] = v
                    qt += 1
                e = nxt[e]
        else:
            # From R to L: matching edge
            pu = pair_v[u]
            if pu != 0 and not visited[pu]:
                visited[pu] = True
                q[qt] = pu
                qt += 1

    # Max independent set:
    # (L and visited) OR (R and not visited)
    mis = []
    for i in range(1, n + 1):
        if color[i] == 0:
            if visited[i]:
                mis.append(i)
        else:
            if not visited[i]:
                mis.append(i)

    if len(mis) < k:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, mis[:k])) + "\n")


if __name__ == "__main__":
    solve()
