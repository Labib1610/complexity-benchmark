We can model each valve as a Boolean variable and each constraint as a clause with three literals.  
This is exactly a **3-SAT** problem, but with a special structure: each clause has exactly three literals, and variables can repeat within a clause.

Since \( n \le 90 \) and \( m \le 400 \), we can solve it using a backtracking algorithm with unit propagation and pure literal elimination, which is efficient enough for this size.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        v1 = int(next(it))
        v2 = int(next(it))
        v3 = int(next(it))
        clause = []
        for v in (v1, v2, v3):
            var = abs(v) - 1
            val = 1 if v > 0 else 0
            clause.append((var, val))
        clauses.append(clause)
    
    assignment = [-1] * n  # -1 unassigned, 0 closed, 1 open
    
    # Check if a clause is satisfied
    def clause_satisfied(clause, assignment):
        for var, val in clause:
            if assignment[var] == val:
                return True
        return False
    
    # Check if a clause is unsatisfiable under current assignment
    def clause_conflict(clause, assignment):
        for var, val in clause:
            if assignment[var] == -1 or assignment[var] == val:
                return False
        return True
    
    # Unit propagation: find clauses with only one unassigned literal
    def propagate(assignment, clauses):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                if clause_satisfied(clause, assignment):
                    continue
                unassigned = []
                for var, val in clause:
                    if assignment[var] == -1:
                        unassigned.append((var, val))
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    var, val = unassigned[0]
                    if assignment[var] == -1:
                        assignment[var] = val
                        changed = True
                    elif assignment[var] != val:
                        return False
        return True
    
    # Pure literal elimination
    def pure_literal_elimination(assignment, clauses):
        # Count occurrences of each literal
        pos_count = [0] * n
        neg_count = [0] * n
        for clause in clauses:
            if clause_satisfied(clause, assignment):
                continue
            for var, val in clause:
                if assignment[var] == -1:
                    if val == 1:
                        pos_count[var] += 1
                    else:
                        neg_count[var] += 1
        changed = False
        for var in range(n):
            if assignment[var] == -1:
                if pos_count[var] > 0 and neg_count[var] == 0:
                    assignment[var] = 1
                    changed = True
                elif neg_count[var] > 0 and pos_count[var] == 0:
                    assignment[var] = 0
                    changed = True
        return changed
    
    def backtrack(assignment, clauses):
        # Propagate
        if not propagate(assignment, clauses):
            return False
        # Pure literal elimination
        while pure_literal_elimination(assignment, clauses):
            if not propagate(assignment, clauses):
                return False
        
        # Check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            if not clause_satisfied(clause, assignment):
                all_sat = False
                break
        if all_sat:
            return True
        
        # Choose an unassigned variable
        var = -1
        for i in range(n):
            if assignment[i] == -1:
                var = i
                break
        if var == -1:
            return False
        
        # Try both values
        for val in (0, 1):
            assignment[var] = val
            if backtrack(assignment, clauses):
                return True
            assignment[var] = -1
        return False
    
    if backtrack(assignment, clauses):
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
