def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    clauses = []
    idx = 2
    for _ in range(m):
        clause = []
        for _ in range(3):
            lit = int(data[idx])
            idx += 1
            clause.append(lit)
        clauses.append(clause)
    
    # We'll use a backtracking approach with pruning
    # Let's try all 2^n possibilities, but we can do better with constraint checking
    
    # Since n <= 90 and m <= 400, a brute force won't work.
    # This is essentially a 3-SAT problem. We need a smarter approach.
    
    # Let's implement a simple backtracking with unit propagation for speed
    
    # First, let's build a mapping from variables to clauses they appear in
    # and also represent clauses in terms of boolean variables (0 for D, 1 for L)
    
    # For each position i (1-based), we have variable i.
    # Positive literal i means position i is L (value 1)
    # Negative literal -i means position i is D (value 0)
    
    # Convert the problem to a satisfiability problem
    # Each clause is a disjunction of 3 literals
    
    # We'll use a recursive backtracking algorithm
    
    def is_satisfied(assignments):
        # Check if all clauses are satisfied
        for clause in clauses:
            satisfied = False
            for lit in clause:
                var = abs(lit)
                required_val = 1 if lit > 0 else 0
                if var-1 < len(assignments) and assignments[var-1] == required_val:
                    satisfied = True
                    break
            if not satisfied:
                return False
        return True
    
    def backtrack(pos, assignments):
        if pos == n:
            return is_satisfied(assignments)
        
        # Try setting position pos to 1 (L)
        assignments.append(1)
        if backtrack(pos+1, assignments):
            return True
        assignments.pop()
        
        # Try setting position pos to 0 (D)
        assignments.append(0)
        if backtrack(pos+1, assignments):
            return True
        assignments.pop()
        
        return False
    
    # This is too slow for large n. Let's optimize with early clause checking.
    
    def backtrack_opt(pos, assignments):
        # Check current partial assignment against all clauses
        # Only check clauses where all literals are decided
        for clause in clauses:
            unknown = False
            satisfied = False
            for lit in clause:
                var = abs(lit)
                if var-1 >= len(assignments):
                    unknown = True
                    break
                required_val = 1 if lit > 0 else 0
                if assignments[var-1] == required_val:
                    satisfied = True
                    break
            if not unknown and not satisfied:
                return False  # Current partial assignment violates this clause
                
        if pos == n:
            return True  # All positions assigned and no violated clauses
            
        # Try setting position pos to 1 (L)
        assignments.append(1)
        if backtrack_opt(pos+1, assignments):
            return True
        assignments.pop()
        
        # Try setting position pos to 0 (D)
        assignments.append(0)
        if backtrack_opt(pos+1, assignments):
            return True
        assignments.pop()
        
        return False
    
    # Even better optimization: for each clause, if only one variable is unassigned,
    # we can determine what value it must have to satisfy the clause (if any)
    
    def unit_propagate(assignments):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unknown_count = 0
                unknown_var = -1
                unknown_sign = 0
                satisfied = False
                
                for lit in clause:
                    var = abs(lit)
                    required_val = 1 if lit > 0 else 0
                    
                    if var-1 >= len(assignments):
                        unknown_count += 1
                        unknown_var = var
                        unknown_sign = lit
                    elif assignments[var-1] == required_val:
                        satisfied = True
                        break
                
                if satisfied:
                    continue
                    
                if unknown_count == 0:
                    return False  # Clause is unsatisfiable
                    
                if unknown_count == 1:
                    # Unit clause: only one way to satisfy it
                    req_val = 1 if unknown_sign > 0 else 0
                    if assignments[unknown_var-1] == -1:  # Unassigned
                        assignments[unknown_var-1] = req_val
                        changed = True
                    elif assignments[unknown_var-1] != req_val:
                        return False  # Conflict
                        
        return True
    
    def backtrack_cnf(pos, assignments):
        # Apply unit propagation
        original = list(assignments)
        if not unit_propagate(assignments):
            return False
            
        # Find next unassigned variable
        next_pos = pos
        while next_pos < n and assignments[next_pos] != -1:
            next_pos += 1
            
        if next_pos == n:
            return True  # All assigned
            
        # Try both values
        assignments[next_pos] = 1
        if backtrack_cnf(next_pos+1, assignments):
            return True
            
        assignments[next_pos] = 0
        if backtrack_cnf(next_pos+1, assignments):
            return True
            
        # Restore
        for i in range(n):
            assignments[i] = original[i]
        return False
    
    # Initialize assignments: -1 for unassigned
    assignments = [-1] * n
    if backtrack_cnf(0, assignments):
        print("YES")
        print(" ".join(str(x) for x in assignments))
    else:
        print("NO")

solve()
