import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        lits = [int(next(it)) for _ in range(3)]
        clauses.append(lits)

    # assignment: -1 unassigned, 0 false (D), 1 true (L)
    assign = [-1] * (n + 1)

    # For propagation we keep a stack of forced assignments to undo
    def literal_value(lit):
        var = abs(lit)
        val = assign[var]
        if val == -1:
            return None
        return val if lit > 0 else 1 - val

    def clause_status(cl):
        # returns (satisfied?, unit_lit or None, conflict?)
        sat = False
        unassigned = []
        for lit in cl:
            val = literal_value(lit)
            if val is None:
                unassigned.append(lit)
            elif val == 1:
                sat = True
                break
        if sat:
            return True, None, False
        if not unassigned:
            return False, None, True  # all false -> conflict
        if len(unassigned) == 1:
            return False, unassigned[0], False  # unit
        return False, None, False

    # propagate returns False on conflict, True otherwise
    def propagate(stack):
        while True:
            unit_found = False
            for cl in clauses:
                sat, unit_lit, conflict = clause_status(cl)
                if conflict:
                    return False
                if not sat and unit_lit is not None:
                    var = abs(unit_lit)
                    val = 1 if unit_lit > 0 else 0
                    if assign[var] == -1:
                        assign[var] = val
                        stack.append(var)
                        unit_found = True
                    elif assign[var] != val:
                        return False
            if not unit_found:
                break
        return True

    def dfs():
        # find unassigned var
        try:
            var = next(i for i in range(1, n + 1) if assign[i] == -1)
        except StopIteration:
            return True  # all assigned, solution found
        for val in (1, 0):  # try true then false
            stack = []
            assign[var] = val
            stack.append(var)
            if propagate(stack):
                if dfs():
                    return True
            # undo
            for v in stack:
                assign[v] = -1
        return False

    if dfs():
        print("YES")
        print(' '.join(str(assign[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
