```python
import sys

sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        lits = [int(next(it)), int(next(it)), int(next(it))]
        clauses.append(lits)

    # helper to get variable index 0-based and its sign
    def var(lit):
        return abs(lit) - 1

    # DPLL recursive solver
    def dpll(clauses, assign):
        # unit propagation & pure literal elimination loop
        while True:
            # remove satisfied clauses, detect conflicts, collect unit literals
            new_clauses = []
            unit_lit = None
            pure_counts = {}
            for cl in clauses:
                satisfied = False
                unassigned = []
                for lit in cl:
                    v = var(lit)
                    if v in assign:
                        val = assign[v]
                        if (lit > 0 and val) or (lit < 0 and not val):
                            satisfied = True
                            break
                    else:
                        unassigned.append(lit)
                        pure_counts[lit] = pure_counts.get(lit, 0) + 1
                if satisfied:
                    continue
                if not unassigned:
                    return None  # conflict
                if len(unassigned) == 1:
                    unit_lit = unassigned[0]
                    break
                new_clauses.append(unassigned)
            if unit_lit is not None:
                v = var(unit_lit)
                val = unit_lit > 0
                if v in assign:
                    if assign[v] != val:
                        return None
                else:
                    assign[v] = val
                clauses = new_clauses
                continue

            # pure literal elimination
            pure = None
            for lit, cnt in pure_counts.items():
                if -lit not in pure_counts:
                    pure = lit
                    break
            if pure is not None:
                v = var(pure)
                val = pure > 0
                assign[v] = val
                # remove clauses containing pure literal
                clauses = [cl for cl in new_clauses if pure not in cl]
                continue

            clauses = new_clauses
            break

        if not clauses:
            return assign

        # choose variable heuristically: first literal of first clause
        for lit in clauses[0]:
            v = var(lit)
            if v not in assign:
                break

        # try true then false
        for val in (True, False):
            new_assign = assign.copy()
            new_assign[v] = val
            res = dpll(clauses, new_assign)
            if res is not None:
                return res
        return None

    result = dpll(clauses, {})
    if result is None:
        print("NO")
    else:
        print("YES")
        out = []
        for i in range(n):
            out.append('1' if result.get(i, False) else '0')
        print(' '.join(out))

if __name__ == "__main__":
    solve()
```
