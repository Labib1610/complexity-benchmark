from collections import defaultdict
import sys

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    constraints = []
    for _ in range(m):
        l1, l2, l3 = map(int, sys.stdin.readline().split())
        constraints.append((l1, l2, l3))
    return n, m, constraints

def solve_2sat(n, clauses):
    # Create implication graph
    # Variables: 1..n (positive), -1..-n (negative)
    # Nodes: 1..n for x_i, n+1..2n for not x_i
    # So variable i maps to node i (if i>0) or node (n + |i|) (if i<0)
    def var_to_node(v):
        if v > 0:
            return v
        else:
            return n + (-v)
    
    # Adjacency list for implication graph
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    # Convert 3-SAT clause to implications
    # For clause (a or b or c), we add:
    # (not a => b or c) which is (not a => b) or (not a => c)
    # But this is still not 2-SAT. We need to reduce 3-SAT to 2-SAT.
    # We introduce auxiliary variables.
    
    # Actually, let's think differently. This is a 3-SAT problem, not 2-SAT.
    # But the problem asks us to solve it, so let's try a different approach.
    # Since n <= 90, we can try backtracking with pruning.
    # But that might be too slow for some cases.
    # Let's try to model it properly.
    
    # Each clause is (a or b or c).
    # We can convert to 2-SAT by introducing new variables, but that complicates things.
    # Let's use a SAT solver approach - specifically, since each clause has 3 literals,
    # this is 3-SAT, which is NP-complete. However, given the small constraints,
    # we can try a backtracking solution.
    
    # Let's implement a simple backtracking algorithm.
    def is_clause_satisfied(clause, assignment):
        for lit in clause:
            var = abs(lit)
            required = lit > 0
            if var in assignment:
                if assignment[var] == required:
                    return True
            else:
                # Variable not assigned yet, could be satisfied
                pass
        # Check if all literals are assigned and none are satisfied
        all_assigned = True
        for lit in clause:
            var = abs(lit)
            if var not in assignment:
                all_assigned = False
                break
        
        if all_assigned:
            # All literals are assigned, check if any is satisfied
            for lit in clause:
                var = abs(lit)
                required = lit > 0
                if assignment[var] == required:
                    return True
            return False  # None satisfied
        else:
            # Not all assigned, so potentially satisfiable
            return True
    
    def is_all_clauses_satisfied(assignment):
        for clause in clauses:
            if not is_clause_satisfied(clause, assignment):
                return False
        return True
    
    def backtrack(index, assignment):
        if index > n:
            if is_all_clauses_satisfied(assignment):
                return assignment.copy()
            return None
        
        # Try setting variable index to True (engaged)
        assignment[index] = 1
        if is_all_clauses_satisfied_partial(assignment, index):
            result = backtrack(index + 1, assignment)
            if result is not None:
                return result
        
        # Try setting variable index to False (disengaged)
        assignment[index] = 0
        if is_all_clauses_satisfied_partial(assignment, index):
            result = backtrack(index + 1, assignment)
            if result is not None:
                return result
        
        del assignment[index]
        return None
    
    def is_all_clauses_satisfied_partial(assignment, max_var):
        # Only check clauses that involve variables up to max_var
        for clause in clauses:
            # Check if clause can still be satisfied
            has_unassigned = False
            satisfied = False
            for lit in clause:
                var = abs(lit)
                required = lit > 0
                if var in assignment:
                    if assignment[var] == required:
                        satisfied = True
                        break
                else:
                    has_unassigned = True
            
            if not satisfied and not has_unassigned:
                return False
        return True
    
    # Optimized version
    def backtrack_opt(assignment):
        # Check if current assignment violates any clause
        for clause in clauses:
            satisfied = False
            unknown = False
            for lit in clause:
                var = abs(lit)
                required = lit > 0
                if var in assignment:
                    if assignment[var] == required:
                        satisfied = True
                        break
                else:
                    unknown = True
            
            if not satisfied and not unknown:
                return None  # Current assignment violates this clause
        
        # If all variables are assigned, we have a solution
        if len(assignment) == n:
            return assignment.copy()
        
        # Pick next unassigned variable
        # Simple heuristic: pick the first unassigned
        next_var = None
        for i in range(1, n+1):
            if i not in assignment:
                next_var = i
                break
        
        # Try setting to 1
        assignment[next_var] = 1
        result = backtrack_opt(assignment)
        if result is not None:
            return result
        
        # Try setting to 0
        assignment[next_var] = 0
        result = backtrack_opt(assignment)
        if result is not None:
            return result
        
        del assignment[next_var]
        return None
    
    final_assignment = backtrack_opt({})
    return final_assignment

def main():
    n, m, constraints = read_input()
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    assignment = solve_2sat(n, constraints)
    
    if assignment is None:
        print("NO")
    else:
        print("YES")
        result = []
        for i in range(1, n+1):
            result.append(str(assignment[i]))
        print(" ".join(result))

if __name__ == "__main__":
    main()
