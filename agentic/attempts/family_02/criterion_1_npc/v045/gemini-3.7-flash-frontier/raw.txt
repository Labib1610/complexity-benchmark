import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Simplify clause: remove duplicates
        c_unique = []
        tautology = False
        for lit in c:
            if -lit in c_unique:
                tautology = True
                break
            if lit not in c_unique:
                c_unique.append(lit)
        if not tautology:
            clauses.append(c_unique)

    # WalkSAT for quick SAT check
    def walksat(clauses, n, max_tries=50, max_flips=3000, p=0.3):
        if not clauses:
            return [0] * n
        for _ in range(max_tries):
            assignment = {i: random.choice([True, False]) for i in range(1, n + 1)}
            for _ in range(max_flips):
                unsat = []
                for c in clauses:
                    if not any((assignment[abs(lit)] if lit > 0 else not assignment[abs(lit)]) for lit in c):
                        unsat.append(c)
                if not unsat:
                    return [1 if assignment[i] else 0 for i in range(1, n + 1)]
                
                c = random.choice(unsat)
                if random.random() < p:
                    var = abs(random.choice(c))
                else:
                    # Pick var that minimizes number of broken clauses
                    best_var = abs(c[0])
                    min_breaks = float('inf')
                    for lit in c:
                        var = abs(lit)
                        assignment[var] = not assignment[var]
                        breaks = 0
                        for cl in clauses:
                            if not any((assignment[abs(l)] if l > 0 else not assignment[abs(l)]) for l in cl):
                                breaks += 1
                        if breaks < min_breaks:
                            min_breaks = breaks
                            best_var = var
                        assignment[var] = not assignment[var]
                    var = best_var
                assignment[var] = not assignment[var]
        return None

    # DPLL solver for completeness
    def dpll(clauses, assignment):
        # Unit propagation
        while True:
            unit_clauses = [c for c in clauses if len(c) == 1]
            if not unit_clauses:
                break
            unit = unit_clauses[0][0]
            val = unit > 0
            var = abs(unit)
            assignment[var] = val
            new_clauses = []
            for c in clauses:
                if unit in c:
                    continue
                if -unit in c:
                    new_c = [l for l in c if l != -unit]
                    if not new_c:
                        return False, None
                    new_clauses.append(new_c)
                else:
                    new_clauses.append(c)
            clauses = new_clauses

        if not clauses:
            return True, assignment

        # Choose variable
        counts = {}
        for c in clauses:
            weight = 1.0 / (2 ** len(c))
            for lit in c:
                counts[abs(lit)] = counts.get(abs(lit), 0) + weight
        var = max(counts, key=counts.get)

        for val in [True, False]:
            lit = var if val else -var
            new_clauses = []
            conflict = False
            for c in clauses:
                if lit in c:
                    continue
                if -lit in c:
                    new_c = [l for l in c if l != -lit]
                    if not new_c:
                        conflict = True
                        break
                    new_clauses.append(new_c)
                else:
                    new_clauses.append(c)
            if not conflict:
                new_assignment = assignment.copy()
                new_assignment[var] = val
                sat, res = dpll(new_clauses, new_assignment)
                if sat:
                    return True, res
        return False, None

    # Check trivial contradiction (empty clause)
    for c in clauses:
        if len(c) == 0:
            print("NO")
            return

    # Try WalkSAT first
    sol = walksat(clauses, n)
    if sol is not None:
        print("YES")
        print(*(sol))
        return

    # Fallback to DPLL
    sat, assignment = dpll(clauses, {})
    if sat:
        print("YES")
        res = [1 if assignment.get(i, False) else 0 for i in range(1, n + 1)]
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
