import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    idx = 2
    clauses = []
    for _ in range(m):
        a = int(data[idx]); b = int(data[idx+1]); c = int(data[idx+2])
        idx += 3
        clauses.append((a, b, c))
    
    # 2-SAT: variable i (1-indexed) engaged=True, disengaged=False
    # literal x>0 means variable x is True; x<0 means variable -x is False
    # clause (a or b or c) -> 3 implications:
    #   not a -> b or c  => not a -> b, not a -> c
    #   not b -> a or c  => not b -> a, not b -> c
    #   not c -> a or b  => not c -> a, not b -> b
    
    size = 2 * (n + 1)
    graph = [[] for _ in range(size)]
    
    def lit_node(lit):
        # positive lit -> 2*var, negative lit -> 2*var+1
        if lit > 0:
            return 2 * lit
        else:
            return 2 * (-lit) + 1
    
    def neg_node(node):
        return node ^ 1
    
    for a, b, c in clauses:
        na = lit_node(a)
        nb = lit_node(b)
        nc = lit_node(c)
        # not a -> b, not a -> c
        graph[neg_node(na)].append(nb)
        graph[neg_node(na)].append(nc)
        # not b -> a, not b -> c
        graph[neg_node(nb)].append(na)
        graph[neg_node(nb)].append(nc)
        # not c -> a, not c -> b
        graph[neg_node(nc)].append(na)
        graph[neg_node(nc)].append(nb)
    
    # Tarjan's SCC
    index_counter = [0]
    stack = []
    on_stack = [False] * size
    indices = [-1] * size
    lowlink = [0] * size
    scc_id = [-1] * size
    scc_count = [0]
    
    import sys
    sys.setrecursionlimit(10000)
    
    # Iterative Tarjan to avoid recursion limit issues
    for start in range(2, 2*(n+1)):
        if indices[start] != -1:
            continue
        call_stack = [(start, 0)]
        while call_stack:
            v, pi = call_stack[-1]
            if pi == 0:
                indices[v] = index_counter[0]
                lowlink[v] = index_counter[0]
                index_counter[0] += 1
                stack.append(v)
                on_stack[v] = True
            recursed = False
            for i in range(pi, len(graph[v])):
                w = graph[v][i]
                if indices[w] == -1:
                    call_stack[-1] = (v, i+1)
                    call_stack.append((w, 0))
                    recursed = True
                    break
                elif on_stack[w]:
                    lowlink[v] = min(lowlink[v], indices[w])
            if recursed:
                continue
            # done with v
            if lowlink[v] == indices[v]:
                while True:
                    w = stack.pop()
                    on_stack[w] = False
                    scc_id[w] = scc_count[0]
                    if w == v:
                        break
                scc_count[0] += 1
            call_stack.pop()
            if call_stack:
                u = call_stack[-1][0]
                lowlink[u] = min(lowlink[u], lowlink[v])
    
    # Check satisfiability
    assignment = [False] * (n + 1)
    for i in range(1, n + 1):
        if scc_id[2*i] == scc_id[2*i+1]:
            print("NO")
            return
        assignment[i] = scc_id[2*i] < scc_id[2*i+1]
    
    print("YES")
    print(' '.join('1' if assignment[i] else '0' for i in range(1, n+1)))

solve()
