```python
import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c1 = int(input_data[idx])
        c2 = int(input_data[idx+1])
        c3 = int(input_data[idx+2])
        idx += 3
        # Simplify clause (remove duplicates)
        clause = list(set([c1, c2, c3]))
        # If clause contains both x and -x, it's trivially satisfied
        has_tautology = False
        seen = set()
        for lit in clause:
            if -lit in seen:
                has_tautology = True
                break
            seen.add(lit)
        if not has_tautology:
            clauses.append(clause)

    # WalkSAT
    def walksat(max_restarts=150, max_flips=1500, p=0.4):
        if not clauses:
            return [0] * (n + 1)
        
        # Precompute clause occurrences
        var_clauses = [[] for _ in range(n + 1)]
        for ci, cl in enumerate(clauses):
            for lit in cl:
                var_clauses[abs(lit)].append(ci)
                
        num_clauses = len(clauses)
        
        for _ in range(max_restarts):
            # Random initial assignment: True (1) or False (0)
            assign = [random.choice([0, 1]) for _ in range(n + 1)]
            
            # Count true literals per clause
            sat_count = [0] * num_clauses
            unsat_clauses = []
            unsat_idx_in_list = [-1] * num_clauses
            
            for ci, cl in enumerate(clauses):
                cnt = 0
                for lit in cl:
                    v = abs(lit)
                    val = assign[v]
                    if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                        cnt += 1
                sat_count[ci] = cnt
                if cnt == 0:
                    unsat_idx_in_list[ci] = len(unsat_clauses)
                    unsat_clauses.append(ci)
            
            if not unsat_clauses:
                return assign

            for _ in range(max_flips):
                if not unsat_clauses:
                    return assign
                
                # Pick random unsatisfied clause
                ci = random.choice(unsat_clauses)
                cl = clauses[ci]
                
                # Find best flip
                best_var = None
                if random.random() < p:
                    best_var = abs(random.choice(cl))
                else:
                    best_break = float('inf')
                    candidates = []
                    for lit in cl:
                        v = abs(lit)
                        # Compute break-count: number of clauses that become unsatisfied if v is flipped
                        # Flipping v:
                        # For clauses where lit matches current assign[v], sat_count decreases by 1.
                        # If it was 1, it becomes 0 (break count increases).
                        break_cnt = 0
                        for c_idx in var_clauses[v]:
                            c_lits = clauses[c_idx]
                            # Check if current assignment satisfies c_idx via v
                            # If so, sat_count[c_idx] would decrease
                            for l in c_lits:
                                if abs(l) == v:
                                    if (l > 0 and assign[v] == 1) or (l < 0 and assign[v] == 0):
                                        if sat_count[c_idx] == 1:
                                            break_cnt += 1
                                    break
                        if break_cnt < best_break:
                            best_break = break_cnt
                            candidates = [v]
                        elif break_cnt == best_break:
                            candidates.append(v)
                    best_var = random.choice(candidates)
                
                # Flip best_var
                old_val = assign[best_var]
                new_val = 1 - old_val
                assign[best_var] = new_val
                
                for c_idx in var_clauses[best_var]:
                    c_lits = clauses[c_idx]
                    for l in c_lits:
                        if abs(l) == best_var:
                            old_sat = (l > 0 and old_val == 1) or (l < 0 and old_val == 0)
                            new_sat = (l > 0 and new_val == 1) or (l < 0 and new_val == 0)
                            if old_sat and not new_sat:
                                sat_count[c_idx] -= 1
                                if sat_count[c_idx] == 0:
                                    unsat_idx_in_list[c_idx] = len(unsat_clauses)
                                    unsat_clauses.append(c_idx)
                            elif not old_sat and new_sat:
                                if sat_count[c_idx] == 0:
                                    idx_remove = unsat_idx_in_list[c_idx]
                                    last_c = unsat_clauses[-1]
                                    unsat_clauses[idx_remove] = last_c
                                    unsat_idx_in_list[last_c] = idx_remove
                                    unsat_clauses.pop()
                                    unsat_idx_in_list[c_idx] = -1
                                sat_count[c_idx] += 1
                            break
                            
            if not unsat_clauses:
                return assign
        return None

    # DPLL Solver
    def dpll_solve():
        # assignment: list of size n+1, values: None, 0, 1
        assignment = [None] * (n + 1)
        
        def get_clause_status(clause):
            # Returns:
            # 'SAT' if at least one lit is True
            # 'UNSAT' if all lits are False
            # 'UNIT', lit if exactly one unassigned and rest False
            # 'UNDET', unassigned_lits otherwise
            unassigned = []
            for lit in clause:
                v = abs(lit)
                val = assignment[v]
                if val is not None:
                    if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                        return 'SAT', None
                else:
                    unassigned.append(lit)
            if not unassigned:
                return 'UNSAT', None
            if len(unassigned) == 1:
                return 'UNIT', unassigned[0]
            return 'UNDET', unassigned

        def unit_propagate():
            trail = []
            while True:
                unit_found = False
                for clause in clauses:
                    st, lit = get_clause_status(clause)
                    if st == 'UNSAT':
                        return False, trail
                    if st == 'UNIT':
                        v = abs(lit)
                        req_val = 1 if lit > 0 else 0
                        if assignment[v] is None:
                            assignment[v] = req_val
                            trail.append(v)
                            unit_found = True
                        elif assignment[v] != req_val:
                            return False, trail
                if not unit_found:
                    break
            return True, trail

        def backtrack():
            ok, trail = unit_propagate()
            if not ok:
                for v in trail:
                    assignment[v] = None
                return False
            
            # Check if all clauses are satisfied
            all_sat = True
            for clause in clauses:
                st, _ = get_clause_status(clause)
                if st == 'UNSAT':
                    for v in trail:
                        assignment[v] = None
                    return False
                if st != 'SAT':
                    all_sat = False

            if all_sat:
                return True

            # Jeroslow-Wang Heuristic for branching variable
            scores = {}
            for clause in clauses:
                st, unassigned = get_clause_status(clause)
                if st == 'UNDET':
                    weight = 2 ** (-len(unassigned))
                    for lit in unassigned:
                        scores[lit] = scores.get(lit, 0) + weight
                elif st == 'UNIT':
                    weight = 0.5
                    scores[unassigned] = scores.get(unassigned, 0) + weight

            if not scores:
                for i in range(1, n + 1):
                    if assignment[i] is None:
                        scores[i] = 1
                        break
            
            if not scores:
                return True

            best_lit = max(scores, key=scores.get)
            best_var = abs(best_lit)
            first_val = 1 if best_lit > 0 else 0

            # Try first_val
            assignment[best_var] = first_val
            if backtrack():
                return True
            assignment[best_var] = None

            # Try 1 - first_val
            assignment[best_var] = 1 - first_val
            if backtrack():
                return True
            assignment[best_var] = None

            for v in trail:
                assignment[v] = None
            return False

        if backtrack():
            for i in range(1, n + 1):
                if assignment[i] is None:
                    assignment[i] = 0
            return assignment
        return None

    # First try WalkSAT
    model = walksat()
    if model is None:
        model = dpll_solve()

    if model is not None:
        print("YES")
        print(*(model[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
