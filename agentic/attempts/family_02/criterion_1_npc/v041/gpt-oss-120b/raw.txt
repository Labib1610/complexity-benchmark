```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        lits = []
        for _ in range(3):
            x = int(next(it))
            var = abs(x) - 1
            sign = x > 0          # True means variable should be 1
            lits.append((var, sign))
        clauses.append(lits)

    # assignment: -1 unassigned, 0 or 1 assigned
    assign = [-1] * n

    # For each variable, list of clause indices where it appears
    var_clauses = [[] for _ in range(n)]
    for idx, cl in enumerate(clauses):
        for v, _ in cl:
            var_clauses[v].append(idx)

    # Helper to evaluate literal under current assignment
    def lit_val(lit):
        v, sign = lit
        a = assign[v]
        if a == -1:
            return None
        return a == 1 if sign else a == 0

    # Unit propagation and pure literal elimination
    def propagate(stack):
        while True:
            changed = False
            # unit clauses
            for idx, cl in enumerate(clauses):
                cnt_un = 0
                last_un = None
                satisfied = False
                for lit in cl:
                    val = lit_val(lit)
                    if val is True:
                        satisfied = True
                        break
                    if val is None:
                        cnt_un += 1
                        last_un = lit
                if satisfied:
                    continue
                if cnt_un == 0:          # all false -> conflict
                    return False
                if cnt_un == 1:          # unit clause
                    v, sign = last_un
                    needed = 1 if sign else 0
                    if assign[v] == -1:
                        assign[v] = needed
                        stack.append(v)
                        changed = True
                    elif assign[v] != needed:
                        return False
            if not changed:
                break
        return True

    # DPLL recursive search
    def dfs():
        # first propagate
        trail = []
        if not propagate(trail):
            for v in trail:
                assign[v] = -1
            return False

        # check if all variables assigned
        if all(a != -1 for a in assign):
            return True

        # choose variable (most occurring in unsatisfied clauses)
        # simple heuristic: first unassigned
        for v in range(n):
            if assign[v] == -1:
                var = v
                break

        for val in (0, 1):
            assign[var] = val
            if dfs():
                return True
            assign[var] = -1

        # backtrack assignments made during propagation
        for v in trail:
            assign[v] = -1
        return False

    if dfs():
        print("YES")
        print(' '.join(str(assign[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
