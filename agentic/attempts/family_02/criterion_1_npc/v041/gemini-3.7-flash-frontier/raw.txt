import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    clauses = []
    idx = 2
    for _ in range(m):
        c = [
            int(input_data[idx]),
            int(input_data[idx + 1]),
            int(input_data[idx + 2]),
        ]
        idx += 3

        s = set(c)
        tautology = False
        for lit in s:
            if -lit in s:
                tautology = True
                break
        if not tautology:
            clauses.append(list(s))

    def unit_propagate(clauses, assignment):
        assignment = dict(assignment)
        while True:
            unit_clauses = [c for c in clauses if len(c) == 1]
            if not unit_clauses:
                break
            unit = unit_clauses[0][0]
            assignment[abs(unit)] = 1 if unit > 0 else 0

            new_clauses = []
            for c in clauses:
                if unit in c:
                    continue
                if -unit in c:
                    new_c = [x for x in c if x != -unit]
                    if not new_c:
                        return None, None
                    new_clauses.append(new_c)
                else:
                    new_clauses.append(c)
            clauses = new_clauses
        return clauses, assignment

    def dpll(clauses, assignment):
        clauses, assignment = unit_propagate(clauses, assignment)
        if clauses is None:
            return None
        if not clauses:
            return assignment

        min_len = min(len(c) for c in clauses)
        counts = {}
        for c in clauses:
            if len(c) == min_len:
                for lit in c:
                    counts[lit] = counts.get(lit, 0) + 1

        best_lit = max(counts, key=counts.get)

        res = dpll(clauses + [[best_lit]], assignment)
        if res is not None:
            return res
        return dpll(clauses + [[-best_lit]], assignment)

    ans = dpll(clauses, {})
    if ans is None:
        print("NO")
    else:
        print("YES")
        res = [str(ans.get(i, 0)) for i in range(1, n + 1)]
        print(" ".join(res))


if __name__ == "__main__":
    solve()
