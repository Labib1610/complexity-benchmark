import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    clauses = []
    for _ in range(m):
        f1 = int(next(it))
        f2 = int(next(it))
        f3 = int(next(it))
        clauses.append((f1, f2, f3))
    
    # State: 0 = unassigned, 1 = True, -1 = False
    state = [0] * (n + 1)
    
    def get_feed_active(feed, current_state):
        sensor = abs(feed)
        if current_state[sensor] == 0:
            return None
        is_one = (current_state[sensor] == 1)
        if feed > 0:
            return is_one
        else:
            return not is_one
    
    def is_clause_satisfied(clause, current_state):
        for feed in clause:
            val = get_feed_active(feed, current_state)
            if val is True:
                return True
        return False
    
    def is_clause_falsified(clause, current_state):
        for feed in clause:
            val = get_feed_active(feed, current_state)
            if val is not False:
                return False
        return True
    
    def get_unit_literal(clause, current_state):
        unassigned = []
        for feed in clause:
            sensor = abs(feed)
            if current_state[sensor] == 0:
                unassigned.append(feed)
            elif get_feed_active(feed, current_state) is True:
                return None  # Already satisfied
        if len(unassigned) == 1:
            return unassigned[0]
        return None
    
    def solve_recursive(idx):
        # Unit propagation
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                if is_clause_satisfied(clause, state):
                    continue
                if is_clause_falsified(clause, state):
                    return False
                unit = get_unit_literal(clause, state)
                if unit is not None:
                    sensor = abs(unit)
                    if unit > 0:
                        state[sensor] = 1
                    else:
                        state[sensor] = -1
                    changed = True
        
        # Find next unassigned variable
        while idx <= n and state[idx] != 0:
            idx += 1
        
        if idx > n:
            return True
        
        # Try both values
        for val in [1, -1]:
            state[idx] = val
            if solve_recursive(idx + 1):
                return True
            state[idx] = 0
        
        return False
    
    if solve_recursive(1):
        print("YES")
        result = []
        for i in range(1, n + 1):
            if state[i] == 1:
                result.append('1')
            else:
                result.append('0')
        print(' '.join(result))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
