We can model each interlock as a clause with three literals, where each literal is either a variable or its negation.  
This is exactly the **3-SAT** problem. Since \( n \le 90 \) and \( m \le 400 \), we can solve it using a backtracking algorithm with unit propagation and pure literal elimination, which is efficient enough for this size.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    clauses = []
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx]); l2 = int(input_data[idx+1]); l3 = int(input_data[idx+2])
        idx += 3
        clauses.append((l1, l2, l3))
    
    assignment = [None] * (n + 1)  # None = unassigned, True/False = value
    
    def propagate(clauses, assignment):
        changed = True
        while changed:
            changed = False
            new_clauses = []
            for clause in clauses:
                # Check if clause is satisfied
                satisfied = False
                unassigned = []
                for lit in clause:
                    var = abs(lit)
                    val = assignment[var]
                    if val is not None:
                        if (lit > 0 and val) or (lit < 0 and not val):
                            satisfied = True
                            break
                    else:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if not unassigned:
                    return None  # conflict
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    val = lit > 0
                    if assignment[var] is not None and assignment[var] != val:
                        return None
                    assignment[var] = val
                    changed = True
                else:
                    new_clauses.append(clause)
            clauses = new_clauses
        return clauses
    
    def pure_literal_elimination(clauses, assignment):
        # Count occurrences
        pos_count = {}
        neg_count = {}
        for clause in clauses:
            for lit in clause:
                var = abs(lit)
                if lit > 0:
                    pos_count[var] = pos_count.get(var, 0) + 1
                else:
                    neg_count[var] = neg_count.get(var, 0) + 1
        changed = False
        new_clauses = []
        for clause in clauses:
            keep = False
            for lit in clause:
                var = abs(lit)
                if assignment[var] is not None:
                    continue
                if lit > 0 and neg_count.get(var, 0) == 0:
                    assignment[var] = True
                    changed = True
                    keep = True
                    break
                elif lit < 0 and pos_count.get(var, 0) == 0:
                    assignment[var] = False
                    changed = True
                    keep = True
                    break
            if not keep:
                new_clauses.append(clause)
        return new_clauses, changed
    
    def backtrack(clauses):
        clauses = propagate(clauses, assignment)
        if clauses is None:
            return False
        clauses, changed = pure_literal_elimination(clauses, assignment)
        if changed:
            clauses = propagate(clauses, assignment)
            if clauses is None:
                return False
        # Check if all satisfied
        if not clauses:
            return True
        # Choose an unassigned variable
        var = None
        for clause in clauses:
            for lit in clause:
                v = abs(lit)
                if assignment[v] is None:
                    var = v
                    break
            if var is not None:
                break
        if var is None:
            return True
        
        # Try True
        assignment[var] = True
        saved = [a for a in assignment]
        if backtrack([c for c in clauses]):
            return True
        # Restore
        for i in range(1, n+1):
            assignment[i] = saved[i]
        # Try False
        assignment[var] = False
        saved = [a for a in assignment]
        if backtrack([c for c in clauses]):
            return True
        for i in range(1, n+1):
            assignment[i] = saved[i]
        return False
    
    if backtrack(clauses):
        print("YES")
        print(" ".join("1" if assignment[i] else "0" for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
