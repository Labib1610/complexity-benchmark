import sys

sys.setrecursionlimit(10000)

def read_input():
    data = sys.stdin.read().strip().split()
    if not data:
        return 0, 0, []
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        clause = [int(next(it)), int(next(it)), int(next(it))]
        clauses.append(clause)
    return n, m, clauses

def simplify(clauses, assign):
    """apply current assignment, return new clause list.
       If a clause becomes empty -> conflict (return None)."""
    new_clauses = []
    for cl in clauses:
        satisfied = False
        new_clause = []
        for lit in cl:
            var = abs(lit)
            if var in assign:
                val = assign[var]
                if (lit > 0 and val) or (lit < 0 and not val):
                    satisfied = True
                    break
                else:
                    continue
            else:
                new_clause.append(lit)
        if satisfied:
            continue
        if not new_clause:   # empty clause -> conflict
            return None
        new_clauses.append(new_clause)
    return new_clauses

def unit_propagate(clauses, assign):
    while True:
        unit = None
        for cl in clauses:
            if len(cl) == 1:
                unit = cl[0]
                break
        if unit is None:
            break
        var = abs(unit)
        val = unit > 0
        if var in assign:
            if assign[var] != val:
                return None, None
        else:
            assign[var] = val
        clauses = simplify(clauses, assign)
        if clauses is None:
            return None, None
    return clauses, assign

def pure_literal_assign(clauses, assign):
    counts = {}
    for cl in clauses:
        for lit in cl:
            counts[lit] = counts.get(lit, 0) + 1
    changed = False
    for var in range(1, max_var+1):
        if var in assign:
            continue
        pos = counts.get(var, 0)
        neg = counts.get(-var, 0)
        if pos and not neg:
            assign[var] = True
            changed = True
        elif neg and not pos:
            assign[var] = False
            changed = True
    if changed:
        clauses = simplify(clauses, assign)
    return clauses, assign, changed

def dpll(clauses, assign):
    clauses, assign = unit_propagate(clauses, assign)
    if clauses is None:
        return None
    if not clauses:
        return assign
    # pure literal elimination
    clauses, assign, changed = pure_literal_assign(clauses, assign)
    if clauses is None:
        return None
    if not clauses:
        return assign
    # choose variable
    for cl in clauses:
        for lit in cl:
            var = abs(lit)
            if var not in assign:
                break
        else:
            continue
        break
    # try true then false
    for val in (True, False):
        new_assign = assign.copy()
        new_assign[var] = val
        new_clauses = simplify(clauses, {var: val})
        if new_clauses is None:
            continue
        res = dpll(new_clauses, new_assign)
        if res is not None:
            return res
    return None

if __name__ == "__main__":
    n, m, clauses = read_input()
    max_var = n
    result = dpll(clauses, {})
    if result is None:
        print("NO")
    else:
        print("YES")
        out = []
        for i in range(1, n+1):
            out.append('1' if result.get(i, False) else '0')
        print(' '.join(out))
