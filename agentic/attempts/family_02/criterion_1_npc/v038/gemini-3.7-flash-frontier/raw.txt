import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    raw_clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx + 1]), int(input_data[idx + 2])]
        idx += 3
        # Simplify clause: remove duplicates, check if tautology
        c_set = set(c)
        tautology = False
        for lit in c_set:
            if -lit in c_set:
                tautology = True
                break
        if not tautology:
            raw_clauses.append(list(c_set))

    # CDCL Solver
    # Literals: 1..n and -1..-n
    # Internal representation:
    # Var: 1..n
    # Lit: lit > 0 -> 2*lit, lit < 0 -> 2*(-lit) + 1

    def to_lit(l):
        return 2 * l if l > 0 else 2 * (-l) + 1

    def from_lit(l):
        return (l // 2) if (l % 2 == 0) else -(l // 2)

    def var_of(l):
        return l // 2

    def neg_lit(l):
        return l ^ 1

    clauses = []
    for c in raw_clauses:
        if len(c) == 0:
            print("NO")
            return
        clauses.append([to_lit(x) for x in c])

    # Data structures
    # assignment: -1 unassigned, 0 false, 1 true (for literals: value[l])
    # To make it simple: assigns[v] in { -1, 0, 1 } where 1 means var v is True (lit 2*v is True).
    # val_lit(l): if l is 2*v, return assigns[v]; if l is 2*v+1, return 1 - assigns[v] if assigns[v] != -1 else -1

    assigns = [-1] * (n + 1)
    trail = []
    trail_lim = []
    reason = [None] * (n + 1)
    level = [-1] * (n + 1)
    activity = [0.0] * (n + 1)
    var_inc = 1.0
    var_decay = 0.95

    # Count initial occurrences for initial activity
    for c in clauses:
        for l in c:
            activity[var_of(l)] += 1.0

    # Watch lists: watches[lit] contains list of clause indices watching lit
    watches = [[] for _ in range(2 * n + 2)]

    for c_idx, c in enumerate(clauses):
        if len(c) == 1:
            watches[c[0]].append(c_idx)
        else:
            watches[c[0]].append(c_idx)
            watches[c[1]].append(c_idx)

    def val_lit(l):
        v = var_of(l)
        val = assigns[v]
        if val == -1:
            return -1
        return val if (l % 2 == 0) else (1 - val)

    def assign_lit(l, r=None):
        v = var_of(l)
        val = 1 if (l % 2 == 0) else 0
        assigns[v] = val
        reason[v] = r
        level[v] = len(trail_lim)
        trail.append(v)

    def unassign(v):
        assigns[v] = -1
        reason[v] = None
        level[v] = -1

    def bcp():
        head = 0
        while head < len(trail):
            v = trail[head]
            head += 1
            # Variable v was assigned. The literal that became FALSE is:
            false_lit = 2 * v + 1 if assigns[v] == 1 else 2 * v
            # Iterate through clauses watching false_lit
            ws = watches[false_lit]
            watches[false_lit] = []
            i = 0
            while i < len(ws):
                c_idx = ws[i]
                c = clauses[c_idx]

                if len(c) == 1:
                    watches[false_lit].append(c_idx)
                    # unit clause became false -> conflict
                    # restore remaining watches
                    for j in range(i + 1, len(ws)):
                        watches[false_lit].append(ws[j])
                    return c_idx

                # Ensure false_lit is at c[1]
                if c[0] == false_lit:
                    c[0], c[1] = c[1], c[0]

                # If the other watched literal c[0] is TRUE, clause is satisfied
                if val_lit(c[0]) == 1:
                    watches[false_lit].append(c_idx)
                    i += 1
                    continue

                # Look for a new literal to watch
                found = False
                for k in range(2, len(c)):
                    if val_lit(c[k]) != 0:
                        c[1], c[k] = c[k], c[1]
                        watches[c[1]].append(c_idx)
                        found = True
                        break

                if not found:
                    watches[false_lit].append(c_idx)
                    # Cannot find another watched literal
                    if val_lit(c[0]) == 0:
                        # Conflict!
                        for j in range(i + 1, len(ws)):
                            watches[false_lit].append(ws[j])
                        return c_idx
                    else:
                        # Unit propagation on c[0]
                        assign_lit(c[0], c_idx)
                i += 1

        return -1

    def analyze(conflict_c_idx):
        learnt = []
        c = clauses[conflict_c_idx]
        cur_level = len(trail_lim)
        path_cnt = 0
        seen = [False] * (n + 1)
        p = None

        for l in c:
            v = var_of(l)
            activity[v] += var_inc
            if level[v] == cur_level:
                seen[v] = True
                path_cnt += 1
            elif level[v] > 0:
                learnt.append(l)
                seen[v] = True

        trail_idx = len(trail) - 1
        while path_cnt > 0:
            while not seen[trail[trail_idx]]:
                trail_idx -= 1
            p = trail[trail_idx]
            trail_idx -= 1
            seen[p] = False
            path_cnt -= 1

            if path_cnt > 0:
                r = reason[p]
                for l in clauses[r]:
                    v = var_of(l)
                    if v != p and not seen[v]:
                        activity[v] += var_inc
                        seen[v] = True
                        if level[v] == cur_level:
                            path_cnt += 1
                        elif level[v] > 0:
                            learnt.append(l)

        # p is the 1-UIP
        # The literal to assert is the negation of p's assignment
        uip_lit = 2 * p + 1 if assigns[p] == 1 else 2 * p
        learnt.insert(0, uip_lit)

        # Backjump level is max level of other literals in learnt
        backtrack_level = 0
        if len(learnt) > 1:
            max_i = 1
            for i in range(2, len(learnt)):
                if level[var_of(learnt[i])] > level[var_of(learnt[max_i])]:
                    max_i = i
            learnt[1], learnt[max_i] = learnt[max_i], learnt[1]
            backtrack_level = level[var_of(learnt[1])]

        return learnt, backtrack_level

    def cancel_until(lvl):
        while len(trail_lim) > lvl:
            t_border = trail_lim.pop()
            while len(trail) > t_border:
                unassign(trail.pop())

    # Pre-propagation for level 0
    # Add initial unit clauses
    for c_idx, c in enumerate(clauses):
        if len(c) == 1:
            if val_lit(c[0]) == -1:
                assign_lit(c[0], c_idx)
            elif val_lit(c[0]) == 0:
                print("NO")
                return

    if bcp() != -1:
        print("NO")
        return

    # Solve loop
    restart_limit = 100
    restart_inc = 1.5
    conflict_count = 0

    while True:
        # Check if all variables assigned
        v_pick = -1
        max_act = -1
        for v in range(1, n + 1):
            if assigns[v] == -1:
                if activity[v] > max_act:
                    max_act = activity[v]
                    v_pick = v

        if v_pick == -1:
            # Satisfiable!
            print("YES")
            res = [str(max(0, assigns[i])) for i in range(1, n + 1)]
            print(*(res))
            return

        # Decision
        trail_lim.append(len(trail))
        assign_lit(2 * v_pick)  # pick True

        while True:
            conflict = bcp()
            if conflict != -1:
                conflict_count += 1
                if len(trail_lim) == 0:
                    print("NO")
                    return

                learnt, backtrack_level = analyze(conflict)
                cancel_until(backtrack_level)

                # Add learnt clause
                c_idx = len(clauses)
                clauses.append(learnt)

                if len(learnt) == 1:
                    watches[learnt[0]].append(c_idx)
                    assign_lit(learnt[0], None)
                else:
                    watches[learnt[0]].append(c_idx)
                    watches[learnt[1]].append(c_idx)
                    assign_lit(learnt[0], c_idx)

                nonlocal_var_inc = var_inc * (1.0 / var_decay)
                var_inc = nonlocal_var_inc

                if conflict_count >= restart_limit:
                    restart_limit = int(restart_limit * restart_inc)
                    cancel_until(0)
                    break
            else:
                break


solve()
