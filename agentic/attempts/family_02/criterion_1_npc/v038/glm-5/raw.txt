                if score > best_score:
                    best_score = score
                    best_toggle = toggle
        
        if best_toggle == -1:
            return True
        
        for val in [0, 1]:
            changes = assign(best_toggle, val)
            if backtrack():
                return True
            undo(best_toggle, changes)
        
        for t, ch in reversed(propagated):
            undo(t, ch)
        return False
    
    if backtrack():
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(1, n+1)))
    else:
        print("NO")
