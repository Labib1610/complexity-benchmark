We can model each check as a clause in a 3-SAT problem, where each literal is a variable or its negation.  
Since \( n \leq 90 \), we can use a simple backtracking or DPLL-style solver, but we must be careful with time.

A straightforward approach:  
- Represent each variable as an integer index (1-based).  
- Each clause has three literals.  
- Use recursive search with unit propagation and pure literal elimination to speed up.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # Variable assignment: 0 = unassigned, 1 = True, -1 = False
    assignment = [0] * (n + 1)
    
    # Quick check: if m == 0, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # DPLL with unit propagation
    def dpll(clauses, assignment):
        # Unit propagation
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    var = abs(lit)
                    val = assignment[var]
                    if val == 0:
                        unassigned.append(lit)
                    elif (lit > 0 and val == 1) or (lit < 0 and val == -1):
                        satisfied = True
                        break
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return None  # conflict
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else -1
                    if assignment[var] != 0 and assignment[var] != val:
                        return None
                    assignment[var] = val
                    changed = True
        
        # Check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            sat = False
            for lit in clause:
                var = abs(lit)
                val = assignment[var]
                if (lit > 0 and val == 1) or (lit < 0 and val == -1):
                    sat = True
                    break
            if not sat:
                all_sat = False
                break
        if all_sat:
            return assignment
        
        # Choose unassigned variable
        for var in range(1, n + 1):
            if assignment[var] == 0:
                # Try True
                assignment[var] = 1
                res = dpll(clauses, assignment[:])
                if res is not None:
                    return res
                # Try False
                assignment[var] = -1
                res = dpll(clauses, assignment[:])
                if res is not None:
                    return res
                assignment[var] = 0
                return None
        return None
    
    result = dpll(clauses, assignment)
    if result is None:
        print("NO")
    else:
        print("YES")
        print(" ".join("1" if result[i] == 1 else "0" for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
