import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # simplify clause: remove duplicates, check if tautology
        c_set = set(c)
        tautology = False
        for lit in c:
            if -lit in c_set:
                tautology = True
                break
        if not tautology:
            clauses.append(list(c_set))
            
    # Check for empty clauses
    for c in clauses:
        if len(c) == 0:
            print("NO")
            return

    # Solver implementation: CDCL
    # Literals: +v -> 2*v, -v -> 2*v+1
    # Variable 1..n
    def lit2idx(l):
        if l > 0:
            return 2 * l
        else:
            return 2 * (-l) + 1

    def idx2lit(idx):
        v = idx // 2
        return v if (idx % 2 == 0) else -v

    def neg(l_idx):
        return l_idx ^ 1

    # Initialize data structures
    all_clauses = []
    watches = [[] for _ in range(2 * n + 2)]
    
    for c in clauses:
        c_lits = [lit2idx(l) for l in c]
        all_clauses.append(c_lits)
    
    # 0 = unassigned, 1 = true, -1 = false
    # values indexed by lit_idx: value of lit_idx
    assignment = [0] * (2 * n + 2)
    decision_level = 0
    trail = []
    trail_lim = []
    reason = [None] * (n + 1)
    level = [0] * (n + 1)
    
    activity = [0.0] * (n + 1)
    decay = 0.95
    act_inc = 1.0

    # Initial frequency-based activity
    for c in all_clauses:
        for l in c:
            activity[l // 2] += 1.0

    # Set up watched literals
    for i, c in enumerate(all_clauses):
        if len(c) == 1:
            watches[c[0]].append(i)
        else:
            watches[c[0]].append(i)
            watches[c[1]].append(i)

    def assign(l_idx, r):
        v = l_idx // 2
        val = 1 if (l_idx % 2 == 0) else -1
        assignment[2 * v] = val
        assignment[2 * v + 1] = -val
        level[v] = decision_level
        reason[v] = r
        trail.append(l_idx)

    # Unit propagation
    def propagate():
        q_head = propagate.q_head
        while q_head < len(trail):
            p = trail[q_head]
            q_head += 1
            false_lit = neg(p)
            
            # We look at clauses watching false_lit
            ws = watches[false_lit]
            i = 0
            while i < len(ws):
                cr = ws[i]
                c = all_clauses[cr]
                
                # Make sure false_lit is at c[1]
                if c[0] == false_lit:
                    c[0], c[1] = c[1], c[0]
                
                # If 0th literal is true, clause is satisfied
                first = c[0]
                if assignment[first] == 1:
                    i += 1
                    continue
                
                # Look for a new watch
                found = False
                for k in range(2, len(c)):
                    if assignment[c[k]] != -1:
                        c[1], c[k] = c[k], c[1]
                        watches[c[1]].append(cr)
                        ws[i] = ws[-1]
                        ws.pop()
                        found = True
                        break
                
                if found:
                    continue
                
                # Clause is unit or conflict
                if assignment[first] == -1:
                    # Conflict!
                    propagate.q_head = q_head
                    return cr
                else:
                    # Unit
                    assign(first, cr)
                    i += 1
        propagate.q_head = q_head
        return None

    propagate.q_head = 0

    # Initial unit propagation
    for i, c in enumerate(all_clauses):
        if len(c) == 1:
            l = c[0]
            if assignment[l] == -1:
                print("NO")
                return
            elif assignment[l] == 0:
                assign(l, None)
                conf = propagate()
                if conf is not None:
                    print("NO")
                    return

    def analyze(conflict_cr):
        nonlocal act_inc
        p = None
        c = 0
        learnt = [0]
        confl = conflict_cr
        idx_t = len(trail) - 1
        
        seen = [False] * (n + 1)
        
        while True:
            c_clause = all_clauses[confl]
            for j in range(0 if p is None else 1, len(c_clause)):
                q = c_clause[j]
                v = q // 2
                if not seen[v] and level[v] > 0:
                    activity[v] += act_inc
                    seen[v] = True
                    if level[v] >= decision_level:
                        c += 1
                    else:
                        learnt.append(q)
            
            while not seen[trail[idx_t] // 2]:
                idx_t -= 1
            
            p = trail[idx_t]
            v = p // 2
            confl = reason[v]
            seen[v] = False
            idx_t -= 1
            c -= 1
            if c <= 0:
                break
                
        learnt[0] = neg(p)
        
        # Find backtrack level
        if len(learnt) == 1:
            bt_level = 0
        else:
            max_i = 1
            for k in range(2, len(learnt)):
                if level[learnt[k] // 2] > level[learnt[max_i] // 2]:
                    max_i = k
            learnt[1], learnt[max_i] = learnt[max_i], learnt[1]
            bt_level = level[learnt[1] // 2]
            
        act_inc /= decay
        return learnt, bt_level

    def backtrack_to(bt_level):
        nonlocal decision_level
        while len(trail) > (trail_lim[bt_level] if bt_level < len(trail_lim) else 0):
            p = trail.pop()
            v = p // 2
            assignment[2 * v] = 0
            assignment[2 * v + 1] = 0
            reason[v] = None
        propagate.q_head = len(trail)
        decision_level = bt_level
        del trail_lim[bt_level:]

    def pick_branch_lit():
        max_act = -1.0
        best_v = -1
        for v in range(1, n + 1):
            if assignment[2 * v] == 0:
                if activity[v] > max_act:
                    max_act = activity[v]
                    best_v = v
        if best_v == -1:
            return None
        return 2 * best_v # default try True

    # Main CDCL loop
    conflict_count = 0
    restart_limit = 100
    
    while True:
        confl = propagate()
        if confl is not None:
            conflict_count += 1
            if decision_level == 0:
                print("NO")
                return
            learnt, bt_level = analyze(confl)
            backtrack_to(bt_level)
            
            new_cr = len(all_clauses)
            all_clauses.append(learnt)
            if len(learnt) == 1:
                watches[learnt[0]].append(new_cr)
                assign(learnt[0], None)
            else:
                watches[learnt[0]].append(new_cr)
                watches[learnt[1]].append(new_cr)
                assign(learnt[0], new_cr)
                
            if conflict_count >= restart_limit:
                # restart
                backtrack_to(0)
                restart_limit = int(restart_limit * 1.5)
        else:
            next_lit = pick_branch_lit()
            if next_lit is None:
                # All assigned!
                print("YES")
                res = []
                for v in range(1, n + 1):
                    res.append(1 if assignment[2 * v] == 1 else 0)
                print(*(res))
                return
            
            decision_level += 1
            trail_lim.append(len(trail))
            assign(next_lit, None)

if __name__ == '__main__':
    solve()
