```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # 2-SAT setup
    # Variable i (1-based): 2*i-2 for False (Southern), 2*i-1 for True (Northern)
    N = 2 * n
    adj = [[] for _ in range(N)]
    radj = [[] for _ in range(N)]
    
    def get_node(p):
        if p > 0:
            return 2 * (p - 1) + 1
        else:
            return 2 * (-p - 1)
    
    idx = 2
    for _ in range(m):
        p1 = int(input_data[idx])
        p2 = int(input_data[idx+1])
        p3 = int(input_data[idx+2])
        idx += 3
        
        u1 = get_node(p1)
        u2 = get_node(p2)
        u3 = get_node(p3)
        
        # u1 OR u2
        adj[u1 ^ 1].append(u2)
        adj[u2 ^ 1].append(u1)
        # u1 OR u3
        adj[u1 ^ 1].append(u3)
        adj[u3 ^ 1].append(u1)
        # u2 OR u3
        adj[u2 ^ 1].append(u3)
        adj[u3 ^ 1].append(u2)
        
    for u in range(N):
        for v in adj[u]:
            radj[v].append(u)
            
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, i = stack[-1]
            if i == 0:
                visited[node] = True
            if i < len(adj[node]):
                stack[-1] = (node, i + 1)
                v = adj[node][i]
                if not visited[v]:
                    stack.append((v, 0))
            else:
                stack.pop()
                order.append(node)
                
    for i in range(N):
        if not visited[i]:
            dfs1(i)
            
    comp_id = [-1] * N
    c = 0
    
    for u in reversed(order):
        if comp_id[u] == -1:
            stack = [u]
            comp_id[u] = c
            while stack:
                node = stack.pop()
                for v in radj[node]:
                    if comp_id[v] == -1:
                        comp_id[v] = c
                        stack.append(v)
            c += 1
            
    assignment = [0] * n
    for i in range(n):
        true_node = 2 * i + 1
        false_node = 2 * i
        if comp_id[true_node] == comp_id[false_node]:
            print("NO")
            return
        if comp_id[true_node] > comp_id[false_node]:
            assignment[i] = 1
        else:
            assignment[i] = 0
            
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == '__main__':
    solve()
```
