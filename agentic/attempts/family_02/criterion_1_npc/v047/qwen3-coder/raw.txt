def solve():
    import sys
    from itertools import product
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    vulnerabilities = []
    idx = 2
    for _ in range(m):
        v1 = int(data[idx])
        v2 = int(data[idx+1])
        v3 = int(data[idx+2])
        vulnerabilities.append((v1, v2, v3))
        idx += 3
    
    # For each vulnerability, we need at least one of the three conditions to be true
    # This is a 3-SAT-like problem. We can try all 2^n possibilities since n <= 90,
    # but actually, we can do better by considering the constraints.
    
    # The key insight is that for each vulnerability, we have a clause with 3 literals.
    # We can use the fact that for any vulnerability, the difference between max and min
    # valve indices is < 90. Since n <= 90, this means all referenced valves are in a
    # window of size < 90.
    
    # However, given the small constraints on n (<= 90) and the structure, let's 
    # consider that we can try all possible assignments.
    # But even 2^90 is too large. Let's think differently.
    
    # Let's build a graph where each vulnerability is a constraint that at least one
    # of its three literals must be true. This is exactly 3-SAT.
    
    # However, given that this is a competition problem, there might be a polynomial
    # solution or we can exploit the structure.
    
    # Let's re-examine the problem:
    # - We have m vulnerabilities, each with 3 literals (valve settings)
    # - For each vulnerability, at least one literal must be satisfied
    # - The sum of all valve indices across vulnerabilities is <= 108000
    # - For each vulnerability, max_index - min_index < 90
    
    # Since n <= 90 and for each vulnerability the span is < 90, we can consider
    # that the variables involved are localized.
    
    # A simpler approach: since n <= 90, we can try all 2^n assignments.
    # But 2^90 is still too big. However, many variables might not appear.
    # Let's find which variables actually appear.
    
    appeared = set()
    for v1, v2, v3 in vulnerabilities:
        appeared.add(abs(v1))
        appeared.add(abs(v2))
        appeared.add(abs(v3))
    
    variables = sorted(list(appeared))
    var_to_idx = {v: i for i, v in enumerate(variables)}
    num_vars = len(variables)
    
    if num_vars == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # Now we can try all 2^num_vars assignments
    for assignment in product([0, 1], repeat=num_vars):
        # Check if this assignment satisfies all vulnerabilities
        valid = True
        for v1, v2, v3 in vulnerabilities:
            val1 = assignment[var_to_idx[abs(v1)]]
            val2 = assignment[var_to_idx[abs(v2)]]
            val3 = assignment[var_to_idx[abs(v3)]]
            
            cond1 = (val1 == 1) if v1 > 0 else (val1 == 0)
            cond2 = (val2 == 1) if v2 > 0 else (val2 == 0)
            cond3 = (val3 == 1) if v3 > 0 else (val3 == 0)
            
            if not (cond1 or cond2 or cond3):
                valid = False
                break
        
        if valid:
            # Construct full assignment
            result = [0] * (n + 1)  # 1-indexed
            for i, var in enumerate(variables):
                result[var] = assignment[i]
            
            # For variables not appearing in any vulnerability, set to 0
            print("YES")
            print(" ".join(map(str, result[1:])))
            return
    
    print("NO")

solve()
