import sys

def solve():
    sys.setrecursionlimit(2000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Remove duplicates within the clause
        c_set = list(set(c))
        # If clause contains both x and -x, it's trivially satisfied, ignore it
        has_both = False
        s = set(c_set)
        for lit in c_set:
            if -lit in s:
                has_both = True
                break
        if not has_both:
            clauses.append(c_set)

    # DPLL Implementation
    def dpll(clauses, assignment):
        # Unit propagation
        while True:
            unit_found = False
            for c in clauses:
                unfalsified = []
                satisfied = False
                for lit in c:
                    var = abs(lit)
                    val = assignment[var]
                    if val != -1:
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                    else:
                        unfalsified.append(lit)
                if satisfied:
                    continue
                if len(unfalsified) == 0:
                    return False
                if len(unfalsified) == 1:
                    lit = unfalsified[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else 0
                    if assignment[var] == -1:
                        assignment[var] = val
                        unit_found = True
                        break
            if not unit_found:
                break

        # Check if all clauses satisfied
        all_satisfied = True
        for c in clauses:
            sat = False
            for lit in c:
                var = abs(lit)
                val = assignment[var]
                if val != -1 and ((lit > 0 and val == 1) or (lit < 0 and val == 0)):
                    sat = True
                    break
            if not sat:
                all_satisfied = False
                break
        
        if all_satisfied:
            return True

        # Choose a variable using heuristic (MOM's or occurrence in shortest unsatisfied clauses)
        # Find minimum length of unsatisfied clause
        min_len = 1000
        for c in clauses:
            unfalsified = []
            satisfied = False
            for lit in c:
                var = abs(lit)
                val = assignment[var]
                if val != -1:
                    if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                        satisfied = True
                        break
                else:
                    unfalsified.append(lit)
            if not satisfied and len(unfalsified) < min_len:
                min_len = len(unfalsified)

        # Count literal occurrences in clauses of min_len
        lit_count = {}
        for c in clauses:
            unfalsified = []
            satisfied = False
            for lit in c:
                var = abs(lit)
                val = assignment[var]
                if val != -1:
                    if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                        satisfied = True
                        break
                else:
                    unfalsified.append(lit)
            if not satisfied and len(unfalsified) == min_len:
                for lit in unfalsified:
                    lit_count[lit] = lit_count.get(lit, 0) + 1

        if not lit_count:
            # All remaining variables can be set arbitrarily
            return True

        # Pick literal with highest frequency
        best_lit = max(lit_count, key=lit_count.get)
        best_var = abs(best_lit)
        best_val = 1 if best_lit > 0 else 0

        # Try best_val first
        assign_copy = list(assignment)
        assign_copy[best_var] = best_val
        if dpll(clauses, assign_copy):
            for i in range(1, n + 1):
                assignment[i] = assign_copy[i]
            return True

        # Try opposite value
        assign_copy = list(assignment)
        assign_copy[best_var] = 1 - best_val
        if dpll(clauses, assign_copy):
            for i in range(1, n + 1):
                assignment[i] = assign_copy[i]
            return True

        return False

    assignment = [-1] * (n + 1)
    if dpll(clauses, assignment):
        print("YES")
        res = [str(assignment[i] if assignment[i] != -1 else 0) for i in range(1, n + 1)]
        print(" ".join(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
