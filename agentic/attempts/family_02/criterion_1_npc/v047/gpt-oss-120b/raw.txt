```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        clause = [int(next(it)), int(next(it)), int(next(it))]
        clauses.append(clause)

    # assignment: 0 = False (CLOSED), 1 = True (OPEN), None = unassigned
    assign = [None] * (n + 1)

    # helper to get literal value under current assignment
    def lit_val(lit):
        v = abs(lit)
        a = assign[v]
        if a is None:
            return None
        return a if lit > 0 else (1 - a)

    # unit propagation and pure literal elimination
    def propagate():
        while True:
            # unit clause detection
            unit_lit = None
            for cl in clauses:
                val0 = lit_val(cl[0])
                val1 = lit_val(cl[1])
                val2 = lit_val(cl[2])
                # if any true, clause satisfied
                if (val0 == 1) or (val1 == 1) or (val2 == 1):
                    continue
                # count unassigned
                unassigned = []
                for lit, val in zip(cl, (val0, val1, val2)):
                    if val is None:
                        unassigned.append(lit)
                if not unassigned:  # all false -> conflict
                    return False
                if len(unassigned) == 1:
                    unit_lit = unassigned[0]
                    break
            if unit_lit is not None:
                v = abs(unit_lit)
                val = 1 if unit_lit > 0 else 0
                if assign[v] is not None:
                    if assign[v] != val:
                        return False
                else:
                    assign[v] = val
                unit_lit = None
                continue

            # pure literal elimination
            pos = [False] * (n + 1)
            neg = [False] * (n + 1)
            for cl in clauses:
                for lit in cl:
                    v = abs(lit)
                    if assign[v] is not None:
                        continue
                    if lit > 0:
                        pos[v] = True
                    else:
                        neg[v] = True
            pure_found = False
            for v in range(1, n + 1):
                if assign[v] is None:
                    if pos[v] and not neg[v]:
                        assign[v] = 1
                        pure_found = True
                    elif neg[v] and not pos[v]:
                        assign[v] = 0
                        pure_found = True
            if pure_found:
                continue
            break
        return True

    # check if all clauses satisfied
    def all_satisfied():
        for cl in clauses:
            if not any(lit_val(l) == 1 for l in cl):
                return False
        return True

    # DPLL recursive
    def dpll():
        if not propagate():
            return False
        if all(assign[v] is not None for v in range(1, n + 1)):
            return all_satisfied()
        # choose unassigned variable
        for v in range(1, n + 1):
            if assign[v] is None:
                var = v
                break
        # try True then False
        saved = assign[:]
        assign[var] = 1
        if dpll():
            return True
        assign[:] = saved
        assign[var] = 0
        if dpll():
            return True
        assign[:] = saved
        return False

    if dpll():
        print("YES")
        print(' '.join(str(assign[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
