import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(['0'] * n))
        return
        
    clauses = []
    idx = 2
    for _ in range(m):
        v1, v2, v3 = int(data[idx]), int(data[idx+1]), int(data[idx+2])
        idx += 3
        clauses.append((v1, v2, v3))
        
    # 2-SAT solver using SCC (Kosaraju's algorithm)
    num_vars = n
    num_nodes = 2 * num_vars
    
    def get_node(x):
        if x > 0:
            return 2 * (x - 1)
        else:
            return 2 * (-x - 1) + 1
            
    adj = [[] for _ in range(num_nodes)]
    radj = [[] for _ in range(num_nodes)]
    
    # Add implications for each vulnerability
    # A vulnerability (a, b, c) means (a OR b OR c)
    # Which translates to:
    # ~a -> b, ~b -> a
    # ~a -> c, ~c -> a
    # ~b -> c, ~c -> b
    for v1, v2, v3 in clauses:
        na, nb, nc = get_node(-v1), get_node(-v2), get_node(-v3)
        a, b, c = get_node(v1), get_node(v2), get_node(v3)
        
        adj[na].append(b); radj[b].append(na)
        adj[na].append(c); radj[c].append(na)
        adj[nb].append(a); radj[a].append(nb)
        adj[nb].append(c); radj[c].append(nb)
        adj[nc].append(a); radj[a].append(nc)
        adj[nc].append(b); radj[b].append(nc)
        
    # Kosaraju's algorithm to find SCCs
    visited = [False] * num_nodes
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            v, i = stack[-1]
            if i == 0:
                visited[v] = True
            if i < len(adj[v]):
                neighbor = adj[v][i]
                stack[-1] = (v, i + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(v)
                
    for i in range(num_nodes):
        if not visited[i]:
            dfs1(i)
            
    comp_id = [-1] * num_nodes
    current_comp = 0
    
    def dfs2(u, cid):
        stack = [u]
        comp_id[u] = cid
        while stack:
            v = stack[-1]
            unvisited_found = False
            for neighbor in radj[v]:
                if comp_id[neighbor] == -1:
                    comp_id[neighbor] = cid
                    stack.append(neighbor)
                    unvisited_found = True
                    break
            if not unvisited_found:
                stack.pop()
                
    for u in reversed(order):
        if comp_id[u] == -1:
            dfs2(u, current_comp)
            current_comp += 1
            
    # Check for unsatisfiability
    assignment = [-1] * num_vars
    for i in range(num_vars):
        if comp_id[2 * i] == comp_id[2 * i + 1]:
            print("NO")
            return
        assignment[i] = 1 if comp_id[2 * i] > comp_id[2 * i + 1] else 0
        
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == '__main__':
    solve()
