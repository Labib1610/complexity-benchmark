We can model each perspective as a clause with three literals, where each literal is either a variable (zone) or its negation.  
This is exactly the **3-SAT** problem.  

Since the constraints are small (at most 90 zones and 400 perspectives), we can solve it with a backtracking search, assigning values to variables and checking clauses.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    clauses = []
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx+1])
        l3 = int(input_data[idx+2])
        idx += 3
        clauses.append((l1, l2, l3))
    
    assignment = [-1] * (n + 1)  # -1 unassigned, 0 cool, 1 warm
    
    def check_clause(clause, assignment):
        for lit in clause:
            var = abs(lit)
            val = assignment[var]
            if val == -1:
                continue
            if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                return True
        return False
    
    def backtrack(var_idx):
        if var_idx > n:
            # all assigned, check all clauses
            for clause in clauses:
                if not check_clause(clause, assignment):
                    return False
            return True
        
        # Try both values
        for val in (0, 1):
            assignment[var_idx] = val
            # Quick check: if any clause is already impossible (all assigned and false)
            possible = True
            for clause in clauses:
                all_assigned = all(assignment[abs(lit)] != -1 for lit in clause)
                if all_assigned and not check_clause(clause, assignment):
                    possible = False
                    break
            if possible:
                if backtrack(var_idx + 1):
                    return True
        assignment[var_idx] = -1
        return False
    
    if backtrack(1):
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
