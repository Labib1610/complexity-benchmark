```python
import sys

# Set recursion depth for DPLL
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx + 1]), int(input_data[idx + 2])]
        idx += 3
        c_set = set(c)
        tautology = False
        for lit in c_set:
            if -lit in c_set:
                tautology = True
                break
        if not tautology:
            clauses.append(list(c_set))

    def dpll(clauses, assignment):
        # Unit propagation
        while True:
            unit_found = False
            for c in clauses:
                if len(c) == 0:
                    return None
                if len(c) == 1:
                    unit_lit = c[0]
                    var = abs(unit_lit)
                    val = 1 if unit_lit > 0 else 0
                    if var in assignment and assignment[var] != val:
                        return None
                    assignment[var] = val

                    new_clauses = []
                    for cl in clauses:
                        if unit_lit in cl:
                            continue
                        if -unit_lit in cl:
                            new_cl = [l for l in cl if l != -unit_lit]
                            new_clauses.append(new_cl)
                        else:
                            new_clauses.append(cl)
                    clauses = new_clauses
                    unit_found = True
                    break
            if not unit_found:
                break

        if len(clauses) == 0:
            return assignment

        for c in clauses:
            if len(c) == 0:
                return None

        # Jeroslow-Wang branching heuristic
        counts = {}
        for c in clauses:
            w = 2 ** (-len(c))
            for lit in c:
                counts[lit] = counts.get(lit, 0) + w

        best_lit = max(counts, key=counts.get)
        var = abs(best_lit)

        for chosen_lit in [best_lit, -best_lit]:
            val = 1 if chosen_lit > 0 else 0
            new_assign = assignment.copy()
            new_assign[var] = val

            new_clauses = []
            conflict = False
            for cl in clauses:
                if chosen_lit in cl:
                    continue
                if -chosen_lit in cl:
                    new_cl = [l for l in cl if l != -chosen_lit]
                    if len(new_cl) == 0:
                        conflict = True
                        break
                    new_clauses.append(new_cl)
                else:
                    new_clauses.append(cl)

            if not conflict:
                res = dpll(new_clauses, new_assign)
                if res is not None:
                    return res
        return None

    res = dpll(clauses, {})
    if res is None:
        print("NO")
    else:
        print("YES")
        ans = [str(res.get(i, 0)) for i in range(1, n + 1)]
        print(" ".join(ans))


if __name__ == "__main__":
    solve()
```
