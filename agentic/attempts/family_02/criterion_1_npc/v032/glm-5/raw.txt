```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Each clause is (a, b, c) where each is (channel, expected_phase)
    # expected_phase is 1 if positive, 0 if negative
    clauses = []
    for _ in range(m):
        clause = []
        for _ in range(3):
            val = int(next(it))
            channel = abs(val)
            expected = 1 if val > 0 else 0
            clause.append((channel, expected))
        clauses.append(clause)
    
    # Try all 2^n configurations
    for mask in range(1 << n):
        valid = True
        for clause in clauses:
            # Check if at least one connection matches
            clause_satisfied = False
            for channel, expected in clause:
                actual = (mask >> (channel - 1)) & 1
                if actual == expected:
                    clause_satisfied = True
                    break
            if not clause_satisfied:
                valid = False
                break
        if valid:
            result = [(mask >> i) & 1 for i in range(n-1, -1, -1)]
            print("YES")
            print(" ".join(map(str, result)))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
```
