import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c1 = int(input_data[idx])
        c2 = int(input_data[idx+1])
        c3 = int(input_data[idx+2])
        idx += 3
        clauses.append([c1, c2, c3])
        
    assignment = {}
    
    def get_lit_val(lit, assign):
        var = abs(lit)
        if var not in assign:
            return None
        val = assign[var]
        return val if lit > 0 else (1 - val)

    def unit_propagate(assign):
        while True:
            unit_found = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    val = get_lit_val(lit, assign)
                    if val == 1:
                        satisfied = True
                        break
                    elif val is None:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False, assign # Conflict
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else 0
                    if var in assign:
                        if assign[var] != val:
                            return False, assign
                    else:
                        assign[var] = val
                        unit_found = True
            if not unit_found:
                break
        return True, assign

    def dpll(assign):
        sat, assign = unit_propagate(assign)
        if not sat:
            return None
        
        # Check if all clauses are satisfied
        all_satisfied = True
        for clause in clauses:
            if not any(get_lit_val(lit, assign) == 1 for lit in clause):
                all_satisfied = False
                break
        if all_satisfied:
            # Assign any remaining variables
            for v in range(1, n + 1):
                if v not in assign:
                    assign[v] = 0
            return assign
        
        # Choose next variable using a heuristic (most frequent unassigned literal in shortest unsatisfied clauses)
        counts = {}
        for clause in clauses:
            sat = False
            unassigned = []
            for lit in clause:
                val = get_lit_val(lit, assign)
                if val == 1:
                    sat = True
                    break
                elif val is None:
                    unassigned.append(lit)
            if not sat:
                weight = 1.0 / (2 ** len(unassigned)) if unassigned else 1.0
                for lit in unassigned:
                    counts[lit] = counts.get(lit, 0) + weight
                    
        if not counts:
            for v in range(1, n + 1):
                if v not in assign:
                    assign[v] = 0
            return assign
        
        # Pick literal with highest count
        best_lit = max(counts, key=counts.get)
        best_var = abs(best_lit)
        preferred_val = 1 if best_lit > 0 else 0
        
        # Try preferred value first
        new_assign = assign.copy()
        new_assign[best_var] = preferred_val
        res = dpll(new_assign)
        if res is not None:
            return res
        
        # Try other value
        new_assign = assign.copy()
        new_assign[best_var] = 1 - preferred_val
        return dpll(new_assign)

    res = dpll(assignment)
    if res is None:
        print("NO")
    else:
        print("YES")
        ans = [str(res[i]) for i in range(1, n + 1)]
        print(" ".join(ans))

if __name__ == '__main__':
    solve()
