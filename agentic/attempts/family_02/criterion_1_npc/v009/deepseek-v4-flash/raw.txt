We can model each rule as a clause in 3-SAT, where each literal is a switch variable or its negation.  
Since each clause is satisfied if at least one literal is true, we can solve it using a SAT solver approach.  
For this problem, we can use a simple backtracking with unit propagation, or use a library like `pycosat` if available, but to be safe we can implement a basic DPLL.

Here’s the Python solution:

```python
import sys

def main():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clause = []
        for lit in (l1, l2, l3):
            if lit > 0:
                clause.append((lit - 1, True))
            else:
                clause.append((-lit - 1, False))
        clauses.append(clause)

    # DPLL solver
    assignment = [None] * n

    def propagate(assign):
        # Returns (new_assign, conflict)
        assign = assign[:]
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for var, val in clause:
                    if assign[var] is None:
                        unassigned.append((var, val))
                    elif assign[var] == val:
                        satisfied = True
                        break
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return None, True
                if len(unassigned) == 1:
                    var, val = unassigned[0]
                    if assign[var] is None:
                        assign[var] = val
                        changed = True
                    elif assign[var] != val:
                        return None, True
        return assign, False

    def solve(assign):
        assign, conflict = propagate(assign)
        if conflict:
            return None
        # Check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            sat = False
            for var, val in clause:
                if assign[var] is not None and assign[var] == val:
                    sat = True
                    break
            if not sat:
                all_sat = False
                break
        if all_sat:
            return assign
        # Choose unassigned variable
        for i in range(n):
            if assign[i] is None:
                # Try True
                a1 = assign[:]
                a1[i] = True
                res = solve(a1)
                if res is not None:
                    return res
                # Try False
                a2 = assign[:]
                a2[i] = False
                res = solve(a2)
                if res is not None:
                    return res
                return None
        return None

    result = solve(assignment)
    if result is None:
        print("NO")
    else:
        print("YES")
        print(" ".join('1' if result[i] else '0' for i in range(n)))

if __name__ == "__main__":
    main()
```
