```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    if m == 0:
        print("YES")
        print(" ".join("0" for _ in range(n)))
        return

    # Convert literal to internal 0-based integer
    # Variable i (1..n): +i -> 2*(i-1), -i -> 2*(i-1) + 1
    def to_lit(x):
        v = abs(x) - 1
        return 2 * v if x > 0 else 2 * v + 1

    clauses = []
    idx = 2
    for _ in range(m):
        l1 = to_lit(int(input_data[idx]))
        l2 = to_lit(int(input_data[idx+1]))
        l3 = to_lit(int(input_data[idx+2]))
        idx += 3
        # Remove duplicate literals in the same clause
        c = list(set([l1, l2, l3]))
        # If clause contains both x and ~x, it's always satisfied, can be ignored
        tautology = False
        for lit in c:
            if (lit ^ 1) in c:
                tautology = True
                break
        if not tautology:
            clauses.append(c)

    # CDCL SAT Solver
    num_vars = n
    
    # watches[lit] = list of clause indices watching lit
    watches = [[] for _ in range(2 * num_vars)]
    
    # clause database
    clause_db = []
    
    for c in clauses:
        clause_db.append(c)
    
    # assignments: -1 unassigned, 0 false, 1 true
    assigns = [-1] * num_vars
    # trail of assigned literals
    trail = []
    # decision level of each variable
    level = [-1] * num_vars
    # reason clause of each variable (clause index or -1 for decision)
    reason = [-1] * num_vars
    
    # Activity for VSIDS
    activity = [0.0] * num_vars
    var_inc = 1.0
    var_decay = 0.95

    def lit_val(lit):
        v = lit >> 1
        s = lit & 1
        a = assigns[v]
        if a == -1:
            return -1
        return 1 if (a == 1 and s == 0) or (a == 0 and s == 1) else 0

    # Attach initial clauses
    for ci, c in enumerate(clause_db):
        if len(c) == 0:
            print("NO")
            return
        elif len(c) == 1:
            watches[c[0]].append(ci)
        else:
            watches[c[0]].append(ci)
            watches[c[1]].append(ci)

    decision_level = 0

    def assign_lit(lit, r):
        v = lit >> 1
        s = lit & 1
        assigns[v] = 1 if s == 0 else 0
        level[v] = decision_level
        reason[v] = r
        trail.append(lit)

    def propagate():
        nonlocal trail_head
        while trail_head < len(trail):
            p = trail[trail_head]
            trail_head += 1
            false_lit = p ^ 1
            
            ws = watches[false_lit]
            watches[false_lit] = []
            
            i = 0
            while i < len(ws):
                ci = ws[i]
                c = clause_db[ci]
                
                # Make sure false_lit is at c[1]
                if c[0] == false_lit:
                    c[0], c[1] = c[1], c[0]
                
                # If c[0] is true, clause is satisfied
                if lit_val(c[0]) == 1:
                    watches[false_lit].append(ci)
                    i += 1
                    continue
                
                # Look for a new literal to watch
                found = False
                for k in range(2, len(c)):
                    if lit_val(c[k]) != 0:
                        c[1], c[k] = c[k], c[1]
                        watches[c[1]].append(ci)
                        found = True
                        break
                
                if found:
                    i += 1
                    continue
                
                watches[false_lit].append(ci)
                i += 1
                
                # Clause is unit or conflict
                if lit_val(c[0]) == 0:
                    # Conflict! Put remaining watches back
                    for j in range(i, len(ws)):
                        watches[false_lit].append(ws[j])
                    return ci
                else:
                    assign_lit(c[0], ci)
        return -1

    def analyze(confl):
        nonlocal var_inc
        learnt = [0]
        path_c = 0
        p = -1
        idx = len(trail) - 1
        
        c = clause_db[confl]
        while True:
            for lit in c:
                v = lit >> 1
                if lit == p:
                    continue
                if level[v] > 0 and v not in seen:
                    seen.add(v)
                    activity[v] += var_inc
                    if level[v] == decision_level:
                        path_c += 1
                    else:
                        learnt.append(lit)
            
            while True:
                p = trail[idx]
                idx -= 1
                if (p >> 1) in seen:
                    break
            
            seen.remove(p >> 1)
            path_c -= 1
            if path_c == 0:
                learnt[0] = p ^ 1
                break
            c = clause_db[reason[p >> 1]]
            
        for v in seen:
            pass # seen will be cleared
        seen.clear()
        
        # Find backtrack level
        if len(learnt) <= 1:
            bt_level = 0
        else:
            max_i = 1
            for k in range(2, len(learnt)):
                if level[learnt[k] >> 1] > level[learnt[max_i] >> 1]:
                    max_i = k
            learnt[1], learnt[max_i] = learnt[max_i], learnt[1]
            bt_level = level[learnt[1] >> 1]
            
        return learnt, bt_level

    seen = set()
    trail_head = 0

    # Initial unit propagation
    for ci, c in enumerate(clause_db):
        if len(c) == 1:
            if lit_val(c[0]) == 0:
                print("NO")
                return
            elif lit_val(c[0]) == -1:
                assign_lit(c[0], ci)

    if propagate() != -1:
        print("NO")
        return

    def backtrack_to(target_level):
        nonlocal trail_head, decision_level
        while len(trail) > 0 and level[trail[-1] >> 1] > target_level:
            lit = trail.pop()
            v = lit >> 1
            assigns[v] = -1
            level[v] = -1
            reason[v] = -1
        trail_head = len(trail)
        decision_level = target_level

    def pick_branch_lit():
        max_act = -1
        best_v = -1
        for v in range(num_vars):
            if assigns[v] == -1:
                if activity[v] > max_act:
                    max_act = activity[v]
                    best_v = v
        if best_v == -1:
            return -1
        return 2 * best_v # default try positive

    # Main CDCL loop
    while True:
        confl = propagate()
        if confl != -1:
            if decision_level == 0:
                print("NO")
                return
            learnt, bt_level = analyze(confl)
            backtrack_to(bt_level)
            
            ci = len(clause_db)
            clause_db.append(learnt)
            if len(learnt) == 1:
                watches[learnt[0]].append(ci)
                assign_lit(learnt[0], ci)
            else:
                watches[learnt[0]].append(ci)
                watches[learnt[1]].append(ci)
                assign_lit(learnt[0], ci)
                
            var_inc /= var_decay
        else:
            p = pick_branch_lit()
            if p == -1:
                # All assigned, solution found
                print("YES")
                res = [str(assigns[i]) if assigns[i] != -1 else "0" for i in range(n)]
                print(" ".join(res))
                return
            decision_level += 1
            assign_lit(p, -1)

solve()
```
