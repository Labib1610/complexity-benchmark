def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    clauses = []
    index = 2
    for _ in range(m):
        clause = []
        for _ in range(3):
            val = int(data[index])
            index += 1
            clause.append(val)
        clauses.append(clause)
    
    # Convert to SAT problem
    # Variable i (1-based) -> valve i is OPEN -> variable i is True
    # Variable i (1-based) -> valve i is CLOSED -> variable i is False -> literal -i
    
    # Each clause is a disjunction of literals
    # For clause [a, b, c], we want at least one of them to be satisfied
    # This is directly a 3-SAT clause
    
    # We'll use a simple backtracking algorithm since n <= 90 but in practice,
    # the constraints make it feasible
    
    # Let's try all assignments (not feasible for n=90, so we need a better approach)
    # Actually, let's implement a simple DPLL-style solver
    
    # But given the constraints, especially that each variable appears in at most 1200 clauses,
    # and sum of |ID| <= 108000, this suggests a sparse formula.
    
    # Let's build a more efficient implementation
    
    # Map from variable (1-based) to list of clauses where it appears positively/negatively
    from collections import defaultdict
    
    pos_clauses = defaultdict(list)  # var -> list of clause indices where var appears positively
    neg_clauses = defaultdict(list)  # var -> list of clause indices where var appears negatively
    
    # Also keep track of clauses as sets of literals for fast checking
    clause_sets = []
    original_clauses = []
    
    for idx, clause in enumerate(clauses):
        original_clauses.append(clause)
        clause_set = set()
        for lit in clause:
            clause_set.add(lit)
            var = abs(lit)
            if lit > 0:
                pos_clauses[var].append(idx)
            else:
                neg_clauses[var].append(idx)
        clause_sets.append(clause_set)
    
    # Current assignment: 0 = unassigned, 1 = True, -1 = False
    assignment = [0] * (n + 1)
    # Stack for backtracking: (var, value, clauses_satisfied_at_this_decision)
    stack = []
    # Clauses that are already satisfied
    satisfied = [False] * m
    
    def is_clause_satisfied(c_idx):
        clause = original_clauses[c_idx]
        for lit in clause:
            var = abs(lit)
            val = assignment[var]
            if val == 0:
                continue
            if (lit > 0 and val == 1) or (lit < 0 and val == -1):
                return True
        return False
    
    def unit_propagation():
        changed = True
        while changed:
            changed = False
            for i in range(m):
                if satisfied[i]:
                    continue
                num_unassigned = 0
                unassigned_lit = 0
                num_false = 0
                
                clause = original_clauses[i]
                for lit in clause:
                    var = abs(lit)
                    val = assignment[var]
                    if val == 0:
                        num_unassigned += 1
                        unassigned_lit = lit
                    elif (lit > 0 and val == 1) or (lit < 0 and val == -1):
                        # Literal is true, so clause is satisfied
                        satisfied[i] = True
                        break
                    else:
                        # Literal is false
                        num_false += 1
                
                if satisfied[i]:
                    continue
                    
                if num_unassigned == 0:
                    # All literals are false, conflict
                    return False
                elif num_unassigned == 1:
                    # Unit clause
                    var = abs(unassigned_lit)
                    val = 1 if unassigned_lit > 0 else -1
                    if assignment[var] == 0:
                        assignment[var] = val
                        stack[-1][2].append(i)  # Track which clause is satisfied by this decision
                        changed = True
                    elif assignment[var] != val:
                        # Conflict
                        return False
        return True
    
    def pick_branching_variable():
        # Pick the first unassigned variable
        for i in range(1, n + 1):
            if assignment[i] == 0:
                return i
        return None
    
    def solve_rec():
        # First do unit propagation
        if not unit_propagation():
            return False
            
        # Check if all clauses are satisfied
        if all(satisfied):
            return True
            
        # Pick a variable to branch on
        var = pick_branching_variable()
        if var is None:
            return all(satisfied)
            
        # Try var = True
        assignment[var] = 1
        newly_satisfied = []
        stack.append((var, 1, newly_satisfied))
        
        if solve_rec():
            return True
            
        # Backtrack
        assignment[var] = 0
        for c_idx in stack.pop()[2]:
            satisfied[c_idx] = False
            
        # Try var = False
        assignment[var] = -1
        newly_satisfied = []
        stack.append((var, -1, newly_satisfied))
        
        if solve_rec():
            return True
            
        # Backtrack
        assignment[var] = 0
        for c_idx in stack.pop()[2]:
            satisfied[c_idx] = False
            
        return False
    
    # Start solving
    if solve_rec():
        print("YES")
        result = []
        for i in range(1, n + 1):
            if assignment[i] == 1:
                result.append("1")
            else:
                result.append("0")
        print(" ".join(result))
    else:
        print("NO")

solve()
