We can model each valve as a Boolean variable and each interlock as a clause of three literals.  
This becomes a **3-SAT** problem, but with special constraints that allow a simpler approach.

We can solve it using **backtracking with unit propagation** (DPLL), which is efficient enough given the limits.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    clauses = []
    idx = 2
    for _ in range(m):
        c1 = int(input_data[idx])
        c2 = int(input_data[idx+1])
        c3 = int(input_data[idx+2])
        idx += 3
        clauses.append((c1, c2, c3))
    
    assignment = [None] * (n + 1)  # 1-indexed, None = unassigned
    
    # DPLL with unit propagation
    def propagate(clauses, assignment):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    var = abs(lit)
                    val = assignment[var]
                    if val is not None:
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                    else:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else 0
                    if assignment[var] is not None and assignment[var] != val:
                        return False
                    assignment[var] = val
                    changed = True
        return True
    
    def backtrack(clauses, assignment):
        if not propagate(clauses, assignment):
            return False
        # Find unassigned variable
        for var in range(1, n+1):
            if assignment[var] is None:
                # Try true
                assignment[var] = 1
                if backtrack(clauses, assignment):
                    return True
                # Try false
                assignment[var] = 0
                if backtrack(clauses, assignment):
                    return True
                assignment[var] = None
                return False
        # All assigned, check all clauses satisfied
        for clause in clauses:
            ok = False
            for lit in clause:
                var = abs(lit)
                val = assignment[var]
                if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                    ok = True
                    break
            if not ok:
                return False
        return True
    
    if backtrack(clauses, assignment):
        print("YES")
        print(" ".join(str(1 if assignment[i] == 1 else 0) for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
