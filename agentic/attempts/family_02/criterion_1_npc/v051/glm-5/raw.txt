import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT: variables are 1..n, each has two literals: i (true) and -i (false)
    # We map literal x to node 2*(|x|-1) + (1 if x > 0 else 0)
    # So for gene i: i (expressed) -> 2*(i-1)+1, -i (suppressed) -> 2*(i-1)
    # Total nodes: 2*n
    
    N = 2 * n
    adj = [[] for _ in range(N)]
    radj = [[] for _ in range(N)]
    
    def get_node(lit):
        # lit is non-zero, |lit| <= n
        idx = abs(lit) - 1
        if lit > 0:
            return 2 * idx + 1
        else:
            return 2 * idx
    
    def get_neg(node):
        return node ^ 1
    
    # For each stressor, we have a clause (l1 OR l2 OR l3)
    # This is equivalent to 3 2-clauses:
    # (!l1 -> l2), (!l1 -> l3), (!l2 -> l3)
    # Actually, (A OR B OR C) is equivalent to:
    # (!A -> B) AND (!A -> C) AND (!B -> C)
    # Wait, let's check: (A OR B OR C) is true if at least one is true.
    # If A is false, then B OR C must be true.
    # If B is false, then A OR C must be true.
    # If C is false, then A OR B must be true.
    # So the implications are:
    # !A -> B, !A -> C
    # !B -> A, !B -> C
    # !C -> A, !C -> B
    # Total 6 implications.
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        
        u1, u2, u3 = get_node(l1), get_node(l2), get_node(l3)
        nu1, nu2, nu3 = get_neg(u1), get_neg(u2), get_neg(u3)
        
        # Add implications
        adj[nu1].append(u2)
        adj[nu2].append(u1)
        adj[nu1].append(u3)
        adj[nu3].append(u1)
        adj[nu2].append(u3)
        adj[nu3].append(u2)
        
        radj[u2].append(nu1)
        radj[u1].append(nu2)
        radj[u3].append(nu1)
        radj[u1].append(nu3)
        radj[u3].append(nu2)
        radj[u2].append(nu3)
    
    # Kosaraju's algorithm to find SCCs
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if idx == 0:
                if visited[node]:
                    stack.pop()
                    continue
                visited[node] = True
            if idx < len(adj[node]):
                nei = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[nei]:
                    stack.append((nei, 0))
            else:
                order.append(node)
                stack.pop()
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * N
    current_comp = 0
    
    def dfs2(u, label):
        stack = [u]
        comp[u] = label
        while stack:
            node = stack.pop()
            for nei in radj[node]:
                if comp[nei] == -1:
                    comp[nei] = label
                    stack.append(nei)
    
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, current_comp)
            current_comp += 1
    
    # Check for contradiction
    for i in range(n):
        if comp[2*i] == comp[2*i + 1]:
            print("NO")
            return
    
    # Assign values
    # If comp[x] > comp[!x], then x must be true
    # Because in topological order of SCCs, x comes after !x
    # So !x -> x, meaning x must be true
    assignment = [0] * n
    for i in range(n):
        if comp[2*i + 1] > comp[2*i]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
