We can model each facility as a boolean variable, and each checkpoint as a clause that forbids exactly one combination of three literals.  
That means each checkpoint is a clause of the form:  

\[
(x_a \lor x_b \lor x_c)
\]

where each literal is either a variable or its negation.  
We need to satisfy all clauses — this is a **3-SAT** problem.  

Since \( n \le 90 \) and \( m \le 400 \), we can use a backtracking algorithm with unit propagation and pure literal elimination, which is fast enough for this size.

Here’s the Python solution:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # Convert literals: positive v -> variable v-1, negative v -> variable -v-1
    # We'll store as (var, is_true) where is_true means literal is positive.
    def lit_to_pair(lit):
        if lit > 0:
            return (lit - 1, True)
        else:
            return (-lit - 1, False)
    
    clause_list = []
    for c in clauses:
        clause_list.append([lit_to_pair(x) for x in c])
    
    assignment = [None] * n  # None = unassigned, True/False = value
    
    # Unit propagation and pure literal elimination
    def propagate():
        changed = True
        while changed:
            changed = False
            # Unit propagation
            for clause in clause_list:
                unassigned = []
                satisfied = False
                for var, val in clause:
                    if assignment[var] is None:
                        unassigned.append((var, val))
                    elif assignment[var] == val:
                        satisfied = True
                        break
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    var, val = unassigned[0]
                    if assignment[var] is None:
                        assignment[var] = val
                        changed = True
                    elif assignment[var] != val:
                        return False
            # Pure literal elimination
            # Count occurrences
            pos_count = {}
            neg_count = {}
            for clause in clause_list:
                for var, val in clause:
                    if assignment[var] is not None:
                        continue
                    if val:
                        pos_count[var] = pos_count.get(var, 0) + 1
                    else:
                        neg_count[var] = neg_count.get(var, 0) + 1
            for var in range(n):
                if assignment[var] is None:
                    p = pos_count.get(var, 0)
                    q = neg_count.get(var, 0)
                    if p > 0 and q == 0:
                        assignment[var] = True
                        changed = True
                    elif q > 0 and p == 0:
                        assignment[var] = False
                        changed = True
        return True
    
    def backtrack():
        # Find first unassigned variable
        try:
            var = next(i for i in range(n) if assignment[i] is None)
        except StopIteration:
            return True  # all assigned
        # Try both values
        for val in [True, False]:
            assignment[var] = val
            saved = assignment[:]
            if propagate():
                if backtrack():
                    return True
            assignment[:] = saved
        assignment[var] = None
        return False
    
    if not propagate():
        print("NO")
        return
    if backtrack():
        print("YES")
        print(' '.join('1' if x else '0' for x in assignment))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
