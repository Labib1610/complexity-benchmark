import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Each checkpoint is a clause: (l1 OR l2 OR l3)
    # Literal v: if v > 0, variable v must be True (1)
    #            if v < 0, variable -v must be False (0)
    
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # 2-SAT setup
    # Variables are 1..n
    # Implication graph nodes: 2*i for False, 2*i+1 for True (0-indexed nodes)
    # Or we can use 2*i and 2*i+1 for i in 0..n-1
    
    N = 2 * n
    adj = [[] for _ in range(N)]
    adj_rev = [[] for _ in range(N)]
    
    def get_node(var, is_true):
        # var is 1-indexed
        idx = (var - 1) * 2
        return idx + (1 if is_true else 0)
    
    def add_implication(u, v):
        adj[u].append(v)
        adj_rev[v].append(u)
    
    for l1, l2, l3 in clauses:
        # Clause: (l1 or l2 or l3)
        # Transform to implications
        # (not l1 -> l2), (not l2 -> l1)
        # (not l1 -> l3), (not l3 -> l1)
        # (not l2 -> l3), (not l3 -> l2)
        
        vars_lit = []
        for lit in [l1, l2, l3]:
            var = abs(lit)
            is_true = (lit > 0)
            vars_lit.append((var, is_true))
        
        v1, t1 = vars_lit[0]
        v2, t2 = vars_lit[1]
        v3, t3 = vars_lit[2]
        
        # (not l1 -> l2)
        add_implication(get_node(v1, not t1), get_node(v2, t2))
        # (not l2 -> l1)
        add_implication(get_node(v2, not t2), get_node(v1, t1))
        
        # (not l1 -> l3)
        add_implication(get_node(v1, not t1), get_node(v3, t3))
        # (not l3 -> l1)
        add_implication(get_node(v3, not t3), get_node(v1, t1))
        
        # (not l2 -> l3)
        add_implication(get_node(v2, not t2), get_node(v3, t3))
        # (not l3 -> l2)
        add_implication(get_node(v3, not t3), get_node(v2, t2))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * N
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    comp_id = [-1] * N
    current_comp = 0
    
    def dfs2(u):
        comp_id[u] = current_comp
        for v in adj_rev[u]:
            if comp_id[v] == -1:
                dfs2(v)
    
    for u in reversed(order):
        if comp_id[u] == -1:
            dfs2(u)
            current_comp += 1
    
    # Check for contradiction: x and not x in same SCC
    assignment = [False] * n
    possible = True
    
    for i in range(n):
        node_true = get_node(i + 1, True)
        node_false = get_node(i + 1, False)
        if comp_id[node_true] == comp_id[node_false]:
            possible = False
            break
        # If comp_id[True] > comp_id[False], then True is topologically later, so we pick True
        # Since we process in reverse topological order, higher comp_id comes first in topo order
        # So if comp_id[True] > comp_id[False], True is earlier in topo order (processed first in reverse)
        # Wait, standard Kosaraju: order is finishing times. Reverse order is topological order of SCCs.
        # The first SCC in topological order has the smallest finishing time? No.
        # Let's trace: order has nodes in increasing finishing time.
        # Reverse order processes nodes in decreasing finishing time.
        # The first SCC found in dfs2 is the one with the largest finishing time node, which is a source SCC.
        # So comp_ids are assigned in topological order (source first).
        # We want to assign True if comp_id[True] > comp_id[False]?
        # No. If comp_id[True] > comp_id[False], then False is in an earlier SCC (topo order).
        # Implication: False -> True. If we pick False, we must pick True. Contradiction.
        # So we must pick True.
        # Correct logic: assign True if comp_id[True] > comp_id[False].
        assignment[i] = comp_id[node_true] > comp_id[node_false]
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(" ".join("1" if x else "0" for x in assignment))

if __name__ == "__main__":
    solve()
