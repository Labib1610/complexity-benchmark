We can model each rule as a 3-literal clause, where each literal is either a variable or its negation.  
This is exactly the **3-SAT** problem. Since the constraints are small (n up to 99, m up to 400), we can use a backtracking search with unit propagation.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    clauses = []
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx+1])
        l3 = int(input_data[idx+2])
        idx += 3
        clauses.append((l1, l2, l3))
    
    assignment = [None] * (n + 1)  # 1-indexed, None = unassigned
    
    def propagate():
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    var = abs(lit)
                    val = assignment[var]
                    if val is not None:
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                    else:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else 0
                    if assignment[var] is not None and assignment[var] != val:
                        return False
                    if assignment[var] is None:
                        assignment[var] = val
                        changed = True
        return True
    
    def backtrack():
        if not propagate():
            return False
        # find unassigned variable
        try:
            var = next(i for i in range(1, n+1) if assignment[i] is None)
        except StopIteration:
            return True
        # try both values
        for val in (0, 1):
            assignment[var] = val
            if backtrack():
                return True
            assignment[var] = None
        return False
    
    if backtrack():
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
