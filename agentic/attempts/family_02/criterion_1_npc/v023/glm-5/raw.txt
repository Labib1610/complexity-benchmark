import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Build implication graph for 2-SAT
    # Variables: x_1, ..., x_n (True = Refrigerated, False = Ambient)
    # Node indices: 2*i for x_i (False state), 2*i+1 for x_i (True state)
    # For i in 0..n-1: x_i uses nodes 2*i and 2*i+1
    
    N = 2 * n
    adj = [[] for _ in range(N)]
    adj_rev = [[] for _ in range(N)]
    
    def get_nodes(lit):
        """Convert literal to (node_index, is_true)"""
        var = abs(lit) - 1
        is_true = lit > 0
        return var, is_true
    
    def add_implication(u_node, v_node):
        adj[u_node].append(v_node)
        adj_rev[v_node].append(u_node)
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        
        # Each rule: (l1 OR l2 OR l3)
        # Convert to 2-SAT clauses:
        # (not l1 -> l2) and (not l1 -> l3)
        # (not l2 -> l1) and (not l2 -> l3)
        # (not l3 -> l1) and (not l3 -> l2)
        
        literals = [l1, l2, l3]
        for i in range(3):
            for j in range(3):
                if i != j:
                    # not literals[i] -> literals[j]
                    var_i, is_true_i = get_nodes(literals[i])
                    var_j, is_true_j = get_nodes(literals[j])
                    
                    # Node for literals[i] being true
                    node_i_true = 2 * var_i + 1
                    # Node for literals[i] being false
                    node_i_false = 2 * var_i
                    
                    # Node for literals[j] being true
                    node_j_true = 2 * var_j + 1
                    # Node for literals[j] being false
                    node_j_false = 2 * var_j
                    
                    # not literals[i]: if literals[i] is positive, not means false
                    # if literals[i] is negative, not means true
                    if is_true_i:
                        # not literals[i] means x_i = false
                        u = node_i_false
                    else:
                        # not literals[i] means x_i = true
                        u = node_i_true
                    
                    # literals[j]
                    if is_true_j:
                        v = node_j_true
                    else:
                        v = node_j_false
                    
                    add_implication(u, v)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * N
    order = []
    
    def dfs1(v):
        stack = [(v, 0)]
        while stack:
            node, idx = stack[-1]
            if idx == 0:
                visited[node] = True
            if idx < len(adj[node]):
                neighbor = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * N
    comp = [-1] * N
    comp_id = 0
    
    def dfs2(v, cid):
        stack = [v]
        visited[v] = True
        while stack:
            node = stack.pop()
            comp[node] = cid
            for neighbor in adj_rev[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    stack.append(neighbor)
    
    for v in reversed(order):
        if not visited[v]:
            dfs2(v, comp_id)
            comp_id += 1
    
    # Check if any variable has both states in the same SCC
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * n
    for i in range(n):
        # If x_i=True has higher component id, set to True
        if comp[2*i+1] > comp[2*i]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
