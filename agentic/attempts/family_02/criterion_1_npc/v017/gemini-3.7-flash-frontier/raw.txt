```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))

    def lit_to_internal(x):
        v = abs(x)
        return (v << 1) if x > 0 else (v << 1) | 1

    def neg(l):
        return l ^ 1

    def var(l):
        return l >> 1

    raw_clauses = []
    for _ in range(m):
        c = [int(next(it)), int(next(it)), int(next(it))]
        raw_clauses.append([lit_to_internal(x) for x in c])

    clauses = []
    watches = [[] for _ in range(2 * n + 2)]
    val = [-1] * (2 * n + 2)
    decision_level = [0] * (n + 1)
    reason = [None] * (n + 1)
    trail = []
    trail_lim = []
    activity = [0.0] * (n + 1)
    var_inc = 1.0

    def assign(l, r=None):
        v = var(l)
        val[l] = 1
        val[neg(l)] = 0
        decision_level[v] = len(trail_lim)
        reason[v] = r
        trail.append(l)

    def undo_one():
        l = trail.pop()
        v = var(l)
        val[l] = -1
        val[neg(l)] = -1
        reason[v] = None

    def propagate():
        q_head = propagate.head
        while q_head < len(trail):
            p = trail[q_head]
            q_head += 1
            false_lit = neg(p)
            i = 0
            ws = watches[false_lit]
            while i < len(ws):
                cr = ws[i]
                c = clauses[cr]

                if c[0] == false_lit:
                    c[0], c[1] = c[1], c[0]

                if val[c[0]] == 1:
                    i += 1
                    continue

                found = False
                for k in range(2, len(c)):
                    if val[c[k]] != 0:
                        c[1], c[k] = c[k], c[1]
                        watches[c[1]].append(cr)
                        ws[i] = ws[-1]
                        ws.pop()
                        found = True
                        break

                if not found:
                    if val[c[0]] == 0:
                        propagate.head = q_head
                        return cr
                    else:
                        assign(c[0], cr)
                    i += 1
        propagate.head = q_head
        return None

    propagate.head = 0

    seen = [False] * (n + 1)

    for c in raw_clauses:
        c = list(dict.fromkeys(c))
        has_tautology = False
        for l in c:
            if neg(l) in c:
                has_tautology = True
                break
        if has_tautology:
            continue

        if len(c) == 0:
            print("NO")
            return
        elif len(c) == 1:
            l = c[0]
            if val[l] == 0:
                print("NO")
                return
            if val[l] == -1:
                assign(l, None)
        else:
            cr = len(clauses)
            clauses.append(c)
            watches[c[0]].append(cr)
            watches[c[1]].append(cr)

    if propagate() is not None:
        print("NO")
        return

    while True:
        confl = propagate()
        if confl is not None:
            if len(trail_lim) == 0:
                print("NO")
                return

            curr_level = len(trail_lim)
            confl_clause = clauses[confl]
            learned = [0]
            counter = 0
            p = None
            trail_idx = len(trail) - 1

            for l in confl_clause:
                activity[var(l)] += var_inc
                v = var(l)
                if not seen[v]:
                    seen[v] = True
                    if decision_level[v] == curr_level:
                        counter += 1
                    elif decision_level[v] > 0:
                        learned.append(l)

            while counter > 0:
                while not seen[var(trail[trail_idx])]:
                    trail_idx -= 1
                p = trail[trail_idx]
                trail_idx -= 1
                v = var(p)
                seen[v] = False
                counter -= 1

                if counter == 0:
                    learned[0] = neg(p)
                    break

                r = reason[v]
                if r is not None:
                    for l in clauses[r]:
                        if var(l) != v:
                            activity[var(l)] += var_inc
                            v2 = var(l)
                            if not seen[v2]:
                                seen[v2] = True
                                if decision_level[v2] == curr_level:
                                    counter += 1
                                elif decision_level[v2] > 0:
                                    learned.append(l)

            for l in learned:
                seen[var(l)] = False

            var_inc *= 1.05

            if len(learned) == 1:
                bt_level = 0
            else:
                max_i = 1
                for i in range(2, len(learned)):
                    if decision_level[var(learned[i])] > decision_level[var(learned[max_i])]:
                        max_i = i
                learned[1], learned[max_i] = learned[max_i], learned[1]
                bt_level = decision_level[var(learned[1])]

            while len(trail_lim) > bt_level:
                target = trail_lim.pop()
                while len(trail) > target:
                    undo_one()
            propagate.head = len(trail)

            cr = len(clauses)
            clauses.append(learned)
            watches[learned[0]].append(cr)
            if len(learned) > 1:
                watches[learned[1]].append(cr)
                assign(learned[0], cr)
            else:
                assign(learned[0], None)
        else:
            best_v = None
            best_act = -1
            for v in range(1, n + 1):
                if val[v << 1] == -1:
                    if activity[v] > best_act:
                        best_act = activity[v]
                        best_v = v

            if best_v is None:
                print("YES")
                res = []
                for v in range(1, n + 1):
                    res.append(1 if val[v << 1] == 1 else 0)
                print(*(res))
                return

            trail_lim.append(len(trail))
            assign(best_v << 1, None)


if __name__ == "__main__":
    solve()
```
