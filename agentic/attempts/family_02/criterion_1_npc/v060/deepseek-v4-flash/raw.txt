We can model this as a 3-SAT problem where each clause has exactly three literals.  
Since \( n \le 90 \), we can use a backtracking algorithm with unit propagation and pure literal elimination, which is efficient enough for this size.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # Variable assignment: 1 = True, 0 = False, -1 = unassigned
    assignment = [-1] * (n + 1)
    
    # Convert literal to (var, is_true)
    def lit_to_var(lit):
        if lit > 0:
            return lit, True
        else:
            return -lit, False
    
    # Check if clause is satisfied
    def clause_satisfied(clause, assignment):
        for lit in clause:
            var, val = lit_to_var(lit)
            if assignment[var] != -1 and assignment[var] == (1 if val else 0):
                return True
        return False
    
    # Check if clause is unsatisfiable under current assignment
    def clause_conflict(clause, assignment):
        for lit in clause:
            var, val = lit_to_var(lit)
            if assignment[var] == -1 or assignment[var] == (1 if val else 0):
                return False
        return True
    
    # Unit propagation
    def propagate(assignment, clauses):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                if clause_satisfied(clause, assignment):
                    continue
                unassigned = []
                for lit in clause:
                    var, val = lit_to_var(lit)
                    if assignment[var] == -1:
                        unassigned.append((var, val))
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    var, val = unassigned[0]
                    assignment[var] = 1 if val else 0
                    changed = True
        return True
    
    # Pure literal elimination
    def pure_literal_elimination(assignment, clauses):
        # Count occurrences of each literal
        pos_count = [0] * (n + 1)
        neg_count = [0] * (n + 1)
        for clause in clauses:
            if clause_satisfied(clause, assignment):
                continue
            for lit in clause:
                var, val = lit_to_var(lit)
                if assignment[var] == -1:
                    if val:
                        pos_count[var] += 1
                    else:
                        neg_count[var] += 1
        changed = False
        for var in range(1, n + 1):
            if assignment[var] == -1:
                if pos_count[var] > 0 and neg_count[var] == 0:
                    assignment[var] = 1
                    changed = True
                elif neg_count[var] > 0 and pos_count[var] == 0:
                    assignment[var] = 0
                    changed = True
        return changed
    
    # Backtracking search
    def backtrack(assignment, clauses):
        # Propagate
        if not propagate(assignment, clauses):
            return False
        # Pure literal elimination
        while pure_literal_elimination(assignment, clauses):
            if not propagate(assignment, clauses):
                return False
        
        # Check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            if not clause_satisfied(clause, assignment):
                all_sat = False
                break
        if all_sat:
            return True
        
        # Choose an unassigned variable
        var = -1
        for i in range(1, n + 1):
            if assignment[i] == -1:
                var = i
                break
        if var == -1:
            return True  # no unassigned, but not all satisfied -> should not happen
        
        # Try both values
        for val in (0, 1):
            assignment[var] = val
            if backtrack(assignment, clauses):
                return True
            assignment[var] = -1
        return False
    
    if backtrack(assignment, clauses):
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
