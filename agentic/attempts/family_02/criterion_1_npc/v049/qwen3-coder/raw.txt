def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    rules = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        l3 = int(data[index+2])
        index += 3
        rules.append((l1, l2, l3))
    
    # For each light source, we have two variables:
    # x_i (light i is warm) and ~x_i (light i is cool)
    # We'll use 2*i for x_i and 2*i+1 for ~x_i
    # But actually, let's think of this as a 2-SAT problem.
    # Each rule is a clause with 3 literals: (l1 OR l2 OR l3)
    # We need to convert this to 2-SAT by breaking 3-clauses into 2-clauses
    
    # However, there's a simpler approach: try all 2^n possibilities for small n
    # But n can be up to 90, so that's not feasible.
    
    # Let's model this properly as a 2-SAT problem by converting 3-CNF to implications
    # For a clause (A OR B OR C), it's equivalent to:
    # (~A => B OR C) AND (~B => A OR C) AND (~C => A OR B)
    # And (X OR Y) is equivalent to (~X => Y) AND (~Y => X)
    
    # So for clause (A OR B OR C):
    # (~A => (B OR C)) which is (~A => (~(~B) OR C)) which is (~A => (~B => C))
    # This gives us: (~A AND ~B) => C, which is A OR B OR C
    
    # Actually, let's build an implication graph directly.
    # For clause (A OR B OR C), we add edges:
    # ~A => B OR C. To express B OR C in implication graph:
    # We introduce an auxiliary variable, but since we're solving 3-SAT,
    # it's easier to check satisfiability by trying assignments.
    
    # Given the constraints, let's implement a solution using brute force for small n,
    # but since n can be 90, that won't work.
    
    # Let's reframe: This is a 3-SAT problem where each clause has 3 literals.
    # But we need to determine if there's a satisfying assignment.
    
    # A better approach: Since this is a constraint satisfaction problem,
    # we can model it as finding a solution to a system where each rule must be satisfied.
    
    # Let's think of it differently. Each rule is a disjunction of 3 literals.
    # We want to find an assignment to variables (lights) such that all clauses are true.
    
    # We can solve this with a backtracking algorithm with pruning.
    
    # But given the time limits and constraints, let's try to implement a smart backtracking.
    
    # Actually, let's look at the examples:
    # Example 1: 
    # Rule 1: 1 2 3 (warm(1) OR warm(2) OR warm(3))
    # Rule 2: -1 2 -3 (cool(1) OR warm(2) OR cool(3))
    # Solution: light 1=cool(0), light 2=cool(0), light 3=warm(1)
    # Check: Rule 1: 0 or 0 or 1 = True. Rule 2: 1 or 0 or 0 = True. OK.
    
    # Example 2:
    # Rule 1: 1 1 1 (warm(1) OR warm(1) OR warm(1)) = warm(1)
    # Rule 2: -1 -1 -1 (cool(1) OR cool(1) OR cool(1)) = cool(1)
    # These contradict, so NO.
    
    # So we need a SAT solver for 3-CNF formulas.
    # Let's implement a simple DPLL-style algorithm.
    
    # Map literal to variable index and polarity
    def lit_to_var(lit):
        var = abs(lit) - 1  # 0-indexed
        polarity = 1 if lit > 0 else 0
        return var, polarity
    
    # Try all 2^n assignments - not feasible for n=90
    # Need a smarter approach
    
    # Let's implement a recursive backtracking with unit propagation
    
    def is_satisfied(assignment, rules):
        for l1, l2, l3 in rules:
            val1 = assignment[abs(l1)-1] if l1 > 0 else 1 - assignment[abs(l1)-1]
            val2 = assignment[abs(l2)-1] if l2 > 0 else 1 - assignment[abs(l2)-1]
            val3 = assignment[abs(l3)-1] if l3 > 0 else 1 - assignment[abs(l3)-1]
            if not (val1 or val2 or val3):
                return False
        return True
    
    # Simple backtracking without optimization will be too slow
    # Let's try a different approach: for small n, we can do better
    
    # Since n <= 90 and m <= 400, we might need a more sophisticated algorithm
    # But given that this is problem A, maybe there's a simpler way
    
    # Let's just implement a basic backtracking with early termination
    
    def backtrack(assignment, idx):
        if idx == n:
            return is_satisfied(assignment, rules)
            
        # Try setting light idx to cool (0)
        assignment[idx] = 0
        if backtrack(assignment, idx + 1):
            return True
            
        # Try setting light idx to warm (1)
        assignment[idx] = 1
        if backtrack(assignment, idx + 1):
            return True
            
        return False
    
    # This is still exponential. For n=90, it won't work.
    
    # Let's think of the problem structure:
    # Each rule is a 3-clause. We need to satisfy all clauses.
    # This is exactly 3-SAT. But we need to solve it efficiently.
    
    # Maybe the test cases are designed to be solvable with some heuristics?
    
    # Let's try implementing a local search or a greedy approach first
    # But that might not find the optimal solution.
    
    # For now, let's implement the backtracking for correctness and see:
    
    # Optimization: Unit propagation
    def propagate_unit_clauses(assignment, rules):
        changed = True
        while changed:
            changed = False
            for l1, l2, l3 in rules:
                # Count unset variables and satisfied literals
                unset_count = 0
                unset_lit = None
                satisfied = False
                
                lits = [l1, l2, l3]
                vals = []
                for lit in lits:
                    var_idx = abs(lit) - 1
                    if assignment[var_idx] == -1:  # Unassigned
                        unset_count += 1
                        unset_lit = lit
                        vals.append(None)
                    else:
                        val = assignment[var_idx] if lit > 0 else 1 - assignment[var_idx]
                        vals.append(val)
                        if val == 1:
                            satisfied = True
                            
                if satisfied:
                    continue
                    
                if unset_count == 1:
                    # Unit clause
                    var_idx = abs(unset_lit) - 1
                    required_val = 1 if unset_lit > 0 else 0
                    if assignment[var_idx] == -1:
                        assignment[var_idx] = required_val
                        changed = True
                    elif assignment[var_idx] != required_val:
                        return False  # Conflict
        return True
    
    def backtrack_opt(assignment, idx):
        # Propagate unit clauses
        if not propagate_unit_clauses(assignment, rules):
            return False
            
        # Find next unassigned variable
        while idx < n and assignment[idx] != -1:
            idx += 1
            
        if idx == n:
            # Check if all clauses are satisfied
            for l1, l2, l3 in rules:
                val1 = assignment[abs(l1)-1] if l1 > 0 else 1 - assignment[abs(l1)-1]
                val2 = assignment[abs(l2)-1] if l2 > 0 else 1 - assignment[abs(l2)-1]
                val3 = assignment[abs(l3)-1] if l3 > 0 else 1 - assignment[abs(l3)-1]
                if not (val1 or val2 or val3):
                    return False
            return True
            
        # Try both values
        for val in [0, 1]:
            assignment[idx] = val
            if backtrack_opt(assignment[:], idx + 1):
                return True
            assignment[idx] = -1
            
        return False
    
    # Start with all unassigned (-1)
    assignment = [-1] * n
    if backtrack_opt(assignment, 0):
        # Convert -1s to 0s (arbitrary choice)
        result = [0 if x == -1 else x for x in assignment]
        print("YES")
        print(" ".join(map(str, result)))
    else:
        print("NO")

solve()
