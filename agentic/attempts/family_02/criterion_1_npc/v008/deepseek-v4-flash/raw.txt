We can model each rule as a 3-clause in CNF, where a positive integer `k` means switch `k` is ON and a negative integer `-k` means switch `k` is OFF. The clause is satisfied if at least one of these literals is true. This is a standard **3-SAT** problem.  

Since n ≤ 90 and m ≤ 400, we can solve it efficiently with a backtracking algorithm and unit propagation, or use a 2-SAT-like reduction (but 3-SAT is NP-complete in general, but constraints are small enough for a DPLL-style solver).

I'll implement a DPLL SAT solver with:
- Unit propagation
- Pure literal elimination
- Recursive branching on a variable

Here’s the code:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # DPLL SAT solver
    def dpll(clauses, assignment):
        # Unit propagation
        while True:
            unit = None
            # Find unit clauses
            new_clauses = []
            changed = False
            for clause in clauses:
                vals = []
                unresolved = False
                for lit in clause:
                    var = abs(lit)
                    if var in assignment:
                        val = assignment[var]
                        # lit true?
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            vals.append(True)
                        else:
                            vals.append(False)
                    else:
                        unresolved = True
                        vals.append(None)
                if True in vals:
                    continue  # clause satisfied
                if all(v is False for v in vals):
                    return None  # conflict
                if unresolved and sum(1 for v in vals if v is None) == 1:
                    # unit clause
                    for idx, v in enumerate(vals):
                        if v is None:
                            lit = clause[idx]
                            var = abs(lit)
                            val = 1 if lit > 0 else 0
                            assignment[var] = val
                            changed = True
                            unit = True
                            break
                else:
                    new_clauses.append(clause)
            clauses = new_clauses
            if not changed:
                break
        # Check if all satisfied
        if not clauses:
            return assignment
        # Pure literal elimination
        # Count literals
        pos_count = {}
        neg_count = {}
        for clause in clauses:
            for lit in clause:
                var = abs(lit)
                if lit > 0:
                    pos_count[var] = pos_count.get(var, 0) + 1
                else:
                    neg_count[var] = neg_count.get(var, 0) + 1
        pure_vars = set()
        for var in set(pos_count) | set(neg_count):
            if var not in pos_count:
                assignment[var] = 0
                pure_vars.add(var)
            elif var not in neg_count:
                assignment[var] = 1
                pure_vars.add(var)
        if pure_vars:
            new_clauses = []
            for clause in clauses:
                sat = False
                new_lits = []
                for lit in clause:
                    var = abs(lit)
                    if var in assignment:
                        if (lit > 0 and assignment[var] == 1) or (lit < 0 and assignment[var] == 0):
                            sat = True
                            break
                        else:
                            continue
                    else:
                        new_lits.append(lit)
                if not sat:
                    if not new_lits:
                        return None
                    new_clauses.append(tuple(new_lits))
            clauses = new_clauses
            return dpll(clauses, assignment)
        
        # Choose a variable to branch on
        var = None
        for clause in clauses:
            for lit in clause:
                v = abs(lit)
                if v not in assignment:
                    var = v
                    break
            if var is not None:
                break
        # Try true first
        for val in [1, 0]:
            new_assignment = assignment.copy()
            new_assignment[var] = val
            # Remove satisfied clauses
            new_clauses = []
            for clause in clauses:
                sat = False
                new_lits = []
                for lit in clause:
                    if abs(lit) == var:
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            sat = True
                            break
                        else:
                            continue
                    else:
                        new_lits.append(lit)
                if not sat:
                    if not new_lits:
                        new_clauses = None
                        break
                    new_clauses.append(tuple(new_lits))
            if new_clauses is None:
                continue
            result = dpll(new_clauses, new_assignment)
            if result is not None:
                return result
        return None
    
    assignment = {}
    result = dpll(clauses, assignment)
    if result is None:
        print("NO")
    else:
        print("YES")
        # Fill unassigned variables with 0
        out = []
        for i in range(1, n+1):
            out.append(str(result.get(i, 0)))
        print(" ".join(out))

if __name__ == "__main__":
    solve()
```
