import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT: each switch i has two variables: i (ON) and i+n (OFF)
    # Variable i is true means switch i is ON
    # Variable i+n is true means switch i is OFF
    # For each switch, exactly one of i and i+n must be true
    # So we add clauses: (i OR i+n) and (NOT i OR NOT i+n)
    
    # Total variables: 2n (1-indexed for convenience)
    N = 2 * n
    adj = [[] for _ in range(N + 1)]
    radj = [[] for _ in range(N + 1)]
    
    def var_on(i):
        return i
    
    def var_off(i):
        return i + n
    
    def neg(v):
        if v <= n:
            return v + n
        else:
            return v - n
    
    def add_implication(u, v):
        adj[u].append(v)
        radj[v].append(u)
    
    def add_clause(u, v):
        # Add clause (u OR v) = (NOT u -> v) AND (NOT v -> u)
        add_implication(neg(u), v)
        add_implication(neg(v), u)
    
    # Each switch must be either ON or OFF
    for i in range(1, n + 1):
        add_clause(var_on(i), var_off(i))  # (i OR i+n)
        add_clause(neg(var_on(i)), neg(var_off(i)))  # (NOT i OR NOT i+n)
    
    # Read rules
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        
        # Convert to variables
        vars_in_clause = []
        for lit in [l1, l2, l3]:
            switch = abs(lit)
            if lit > 0:
                vars_in_clause.append(var_on(switch))
            else:
                vars_in_clause.append(var_off(switch))
        
        # Add clause (v1 OR v2 OR v3) = ((v1 OR v2) OR v3)
        v1, v2, v3 = vars_in_clause
        add_clause(v1, v2)
        add_clause(neg(v1), v3)
        add_clause(neg(v2), v3)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (N + 1)
    order = []
    
    def dfs1(v):
        stack = [(v, 0)]
        while stack:
            node, idx = stack[-1]
            if idx == 0:
                visited[node] = True
            if idx < len(adj[node]):
                neighbor = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(1, N + 1):
        if not visited[i]:
            dfs1(i)
    
    comp = [0] * (N + 1)
    comp_id = 0
    
    def dfs2(v, cid):
        stack = [v]
        comp[v] = cid
        while stack:
            node = stack.pop()
            for neighbor in radj[node]:
                if comp[neighbor] == 0:
                    comp[neighbor] = cid
                    stack.append(neighbor)
    
    for v in reversed(order):
        if comp[v] == 0:
            comp_id += 1
            dfs2(v, comp_id)
    
    # Check for contradiction
    for i in range(1, n + 1):
        if comp[var_on(i)] == comp[var_off(i)]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        # If var_on(i) has higher component number, it's assigned true
        if comp[var_on(i)] > comp[var_off(i)]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment[1:])))

solve()
