import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        lits = [int(next(it)), int(next(it)), int(next(it))]
        clauses.append(lits)

    # assignment: -1 unassigned, 0 false, 1 true
    assign = [-1] * (n + 1)

    # For backtracking we store a list of (var, previous_value)
    trail = []

    # Helper to get literal value under current assignment
    def lit_val(lit):
        v = abs(lit)
        a = assign[v]
        if a == -1:
            return None
        return a if lit > 0 else 1 - a

    # Propagation: unit clauses and conflict detection
    def propagate():
        while True:
            unit_var = None
            unit_val = None
            for clause in clauses:
                satisfied = False
                unassigned = []
                for lit in clause:
                    val = lit_val(lit)
                    if val == 1:
                        satisfied = True
                        break
                    elif val is None:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if not unassigned:          # all false -> conflict
                    return False
                if len(unassigned) == 1:   # unit clause
                    lit = unassigned[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else 0
                    cur = assign[var]
                    if cur == -1:
                        unit_var, unit_val = var, val
                        break
                    elif cur != val:
                        return False
            if unit_var is None:
                break
            # assign unit literal
            trail.append((unit_var, assign[unit_var]))
            assign[unit_var] = unit_val
        return True

    # Choose unassigned variable (simple heuristic)
    def choose_var():
        for v in range(1, n + 1):
            if assign[v] == -1:
                return v
        return None

    def backtrack():
        if not propagate():
            return False
        var = choose_var()
        if var is None:          # all assigned, solution found
            return True
        # try true then false
        for val in (1, 0):
            trail.append((var, assign[var]))
            assign[var] = val
            if backtrack():
                return True
            # undo to previous state
            while trail:
                v, prev = trail.pop()
                assign[v] = prev
                if v == var:
                    break
        return False

    if backtrack():
        print("YES")
        print(' '.join(str(assign[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
