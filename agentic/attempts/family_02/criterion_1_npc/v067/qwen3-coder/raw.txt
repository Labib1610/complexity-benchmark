from collections import defaultdict
import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each check, we have a clause with 3 literals
    # Literal is positive if we want ACTIVE, negative if we want INACTIVE
    clauses = []
    for _ in range(m):
        l1, l2, l3 = map(int, sys.stdin.readline().split())
        clauses.append((l1, l2, l3))
    
    # Try all 2^n possible assignments
    # But n can be up to 90, so 2^90 is too large
    # However, since each clause has size 3, this is a 3-SAT variant
    # But we're looking for a solution that satisfies ALL clauses, not MAX-SAT
    
    # Let's think differently. This is actually a 3-SAT problem where each clause
    # must be satisfied. In 3-SAT, each clause is a disjunction of 3 literals.
    # We want to find if there exists an assignment satisfying all clauses.
    
    # Since n <= 90 and m <= 400, trying all assignments is infeasible.
    # But maybe we can use the structure or small m to our advantage.
    
    # Actually, let's try a randomized approach or backtracking with pruning.
    # But given constraints, let's implement a simple backtracking with early termination.
    
    # Let's represent variables as 0 to n-1 internally
    # And literals: positive means variable should be True (ACTIVE)
    # Negative means variable should be False (INACTIVE)
    
    # Convert clauses to internal representation
    internal_clauses = []
    for l1, l2, l3 in clauses:
        clause = []
        for lit in [l1, l2, l3]:
            if lit > 0:
                clause.append((lit - 1, True))  # variable index, should be True
            else:
                clause.append((-lit - 1, False))  # variable index, should be False
        internal_clauses.append(clause)
    
    # Simple backtracking solver
    def is_satisfied(assignment, clause):
        for var_idx, should_be_true in clause:
            if var_idx >= len(assignment):
                return True  # Variable not assigned yet, could be satisfied
            if (assignment[var_idx] == 1) == should_be_true:
                return True
        return False
    
    def backtrack(assignment):
        # Check if current assignment violates any clause
        for clause in internal_clauses:
            if all(var_idx < len(assignment) for var_idx, _ in clause):
                if not is_satisfied(assignment, clause):
                    return None
        
        if len(assignment) == n:
            return assignment[:]
        
        # Try both values for the next variable
        for val in [0, 1]:
            assignment.append(val)
            result = backtrack(assignment)
            if result is not None:
                return result
            assignment.pop()
        
        return None
    
    solution = backtrack([])
    
    if solution is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, solution)))

solve()
