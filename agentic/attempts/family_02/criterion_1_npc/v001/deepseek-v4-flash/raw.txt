We can model each valve as a Boolean variable and each rule as a clause with three literals.  
This is exactly a **3-SAT** problem. Since \( n \leq 90 \) and \( m \leq 400 \), we can solve it with a backtracking algorithm that uses unit propagation and pure literal elimination.

Here’s the Python solution:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clause = []
        for lit in (l1, l2, l3):
            var = abs(lit) - 1
            val = 1 if lit > 0 else 0
            clause.append((var, val))
        clauses.append(clause)
    
    assignment = [-1] * n  # -1 unassigned, 0/1 assigned
    
    # Unit propagation and pure literal elimination
    def propagate():
        changed = True
        while changed:
            changed = False
            # Unit propagation
            for clause in clauses:
                unassigned = []
                satisfied = False
                for var, val in clause:
                    if assignment[var] == val:
                        satisfied = True
                        break
                    if assignment[var] == -1:
                        unassigned.append((var, val))
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False
                if len(unassigned) == 1:
                    var, val = unassigned[0]
                    if assignment[var] == -1:
                        assignment[var] = val
                        changed = True
                    elif assignment[var] != val:
                        return False
            # Pure literal elimination
            # Count occurrences
            pos_count = {}
            neg_count = {}
            for clause in clauses:
                satisfied = False
                for var, val in clause:
                    if assignment[var] == val:
                        satisfied = True
                        break
                if satisfied:
                    continue
                for var, val in clause:
                    if assignment[var] == -1:
                        if val == 1:
                            pos_count[var] = pos_count.get(var, 0) + 1
                        else:
                            neg_count[var] = neg_count.get(var, 0) + 1
            for var in pos_count:
                if var not in neg_count and assignment[var] == -1:
                    assignment[var] = 1
                    changed = True
            for var in neg_count:
                if var not in pos_count and assignment[var] == -1:
                    assignment[var] = 0
                    changed = True
        return True
    
    def backtrack():
        if not propagate():
            return False
        # Find unassigned variable
        var = -1
        for i in range(n):
            if assignment[i] == -1:
                var = i
                break
        if var == -1:
            # Check all clauses satisfied
            for clause in clauses:
                ok = False
                for v, val in clause:
                    if assignment[v] == val:
                        ok = True
                        break
                if not ok:
                    return False
            return True
        
        # Try both values
        for val in (0, 1):
            assignment[var] = val
            saved = assignment[:]
            if backtrack():
                return True
            assignment[:] = saved
        assignment[var] = -1
        return False
    
    if backtrack():
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
