def solve():

        return
    n = int(data[0])
    m = int(data[1])
    
    # Build 2-SAT: each rule is a clause (a OR b OR c)
    # Convert 3-SAT clauses to 2-SAT by introducing auxiliary variables
    # For each rule j, introduce aux variable x_j
    # Clause: (a OR b OR c) becomes:
    #   (a OR x_j) AND (b OR x_j) AND (c OR x_j) AND (NOT x_j OR NOT x_j)

    # Actually for 3-SAT to 2-SAT, we can't do it in general.
    # But this is a special case: each clause has exactly 3 literals.

    # But n <= 90, m <= 400, so we need an efficient algorithm.
    
    # Wait, let me re-read. Each rule has exactly 3 conditions, at least one must hold.
    # This is exactly 3-SAT. With n <= 90 and m <= 400, we need a good 3-SAT solver.
    
    # Let's use a DPLL-based approach or just brute force with pruning.
    # n=90 is too large for brute force (2^90).
    # But m=400 clauses with 3 literals each...
    
    # Actually, let's think about this differently. 
    # We can use a SAT solver approach with unit propagation and backtracking.
    
    # Let's implement a simple DPLL solver.
    
    # Variables are 1..n, value True means OPEN(1), False means CLOSED(0)
    # Literal: positive int v means variable v is True, negative means False
    
    clauses = []
    idx = 2
    for j in range(m):
        clause = []
        for k in range(3):
            val = int(data[idx])
            idx += 1
            clause.append(val)
        clauses.append(clause)
    
    # DPLL with unit propagation
    assignment = {}  # var -> bool
    
    def get_lit_val(lit, assignment):
        v = abs(lit)
        if v not in assignment:
            return None
        val = assignment[v]
        return val if lit > 0 else not val
    
    def unit_propagate(assignment, clauses):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    val = get_lit_val(lit, assignment)
                    if val is True:
                        satisfied = True
                        break
                    elif val is None:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    v = abs(lit)
                    assignment[v] = (lit > 0)
                    changed = True
        return True
    
    def dpll(assignment, clauses):
        # Unit propagation
        if not unit_propagate(assignment, clauses):
            return None
        
        # Check if all clauses satisfied
        unassigned_var = None
        all_satisfied = True
        for clause in clauses:
            satisfied = False
            unassigned = []
            for lit in clause:
                val = get_lit_val(lit, assignment)
                if val is True:
                    satisfied = True
                    break
                elif val is None:
                    unassigned.append(lit)
            if not satisfied:
                all_satisfied = False
                if len(unassigned) > 0:
                    # Pick first unassigned variable
                    v = abs(unassigned[0])
                    if v not in assignment:
                        unassigned_var = v
                        break
        
        if all_satisfied:
            return dict(assignment)
        
        if unassigned_var is None:
            return None
        
        # Try True first
        for val in [True, False]:
            new_assignment = dict(assignment)
            new_assignment[unassigned_var] = val
            result = dpll(new_assignment, clauses)
            if result is not None:
                return result
        
        return None
    
    result = dpll({}, clauses)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        states = []
        for i in range(1, n+1):
            if i in result:
                states.append(1 if result[i] else 0)
            else:
                states.append(0)
        print(' '.join(map(str, states)))

solve()
