```python
import sys
import random

def solve():
    sys.setrecursionlimit(2000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Simplify clause: remove duplicates, check if tautology
        c_set = set(c)
        tautology = False
        for lit in c_set:
            if -lit in c_set:
                tautology = True
                break
        if not tautology:
            clauses.append(list(c_set))
            
    # Quick check for empty clause
    for c in clauses:
        if len(c) == 0:
            print("NO")
            return

    # Try WalkSAT first for quick SAT finding
    def walksat(max_flips=100000, p=0.4):
        if not clauses:
            return [0] * (n + 1)
        
        # assignment: 1..n -> 0 or 1
        assign = [random.randint(0, 1) for _ in range(n + 1)]
        
        # clause truth count
        # For each clause, count how many literals are true
        def eval_lit(lit):
            if lit > 0:
                return assign[lit] == 1
            else:
                return assign[-lit] == 0

        # Precompute occurrences of literals in clauses
        lit_to_clauses = {}
        for c_idx, c in enumerate(clauses):
            for lit in c:
                lit_to_clauses.setdefault(lit, []).append(c_idx)

        sat_count = [0] * len(clauses)
        unsat_clauses = set()
        
        for c_idx, c in enumerate(clauses):
            cnt = sum(1 for lit in c if eval_lit(lit))
            sat_count[c_idx] = cnt
            if cnt == 0:
                unsat_clauses.add(c_idx)

        if not unsat_clauses:
            return assign

        for _ in range(max_flips):
            if not unsat_clauses:
                return assign
            
            c_idx = random.choice(list(unsat_clauses))
            c = clauses[c_idx]
            
            # Pick variable to flip
            if random.random() < p:
                var_to_flip = abs(random.choice(c))
            else:
                # Find variable that minimizes the number of broken clauses
                best_var = abs(c[0])
                min_break = float('inf')
                for lit in c:
                    var = abs(lit)
                    # Count how many clauses become unsatisfied if we flip var
                    # A clause is broken if sat_count is 1 and the true lit is this var's current lit
                    cur_lit = var if assign[var] == 1 else -var
                    broken = 0
                    if cur_lit in lit_to_clauses:
                        for cl in lit_to_clauses[cur_lit]:
                            if sat_count[cl] == 1:
                                broken += 1
                    if broken < min_break:
                        min_break = broken
                        best_var = var
                var_to_flip = best_var
            
            # Flip var_to_flip
            old_lit = var_to_flip if assign[var_to_flip] == 1 else -var_to_flip
            new_lit = -old_lit
            assign[var_to_flip] = 1 - assign[var_to_flip]
            
            # Update sat_count
            if old_lit in lit_to_clauses:
                for cl in lit_to_clauses[old_lit]:
                    sat_count[cl] -= 1
                    if sat_count[cl] == 0:
                        unsat_clauses.add(cl)
            if new_lit in lit_to_clauses:
                for cl in lit_to_clauses[new_lit]:
                    if sat_count[cl] == 0:
                        unsat_clauses.remove(cl)
                    sat_count[cl] += 1
                    
        return None

    # Run WalkSAT with a few restarts
    for _ in range(5):
        sol = walksat(max_flips=10000, p=0.4)
        if sol is not None:
            print("YES")
            print(*(sol[1:]))
            return

    # If WalkSAT didn't find a solution, run DPLL with 2-watched literals or simple DPLL
    # Given n <= 90, m <= 400, a standard DPLL with Jeroslow-Wang heuristic is very fast.
    
    assignment = {} # var -> True/False
    
    def get_jw_weights(cur_clauses):
        scores = {}
        for c in cur_clauses:
            weight = 2 ** (-len(c))
            for lit in c:
                scores[lit] = scores.get(lit, 0.0) + weight
        return scores

    def dpll(cur_clauses):
        # Unit propagation
        changed = True
        units_assigned = []
        
        while changed:
            changed = False
            # Find unit clauses
            unit_lit = None
            for c in cur_clauses:
                if len(c) == 0:
                    for lit in units_assigned:
                        del assignment[abs(lit)]
                    return False
                if len(c) == 1:
                    unit_lit = c[0]
                    break
            
            if unit_lit is not None:
                changed = True
                units_assigned.append(unit_lit)
                assignment[abs(unit_lit)] = (unit_lit > 0)
                new_clauses = []
                for c in cur_clauses:
                    if unit_lit in c:
                        continue
                    if -unit_lit in c:
                        new_clauses.append([l for l in c if l != -unit_lit])
                    else:
                        new_clauses.append(c)
                cur_clauses = new_clauses

        if not cur_clauses:
            return True

        for c in cur_clauses:
            if len(c) == 0:
                for lit in units_assigned:
                    del assignment[abs(lit)]
                return False

        # Pick branching variable using Jeroslow-Wang
        jw = get_jw_weights(cur_clauses)
        best_lit = max(jw.keys(), key=lambda l: jw[l] + jw.get(-l, 0.0))
        best_var = abs(best_lit)
        
        # Decide which polarity to try first
        if jw.get(best_var, 0) >= jw.get(-best_var, 0):
            first_lit = best_var
        else:
            first_lit = -best_var
            
        for branch_lit in [first_lit, -first_lit]:
            assignment[best_var] = (branch_lit > 0)
            next_clauses = []
            conflict = False
            for c in cur_clauses:
                if branch_lit in c:
                    continue
                if -branch_lit in c:
                    nc = [l for l in c if l != -branch_lit]
                    if len(nc) == 0:
                        conflict = True
                        break
                    next_clauses.append(nc)
                else:
                    next_clauses.append(c)
            
            if not conflict:
                if dpll(next_clauses):
                    return True
            del assignment[best_var]

        for lit in units_assigned:
            del assignment[abs(lit)]
        return False

    if dpll(clauses):
        print("YES")
        res = [1 if assignment.get(i, False) else 0 for i in range(1, n + 1)]
        print(*res)
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
