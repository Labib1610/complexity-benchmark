We can model each switch as a Boolean variable, and each interlock as a clause that must have at least one true literal.  
This is exactly a **3-SAT** problem. Since \( n \leq 90 \) and \( m \leq 400 \), we can solve it using a backtracking algorithm with unit propagation and pure literal elimination.

Here’s the Python solution:  

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        d1 = int(next(it))
        d2 = int(next(it))
        d3 = int(next(it))
        clauses.append((d1, d2, d3))
    
    # Variable assignment: 1 = engaged, 0 = isolated
    assignment = [None] * (n + 1)
    
    # Unit propagation and pure literal elimination
    def propagate(clauses, assignment):
        changed = True
        while changed:
            changed = False
            # Unit clauses
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    var = abs(lit)
                    val = assignment[var]
                    if val is not None:
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                    else:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    assignment[var] = 1 if lit > 0 else 0
                    changed = True
            # Pure literal elimination
            if not changed:
                pos_count = {}
                neg_count = {}
                for clause in clauses:
                    for lit in clause:
                        var = abs(lit)
                        if assignment[var] is not None:
                            continue
                        if lit > 0:
                            pos_count[var] = pos_count.get(var, 0) + 1
                        else:
                            neg_count[var] = neg_count.get(var, 0) + 1
                for var in set(pos_count) | set(neg_count):
                    if assignment[var] is not None:
                        continue
                    if var in pos_count and var not in neg_count:
                        assignment[var] = 1
                        changed = True
                    elif var in neg_count and var not in pos_count:
                        assignment[var] = 0
                        changed = True
        return True
    
    def simplify(clauses, assignment):
        new_clauses = []
        for clause in clauses:
            satisfied = False
            new_clause = []
            for lit in clause:
                var = abs(lit)
                val = assignment[var]
                if val is not None:
                    if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                        satisfied = True
                        break
                    # else literal false, skip
                else:
                    new_clause.append(lit)
            if satisfied:
                continue
            if not new_clause:
                return None
            new_clauses.append(tuple(new_clause))
        return new_clauses
    
    def backtrack(clauses, assignment):
        clauses = simplify(clauses, assignment)
        if clauses is None:
            return False
        if not clauses:
            return True
        if not propagate(clauses, assignment):
            return False
        clauses = simplify(clauses, assignment)
        if clauses is None:
            return False
        if not clauses:
            return True
        
        # Choose variable with most occurrences
        var_count = {}
        for clause in clauses:
            for lit in clause:
                var = abs(lit)
                if assignment[var] is None:
                    var_count[var] = var_count.get(var, 0) + 1
        if not var_count:
            return True
        var = max(var_count, key=var_count.get)
        
        for val in (1, 0):
            assignment[var] = val
            if backtrack(clauses, assignment):
                return True
            assignment[var] = None
        return False
    
    if backtrack(clauses, assignment):
        print("YES")
        print(" ".join(str(1 if assignment[i] == 1 else 0) for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
