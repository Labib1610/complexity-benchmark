import sys

# Increase recursion depth for deep DFS calls
sys.setrecursionlimit(200000)

def solve():
    g_data = sys.stdin.read().split()
    if not g_data:
        return
    iterator = iter(g_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return
    
    # Number of variables in the 2-SAT problem: 2 * n (for relay i: (i-1)*2 is primary, (i-1)*2+1 is secondary)
    num_nodes = 2 * n
    
    # Graph for finding Strongly Connected Components (SCS)
    adj = [[] for _ in range(num_nodes)]
    adj_transpose = [[] for _ in range(num_nodes)]
    
    # Helper to convert 1-based relay index to node index
    # Primary mode (positive input) maps to 2*(i-1)
    # Secondary mode (negative input) maps to 2*(i-1) + 1
    def get_node(relay_entry):
        idx = abs(relay_entry) - 1
        if relay_entry > 0:
            return 2 * idx      # Primary
        else:
            return 2 * idx + 1  # Secondary
    
    # Process each limit (clause with 3 literals)
    for _ in range(m):
        l1 = int(next(iterator))
        l2 = int(next(iterator))
        l3 = int(next(iterator))
        
        # Nodes representing the literals
        n1 = get_node(l1)
        n2 = get_node(l2)
        n3 = get_node(l3)
        
        # For a clause (A or B or C), the implication graph edges are:
        # !A -> B, !B -> A
        # !A -> C, !C -> A
        # !B -> C, !C -> B
        # And symmetry of nodes: for node v, v^1 is its negation
        adj[n1 ^ 1].append(n2)
        adj[n2 ^ 1].append(n1)
        adj_transpose[n2].append(n1 ^ 1)
        adj_transpose[n1].append(n2 ^ 1)
        
        adj[n1 ^ 1].append(n3)
        adj[n3 ^ 1].append(n1)
        adj_transpose[n3].append(n1 ^ 1)
        adj_transpose[n1].append(n3 ^ 1)
        
        adj[n2 ^ 1].append(n3)
        adj[n3 ^ 1].append(n2)
        adj_transpose[n3].append(n2 ^ 1)
        adj_transpose[n2].append(n3 ^ 1)
    
    # Kosaraju's algorithm to find SCCs
    visited = [False] * num_nodes
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(num_nodes):
        if not visited[i]:
            dfs1(i)
    
    comp_id = [-1] * num_nodes
    current_id = 0
    
    def dfs2(u, c_id):
        comp_id[u] = c_id
        for v in adj_transpose[u]:
            if comp_id[v] == -1:
                dfs2(v, c_id)
    
    for u in reversed(order):
        if comp_id[u] == -1:
            dfs2(u, current_id)
            current_id += 1
            
    # Check for unsatisfiability: if x and !x are in the same SCC, it's impossible
    for i in range(n):
        if comp_id[2 * i] == comp_id[2 * i + 1]:
            print("NO")
            return
    
    # Construct the solution using the topological order of SCCs
    # Components are assigned IDs in reverse topological order (since we processed reversed graph)
    # We want to assign 'True' to variables in components that appear later (higher ID)
    assignment = [0] * n
    
    for i in range(n):
        if comp_id[2 * i] > comp_id[2 * i + 1]:
            assignment[i] = 1  # Primary mode is selected
        else:
            assignment[i] = 0  # Secondary mode is selected
    
    print("YES")
    print(*(assignment))

if __name__ == '__main__':
    solve()
