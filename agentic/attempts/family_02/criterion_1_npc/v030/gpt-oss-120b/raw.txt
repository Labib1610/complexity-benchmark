```python
import sys

sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        c = [int(next(it)) for _ in range(3)]
        clauses.append(c)

    # Precompute variable occurrences
    var_pos = [set() for _ in range(n+1)]
    var_neg = [set() for _ in range(n+1)]
    for idx, cl in enumerate(clauses):
        for lit in cl:
            v = abs(lit)
            if lit > 0:
                var_pos[v].add(idx)
            else:
                var_neg[v].add(idx)

    # DPLL
    assignment = [None] * (n + 1)  # None, 0, 1

    # Helper to get literal value under current assignment
    def lit_val(lit):
        v = abs(lit)
        val = assignment[v]
        if val is None:
            return None
        return val if lit > 0 else 1 - val

    # Unit propagation and pure literal elimination
    def propagate():
        changed = True
        while changed:
            changed = False
            # unit clauses
            for cl in clauses:
                sat = False
                unassigned = []
                for lit in cl:
                    v = lit_val(lit)
                    if v == 1:
                        sat = True
                        break
                    elif v is None:
                        unassigned.append(lit)
                if sat:
                    continue
                if not unassigned:  # all false -> conflict
                    return False
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    v = abs(lit)
                    val = 1 if lit > 0 else 0
                    if assignment[v] is None:
                        assignment[v] = val
                        changed = True
            # pure literals
            for v in range(1, n+1):
                if assignment[v] is not None:
                    continue
                pos = any(lit_val(lit) is None for idx in var_pos[v] for lit in clauses[idx])
                neg = any(lit_val(lit) is None for idx in var_neg[v] for lit in clauses[idx])
                if pos and not neg:
                    assignment[v] = 1
                    changed = True
                elif neg and not pos:
                    assignment[v] = 0
                    changed = True
        return True

    # Check if all clauses satisfied
    def all_satisfied():
        for cl in clauses:
            if not any(lit_val(lit) == 1 for lit in cl):
                return False
        return True

    # Choose unassigned variable (simple heuristic)
    def choose_var():
        # variable with most occurrences
        best = None
        best_cnt = -1
        for v in range(1, n+1):
            if assignment[v] is None:
                cnt = len(var_pos[v]) + len(var_neg[v])
                if cnt > best_cnt:
                    best_cnt = cnt
                    best = v
        return best

    def dfs():
        if not propagate():
            return False
        if all(assignment[v] is not None for v in range(1, n+1)):
            return all_satisfied()
        v = choose_var()
        saved = assignment[:]
        for val in (1, 0):
            assignment[v] = val
            if dfs():
                return True
            assignment[:] = saved[:]
        return False

    if dfs():
        print("YES")
        print(' '.join(str(assignment[i]) for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
