import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Simplify clause: remove duplicate literals
        # If clause contains both x and -x, it is trivially satisfied
        c_set = set(c)
        tautology = False
        for lit in c_set:
            if -lit in c_set:
                tautology = True
                break
        if not tautology:
            clauses.append(list(c_set))
            
    # DPLL solver
    assignment = {} # var -> bool
    
    def get_lit_val(lit):
        var = abs(lit)
        if var not in assignment:
            return None
        val = assignment[var]
        return val if lit > 0 else (not val)

    def unit_propagate():
        while True:
            unit_found = False
            for c in clauses:
                unassigned = []
                satisfied = False
                for lit in c:
                    val = get_lit_val(lit)
                    if val is True:
                        satisfied = True
                        break
                    elif val is None:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False  # Conflict
                if len(unassigned) == 1:
                    unit_lit = unassigned[0]
                    var = abs(unit_lit)
                    val = (unit_lit > 0)
                    if var in assignment:
                        if assignment[var] != val:
                            return False
                    else:
                        assignment[var] = val
                        unit_found = True
            if not unit_found:
                break
        return True

    def dpll():
        # Save current assignment
        saved = dict(assignment)
        
        if not unit_propagate():
            assignment.clear()
            assignment.update(saved)
            return False
            
        # Check if all clauses are satisfied
        all_sat = True
        for c in clauses:
            sat = False
            for lit in c:
                if get_lit_val(lit) is True:
                    sat = True
                    break
            if not sat:
                all_sat = False
                break
                
        if all_sat:
            return True
            
        # Jeroslow-Wang heuristic for variable selection
        scores_pos = [0.0] * (n + 1)
        scores_neg = [0.0] * (n + 1)
        
        for c in clauses:
            sat = False
            unassigned = []
            for lit in c:
                val = get_lit_val(lit)
                if val is True:
                    sat = True
                    break
                elif val is None:
                    unassigned.append(lit)
            if not sat:
                weight = 2.0 ** (-len(unassigned))
                for lit in unassigned:
                    if lit > 0:
                        scores_pos[lit] += weight
                    else:
                        scores_neg[-lit] += weight
                        
        best_var = None
        best_score = -1.0
        for v in range(1, n + 1):
            if v not in assignment:
                score = scores_pos[v] + scores_neg[v]
                if score > best_score:
                    best_score = score
                    best_var = v
                    
        if best_var is None:
            return True
            
        first_val = scores_pos[best_var] >= scores_neg[best_var]
        
        # Branch 1
        assignment_before_branch = dict(assignment)
        assignment[best_var] = first_val
        if dpll():
            return True
            
        # Branch 2
        assignment.clear()
        assignment.update(assignment_before_branch)
        assignment[best_var] = not first_val
        if dpll():
            return True
            
        assignment.clear()
        assignment.update(saved)
        return False

    sys.setrecursionlimit(2000)
    if dpll():
        print("YES")
        res = []
        for i in range(1, n + 1):
            res.append(1 if assignment.get(i, False) else 0)
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
