We can model each component as a Boolean variable, and each clause as an OR of three literals.

This is a **3-SAT** problem, but since the number of variables $n \le 90$ and clauses $m \le 400$, we need an efficient approach deducing that we can map it to a graph reachability problem.

We can solve it using **implication graph** (like 2-SAT), but here clauses have 3 literals — that’s not directly reducible to 2-SAT.  

However, since $n$ is small (≤ 90), we can use **backtracking with unit propagation** and early pruning, which will be fast enough for $n=90$ and $m=400$ in practice because typical 3-SAT instances at that size are manageable.

We'll implement a simple **DPLL** algorithm with **pure literal** and **unit clause** heuristics.

```python
import sys
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        lits = [int(next(it)) for _ in range(3)]
        clauses.append(lits)
    
    assignment = [None] * (n + 1)  # None = unassigned, True=1, False=0

    def check_clause(clause, assign):
        # clause: list of literals, positive means var, negative means not var
        for lit in clause:
            var = abs(lit)
            val = assign[var]
            if val is not None:
                if lit > 0 and val:
                    return True
                if lit < 0 and not val:
                    return True
        return False

    def is_satisfied(assign):
        for clause in clauses:
            if not check_clause(clause, assign):
                return False
        return True

    def unit_propagate(assign):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    var = abs(lit)
                    val = assign[var]
                    if val is not None:
                        if (lit > 0 and val) or (lit < 0 and not val):
                            satisfied = True
                            break
                    else:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    if lit > 0:
                        assign[var] = True
                    else:
                        assign[var] = False
                    changed = True
        return True

    def pure_literal_assign(assign):
        # Find pure literals (appear only in one polarity)
        pos_count = {}
        neg_count = {}
        for clause in clauses:
            for lit in clause:
                var = abs(lit)
                if assign[var] is not None:
                    continue
                if lit > 0:
                    pos_count[var] = pos_count.get(var, 0) + 1
                else:
                    neg_count[var] = neg_count.get(var, 0) + 1
        changed = False
        for var in pos_count:
            if assign[var] is None and neg_count.get(var, 0) == 0:
                assign[var] = True
                changed = True
        for var in neg_count:
            if assign[var] is None and pos_count.get(var, 0) == 0:
                assign[var] = False
                changed = True
        return changed

    def backtrack(assign):
        if not unit_propagate(assign):
            return False
        # If all clauses satisfied
        if all(check_clause(clause, assign) for clause in clauses):
            return True
        # Pure literal assignment
        if pure_literal_assign(assign):
            return backtrack(assign)
        # Choose an unassigned variable
        for var in range(1, n+1):
            if assign[var] is None:
                break
        # Try True then False
        assign[var] = True
        if backtrack(assign):
            return True
        assign[var] = False
        if backtrack(assign):
            return True
        assign[var] = None
        return False

    if backtrack(assignment):
        print("YES")
        print(" ".join("1" if assignment[i] else "0" for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
