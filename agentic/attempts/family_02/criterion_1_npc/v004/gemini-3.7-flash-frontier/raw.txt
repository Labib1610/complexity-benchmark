import sys

sys.setrecursionlimit(5000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx + 1]), int(input_data[idx + 2])]
        idx += 3
        c_set = set(c)
        tautology = any(-lit in c_set for lit in c_set)
        if not tautology:
            clauses.append(list(c_set))

    def dpll(cls, assignment):
        while True:
            unit_clauses = [c for c in cls if len(c) == 1]
            if not unit_clauses:
                break
            unit_lit = unit_clauses[0][0]
            var = abs(unit_lit)
            val = unit_lit > 0
            if var in assignment:
                if assignment[var] != val:
                    return False, None
            assignment[var] = val

            new_clauses = []
            for c in cls:
                if unit_lit in c:
                    continue
                if -unit_lit in c:
                    new_c = [l for l in c if l != -unit_lit]
                    if not new_c:
                        return False, None
                    new_clauses.append(new_c)
                else:
                    new_clauses.append(c)
            cls = new_clauses

        if not cls:
            return True, assignment

        min_len = min(len(c) for c in cls)
        shortest = [c for c in cls if len(c) == min_len]
        freq = {}
        for c in shortest:
            for lit in c:
                freq[lit] = freq.get(lit, 0) + 1

        best_lit = max(freq.keys(), key=lambda l: (freq[l], freq.get(-l, 0)))

        for lit in [best_lit, -best_lit]:
            res, final_assign = dpll(cls + [[lit]], assignment.copy())
            if res:
                return True, final_assign

        return False, None

    res, final_assign = dpll(clauses, {})
    if res:
        print("YES")
        ans = [1 if final_assign.get(i, False) else 0 for i in range(1, n + 1)]
        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
