```python
import sys

# Increase recursion depth for DPLL search
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx + 1]), int(input_data[idx + 2])]
        idx += 3
        clauses.append(c)

    cleaned_clauses = []
    for c in clauses:
        s = set(c)
        if any(-x in s for x in s):
            continue
        cleaned_clauses.append(list(s))

    val = [0] * (n + 1)

    def dpll():
        assigned = []

        while True:
            unit_lit = None
            conflict = False

            for c in cleaned_clauses:
                c_val = False
                unassigned_lits = []
                for lit in c:
                    v = abs(lit)
                    sign = 1 if lit > 0 else -1
                    if val[v] == sign:
                        c_val = True
                        break
                    elif val[v] == 0:
                        unassigned_lits.append(lit)
                if c_val:
                    continue
                if len(unassigned_lits) == 0:
                    conflict = True
                    break
                elif len(unassigned_lits) == 1:
                    unit_lit = unassigned_lits[0]
                    break

            if conflict:
                for v in assigned:
                    val[v] = 0
                return False

            if unit_lit is not None:
                v = abs(unit_lit)
                sign = 1 if unit_lit > 0 else -1
                val[v] = sign
                assigned.append(v)
            else:
                break

        all_sat = True
        min_size = 999
        for c in cleaned_clauses:
            c_val = False
            unassigned_lits = []
            for lit in c:
                v = abs(lit)
                sign = 1 if lit > 0 else -1
                if val[v] == sign:
                    c_val = True
                    break
                elif val[v] == 0:
                    unassigned_lits.append(lit)
            if not c_val:
                all_sat = False
                if len(unassigned_lits) < min_size:
                    min_size = len(unassigned_lits)

        if all_sat:
            return True

        pos_count = [0] * (n + 1)
        neg_count = [0] * (n + 1)
        for c in cleaned_clauses:
            c_val = False
            unassigned_lits = []
            for lit in c:
                v = abs(lit)
                sign = 1 if lit > 0 else -1
                if val[v] == sign:
                    c_val = True
                    break
                elif val[v] == 0:
                    unassigned_lits.append(lit)
            if not c_val and len(unassigned_lits) == min_size:
                for lit in unassigned_lits:
                    v = abs(lit)
                    if lit > 0:
                        pos_count[v] += 1
                    else:
                        neg_count[v] += 1

        best_var = 0
        best_score = -1
        for v in range(1, n + 1):
            if val[v] == 0:
                p, q = pos_count[v], neg_count[v]
                score = (p + q) * 1024 + p * q
                if score > best_score:
                    best_score = score
                    best_var = v

        if best_var == 0:
            return True

        branch_order = (
            [1, -1] if pos_count[best_var] >= neg_count[best_var] else [-1, 1]
        )
        for branch_sign in branch_order:
            val[best_var] = branch_sign
            if dpll():
                return True
            val[best_var] = 0

        for v in assigned:
            val[v] = 0
        return False

    if dpll():
        ans = [1 if val[i] == 1 else 0 for i in range(1, n + 1)]
        print("YES")
        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
