We can model each rule as a clause in a 3-SAT problem, where each literal is a switch being ON or OFF.  
Since each variable appears at most 1200 times and the total literal count is bounded, we can use a SAT solver approach.  
We’ll implement a simple DPLL-style backtracking with unit propagation and pure literal elimination.

```python
import sys
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # Convert to 0-indexed variables: positive x -> var*2, negative x -> var*2+1
    # We'll use assignment: 0 = False, 1 = True
    # Literal representation: (var, is_true) where is_true means the literal is satisfied if var == is_true
    # For positive literal x: satisfied if var == True (1)
    # For negative literal -x: satisfied if var == False (0)
    
    # We'll store clauses as list of (var, val) pairs
    cnf = []
    for clause in clauses:
        c = []
        for lit in clause:
            var = abs(lit) - 1
            val = 1 if lit > 0 else 0
            c.append((var, val))
        cnf.append(c)
    
    assignment = [-1] * n  # -1 unassigned, 0 False, 1 True
    
    # Count occurrences for pure literal heuristic
    from collections import defaultdict
    pos_count = defaultdict(int)
    neg_count = defaultdict(int)
    for clause in cnf:
        for var, val in clause:
            if val == 1:
                pos_count[var] += 1
            else:
                neg_count[var] += 1
    
    def propagate(assignment, cnf):
        changed = True
        while changed:
            changed = False
            # Unit propagation
            for clause in cnf:
                unassigned = []
                satisfied = False
                for var, val in clause:
                    if assignment[var] == -1:
                        unassigned.append((var, val))
                    elif assignment[var] == val:
                        satisfied = True
                        break
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False  # conflict
                if len(unassigned) == 1:
                    var, val = unassigned[0]
                    if assignment[var] == -1:
                        assignment[var] = val
                        changed = True
                    # if already assigned opposite, conflict would have been caught
        return True
    
    def pure_literal_assign(assignment, cnf):
        # Find pure literals among unassigned vars
        pos_seen = set()
        neg_seen = set()
        for clause in cnf:
            sat = False
            for var, val in clause:
                if assignment[var] != -1:
                    if assignment[var] == val:
                        sat = True
                        break
                else:
                    if val == 1:
                        pos_seen.add(var)
                    else:
                        neg_seen.add(var)
            # If clause already satisfied, we don't need to consider it for pure literal
        changed = False
        for var in range(n):
            if assignment[var] == -1:
                if var in pos_seen and var not in neg_seen:
                    assignment[var] = 1
                    changed = True
                elif var in neg_seen and var not in pos_seen:
                    assignment[var] = 0
                    changed = True
        return changed
    
    def simplify(cnf, assignment):
        new_cnf = []
        for clause in cnf:
            sat = False
            new_clause = []
            for var, val in clause:
                if assignment[var] == -1:
                    new_clause.append((var, val))
                elif assignment[var] == val:
                    sat = True
                    break
            if sat:
                continue
            if not new_clause:
                return None  # conflict
            new_cnf.append(new_clause)
        return new_cnf
    
    def dpll(assignment, cnf):
        # Propagate units
        if not propagate(assignment, cnf):
            return False
        # Pure literal
        while pure_literal_assign(assignment, cnf):
            if not propagate(assignment, cnf):
                return False
        # Simplify
        cnf = simplify(cnf, assignment)
        if cnf is None:
            return False
        if not cnf:
            return True
        # Choose unassigned variable with most occurrences
        var_count = defaultdict(int)
        for clause in cnf:
            for var, val in clause:
                if assignment[var] == -1:
                    var_count[var] += 1
        if not var_count:
            return True
        var = max(var_count, key=var_count.get)
        # Try True
        assignment[var] = 1
        if dpll(assignment, [list(c) for c in cnf]):
            return True
        assignment[var] = 0
        if dpll(assignment, [list(c) for c in cnf]):
            return True
        assignment[var] = -1
        return False
    
    if dpll(assignment, cnf):
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
