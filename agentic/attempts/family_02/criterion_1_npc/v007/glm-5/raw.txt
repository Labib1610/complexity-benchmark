import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    rules = []
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        rules.append((a, b, c))
    
    # 2-SAT: Variable x_i is 0 if node 2i, 1 if node 2i+1
    # Indices: 0..2n-1
    # Implies: (u -> v) means if u is true then v must be true.
    # Clause (a v b v c): add edges (~a -> b), (~a -> c), (~b -> a), (~b -> c), (~c -> a), (~c -> b)
    
    N = 2 * n
    adj = [[] for _ in range(N)]
    radj = [[] for _ in range(N)]
    
    def get_node(lit):
        # lit is non-zero integer. |lit| is 1-based switch index.
        idx = abs(lit) - 1
        if lit > 0: # Switch ON (state 1)
            return 2 * idx + 1
        else: # Switch OFF (state 0)
            return 2 * idx
    
    def add_imp(u, v):
        adj[u].append(v)
        radj[v].append(u)
        
    for r in rules:
        a, b, c = r
        u_a = get_node(a)
        u_na = u_a ^ 1
        u_b = get_node(b)
        u_nb = u_b ^ 1
        u_c = get_node(c)
        u_nc = u_c ^ 1
        
        add_imp(u_na, u_b)
        add_imp(u_na, u_c)
        add_imp(u_nb, u_a)
        add_imp(u_nb, u_c)
        add_imp(u_nc, u_a)
        add_imp(u_nc, u_b)
        
    # Kosaraju's algorithm for SCCs
    visited = [False] * N
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
        
    for i in range(N):
        if not visited[i]:
            dfs1(i)
            
    comp_id = [-1] * N
    cid = 0
    # Process in reverse order of finishing times
    for u in reversed(order):
        if comp_id[u] == -1:
            stack = [u]
            comp_id[u] = cid
            while stack:
                curr = stack.pop()
                for v in radj[curr]:
                    if comp_id[v] == -1:
                        comp_id[v] = cid
                        stack.append(v)
            cid += 1
            
    # Check validity
    for i in range(n):
        if comp_id[2*i] == comp_id[2*i+1]:
            print("NO")
            return
            
    # Topological order of components is(cid) descending
    # To assign True to the literal with component processed later (higher cid),
    # we need to check which component is assigned later.
    assignment = [0] * n
    for i in range(n):
        if comp_id[2*i] > comp_id[2*i+1]:
            assignment[i] = 0 # OFF literal is true
        else:
            assignment[i] = 1 # ON literal is true
            
    print("YES")
    print(*(assignment))

if __name__ == '__main__':
    solve()
