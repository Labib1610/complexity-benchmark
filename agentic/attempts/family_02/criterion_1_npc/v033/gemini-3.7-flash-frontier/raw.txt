import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Simplify clause: remove duplicates, check if tautology
        c_set = set(c)
        tautology = False
        for lit in c:
            if -lit in c_set:
                tautology = True
                break
        if not tautology:
            clauses.append(list(c_set))
    
    # DPLL / CDCL Implementation
    # Variables: 1 to n
    assignment = [None] * (n + 1)
    
    # Watch list: for each literal, list of clauses where this literal is watched
    # Literals can be from -n to n (excluding 0)
    watches = {lit: [] for lit in range(-n, n + 1) if lit != 0}
    
    cleaned_clauses = []
    for c in clauses:
        if len(c) == 0:
            print("NO")
            return
        cleaned_clauses.append(c)
    
    clauses = cleaned_clauses
    
    # Check for initial unit clauses
    for i, c in enumerate(clauses):
        if len(c) >= 2:
            watches[c[0]].append(i)
            watches[c[1]].append(i)
        elif len(c) == 1:
            watches[c[0]].append(i)

    # Activity for VSIDS
    activity = {lit: 0.0 for lit in range(-n, n + 1) if lit != 0}
    for c in clauses:
        for lit in c:
            activity[lit] += 1.0

    # Trail for assignments: list of (lit, reason_clause_idx)
    trail = []
    trail_lim = [] # decision levels
    
    def value(lit):
        var = abs(lit)
        val = assignment[var]
        if val is None:
            return None
        return val if lit > 0 else 1 - val

    def assign(lit, reason):
        var = abs(lit)
        assignment[var] = 1 if lit > 0 else 0
        trail.append((lit, reason))

    def unassign():
        lit, _ = trail.pop()
        assignment[abs(lit)] = None

    def propagate():
        i = propagate.qhead
        while i < len(trail):
            lit, _ = trail[i]
            i += 1
            false_lit = -lit
            watch_list = watches[false_lit]
            watches[false_lit] = []
            
            j = 0
            while j < len(watch_list):
                ci = watch_list[j]
                c = clauses[ci]
                
                # Make sure false_lit is at c[1]
                if c[0] == false_lit:
                    c[0], c[1] = c[1], c[0]
                
                # If 0th watch is true, clause is satisfied
                if value(c[0]) == 1:
                    watches[false_lit].append(ci)
                    j += 1
                    continue
                
                # Look for a new watch
                found = False
                for k in range(2, len(c)):
                    if value(c[k]) != 0:
                        c[1], c[k] = c[k], c[1]
                        watches[c[1]].append(ci)
                        found = True
                        break
                
                if found:
                    j += 1
                    continue
                
                watches[false_lit].append(ci)
                j += 1
                
                # Clause is unit or conflict
                if value(c[0]) == 0:
                    # Conflict!
                    watches[false_lit].extend(watch_list[j:])
                    propagate.qhead = i - 1
                    return ci
                else:
                    if value(c[0]) is None:
                        assign(c[0], ci)
                        
        propagate.qhead = i
        return None

    propagate.qhead = 0

    # Initial propagation
    for i, c in enumerate(clauses):
        if len(c) == 1:
            if value(c[0]) == 0:
                print("NO")
                return
            elif value(c[0]) is None:
                assign(c[0], i)
    
    confl = propagate()
    if confl is not None:
        print("NO")
        return

    # VSIDS decay factor
    decay = 0.95

    while True:
        confl = propagate()
        if confl is not None:
            if len(trail_lim) == 0:
                print("NO")
                return
            
            # Simple backtracking with learned clause (CDCL 1UIP or simple chronological DPLL + activity)
            # Let's do 1UIP conflict analysis
            confl_clause = clauses[confl]
            seen = [False] * (n + 1)
            learnt = []
            path_cnt = 0
            p = None
            
            # Level of literals
            level = {}
            cur_lvl = len(trail_lim)
            
            # Calculate levels for trail
            lvl = 0
            idx_in_trail = 0
            for decision_point in trail_lim:
                while idx_in_trail < decision_point:
                    level[abs(trail[idx_in_trail][0])] = lvl
                    idx_in_trail += 1
                lvl += 1
            while idx_in_trail < len(trail):
                level[abs(trail[idx_in_trail][0])] = lvl
                idx_in_trail += 1
                
            idx_trail = len(trail) - 1
            c = confl_clause
            
            while True:
                for lit in c:
                    var = abs(lit)
                    if not seen[var] and level.get(var, 0) > 0:
                        seen[var] = True
                        activity[lit] += 1.0
                        activity[-lit] += 1.0
                        if level.get(var, 0) >= cur_lvl:
                            path_cnt += 1
                        else:
                            learnt.append(lit)
                            
                while idx_trail >= 0 and not seen[abs(trail[idx_trail][0])]:
                    idx_trail -= 1
                    
                if idx_trail < 0:
                    break
                    
                p, reason_idx = trail[idx_trail]
                idx_trail -= 1
                seen[abs(p)] = False
                path_cnt -= 1
                
                if path_cnt <= 0:
                    break
                c = clauses[reason_idx]
                
            learnt.insert(0, -p)
            
            # Bump decay
            for k in activity:
                activity[k] *= decay
                
            # Backtrack level: second highest level in learnt
            backtrack_lvl = 0
            if len(learnt) > 1:
                max_i = 1
                for i_l in range(1, len(learnt)):
                    if level[abs(learnt[i_l])] > level[abs(learnt[max_i])]:
                        max_i = i_l
                learnt[1], learnt[max_i] = learnt[max_i], learnt[1]
                backtrack_lvl = level[abs(learnt[1])]
                
            # Backjump
            while len(trail_lim) > backtrack_lvl:
                target_len = trail_lim.pop()
                while len(trail) > target_len:
                    unassign()
            propagate.qhead = len(trail)
            
            new_ci = len(clauses)
            clauses.append(learnt)
            if len(learnt) >= 2:
                watches[learnt[0]].append(new_ci)
                watches[learnt[1]].append(new_ci)
            elif len(learnt) == 1:
                watches[learnt[0]].append(new_ci)
            assign(learnt[0], new_ci)
            
        else:
            # Pick branching variable
            best_var = None
            best_act = -1
            for var in range(1, n + 1):
                if assignment[var] is None:
                    act = activity[var] + activity[-var]
                    if act > best_act:
                        best_act = act
                        best_var = var
                        
            if best_var is None:
                # All variables assigned
                print("YES")
                res = [assignment[i] if assignment[i] is not None else 0 for i in range(1, n + 1)]
                print(*(res))
                return
                
            # Decision
            trail_lim.append(len(trail))
            chosen_lit = best_var if activity[best_var] >= activity[-best_var] else -best_var
            assign(chosen_lit, None)

if __name__ == '__main__':
    solve()
