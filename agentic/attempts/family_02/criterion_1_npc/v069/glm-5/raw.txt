import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    groups = []
    for _ in range(m):
        d1 = int(next(it))
        d2 = int(next(it))
        d3 = int(next(it))
        groups.append((d1, d2, d3))
    
    # Try all 2^n possible assignments
    for mask in range(1 << n):
        ok = True
        for d1, d2, d3 in groups:
            # Check if at least one demand is satisfied
            satisfied = False
            
            # Demand 1
            if d1 > 0:
                if mask & (1 << (d1 - 1)):
                    satisfied = True
            else:
                if not (mask & (1 << ((-d1) - 1))):
                    satisfied = True
            
            if satisfied:
                continue
            
            # Demand 2
            if d2 > 0:
                if mask & (1 << (d2 - 1)):
                    satisfied = True
            else:
                if not (mask & (1 << ((-d2) - 1))):
                    satisfied = True
            
            if satisfied:
                continue
            
            # Demand 3
            if d3 > 0:
                if mask & (1 << (d3 - 1)):
                    satisfied = True
            else:
                if not (mask & (1 << ((-d3) - 1))):
                    satisfied = True
            
            if not satisfied:
                ok = False
                break
        
        if ok:
            print("YES")
            result = []
            for i in range(n):
                if mask & (1 << i):
                    result.append('1')
                else:
                    result.append('0')
            print(' '.join(result))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
