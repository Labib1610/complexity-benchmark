We can model each proposal as a Boolean variable and each interest group as a clause of three literals.  
This is exactly the **3-SAT** problem. Since \( n \leq 90 \) and \( m \leq 400 \), we can solve it using a backtracking algorithm with unit propagation and pure literal elimination, which is efficient enough for this size.

Here’s the Python solution:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    clauses = []
    for _ in range(m):
        d1 = int(next(it))
        d2 = int(next(it))
        d3 = int(next(it))
        clause = []
        for d in (d1, d2, d3):
            var = abs(d) - 1
            val = 1 if d > 0 else 0
            clause.append((var, val))
        clauses.append(clause)
    
    assignment = [-1] * n  # -1 unassigned, 0 false, 1 true
    
    def propagate(assign):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for var, val in clause:
                    if assign[var] == -1:
                        unassigned.append((var, val))
                    elif assign[var] == val:
                        satisfied = True
                        break
                if satisfied:
                    continue
                if not unassigned:
                    return False
                if len(unassigned) == 1:
                    var, val = unassigned[0]
                    if assign[var] == -1:
                        assign[var] = val
                        changed = True
                    elif assign[var] != val:
                        return False
        return True
    
    def pure_literal(assign):
        # Find pure literals
        pos_count = {}
        neg_count = {}
        for clause in clauses:
            for var, val in clause:
                if assign[var] != -1:
                    continue
                if val == 1:
                    pos_count[var] = pos_count.get(var, 0) + 1
                else:
                    neg_count[var] = neg_count.get(var, 0) + 1
        changed = False
        for var in range(n):
            if assign[var] != -1:
                continue
            p = pos_count.get(var, 0)
            q = neg_count.get(var, 0)
            if p > 0 and q == 0:
                assign[var] = 1
                changed = True
            elif q > 0 and p == 0:
                assign[var] = 0
                changed = True
        return changed
    
    def backtrack(assign):
        # Propagate
        if not propagate(assign):
            return False
        # Pure literal
        while pure_literal(assign):
            if not propagate(assign):
                return False
        
        # Check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            sat = False
            for var, val in clause:
                if assign[var] == val:
                    sat = True
                    break
            if not sat:
                all_sat = False
                break
        if all_sat:
            return True
        
        # Choose unassigned variable
        var = -1
        for i in range(n):
            if assign[i] == -1:
                var = i
                break
        if var == -1:
            return True
        
        # Try both values
        for val in (0, 1):
            assign[var] = val
            if backtrack(assign):
                return True
            assign[var] = -1
        return False
    
    if backtrack(assignment):
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
