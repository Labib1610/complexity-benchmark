```python
import sys
import random
import time

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Simplify clause: remove duplicate literals
        c = list(set(c))
        # If clause contains both x and -x, it's always satisfied (tautology)
        has_tautology = False
        lit_set = set(c)
        for lit in lit_set:
            if -lit in lit_set:
                has_tautology = True
                break
        if not has_tautology:
            clauses.append(c)
            
    # Try WalkSAT first
    start_time = time.time()
    
    def walksat(max_flips=2000, p=0.4):
        if not clauses:
            return [1] * (n + 1)
        
        assignment = [random.choice([False, True]) for _ in range(n + 1)]
        
        # Track satisfied count per clause
        sat_count = []
        clause_is_sat = []
        for c in clauses:
            sc = sum(1 for lit in c if (assignment[abs(lit)] if lit > 0 else not assignment[abs(lit)]))
            sat_count.append(sc)
            clause_is_sat.append(sc > 0)
            
        unsat_clauses = [i for i, is_sat in enumerate(clause_is_sat) if not is_sat]
        unsat_pos = {ci: i for i, ci in enumerate(unsat_clauses)}
        
        # Occurrence list: for each var, list of clauses it appears in positively and negatively
        pos_occ = [[] for _ in range(n + 1)]
        neg_occ = [[] for _ in range(n + 1)]
        for ci, c in enumerate(clauses):
            for lit in c:
                if lit > 0:
                    pos_occ[lit].append(ci)
                else:
                    neg_occ[-lit].append(ci)
                    
        for _ in range(max_flips):
            if not unsat_clauses:
                return assignment
            
            # Pick a random unsatisfied clause
            ci = random.choice(unsat_clauses)
            c = clauses[ci]
            
            # Find the best variable to flip
            best_var = None
            if random.random() < p:
                best_var = abs(random.choice(c))
            else:
                best_break = float('inf')
                best_vars = []
                for lit in c:
                    var = abs(lit)
                    # Count break: clauses that currently have sat_count == 1 and are satisfied by var's current value
                    break_count = 0
                    current_true_occ = pos_occ[var] if assignment[var] else neg_occ[var]
                    for occ_ci in current_true_occ:
                        if sat_count[occ_ci] == 1:
                            break_count += 1
                    
                    if break_count < best_break:
                        best_break = break_count
                        best_vars = [var]
                    elif break_count == best_break:
                        best_vars.append(var)
                best_var = random.choice(best_vars)
                
            # Flip best_var
            curr_val = assignment[best_var]
            assignment[best_var] = not curr_val
            
            # Update clauses
            now_false_occ = pos_occ[best_var] if curr_val else neg_occ[best_var]
            now_true_occ = neg_occ[best_var] if curr_val else pos_occ[best_var]
            
            for occ_ci in now_false_occ:
                sat_count[occ_ci] -= 1
                if sat_count[occ_ci] == 0:
                    unsat_pos[occ_ci] = len(unsat_clauses)
                    unsat_clauses.append(occ_ci)
                    
            for occ_ci in now_true_occ:
                sat_count[occ_ci] += 1
                if sat_count[occ_ci] == 1:
                    pos = unsat_pos[occ_ci]
                    last_ci = unsat_clauses[-1]
                    unsat_clauses[pos] = last_ci
                    unsat_pos[last_ci] = pos
                    unsat_clauses.pop()
                    del unsat_pos[occ_ci]
                    
        return None

    # Run WalkSAT for up to 0.4 seconds
    while time.time() - start_time < 0.4:
        sol = walksat(max_flips=1500, p=0.4)
        if sol is not None:
            print("YES")
            print(*(1 if sol[i] else 0 for i in range(1, n + 1)))
            return

    # Fallback to DPLL
    def dpll_solve():
        # assignment: None, True, False
        assign = [None] * (n + 1)
        
        def simplify(cls, assign):
            new_cls = []
            for c in cls:
                satisfied = False
                new_c = []
                for lit in c:
                    v = abs(lit)
                    val = assign[v]
                    if val is not None:
                        if (lit > 0 and val) or (lit < 0 and not val):
                            satisfied = True
                            break
                    else:
                        new_c.append(lit)
                if not satisfied:
                    if not new_c:
                        return None # conflict
                    new_cls.append(new_c)
            return new_cls

        def unit_propagate(cls, assign):
            while True:
                unit = None
                for c in cls:
                    if len(c) == 1:
                        unit = c[0]
                        break
                if unit is None:
                    break
                var = abs(unit)
                val = (unit > 0)
                if assign[var] is not None:
                    if assign[var] != val:
                        return None, None
                else:
                    assign[var] = val
                cls = simplify(cls, assign)
                if cls is None:
                    return None, None
            return cls, assign

        def dpll(cls, assign):
            cls, assign = unit_propagate(cls, assign)
            if cls is None:
                return None
            if not cls:
                # All satisfied
                for i in range(1, n + 1):
                    if assign[i] is None:
                        assign[i] = True
                return assign
            
            # Choose unassigned variable (MOM or frequency heuristic)
            counts = {}
            for c in cls:
                weight = 2 ** (-len(c))
                for lit in c:
                    counts[abs(lit)] = counts.get(abs(lit), 0) + weight
            
            var = max(counts, key=counts.get)
            
            # Try True then False
            for val in [True, False]:
                new_assign = list(assign)
                new_assign[var] = val
                new_cls = simplify(cls, new_assign)
                if new_cls is not None:
                    res = dpll(new_cls, new_assign)
                    if res is not None:
                        return res
            return None

        init_assign = [None] * (n + 1)
        init_cls = simplify(clauses, init_assign)
        if init_cls is None:
            return None
        return dpll(init_cls, init_assign)

    sol = dpll_solve()
    if sol is not None:
        print("YES")
        print(*(1 if sol[i] else 0 for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
