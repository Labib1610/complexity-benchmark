from collections import defaultdict
import sys

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each clause (test suite), we have three literals
    # Literal is positive if flag should be ENABLED, negative if DISABLED
    clauses = []
    for _ in range(m):
        clause = list(map(int, input().split()))
        clauses.append(clause)
    
    # We'll try all 2^n possible assignments
    # But n can be up to 90, so 2^90 is too large
    # However, since each clause has exactly 3 literals, this is a 3-SAT variant
    # Let's use a more intelligent approach
    
    # Try to solve using brute force for small n, but since n <= 90, we need something better
    # Actually, let's implement a simple backtracking solution
    
    def is_satisfied(assignment):
        # assignment is a list where assignment[i] = 0 (DISABLED) or 1 (ENABLED)
        for clause in clauses:
            satisfied = False
            for literal in clause:
                flag_index = abs(literal) - 1  # Convert to 0-based indexing
                required_value = 1 if literal > 0 else 0
                if assignment[flag_index] == required_value:
                    satisfied = True
                    break
            if not satisfied:
                return False
        return True
    
    # Since n can be up to 90, trying all combinations is infeasible
    # We need a SAT solver approach. Let's implement a simple one.
    
    # Let's try a different approach: since this is 3-SAT, and constraints are small,
    # we can use a randomized or heuristic approach, but that might not be reliable.
    
    # For correctness, let's implement a systematic backtracking with pruning
    
    def backtrack(assignment, index):
        if index == n:
            return is_satisfied(assignment)
        
        # Try setting flag index to 0 (DISABLED)
        assignment[index] = 0
        if backtrack(assignment, index + 1):
            return True
            
        # Try setting flag index to 1 (ENABLED)
        assignment[index] = 1
        if backtrack(assignment, index + 1):
            return True
            
        return False
    
    # This will be too slow for large n. Let's check if there's a smarter way.
    # Since the time limit is 2 seconds and n <= 90, complete search won't work.
    
    # Let's re-attempt with a more optimized backtracking that checks partial assignments
    
    def is_currently_satisfiable(assignment, decided_count):
        # Check if any clause is already unsatisfiable with current partial assignment
        for clause in clauses:
            unsat_count = 0
            unknown_count = 0
            for literal in clause:
                flag_index = abs(literal) - 1
                if flag_index < decided_count:
                    required_value = 1 if literal > 0 else 0
                    if assignment[flag_index] != required_value:
                        unsat_count += 1
                else:
                    unknown_count += 1
            
            # If all literals in clause are decided and none are satisfied
            if unknown_count == 0 and unsat_count == 3:
                return False
        return True
    
    def backtrack_optimized(assignment, index):
        if not is_currently_satisfiable(assignment, index):
            return False
            
        if index == n:
            return is_satisfied(assignment)
        
        # Try setting flag index to 0 (DISABLED)
        assignment[index] = 0
        if backtrack_optimized(assignment, index + 1):
            return True
            
        # Try setting flag index to 1 (ENABLED)
        assignment[index] = 1
        if backtrack_optimized(assignment, index + 1):
            return True
            
        return False
    
    # Even with optimization, for n=90 this might be too slow
    # But given the constraints of the problem and that it's likely designed to be solvable,
    # let's proceed with the optimized backtracking
    
    assignment = [0] * n
    if backtrack_optimized(assignment, 0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

solve()
