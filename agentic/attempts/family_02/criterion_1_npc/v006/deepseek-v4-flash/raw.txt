We can model each flag as a Boolean variable and each test suite as a clause with three literals.  
This is exactly a **3-SAT** problem. Since \( n \le 90 \) and \( m \le 400 \), we can solve it with a backtracking algorithm that uses unit propagation and pure literal elimination.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    clauses = []
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx+1])
        l3 = int(input_data[idx+2])
        idx += 3
        clauses.append((l1, l2, l3))
    
    assignment = [None] * (n + 1)  # 1-indexed, None = unassigned
    
    def propagate(assign):
        # returns (new_assign, conflict)
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                # Check if clause is already satisfied
                satisfied = False
                unassigned_lits = []
                for lit in clause:
                    var = abs(lit)
                    val = assign[var]
                    if val is not None:
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                    else:
                        unassigned_lits.append(lit)
                if satisfied:
                    continue
                if len(unassigned_lits) == 0:
                    return assign, True  # conflict
                if len(unassigned_lits) == 1:
                    lit = unassigned_lits[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else 0
                    if assign[var] is not None and assign[var] != val:
                        return assign, True
                    if assign[var] is None:
                        assign[var] = val
                        changed = True
        return assign, False
    
    def pure_literal_elimination(assign):
        # Find pure literals among unassigned variables
        pos_count = {}
        neg_count = {}
        for clause in clauses:
            satisfied = False
            for lit in clause:
                var = abs(lit)
                val = assign[var]
                if val is not None:
                    if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                        satisfied = True
                        break
            if satisfied:
                continue
            for lit in clause:
                var = abs(lit)
                if assign[var] is None:
                    if lit > 0:
                        pos_count[var] = pos_count.get(var, 0) + 1
                    else:
                        neg_count[var] = neg_count.get(var, 0) + 1
        changed = False
        for var in list(pos_count.keys()):
            if assign[var] is None:
                if var not in neg_count:
                    assign[var] = 1
                    changed = True
        for var in list(neg_count.keys()):
            if assign[var] is None:
                if var not in pos_count:
                    assign[var] = 0
                    changed = True
        return assign, changed
    
    def backtrack(assign):
        assign, conflict = propagate(assign[:])
        if conflict:
            return None
        # Pure literal elimination
        changed = True
        while changed:
            assign, changed = pure_literal_elimination(assign)
            if changed:
                assign, conflict = propagate(assign[:])
                if conflict:
                    return None
        # Check if all clauses satisfied
        all_satisfied = True
        for clause in clauses:
            sat = False
            for lit in clause:
                var = abs(lit)
                val = assign[var]
                if val is not None and ((lit > 0 and val == 1) or (lit < 0 and val == 0)):
                    sat = True
                    break
            if not sat:
                all_satisfied = False
                break
        if all_satisfied:
            return assign
        # Choose an unassigned variable
        for var in range(1, n+1):
            if assign[var] is None:
                # Try true
                assign1 = assign[:]
                assign1[var] = 1
                res = backtrack(assign1)
                if res is not None:
                    return res
                # Try false
                assign2 = assign[:]
                assign2[var] = 0
                res = backtrack(assign2)
                if res is not None:
                    return res
                return None
        return None
    
    result = backtrack(assignment)
    if result is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(str(result[i]) for i in range(1, n+1)))

if __name__ == "__main__":
    solve()
```
