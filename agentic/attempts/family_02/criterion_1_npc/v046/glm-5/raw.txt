```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Each clause is (a OR b OR c) where each literal is valve index (1-based)
    # Positive means valve should be OPEN (1), negative means CLOSED (0)
    clauses = []
    for _ in range(m):
        c1 = int(next(it))
        c2 = int(next(it))
        c3 = int(next(it))
        clauses.append((c1, c2, c3))
    
    # Try all 2^n configurations
    for mask in range(1 << n):
        valid = True
        for c1, c2, c3 in clauses:
            # Check if at least one literal is satisfied
            sat = False
            
            # Literal c1
            v = abs(c1) - 1
            bit = (mask >> v) & 1
            if c1 > 0 and bit == 1:
                sat = True
            elif c1 < 0 and bit == 0:
                sat = True
            
            # Literal c2
            if not sat:
                v = abs(c2) - 1
                bit = (mask >> v) & 1
                if c2 > 0 and bit == 1:
                    sat = True
                elif c2 < 0 and bit == 0:
                    sat = True
            
            # Literal c3
            if not sat:
                v = abs(c3) - 1
                bit = (mask >> v) & 1
                if c3 > 0 and bit == 1:
                    sat = True
                elif c3 < 0 and bit == 0:
                    sat = True
            
            if not sat:
                valid = False
                break
        
        if valid:
            print("YES")
            result = [(mask >> i) & 1 for i in range(n)]
            print(' '.join(map(str, result)))
            return
    
    print("NO")

solve()
```
