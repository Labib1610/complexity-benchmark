import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx + 1]), int(input_data[idx + 2])]
        idx += 3
        # Remove tautological clauses (e.g., contains both x and -x)
        s = set(c)
        if not any(-x in s for x in s):
            clauses.append(list(s))

    assignment = [0] * (n + 1)

    def dpll(assignment):
        assignment = assignment.copy()
        while True:
            unit_clauses = []
            for c in clauses:
                sat = False
                unassigned = []
                for lit in c:
                    val = assignment[abs(lit)]
                    if (lit > 0 and val == 1) or (lit < 0 and val == -1):
                        sat = True
                        break
                    elif val == 0:
                        unassigned.append(lit)
                if sat:
                    continue
                if len(unassigned) == 0:
                    return None
                if len(unassigned) == 1:
                    unit_clauses.append(unassigned[0])

            if not unit_clauses:
                break

            for lit in unit_clauses:
                var = abs(lit)
                val = 1 if lit > 0 else -1
                if assignment[var] != 0:
                    if assignment[var] != val:
                        return None
                else:
                    assignment[var] = val

        # Check if all clauses are satisfied
        all_sat = True
        scores = {}
        for c in clauses:
            sat = False
            unassigned = []
            for lit in c:
                val = assignment[abs(lit)]
                if (lit > 0 and val == 1) or (lit < 0 and val == -1):
                    sat = True
                    break
                elif val == 0:
                    unassigned.append(lit)
            if sat:
                continue
            all_sat = False
            w = 2 ** (-len(unassigned))
            for lit in unassigned:
                scores[lit] = scores.get(lit, 0.0) + w

        if all_sat:
            return assignment

        # Jeroslow-Wang heuristic
        best_var = None
        best_score = -1.0
        for var in range(1, n + 1):
            if assignment[var] == 0:
                s = scores.get(var, 0.0) + scores.get(-var, 0.0)
                if s > best_score:
                    best_score = s
                    best_var = var

        if best_var is None:
            return assignment

        first_val = 1 if scores.get(best_var, 0.0) >= scores.get(-best_var, 0.0) else -1

        for val in [first_val, -first_val]:
            assignment[best_var] = val
            res = dpll(assignment)
            if res is not None:
                return res
            assignment[best_var] = 0

        return None

    res = dpll(assignment)
    if res is None:
        print("NO")
    else:
        print("YES")
        ans = [1 if res[i] == 1 else 0 for i in range(1, n + 1)]
        print(*(ans))


if __name__ == "__main__":
    solve()
