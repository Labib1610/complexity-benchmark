We can model this as a **3-SAT** problem where each clause has 3 literals, and a clause is *unsatisfied* if all three literals are false.

Here’s one efficient approach:  
Since each protocol halts if **all three** are unfulfilled, we need at least one literal true per clause/year.

We can solve it using brute-force or **backtracking with unit propagation**, but `n` and `m` are small (n ≤ 90, m ≤ 400), so we can actually use a complete 3-SAT solver via recursion on variables with pruning.

Let’s code it:

```python
import sys
sys.setrecursionlimit(1000000)

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    clauses = []
    idx = 2
    for _ in range(m):
        v1 = int(input_data[idx])
        v2 = int(input_data[idx+1])
        v3 = int(input_data[idx+2])
        idx += 3
        clauses.append((v1, v2, v3))
    
    assignment = [-1] * (n + 1)  # -1 unassigned, 0 false, 1 true

    def is_clause_satisfied(clause):
        for lit in clause:
            var = abs(lit)
            val = assignment[var]
            if val == -1:
                continue
            if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                return True
        return False

    def backtrack(var_index):
        if var_index > n:
            # Check all clauses satisfied
            for clause in clauses:
                if not is_clause_satisfied(clause):
                    return False
            return True
        
        # Try true
        assignment[var_index] = 1
        # quick prune: if any clause already all false (but complete) with assigned vars only
        possible = True
        for clause in clauses:
            assigned_vals = []
            all_assigned = True
            for lit in clause:
                v = abs(lit)
                if assignment[v] == -1:
                    all_assigned = False
                    break
            if all_assigned and not is_clause_satisfied(clause):
                possible = False
                break
        if possible:
            if backtrack(var_index + 1):
                return True
        
        # Try false
        assignment[var_index] = 0
        possible = True
        for clause in clauses:
            all_assigned = True
            for lit in clause:
                v = abs(lit)
                if assignment[v] == -1:
                    all_assigned = False
                    break
            if all_assigned and not is_clause_satisfied(clause):
                possible = False
                break
        if possible:
            if backtrack(var_index + 1):
                return True
        
        assignment[var_index] = -1
        return False

    if backtrack(1):
        print("YES")
        print(" ".join(str(assignment[i]) for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
