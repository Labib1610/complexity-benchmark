```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        clause = [int(next(it)), int(next(it)), int(next(it))]
        clauses.append(clause)

    # assignment: None=unassigned, True=warm(1), False=cool(0)
    assign = [None] * (n + 1)

    # For each variable, keep clauses where it appears positively/negatively
    pos_occ = [[] for _ in range(n + 1)]
    neg_occ = [[] for _ in range(n + 1)]
    for idx, cl in enumerate(clauses):
        for lit in cl:
            var = abs(lit)
            if lit > 0:
                pos_occ[var].append(idx)
            else:
                neg_occ[var].append(idx)

    # Helper to get literal value under current assignment
    def lit_val(lit):
        v = assign[abs(lit)]
        if v is None:
            return None
        return v if lit > 0 else (not v)

    # Unit propagation stack
    def propagate(stack):
        while stack:
            var, val = stack.pop()
            if assign[var] is not None:
                if assign[var] != val:
                    return False
                continue
            assign[var] = val
            # check clauses where this literal appears positively/negatively
            affected = pos_occ[var] if val else neg_occ[var]
            for ci in affected:
                # clause already satisfied? skip
                cl = clauses[ci]
                if any(lit_val(l) is True for l in cl):
                    continue
                # count unassigned literals
                unassigned = [l for l in cl if lit_val(l) is None]
                if not unassigned:  # all false
                    return False
                if len(unassigned) == 1:
                    # force this literal to be true
                    l = unassigned[0]
                    v = abs(l)
                    needed = True if l > 0 else False
                    stack.append((v, needed))
        return True

    # Choose next variable (simple heuristic)
    def choose_var():
        for v in range(1, n + 1):
            if assign[v] is None:
                return v
        return None

    def dfs():
        # find any clause unsatisfied?
        for cl in clauses:
            vals = [lit_val(l) for l in cl]
            if any(v is True for v in vals):
                continue
            if all(v is False for v in vals):
                return False
        # all clauses satisfied?
        if all(assign[v] is not None for v in range(1, n + 1)):
            return True
        var = choose_var()
        # try True then False
        saved = assign[:]
        for val in (True, False):
            assign[:] = saved[:]
            if propagate([(var, val)]):
                if dfs():
                    return True
        assign[:] = saved[:]
        return False

    if dfs():
        print("YES")
        print(' '.join('1' if assign[i] else '0' for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
