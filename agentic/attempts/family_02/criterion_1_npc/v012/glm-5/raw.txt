import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    idx = 2
    clauses = []
    for _ in range(m):
        a = int(data[idx]); b = int(data[idx+1]); c = int(data[idx+2])
        idx += 3
        clauses.append((a, b, c))
    
    # Each clause: at least one of the three literals must be true.
    # Literal positive -> variable True (warm=1). Negative -> variable False (cool=0).
    # We try all 2^n assignments. n <= 90 is too large for brute force in general,
    # but we can use a smarter approach: since each clause has 3 literals,
    # we can reduce the problem.
    #
    # Key insight: if m == 0, any assignment works.
    # Otherwise, pick the first clause. At least one of its 3 literals must be true.
    # Try each of the 3 possibilities, fix that variable, and propagate / check remaining.
    #
    # But propagation isn't straightforward for "at least one true" (it's not 2-SAT).
    # However, we can use the following: after fixing one literal to true in the first clause,
    # we still need to satisfy all other clauses. This is still a 3-SAT-like problem.
    #
    # Given constraints n<=90, m<=400, we need a different approach.
    # 
    # Observation: We can model this as hitting set / set cover complement.
    # Each clause needs at least one of its 3 literals satisfied.
    # Total literals across all clauses: at most 1200.
    # 
    # Alternative: Since each clause has exactly 3 literals and we need at least 1 true,
    # the complement is: at most 2 false per clause. 
    # 
    # Let's try: for each clause, at least one literal true. 
    # We can try all combinations for a small subset of variables that appear in clauses.
    # 
    # Actually, let's think about it differently. The number of distinct variables 
    # that appear is at most min(n, 3*m) = min(90, 1200) = 90. Still large.
    #
    # Better approach: This is exactly the "one-in-three positive" variant... no.
    # It's "at least one of three" which is standard 3-SAT (monotone-ish but with negations).
    #
    # Wait - let's reconsider. n <= 90 but maybe we can use the structure.
    # Each clause has 3 literals. We need at least 1 satisfied.
    # 
    # DPLL-style: pick a clause with fewest unassigned, branch on its literals.
    # With m=400 clauses and 3 literals each, and good unit propagation,
    # this can be efficient in practice.
    #
    # Let's implement a recursive DPLL with unit propagation.
    
    assignment = [None] * (n + 1)  # None = unassigned, True/False
    
    def get_lit_val(lit, assignment):
        var = abs(lit)
        if assignment[var] is None:
            return None
        return assignment[var] if lit > 0 else (not assignment[var])
    
    def clause_status(clause, assignment):
        # Returns 'sat', 'unsat', or 'unknown' with count of unassigned
        unassigned = []
        for lit in clause:
            v = get_lit_val(lit, assignment)
            if v is True:
                return 'sat', None
            elif v is None:
                unassigned.append(lit)
        if not unassigned:
            return 'unsat', None
        return 'unknown', unassigned
    
    def propagate(assignment, clauses):
        # Unit propagation: if a clause has all but one literal false, 
        # the remaining must be true.
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                status, unassigned = clause_status(clause, assignment)
                if status == 'unsat':
                    return False
                elif status == 'unknown' and len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    assignment[var] = (lit > 0)
                    changed = True
        return True
    
    def dpll(assignment, clauses):
        # Propagate
        if not propagate(assignment, clauses):
            return None
        
        # Check all satisfied
        best_clause = None
        best_unassigned_count = float('inf')
        all_sat = True
        for clause in clauses:
            status, unassigned = clause_status(clause, assignment)
            if status != 'sat':
                all_sat = False
                if unassigned and len(unassigned) < best_unassigned_count:
                    best_unassigned_count = len(unassigned)
                    best_clause = unassigned
        
        if all_sat:
            return list(assignment)
        
        # Branch on first unassigned literal of the best clause
        for lit in best_clause:
            var = abs(lit)
            if assignment[var] is not None:
                continue
            new_assignment = assignment[:]
            new_assignment[var] = (lit > 0)
            result = dpll(new_assignment, clauses)
            if result is not None:
                return result
        
        return None
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    result = dpll(assignment, clauses)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        vals = []
        for i in range(1, n+1):
            if result[i] is None:
                vals.append("1")  # arbitrary
            else:
                vals.append("1" if result[i] else "0")
        print(" ".join(vals))

solve()
