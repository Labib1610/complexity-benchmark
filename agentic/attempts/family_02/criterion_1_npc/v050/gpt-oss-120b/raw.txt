```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        c = [int(next(it)), int(next(it)), int(next(it))]
        clauses.append(c)

    # assignment: list of None/0/1 for variables 1..n (index 0 unused)
    assign = [None] * (n + 1)

    # helper to get literal value under current assignment
    def lit_val(lit):
        v = abs(lit)
        val = assign[v]
        if val is None:
            return None
        return val if lit > 0 else 1 - val

    # unit propagation and pure literal elimination
    def propagate():
        changed = True
        while changed:
            changed = False
            # unit clauses
            for clause in clauses:
                vals = [lit_val(l) for l in clause]
                if any(v == 1 for v in vals):
                    continue  # already satisfied
                unknown = [clause[i] for i, v in enumerate(vals) if v is None]
                if not unknown:  # all false -> conflict
                    return False
                if len(unknown) == 1:
                    lit = unknown[0]
                    var = abs(lit)
                    needed = 1 if lit > 0 else 0
                    if assign[var] is None:
                        assign[var] = needed
                        changed = True
                    elif assign[var] != needed:
                        return False
            # pure literals
            count_pos = [0] * (n + 1)
            count_neg = [0] * (n + 1)
            for clause in clauses:
                for lit in clause:
                    v = abs(lit)
                    if assign[v] is not None:
                        continue
                    if lit > 0:
                        count_pos[v] += 1
                    else:
                        count_neg[v] += 1
            for v in range(1, n + 1):
                if assign[v] is None:
                    if count_pos[v] and not count_neg[v]:
                        assign[v] = 1
                        changed = True
                    elif count_neg[v] and not count_pos[v]:
                        assign[v] = 0
                        changed = True
        return True

    # recursive DPLL
    def dpll():
        if not propagate():
            return False
        # check if all clauses satisfied
        all_sat = True
        for clause in clauses:
            vals = [lit_val(l) for l in clause]
            if any(v == 1 for v in vals):
                continue
            if any(v is None for v in vals):
                all_sat = False
                break
            # all false
            return False
        if all_sat:
            return True
        # choose unassigned variable (heuristic: first)
        for v in range(1, n + 1):
            if assign[v] is None:
                var = v
                break
        # try 0 then 1
        saved = assign[:]
        assign[var] = 0
        if dpll():
            return True
        assign[:] = saved
        assign[var] = 1
        if dpll():
            return True
        assign[:] = saved
        return False

    if dpll():
        print("YES")
        print(' '.join(str(assign[i] if assign[i] is not None else 0) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
