from collections import defaultdict
import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each relay, we have a clause that at least one of three literals must be true
    # This is a 3-SAT variant where each clause is an "at least one" constraint
    # We can model this as 3-CNF SAT where each clause is (a OR b OR c)
    
    clauses = []
    for _ in range(m):
        w1, w2, w3 = map(int, sys.stdin.readline().split())
        clause = []
        for w in [w1, w2, w3]:
            if w > 0:
                clause.append((w-1, True))  # variable w-1 should be True (valve w set to 1)
            else:
                clause.append((-w-1, False)) # variable -w-1 should be False (valve -w set to 0)
        clauses.append(clause)
    
    # Try all 2^n possible assignments
    # But n can be up to 90, so 2^90 is too large
    # However, since each clause has exactly 3 literals, we might be able to use a more efficient approach
    
    # Let's try a backtracking approach with pruning
    
    def is_satisfied(assignment, clause):
        """Check if a clause is satisfied by the current assignment"""
        for var, value in clause:
            if var >= len(assignment):
                # Variable not yet assigned, so clause could still be satisfied
                return True
            if assignment[var] == value:
                return True
        return False
    
    def backtrack(assignment):
        # Check if current partial assignment violates any clause
        for clause in clauses:
            if all(var < len(assignment) for var, _ in clause):  # All variables in clause are assigned
                if not is_satisfied(assignment, clause):
                    return False
        
        if len(assignment) == n:
            return True
            
        # Try both values for the next unassigned variable
        for val in [False, True]:
            assignment.append(val)
            if backtrack(assignment):
                return True
            assignment.pop()
            
        return False
    
    # This approach might still be too slow. Let's think differently.
    # Since each clause has exactly 3 literals, and we need at least one to be true,
    # we can convert this to a graph problem or use a SAT solver approach.
    
    # Let's implement a more efficient backtracking with better pruning
    
    # Preprocess clauses to make indexing easier
    # Convert to 0-based indexing
    processed_clauses = []
    for clause in clauses:
        processed_clause = []
        for w in clause:
            var, sign = w
            processed_clause.append((var, sign))
        processed_clauses.append(processed_clause)
    
    assignment = [None] * n
    
    def is_currently_satisfiable():
        for clause in processed_clauses:
            satisfied = False
            for var, required_value in clause:
                if assignment[var] is not None:
                    if assignment[var] == required_value:
                        satisfied = True
                        break
                else:
                    # Variable not assigned yet, clause could be satisfied
                    satisfied = True
                    break
            if not satisfied:
                return False
        return True
    
    def backtrack_opt(index):
        if index == n:
            return True
            
        if not is_currently_satisfiable():
            return False
            
        # Try setting variable to False (0)
        assignment[index] = False
        if backtrack_opt(index + 1):
            return True
            
        # Try setting variable to True (1)
        assignment[index] = True
        if backtrack_opt(index + 1):
            return True
            
        # Backtrack
        assignment[index] = None
        return False
    
    # This is still potentially exponential but with better pruning
    if backtrack_opt(0):
        print("YES")
        result = []
        for val in assignment:
            result.append("1" if val else "0")
        print(" ".join(result))
    else:
        print("NO")

solve()
