import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    idx = 2
    clauses = []
    for _ in range(m):
        a = int(data[idx]); b = int(data[idx+1]); c = int(data[idx+2])
        idx += 3
        clauses.append((a, b, c))
    
    # 2-SAT with at-most-one auxiliary encoding for exactly-3 clauses.
    # Variables: 1..n are original. n+1.. are auxiliaries.
    size = n
    aux_start = size + 1
    for _ in range(m):
        size += 2  # two auxiliaries per clause
    total = size
    
    adj = [[] for _ in range(2 * total + 2)]
    
    def neg(x):
        if x > 0:
            return x + total
        else:
            return -x
    
    def add_impl(u, v):
        adj[neg(u)].append(v)
        adj[neg(v)].append(u)
    
    def add_or(u, v):
        add_impl(u, v)
    
    cur = aux_start
    for (a, b, c) in clauses:
        x = cur
        y = cur + 1
        cur += 2
        # x false => a true, b true
        add_impl(-x, a)
        add_impl(-x, b)
        # y false => b true, c true
        add_impl(-y, b)
        add_impl(-y, c)
        # at most one of x,y true
        add_or(-x, -y)
        # at least one of x,y true
        add_or(x, y)
    
    # Tarjan's SCC
    index_counter = [0]
    stack = []
    on_stack = [False] * (2 * total + 2)
    index = [-1] * (2 * total + 2)
    lowlink = [0] * (2 * total + 2)
    comp = [-1] * (2 * total + 2)
    comp_cnt = [0]
    
    sys.setrecursionlimit(1000000)
    
    def strongconnect(v):
        work = [(v, 0)]
        while work:
            node, pi = work[-1]
            if pi == 0:
                index[node] = index_counter[0]
                lowlink[node] = index_counter[0]
                index_counter[0] += 1
                stack.append(node)
                on_stack[node] = True
            recurse = False
            i = pi
            while i < len(adj[node]):
                w = adj[node][i]
                if index[w] == -1:
                    work[-1] = (node, i + 1)
                    work.append((w, 0))
                    recurse = True
                    break
                elif on_stack[w]:
                    if index[w] < lowlink[node]:
                        lowlink[node] = index[w]
                i += 1
            if recurse:
                continue
            if lowlink[node] == index[node]:
                while True:
                    w = stack.pop()
                    on_stack[w] = False
                    comp[w] = comp_cnt[0]
                    if w == node:
                        break
                comp_cnt[0] += 1
            work.pop()
            if work:
                parent = work[-1][0]
                if lowlink[node] < lowlink[parent]:
                    lowlink[parent] = lowlink[node]
    
    for v in range(1, 2 * total + 1):
        if index[v] == -1:
            strongconnect(v)
    
    for i in range(1, n + 1):
        if comp[i] == comp[i + total]:
            print("NO")
            return
    
    assignment = [0] * (total + 1)
    for i in range(1, total + 1):
        if comp[i] < comp[i + total]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(' '.join(str(assignment[i]) for i in range(1, n + 1)))

solve()
