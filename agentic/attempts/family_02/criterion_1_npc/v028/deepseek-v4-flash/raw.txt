We can model each truck’s assignment as a boolean variable (0 or 1).  
Each photographer gives a clause of three literals, and we need all clauses satisfied — this is a 3-SAT problem.  

Since \( n \le 90 \) and \( m \le 400 \), we can use a backtracking search with unit propagation and pure literal elimination, which is efficient enough for this size.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # variable assignment: 0 = unassigned, 1 = True, -1 = False
    assignment = [0] * (n + 1)
    
    def evaluate_literal(lit, assign):
        var = abs(lit)
        val = assign[var]
        if val == 0:
            return None  # unassigned
        if lit > 0:
            return val == 1
        else:
            return val == -1
    
    def clause_status(clause, assign):
        # returns True if satisfied, False if unsatisfied, None if unknown
        status = None
        for lit in clause:
            res = evaluate_literal(lit, assign)
            if res is True:
                return True
            if res is False:
                continue
            status = False  # at least one unassigned
        return status if status is not None else False
    
    def propagate(assign):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                status = clause_status(clause, assign)
                if status is False:
                    return False
                if status is None:
                    # exactly one unassigned literal, others false
                    unassigned = None
                    count_unassigned = 0
                    for lit in clause:
                        res = evaluate_literal(lit, assign)
                        if res is None:
                            unassigned = lit
                            count_unassigned += 1
                        elif res is False:
                            continue
                        else:
                            break
                    else:
                        if count_unassigned == 1:
                            var = abs(unassigned)
                            val = 1 if unassigned > 0 else -1
                            if assign[var] == 0:
                                assign[var] = val
                                changed = True
                            elif assign[var] != val:
                                return False
        return True
    
    def pure_literal_elimination(assign):
        # find pure literals among unassigned variables
        pos_count = {}
        neg_count = {}
        for clause in clauses:
            status = clause_status(clause, assign)
            if status is True:
                continue
            for lit in clause:
                var = abs(lit)
                if assign[var] != 0:
                    continue
                if lit > 0:
                    pos_count[var] = pos_count.get(var, 0) + 1
                else:
                    neg_count[var] = neg_count.get(var, 0) + 1
        for var in range(1, n+1):
            if assign[var] == 0:
                p = pos_count.get(var, 0)
                q = neg_count.get(var, 0)
                if p > 0 and q == 0:
                    assign[var] = 1
                elif q > 0 and p == 0:
                    assign[var] = -1
    
    def backtrack(assign):
        if not propagate(assign):
            return False
        pure_literal_elimination(assign)
        if not propagate(assign):
            return False
        
        # find unassigned variable
        var = None
        for i in range(1, n+1):
            if assign[i] == 0:
                var = i
                break
        if var is None:
            return True
        
        # try True
        assign[var] = 1
        if backtrack(assign):
            return True
        assign[var] = -1
        if backtrack(assign):
            return True
        assign[var] = 0
        return False
    
    if backtrack(assignment):
        print("YES")
        print(" ".join("1" if assignment[i] == 1 else "0" for i in range(1, n+1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
