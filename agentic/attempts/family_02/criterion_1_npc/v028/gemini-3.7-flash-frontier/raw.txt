import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    raw_clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        raw_clauses.append(c)
        
    # Simplify clauses
    clauses = []
    for c in raw_clauses:
        # Check if tautology (contains both x and -x)
        lits = set(c)
        tautology = False
        for lit in lits:
            if -lit in lits:
                tautology = True
                break
        if not tautology:
            clauses.append(list(lits))
            
    # Quick check with WalkSAT
    def walksat(max_flips=10000, max_tries=20, p=0.5):
        if not clauses:
            return [0] * (n + 1)
        for _ in range(max_tries):
            assignment = [random.choice([0, 1]) for _ in range(n + 1)]
            for _ in range(max_flips):
                unsat = []
                for ci, clause in enumerate(clauses):
                    satisfied = False
                    for lit in clause:
                        val = assignment[abs(lit)]
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                    if not satisfied:
                        unsat.append(ci)
                if not unsat:
                    return assignment
                
                # Pick a random unsatisfied clause
                ci = random.choice(unsat)
                clause = clauses[ci]
                
                # With probability p, pick random variable from clause
                if random.random() < p:
                    var = abs(random.choice(clause))
                else:
                    # Pick variable that minimizes the number of broken clauses
                    best_var = abs(clause[0])
                    min_breaks = float('inf')
                    for lit in clause:
                        var_candidate = abs(lit)
                        assignment[var_candidate] ^= 1
                        # Count breaks
                        breaks = 0
                        for c_other in clauses:
                            sat = False
                            for l in c_other:
                                val = assignment[abs(l)]
                                if (l > 0 and val == 1) or (l < 0 and val == 0):
                                    sat = True
                                    break
                            if not sat:
                                breaks += 1
                        assignment[var_candidate] ^= 1
                        if breaks < min_breaks:
                            min_breaks = breaks
                            best_var = var_candidate
                    var = best_var
                assignment[var] ^= 1
        return None

    # DPLL solver as complete fallback
    def dpll(clauses_state, assignment):
        # Unit propagation
        while True:
            unit_clauses = [c for c in clauses_state if len(c) == 1]
            if not unit_clauses:
                break
            for c in unit_clauses:
                lit = c[0]
                var = abs(lit)
                val = 1 if lit > 0 else 0
                if assignment[var] is not None and assignment[var] != val:
                    return False, None
                assignment[var] = val
                
            new_clauses = []
            for c in clauses_state:
                c_sat = False
                c_new = []
                for lit in c:
                    var = abs(lit)
                    if assignment[var] is not None:
                        if (lit > 0 and assignment[var] == 1) or (lit < 0 and assignment[var] == 0):
                            c_sat = True
                            break
                    else:
                        c_new.append(lit)
                if c_sat:
                    continue
                if not c_new:
                    return False, None
                new_clauses.append(c_new)
            clauses_state = new_clauses

        if not clauses_state:
            # Fill unassigned with 0
            for i in range(1, n + 1):
                if assignment[i] is None:
                    assignment[i] = 0
            return True, assignment

        # Pick literal using MOM / frequency heuristic
        counts = {}
        min_len = min(len(c) for c in clauses_state)
        for c in clauses_state:
            if len(c) == min_len:
                for lit in c:
                    counts[lit] = counts.get(lit, 0) + 1
        chosen_lit = max(counts.keys(), key=lambda l: counts[l])
        chosen_var = abs(chosen_lit)
        val = 1 if chosen_lit > 0 else 0

        # Branch 1
        a1 = list(assignment)
        a1[chosen_var] = val
        c1 = []
        conflict = False
        for c in clauses_state:
            c_sat = False
            c_new = []
            for lit in c:
                v = abs(lit)
                if a1[v] is not None:
                    if (lit > 0 and a1[v] == 1) or (lit < 0 and a1[v] == 0):
                        c_sat = True
                        break
                else:
                    c_new.append(lit)
            if c_sat:
                continue
            if not c_new:
                conflict = True
                break
            c1.append(c_new)
        if not conflict:
            res, sol = dpll(c1, a1)
            if res:
                return True, sol

        # Branch 2
        a2 = list(assignment)
        a2[chosen_var] = 1 - val
        c2 = []
        conflict = False
        for c in clauses_state:
            c_sat = False
            c_new = []
            for lit in c:
                v = abs(lit)
                if a2[v] is not None:
                    if (lit > 0 and a2[v] == 1) or (lit < 0 and a2[v] == 0):
                        c_sat = True
                        break
                else:
                    c_new.append(lit)
            if c_sat:
                continue
            if not c_new:
                conflict = True
                break
            c2.append(c_new)
        if not conflict:
            res, sol = dpll(c2, a2)
            if res:
                return True, sol

        return False, None

    sol = walksat(max_flips=1000, max_tries=15)
    if sol is not None:
        print("YES")
        print(*(sol[i] for i in range(1, n + 1)))
        return

    init_assign = [None] * (n + 1)
    sat, sol = dpll([list(c) for c in clauses], init_assign)
    if sat:
        print("YES")
        print(*(sol[i] for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
