```python
import sys

sys.setrecursionlimit(10000)

def parse():
    data = sys.stdin.read().strip().split()
    if not data:
        return 0,0,[]
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        lits = [int(next(it)) for _ in range(3)]
        clauses.append(lits)
    return n, m, clauses

def simplify(clauses, assign):
    """Return new clause list after applying current assignment.
    If a clause becomes empty -> return None (conflict)."""
    new_clauses = []
    for clause in clauses:
        satisfied = False
        new_clause = []
        for lit in clause:
            var = abs(lit)
            val = assign.get(var)
            if val is None:
                new_clause.append(lit)
            else:
                lit_val = (val if lit > 0 else not val)
                if lit_val:
                    satisfied = True
                    break
        if satisfied:
            continue
        if not new_clause:
            return None
        new_clauses.append(new_clause)
    return new_clauses

def unit_propagate(clauses, assign):
    while True:
        unit = None
        for clause in clauses:
            if len(clause) == 1:
                unit = clause[0]
                break
        if unit is None:
            break
        var = abs(unit)
        val = unit > 0
        if var in assign:
            if assign[var] != val:
                return None, None
        else:
            assign[var] = val
        clauses = simplify(clauses, assign)
        if clauses is None:
            return None, None
    return clauses, assign

def pure_literal_eliminate(clauses, assign):
    # find pure literals
    count = {}
    for clause in clauses:
        for lit in clause:
            count[lit] = count.get(lit,0)+1
    pure = {}
    for lit in count:
        var = abs(lit)
        if var in assign:
            continue
        if -lit not in count:
            pure[var] = lit>0
    if not pure:
        return clauses, assign
    for var,val in pure.items():
        assign[var]=val
    clauses = simplify(clauses, assign)
    return clauses, assign

def dpll(clauses, assign, n):
    clauses, assign = unit_propagate(clauses, assign)
    if clauses is None:
        return None
    clauses, assign = pure_literal_eliminate(clauses, assign)
    if clauses is None:
        return None
    if not clauses:
        # all satisfied
        # fill unassigned with False arbitrarily
        result = [0]*n
        for i in range(1,n+1):
            if assign.get(i, False):
                result[i-1]=1
        return result
    # choose variable
    for clause in clauses:
        for lit in clause:
            var = abs(lit)
            if var not in assign:
                chosen = var
                break
        else:
            continue
        break
    for val in (True, False):
        new_assign = assign.copy()
        new_assign[chosen]=val
        new_clauses = simplify(clauses, new_assign)
        if new_clauses is None:
            continue
        res = dpll(new_clauses, new_assign, n)
        if res is not None:
            return res
    return None

def main():
    n,m,clauses = parse()
    assign = {}
    res = dpll(clauses, assign, n)
    if res is None:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str,res)))

if __name__ == "__main__":
    main()
```
