```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        lits = []
        for _ in range(3):
            x = int(next(it))
            var = abs(x) - 1
            val = 1 if x > 0 else 0
            lits.append((var, val))
        clauses.append(lits)

    # assignment: -1 unassigned, 0 or 1 assigned
    assign = [-1] * n

    # For each variable, list of clauses indices where it appears
    var_clauses = [[] for _ in range(n)]
    for idx, cl in enumerate(clauses):
        for var, _ in cl:
            var_clauses[var].append(idx)

    # Helper to check clause status
    def clause_status(cl):
        """return 1 if satisfied, 0 if unsatisfied, -1 if undetermined"""
        unknown = False
        for var, val in cl:
            a = assign[var]
            if a == -1:
                unknown = True
            elif a == val:
                return 1
        return 0 if not unknown else -1

    # Unit propagation queue
    from collections import deque

    def propagate(queue):
        while queue:
            var, val = queue.popleft()
            if assign[var] != -1:
                if assign[var] != val:
                    return False
                continue
            assign[var] = val
            # check affected clauses
            for ci in var_clauses[var]:
                st = clause_status(clauses[ci])
                if st == 0:  # unsatisfied
                    return False
                if st == -1:
                    # maybe unit clause?
                    unassigned = []
                    for v, vval in clauses[ci]:
                        if assign[v] == -1:
                            unassigned.append((v, vval))
                        elif assign[v] == vval:
                            break
                    else:
                        # none satisfied yet
                        if len(unassigned) == 1:
                            queue.append(unassigned[0])
        return True

    # Choose next variable (simple heuristic: first unassigned)
    def choose_var():
        for i in range(n):
            if assign[i] == -1:
                return i
        return -1

    def dfs():
        # initial unit propagation already done before call
        var = choose_var()
        if var == -1:
            return True
        # try 0 then 1
        saved = assign[:]
        for val in (0, 1):
            assign[:] = saved[:]
            q = deque()
            q.append((var, val))
            if propagate(q):
                if dfs():
                    return True
        assign[:] = saved[:]
        return False

    # initial propagation (none)
    if dfs():
        print("YES")
        print(' '.join(str(v if v != -1 else 0) for v in assign))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
