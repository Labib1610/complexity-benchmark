from collections import defaultdict

def main():

    m = int(data[1])

        print("YES")
        print(" ".join(["0"] * n))
        return

    constraints = []
    idx = 2

        l1 = int(data[idx])
        l2 = int(data[idx+1])
        l3 = int(data[idx+2])
        idx += 3
        constraints.append((l1, l2, l3))
    
    # For each truck, we maintain possible assignments
    # 0: lower corridor, 1: upper corridor
    # Initially, all trucks can be either 0 or 1
    possible = [set([0,1]) for _ in range(n+1)]  # index 0 unused
    
    # We'll use a greedy approach with backtracking simulation
    # But since we need to satisfy all constraints, let's think differently.
    # This is essentially a 2-SAT problem or can be solved with implication graph.
    # However, given small constraints, we can try a different approach.

    # But perhaps simpler: for each constraint (a or b or c), at least one must be satisfied.
    # We can try all 2^n combinations, but n=90 is too large.
    # However, notice that each constraint only involves 3 variables.
    # We can build an implication graph and check 2-SAT.
    
    # Actually, this isn't directly 2-SAT because we have 3 terms in OR.
    # But we can convert it.
    # (a or b or c) is equivalent to (!a => (b or c)) and (!b => (a or c)) and (!c => (a or b))
    # Which is equivalent to (!a => b or c) = (!a => b) or (!a => c)
    # In 2-SAT form: (a or b) and (a or c) if we have (!a => b or c)

    # (a or b or c) = (a or b or c)
    # To make it 2-SAT, we introduce auxiliary variables.
    # But for implementation simplicity, let's use a different approach.
    
    # Let's think of it this way:
    # For each constraint, at least one of the literals must be true.
    # A literal is either x_i or not x_i.
    # So for constraint (l1 or l2 or l3), we want at least one to be satisfied.
    
    # We can model this as finding an assignment such that for every constraint,
    # at least one of its literals is true.
    # This is a satisfiability problem with clauses of size 3.
    
    # One approach: for each constraint, pick one literal that we "try" to satisfy.
    # But that doesn't guarantee global solution.
    
    # Let's use backtracking with constraint propagation.
    # But with n up to 90, that's infeasible.
    
    # Wait, let's re-examine the problem.
    # Each constraint is an OR of 3 literals. We need all constraints to be satisfied.
    # This is MAX-3-SAT decision problem for all clauses satisfiable.
    
    # There's a better way. Since each clause has 3 literals, we can encode it properly.
    # But let's think of a simpler method.
    
    # Key insight: for each constraint, we must satisfy at least one literal.

    # Let's try a greedy + backtracking hybrid.
    # First, identify forced assignments.
    # Then, for remaining, try both values.

    # (a or b or c) is equivalent to:
    # (a or y) and (!y or b or c)

    # (!y or z) and (!z or b or c), etc.

    # (!a and !b) => c

    # To convert 3-CNF to 2-CNF equisatisfiable:

    # (a or b or z) and (!z or c or false) - No, that's wrong.

    # (a or b or c) becomes
    # (a or b or z) and (!z or c)
    # where z is a new boolean variable.

    # The original variables still need to have a consistent assignment.

    # But n=90 is too big.

    # Let's consider that in each constraint (at most 3 literals),

    # If setting truck i to 0 violates a constraint where it's required to be 1,

    # If we have two constraints like:
    # (1) (truck 1 in upper) OR ...
    # (2) (truck 1 in lower) OR ...
    # And these are the ONLY options in their respective constraints,
    # and the rest of the literals in those constraints can never be satisfied,
    # then we have a problem.

    # Better yet, let's just implement a quick 3-SAT checker for small cases,
    # but since n can be 90, that won't work.
    
    # I think the intended solution is to model this as a 2-SAT with auxiliary variables,
    # or realize that due to the structure, we can solve it efficiently.
    
    # Let me try to implement the proper 2-SAT reduction.
    
    # For clause (a or b or c), introduce new variable y and replace with:
    # (a or b or y) and (!y or c)
    # But the first is still a 3-clause.
    
    # Do it again:
    # (a or b or y) becomes (a or z) and (!z or b or y) 
    # and that becomes (!z or b or y) = (!z or w) and (!w or b or y)...
    # This is getting messy.
    
    # Textbook way:
    # (x1 or x2 or x3) is replaced by:
    # (x1 or x2 or y) and (!y or x3)
    # But first is still 3-clause.

    # Any 3-clause (a or b or c) gets replaced by (a or b or y) and (!y or c).
    # Now we have one 3-clause and one 2-clause.
    # Replace the 3-clause again: (a or z) and (!z or b or y).
    # Now we have (a or z), (!z or b or y), (!y or c).
    # Replace (!z or b or y): (!z or w) and (!w or b or y) - still 3-clause.
    # This process continues and creates O(k) new variables for a k-clause.

    # (a or b or c) => (a or b or y) and (!y or c)
    # But (a or b or y) is still 3-clause.

    # Introduce new variable y.
    # Clauses: (a or b or y) and (!y or c)

    # To convert (x1 or x2 or x3) into 2-clauses equisatisfiable:

    # Clauses: (x1 or x2 or y) and (!y or x3)

    # The point is that we can convert any 3-CNF to a 2-CNF formula (with new variables)

    # Third clause: (!T or F) = (F or F) = F

    # So for each 3-literal clause (a or b or c), we add:

    clauses = []
    aux_var_counter = n + 1
    
    for l1, l2, l3 in constraints:
        # Clause is (l1 or l2 or l3)
        # Convert to (l1 or y) and (l2 or y) and (!y or l3)
        y = aux_var_counter
        aux_var_counter += 1
        clauses.append((l1, y))
        clauses.append((l2, y))
        clauses.append((-y, l3))
        
    # Now solve 2-SAT
    # Number of variables: up to aux_var_counter - 1
    num_vars = aux_var_counter - 1
    
    # Build implication graph
    # Nodes: 1..num_vars for True, -(1..num_vars) for False
    # For clause (a or b), it's equivalent to (!a => b) and (!b => a)
    
    adj = defaultdict(list)
    for a, b in clauses:
        # (!a => b) and (!b => a)
        adj[-a].append(b)
        adj[-b].append(a)
        
    # Check for contradiction: if x and -x are in the same SCC
    def scc():
        # Kosaraju's algorithm
        visited = [False] * (2 * num_vars + 1)
        order = []
        
        def dfs1(u):
            visited[u] = True
            for v in adj[u]:
                if not visited[v]:
                    dfs1(v)
            order.append(u)
            
        for i in range(1, num_vars + 1):
            if not visited[i]:
                dfs1(i)
            if not visited[-i]:
                dfs1(-i)
                
        # Transpose graph
        adj_t = defaultdict(list)
        for u in adj:
            for v in adj[u]:
                adj_t[v].append(u)
                
        visited = [False] * (2 * num_vars + 1)
        scc_id = {}
        scc_count = 0
        
        def dfs2(u):
            visited[u] = True
            scc_id[u] = scc_count
            for v in adj_t[u]:
                if not visited[v]:
                    dfs2(v)
                    
        while order:
            u = order.pop()
            if not visited[u]:
                scc_count += 1
                dfs2(u)
                
        # Check for contradictions
        for i in range(1, n + 1):
            if scc_id.get(i, 0) == scc_id.get(-i, 0):
                return None  # Unsatisfiable
                
        # Extract assignment
        assignment = {}
        # Process SCCs in reverse topological order
        # We already have the order from Kosaraju
        for i in range(1, num_vars + 1):
            if i not in assignment:
                # If -i was processed earlier (has higher SCC id), set i = False
                # Else set i = True
                if scc_id.get(-i, 0) > scc_id.get(i, 0):
                    assignment[i] = False
                    assignment[-i] = True
                else:
                    assignment[i] = True
                    assignment[-i] = False
                    
        return assignment
        
    result = scc()
    if result is None:
        print("NO")
    else:
        print("YES")
        ans = []
        for i in range(1, n + 1):
            # result[i] = True means truck i is in corridor 1 (upper)
            # result[i] = False means truck i is in corridor 0 (lower)
            if result[i]:
                ans.append("1")
            else:
                ans.append("0")
        print(" ".join(ans))
