import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT setup
    # Variables: x_i = True means truck i is in corridor 1, False means corridor 0
    # Node indices: 2*i for x_i, 2*i+1 for NOT x_i
    # i ranges from 0 to n-1
    
    N = 2 * n
    adj = [[] for _ in range(N)]
    radj = [[] for _ in range(N)]
    
    def add_implication(u, v):
        adj[u].append(v)
        radj[v].append(u)
    
    def var_node(truck, corridor):
        # truck: 1-indexed, corridor: 0 or 1
        # returns node index for (truck is in corridor)
        i = truck - 1
        if corridor == 1:
            return 2 * i
        else:
            return 2 * i + 1
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        
        shots = [l1, l2, l3]
        
        # For each shot, determine truck and required corridor
        # shot satisfied if truck is in required corridor
        # Contract honored if at least one shot is satisfied
        # So we need: (shot1 OR shot2 OR shot3)
        
        # Convert to 2-SAT:
        # (a OR b OR c) = (a OR b) OR c
        # We introduce auxiliary variables
        
        # shot1: truck1 in corridor c1
        t1 = abs(shots[0])
        c1 = 1 if shots[0] > 0 else 0
        
        # shot2: truck2 in corridor c2
        t2 = abs(shots[1])
        c2 = 1 if shots[1] > 0 else 0
        
        # shot3: truck3 in corridor c3
        t3 = abs(shots[2])
        c2_val = c2  # save for later
        c3 = 1 if shots[2] > 0 else 0
        
        # Let a = (t1 in c1), b = (t2 in c2), c = (t3 in c3)
        # We need a OR b OR c
        
        # Using auxiliary variable y:
        # y = a OR b
        # Then we need y OR c
        
        # y = a OR b:
        # NOT y => NOT a, NOT y => NOT b
        # a => y, b => y
        
        # y OR c:
        # NOT y => c
        
        # Allocate auxiliary variable y (truck n+1, but we use indices directly)
        # y_true node: N, y_false node: N+1
        y_true = N
        y_false = N + 1
        N += 2
        
        adj.append([])
        adj.append([])
        radj.append([])
        radj.append([])
        
        a_true = var_node(t1, c1)
        a_false = a_true ^ 1
        b_true = var_node(t2, c2_val)
        b_false = b_true ^ 1
        c_true = var_node(t3, c3)
        c_false = c_true ^ 1
        
        # y = a OR b
        add_implication(y_false, a_false)  # NOT y => NOT a
        add_implication(y_false, b_false)  # NOT y => NOT b
        add_implication(a_true, y_true)     # a => y
        add_implication(b_true, y_true)     # b => y
        
        # y OR c
        add_implication(y_false, c_true)    # NOT y => c
        
        # Also need: NOT c => y (from y OR c)
        add_implication(c_false, y_true)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                v = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[v]:
                    stack.append((v, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * N
    comp_id = 0
    
    def dfs2(u, cid):
        stack = [u]
        while stack:
            node = stack.pop()
            if comp[node] == -1:
                comp[node] = cid
                for v in radj[node]:
                    if comp[v] == -1:
                        stack.append(v)
    
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, comp_id)
            comp_id += 1
    
    # Check for contradiction
    assignment = [False] * n
    possible = True
    
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            possible = False
            break
        assignment[i] = comp[2*i] > comp[2*i+1]
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join('1' if assignment[i] else '0' for i in range(n)))

solve()
