import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        c1 = int(next(it)); c2 = int(next(it)); c3 = int(next(it))
        clauses.append([c1, c2, c3])

    # DPLL
    def simplify(clauses, assign):
        """apply current assignment, return (new_clauses, conflict_flag)"""
        new_clauses = []
        for cl in clauses:
            satisfied = False
            new_clause = []
            for lit in cl:
                var = abs(lit)
                if var in assign:
                    val = assign[var]
                    if (lit > 0 and val) or (lit < 0 and not val):
                        satisfied = True
                        break
                    else:
                        continue  # literal false, drop it
                else:
                    new_clause.append(lit)
            if satisfied:
                continue
            if not new_clause:   # empty clause -> conflict
                return None, True
            new_clauses.append(new_clause)
        return new_clauses, False

    def unit_propagate(clauses, assign):
        while True:
            unit = None
            for cl in clauses:
                if len(cl) == 1:
                    unit = cl[0]
                    break
            if unit is None:
                break
            var = abs(unit)
            val = unit > 0
            if var in assign:
                if assign[var] != val:
                    return None, True
            else:
                assign[var] = val
                clauses, conflict = simplify(clauses, assign)
                if conflict:
                    return None, True
        return clauses, False

    def pure_literal_eliminate(clauses, assign):
        while True:
            counts = {}
            for cl in clauses:
                for lit in cl:
                    counts[lit] = counts.get(lit, 0) + 1
            pure = None
            for lit, cnt in counts.items():
                if -lit not in counts:
                    pure = lit
                    break
            if pure is None:
                break
            var = abs(pure)
            val = pure > 0
            assign[var] = val
            clauses, conflict = simplify(clauses, assign)
            if conflict:
                return None, True
        return clauses, False

    def dpll(clauses, assign):
        clauses, conflict = simplify(clauses, assign)
        if conflict:
            return None
        if not clauses:
            return assign
        clauses, conflict = unit_propagate(clauses, assign)
        if conflict:
            return None
        clauses, conflict = pure_literal_eliminate(clauses, assign)
        if conflict:
            return None
        # choose variable heuristically: first literal of first clause
        var = abs(clauses[0][0])
        for val in (True, False):
            new_assign = assign.copy()
            new_assign[var] = val
            res = dpll(clauses, new_assign)
            if res is not None:
                return res
        return None

    result = dpll(clauses, {})
    if result is None:
        print("NO")
    else:
        print("YES")
        out = []
        for i in range(1, n + 1):
            out.append('1' if result.get(i, False) else '0')
        print(' '.join(out))

if __name__ == "__main__":
    solve()
