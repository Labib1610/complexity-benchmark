from collections import defaultdict
import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each constraint, we have a clause with 3 literals
    # Positive literal means switch should be 1
    # Negative literal means switch should be 0
    # We need to satisfy all clauses (at least one literal true in each clause)
    
    # This is a 3-SAT problem. We'll use a brute force approach since n <= 90
    # but we can be smart about it - try all 2^n possibilities? No, that's too much.
    # Let's use a more systematic approach like backtracking with unit propagation
    
    # Let's represent each constraint as a set of literals
    constraints = []
    for _ in range(m):
        literals = list(map(int, sys.stdin.readline().split()))
        constraints.append(literals)
    
    # Try to solve using backtracking
    def is_satisfied(assignment):
        # Check if all constraints are satisfied
        for clause in constraints:
            satisfied = False
            for literal in clause:
                switch_idx = abs(literal)
                required_value = 1 if literal > 0 else 0
                if switch_idx in assignment:
                    if assignment[switch_idx] == required_value:
                        satisfied = True
                        break
            if not satisfied:
                return False
        return True
    
    def backtrack(assignment, unassigned):
        # If all variables are assigned, check if it satisfies all constraints
        if not unassigned:
            return is_satisfied(assignment)
        
        # Pick an unassigned variable
        var = unassigned.pop()
        
        # Try assigning 0
        assignment[var] = 0
        if backtrack(assignment, unassigned):
            return True
        
        # Try assigning 1
        assignment[var] = 1
        if backtrack(assignment, unassigned):
            return True
        
        # Backtrack
        del assignment[var]
        unassigned.add(var)
        return False
    
    # Start with no assignments
    assignment = {}
    unassigned = set(range(1, n+1))
    
    # Try to find a solution
    if backtrack(assignment, unassigned):
        print("YES")
        result = []
        for i in range(1, n+1):
            result.append(str(assignment.get(i, 0)))
        print(" ".join(result))
    else:
        print("NO")

            satisfied_with_0 = False

def solve_proper():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    constraints = []
    for _ in range(m):
        literals = list(map(int, input().split()))
        constraints.append(literals)
    
    def is_clause_satisfied(clause, assignment):
        for literal in clause:
            switch_idx = abs(literal)
            required_value = 1 if literal > 0 else 0
            if switch_idx in assignment and assignment[switch_idx] == required_value:
                return True
        return False
    
    def is_all_satisfied(assignment):
        for clause in constraints:
            if not is_clause_satisfied(clause, assignment):
                return False
        return True
    
    def backtrack(pos, assignment):
        if pos > n:
            return is_all_satisfied(assignment)
        
        # Try 0
        assignment[pos] = 0
        if backtrack(pos + 1, assignment):
            return True
        
        # Try 1
        assignment[pos] = 1
        if backtrack(pos + 1, assignment):
            return True
        
        del assignment[pos]
        return False
    
    assignment = {}
    if backtrack(1, assignment):
        print("YES")
        result = []
        for i in range(1, n + 1):
            result.append(str(assignment[i]))
        print(" ".join(result))
    else:
        print("NO")

solve_proper()
