import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve_sat():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx + 1]), int(input_data[idx + 2])]
        idx += 3
        clauses.append(c)

    # Pre-simplify initial clauses (remove duplicate literals, tautologies)
    initial_clauses = []
    for c in clauses:
        c_set = set(c)
        tautology = any(-l in c_set for l in c_set)
        if not tautology:
            initial_clauses.append(list(c_set))

    def dpll(clauses, assignment):
        # Unit propagation
        while True:
            unit_found = False
            for c in clauses:
                if len(c) == 0:
                    return None
                if len(c) == 1:
                    unit = c[0]
                    var = abs(unit)
                    val = 1 if unit > 0 else 0
                    if var in assignment:
                        if assignment[var] != val:
                            return None
                    else:
                        assignment[var] = val
                    unit_found = True

                    new_clauses = []
                    for clause in clauses:
                        if unit in clause:
                            continue
                        if -unit in clause:
                            new_c = [l for l in clause if l != -unit]
                            new_clauses.append(new_c)
                        else:
                            new_clauses.append(clause)
                    clauses = new_clauses
                    break
            if not unit_found:
                break

        for c in clauses:
            if len(c) == 0:
                return None

        if not clauses:
            for v in range(1, n + 1):
                if v not in assignment:
                    assignment[v] = 0
            return assignment

        # Jeroslow-Wang heuristic for literal choice
        jw = {}
        for c in clauses:
            weight = 2 ** (-len(c))
            for l in c:
                jw[l] = jw.get(l, 0.0) + weight

        best_lit = max(jw, key=jw.get)
        best_var = abs(best_lit)
        best_val = 1 if best_lit > 0 else 0

        # Branch on the most promising value first
        for val in (best_val, 1 - best_val):
            lit = best_var if val == 1 else -best_var
            new_assignment = assignment.copy()
            new_assignment[best_var] = val

            new_clauses = []
            possible = True
            for c in clauses:
                if lit in c:
                    continue
                if -lit in c:
                    rem = [l for l in c if l != -lit]
                    if not rem:
                        possible = False
                        break
                    new_clauses.append(rem)
                else:
                    new_clauses.append(c)

            if possible:
                res = dpll(new_clauses, new_assignment)
                if res is not None:
                    return res

        return None

    res = dpll(initial_clauses, {})
    if res is None:
        print("NO")
    else:
        print("YES")
        ans = [res[i] for i in range(1, n + 1)]
        print(*ans)


if __name__ == "__main__":
    solve_sat()
