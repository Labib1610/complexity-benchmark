from collections import defaultdict
import sys

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    rules = []
    for _ in range(m):
        rule = list(map(int, sys.stdin.readline().split()))
        rules.append(rule)
    return n, m, rules

def solve_2sat(n, rules):
    # For each variable i, we have nodes 2*i (for False) and 2*i+1 (for True)
    # But we'll use 2*i for x_i and 2*i+1 for not x_i
    
    # Convert rules to 2-SAT clauses
    # Each rule is (l1 or l2 or l3)
    # We convert to 2-SAT: (l1 or l2) and (l2 or l3) and (l1 or l3)
    # But actually, we need to convert 3-SAT to 2-SAT properly
    # (a or b or c) is equivalent to (a or b or y) and (not y or c)
    # But for simplicity, we'll use the standard reduction:
    # (a or b or c) = (a or b) and (not(not a and not b) or c)
    # Let's use auxiliary variables approach
    
    # Actually, let's think differently. 
    # A rule (l1 or l2 or l3) means at least one of them must be true
    # This is equivalent to saying it's NOT the case that all are false
    # So we can add clauses (l1 or l2 or l3)
    
    # But 2-SAT solver works with 2-clauses, so we need to reduce 3-clauses to 2-clauses
    # (a or b or c) is equivalent to (a or b or y) and (not y or c) where y is a new variable
    # But this increases the number of variables...
    
    # Let's implement a proper 3-SAT approach instead of reducing to 2-SAT
    
    # We'll use a brute force approach since n <= 90, but that's too much
    # Let's try a more efficient approach
    
    # Actually, let's reduce 3-SAT to 2-SAT:
    # For each clause (a or b or c), introduce a new variable y
    # Add clauses: (a or b or y) and (not y or c)
    # But (a or b or y) still needs to be broken down:
    # This is equivalent to (a or b or y) which is 3-clause again
    
    # Better approach for 3-SAT to 2-SAT:
    # (x1 or x2 or x3) = (x1 or x2 or y) and (not y or x3)
    # But (x1 or x2 or y) = (x1 or z) and (not z or x2 or y)
    # This is getting complex. Let's use a different approach.
    
    # Let's use the fact that we can represent any 3-clause as conjunction of 2-clauses
    # with auxiliary variables, but that changes our variable count.
    
    # Let's implement a direct backtracking solution with pruning
    
    def literal_to_index(lit):
        # lit is +i or -i
        # returns (variable_index, value)
        # variable_index is 0-based
        if lit > 0:
            return (lit - 1, 1)
        else:
            return (-lit - 1, 0)
    
    # For small n, we can try all 2^n assignments, but n=90 is too large
    # We need a more sophisticated approach
    
    # Let's model this as a 2-SAT problem by converting 3-clauses to 2-clauses
    # For clause (a or b or c), we add:
    # (a or b) or c -> introduce auxiliary variable y
    # Add clauses: (a or b or y) - but this is still 3-clause
    # Actually: (a or b or c) is equivalent to 
    # (a or b) if not c, and (a or c) if not b, and (b or c) if not a
    # No, that's not right.
    
    # The correct reduction:
    # (a or b or c) is equivalent to (a or b or y) and (not y or c)
    # But (a or b or y) is still 3-clause.
    
    # Simpler reduction:
    # (a or b or c) = (a or b) or c
    # This is true if (a or b) is true, or if c is true
    # We can't directly convert to 2-clauses without auxiliary variables
    
    # Let's use a different approach: implement a SAT solver for 3-SAT
    
    # But given the constraints, let's try to model this properly
    
    # Each rule is a 3-clause: at least one literal must be true
    # We can solve this with a SAT solver approach
    
    # Let's implement a simple backtracking with unit propagation
    
    assignment = [None] * n  # None means unassigned
    
    def is_clause_satisfied(clause):
        # Check if a 3-clause is satisfied by current assignment
        for lit in clause:
            var_idx, expected_val = literal_to_index(lit)
            if assignment[var_idx] is not None:
                if assignment[var_idx] == expected_val:
                    return True  # Literal is satisfied
        return False  # No satisfied literal found
    
    def is_all_clauses_satisfied():
        for clause in rules:
            if not is_clause_satisfied(clause):
                return False
        return True
    
    def find_unit_clause():
        # Find a clause where all but one literals are false, 
        # and the remaining one is unassigned
        for clause in rules:
            unassigned_count = 0
            unassigned_literal = None
            satisfied = False
            
            for lit in clause:
                var_idx, expected_val = literal_to_index(lit)
                if assignment[var_idx] is not None:
                    if assignment[var_idx] == expected_val:
                        satisfied = True
                        break
                else:
                    unassigned_count += 1
                    unassigned_literal = lit
            
            if satisfied:
                continue
                
            if unassigned_count == 1:
                return unassigned_literal
                
        return None
    
    def backtrack():
        # Unit propagation
        while True:
            unit_lit = find_unit_clause()
            if unit_lit is None:
                break
            var_idx, expected_val = literal_to_index(unit_lit)
            if assignment[var_idx] is not None:
                if assignment[var_idx] != expected_val:
                    return False  # Conflict
                continue
            assignment[var_idx] = expected_val
        
        # Check if all clauses are satisfied
        if is_all_clauses_satisfied():
            return True
            
        # Find an unassigned variable
        var_to_assign = None
        for i in range(n):
            if assignment[i] is None:
                var_to_assign = i
                break
                
        if var_to_assign is None:
            return False  # All assigned but not satisfied
            
        # Try assigning True
        assignment[var_to_assign] = 1
        if backtrack():
            return True
        assignment[var_to_assign] = None
        
        # Try assigning False
        assignment[var_to_assign] = 0
        if backtrack():
            return True
        assignment[var_to_assign] = None
        
        return False
    
    # But this might be too slow. Let's try a more efficient approach.
    
    # Let's reduce to 2-SAT by introducing auxiliary variables
    # For each 3-clause (a or b or c), we introduce a new variable y
    # and add constraints: (not y or a or b) and (y or c)
    # But (not y or a or b) is still a 3-clause.
    
    # The standard approach:
    # (x1 or x2 or x3) is equivalent to:
    # (y1 or x3) and (not y1 or x1 or x2)
    # where y1 is a new variable
    # Then (not y1 or x1 or x2) is equivalent to:
    # (y2 or x2) and (not y2 or not y1 or x1)
    # where y2 is another new variable
    
    # This is getting complex. Let's implement a simple solution
    # and optimize if needed.
    
    # Let's try a different approach: since n <= 90 but that's too large for brute force,
    # and m <= 400, maybe we can do some smart pruning.
    
    # Let's go back to the 2-SAT reduction properly.
    
    # For clause (a or b or c), we can add clauses:
    # (a or b) or c
    # This is equivalent to: if not (a or b), then c
    # Which is: (not a and not b) implies c
    # Which is: (a or b or c) - we're back to square one.
    
    # The correct reduction to 2-SAT with auxiliary variables:
    # Clause (x1 or x2 or x3) becomes:
    # Add new variable y
    # Clauses: (x1 or x2 or y) and (not y or x3)
    # But (x1 or x2 or y) is still 3-clause.
    
    # Let's do it step by step:
    # (x1 or x2 or x3) = (x1 or (x2 or x3))
    # To make it 2-SAT, we introduce y1 to represent (x2 or x3)
    # So we have (x1 or y1) and y1 = (x2 or x3)
    # But y1 = (x2 or x3) is not a clause - it's an equivalence.
    # We need to represent y1 <=> (x2 or x3)
    # Which is (not y1 or x2 or x3) and (not x2 or y1) and (not x3 or y1)
    # The first one is still 3-clause.
    
    # This is not working. Let's implement a direct solution.
    
    # Since n can be up to 90, we can't enumerate all assignments.
    # We need to use a SAT solving technique.
    
    # Let's implement a simple DPLL-style algorithm
    
    def dpll():
        # Unit propagation
        changed = True
        while changed:
            changed = False
            for clause in rules:
                # Count satisfied, unsatisfied, and unassigned literals
                satisfied = False
                unassigned_count = 0
                unassigned_literal = None
                
                for lit in clause:
                    var_idx, expected_val = literal_to_index(lit)
                    if assignment[var_idx] is not None:
                        if assignment[var_idx] == expected_val:
                            satisfied = True
                            break
                    else:
                        unassigned_count += 1
                        unassigned_literal = lit
                
                if satisfied:
                    continue
                    
                if unassigned_count == 0:
                    return False  # Unsatisfiable clause
                    
                if unassigned_count == 1:
                    # Unit clause
                    var_idx, expected_val = literal_to_index(unassigned_literal)
                    if assignment[var_idx] is not None:
                        if assignment[var_idx] != expected_val:
                            return False  # Conflict
                    else:
                        assignment[var_idx] = expected_val
                        changed = True
        
        # Check if all clauses are satisfied
        all_satisfied = True
        for clause in rules:
            satisfied = False
            for lit in clause:
                var_idx, expected_val = literal_to_index(lit)
                if assignment[var_idx] is not None and assignment[var_idx] == expected_val:
                    satisfied = True
                    break
            if not satisfied:
                all_satisfied = False
                break
                
        if all_satisfied:
            return True
            
        # Choose an unassigned variable
        for i in range(n):
            if assignment[i] is None:
                var_idx = i
                break
        else:
            return False  # All assigned but not all clauses satisfied
            
        # Try True
        old_assignment = assignment[var_idx]
        assignment[var_idx] = 1
        if dpll():
            return True
            
        # Try False
        assignment[var_idx] = 0
        if dpll():
            return True
            
        # Backtrack
        assignment[var_idx] = old_assignment
        return False
    
    if dpll():
        return [0 if x is None else x for x in assignment]
    else:
        return None

def main():
    n, m, rules = read_input()
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
        
    result = solve_2sat(n, rules)
    if result is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, result)))

if __name__ == "__main__":
    main()
