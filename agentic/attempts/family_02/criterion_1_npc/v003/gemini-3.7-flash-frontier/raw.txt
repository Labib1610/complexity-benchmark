import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve_sat():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Simplify clause: remove duplicate literals
        # If clause contains both x and -x, it is always satisfied (can be ignored)
        s = set(c)
        tautology = False
        for lit in s:
            if -lit in s:
                tautology = True
                break
        if not tautology:
            clauses.append(list(s))

    assignment = {}

    def unit_propagate(current_clauses, current_assignment):
        assignment = dict(current_assignment)
        clauses = [list(c) for c in current_clauses]
        
        while True:
            unit_found = False
            for c in clauses:
                if len(c) == 0:
                    return None, None
                if len(c) == 1:
                    lit = c[0]
                    var = abs(lit)
                    val = 1 if lit > 0 else 0
                    if var in assignment:
                        if assignment[var] != val:
                            return None, None
                    else:
                        assignment[var] = val
                    unit_found = True
                    # Simplify clauses with this literal
                    new_clauses = []
                    for clause in clauses:
                        if lit in clause:
                            continue
                        if -lit in clause:
                            new_clause = [x for x in clause if x != -lit]
                            if len(new_clause) == 0:
                                return None, None
                            new_clauses.append(new_clause)
                        else:
                            new_clauses.append(clause)
                    clauses = new_clauses
                    break
            if not unit_found:
                break
        return clauses, assignment

    # Initial unit propagation
    simplified_clauses = []
    for c in clauses:
        simplified_clauses.append(list(c))
    
    clauses, assignment = unit_propagate(simplified_clauses, {})
    if clauses is None:
        print("NO")
        return

    def dpll(clauses, assignment):
        clauses, assignment = unit_propagate(clauses, assignment)
        if clauses is None:
            return None
        if len(clauses) == 0:
            return assignment

        # Jeroslow-Wang Heuristic
        scores = {}
        for c in clauses:
            weight = 2.0 ** (-len(c))
            for lit in c:
                scores[lit] = scores.get(lit, 0.0) + weight

        best_var = None
        best_lit = None
        max_score = -1.0
        
        for lit, score in scores.items():
            var = abs(lit)
            var_score = scores.get(var, 0.0) + scores.get(-var, 0.0)
            if var_score > max_score:
                max_score = var_score
                best_var = var
                best_lit = var if scores.get(var, 0.0) >= scores.get(-var, 0.0) else -var

        if best_lit is None:
            return assignment

        # Try best_lit first
        res = dpll(clauses + [[best_lit]], assignment)
        if res is not None:
            return res
        return dpll(clauses + [[-best_lit]], assignment)

    res = dpll(clauses, assignment)
    if res is None:
        print("NO")
    else:
        print("YES")
        ans = []
        for i in range(1, n + 1):
            ans.append(res.get(i, 0))
        print(*(ans))

if __name__ == '__main__':
    solve_sat()
