import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # Variable mapping for 2-SAT
    # Each station has two nodes: id*2 (false, mode=0) and id*2+1 (true, mode=1)
    total_nodes = n * 2
    
    adj = [[] for _ in range(total_nodes)]
    
    def get_node(var, val):
        # var is 1-indexed station number
        # val is 1 for alkaline, 0 for acidic
        # returns node index (0-indexed)
        return (var - 1) * 2 + val
    
    def add_implication(u, v):
        adj[u].append(v)
    
    # Build implication graph
    for l1, l2, l3 in clauses:
        vars_in_clause = set()
        vals_in_clause = set()
        
        # Each behavior l requires station |l| to have value (1 if l>0 else 0)
        var1, val1 = abs(l1), 1 if l1 > 0 else 0
        var2, val2 = abs(l2), 1 if l2 > 0 else 0
        var3, val3 = abs(l3), 1 if l3 > 0 else 0
        
        behaviors = [(var1, val1), (var2, val2), (var3, val3)]
        
        for var, val in behaviors:
            vars_in_clause.add(var)
            if val == 1:
                vals_in_clause.add(var)  # alkaline
            else:
                vals_in_clause.add(-var)  # acidic
        
        # If only one variable, must match at least one behavior
        if len(vars_in_clause) == 1:
            var = var1
            possible_vals = set()
            for _, val in behaviors:
                possible_vals.add(val)
            if len(possible_vals) == 1:
                # Only one value possible
                val = possible_vals.pop()
                u = get_node(var, 1 - val)  # not chosen value
                v = get_node(var, val)      # must be chosen value
                add_implication(u, v)
            # else: both values possible, no constraint
        elif len(vars_in_clause) == 2:
            # Two variables, add implications
            # If both behaviors have same variable with different values, that variable is unconstrained
            # Otherwise, add implication
            b0_var, b0_val = behaviors[0]
            b1_var, b1_val = behaviors[1]
            b2_var, b2_val = behaviors[2]
            
            # Count occurrences
            from collections import Counter
            var_count = Counter([b0_var, b1_var, b2_var])
            
            if var_count[var1] == 2:
                # var1 appears twice, var2 appears once
                # var1 must be able to take at least one of the values it's assigned twice
                vals_for_var1 = set()
                for var, val in behaviors:
                    if var == var1:
                        vals_for_var1.add(val)
                if len(vals_for_var1) == 1:
                    # var1 is fixed to this value
                    val = vals_for_var1.pop()
                    add_implication(get_node(var1, 1 - val), get_node(var1, val))
                # Find which behavior has var2
                for var, val in behaviors:
                    if var == var2:
                        val2_actual = val
                        break
                # If var1 takes the value not in vals_for_var1 (or either if both), var2 must follow
                # Simplify: if var1 is forced away from critical state
                if len(vals_for_var1) == 1:
                    # var1 fixed, var2 must match remaining behavior
                    pass  # already handled above
                elif len(vals_for_var1) == 2:
                    # var1 can be 0 or 1, no constraint on var1 itself from this
                    # But if var1 is NOT the value from the single var2 behavior's setting, 
                    # then we need var2 to match its behavior
                    pass  # Actually this gives no constraint since var1 has freedom
                
                # Better approach: go through all pairs
                beh_pairs = [(behaviors[0], behaviors[1]), (behaviors[0], behaviors[2]), (behaviors[1], behaviors[2])]
                for a, b in beh_pairs:
                    va, vl_a = a
                    vb, vl_b = b
                    if va != vb:
                        # If var va is NOT vl_a, then var vb must be vl_b (to satisfy this pair)
                        add_implication(get_node(va, 1 - vl_a), get_node(vb, vl_b))
            else:
                # var2 appears twice, var1 appears once (symmetric)
                vals_for_var2 = set()
                for var, val in behaviors:
                    if var == var2:
                        vals_for_var2.add(val)
                if len(vals_for_var2) == 1:
                    val = vals_for_var2.pop()
                    add_implication(get_node(var2, 1 - val), get_node(var2, val))
                
                beh_pairs = [(behaviors[0], behaviors[1]), (behaviors[0], behaviors[2]), (behaviors[1], behaviors[2])]
                for a, b in beh_pairs:
                    va, vl_a = a
                    vb, vl_b = b
                    if va != vb:
                        add_implication(get_node(va, 1 - vl_a), get_node(vb, vl_b))
        else:
            # Three different variables
            # For each pair, add implication
            beh_pairs = [(behaviors[0], behaviors[1]), (behaviors[0], behaviors[2]), (behaviors[1], behaviors[2])]
            for a, b in beh_pairs:
                va, vl_a = a
                vb, vl_b = b
                add_implication(get_node(va, 1 - vl_a), get_node(vb, vl_b))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * total_nodes
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(total_nodes):
        if not visited[i]:
            dfs1(i)
    
    radj = [[] for _ in range(total_nodes)]
    for u in range(total_nodes):
        for v in adj[u]:
            radj[v].append(u)
    
    comp_id = [-1] * total_nodes
    current_comp = 0
    
    def dfs2(u, cid):
        comp_id[u] = cid
        for v in radj[u]:
            if comp_id[v] == -1:
                dfs2(v, cid)
    
    for u in reversed(order):
        if comp_id[u] == -1:
            dfs2(u, current_comp)
            current_comp += 1
    
    # Check if any variable and its negation are in the same SCC
    for i in range(n):
        if comp_id[i * 2] == comp_id[i * 2 + 1]:
            print("NO")
            return
    
    print("YES")
    
    # Construct assignment
    assignment = []
    for i in range(n):
        # Choose the value with higher (or lower) component id
        # In 2-SAT, we choose: if comp_id[false] < comp_id[true], choose true
        if comp_id[i * 2] < comp_id[i * 2 + 1]:
            assignment.append(1)
        else:
            assignment.append(0)
    
    print(' '.join(map(str, assignment)))

solve()
