import sys

# Set recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Remove duplicate literals in the same clause
        c = list(set(c))
        # If clause contains both x and -x, it's always satisfied, so ignore it
        has_tautology = False
        for lit in c:
            if -lit in c:
                has_tautology = True
                break
        if not has_tautology:
            clauses.append(c)

    def unit_propagate(clauses, assignment):
        assignment = dict(assignment)
        while True:
            unit_clauses = [c for c in clauses if len(c) == 1]
            if not unit_clauses:
                break
            for c in unit_clauses:
                unit = c[0]
                var = abs(unit)
                val = 1 if unit > 0 else 0
                if var in assignment:
                    if assignment[var] != val:
                        return None, None
                else:
                    assignment[var] = val
            
            new_clauses = []
            for c in clauses:
                satisfied = False
                new_c = []
                for lit in c:
                    var = abs(lit)
                    if var in assignment:
                        val = assignment[var]
                        if (lit > 0 and val == 1) or (lit < 0 and val == 0):
                            satisfied = True
                            break
                    else:
                        new_c.append(lit)
                if satisfied:
                    continue
                if not new_c:
                    return None, None
                new_clauses.append(new_c)
            clauses = new_clauses
        return clauses, assignment

    def dpll(clauses, assignment):
        clauses, assignment = unit_propagate(clauses, assignment)
        if clauses is None:
            return None
        if not clauses:
            return assignment

        # Jeroslow-Wang Heuristic
        jw_pos = {}
        jw_neg = {}
        for c in clauses:
            weight = 2.0 ** (-len(c))
            for lit in c:
                var = abs(lit)
                if lit > 0:
                    jw_pos[var] = jw_pos.get(var, 0.0) + weight
                else:
                    jw_neg[var] = jw_neg.get(var, 0.0) + weight

        all_vars = set(jw_pos.keys()) | set(jw_neg.keys())
        best_var = max(all_vars, key=lambda v: jw_pos.get(v, 0.0) + jw_neg.get(v, 0.0))
        
        preferred_val = 1 if jw_pos.get(best_var, 0.0) >= jw_neg.get(best_var, 0.0) else 0

        for val in [preferred_val, 1 - preferred_val]:
            lit = best_var if val == 1 else -best_var
            new_assignment = dict(assignment)
            new_assignment[best_var] = val
            res = dpll(clauses + [[lit]], new_assignment)
            if res is not None:
                return res

        return None

    res = dpll(clauses, {})
    if res is None:
        print("NO")
    else:
        print("YES")
        ans = [res.get(i, 0) for i in range(1, n + 1)]
        print(*(ans))

if __name__ == '__main__':
    solve()
