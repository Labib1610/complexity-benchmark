def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    clauses = []
    index = 2
    for _ in range(m):
        clause = []
        for _ in range(3):
            lit = int(data[index])
            clause.append(lit)
            index += 1
        clauses.append(clause)
    
    # For each variable, we'll try both values (0 and 1)
    # We can model this as a 2-SAT or 3-SAT problem, but since n can be up to 90,
    # and m up to 400, brute force might be feasible for small cases,
    # but let's implement a proper solution using implication graph and SCC.
    
    # However, given the constraints, let's use a SAT solver approach.
    # But since we cannot use external libraries, we'll implement a simple backtracking solution.
    
    # Since n <= 90, and 2^90 is too large, we need a smarter approach.
    # Let's try to implement a 3-SAT solver with unit propagation and pure literal elimination.
    
    # Actually, let's think differently. Since this is a 3-SAT problem,
    # and the constraints are not too high (m<=400), we can try a randomized or heuristic approach.
    # But the most reliable way without external libraries is to implement a simple DPLL-style algorithm.
    
    # Let's proceed with a backtracking solution with some optimizations.
    
    # Map literals to indices for easier handling
    # We'll represent variables from 1 to n, and their negations as negative numbers
    
    def is_satisfied(assignment):
        for clause in clauses:
            satisfied = False
            for lit in clause:
                var = abs(lit)
                sign = 1 if lit > 0 else 0
                if var in assignment:
                    if assignment[var] == sign:
                        satisfied = True
                        break
                else:
                    # Variable not assigned yet, so clause could still be satisfied
                    pass
            if not satisfied:
                # Check if all literals in the clause have been assigned and none satisfy it
                all_assigned = True
                for lit in clause:
                    var = abs(lit)
                    if var not in assignment:
                        all_assigned = False
                        break
                if all_assigned:
                    return False
        return True
    
    def is_clause_satisfied(clause, assignment):
        for lit in clause:
            var = abs(lit)
            sign = 1 if lit > 0 else 0
            if var in assignment and assignment[var] == sign:
                return True
        return False
    
    def find_unsatisfied_clause(assignment):
        for clause in clauses:
            if not is_clause_satisfied(clause, assignment):
                # Check if all variables in clause are assigned
                all_assigned = True
                for lit in clause:
                    var = abs(lit)
                    if var not in assignment:
                        all_assigned = False
                        break
                if all_assigned:
                    return clause
        return None
    
    def backtrack(assignment):
        # Unit propagation
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned_count = 0
                unassigned_lit = None
                satisfied = False
                for lit in clause:
                    var = abs(lit)
                    sign = 1 if lit > 0 else 0
                    if var in assignment:
                        if assignment[var] == sign:
                            satisfied = True
                            break
                    else:
                        unassigned_count += 1
                        unassigned_lit = lit
                
                if satisfied:
                    continue
                
                if unassigned_count == 0:
                    return False  # Conflict
                
                if unassigned_count == 1:
                    var = abs(unassigned_lit)
                    sign = 1 if unassigned_lit > 0 else 0
                    assignment[var] = sign
                    changed = True
        
        # Pure literal elimination would be complex, so we skip it for now
        
        # Find the first unassigned variable
        for i in range(1, n+1):
            if i not in assignment:
                # Try assigning it to 1
                assignment[i] = 1
                result = backtrack(assignment)
                if result:
                    return result
                # Backtrack
                del assignment[i]
                
                # Try assigning it to 0
                assignment[i] = 0
                result = backtrack(assignment)
                if result:
                    return result
                # Backtrack
                del assignment[i]
                return False
        
        # All variables assigned and no conflicts found so far
        # Check if all clauses are satisfied
        for clause in clauses:
            if not is_clause_satisfied(clause, assignment):
                return False
        return assignment.copy()
    
    # Start with empty assignment
    result = backtrack({})
    
    if result:
        print("YES")
        ans = []
        for i in range(1, n+1):
            ans.append(str(result.get(i, 0)))  # Default to 0 if not in assignment (should not happen)
        print(" ".join(ans))
    else:
        print("NO")

solve()
