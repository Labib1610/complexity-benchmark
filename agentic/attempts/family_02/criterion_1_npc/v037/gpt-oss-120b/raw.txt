```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        c = (int(next(it)), int(next(it)), int(next(it)))
        clauses.append(c)

    # assignment: None=unassigned, 0=OFF,1=ON
    assign = [None] * (n + 1)

    # helper to get literal value under current assignment
    def lit_val(lit):
        v = abs(lit)
        a = assign[v]
        if a is None:
            return None
        return a if lit > 0 else 1 - a

    # unit propagation
    def propagate():
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                # evaluate clause
                vals = [lit_val(l) for l in clause]
                if any(v == 1 for v in vals):
                    continue  # already satisfied
                # count unassigned
                unassigned = [l for l, v in zip(clause, vals) if v is None]
                if not unassigned:
                    # all false -> conflict
                    return False
                if len(unassigned) == 1:
                    # unit clause, must set to make it true
                    lit = unassigned[0]
                    var = abs(lit)
                    needed = 1 if lit > 0 else 0
                    if assign[var] is None:
                        assign[var] = needed
                        changed = True
                    elif assign[var] != needed:
                        return False
        return True

    # recursive DPLL
    def dpll():
        if not propagate():
            return False
        # check if all variables assigned
        if all(v is not None for v in assign[1:]):
            return True
        # choose unassigned variable (simple heuristic)
        var = next(i for i in range(1, n + 1) if assign[i] is None)
        # try true then false
        saved = assign[:]
        assign[var] = 1
        if dpll():
            return True
        assign[:] = saved
        assign[var] = 0
        if dpll():
            return True
        assign[:] = saved
        return False

    if dpll():
        print("YES")
        print(' '.join(str(assign[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
