We can model each flag as a Boolean variable and each module as a clause of three literals.  
The condition “at least one of the three must be true” is exactly a 3-SAT clause.  
So the problem reduces to checking satisfiability of a 3-SAT instance with up to 90 variables and 400 clauses.

Since n ≤ 90, we can use a backtracking search with unit propagation and pure literal elimination, which is efficient enough for this size.

Here’s the Python solution:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        l3 = int(next(it))
        clauses.append((l1, l2, l3))
    
    # Variable assignment: 0 = unassigned, 1 = True, -1 = False
    assignment = [0] * (n + 1)
    
    def propagate(assignment):
        changed = True
        while changed:
            changed = False
            for clause in clauses:
                unassigned = []
                satisfied = False
                for lit in clause:
                    var = abs(lit)
                    val = assignment[var]
                    if val == 0:
                        unassigned.append(lit)
                    elif (lit > 0 and val == 1) or (lit < 0 and val == -1):
                        satisfied = True
                        break
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return False
                if len(unassigned) == 1:
                    lit = unassigned[0]
                    var = abs(lit)
                    if lit > 0:
                        assignment[var] = 1
                    else:
                        assignment[var] = -1
                    changed = True
        return True
    
    def pure_literal_elimination(assignment):
        # Count occurrences of each literal
        pos_count = {}
        neg_count = {}
        for clause in clauses:
            for lit in clause:
                var = abs(lit)
                if assignment[var] != 0:
                    continue
                if lit > 0:
                    pos_count[var] = pos_count.get(var, 0) + 1
                else:
                    neg_count[var] = neg_count.get(var, 0) + 1
        for var in range(1, n + 1):
            if assignment[var] != 0:
                continue
            p = pos_count.get(var, 0)
            q = neg_count.get(var, 0)
            if p > 0 and q == 0:
                assignment[var] = 1
            elif q > 0 and p == 0:
                assignment[var] = -1
    
    def backtrack(assignment):
        # Propagate
        if not propagate(assignment):
            return False
        # Pure literal elimination
        pure_literal_elimination(assignment)
        # Find an unassigned variable
        var = -1
        for i in range(1, n + 1):
            if assignment[i] == 0:
                var = i
                break
        if var == -1:
            # All assigned, check all clauses
            for clause in clauses:
                ok = False
                for lit in clause:
                    v = abs(lit)
                    val = assignment[v]
                    if (lit > 0 and val == 1) or (lit < 0 and val == -1):
                        ok = True
                        break
                if not ok:
                    return False
            return True
        # Try True
        assignment[var] = 1
        if backtrack(assignment):
            return True
        assignment[var] = -1
        if backtrack(assignment):
            return True
        assignment[var] = 0
        return False
    
    if backtrack(assignment):
        print("YES")
        print(" ".join("1" if assignment[i] == 1 else "0" for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
