import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    clauses = []
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        c = int(next(it))
        clauses.append((a, b, c))

    # assignment: -1 unassigned, 0 false, 1 true
    assign = [-1] * (n + 1)

    # For each variable, keep list of clause indices where it appears positively/negatively
    pos_occ = [[] for _ in range(n + 1)]
    neg_occ = [[] for _ in range(n + 1)]
    for idx, (a, b, c) in enumerate(clauses):
        for lit in (a, b, c):
            var = abs(lit)
            if lit > 0:
                pos_occ[var].append(idx)
            else:
                neg_occ[var].append(idx)

    # Helper to get literal value under current assignment
    def lit_val(lit):
        v = assign[abs(lit)]
        if v == -1:
            return None
        return v if lit > 0 else 1 - v

    # Unit propagation; returns False if conflict
    def propagate(stack):
        while stack:
            var = stack.pop()
            val = assign[var]
            # check clauses where this literal could satisfy or become unit/false
            for idx in (pos_occ[var] if val == 1 else neg_occ[var]):
                # clause already satisfied? skip
                a, b, c = clauses[idx]
                if any(lit_val(l) == 1 for l in (a, b, c)):
                    continue
                # count unassigned literals
                unassigned = []
                for l in (a, b, c):
                    v = lit_val(l)
                    if v is None:
                        unassigned.append(l)
                if not unassigned:  # all false -> conflict
                    return False
                if len(unassigned) == 1:
                    # force this literal to be true
                    l = unassigned[0]
                    var2 = abs(l)
                    needed = 1 if l > 0 else 0
                    if assign[var2] == -1:
                        assign[var2] = needed
                        stack.append(var2)
                    elif assign[var2] != needed:
                        return False
        return True

    # Recursive DPLL
    def dpll():
        # find any clause that is unsatisfied or unit
        # first, check for conflict
        for a, b, c in clauses:
            vals = [lit_val(a), lit_val(b), lit_val(c)]
            if any(v == 1 for v in vals):
                continue
            if all(v == 0 for v in vals):
                return False
        # check if all variables assigned
        if all(v != -1 for v in assign[1:]):
            return True
        # choose variable (simple heuristic: first unassigned)
        var = next(i for i in range(1, n + 1) if assign[i] == -1)
        # try true then false
        saved = assign[:]
        for val in (1, 0):
            assign[var] = val
            stack = [var]
            if propagate(stack):
                if dpll():
                    return True
            # backtrack
            assign[:] = saved[:]
        return False

    if dpll():
        print("YES")
        print(' '.join(str(assign[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
