import sys

def solve_3sat():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Remove duplicate literals in the same clause
        c = list(set(c))
        # If clause contains both x and -x, it is trivially true, ignore it
        tautology = False
        seen = set()
        for lit in c:
            if -lit in seen:
                tautology = True
                break
            seen.add(lit)
        if not tautology:
            clauses.append(c)
            
    # Check if empty clause exists
    if any(len(c) == 0 for c in clauses):
        print("NO")
        return

    assignment = {} # var -> bool

    def unit_propagate(clauses, assignment):
        assignment = assignment.copy()
        while True:
            unit_clauses = []
            for c in clauses:
                unassigned = []
                satisfied = False
                for lit in c:
                    var = abs(lit)
                    val = assignment.get(var)
                    if val is not None:
                        if (lit > 0 and val) or (lit < 0 and not val):
                            satisfied = True
                            break
                    else:
                        unassigned.append(lit)
                if satisfied:
                    continue
                if len(unassigned) == 0:
                    return None, None # Conflict
                if len(unassigned) == 1:
                    unit_clauses.append(unassigned[0])
            
            if not unit_clauses:
                break
            
            for lit in unit_clauses:
                var = abs(lit)
                val = (lit > 0)
                if var in assignment:
                    if assignment[var] != val:
                        return None, None
                else:
                    assignment[var] = val
                    
        # Filter clauses
        new_clauses = []
        for c in clauses:
            satisfied = False
            unassigned = []
            for lit in c:
                var = abs(lit)
                val = assignment.get(var)
                if val is not None:
                    if (lit > 0 and val) or (lit < 0 and not val):
                        satisfied = True
                        break
                else:
                    unassigned.append(lit)
            if not satisfied:
                new_clauses.append(unassigned)
        return new_clauses, assignment

    def dpll(clauses, assignment):
        clauses, assignment = unit_propagate(clauses, assignment)
        if clauses is None:
            return None
        if len(clauses) == 0:
            return assignment
        
        # Choose literal with maximum occurrences (MOM's / most frequent)
        counts = {}
        for c in clauses:
            for lit in c:
                counts[lit] = counts.get(lit, 0) + 1
        
        # Pick literal that appears most frequently
        best_lit = max(counts, key=counts.get)
        var = abs(best_lit)
        pref_val = (best_lit > 0)
        
        # Try pref_val first
        assign1 = assignment.copy()
        assign1[var] = pref_val
        res = dpll(clauses, assign1)
        if res is not None:
            return res
        
        # Try opposite
        assign2 = assignment.copy()
        assign2[var] = not pref_val
        return dpll(clauses, assign2)

    res = dpll(clauses, {})
    if res is None:
        print("NO")
    else:
        print("YES")
        out = []
        for i in range(1, n + 1):
            val = res.get(i, False) # default to 0 (False)
            out.append("1" if val else "0")
        print(" ".join(out))

if __name__ == '__main__':
    solve_3sat()
