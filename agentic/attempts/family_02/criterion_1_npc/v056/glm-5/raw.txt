import sys
input = sys.stdin.readline

def solve():
    n, m = map(int, input().split())
    
    # Each clause is (actuator_index, required_value)
    # clause[0], clause[1], clause[2] are the three literals
    clauses = []
    for _ in range(m):
        parts = list(map(int, input().split()))
        clause = []
        for x in parts:
            idx = abs(x) - 1
            val = 1 if x > 0 else 0
            clause.append((idx, val))
        clauses.append(clause)
    
    assignment = [-1] * n
    
    def can_satisfy(start_clause_idx):
        """
        Check if remaining clauses can be satisfied using two SAT solvers.
        Returns True if satisfiable (solution stored in 'assignment'),
        False otherwise and restores assignment values upon backtracking.
        """
        changed_indices = []
        
        # Iterate through all clauses to find unit clauses
        progress = True
        while progress:
            progress = False
            for ci in range(start_clause_idx, m):
                clause = clauses[ci]
                first_unset = -1
                satisfied = False
                unset_count = 0
                
                for idx, val in clause:
                    if assignment[idx] == -1:
                        unset_count += 1
                        if first_unset == -1:
                            first_unset = (idx, val)
                    elif assignment[idx] == val:
                        satisfied = True
                        break
                
                if satisfied:
                    continue
                
                if unset_count == 0:
                    # Conflict: restore and return
                    for i in changed_indices:
                        assignment[i] = -1
                    return False
                
                if unset_count == 1:
                    # Unit clause: set the remaining variable
                    idx, val = first_unset
                    assignment[idx] = val
                    changed_indices.append(idx)
                    progress = True
        
        # Find first unset variable
        unset_var = -1
        for i in range(n):
            if assignment[i] == -1:
                unset_var = i
                break
        
        if unset_var == -1:
            return True
        
        # Try both values for the unset variable
        for val_to_try in [0, 1]:
            assignment[unset_var] = val_to_try
            if can_satisfy(start_clause_idx):
                return True
        
        # Backtrack
        assignment[unset_var] = -1
        for i in changed_indices:
            assignment[i] = -1
        return False
    
    if can_satisfy(0):
        # Fill remaining unset variables with 0
        for i in range(n):
            if assignment[i] == -1:
                assignment[i] = 0
        print("YES")
        print(' '.join(map(str, assignment)))
    else:
        print("NO")

solve()
