import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    raw_clauses = []
    idx = 2
    for _ in range(m):
        c = [int(input_data[idx]), int(input_data[idx+1]), int(input_data[idx+2])]
        idx += 3
        # Remove duplicates
        c = list(set(c))
        # Check for tautology (e.g. x and -x in same clause)
        has_tautology = False
        pos = set()
        for lit in c:
            if -lit in pos:
                has_tautology = True
                break
            pos.add(lit)
        if not has_tautology:
            raw_clauses.append(c)

    # 3-SAT Solver using DPLL with Jeroslow-Wang heuristic and two-watched literals
    # Literals: 1..n are positive, -1..-n are negative.
    # We map lit to internal index: lit > 0 -> 2*(lit-1), lit < 0 -> 2*(-lit-1) + 1
    def lit_to_idx(lit):
        if lit > 0:
            return 2 * (lit - 1)
        else:
            return 2 * (-lit - 1) + 1

    def not_lit(lit_idx):
        return lit_idx ^ 1

    # Assignment: -1 = unassigned, 1 = true, 0 = false for each variable 0..n-1
    assign = [-1] * n

    # Clauses represented as list of literal indices
    clauses = []
    for c in raw_clauses:
        clauses.append([lit_to_idx(lit) for lit in c])

    # Watch lists: for each literal idx, list of clause indices it watches
    watches = [[] for _ in range(2 * n)]

    # Handle unit clauses initially
    unit_queue = []
    
    for ci, c in enumerate(clauses):
        if len(c) == 0:
            print("NO")
            return
        elif len(c) == 1:
            unit_queue.append((c[0], None))
        else:
            watches[c[0]].append(ci)
            watches[c[1]].append(ci)

    # DPLL with Trail
    trail = []
    trail_lim = []

    def lit_val(l_idx):
        v = l_idx >> 1
        sign = l_idx & 1
        val = assign[v]
        if val == -1:
            return -1
        return 1 if (val ^ sign) == 1 else 0

    def assign_lit(l_idx):
        v = l_idx >> 1
        sign = l_idx & 1
        val = 1 if sign == 0 else 0
        assign[v] = val
        trail.append(l_idx)

    def undo_one():
        l_idx = trail.pop()
        v = l_idx >> 1
        assign[v] = -1

    def propagate():
        p = len(trail) - len(prop_queue)
        while prop_queue:
            l_idx = prop_queue.pop(0)
            nl_idx = not_lit(l_idx)
            
            # nl_idx has become false, so clauses watching nl_idx need updating
            i = 0
            watch_list = watches[nl_idx]
            while i < len(watch_list):
                ci = watch_list[i]
                c = clauses[ci]
                
                # Make sure nl_idx is at c[1]
                if c[0] == nl_idx:
                    c[0], c[1] = c[1], c[0]
                
                # If first watched literal is already true, clause is satisfied
                if lit_val(c[0]) == 1:
                    i += 1
                    continue
                
                # Look for a new literal to watch
                found = False
                for k in range(2, len(c)):
                    if lit_val(c[k]) != 0:
                        c[1], c[k] = c[k], c[1]
                        watches[c[1]].append(ci)
                        watch_list[i] = watch_list[-1]
                        watch_list.pop()
                        found = True
                        break
                
                if found:
                    continue
                
                # No new literal found. Check c[0]
                val0 = lit_val(c[0])
                if val0 == 0:
                    # Conflict!
                    return False
                elif val0 == -1:
                    # Unit propagation
                    assign_lit(c[0])
                    prop_queue.append(c[0])
                i += 1
        return True

    prop_queue = []
    # Initial unit propagation
    for l_idx, _ in unit_queue:
        val = lit_val(l_idx)
        if val == 0:
            print("NO")
            return
        elif val == -1:
            assign_lit(l_idx)
            prop_queue.append(l_idx)

    if not propagate():
        print("NO")
        return

    def pick_branch_lit():
        # Two-sided Jeroslow-Wang heuristic
        # Compute weight for each unassigned literal
        # J(l) = sum_{c: l in c, c not sat} 2^(-len_unassigned)
        scores = [0.0] * (2 * n)
        for c in clauses:
            sat = False
            unassigned_count = 0
            for l in c:
                v = lit_val(l)
                if v == 1:
                    sat = True
                    break
                elif v == -1:
                    unassigned_count += 1
            if not sat and unassigned_count > 0:
                w = 2.0 ** (-unassigned_count)
                for l in c:
                    if lit_val(l) == -1:
                        scores[l] += w
        
        best_v = -1
        best_score = -1.0
        for v in range(n):
            if assign[v] == -1:
                pos_l = 2 * v
                neg_l = 2 * v + 1
                s = scores[pos_l] + scores[neg_l]
                if s > best_score:
                    best_score = s
                    best_v = v
        if best_v == -1:
            return -1
        
        pos_l = 2 * best_v
        neg_l = 2 * best_v + 1
        return pos_l if scores[pos_l] >= scores[neg_l] else neg_l

    def dpll():
        next_lit = pick_branch_lit()
        if next_lit == -1:
            return True
        
        # Try best polarity first, then opposite
        for lit in [next_lit, not_lit(next_lit)]:
            trail_mark = len(trail)
            assign_lit(lit)
            prop_queue.append(lit)
            
            if propagate():
                if dpll():
                    return True
            
            # Backtrack
            prop_queue.clear()
            while len(trail) > trail_mark:
                undo_one()
                
        return False

    if dpll():
        print("YES")
        # Any remaining unassigned variables can be set to 0
        res = [assign[i] if assign[i] != -1 else 0 for i in range(n)]
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
