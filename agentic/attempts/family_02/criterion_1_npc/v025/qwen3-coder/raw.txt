from collections import defaultdict
import sys

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    clauses = []
    for _ in range(m):
        clause = list(map(int, sys.stdin.readline().split()))
        clauses.append(clause)
    return n, m, clauses

def solve_2sat(n, clauses):
    # Convert to 2-SAT problem
    # Each clause has 3 literals. We need to convert it to equivalent 2-SAT clauses.
    # For a clause (a OR b OR c), it's equivalent to (a OR b) AND (¬(a OR b) OR c)
    # Which is (a OR b) AND (¬a OR c) AND (¬b OR c)
    
    # But let's think differently. We can enumerate all possibilities for small n
    # However, n can be up to 90, so we need a better approach.
    
    # Let's model this as a graph problem for 2-SAT
    # But since we have 3-SAT clauses, we'll convert them to 2-SAT
    
    # Actually, let's use the standard reduction:
    # (a OR b OR c) is equivalent to 
    # (a OR b OR y) AND (¬y OR c) where y is a new variable
    # But this introduces new variables...
    
    # Let's try a different approach: Use the fact that this is essentially a MAX-SAT
    # But we need ALL clauses to be satisfied, so it's a decision problem.
    
    # Let's implement a backtracking solution with pruning, but that might be too slow.
    
    # Better approach: Convert to implication graph for 2-SAT by introducing auxiliary variables
    # But that complicates things.
    
    # Let's try a randomized approach or brute force for small cases, but n=90 is too big.
    
    # Let's model this properly:
    # For each clause (a OR b OR c), we can rewrite it as:
    # (a OR b) OR c
    # This is satisfied if (a OR b) is true OR c is true
    # Which means: if both a and b are false, then c must be true
    # So we get constraints like: if NOT a AND NOT b, then c
    
    # This is getting complex. Let's use a SAT solver approach.
    # We'll reduce to 2-SAT by adding auxiliary variables.
    
    # For clause (a OR b OR c), introduce a new variable y
    # Add clauses: (a OR b OR y) AND (NOT y OR c)
    # But the first one is still 3-SAT...
    
    # Correct reduction:
    # (a OR b OR c) is equivalent to:
    # (a OR b OR y) AND (NOT y OR c)
    # Where the first one needs another reduction:
    # (a OR b OR y) = (a OR b) OR y
    # If both a,b false => y must be true
    # If a or b true => clause satisfied regardless of y
    # So add edges: NOT a -> b, NOT b -> a, NOT a -> y, NOT y -> a, NOT b -> y, NOT y -> b
    # And for (NOT y OR c): y -> c, NOT c -> NOT y
    
    # This is getting very complex. Let's look for a simpler way.
    
    # Let's just implement a direct 3-SAT solver using backtracking with optimizations.
    
    def var_to_index(v):
        # v is +i or -i
        # Returns (index, is_positive)
        if v > 0:
            return (v - 1, True)
        else:
            return (-v - 1, False)
    
    def evaluate_clause(clause, assignment):
        for lit in clause:
            idx, is_pos = var_to_index(lit)
            if idx >= len(assignment):
                continue
            val = assignment[idx]
            if val == -1:  # unassigned
                continue
            if (val == 1 and is_pos) or (val == 0 and not is_pos):
                return True
        return False
    
    def backtrack(assignment, clauses_unsat):
        # Find next unassigned variable
        var = -1
        for i in range(n):
            if assignment[i] == -1:
                var = i
                break
        
        if var == -1:
            # All variables assigned, check if all clauses are satisfied
            for clause in clauses_unsat:
                if not evaluate_clause(clause, assignment):
                    return None
            return assignment[:]
        
        # Try setting var to 1
        assignment[var] = 1
        new_clauses_unsat = []
        for clause in clauses_unsat:
            if not evaluate_clause(clause, assignment):
                new_clauses_unsat.append(clause)
        
        result = backtrack(assignment, new_clauses_unsat)
        if result is not None:
            return result
            
        # Try setting var to 0
        assignment[var] = 0
        new_clauses_unsat = []
        for clause in clauses_unsat:
            if not evaluate_clause(clause, assignment):
                new_clauses_unsat.append(clause)
                
        result = backtrack(assignment, new_clauses_unsat)
        if result is not None:
            return result
            
        # Backtrack
        assignment[var] = -1
        return None
    
    # Start with all variables unassigned
    assignment = [-1] * n
    result = backtrack(assignment, clauses)
    return result

def main():
    n, m, clauses = read_input()
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # Solve as 3-SAT
    solution = solve_2sat(n, clauses)
    
    if solution is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, solution)))

if __name__ == "__main__":
    main()
