We can model this as a **2-SAT** problem.  
Each valve has two possible states:  
- `x_i = true` means valve `i` is set to Primary (1).  
- `x_i = false` means valve `i` is set to Secondary (0).  

A condition like `(a or b)` where `a` and `b` are literals (positive or negative valve IDs) becomes a 2-SAT clause.  

We can solve 2-SAT using **implication graph + SCC (Tarjan’s or Kosaraju’s)**.  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 2*i for false, 2*i+1 for true

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    def node(lit):
        # lit > 0: variable (lit-1) is True
        # lit < 0: variable (-lit-1) is False
        var = abs(lit) - 1
        if lit > 0:
            return 2 * var + 1  # True node
        else:
            return 2 * var      # False node

    for _ in range(m):
        a, b = map(int, input().split())
        # clause: (a or b)
        # implications: !a -> b, !b -> a
        add_imp(node(-a), node(b))
        add_imp(node(-b), node(a))

    # Kosaraju's SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Build assignment
    assignment = [False] * n
    for i in range(n):
        # In Kosaraju, comp id order gives topological order of SCCs in reverse.
        # If True node has higher comp id, set True.
        assignment[i] = comp[2*i+1] > comp[2*i]

    print("YES")
    print(" ".join("1" if x else "0" for x in assignment))

if __name__ == "__main__":
    solve()
```
