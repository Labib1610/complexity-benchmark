import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT setup
    # Variables: 1 to n
    # For each variable x: 2*x is True (Primary), 2*x+1 is False (Secondary)
    # Total nodes: 2*(n+1) to use 1-based indexing for variables
    
    N = 2 * (n + 1)
    adj = [[] for _ in range(N)]
    radj = [[] for _ in range(N)]
    
    def get_node(v):
        # v is non-zero integer
        # positive: valve |v| is Primary (True) -> node 2*|v|
        # negative: valve |v| is Secondary (False) -> node 2*|v|+1
        valve = abs(v)
        if v > 0:
            return 2 * valve
        else:
            return 2 * valve + 1
    
    def neg(node):
        # Returns the negation of a node
        return node ^ 1
    
    for _ in range(m):
        v1 = int(next(it))
        v2 = int(next(it))
        a = get_node(v1)
        b = get_node(v2)
        # Clause: (v1 OR v2)
        # Implications: NOT v1 -> v2, NOT v2 -> v1
        adj[neg(a)].append(b)
        adj[neg(b)].append(a)
        radj[b].append(neg(a))
        radj[a].append(neg(b))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                neighbor = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(1, N):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * N
    current_comp = 0
    
    def dfs2(u, label):
        stack = [u]
        comp[u] = label
        while stack:
            node = stack.pop()
            for neighbor in radj[node]:
                if comp[neighbor] == -1:
                    comp[neighbor] = label
                    stack.append(neighbor)
    
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, current_comp)
            current_comp += 1
    
    # Check for contradiction: x and NOT x in same SCC
    for i in range(1, n + 1):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * (n + 1)
    # Process components in reverse topological order (higher comp number = earlier in order)
    # We need to assign True to nodes that appear earlier in topological order
    # Since we numbered components in reverse topological order, higher number = earlier
    
    # For each variable, choose the literal whose component has higher number
    for i in range(1, n + 1):
        if comp[2 * i] > comp[2 * i + 1]:
            assignment[i] = 1  # Primary
        else:
            assignment[i] = 0  # Secondary
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

solve()
