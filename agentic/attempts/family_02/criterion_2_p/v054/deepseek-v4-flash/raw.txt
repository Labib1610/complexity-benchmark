We can model each toggle as a boolean variable and each rule as a clause of two literals.  
This is exactly the **2-SAT** problem.  

We can solve it by:
1. Building an implication graph.
2. Finding strongly connected components (SCCs).
3. Checking if a variable and its negation are in the same SCC (unsatisfiable).
4. Assigning values based on topological order of SCCs.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 0..n-1 for x_i, n..2n-1 for ~x_i

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def node(var):
        # var > 0: x_var, var < 0: ~x_var
        if var > 0:
            return (var - 1) * 2
        else:
            return (-var - 1) * 2 + 1

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    for _ in range(m):
        a, b = map(int, input().split())
        # clause (a or b) => (~a -> b) and (~b -> a)
        na = node(-a)
        nb = node(-b)
        add_imp(na, node(b))
        add_imp(nb, node(a))

    # Kosaraju's algorithm
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Assign values: if comp[x] > comp[~x] then x = True
    ans = []
    for i in range(n):
        ans.append(1 if comp[2*i] > comp[2*i+1] else 0)

    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
