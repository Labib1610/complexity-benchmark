import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(500000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    # 2*n nodes: for variable i (0 to n-1):
    # 2*i is True, 2*i + 1 is False
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2

        # Convert literal to node
        if l1 > 0:
            u = 2 * (l1 - 1)
        else:
            u = 2 * (-l1 - 1) + 1

        if l2 > 0:
            v = 2 * (l2 - 1)
        else:
            v = 2 * (-l2 - 1) + 1

        # not u => v
        # not v => u
        not_u = u ^ 1
        not_v = v ^ 1

        adj[not_u].append(v)
        adj[not_v].append(u)

    # Tarjan's SCC algorithm (iterative to avoid recursion overhead/issues)
    tin = [-1] * num_nodes
    low = [-1] * num_nodes
    scc = [-1] * num_nodes
    on_stack = [False] * num_nodes
    stack = []
    timer = 0
    scc_cnt = 0

    for i in range(num_nodes):
        if tin[i] != -1:
            continue

        # Iterative DFS
        call_stack = [(i, 0)]
        tin[i] = low[i] = timer
        timer += 1
        stack.append(i)
        on_stack[i] = True

        while call_stack:
            u, edge_idx = call_stack[-1]

            if edge_idx < len(adj[u]):
                call_stack[-1] = (u, edge_idx + 1)
                to = adj[u][edge_idx]
                if tin[to] == -1:
                    tin[to] = low[to] = timer
                    timer += 1
                    stack.append(to)
                    on_stack[to] = True
                    call_stack.append((to, 0))
                elif on_stack[to]:
                    if tin[to] < low[u]:
                        low[u] = tin[to]
            else:
                call_stack.pop()
                if low[u] == tin[u]:
                    while True:
                        node = stack.pop()
                        on_stack[node] = False
                        scc[node] = scc_cnt
                        if node == u:
                            break
                    scc_cnt += 1

                if call_stack:
                    p = call_stack[-1][0]
                    if low[u] < low[p]:
                        low[p] = low[u]

    ans = [0] * n
    for i in range(n):
        if scc[2 * i] == scc[2 * i + 1]:
            print("NO")
            return
        ans[i] = 1 if scc[2 * i] < scc[2 * i + 1] else 0

    print("YES")
    print(*(ans))

if __name__ == '__main__':
    main()
