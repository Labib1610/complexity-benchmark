import sys

# Increase recursion depth for deep DFS calls
sys.setrecursionlimit(300000)

def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Each variable x has two nodes: 2x (True) and 2x+1 (False)
    # We map indices 1..n to 0..n-1 internally
    # Node for variable i being True: 2*i
    # Node for variable i being False: 2*i + 1
    
    # Adjacency list for implication graph
    adj = [[] for _ in range(2 * n)]
    adj_transpose = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        try:
            u_cond = int(next(iterator))
            v_cond = int(next(iterator))
        except StopIteration:
            break
        
        # Process condition u_cond
        # If positive x, it means x is True. Node index: 2*(x-1)
        # If negative -x, it means x is False. Node index: 2*(x-1) + 1
        if u_cond > 0:
            u_node = 2 * (u_cond - 1)
            u_neg_node = 2 * (u_cond - 1) + 1
        else:
            u_node = 2 * (-u_cond - 1) + 1
            u_neg_node = 2 * (-u_cond - 1)
            
        # Process condition v_cond
        if v_cond > 0:
            v_node = 2 * (v_cond - 1)
            v_neg_node = 2 * (v_cond - 1) + 1
        else:
            v_node = 2 * (-v_cond - 1) + 1
            v_neg_node = 2 * (-v_cond - 1)
        
        # Add implication: NOT u -> v
        adj[u_neg_node].append(v_node)
        adj_transpose[v_node].append(u_neg_node)
        
        # Add implication: NOT v -> u
        adj[v_neg_node].append(u_node)
        adj_transpose[u_node].append(v_neg_node)

    # Kosaraju's algorithm to find Strongly Connected Components
    
    # 1. First pass DFS to determine finishing order
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # 2. Second pass DFS on transposed graph in reverse order of finishing times
    visited = [False] * (2 * n)
    component_id = [0] * (2 * n)
    current_id = 0
    
    def dfs2(u, cid):
        visited[u] = True
        component_id[u] = cid
        for v in adj_transpose[u]:
            if not visited[v]:
                dfs2(v, cid)
    
    for u in reversed(order):
        if not visited[u]:
            dfs2(u, current_id)
            current_id += 1
            
    # Check for unsatisfiability: if x and NOT x are in the same SCC
    for i in range(n):
        if component_id[2*i] == component_id[2*i + 1]:
            print("NO")
            return

    # Construct the solution
    # If component_id[x] > component_id[NOT x], then x must be True
    # This works because we process SCCs in reverse topological order
    # and assign True to the component that appears later in the order.
    result = [0] * n
    for i in range(n):
        if component_id[2*i] > component_id[2*i + 1]:
            result[i] = 1
        else:
            result[i] = 0
            
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == '__main__':
    solve()
