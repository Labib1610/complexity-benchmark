import sys

# Increase recursion depth just in case, though we will use an iterative approach or efficient recursion.
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    # Map literal to node index
    # For qubit i (1-based):
    # True  (state 1) -> 2 * (i - 1)
    # False (state 0) -> 2 * (i - 1) + 1
    def get_node(l):
        if l > 0:
            return 2 * (l - 1), 2 * (l - 1) + 1
        else:
            return 2 * (-l - 1) + 1, 2 * (-l - 1)

    adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        u, not_u = get_node(l1)
        v, not_v = get_node(l2)
        
        # (l1 or l2) <=> (~l1 -> l2) and (~l2 -> l1)
        adj[not_u].append(v)
        adj[not_v].append(u)

    # Tarjan's algorithm (iterative to avoid recursion limit and overhead)
    V = 2 * n
    dfn = [-1] * V
    low = [-1] * V
    in_stack = [False] * V
    stack = []
    scc = [-1] * V
    timer = 0
    scc_cnt = 0

    for i in range(V):
        if dfn[i] != -1:
            continue

        call_stack = [(i, 0)]
        while call_stack:
            u, edge_idx = call_stack[-1]
            if edge_idx == 0:
                dfn[u] = low[u] = timer
                timer += 1
                stack.append(u)
                in_stack[u] = True

            pushed = False
            for j in range(edge_idx, len(adj[u])):
                v = adj[u][j]
                if dfn[v] == -1:
                    call_stack[-1] = (u, j + 1)
                    call_stack.append((v, 0))
                    pushed = True
                    break
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]

            if pushed:
                continue

            # Backtracking
            call_stack.pop()
            if call_stack:
                p_u = call_stack[-1][0]
                if low[u] < low[p_u]:
                    low[p_u] = low[u]

            if low[u] == dfn[u]:
                while True:
                    v = stack.pop()
                    in_stack[v] = False
                    scc[v] = scc_cnt
                    if v == u:
                        break
                scc_cnt += 1

    # Check satisfiability and assign values
    ans = [0] * n
    for i in range(n):
        u = 2 * i
        not_u = 2 * i + 1
        if scc[u] == scc[not_u]:
            print("NO")
            return
        # In Tarjan's algorithm, SCCs are produced in reverse topological order.
        # If scc[u] < scc[not_u], then u is in a later component topologically (closer to sinks),
        # so u = True is a valid assignment.
        ans[i] = 1 if scc[u] < scc[not_u] else 0

    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
