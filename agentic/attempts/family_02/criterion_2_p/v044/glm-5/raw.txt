import sys
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))

    # Build implication graph
    # For each toggle i (1-indexed): node 2*i-2 is i=FALSE, node 2*i-1 is i=TRUE
    # A literal x: if x > 0, it means toggle x is ON (TRUE), corresponds to node 2*x-1
    #             if x < 0, it means toggle -x is OFF (FALSE), corresponds to node 2*(-x)-2
    # A clause (a b) means (a OR b). We add edges ~a -> b and ~b -> a.
    
    adj = [[] for _ in range(2 * n)]
    radj = [[] for _ in range(2 * n)]

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        def get_node(x):
            if x > 0:
                return 2 * x - 1  # TRUE node
            else:
                return 2 * (-x) - 2  # FALSE node
        
        u = get_node(l1)
        v = get_node(l2)
        
        nu = u ^ 1
        nv = v ^ 1
        
        adj[nu].append(v)
        adj[nv].append(u)
        radj[v].append(nu)
        radj[u].append(nv)

    # Kosaraju's algorithm for SCCs
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    component_id = [-1] * (2 * n)
    current_id = 0
    
    def dfs2(u, cid):
        component_id[u] = cid
        for v in radj[u]:
            if component_id[v] == -1:
                dfs2(v, cid)
    
    for u in reversed(order):
        if component_id[u] == -1:
            dfs2(u, current_id)
            current_id += 1
    
    # Check for unsatisfiability
    for i in range(n):
        if component_id[2 * i] == component_id[2 * i + 1]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * n
    for i in range(n):
        if component_id[2 * i] > component_id[2 * i + 1]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(*(assignment))

solve()
