import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    num_nodes = 2 * n
    head = [-1] * num_nodes
    to = []
    nxt = []
    
    head_rev = [-1] * num_nodes
    to_rev = []
    nxt_rev = []
    
    def add_edge(u, v):
        to.append(v)
        nxt.append(head[u])
        head[u] = len(to) - 1
        
        to_rev.append(u)
        nxt_rev.append(head_rev[v])
        head_rev[v] = len(to_rev) - 1

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx+1])
        idx += 2
        
        u = 2 * (abs(l1) - 1) + (0 if l1 > 0 else 1)
        v = 2 * (abs(l2) - 1) + (0 if l2 > 0 else 1)
        
        add_edge(u ^ 1, v)
        add_edge(v ^ 1, u)
        
    order = []
    visited = [False] * num_nodes
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_ptr = [head[i]]
            visited[i] = True
            while stack:
                u = stack[-1]
                e = edge_ptr[-1]
                if e != -1:
                    edge_ptr[-1] = nxt[e]
                    v = to[e]
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
                        edge_ptr.append(head[v])
                else:
                    stack.pop()
                    edge_ptr.pop()
                    order.append(u)
                    
    scc = [-1] * num_nodes
    scc_count = 0
    
    for i in reversed(order):
        if scc[i] == -1:
            stack = [i]
            scc[i] = scc_count
            while stack:
                u = stack.pop()
                e = head_rev[u]
                while e != -1:
                    v = to_rev[e]
                    if scc[v] == -1:
                        scc[v] = scc_count
                        stack.append(v)
                    e = nxt_rev[e]
            scc_count += 1
            
    res = []
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        res.append(1 if scc[pos] > scc[neg] else 0)
        
    print("YES")
    print(*(res))

if __name__ == '__main__':
    solve()
