We can model each criterion as a 2-SAT clause:  
`(toggle = value1) OR (toggle = value2)`.

This problem is a direct 2-SAT instance. We’ll build the implication graph and run SCC to check satisfiability. If satisfiable, extract a valid assignment.

Here’s the Python implementation:

```python
import sys
sys.setrecursionlimit(300000)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # variables: 0..n-1 for true, n..2n-1 for false

    # implication graph
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    # variable encoding: index = 2*var for value 1(ON), 2*var+1 for value 0(OFF)
    def node(var, val):
        # var: toggle number (1-indexed)
        idx = var - 1
        if val == 1:
            return 2 * idx
        else:
            return 2 * idx + 1

    def neg(u):
        return u ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        # a = toggle with sign
        var1 = abs(a)
        val1 = 1 if a > 0 else 0
        var2 = abs(b)
        val2 = 1 if b > 0 else 0
        # clause: (x1=val1) OR (x2=val2)
        # implications: !(x1=val1) -> (x2=val2), !(x2=val2) -> (x1=val1)
        u = node(var1, val1)
        v = node(var2, val2)
        add_imp(neg(u), v)
        add_imp(neg(v), u)

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for w in g[u]:
            if not visited[w]:
                dfs1(w)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for w in gr[u]:
            if comp[w] == -1:
                dfs2(w, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # check unsatisfiability: var and its negation in same SCC
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # assignment: pick comp with larger index => true (standard trick)
    assignment = [0] * n
    for i in range(n):
        # comp[2*i] = true for ON, comp[2*i+1] = true for OFF
        # If comp[2*i] > comp[2*i+1], assign true (1)
        assignment[i] = 1 if comp[2*i] > comp[2*i+1] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
