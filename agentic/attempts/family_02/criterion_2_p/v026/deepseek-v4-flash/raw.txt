We can model each requirement as a 2-SAT clause.  
Each component `i` has two possible literals:  
- `x` means component `|x|` is active (true).  
- `-x` means component `|x|` is standby (false).  

A clause `(l1 OR l2)` must be satisfied.  

We can solve 2-SAT using implication graph and SCC (Tarjan or Kosaraju).  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(10**6)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # each variable has two nodes: 2*i for false, 2*i+1 for true

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def node(var):
        # var > 0: true literal, var < 0: false literal
        idx = abs(var) - 1
        if var > 0:
            return 2 * idx + 1
        else:
            return 2 * idx

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    for _ in range(m):
        a, b = map(int, input().split())
        # (a OR b) => (!a -> b) and (!b -> a)
        na = node(-a)
        nb = node(b)
        add_imp(na, nb)
        nb2 = node(-b)
        na2 = node(a)
        add_imp(nb2, na2)

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # If true node has higher comp id, set true
        assignment[i] = 1 if comp[2*i+1] > comp[2*i] else 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
