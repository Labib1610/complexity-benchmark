We can model this as a **2-SAT** problem.  
Each regulator `x` gives two literals:  
- `x` means regulator `x` is active (1).  
- `-x` means regulator `x` is dormant (0).  

Each channel `(l1, l2)` is a clause:  
`(l1 OR l2)` must be true.  

We can solve 2-SAT using implication graph and SCC (Tarjan or Kosaraju).  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(10**6)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 0..n-1 for x, n..2n-1 for ~x

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def node(lit):
        # lit > 0: x -> index lit-1
        # lit < 0: ~x -> index n + (-lit)-1
        if lit > 0:
            return lit - 1
        else:
            return n + (-lit) - 1

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    for _ in range(m):
        a, b = map(int, input().split())
        # (a OR b) => (~a -> b) and (~b -> a)
        add_imp(node(-a), node(b))
        add_imp(node(-b), node(a))

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[i] == comp[i + n]:
            print("NO")
            return

    # Build assignment: choose value based on SCC order
    assignment = [0] * n
    for i in range(n):
        # If comp[i] > comp[i+n], set true (1), else false (0)
        assignment[i] = 1 if comp[i] > comp[i + n] else 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
