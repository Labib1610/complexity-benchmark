import sys
from sys import setrecursionlimit

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    N = 2 * n + 2
    head = [-1] * N
    to = []
    nxt = []
    
    def add_edge(u, v):
        to.append(v)
        nxt.append(head[u])
        head[u] = len(to) - 1
    
    def neg(x):
        if x > 0:
            return x + n
        else:
            return -x
    
    def pos(x):
        if x > 0:
            return x
        else:
            return -x + n
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        u = pos(a)
        v = pos(b)
        nu = neg(a)
        nv = neg(b)
        add_edge(u, nv)
        add_edge(v, nu)
    
    sys.setrecursionlimit(max(1000000, N + 100))
    
    index = [-1] * N
    low = [0] * N
    on_stack = [False] * N
    stack = []
    comp = [-1] * N
    counter = [0]
    comp_counter = [0]
    
    def dfs(start):
        # iterative DFS
        call_stack = [(start, head[start])]
        index[start] = counter[0]
        low[start] = counter[0]
        counter[0] += 1
        stack.append(start)
        on_stack[start] = True
        
        while call_stack:
            v, e = call_stack[-1]
            if e != -1:
                call_stack[-1] = (v, nxt[e])
                w = to[e]
                if index[w] == -1:
                    index[w] = counter[0]
                    low[w] = counter[0]
                    counter[0] += 1
                    stack.append(w)
                    on_stack[w] = True
                    call_stack.append((w, head[w]))
                elif on_stack[w]:
                    if index[w] < low[v]:
                        low[v] = index[w]
            else:
                call_stack.pop()
                if call_stack:
                    parent = call_stack[-1][0]
                    if low[v] < low[parent]:
                        low[parent] = low[v]
                if low[v] == index[v]:
                    while True:
                        w = stack.pop()
                        on_stack[w] = False
                        comp[w] = comp_counter[0]
                        if w == v:
                            break
                    comp_counter[0] += 1
    
    for i in range(1, N):
        if index[i] == -1:
            dfs(i)
    
    for i in range(1, n + 1):
        if comp[i] == comp[i + n]:
            print("NO")
            return
    
    print("YES")
    assignment = []
    for i in range(1, n + 1):
        if comp[i] < comp[i + n]:
            assignment.append('1')
        else:
            assignment.append('0')
    print(' '.join(assignment))

solve()
