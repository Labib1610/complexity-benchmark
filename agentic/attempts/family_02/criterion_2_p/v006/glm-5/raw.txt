import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # variables: 1..n, negations: n+1..2n
    size = 2 * n + 1
    head = [-1] * size
    to = []
    nxt = []
    
    def add_edge(u, v):
        to.append(v)
        nxt.append(head[u])
        head[u] = len(to) - 1
    
    def neg(x):
        return x + n if x <= n else x - n
    
    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        if a > 0:
            ua = a
        else:
            ua = (-a) + n
        if b > 0:
            ub = b
        else:
            ub = (-b) + n
        # clause (ua or ub): add ~ua -> ub and ~ub -> ua
        add_edge(neg(ua), ub)
        add_edge(neg(ub), ua)
    
    # Tarjan's SCC
    index_counter = [0]
    low = [0] * size
    on_stack = [False] * size
    visited = [False] * size
    stack = []
    comp = [-1] * size
    
    sys.setrecursionlimit(1 << 25)
    
    def strongconnect(v):
        # iterative Tarjan to avoid recursion depth issues
        work = [(v, 0)]
        while work:
            node, pi = work[-1]
            if pi == 0:
                low[node] = index_counter[0]
                visited[node] = True
                # we'll store index in low temporarily
                stack.append(node)
                on_stack[node] = True
                index_counter[0] += 1
            recurse = False
            e = head[node]
            # skip edges already processed
            while e != -1:
                w = to[e]
                if not visited[w]:
                    work[-1] = (node, e)
                    work.append((w, 0))
                    recurse = True
                    break
                elif on_stack[w]:
                    if low[w] < low[node]:
                        low[node] = low[w]
                e = nxt[e]
            if recurse:
                continue
            # done with node
            if low[node] == visited[node]:
                # pop stack
                while True:
                    w = stack.pop()
                    on_stack[w] = False
                    comp[w] = node
                    if w == node:
                        break
            work.pop()
            if work:
                parent = work[-1][0]
                if low[node] < low[parent]:
                    low[parent] = low[node]
    
    for i in range(1, size):
        if not visited[i]:
            strongconnect(i)
    
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        if comp[i] == comp[neg(i)]:
            print("NO")
            return
        assignment[i] = 1 if comp[i] < comp[neg(i)] else 0
    
    print("YES")
    print(' '.join(str(assignment[i]) for i in range(1, n + 1)))

solve()
