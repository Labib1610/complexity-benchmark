import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    num_nodes = 2 * n
    
    head = [-1] * num_nodes
    to = []
    nxt = []
    
    def add_edge(u, v):
        nxt.append(head[u])
        to.append(v)
        head[u] = len(to) - 1

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        # l1, l2:
        # positive x -> 2*(x-1) (true), 2*(x-1)+1 (false)
        # negative x -> 2*(-x-1)+1 (true for clause), 2*(-x-1) (false for clause)
        u = 2 * (abs(l1) - 1) if l1 > 0 else 2 * (abs(l1) - 1) + 1
        not_u = u ^ 1
        
        v = 2 * (abs(l2) - 1) if l2 > 0 else 2 * (abs(l2) - 1) + 1
        not_v = v ^ 1
        
        # (l1 or l2) <=> (not l1 => l2) and (not l2 => l1)
        add_edge(not_u, v)
        add_edge(not_v, u)

    # Tarjan's SCC iterative
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    scc_id = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stk = []
    
    timer = 0
    scc_count = 0
    
    # We will simulate call stack
    call_stack = []
    edge_ptr = [-1] * num_nodes
    
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue
            
        call_stack.append(i)
        dfn[i] = low[i] = timer
        timer += 1
        stk.append(i)
        in_stack[i] = True
        edge_ptr[i] = head[i]
        
        while call_stack:
            u = call_stack[-1]
            e = edge_ptr[u]
            if e != -1:
                edge_ptr[u] = nxt[e]
                v = to[e]
                if dfn[v] == -1:
                    dfn[v] = low[v] = timer
                    timer += 1
                    stk.append(v)
                    in_stack[v] = True
                    edge_ptr[v] = head[v]
                    call_stack.append(v)
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]
            else:
                # Finished exploring u
                call_stack.pop()
                if call_stack:
                    p = call_stack[-1]
                    if low[u] < low[p]:
                        low[p] = low[u]
                
                if low[u] == dfn[u]:
                    while True:
                        node = stk.pop()
                        in_stack[node] = False
                        scc_id[node] = scc_count
                        if node == u:
                            break
                    scc_count += 1

    ans = [0] * n
    for i in range(n):
        u = 2 * i
        not_u = 2 * i + 1
        if scc_id[u] == scc_id[not_u]:
            print("NO")
            return
        if scc_id[u] < scc_id[not_u]:
            ans[i] = 1
        else:
            ans[i] = 0
            
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
