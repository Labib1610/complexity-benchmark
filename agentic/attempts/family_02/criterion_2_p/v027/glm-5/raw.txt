import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        
        # u and v are the literals that must be satisfied (at least one)
        # To build implication graph, we add edges: not u -> v and not v -> u
        # Literal x: if x > 0, node is 2*(x-1); if x < 0, node is 2*(-x-1)+1
        # not x: if x > 0, node is 2*(x-1)+1; if x < 0, node is 2*(-x-1)
        
        # Map literal to node index
        def get_node(lit):
            idx = abs(lit) - 1
            if lit > 0:
                return 2 * idx
            else:
                return 2 * idx + 1
        
        u_node = get_node(u)
        v_node = get_node(v)
        not_u_node = u_node ^ 1
        not_v_node = v_node ^ 1
        
        adj[not_u_node].append(v_node)
        adj[not_v_node].append(u_node)
    
    N = 2 * n
    visited = [False] * N
    order = []
    
    # First pass: DFS to get finishing order
    for i in range(N):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                node, state = stack[-1]
                if state == 0:
                    visited[node] = True
                    stack[-1] = (node, 1)
                    for nei in adj[node]:
                        if not visited[nei]:
                            stack.append((nei, 0))
                else:
                    stack.pop()
                    order.append(node)
    
    # Second pass: DFS on reversed graph in decreasing order of finishing times
    visited = [False] * N
    comp_id = [0] * N
    current_comp = 0
    
    # Build reversed graph
    rev_adj = [[] for _ in range(N)]
    for u in range(N):
        for v in adj[u]:
            rev_adj[v].append(u)
    
    for node in reversed(order):
        if not visited[node]:
            current_comp += 1
            stack = [node]
            visited[node] = True
            while stack:
                curr = stack.pop()
                comp_id[curr] = current_comp
                for nei in rev_adj[curr]:
                    if not visited[nei]:
                        visited[nei] = True
                        stack.append(nei)
    
    # Check for contradictions
    assignment = [0] * n
    for i in range(n):
        node_true = 2 * i
        node_false = 2 * i + 1
        if comp_id[node_true] == comp_id[node_false]:
            print("NO")
            return
        # If true node has higher comp_id (processed later in reversed graph), it is assigned True
        # This corresponds to the topological order of SCCs
        assignment[i] = 1 if comp_id[node_true] > comp_id[node_false] else 0
    
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
