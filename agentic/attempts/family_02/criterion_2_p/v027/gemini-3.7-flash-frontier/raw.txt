import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    num_nodes = 2 * n
    head = [-1] * num_nodes
    next_edge = []
    to_edge = []
    
    def add_edge(u, v):
        next_edge.append(head[u])
        head[u] = len(to_edge)
        to_edge.append(v)

    head_rev = [-1] * num_nodes
    next_rev = []
    to_rev = []

    def add_edge_rev(u, v):
        next_rev.append(head_rev[u])
        head_rev[u] = len(to_rev)
        to_rev.append(v)

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx+1])
        idx += 2
        
        u = (l1 - 1) * 2 if l1 > 0 else (-l1 - 1) * 2 + 1
        v = (l2 - 1) * 2 if l2 > 0 else (-l2 - 1) * 2 + 1
        
        not_u = u ^ 1
        not_v = v ^ 1
        
        # not_u -> v
        add_edge(not_u, v)
        add_edge_rev(v, not_u)
        
        # not_v -> u
        add_edge(not_v, u)
        add_edge_rev(u, not_v)
        
    # Kosaraju's Algorithm (Iterative)
    visited = [False] * num_nodes
    order = []
    
    # First pass: ordering by finish time
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_ptr = [head[i]]
            visited[i] = True
            
            while stack:
                u = stack[-1]
                e = edge_ptr[-1]
                
                if e != -1:
                    edge_ptr[-1] = next_edge[e]
                    v = to_edge[e]
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
                        edge_ptr.append(head[v])
                else:
                    stack.pop()
                    edge_ptr.pop()
                    order.append(u)
                    
    # Second pass: SCC identification on reversed graph
    scc_id = [-1] * num_nodes
    current_scc = 0
    
    for i in reversed(order):
        if scc_id[i] == -1:
            stack = [i]
            scc_id[i] = current_scc
            
            while stack:
                u = stack.pop()
                e = head_rev[u]
                while e != -1:
                    v = to_rev[e]
                    if scc_id[v] == -1:
                        scc_id[v] = current_scc
                        stack.append(v)
                    e = next_rev[e]
            current_scc += 1
            
    # Check satisfiability and construct solution
    ans = [0] * n
    for i in range(n):
        pos_var = 2 * i
        neg_var = 2 * i + 1
        if scc_id[pos_var] == scc_id[neg_var]:
            print("NO")
            return
        # If scc_id[pos_var] < scc_id[neg_var], pos_var is later in DAG topological order
        # meaning pos_var is chosen as True (1)
        ans[i] = 1 if scc_id[pos_var] < scc_id[neg_var] else 0
        
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
