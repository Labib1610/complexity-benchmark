We can model each filter as a boolean variable and each constraint as a clause in 2-SAT form.  

A constraint like `(a or b)` where `a` and `b` are literals (positive = true, negative = false) can be transformed into implications:  
- `not a -> b`  
- `not b -> a`  

Then we run SCC (Strongly Connected Components) to check satisfiability. If a variable and its negation are in the same SCC, it’s unsatisfiable. Otherwise, we assign values based on topological order of SCCs.

```python
import sys
sys.setrecursionlimit(1 << 20)
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # nodes: 0..n-1 for x_i, n..2n-1 for not x_i

def node(var):
    # var > 0 -> x_var, var < 0 -> not x_var
    if var > 0:
        return (var - 1) * 2
    else:
        return (-var - 1) * 2 + 1

def neg_node(u):
    return u ^ 1

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

for _ in range(m):
    a, b = map(int, input().split())
    u = node(a)
    v = node(b)
    # not u -> v
    g[neg_node(u)].append(v)
    gr[v].append(neg_node(u))
    # not v -> u
    g[neg_node(v)].append(u)
    gr[u].append(neg_node(v))

# Kosaraju SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for w in g[u]:
        if not visited[w]:
            dfs1(w)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N
def dfs2(u, cid):
    comp[u] = cid
    for w in gr[u]:
        if comp[w] == -1:
            dfs2(w, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2*i] == comp[2*i+1]:
        print("NO")
        sys.exit()

# Assign values: if comp[x] > comp[not x] then true
res = [0] * n
for i in range(n):
    res[i] = 1 if comp[2*i] > comp[2*i+1] else 0

print("YES")
print(" ".join(map(str, res)))
```
