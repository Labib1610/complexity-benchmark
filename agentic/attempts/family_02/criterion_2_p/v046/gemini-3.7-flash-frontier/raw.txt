```python
import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    
    # Helper to get node index
    # Variable i (1-based):
    # True literal: 2 * (i - 1)
    # False literal: 2 * (i - 1) + 1
    def get_lit(l):
        if l > 0:
            return 2 * (l - 1)
        else:
            return 2 * (-l - 1) + 1
            
    def get_neg(l):
        if l > 0:
            return 2 * (l - 1) + 1
        else:
            return 2 * (-l - 1)

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        # (l1 or l2) <=> ~l1 -> l2 and ~l2 -> l1
        adj[get_neg(l1)].append(get_lit(l2))
        adj[get_neg(l2)].append(get_lit(l1))
        
    # Tarjan's SCC
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    scc_id = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stack = []
    timer = 0
    scc_count = 0
    
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue
            
        # Non-recursive / Recursive DFS
        # We can use recursion since sys.setrecursionlimit is large
        def dfs(u):
            nonlocal timer, scc_count
            dfn[u] = low[u] = timer
            timer += 1
            stack.append(u)
            in_stack[u] = True
            
            for v in adj[u]:
                if dfn[v] == -1:
                    dfs(v)
                    low[u] = min(low[u], low[v])
                elif in_stack[v]:
                    low[u] = min(low[u], dfn[v])
                    
            if low[u] == dfn[u]:
                while True:
                    w = stack.pop()
                    in_stack[w] = False
                    scc_id[w] = scc_count
                    if w == u:
                        break
                scc_count += 1
                
        dfs(i)
        
    ans = []
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc_id[pos] == scc_id[neg]:
            print("NO")
            return
        if scc_id[pos] < scc_id[neg]:
            ans.append("1")
        else:
            ans.append("0")
            
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
```
