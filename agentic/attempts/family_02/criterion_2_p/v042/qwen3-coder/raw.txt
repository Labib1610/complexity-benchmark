import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build implication graph
    # Variables: 0 to 2*n-1
    # For variable i (0-based): 
    #   node 2*i represents x_i (amber)
    #   node 2*i+1 represents not x_i (blue)
    
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    index = 2
    for _ in range(m):
        c1 = int(data[index])
        c2 = int(data[index + 1])
        index += 2
        
        # Convert to 0-based indexing and determine if negated
        # c1: if positive, then x_(c1-1); if negative, then not x_(|c1|-1)
        # c2: similar
        
        # The clause is (x_a OR x_b)
        # Which is equivalent to (not x_a -> x_b) AND (not x_b -> x_a)
        
        if c1 > 0:
            u = 2 * (c1 - 1)       # x_a
            not_u = u + 1          # not x_a
        else:
            not_u = 2 * (-c1 - 1)  # x_|c1|
            u = not_u + 1          # not x_|c1|
            
        if c2 > 0:
            v = 2 * (c2 - 1)       # x_b
            not_v = v + 1          # not x_b
        else:
            not_v = 2 * (-c2 - 1)  # x_|c2|
            v = not_v + 1          # not x_|c2|
            
        # Add implications: not_u -> v and not_v -> u
        adj[not_u].append(v)
        adj[not_v].append(u)
        rev_adj[v].append(not_u)
        rev_adj[u].append(not_v)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Reverse graph DFS
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check for contradictions
    for i in range(n):
        if scc_id[2*i] == scc_id[2*i + 1]:
            print("NO")
            return
    
    # Assign values
    assignment = [-1] * n
    # Process SCCs in reverse topological order
    # Since we used Kosaraju, scc_id gives a topological order (lower id = earlier)
    # We want to process later SCCs first to assign values correctly
    
    # Create list of (scc_id, node) and sort by scc_id descending
    scc_nodes = [(scc_id[i], i) for i in range(2 * n)]
    scc_nodes.sort(reverse=True)
    
    for _, node in scc_nodes:
        var = node // 2
        is_neg = node % 2
        if assignment[var] == -1:
            if is_neg:
                assignment[var] = 0  # blue
            else:
                assignment[var] = 1  # amber
    
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
