import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # 2 * n literals:
    # for i in 1..n:
    #   +i -> 2 * (i - 1)
    #   -i -> 2 * (i - 1) + 1
    # negation of lit: lit ^ 1
    
    num_nodes = 2 * n
    head = [-1] * num_nodes
    to = []
    nxt = []
    
    head_rev = [-1] * num_nodes
    to_rev = []
    nxt_rev = []
    
    def add_edge(u, v):
        nxt.append(head[u])
        to.append(v)
        head[u] = len(to) - 1
        
        nxt_rev.append(head_rev[v])
        to_rev.append(u)
        head_rev[v] = len(to_rev) - 1

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        u = 2 * (abs(l1) - 1) if l1 > 0 else 2 * (abs(l1) - 1) + 1
        v = 2 * (abs(l2) - 1) if l2 > 0 else 2 * (abs(l2) - 1) + 1
        
        # (u or v) <=> (~u -> v) and (~v -> u)
        add_edge(u ^ 1, v)
        add_edge(v ^ 1, u)
        
    # Kosaraju's algorithm (iterative)
    visited = [False] * num_nodes
    order = []
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_ptr = [head[i]]
            visited[i] = True
            
            while stack:
                curr = stack[-1]
                e = edge_ptr[-1]
                if e != -1:
                    edge_ptr[-1] = nxt[e]
                    nxt_node = to[e]
                    if not visited[nxt_node]:
                        visited[nxt_node] = True
                        stack.append(nxt_node)
                        edge_ptr.append(head[nxt_node])
                else:
                    order.append(stack.pop())
                    edge_ptr.pop()
                    
    scc = [-1] * num_nodes
    scc_count = 0
    
    for i in reversed(order):
        if scc[i] == -1:
            stack = [i]
            scc[i] = scc_count
            while stack:
                curr = stack.pop()
                e = head_rev[curr]
                while e != -1:
                    nxt_node = to_rev[e]
                    if scc[nxt_node] == -1:
                        scc[nxt_node] = scc_count
                        stack.append(nxt_node)
                    e = nxt_rev[e]
            scc_count += 1
            
    ans = []
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        # In Kosaraju, smaller scc id means topologically earlier in original graph?
        # Actually, in transposed graph, we assign 0, 1, 2... in reverse finishing order of original DFS.
        # This gives a topological order of the SCC DAG.
        # If pos -> neg, then scc[pos] <= scc[neg]. We want to avoid pos=True if pos -> ~pos.
        # If scc[pos] > scc[neg], pos cannot reach neg, so pos=True is valid (or pos is topologically after neg).
        # Specifically, scc[pos] < scc[neg] means pos comes earlier in topological order of implication graph,
        # so pos -> neg is possible, hence pos must be False (0).
        # Thus pos is 1 if scc[pos] > scc[neg] else 0.
        if scc[pos] > scc[neg]:
            ans.append(1)
        else:
            ans.append(0)
            
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
