import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    num_nodes = 2 * n
    head = [-1] * num_nodes
    to = []
    nxt = []
    
    def add_edge(u, v):
        nxt.append(head[u])
        head[u] = len(to)
        to.append(v)

    def lit_to_node(l):
        if l > 0:
            return 2 * (l - 1)
        else:
            return 2 * (-l - 1) + 1

    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        
        u = lit_to_node(l1)
        v = lit_to_node(l2)
        
        # (u or v) is equivalent to:
        # not u -> v
        # not v -> u
        not_u = u ^ 1
        not_v = v ^ 1
        
        add_edge(not_u, v)
        add_edge(not_v, u)

    # Tarjan's SCC iterative
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    scc = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stack = []
    scc_cnt = 0
    timer = 0

    # We will use an explicit call stack for DFS
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue
        
        call_stack = [(i, head[i])]
        dfn[i] = low[i] = timer
        timer += 1
        stack.append(i)
        in_stack[i] = True
        
        while call_stack:
            u, e = call_stack[-1]
            if e != -1:
                call_stack[-1] = (u, nxt[e])
                v = to[e]
                if dfn[v] == -1:
                    dfn[v] = low[v] = timer
                    timer += 1
                    stack.append(v)
                    in_stack[v] = True
                    call_stack.append((v, head[v]))
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]
            else:
                call_stack.pop()
                if call_stack:
                    p = call_stack[-1][0]
                    if low[u] < low[p]:
                        low[p] = low[u]
                
                if low[u] == dfn[u]:
                    while True:
                        v = stack.pop()
                        in_stack[v] = False
                        scc[v] = scc_cnt
                        if v == u:
                            break
                    scc_cnt += 1

    ans = [0] * n
    for i in range(n):
        if scc[2 * i] == scc[2 * i + 1]:
            print("NO")
            return
        ans[i] = 1 if scc[2 * i] < scc[2 * i + 1] else 0

    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
