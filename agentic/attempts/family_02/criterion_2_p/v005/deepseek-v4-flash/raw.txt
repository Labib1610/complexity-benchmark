We can model this as a **2-SAT** problem.  
Each rule `l1 l2` means:  
`(condition l1) OR (condition l2)` must be true.

A positive integer `x` means switch `x` is ON (true).  
A negative integer `-x` means switch `x` is OFF (false).

So each rule is a clause of two literals.  
We can solve 2-SAT using implication graph and SCC (Tarjan or Kosaraju).

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    
    # 2-SAT: variable i (1..n) -> node 2*(i-1) for false, 2*(i-1)+1 for true
    N = 2 * n
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]
    
    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)
    
    def node(var):
        # var > 0: true, var < 0: false
        idx = abs(var) - 1
        if var > 0:
            return 2 * idx + 1
        else:
            return 2 * idx
    
    for _ in range(m):
        a, b = map(int, input().split())
        # (a or b) => (!a -> b) and (!b -> a)
        na = node(-a)
        nb = node(b)
        add_imp(na, nb)
        nb2 = node(-b)
        na2 = node(a)
        add_imp(nb2, na2)
    
    # Kosaraju SCC
    visited = [False] * N
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)
    
    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1
    
    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return
    
    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # If false node has higher comp id, assign false (0), else true (1)
        assignment[i] = 1 if comp[2*i] < comp[2*i+1] else 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
