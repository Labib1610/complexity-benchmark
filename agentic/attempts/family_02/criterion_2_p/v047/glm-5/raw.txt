import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Build implication graph
    # Node 2*i represents x_i = True, node 2*i+1 represents x_i = False
    # For each constraint (a, b), we need a OR b
    # a OR b is equivalent to (not a -> b) AND (not b -> a)
    
    adj = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        # Convert to node indices
        # If l > 0: marker l must be 1 (True), represented by node 2*(l-1)
        # If l < 0: marker |l| must be 0 (False), represented by node 2*(|l|-1)+1
        
        def get_node(val):
            if val > 0:
                return 2 * (val - 1)  # True node
            else:
                return 2 * ((-val) - 1) + 1  # False node
        
        def negate_node(node):
            return node ^ 1
        
        u = get_node(l1)
        v = get_node(l2)
        
        # Add edges: not u -> v and not v -> u
        adj[negate_node(u)].append(v)
        adj[negate_node(v)].append(u)
    
    # Kosaraju's algorithm for SCC
    N = 2 * n
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                nei = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[nei]:
                    stack.append((nei, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    # Build reverse graph
    radj = [[] for _ in range(N)]
    for u in range(N):
        for v in adj[u]:
            radj[v].append(u)
    
    visited = [False] * N
    comp = [-1] * N
    comp_id = 0
    
    def dfs2(u, cid):
        stack = [u]
        visited[u] = True
        while stack:
            node = stack.pop()
            comp[node] = cid
            for nei in radj[node]:
                if not visited[nei]:
                    visited[nei] = True
                    stack.append(nei)
    
    for u in reversed(order):
        if not visited[u]:
            dfs2(u, comp_id)
            comp_id += 1
    
    # Check if x_i and not x_i are in the same SCC
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
    
    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # If comp[True] > comp[False], set to True (since we process in reverse topological order)
        if comp[2 * i] > comp[2 * i + 1]:
            assignment[i] = 1
    
    print("YES")
    print(' '.join(map(str, assignment)))

solve()
