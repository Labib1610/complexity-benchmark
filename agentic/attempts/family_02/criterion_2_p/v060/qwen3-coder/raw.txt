import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        clauses.append((l1, l2))
        index += 2
    
    # Build implication graph
    # Variables: 1 to n (positive for ethanol, negative for glycerin)
    # We'll use nodes: 1 to n for x_i, n+1 to 2n for not x_i
    # But for simplicity, let's map variable v (1-based) to node 2v-2 (for x_v) and 2v-1 (for not x_v)
    # i.e., for variable v:
    #    node 2*(v-1) represents x_v (herb v in ethanol)
    #    node 2*(v-1)+1 represents not x_v (herb v in glycerin)
    
    adj = [[] for _ in range(2 * n)]
    adj_rev = [[] for _ in range(2 * n)]
    
    # For each clause (l1 or l2), we add edges:
    # not l1 -> l2 and not l2 -> l1
    
    for l1, l2 in clauses:
        # Convert literals to node indices
        # l1 is positive: l1 means herb |l1| in ethanol => node 2*(|l1|-1)
        # l1 is negative: l1 means herb |l1| in glycerin => node 2*(|l1|-1)+1
        def lit_to_node(lit):
            var = abs(lit)
            if lit > 0:
                return 2 * (var - 1)     # x_var
            else:
                return 2 * (var - 1) + 1 # not x_var
        
        node_l1 = lit_to_node(l1)
        node_l2 = lit_to_node(l2)
        
        # not l1 -> l2
        not_l1 = node_l1 ^ 1  # flip last bit to get the negation
        adj[not_l1].append(node_l2)
        adj_rev[node_l2].append(not_l1)
        
        # not l2 -> l1
        not_l2 = node_l2 ^ 1
        adj[not_l2].append(node_l1)
        adj_rev[node_l1].append(not_l2)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in adj_rev[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check for contradiction: variable and its negation in same SCC
    for i in range(n):
        node_x = 2 * i       # x_{i+1}
        node_not_x = 2 * i + 1 # not x_{i+1}
        if scc_id[node_x] == scc_id[node_not_x]:
            print("NO")
            return
    
    # Assign values: process SCCs in reverse topological order
    assignment = [-1] * n
    # Get SCCs in topological order by processing order list backwards
    # But we already popped all from order, so we can't use it directly.
    # Instead, we can sort SCC ids by their finishing time in reverse.
    
    # Let's create a list of (finishing_time, node) and sort
    # But since we don't have explicit times, we can use the fact that 
    # in Kosaraju, the order in which we call dfs2 gives reverse topological order of SCC DAG
    
    # So we process SCCs in order from 0 to scc_count-1, which is reverse topological
    for i in range(2 * n):
        scc = scc_id[i]
        var_index = i // 2  # which variable (0-based)
        is_negated = i % 2  # 0 means positive (ethanol), 1 means negative (glycerin)
        
        if assignment[var_index] == -1:
            # If not assigned yet, assign based on whether it's the positive or negative node
            if is_negated == 0:  # This is the positive literal (ethanol)
                assignment[var_index] = 1
            else:  # This is the negative literal (glycerin)
                assignment[var_index] = 0
    
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
