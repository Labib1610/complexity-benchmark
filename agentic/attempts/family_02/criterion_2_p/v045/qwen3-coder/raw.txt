import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # For each lever i, we have two variables:
    # x_i (i-1 in 0-indexed) representing ACTIVE
    # not x_i (i-1+n in 0-indexed) representing STANDBY
    
    clauses = []
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        
        # Convert to 0-indexed variables for SAT solver
        # Positive literal -> ACTIVE
        # Negative literal -> STANDBY
        v1 = abs(l1) - 1
        v2 = abs(l2) - 1
        
        if l1 > 0:
            lit1 = v1  # x_v1
        else:
            lit1 = v1 + n  # not x_v1
            
        if l2 > 0:
            lit2 = v2  # x_v2
        else:
            lit2 = v2 + n  # not x_v2
            
        clauses.append((lit1, lit2))
    
    # Now we solve 2-SAT
    # Number of variables: 2*n (x_0,...,x_{n-1}, not x_0, ..., not x_{n-1})
    # Variables are numbered from 0 to 2*n - 1
    
    # Build implication graph
    # Each variable has index: 0 to 2*n - 1
    # x_i is at index i
    # not x_i is at index i + n
    
    adj = [[] for _ in range(2 * n)]
    adj_rev = [[] for _ in range(2 * n)]
    
    # For clause (a OR b), we add edges (not a -> b) and (not b -> a)
    for a, b in clauses:
        # not a -> b
        adj[a ^ n].append(b)
        adj_rev[b].append(a ^ n)
        # not b -> a
        adj[b ^ n].append(a)
        adj_rev[a].append(b ^ n)
        
    # Kosaraju's algorithm for SCC
    order = []
    visited = [False] * (2 * n)
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
        
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
            
    # Second DFS on transpose graph
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in adj_rev[u]:
            if not visited[v]:
                dfs2(v)
                
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
            
    # Check if satisfiable
    for i in range(n):
        if scc_id[i] == scc_id[i + n]:
            print("NO")
            return
            
    # Extract solution
    # In the implication graph, if scc_id[x_i] > scc_id[not x_i], then x_i is true
    result = [0] * n
    for i in range(n):
        if scc_id[i] > scc_id[i + n]:
            result[i] = 1  # ACTIVE
        else:
            result[i] = 0  # STANDBY
            
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
