import sys
sys.setrecursionlimit(200000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 2)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        u = abs(l1)
        nu = u + n
        v = abs(l2)
        nv = v + n
        
        if l1 > 0:
            u = u + n
            nu = u - n
        if l2 > 0:
            v = v + n
            nv = v - n
            
        adj[nu].append(v)
        adj[nv].append(u)
    
    visited = [False] * (2 * n + 2)
    order = []
    
    def dfs(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                neighbor = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(1, 2 * n + 2):
        if not visited[i]:
            dfs(i)
    
    comp = [-1] * (2 * n + 2)
    cid = 0
    
    radj = [[] for _ in range(2 * n + 2)]
    for u in range(1, 2 * n + 2):
        for v in adj[u]:
            radj[v].append(u)
            
    for u in reversed(order):
        if comp[u] == -1:
            stack = [u]
            comp[u] = cid
            while stack:
                node = stack.pop()
                for neighbor in radj[node]:
                    if comp[neighbor] == -1:
                        comp[neighbor] = cid
                        stack.append(neighbor)
            cid += 1
    
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        if comp[i] == comp[i + n]:
            print("NO")
            return
        if comp[i] > comp[i + n]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

solve()
