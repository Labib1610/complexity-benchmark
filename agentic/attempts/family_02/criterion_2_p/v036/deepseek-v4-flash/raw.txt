We can model each zone as a boolean variable, and each contingency plan as a 2-SAT clause.  
A clause `(l1, l2)` means: at least one of the two literals must be true.  
We can solve 2-SAT using implication graph and SCC (Tarjan or Kosaraju).

```python
import sys
sys.setrecursionlimit(1 << 25)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    # 2-SAT: variable i (1..n) -> node 2*i for true, 2*i+1 for false
    # We'll use 0-based: node 2*i = true, 2*i+1 = false
    N = 2 * n
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    # literal to node
    def lit_to_node(lit):
        var = abs(lit)
        if lit > 0:
            return 2 * (var - 1)      # true
        else:
            return 2 * (var - 1) + 1  # false

    def neg_node(u):
        return u ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        u = lit_to_node(a)
        v = lit_to_node(b)
        # (a or b) => (!a -> b) and (!b -> a)
        add_imp(neg_node(u), v)
        add_imp(neg_node(v), u)

    # Kosaraju SCC
    visited = [False] * N
    order = []
    def dfs1(u):
        visited[u] = True
        for w in g[u]:
            if not visited[w]:
                dfs1(w)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for w in gr[u]:
            if comp[w] == -1:
                dfs2(w, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Assign values: if comp[true] > comp[false] => true
    assignment = [0] * n
    for i in range(n):
        if comp[2*i] > comp[2*i+1]:
            assignment[i] = 1
        else:
            assignment[i] = 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
