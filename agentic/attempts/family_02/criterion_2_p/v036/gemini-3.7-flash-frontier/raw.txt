import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # 2n vertices: for variable i (1-indexed):
    # True is 2*(i-1)
    # False is 2*(i-1) + 1
    # Negation of lit is lit ^ 1
    
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx+1])
        idx += 2
        
        u = (abs(l1) - 1) * 2 if l1 > 0 else (abs(l1) - 1) * 2 + 1
        v = (abs(l2) - 1) * 2 if l2 > 0 else (abs(l2) - 1) * 2 + 1
        
        # (u or v) <=> (~u => v) and (~v => u)
        not_u = u ^ 1
        not_v = v ^ 1
        
        adj[not_u].append(v)
        rev_adj[v].append(not_u)
        
        adj[not_v].append(u)
        rev_adj[u].append(not_v)
        
    num_nodes = 2 * n
    
    # Pass 1: Order by finishing time
    visited = [False] * num_nodes
    order = []
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            visited[i] = True
            it = [0] * num_nodes
            
            # Iterative DFS for topological sort
            dfs_stack = [(i, 0)]
            while dfs_stack:
                u, edge_idx = dfs_stack.pop()
                if edge_idx == 0:
                    visited[u] = True
                
                while edge_idx < len(adj[u]):
                    v = adj[u][edge_idx]
                    edge_idx += 1
                    if not visited[v]:
                        visited[v] = True
                        dfs_stack.append((u, edge_idx))
                        u = v
                        edge_idx = 0
                
                order.append(u)
                
    # Pass 2: Assign SCC components
    scc = [-1] * num_nodes
    scc_id = 0
    
    for i in reversed(order):
        if scc[i] == -1:
            stack = [i]
            scc[i] = scc_id
            while stack:
                u = stack.pop()
                for v in rev_adj[u]:
                    if scc[v] == -1:
                        scc[v] = scc_id
                        stack.append(v)
            scc_id += 1
            
    # Check satisfiability and construct answer
    ans = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        ans[i] = 1 if scc[pos] > scc[neg] else 0
        
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
