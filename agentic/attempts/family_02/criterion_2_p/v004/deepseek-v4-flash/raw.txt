We can model this as a **2-SAT** problem.  
Each rule is a clause of the form `(A OR B)`, where `A` and `B` are literals (a variable or its negation).  

We build an implication graph:
- For each clause `(x OR y)`, add edges `(¬x → y)` and `(¬y → x)`.

Then we run **Tarjan’s SCC** algorithm.  
If a variable and its negation are in the same SCC, the formula is unsatisfiable → output `NO`.  
Otherwise, we assign values based on the topological order of SCCs (or simply by comparing SCC IDs).

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 0..n-1 for x_i, n..2n-1 for ¬x_i

    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def idx(lit):
        # lit > 0: variable lit-1, lit < 0: negation of (-lit)-1
        if lit > 0:
            return (lit - 1) * 2
        else:
            return (-lit - 1) * 2 + 1

    def neg(i):
        return i ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        ia = idx(a)
        ib = idx(b)
        # (a OR b) => (¬a -> b) and (¬b -> a)
        g[neg(ia)].append(ib)
        rg[ib].append(neg(ia))
        g[neg(ib)].append(ia)
        rg[ia].append(neg(ib))

    # Kosaraju's SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in rg[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check contradiction
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Assign values: if comp[2*i] > comp[2*i+1] => True (1), else False (0)
    res = []
    for i in range(n):
        if comp[2*i] > comp[2*i+1]:
            res.append('1')
        else:
            res.append('0')

    print("YES")
    print(' '.join(res))

if __name__ == "__main__":
    solve()
```
