import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    try:
        n = int(next(it))
        m = int(next(it))
    except StopIteration:
        return

    # Build implication graph
    # Node i (1-indexed) represents that switch i is ON
    # Node -i represents that switch i is OFF
    # 2-SAT: clause (a or b) -> edges: not a -> b, not b -> a
    G = [[] for _ in range(2 * n + 1)]
    
    for _ in range(m):
        try:
            l1 = int(next(it))
            l2 = int(next(it))
        except StopIteration:
            break
        
        # not l1 -> l2
        u = -l1
        v = l2
        # Map to positive indices: pos(x) = 2x-1 if x>0 else 2|x|-2
        pu = 2 * abs(u) - (2 if u > 0 else 1)
        pv = 2 * abs(v) - (2 if v > 0 else 1)
        G[pu].append(pv)
        
        # not l2 -> l1
        u = -l2
        v = l1
        pu = 2 * abs(u) - (2 if u > 0 else 1)
        pv = 2 * abs(v) - (2 if v > 0 else 1)
        G[pu].append(pv)

    N = 2 * n
    visited = [False] * N
    order = []
    
    # Kosaraju's algorithm: Step 1 - DFS to get finish order
    for i in range(N):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                u, idx = stack[-1]
                if not visited[u]:
                    visited[u] = True
                
                if idx < len(G[u]):
                    stack[-1] = (u, idx + 1)
                    v = G[u][idx]
                    if not visited[v]:
                        stack.append((v, 0))
                else:
                    stack.pop()
                    order.append(u)

    # Build reverse graph
    GT = [[] for _ in range(N)]
    for u in range(N):
        for v in G[u]:
            GT[v].append(u)

    # Step 2 - Process in reverse finish order to find SCCs
    scc_id = [-1] * N
    current_id = 0
    
    for u in reversed(order):
        if scc_id[u] == -1:
            stack = [u]
            scc_id[u] = current_id
            while stack:
                curr = stack.pop()
                for v in GT[curr]:
                    if scc_id[v] == -1:
                        scc_id[v] = current_id
                        stack.append(v)
            current_id += 1

    # Check for contradiction: x and not x in same SCC
    for i in range(1, n + 1):
        pos_node = 2 * i - 2
        neg_node = 2 * i - 1
        if scc_id[pos_node] == scc_id[neg_node]:
            print("NO")
            return

    # Construct assignment: variable x is true if scc(x) < scc(not x)
    print("YES")
    res = []
    for i in range(1, n + 1):
        pos_node = 2 * i - 2
        neg_node = 2 * i - 1
        if scc_id[pos_node] < scc_id[neg_node]:
            res.append('1')
        else:
            res.append('0')
    
    print(' '.join(res))

if __name__ == "__main__":
    solve()
