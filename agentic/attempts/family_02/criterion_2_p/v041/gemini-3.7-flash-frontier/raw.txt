```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        # l > 0: var = l - 1, true: 2*var, false: 2*var + 1
        # l < 0: var = -l - 1, true: 2*var + 1, false: 2*var
        u = 2 * (l1 - 1) if l1 > 0 else 2 * (-l1 - 1) + 1
        v = 2 * (l2 - 1) if l2 > 0 else 2 * (-l2 - 1) + 1
        
        # clause: u or v => not u -> v, not v -> u
        adj[u ^ 1].append(v)
        adj[v ^ 1].append(u)
        
    num_nodes = 2 * n
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    in_stack = [False] * num_nodes
    sccs = [-1] * num_nodes
    stack = []
    timer = 0
    scc_cnt = 0
    
    for start in range(num_nodes):
        if dfn[start] != -1:
            continue
        call_stack = [(start, 0)]
        dfn[start] = low[start] = timer
        timer += 1
        stack.append(start)
        in_stack[start] = True
        
        while call_stack:
            u, edge_idx = call_stack[-1]
            if edge_idx < len(adj[u]):
                v = adj[u][edge_idx]
                call_stack[-1] = (u, edge_idx + 1)
                if dfn[v] == -1:
                    dfn[v] = low[v] = timer
                    timer += 1
                    stack.append(v)
                    in_stack[v] = True
                    call_stack.append((v, 0))
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]
            else:
                call_stack.pop()
                if call_stack:
                    p, _ = call_stack[-1]
                    if low[u] < low[p]:
                        low[p] = low[u]
                if low[u] == dfn[u]:
                    while True:
                        v = stack.pop()
                        in_stack[v] = False
                        sccs[v] = scc_cnt
                        if v == u:
                            break
                    scc_cnt += 1

    ans = [0] * n
    for i in range(n):
        if sccs[2 * i] == sccs[2 * i + 1]:
            print("NO")
            return
        ans[i] = 1 if sccs[2 * i] < sccs[2 * i + 1] else 0

    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
```
