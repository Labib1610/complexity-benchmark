```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # Build implication graph for 2-SAT
    # Variables: for drone i, var i is upper=1, var -i is lower=0
    # For each clause (a OR b), we add edges (!a -> b) and (!b -> a)
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        
        # Clause: (l1 OR l2)
        # This means: if not l1 then l2, and if not l2 then l1
        # not l1 -> l2  and not l2 -> l1
        
        # not l1 is: -l1 becomes ~upper, which is lower, so if l1>0 it's -l1, if l1<0 it's l1
        # not l1 = -l1 if l1>0, = l1 if l1<0
        not_l1 = -l1 if l1 > 0 else abs(l1)
        not_l2 = -l2 if l2 > 0 else abs(l2)
        
        adj[not_l1].append(l2)
        adj[not_l2].append(l1)
        rev_adj[l2].append(not_l1)
        rev_adj[l1].append(not_l2)
    
    # Kosaraju's algorithm for SCC
    visited = set()
    order = []
    
    def dfs1(u):
        visited.add(u)
        for v in adj[u]:
            if v not in visited:
                dfs1(v)
        order.append(u)
    
    # Run DFS1 on all nodes
    for i in range(1, n+1):
        if i not in visited:
            dfs1(i)
        if -i not in visited:
            dfs1(-i)
    
    # Reverse the graph and do DFS in reverse finish order
    visited.clear()
    scc = {}
    scc_id = 0
    
    def dfs2(u):
        visited.add(u)
        scc[u] = scc_id
        for v in rev_adj[u]:
            if v not in visited:
                dfs2(v)
    
    for u in reversed(order):
        if u not in visited:
            dfs2(u)
            scc_id += 1
    
    # Check if any variable and its negation are in the same SCC
    for i in range(1, n+1):
        if scc[i] == scc[-i]:
            print("NO")
            return
    
    # Assign values based on SCC topological order
    # In Kosaraju, later finishing components in first DFS are "earlier" in condensation
    # So higher SCC id means earlier in topological order of SCC DAG
    assignment = {}
    
    # Process SCCs in reverse topological order (lower SCC id first)
    scc_nodes = defaultdict(list)
    for node, sid in scc.items():
        scc_nodes[sid].append(node)
    
    for sid in sorted(scc_nodes.keys()):
        for node in scc_nodes[sid]:
            if abs(node) not in assignment:
                if node > 0:
                    assignment[node] = 1
                else:
                    assignment[-node] = 0
    
    result = []
    for i in range(1, n+1):
        result.append(str(assignment[i]))
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
