import sys
from collections import deque

def solve():
    # Read all input at once for efficiency
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Build implication graph
    # Nodes: 1 to 2n
    # Node k represents drone k is True (upper corridor)
    # Node k+n represents drone k is False (lower corridor)
    # Total nodes: 2n
    N = 2 * n + 1
    adj = [[] for _ in range(N)]
    adj_rev = [[] for _ in range(N)]
    
    # Helper to convert drone requirement to node index
    # x > 0: drone x is True -> node x
    # x < 0: drone |x| is False -> node |x| + n
    def get_node(x):
        if x > 0:
            return x
        else:
            return -x + n

    for _ in range(m):
        try:
            l1 = int(next(iterator))
            l2 = int(next(iterator))
        except StopIteration:
            break
        
        u = get_node(l1)
        v = get_node(l2)
        
        # Determine negations
        # If u corresponds to drone k being True, u+n is drone k being False
        # If u corresponds to drone k being False, u-n is drone k being True
        # Since 1 <= u <= n implies True, n+1 <= u <= 2n implies False
        
        u_neg = u + n if u <= n else u - n
        v_neg = v + n if v <= n else v - n
        
        # Add implication: not u -> v
        adj[u_neg].append(v)
        adj_rev[v].append(u_neg)
        
        # Add implication: not v -> u
        adj[v_neg].append(u)
        adj_rev[u].append(v_neg)

    # Kosaraju's algorithm to find SCCs
    # Pass 1: DFS to get finish order
    visited = [False] * N
    order = []
    
    # Using iterative DFS to avoid recursion limit
    for i in range(1, N):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                node, idx = stack.pop()
                if idx == 0:
                    if visited[node]:
                        continue
                    visited[node] = True
                
                if idx < len(adj[node]):
                    child = adj[node][idx]
                    stack.append((node, idx + 1))
                    if not visited[child]:
                        stack.append((child, 0))
                else:
                    order.append(node)

    # Pass 2: DFS on reversed graph in reverse finish order
    visited = [False] * N
    component_id = [0] * N
    current_id = 0
    
    for node in reversed(order):
        if not visited[node]:
            current_id += 1
            stack = [node]
            visited[node] = True
            while stack:
                curr = stack.pop()
                component_id[curr] = current_id
                for child in adj_rev[curr]:
                    if not visited[child]:
                        visited[child] = True
                        stack.append(child)

    # Check for contradictions and build result
    result = [0] * (n + 1)
    possible = True
    
    for i in range(1, n + 1):
        # Node i: drone i is True
        # Node i+n: drone i is False
        if component_id[i] == component_id[i + n]:
            possible = False
            break
        # If component of True node is greater, it implies False
        # We want to assign True if component of True is smaller (topological order)
        if component_id[i] < component_id[i + n]:
            result[i] = 1
        else:
            result[i] = 0

    if possible:
        print("YES")
        print(*(result[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
