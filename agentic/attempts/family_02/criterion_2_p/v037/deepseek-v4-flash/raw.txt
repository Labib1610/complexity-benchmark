We can model this as a **2-SAT** problem.  
Each rule `(a, b)` means: `a` OR `b` must be true.  
We convert each literal to a boolean variable and build implication edges, then check satisfiability using SCC (Kosaraju or Tarjan).  

```python
import sys
sys.setrecursionlimit(1 << 20)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 0..n-1 for x_i false, n..2n-1 for x_i true

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    # literal to node: positive x -> node (x-1)+n, negative -x -> node (x-1)
    def node(lit):
        if lit > 0:
            return (lit - 1) + n
        else:
            return (-lit - 1)

    for _ in range(m):
        a, b = map(int, input().split())
        # a OR b  =>  (not a -> b) and (not b -> a)
        na = node(-a)
        nb = node(-b)
        add_imp(na, node(b))
        add_imp(nb, node(a))

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[i] == comp[i + n]:
            print("NO")
            return

    # Build assignment: choose value based on SCC order
    assignment = [0] * n
    for i in range(n):
        # comp[i] < comp[i+n] means false literal comes earlier => assign false
        assignment[i] = 1 if comp[i] > comp[i + n] else 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
