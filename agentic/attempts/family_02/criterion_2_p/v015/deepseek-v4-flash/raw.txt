We can model this as a **2-SAT** problem.  
Each gate has two possible states: OPEN (1) or CLOSED (0).  
Each mandate is a clause of the form `(A OR B)`, where A and B are literals like `x_i` or `¬x_i`.

We can solve 2-SAT using implication graph and SCC (Tarjan or Kosaraju).

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 2*i for x_i false, 2*i+1 for x_i true

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    # literal to node
    # positive v -> node 2*(v-1)+1
    # negative v -> node 2*(v-1)
    def lit_node(lit):
        if lit > 0:
            return 2 * (lit - 1) + 1
        else:
            return 2 * (-lit - 1)

    for _ in range(m):
        a, b = map(int, input().split())
        na = a ^ 1 if a > 0 else (-a)  # not used directly, we use node ids
        # clause (a OR b) => (¬a -> b) and (¬b -> a)
        na_node = lit_node(-a)
        nb_node = lit_node(-b)
        a_node = lit_node(a)
        b_node = lit_node(b)
        add_imp(na_node, b_node)
        add_imp(nb_node, a_node)

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Assign values
    assignment = [0] * n
    for i in range(n):
        # If comp[2*i] > comp[2*i+1], set true
        assignment[i] = 1 if comp[2*i] > comp[2*i+1] else 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
