import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])

    num_nodes = 2 * n
    head = [-1] * num_nodes
    to = []
    nxt = []

    head_t = [-1] * num_nodes
    to_t = []
    nxt_t = []

    def add_edge(u, v):
        nxt.append(head[u])
        to.append(v)
        head[u] = len(to) - 1

        nxt_t.append(head_t[v])
        to_t.append(u)
        head_t[v] = len(to_t) - 1

    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx + 1])
        idx += 2

        # Convert l1, l2 to node indices
        # For variable x (1-indexed):
        # x is True -> 2*(x-1), x is False -> 2*(x-1)+1
        if l1 > 0:
            u = 2 * (l1 - 1)
        else:
            u = 2 * (-l1 - 1) + 1

        if l2 > 0:
            v = 2 * (l2 - 1)
        else:
            v = 2 * (-l2 - 1) + 1

        # not u -> v
        add_edge(u ^ 1, v)
        # not v -> u
        add_edge(v ^ 1, u)

    # Kosaraju's algorithm
    # Step 1: get post-order on original graph
    order = []
    visited = [False] * num_nodes

    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_ptr = [head[i]]
            visited[i] = True

            while stack:
                u = stack[-1]
                e = edge_ptr[-1]
                if e != -1:
                    edge_ptr[-1] = nxt[e]
                    v = to[e]
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
                        edge_ptr.append(head[v])
                else:
                    stack.pop()
                    edge_ptr.pop()
                    order.append(u)

    # Step 2: DFS on transposed graph
    scc = [-1] * num_nodes
    scc_count = 0

    for i in reversed(order):
        if scc[i] == -1:
            stack = [i]
            scc[i] = scc_count
            while stack:
                u = stack.pop()
                e = head_t[u]
                while e != -1:
                    v = to_t[e]
                    if scc[v] == -1:
                        scc[v] = scc_count
                        stack.append(v)
                    e = nxt_t[e]
            scc_count += 1

    # Check satisfiability and build solution
    res = [0] * n
    for i in range(n):
        if scc[2 * i] == scc[2 * i + 1]:
            print("NO")
            return
        if scc[2 * i] > scc[2 * i + 1]:
            res[i] = 1
        else:
            res[i] = 0

    print("YES")
    print(*(res))

if __name__ == '__main__':
    solve()
