import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(2 * n + 2)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
        except StopIteration:
            break
        
        nu = -u
        nv = -v
        
        u_node = u if u > 0 else (n - u)
        v_node = v if v > 0 else (n - v)
        nu_node = nu if nu > 0 else (n - nu)
        nv_node = nv if nv > 0 else (n - nv)
        
        adj[nu_node].append(v_node)
        adj[nv_node].append(u_node)

    total_nodes = 2 * n + 2
    visited = [False] * total_nodes
    order = []
    
    for i in range(1, total_nodes):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                node, idx = stack[-1]
                if not visited[node]:
                    visited[node] = True
                
                while idx < len(adj[node]) and visited[adj[node][idx]]:
                    idx += 1
                
                if idx < len(adj[node]):
                    stack[-1] = (node, idx + 1)
                    neighbor = adj[node][idx]
                    if not visited[neighbor]:
                        stack.append((neighbor, 0))
                else:
                    stack.pop()
                    order.append(node)

    visited = [False] * total_nodes
    comp_id = [0] * total_nodes
    current_id = 0
    
    for node in reversed(order):
        if not visited[node]:
            current_id += 1
            stack = [node]
            visited[node] = True
            while stack:
                curr = stack.pop()
                comp_id[curr] = current_id
                for neighbor in adj[curr]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        stack.append(neighbor)

    assignment = [0] * (n + 1)
    possible = True
    
    for i in range(1, n + 1):
        pos_node = i
        neg_node = n + i
        
        if comp_id[pos_node] == comp_id[neg_node]:
            possible = False
            break
        
        if comp_id[pos_node] > comp_id[neg_node]:
            assignment[i] = 1
        else:
            assignment[i] = 0

    if not possible:
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, assignment[1:])))

if __name__ == "__main__":
    solve()
