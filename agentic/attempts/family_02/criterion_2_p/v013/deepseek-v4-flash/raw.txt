We can model each rule as a 2-SAT clause:  
A rule `(l1, l2)` means at least one of the two literals must be true.  
This is exactly a 2-SAT problem.

We’ll build an implication graph and use SCC (Tarjan’s or Kosaraju’s) to check satisfiability and produce a valid assignment.

```python
import sys
sys.setrecursionlimit(1 << 20)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 0..n-1 for x_i true, n..2n-1 for x_i false

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def node(lit):
        # lit > 0: variable (lit-1) true
        # lit < 0: variable (-lit-1) false
        var = abs(lit) - 1
        if lit > 0:
            return var
        else:
            return var + n

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    for _ in range(m):
        a, b = map(int, input().split())
        # clause (a or b) => (!a -> b) and (!b -> a)
        na = -a
        nb = -b
        add_imp(node(na), node(b))
        add_imp(node(nb), node(a))

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[i] == comp[i + n]:
            print("NO")
            return

    # Build assignment: choose comp order
    assignment = [False] * n
    for i in range(n):
        # If comp[i] > comp[i+n], set true; else false
        assignment[i] = comp[i] > comp[i + n]

    print("YES")
    print(" ".join("1" if assignment[i] else "0" for i in range(n)))

if __name__ == "__main__":
    main()
```
