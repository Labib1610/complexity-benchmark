```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n  # false = 2*i, true = 2*i+1 (i from 0)
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def lit_node(x: int) -> int:
        v = abs(x) - 1
        if x > 0:   # true literal
            return 2 * v + 1
        else:       # false literal (¬var)
            return 2 * v

    def neg(node: int) -> int:
        return node ^ 1  # toggle last bit

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        na = lit_node(a)
        nb = lit_node(b)
        # (na OR nb) => (¬na -> nb) and (¬nb -> na)
        g[neg(na)].append(nb)
        rg[nb].append(neg(na))
        g[neg(nb)].append(na)
        rg[na].append(neg(nb))

    sys.setrecursionlimit(1 << 25)
    visited = [False] * N
    order = []

    def dfs(v: int):
        visited[v] = True
        for to in g[v]:
            if not visited[to]:
                dfs(to)
        order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cur = 0

    def rdfs(v: int, label: int):
        comp[v] = label
        for to in rg[v]:
            if comp[to] == -1:
                rdfs(to, label)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cur)
            cur += 1

    assignment = [0] * n
    for i in range(n):
        false_node = 2 * i
        true_node = false_node + 1
        if comp[false_node] == comp[true_node]:
            print("NO")
            return
        # component with larger topological order (higher label) is earlier in reverse order
        assignment[i] = 1 if comp[true_node] > comp[false_node] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
