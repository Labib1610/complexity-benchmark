import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT: variables 1..n, negation handled by x^1
    # literal v>0 -> 2*(v-1), v<0 -> 2*(-v-1)+1
    size = 2 * n
    head = [-1] * size
    to = []
    nxt = []
    
    def add_edge(a, b):
        to.append(b)
        nxt.append(head[a])
        head[a] = len(to) - 1
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        # literal to node index
        def lit_node(x):
            if x > 0:
                return 2 * (x - 1)
            else:
                return 2 * (-x - 1) + 1
        u = lit_node(l1)
        v = lit_node(l2)
        # clause (l1 or l2): add ~l1 -> l2 and ~l2 -> l1
        add_edge(u ^ 1, v)
        add_edge(v ^ 1, u)
    
    # Tarjan's SCC
    index_counter = [0]
    stack = []
    on_stack = [False] * size
    indices = [-1] * size
    lowlink = [0] * size
    scc_id = [-1] * size
    scc_count = [0]
    
    # iterative DFS
    for start in range(size):
        if indices[start] != -1:
            continue
        call_stack = [(start, head[start], False)]
        while call_stack:
            node, edge_ptr, processed = call_stack.pop()
            if processed:
                # update lowlink from children
                ptr = head[node]
                while ptr != -1:
                    w = to[ptr]
                    if on_stack[w] and scc_id[w] == -1:
                        if lowlink[w] < lowlink[node]:
                            lowlink[node] = lowlink[w]
                    ptr = nxt[ptr]
                if lowlink[node] == indices[node]:
                    while True:
                        w = stack.pop()
                        on_stack[w] = False
                        scc_id[w] = scc_count[0]
                        if w == node:
                            break
                    scc_count[0] += 1
                continue
            
            if indices[node] != -1:
                continue
            
            indices[node] = index_counter[0]
            lowlink[node] = index_counter[0]
            index_counter[0] += 1
            stack.append(node)
            on_stack[node] = True
            
            # push processed marker
            call_stack.append((node, edge_ptr, True))
            
            ptr = edge_ptr
            while ptr != -1:
                w = to[ptr]
                if indices[w] == -1:
                    call_stack.append((w, head[w], False))
                elif on_stack[w]:
                    if indices[w] < lowlink[node]:
                        lowlink[node] = indices[w]
                ptr = nxt[ptr]
    
    # Check satisfiability
    for i in range(n):
        if scc_id[2*i] == scc_id[2*i+1]:
            print("NO")
            return
    
    # Build assignment: if scc_id[x] > scc_id[~x], x is true
    assignment = []
    for i in range(n):
        val = 1 if scc_id[2*i] > scc_id[2*i+1] else 0
        assignment.append(str(val))
    
    print("YES")
    print(" ".join(assignment))

solve()
