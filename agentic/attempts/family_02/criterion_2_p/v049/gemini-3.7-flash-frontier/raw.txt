```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    radj = [[] for _ in range(num_nodes)]
    
    def lit_to_node(lit):
        if lit > 0:
            return 2 * (lit - 1), 2 * (lit - 1) + 1
        else:
            return 2 * (-lit - 1) + 1, 2 * (-lit - 1)

    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx + 1])
        idx += 2
        
        pos1, neg1 = lit_to_node(l1)
        pos2, neg2 = lit_to_node(l2)
        
        # ~l1 => l2
        adj[neg1].append(pos2)
        radj[pos2].append(neg1)
        # ~l2 => l1
        adj[neg2].append(pos1)
        radj[pos1].append(neg2)

    # Kosaraju's algorithm (iterative)
    visited = [False] * num_nodes
    order = []
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_idx = [0] * num_nodes
            visited[i] = True
            while stack:
                u = stack[-1]
                if edge_idx[u] < len(adj[u]):
                    v = adj[u][edge_idx[u]]
                    edge_idx[u] += 1
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
                else:
                    order.append(u)
                    stack.pop()
                    
    scc = [-1] * num_nodes
    scc_id = 0
    
    for i in reversed(order):
        if scc[i] == -1:
            stack = [i]
            scc[i] = scc_id
            while stack:
                u = stack.pop()
                for v in radj[u]:
                    if scc[v] == -1:
                        scc[v] = scc_id
                        stack.append(v)
            scc_id += 1

    ans = [0] * n
    for i in range(n):
        if scc[2 * i] == scc[2 * i + 1]:
            print("NO")
            return
        ans[i] = 1 if scc[2 * i] > scc[2 * i + 1] else 0

    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
```
