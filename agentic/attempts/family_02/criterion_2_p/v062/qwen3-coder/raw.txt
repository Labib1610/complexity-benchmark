import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        clauses.append((l1, l2))
        index += 2
    
    # Build implication graph
    # Variables: 1 to n (positive), -1 to -n (negative)
    # Nodes: 0 to 2*n - 1
    # Node i represents variable i+1, node i+n represents negation of variable i+1
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    def var_to_node(v):
        if v > 0:
            return v - 1
        else:
            return (-v - 1) + n
    
    def node_to_var(node):
        if node < n:
            return node + 1
        else:
            return -(node - n + 1)
    
    # For each clause (a OR b), we add edges (!a -> b) and (!b -> a)
    for l1, l2 in clauses:
        # !l1 -> l2
        adj[var_to_node(-l1)].append(var_to_node(l2))
        rev_adj[var_to_node(l2)].append(var_to_node(-l1))
        # !l2 -> l1
        adj[var_to_node(-l2)].append(var_to_node(l1))
        rev_adj[var_to_node(l1)].append(var_to_node(-l2))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check if satisfiable
    for i in range(n):
        if scc_id[i] == scc_id[i + n]:
            print("NO")
            return
    
    # Assign values
    assignment = [None] * n
    # Process SCCs in reverse topological order
    # We can do this by processing scc_id in order
    scc_order = [[] for _ in range(scc_count)]
    for i in range(2 * n):
        scc_order[scc_id[i]].append(i)
    
    for comp in scc_order:
        for node in comp:
            var_index = node % n
            if assignment[var_index] is None:
                if node < n:
                    assignment[var_index] = 0  # False
                else:
                    assignment[var_index] = 1  # True
    
    print("YES")
    print(' '.join(str(x) for x in assignment))

if __name__ == "__main__":
    main()
