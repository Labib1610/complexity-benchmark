We can model this as a **2-SAT** problem.  
Each species corresponds to a boolean variable, and each criterion is a clause of two literals.  
We need to find a satisfying assignment.

We’ll use implication graph + SCC (Tarjan’s or Kosaraju’s) to solve it in linear time.

```python
import sys
sys.setrecursionlimit(1 << 20)
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # nodes: 2*i for false, 2*i+1 for true

def var(i):
    return 2 * i  # false
def var_not(i):
    return 2 * i + 1  # true

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

def add_imp(u, v):
    g[u].append(v)
    gr[v].append(u)

def add_clause(a, b):
    # a or b
    # not a -> b
    # not b -> a
    if a > 0:
        a_lit = var_not(a - 1)
        a_not = var(a - 1)
    else:
        a_lit = var(-a - 1)
        a_not = var_not(-a - 1)
    if b > 0:
        b_lit = var_not(b - 1)
        b_not = var(b - 1)
    else:
        b_lit = var(-b - 1)
        b_not = var_not(-b - 1)
    add_imp(a_not, b_lit)
    add_imp(b_not, a_lit)

for _ in range(m):
    l1, l2 = map(int, input().split())
    add_clause(l1, l2)

# Kosaraju's SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for v in g[u]:
        if not visited[v]:
            dfs1(v)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N
def dfs2(u, cid):
    comp[u] = cid
    for v in gr[u]:
        if comp[v] == -1:
            dfs2(v, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[var(i)] == comp[var_not(i)]:
        print("NO")
        sys.exit()

print("YES")
res = []
for i in range(n):
    # if var_not(i) comes after var(i) in topological order => true
    if comp[var_not(i)] > comp[var(i)]:
        res.append(1)
    else:
        res.append(0)
print(" ".join(map(str, res)))
```
