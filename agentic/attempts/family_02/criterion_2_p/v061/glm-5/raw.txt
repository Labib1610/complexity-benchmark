import sys

sys.setrecursionlimit(3000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n)]
    
    def get_node(x):
        if x > 0:
            return 2 * (x - 1)
        else:
            return 2 * (-x - 1) + 1
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        u = get_node(l1)
        v = get_node(l2)
        nu = u ^ 1
        nv = v ^ 1
        adj[nu].append(v)
        adj[nv].append(u)
    
    visited = [False] * (2 * n)
    order = []
    
    def dfs(u):
        stack = [u]
        while stack:
            curr = stack[-1]
            if not visited[curr]:
                visited[curr] = True
                for v in adj[curr]:
                    if not visited[v]:
                        stack.append(v)
                continue
            if stack[-1] == curr:
                stack.pop()
                order.append(curr)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs(i)
    
    visited = [False] * (2 * n)
    comp_id = [-1] * (2 * n)
    cid = 0
    
    adj_rev = [[] for _ in range(2 * n)]
    for u in range(2 * n):
        for v in adj[u]:
            adj_rev[v].append(u)
    
    for u in reversed(order):
        if not visited[u]:
            stack = [u]
            visited[u] = True
            component_nodes = []
            while stack:
                node = stack.pop()
                component_nodes.append(node)
                for v in adj_rev[node]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            for node in component_nodes:
                comp_id[node] = cid
            cid += 1
    
    assignment = [0] * n
    for i in range(n):
        u = 2 * i
        v = 2 * i + 1
        if comp_id[u] == comp_id[v]:
            print("NO")
            return
        if comp_id[u] < comp_id[v]:
            assignment[i] = 1
    
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
