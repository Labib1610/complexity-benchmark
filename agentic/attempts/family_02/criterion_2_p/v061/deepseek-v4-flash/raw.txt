We can model each dependency as a 2-SAT clause:  

- A positive literal `+i` means variable `i` is true (exothermic).  
- A negative literal `-i` means variable `i` is false (endothermic).  
- A dependency `(l1, l2)` means at least one of them must be true → clause `(l1 OR l2)`.  

We then solve the 2-SAT problem using implication graph and SCC (Tarjan or Kosaraju).  

```python
import sys
sys.setrecursionlimit(1 << 20)
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # each variable has two nodes: i*2 for false, i*2+1 for true

def node(lit):
    var = abs(lit) - 1
    return var * 2 + (1 if lit > 0 else 0)

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

for _ in range(m):
    a, b = map(int, input().split())
    u = node(a)
    v = node(b)
    # (not a -> b) and (not b -> a)
    g[u ^ 1].append(v)
    g[v ^ 1].append(u)
    gr[v].append(u ^ 1)
    gr[u].append(v ^ 1)

# Kosaraju SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for w in g[u]:
        if not visited[w]:
            dfs1(w)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N
def dfs2(u, cid):
    comp[u] = cid
    for w in gr[u]:
        if comp[w] == -1:
            dfs2(w, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2 * i] == comp[2 * i + 1]:
        print("NO")
        sys.exit()

print("YES")
res = []
for i in range(n):
    # If true node has higher comp id, assign true
    res.append(1 if comp[2 * i + 1] > comp[2 * i] else 0)
print(*res)
```
