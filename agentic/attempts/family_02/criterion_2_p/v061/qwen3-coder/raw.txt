```python
import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each variable x_i, we have two literals: x_i (positive) and not x_i (negative)
    # We'll build a graph for 2-SAT
    # Variables: for reactor i, we use:
    # variable i -> x_i = 1 (exothermic)
    # variable -i -> x_i = 0 (endothermic)
    
    # For each constraint (l1 or l2), we add edges:
    # not l1 -> l2
    # not l2 -> l1
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        # Add clauses: (l1 or l2) means (not not l1 or l2) and (not not l2 or l1)
        # Which translates to edges: not l1 -> l2 and not l2 -> l1
        adj[-l1].append(l2)
        adj[-l2].append(l1)
        rev_adj[l2].append(-l1)
        rev_adj[l1].append(-l2)
        
    # Kosaraju's algorithm for SCC
    visited = set()
    order = []
    
    def dfs1(v):
        if v in visited:
            return
        visited.add(v)
        for u in adj[v]:
            dfs1(u)
        order.append(v)
        
    all_nodes = set(adj.keys()) | set(rev_adj.keys())
    for v in all_nodes:
        dfs1(v)
    
    visited.clear()
    scc = {}
    component = 0
    
    def dfs2(v):
        if v in visited:
            return
        visited.add(v)
        scc[v] = component
        for u in rev_adj[v]:
            dfs2(u)
            
    while order:
        v = order.pop()
        if v not in visited:
            dfs2(v)
            component += 1
            
    # Check for contradiction
    for i in range(1, n+1):
        if scc.get(i, -1) == scc.get(-i, -2):
            print("NO")
            return
            
    # Build assignment
    assignment = {}
    # Process components in reverse topological order
    comp_to_nodes = defaultdict(list)
    for node, c in scc.items():
        comp_to_nodes[c].append(node)
        
    sorted_components = sorted(comp_to_nodes.keys(), key=lambda x: -x)
    
    for c in sorted_components:
        for node in comp_to_nodes[c]:
            if abs(node) <= n and abs(node) > 0:
                var = abs(node)
                if var not in assignment:
                    # If not assigned yet, assign based on sign
                    if node > 0:
                        assignment[var] = 1
                    else:
                        assignment[var] = 0
                        
    print("YES")
    result = []
    for i in range(1, n+1):
        result.append(str(assignment.get(i, 0)))  # default to 0 if not constrained
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
