```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    # 2-SAT construction
    # literal x_i -> 2 * (i - 1)
    # literal ~x_i -> 2 * (i - 1) + 1
    def to_lit(x):
        if x > 0:
            return 2 * (x - 1)
        else:
            return 2 * (-x - 1) + 1

    adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        
        lu = to_lit(u)
        lv = to_lit(v)
        
        # lu or lv  <=>  ~lu -> lv and ~lv -> lu
        adj[lu ^ 1].append(lv)
        adj[lv ^ 1].append(lu)
        
    # Tarjan's SCC algorithm (iterative)
    num_nodes = 2 * n
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    scc = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stack = []
    
    timer = 0
    scc_cnt = 0
    
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue
        
        call_stack = [(i, 0)]
        dfn[i] = low[i] = timer
        timer += 1
        stack.append(i)
        in_stack[i] = True
        
        while call_stack:
            u, edge_idx = call_stack[-1]
            if edge_idx < len(adj[u]):
                call_stack[-1] = (u, edge_idx + 1)
                v = adj[u][edge_idx]
                if dfn[v] == -1:
                    dfn[v] = low[v] = timer
                    timer += 1
                    stack.append(v)
                    in_stack[v] = True
                    call_stack.append((v, 0))
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]
            else:
                call_stack.pop()
                if call_stack:
                    p = call_stack[-1][0]
                    if low[u] < low[p]:
                        low[p] = low[u]
                
                if low[u] == dfn[u]:
                    while True:
                        node = stack.pop()
                        in_stack[node] = False
                        scc[node] = scc_cnt
                        if node == u:
                            break
                    scc_cnt += 1

    ans = []
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        # Tarjan assigns smaller scc index to sinks
        if scc[pos] < scc[neg]:
            ans.append("1")
        else:
            ans.append("0")
            
    print("YES")
    print(" ".join(ans))

if __name__ == '__main__':
    solve()
```
