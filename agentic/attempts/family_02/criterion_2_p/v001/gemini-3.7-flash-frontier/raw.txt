import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    rev_adj = [[] for _ in range(num_nodes)]
    
    def to_var(lit):
        if lit > 0:
            return 2 * (lit - 1)
        else:
            return 2 * (-lit - 1) + 1

    def neg(v):
        return v ^ 1

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        u = to_var(l1)
        v = to_var(l2)
        
        # ~u -> v
        nu = neg(u)
        adj[nu].append(v)
        rev_adj[v].append(nu)
        
        # ~v -> u
        nv = neg(v)
        adj[nv].append(u)
        rev_adj[u].append(nv)

    # Kosaraju's Algorithm
    order = []
    visited = [False] * num_nodes
    
    for i in range(num_nodes):
        if not visited[i]:
            visited[i] = True
            call_stack = [(i, iter(adj[i]))]
            while call_stack:
                u, it = call_stack[-1]
                try:
                    nxt = next(it)
                    if not visited[nxt]:
                        visited[nxt] = True
                        call_stack.append((nxt, iter(adj[nxt])))
                except StopIteration:
                    order.append(u)
                    call_stack.pop()

    scc = [-1] * num_nodes
    scc_count = 0
    
    for u in reversed(order):
        if scc[u] == -1:
            scc[u] = scc_count
            stack = [u]
            while stack:
                curr = stack.pop()
                for nxt in rev_adj[curr]:
                    if scc[nxt] == -1:
                        scc[nxt] = scc_count
                        stack.append(nxt)
            scc_count += 1

    assignment = []
    for i in range(n):
        pos = 2 * i
        neg_val = 2 * i + 1
        if scc[pos] == scc[neg_val]:
            print("NO")
            return
        if scc[pos] < scc[neg_val]:
            assignment.append(1)
        else:
            assignment.append(0)

    print("YES")
    print(*(assignment))

if __name__ == '__main__':
    solve()
