We can model this as a **2-SAT** problem.  
Each species has two possible states: Humid (1) or Arid (0).  
Each criterion is a clause of the form `(a OR b)`, where `a` and `b` are literals (positive or negative).  

We can convert each clause into implications and then run SCC (Strongly Connected Components) to check satisfiability and extract a valid assignment.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(10**6)
input = sys.stdin.readline

n, m = map(int, input().split())
N = 2 * n  # each variable has two nodes: 2*i for false, 2*i+1 for true

def node(var):
    # var > 0: species var is Humid (True)
    # var < 0: species |var| is Arid (False)
    v = abs(var) - 1
    if var > 0:
        return 2 * v + 1  # True node
    else:
        return 2 * v      # False node

g = [[] for _ in range(N)]
gr = [[] for _ in range(N)]

for _ in range(m):
    a, b = map(int, input().split())
    # clause (a OR b) => (!a -> b) and (!b -> a)
    na = node(-a)
    nb = node(-b)
    pa = node(a)
    pb = node(b)
    g[na].append(pb)
    gr[pb].append(na)
    g[nb].append(pa)
    gr[pa].append(nb)

# Kosaraju's SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for v in g[u]:
        if not visited[v]:
            dfs1(v)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N
def dfs2(u, cid):
    comp[u] = cid
    for v in gr[u]:
        if comp[v] == -1:
            dfs2(v, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2*i] == comp[2*i+1]:
        print("NO")
        sys.exit()

print("YES")
res = []
for i in range(n):
    # If True node comes after False node in topological order, assign True
    if comp[2*i+1] > comp[2*i]:
        res.append(1)
    else:
        res.append(0)
print(' '.join(map(str, res)))
```
