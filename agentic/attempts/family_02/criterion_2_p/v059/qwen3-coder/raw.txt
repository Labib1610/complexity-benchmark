```python
import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each database i, we have two variables:
    # x_i represents database i being synchronous (1)
    # not x_i represents database i being asynchronous (0)
    
    # Each clause is of the form (a OR b), where a and b are literals
    # This is a 2-SAT problem
    
    # We'll build implication graph
    # For clause (a OR b), we add edges (not a -> b) and (not b -> a)
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    for _ in range(m):
        v1, v2 = map(int, sys.stdin.readline().split())
        # v1 and v2 are literals
        # v > 0 means variable v is TRUE (synchronous)
        # v < 0 means variable |v| is FALSE (asynchronous)
        
        # Clause is (x_{|v1|} == (v1>0)) OR (x_{|v2|} == (v2>0))
        # Which is equivalent to (x_{|v1|} OR x_{|v2|}) if both v1,v2 > 0
        # Or (x_{|v1|} OR NOT x_{|v2|}) if v1>0, v2<0
        # etc.
        
        # In general, clause is (lit1 OR lit2)
        # Which gives us implications: (NOT lit1 -> lit2) and (NOT lit2 -> lit1)
        
        # Function to get node id for literal
        def lit_to_node(v):
            if v > 0:
                return 2 * v  # positive literal x_v
            else:
                return 2 * (-v) + 1  # negative literal NOT x_v
        
        # Function to get negation of literal
        def negate_lit(v):
            if v > 0:
                return -v
            else:
                return -v  # --v = v
        
        # Add implications for clause (lit1 OR lit2)
        lit1, lit2 = v1, v2
        not_lit1 = negate_lit(lit1)
        not_lit2 = negate_lit(lit2)
        
        # not_lit1 -> lit2
        adj[lit_to_node(not_lit1)].append(lit_to_node(lit2))
        rev_adj[lit_to_node(lit2)].append(lit_to_node(not_lit1))
        
        # not_lit2 -> lit1
        adj[lit_to_node(not_lit2)].append(lit_to_node(lit1))
        rev_adj[lit_to_node(lit1)].append(lit_to_node(not_lit2))
    
    # Now we do Kosaraju's algorithm to find strongly connected components
    visited = [False] * (2 * n + 2)
    order = []
    
    # First DFS on original graph
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    # Second DFS on reversed graph
    scc_id = [-1] * (2 * n + 2)
    scc_count = 0
    
    def dfs2(u, cid):
        scc_id[u] = cid
        for v in rev_adj[u]:
            if scc_id[v] == -1:
                dfs2(v, cid)
    
    # Run first DFS
    for i in range(1, n + 1):
        # Node 2*i represents x_i (database i is synchronous)
        # Node 2*i+1 represents NOT x_i (database i is asynchronous)
        if not visited[2 * i]:
            dfs1(2 * i)
        if not visited[2 * i + 1]:
            dfs1(2 * i + 1)
    
    # Run second DFS in reverse order
    for u in reversed(order):
        if scc_id[u] == -1:
            dfs2(u, scc_count)
            scc_count += 1
    
    # Check if satisfiable
    for i in range(1, n + 1):
        if scc_id[2 * i] == scc_id[2 * i + 1]:
            print("NO")
            return
    
    # Construct solution
    # Assign values based on SCC completion order
    assignment = [None] * (n + 1)
    
    # Create mapping from SCC id to finish time in second DFS
    scc_finish = {}
    for i in range(scc_count):
        scc_finish[i] = 0
    
    # Actually, let's re-think this.
    # In Kosaraju, if a variable and its negation are in the same SCC, it's unsat
    # Otherwise, if SCC of variable finishes AFTER SCC of its negation, then variable is TRUE
    
    for i in range(1, n + 1):
        scc_x_i = scc_id[2 * i]
        scc_not_x_i = scc_id[2 * i + 1]
        # The one in later finishing SCC is true
        if scc_x_i > scc_not_x_i:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(str(assignment[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
```
