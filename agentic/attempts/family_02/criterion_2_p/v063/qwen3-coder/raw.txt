import sys
sys.setrecursionlimit(200000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
    
    # Build implication graph
    # Variables: 1 to n (positive), -1 to -n (negative)
    # Nodes: 0 to 2*n - 1
    # Node i represents variable i+1 if i < n, and -(i-n+1) if i >= n
    # But let's use a clearer mapping:
    # For variable x (1-based), 
    # node 2*(x-1) represents x positive (x is true)
    # node 2*(x-1)+1 represents x negative (x is false, or ~x)
    
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    # For each clause (l1 or l2), we add edges (~l1 -> l2) and (~l2 -> l1)
    for l1, l2 in clauses:
        # Convert literals to nodes
        # l1: if positive, node = 2*(l1-1); if negative, node = 2*(-l1-1)+1
        if l1 > 0:
            node_l1_pos = 2 * (l1 - 1)
            node_l1_neg = node_l1_pos + 1
        else:
            node_l1_neg = 2 * (-l1 - 1)
            node_l1_pos = node_l1_neg + 1
            
        if l2 > 0:
            node_l2_pos = 2 * (l2 - 1)
            node_l2_neg = node_l2_pos + 1
        else:
            node_l2_neg = 2 * (-l2 - 1)
            node_l2_pos = node_l2_neg + 1
        
        # Add implications: (~l1 -> l2) and (~l2 -> l1)
        adj[node_l1_neg].append(node_l2_pos)
        adj[node_l2_neg].append(node_l1_pos)
        rev_adj[node_l2_pos].append(node_l1_neg)
        rev_adj[node_l1_pos].append(node_l2_neg)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
        
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
            
    # Second DFS on reverse graph
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
                
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
            
    # Check for contradiction: variable and its negation in same SCC
    for i in range(n):
        pos_node = 2 * i
        neg_node = 2 * i + 1
        if scc_id[pos_node] == scc_id[neg_node]:
            print("NO")
            return
            
    # Assign values based on SCC topological order
    # Higher SCC id means higher priority in reverse topological order
    assignment = [-1] * n
    
    # Process SCCs in reverse topological order (from highest id to lowest)
    for i in range(2 * n):
        var_index = i // 2
        is_neg = i % 2
        if assignment[var_index] == -1:
            if is_neg:  # This is the negative node
                assignment[var_index] = 1  # So variable is true
            else:
                assignment[var_index] = 0  # Variable is false
                
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
