import sys

# Increase recursion depth just in case
sys.setrecursionlimit(300000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    num_vars = 2 * n

    # Adjacency lists
    head = [-1] * num_vars
    to = []
    nxt = []

    head_rev = [-1] * num_vars
    to_rev = []
    nxt_rev = []

    def add_edge(u, v):
        nxt.append(head[u])
        to.append(v)
        head[u] = len(to) - 1

        nxt_rev.append(head_rev[v])
        to_rev.append(u)
        head_rev[v] = len(to_rev) - 1

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2

        # Convert to 0-based literal index
        # switch x (1-based): active -> 2*(x-1), inactive -> 2*(x-1)+1
        if l1 > 0:
            u = 2 * (l1 - 1)
        else:
            u = 2 * (-l1 - 1) + 1

        if l2 > 0:
            v = 2 * (l2 - 1)
        else:
            v = 2 * (-l2 - 1) + 1

        # (u or v) <=> (~u -> v) and (~v -> u)
        add_edge(u ^ 1, v)
        add_edge(v ^ 1, u)

    # Kosaraju's algorithm - Pass 1: Order by finish time
    order = []
    visited = [False] * num_vars

    for start in range(num_vars):
        if visited[start]:
            continue
        # Iterative DFS
        stack = [(start, head[start])]
        visited[start] = True
        while stack:
            u, e = stack[-1]
            if e != -1:
                stack[-1] = (u, nxt[e])
                v = to[e]
                if not visited[v]:
                    visited[v] = True
                    stack.append((v, head[v]))
            else:
                order.append(u)
                stack.pop()

    # Pass 2: Assign SCC IDs
    scc = [-1] * num_vars
    scc_count = 0

    for start in reversed(order):
        if scc[start] != -1:
            continue
        scc[start] = scc_count
        stack = [start]
        while stack:
            u = stack.pop()
            e = head_rev[u]
            while e != -1:
                v = to_rev[e]
                if scc[v] == -1:
                    scc[v] = scc_count
                    stack.append(v)
                e = nxt_rev[e]
        scc_count += 1

    # Check 2-SAT satisfiability and build solution
    ans = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        # In Kosaraju, scc[x] < scc[~x] means x is topologically after ~x, so x is true
        if scc[pos] < scc[neg]:
            ans[i] = 1
        else:
            ans[i] = 0

    print("YES")
    print(*(ans[i] for i in range(n)))


if __name__ == "__main__":
    solve()
