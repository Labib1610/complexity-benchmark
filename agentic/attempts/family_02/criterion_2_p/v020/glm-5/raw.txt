```python
import sys
sys.setrecursionlimit(2 * 10**5 + 10)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT: variables are 1..n, each has two nodes: 2*i (false), 2*i+1 (true)
    # Implication graph has 2*n nodes
    N = 2 * n
    adj = [[] for _ in range(N)]
    adj_rev = [[] for _ in range(N)]
    
    def get_node(var):
        # var > 0: node 2*var + 1 (true)
        # var < 0: node 2*|var| (false)
        idx = abs(var) - 1
        if var > 0:
            return 2 * idx + 1
        else:
            return 2 * idx
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        # Clause: (u OR v)
        # Add edges: not u -> v, not v -> u
        nu = get_node(-u)
        nv = get_node(-v)
        tu = get_node(u)
        tv = get_node(v)
        adj[nu].append(tv)
        adj_rev[tv].append(nu)
        adj[nv].append(tu)
        adj_rev[tu].append(nv)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                neighbor = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * N
    comp_id = [-1] * N
    current_comp = 0
    
    for u in reversed(order):
        if not visited[u]:
            stack = [u]
            visited[u] = True
            component_nodes = []
            while stack:
                node = stack.pop()
                component_nodes.append(node)
                for neighbor in adj_rev[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        stack.append(neighbor)
            for node in component_nodes:
                comp_id[node] = current_comp
            current_comp += 1
    
    # Check for contradiction
    assignment = [0] * n
    for i in range(n):
        if comp_id[2 * i] == comp_id[2 * i + 1]:
            print("NO")
            return
        assignment[i] = 1 if comp_id[2 * i + 1] > comp_id[2 * i] else 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

solve()
```
