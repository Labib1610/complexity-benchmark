```python
import sys

def solve() -> None:
    sys.setrecursionlimit(1 << 25)
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def lit_to_node(x: int) -> int:
        v = abs(x) - 1
        if x > 0:
            return v * 2      # true
        else:
            return v * 2 + 1  # false

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        A = lit_to_node(a)
        B = lit_to_node(b)
        NA = A ^ 1
        NB = B ^ 1
        # ¬a -> b
        g[NA].append(B)
        rg[B].append(NA)
        # ¬b -> a
        g[NB].append(A)
        rg[A].append(NB)

    visited = [False] * N
    order = []

    # first pass: order by finish time
    stack = []
    for v in range(N):
        if not visited[v]:
            stack.append((v, 0))
            while stack:
                node, state = stack.pop()
                if state == 0:
                    if visited[node]:
                        continue
                    visited[node] = True
                    stack.append((node, 1))
                    for to in g[node]:
                        if not visited[to]:
                            stack.append((to, 0))
                else:
                    order.append(node)

    comp = [-1] * N
    cid = 0
    # second pass: assign components on reversed graph
    for v in reversed(order):
        if comp[v] == -1:
            stack = [v]
            comp[v] = cid
            while stack:
                node = stack.pop()
                for to in rg[node]:
                    if comp[to] == -1:
                        comp[to] = cid
                        stack.append(to)
            cid += 1

    assignment = [0] * n
    for i in range(n):
        t = i * 2
        f = t ^ 1
        if comp[t] == comp[f]:
            print("NO")
            return
        assignment[i] = 1 if comp[t] > comp[f] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
