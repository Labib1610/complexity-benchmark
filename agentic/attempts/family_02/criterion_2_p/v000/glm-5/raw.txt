import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT with n variables, indices 1..n
    # var i true -> node 2*i, false -> node 2*i+1
    size = 2 * n + 2
    adj = [[] for _ in range(size)]
    radj = [[] for _ in range(size)]
    
    def add_or(a, b):
        # (a or b)
        # not a -> b
        # not b -> a
        na = a ^ 1
        nb = b ^ 1
        adj[na].append(b)
        adj[nb].append(a)
        radj[b].append(na)
        radj[a].append(nb)
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        a = 2 * abs(l1) + (0 if l1 > 0 else 1)
        b = 2 * abs(l2) + (0 if l2 > 0 else 1)
        add_or(a, b)
    
    # Kosaraju SCC
    visited = [False] * size
    order = []
    
    sys.setrecursionlimit(1 << 25)
    
    for i in range(2, size):
        if not visited[i]:
            stack = [(i, 0)]
            visited[i] = True
            while stack:
                v, idx = stack[-1]
                if idx < len(adj[v]):
                    to = adj[v][idx]
                    stack[-1] = (v, idx + 1)
                    if not visited[to]:
                        visited[to] = True
                        stack.append((to, 0))
                else:
                    stack.pop()
                    order.append(v)
    
    comp = [-1] * size
    c = 0
    for i in reversed(order):
        if comp[i] == -1:
            stack = [i]
            comp[i] = c
            while stack:
                v = stack.pop()
                for to in radj[v]:
                    if comp[to] == -1:
                        comp[to] = c
                        stack.append(to)
            c += 1
    
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        t_node = 2 * i
        f_node = 2 * i + 1
        if comp[t_node] == comp[f_node]:
            print("NO")
            return
        assignment[i] = 1 if comp[t_node] > comp[f_node] else 0
    
    print("YES")
    print(' '.join(str(assignment[i]) for i in range(1, n + 1)))

solve()
