```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    N = 2 * n + 2
    
    adj = [[] for _ in range(N)]
    adj_rev = [[] for _ in range(N)]
    
    def norm(x):
        if x > 0:
            return 2 * x
        return 2 * (-x) + 1
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        u = norm(l1)
        v = norm(l2)
        
        not_u = u ^ 1
        not_v = v ^ 1
        
        adj[not_u].append(v)
        adj_rev[v].append(not_u)
        
        adj[not_v].append(u)
        adj_rev[u].append(not_v)
    
    # Kosaraju's algorithm
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                nxt = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[nxt]:
                    stack.append((nxt, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(1, 2 * n + 1):
        if not visited[i]:
            dfs1(i)
    
    comp_id = [-1] * N
    current_id = 0
    
    for u in reversed(order):
        if comp_id[u] == -1:
            current_id += 1
            q = deque([u])
            comp_id[u] = current_id
            while q:
                v = q.popleft()
                for nxt in adj_rev[v]:
                    if comp_id[nxt] == -1:
                        comp_id[nxt] = current_id
                        q.append(nxt)
    
    result = [0] * (n + 1)
    
    for i in range(1, n + 1):
        pos = 2 * i
        neg = 2 * i + 1
        if comp_id[pos] == comp_id[neg]:
            sys.stdout.write("NO\n")
            return
        if comp_id[pos] > comp_id[neg]:
            result[i] = 1
    
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, result[1:])) + "\n")

if __name__ == "__main__":
    sys.setrecursionlimit(300000)
    solve()
```
