import sys
sys.setrecursionlimit(1000000)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
    
    # Build implication graph for 2-SAT
    # Variables: 0 to 2*n-1
    # For variable i (0-based), node 2*i is !i, node 2*i+1 is i
    adj = [[] for _ in range(2 * n)]
    adj_rev = [[] for _ in range(2 * n)]
    
    def var_to_node(v):
        # v is signed integer: +k means variable k-1 is True, -k means variable k-1 is False
        # variable index is 0-based
        var_index = abs(v) - 1
        if v > 0:
            return 2 * var_index + 1  # x_k is True
        else:
            return 2 * var_index      # x_k is False
    
    # For each clause (l1 OR l2), we add edges (!l1 -> l2) and (!l2 -> l1)
    for l1, l2 in clauses:
        node_l1 = var_to_node(l1)
        node_l2 = var_to_node(l2)
        not_l1 = node_l1 ^ 1
        not_l2 = node_l2 ^ 1
        adj[not_l1].append(node_l2)
        adj[not_l2].append(node_l1)
        adj_rev[node_l2].append(not_l1)
        adj_rev[node_l1].append(not_l2)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Reverse graph DFS to find SCCs
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    component_nodes = []
    
    def dfs2(u, comp):
        visited[u] = True
        scc_id[u] = scc_count
        comp.append(u)
        for v in adj_rev[u]:
            if not visited[v]:
                dfs2(v, comp)
    
    while order:
        u = order.pop()
        if not visited[u]:
            comp = []
            dfs2(u, comp)
            component_nodes.append(comp)
            scc_count += 1
    
    # Check if any variable and its negation are in the same SCC
    for i in range(n):
        if scc_id[2*i] == scc_id[2*i+1]:
            print("NO")
            return
    
    # Assign values based on topological order of SCCs
    assignment = [-1] * n
    # Process SCCs in reverse topological order (which is the order we found them in Kosaraju)
    for comp in reversed(component_nodes):
        for node in comp:
            var_index = node // 2
            if assignment[var_index] == -1:
                # If node is 2*i+1 (True), set var i to True
                # If node is 2*i (False), set var i to False
                assignment[var_index] = node & 1
    
    print("YES")
    print(' '.join(str(x) for x in assignment))

if __name__ == "__main__":
    main()
