import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # 2-SAT with n variables.
    # Variable i (1-indexed): node 2*i for True, 2*i+1 for False.
    # Literal x: if x > 0 -> 2*x; if x < 0 -> 2*(-x) + 1.
    # Negation: node ^ 1.

    size = 2 * (n + 1)
    graph = [[] for _ in range(size)]

    def lit(x):
        if x > 0:
            return 2 * x
        else:
            return 2 * (-x) + 1

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        la = lit(a)
        lb = lit(b)
        # Clause (a or b): add edges ~a -> b and ~b -> a
        graph[la ^ 1].append(lb)
        graph[lb ^ 1].append(la)

    # Tarjan's SCC
    index_counter = [0]
    stack = []
    on_stack = [False] * size
    index = [-1] * size
    lowlink = [0] * size
    comp = [-1] * size
    comp_counter = [0]

    sys.setrecursionlimit(1 << 25)

    # Iterative Tarjan to avoid recursion depth issues
    for start in range(2, size):
        if index[start] != -1:
            continue
        call_stack = [(start, 0)]
        while call_stack:
            v, pi = call_stack[-1]
            if pi == 0:
                index[v] = index_counter[0]
                lowlink[v] = index_counter[0]
                index_counter[0] += 1
                stack.append(v)
                on_stack[v] = True
            recurse = False
            neighbors = graph[v]
            for i in range(pi, len(neighbors)):
                w = neighbors[i]
                if index[w] == -1:
                    call_stack[-1] = (v, i + 1)
                    call_stack.append((w, 0))
                    recurse = True
                    break
                elif on_stack[w]:
                    if index[w] < lowlink[v]:
                        lowlink[v] = index[w]
            if recurse:
                continue
            if lowlink[v] == index[v]:
                while True:
                    w = stack.pop()
                    on_stack[w] = False
                    comp[w] = comp_counter[0]
                    if w == v:
                        break
                comp_counter[0] += 1
            call_stack.pop()
            if call_stack:
                u = call_stack[-1][0]
                if lowlink[v] < lowlink[u]:
                    lowlink[u] = lowlink[v]

    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        t = 2 * i
        f = 2 * i + 1
        if comp[t] == comp[f]:
            print("NO")
            return
        assignment[i] = 1 if comp[t] < comp[f] else 0

    print("YES")
    print(' '.join(str(assignment[i]) for i in range(1, n + 1)))

solve()
