import sys
sys.setrecursionlimit(10**6)

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    clauses = []
    index = 2
    for _ in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        clauses.append((l1, l2))
        index += 2
    
    # Build implication graph
    # Variables: 1 to n (positive), -1 to -n (negative)
    # Nodes: 0 to 2*n - 1
    # Node i represents variable i+1, node i+n represents negation of variable i+1
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    def var_to_node(v):
        if v > 0:
            return v - 1
        else:
            return (-v - 1) + n
    
    def node_to_var(node):
        if node < n:
            return node + 1
        else:
            return -(node - n + 1)
    
    # For each clause (l1 or l2), we add edges (~l1 -> l2) and (~l2 -> l1)
    for l1, l2 in clauses:
        # ~l1 -> l2
        adj[var_to_node(-l1)].append(var_to_node(l2))
        rev_adj[var_to_node(l2)].append(var_to_node(-l1))
        # ~l2 -> l1
        adj[var_to_node(-l2)].append(var_to_node(l1))
        rev_adj[var_to_node(l1)].append(var_to_node(-l2))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check for contradictions
    for i in range(n):
        if scc_id[i] == scc_id[i + n]:
            print("NO")
            return
    
    # Assign values
    assignment = [None] * n
    scc_order = list(range(scc_count))
    # We need to process SCCs in topological order
    # Since we used Kosaraju, the scc_id already gives reverse topological order
    # So we process from highest scc_id to lowest
    scc_nodes = [[] for _ in range(scc_count)]
    for node in range(2 * n):
        scc_nodes[scc_id[node]].append(node)
    
    assigned = [False] * n
    for sc in reversed(range(scc_count)):
        for node in scc_nodes[sc]:
            var_index = node % n
            if not assigned[var_index]:
                if node < n:  # Positive variable
                    assignment[var_index] = 1
                else:  # Negative variable
                    assignment[var_index] = 0
                assigned[var_index] = True
    
    print("YES")
    print(' '.join(str(x) for x in assignment))

if __name__ == "__main__":
    main()
