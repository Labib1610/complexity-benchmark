import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(2 * n + 2)]

    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        u1 = abs(l1)
        t1 = 1 if l1 > 0 else 0
        nu1 = u1 + (n if t1 == 0 else 0)
        pu1 = u1 + (n if t1 == 1 else 0)

        u2 = abs(l2)
        t2 = 1 if l2 > 0 else 0
        nu2 = u2 + (n if t2 == 0 else 0)
        pu2 = u2 + (n if t2 == 1 else 0)

        adj[nu1].append(pu2)
        adj[nu2].append(pu1)

    visited = [0] * (2 * n + 2)
    order = []

    def dfs(u):
        stack = [(u, 0)]
        while stack:
            curr, state = stack.pop()
            if state == 0:
                if visited[curr] == 2:
                    continue
                if visited[curr] == 1:
                    continue
                visited[curr] = 1
                stack.append((curr, 1))
                for v in adj[curr]:
                    if visited[v] == 0:
                        stack.append((v, 0))
            else:
                visited[curr] = 2
                order.append(curr)

    for i in range(1, 2 * n + 1):
        if visited[i] == 0:
            dfs(i)

    comp_id = [0] * (2 * n + 2)
    visited = [0] * (2 * n + 2)
    cid = 0

    radj = [[] for _ in range(2 * n + 2)]
    for u in range(1, 2 * n + 1):
        for v in adj[u]:
            radj[v].append(u)

    for u in reversed(order):
        if visited[u] == 0:
            cid += 1
            stack = [u]
            visited[u] = 1
            while stack:
                curr = stack.pop()
                comp_id[curr] = cid
                for v in radj[curr]:
                    if visited[v] == 0:
                        visited[v] = 1
                        stack.append(v)

    result = [0] * (n + 1)
    for i in range(1, n + 1):
        if comp_id[i] == comp_id[i + n]:
            print("NO")
            return
        if comp_id[i] > comp_id[i + n]:
            result[i] = 1
        else:
            result[i] = 0

    print("YES")
    print(' '.join(map(str, result[1:])))

solve()
