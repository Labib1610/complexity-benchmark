```python
import sys
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # For each policy (l1 OR l2), we create implications
    # l1 OR l2 is equivalent to (NOT l1 -> l2) AND (NOT l2 -> l1)
    
    # We'll use 2SAT approach
    # Variables: 1..n for upgraded, (n+1)..(2n) for legacy
    # So variable i (1-based) represents service i upgraded
    # Variable i+n represents service i legacy
    
    def var_up(i):
        return i
    
    def var_leg(i):
        return i + n
    
    def negate(v):
        if v <= n:
            return v + n
        else:
            return v - n
    
    # Implication graph: adjacency list
    adj = [[] for _ in range(2 * n + 1)]
    rev_adj = [[] for _ in range(2 * n + 1)]
    
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        # Convert to variables
        if l1 > 0:
            v1 = var_up(l1)
        else:
            v1 = var_leg(-l1)
            
        if l2 > 0:
            v2 = var_up(l2)
        else:
            v2 = var_leg(-l2)
        
        # Add implications: NOT v1 -> v2 and NOT v2 -> v1
        adj[negate(v1)].append(v2)
        adj[negate(v2)].append(v1)
        rev_adj[v2].append(negate(v1))
        rev_adj[v1].append(negate(v2))
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n + 1)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(1, 2 * n + 1):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on transpose graph
    visited = [False] * (2 * n + 1)
    scc_id = [0] * (2 * n + 1)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            scc_count += 1
            dfs2(u)
    
    # Check if satisfiable
    for i in range(1, n + 1):
        if scc_id[var_up(i)] == scc_id[var_leg(i)]:
            print("NO")
            return
    
    # Build solution
    # In the implication graph, if scc_id[u] > scc_id[negate(u)], then u is true
    result = [0] * (n + 1)
    for i in range(1, n + 1):
        if scc_id[var_up(i)] > scc_id[var_leg(i)]:
            result[i] = 1
        else:
            result[i] = 0
    
    print("YES")
    print(' '.join(map(str, result[1:])))

threading.Thread(target=main).start()
```
