```python
import sys

sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    N = 2 * n  # total nodes
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def lit_node(l: int) -> int:
        """return node index for literal l (positive -> true, negative -> false)"""
        v = abs(l) - 1
        if l > 0:
            return 2 * v + 1  # true
        else:
            return 2 * v      # false

    def neg(node: int) -> int:
        """return node index of negated literal"""
        return node ^ 1  # toggle last bit (true<->false)

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        na = lit_node(a)
        nb = lit_node(b)
        # (a or b) => (~a -> b) and (~b -> a)
        na_neg = neg(na)
        nb_neg = neg(nb)
        g[na_neg].append(nb)
        g[nb_neg].append(na)
        rg[nb].append(na_neg)
        rg[na].append(nb_neg)

    visited = [False] * N
    order = []

    sys.setrecursionlimit(1 << 25)

    def dfs(v: int):
        stack = [(v, 0)]
        while stack:
            v, idx = stack.pop()
            if idx == 0:
                if visited[v]:
                    continue
                visited[v] = True
                stack.append((v, 1))  # post processing marker
                for to in g[v]:
                    if not visited[to]:
                        stack.append((to, 0))
            else:
                order.append(v)

    for v in range(N):
        if not visited[v]:
            dfs(v)

    comp = [-1] * N
    cur = 0
    def rdfs(v: int, label: int):
        stack = [v]
        while stack:
            x = stack.pop()
            if comp[x] != -1:
                continue
            comp[x] = label
            for to in rg[x]:
                if comp[to] == -1:
                    stack.append(to)

    for v in reversed(order):
        if comp[v] == -1:
            rdfs(v, cur)
            cur += 1

    assignment = [0] * n
    for i in range(n):
        false_node = 2 * i
        true_node = false_node + 1
        if comp[false_node] == comp[true_node]:
            print("NO")
            return
        # component with larger topological order (higher label) is earlier in reverse topological order
        assignment[i] = 1 if comp[true_node] > comp[false_node] else 0

    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
