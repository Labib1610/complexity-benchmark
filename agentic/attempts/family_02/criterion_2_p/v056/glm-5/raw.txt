```python
import sys
from sys import setrecursionlimit

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT: variables are drones 1..n, each can be 0 (False) or 1 (True)
    # For each drone i: node 2*i = i is True (Heavy-Lift), node 2*i+1 = i is False (Long-Range)
    # A directive (l1, l2) means (l1) OR (l2)
    # If l1 is positive: literal = drone l1 is True
    # If l1 is negative: literal = drone |l1| is False
    
    # Number of nodes = 2*n
    # For each literal, we need implication: NOT l1 -> l2 and NOT l2 -> l1
    # NOT of (drone i is True) = (drone i is False)
    # NOT of (drone i is False) = (drone i is True)
    
    # Node encoding:
    # For drone i (1-indexed): 
    #   node for "i is True" = 2*(i-1)
    #   node for "i is False" = 2*(i-1)+1
    
    def node_true(i):
        return 2 * (i - 1)
    
    def node_false(i):
        return 2 * (i - 1) + 1
    
    def neg(node):
        return node ^ 1
    
    def get_node(lit):
        if lit > 0:
            return node_true(lit)
        else:
            return node_false(-lit)
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    radj = [[] for _ in range(num_nodes)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        u = get_node(l1)
        v = get_node(l2)
        # NOT u -> v
        adj[neg(u)].append(v)
        radj[v].append(neg(u))
        # NOT v -> u
        adj[neg(v)].append(u)
        radj[u].append(neg(v))
    
    # Kosaraju's algorithm
    visited = [False] * num_nodes
    order = []
    
    sys.setrecursionlimit(1 << 25)
    
    # Iterative DFS to avoid recursion limit issues
    for start in range(num_nodes):
        if not visited[start]:
            stack = [(start, 0)]
            while stack:
                node, idx = stack[-1]
                if idx == 0:
                    if visited[node]:
                        stack.pop()
                        continue
                    visited[node] = True
                if idx < len(adj[node]):
                    child = adj[node][idx]
                    stack[-1] = (node, idx + 1)
                    if not visited[child]:
                        stack.append((child, 0))
                else:
                    order.append(node)
                    stack.pop()
    
    visited = [False] * num_nodes
    comp = [-1] * num_nodes
    
    for node in reversed(order):
        if not visited[node]:
            stack = [node]
            visited[node] = True
            component = []
            while stack:
                curr = stack.pop()
                component.append(curr)
                for child in radj[curr]:
                    if not visited[child]:
                        visited[child] = True
                        stack.append(child)
            c = len(component)
            for cnode in component:
                comp[cnode] = c
    
    # Check satisfiability
    assignment = [0] * n
    for i in range(1, n + 1):
        nt = node_true(i)
        nf = node_false(i)
        if comp[nt] == comp[nf]:
            print("NO")
            return
        assignment[i - 1] = 1 if comp[nt] > comp[nf] else 0
    
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
