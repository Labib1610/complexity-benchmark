import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]

    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        u = 2 * (l1 - 1) if l1 > 0 else 2 * (-l1 - 1) + 1
        v = 2 * (l2 - 1) if l2 > 0 else 2 * (-l2 - 1) + 1
        
        # ~u -> v
        neg_u = u ^ 1
        adj[neg_u].append(v)
        rev_adj[v].append(neg_u)
        
        # ~v -> u
        neg_v = v ^ 1
        adj[neg_v].append(u)
        rev_adj[u].append(neg_v)

    visited = [False] * (2 * n)
    order = []

    for i in range(2 * n):
        if not visited[i]:
            stack = [(i, 0)]
            visited[i] = True
            while stack:
                u, edge_idx = stack[-1]
                if edge_idx < len(adj[u]):
                    stack[-1] = (u, edge_idx + 1)
                    v = adj[u][edge_idx]
                    if not visited[v]:
                        visited[v] = True
                        stack.append((v, 0))
                else:
                    stack.pop()
                    order.append(u)

    comp = [-1] * (2 * n)
    comp_id = 0
    for u in reversed(order):
        if comp[u] == -1:
            stack = [u]
            comp[u] = comp_id
            while stack:
                curr = stack.pop()
                for v in rev_adj[curr]:
                    if comp[v] == -1:
                        comp[v] = comp_id
                        stack.append(v)
            comp_id += 1

    res = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if comp[pos] == comp[neg]:
            print("NO")
            return
        res[i] = 1 if comp[pos] > comp[neg] else 0

    print("YES")
    print(*(res))

if __name__ == '__main__':
    solve()
