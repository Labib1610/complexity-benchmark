```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # 2-SAT with n variables, indices 1..n
    # variable i true -> node 2*i, false -> node 2*i+1
    size = 2 * n + 2
    adj = [[] for _ in range(size)]
    radj = [[] for _ in range(size)]

    def node(var):
        # var is non-zero integer, |var| in 1..n
        if var > 0:
            return 2 * var
        else:
            return 2 * (-var) + 1

    def neg(x):
        return x ^ 1

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        # clause (a or b): not a -> b, not b -> a
        na = node(-a)
        nb = node(-b)
        ua = node(a)
        ub = node(b)
        adj[na].append(ub)
        adj[nb].append(ua)
        radj[ub].append(na)
        radj[ua].append(nb)

    # iterative Kosaraju
    order = []
    visited = bytearray(size)
    stack = []

    for i in range(2, size):
        if not visited[i]:
            visited[i] = 1
            stack.append((i, 0))
            while stack:
                v, idx = stack.pop()
                if idx < len(adj[v]):
                    e = adj[v][idx]
                    stack.append((v, idx + 1))
                    if not visited[e]:
                        visited[e] = 1
                        stack.append((e, 0))
                else:
                    order.append(v)

    comp = [-1] * size
    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            comp[u] = cid
            stack.append(u)
            while stack:
                v = stack.pop()
                for w in radj[v]:
                    if comp[w] == -1:
                        comp[w] = cid
                        stack.append(w)
            cid += 1

    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        t = 2 * i
        f = 2 * i + 1
        if comp[t] == comp[f]:
            sys.stdout.write("NO\n")
            return
        assignment[i] = 1 if comp[t] > comp[f] else 0

    out = ["YES"]
    out.append(" ".join(map(str, assignment[1:])))
    sys.stdout.write("\n".join(out) + "\n")

solve()
```
