import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # 2-SAT variable indexing:
    # +i -> 2*(i-1)
    # -i -> 2*(i-1) + 1
    # not(u) = u ^ 1
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    rev_adj = [[] for _ in range(num_nodes)]
    
    def to_literal(x):
        if x > 0:
            return 2 * (x - 1)
        else:
            return 2 * (-x - 1) + 1

    idx = 2
    for _ in range(m):
        c1 = int(input_data[idx])
        c2 = int(input_data[idx + 1])
        idx += 2
        
        u = to_literal(c1)
        v = to_literal(c2)
        
        # (c1 or c2) <=> (~c1 => c2) and (~c2 => c1)
        not_u = u ^ 1
        not_v = v ^ 1
        
        adj[not_u].append(v)
        adj[not_v].append(u)
        rev_adj[v].append(not_u)
        rev_adj[u].append(not_v)
        
    # Kosaraju's algorithm (iterative to avoid recursion limit)
    order = []
    visited = [False] * num_nodes
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_idx = [0] * num_nodes
            visited[i] = True
            
            # DFS using explicit stack
            curr_stack = [i]
            while curr_stack:
                u = curr_stack[-1]
                if edge_idx[u] < len(adj[u]):
                    nxt = adj[u][edge_idx[u]]
                    edge_idx[u] += 1
                    if not visited[nxt]:
                        visited[nxt] = True
                        curr_stack.append(nxt)
                else:
                    order.append(u)
                    curr_stack.pop()

    scc = [-1] * num_nodes
    scc_count = 0
    
    for u in reversed(order):
        if scc[u] == -1:
            curr_stack = [u]
            scc[u] = scc_count
            while curr_stack:
                curr = curr_stack.pop()
                for nxt in rev_adj[curr]:
                    if scc[nxt] == -1:
                        scc[nxt] = scc_count
                        curr_stack.append(nxt)
            scc_count += 1
            
    ans = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        # In Kosaraju, earlier SCC in reversed order has higher topological order
        # So scc[pos] < scc[neg] means pos comes after neg topologically
        ans[i] = 1 if scc[pos] < scc[neg] else 0
        
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
