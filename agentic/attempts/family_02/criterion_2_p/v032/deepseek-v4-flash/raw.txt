We can model this as a **2-SAT** problem.  
Each variable is a subsystem, and each dependency is a clause of two literals.  
We need to find a satisfying assignment.

We’ll use implication graph + SCC (Tarjan’s or Kosaraju’s) to solve it in linear time.

```python
import sys
sys.setrecursionlimit(1 << 20)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # each variable has two nodes: x_i and ¬x_i

    # node mapping: 
    # for variable i (1-indexed), 
    #   positive literal  i  -> node (i-1)*2
    #   negative literal -i  -> node (i-1)*2 + 1
    def node(lit):
        if lit > 0:
            return (lit - 1) * 2
        else:
            return (-lit - 1) * 2 + 1

    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    for _ in range(m):
        a, b = map(int, input().split())
        # clause (a OR b) => (¬a -> b) and (¬b -> a)
        na = -a
        nb = -b
        u = node(na)
        v = node(b)
        g[u].append(v)
        gr[v].append(u)

        u = node(nb)
        v = node(a)
        g[u].append(v)
        gr[v].append(u)

    # Kosaraju's SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for w in g[u]:
            if not visited[w]:
                dfs1(w)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for w in gr[u]:
            if comp[w] == -1:
                dfs2(w, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # In Kosaraju, comp id order is topological for condensation
        # If comp[¬x] > comp[x], then x is true
        if comp[2*i+1] > comp[2*i]:
            assignment[i] = 1
        else:
            assignment[i] = 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
