```python
import sys
from collections import defaultdict

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    # For each toggle, we'll have two variables: x_i (active) and not x_i (dormant)
    # We'll use 2*i for x_i (active) and 2*i+1 for not x_i (dormant)
    # But since toggle ids start from 1, we'll map toggle i to nodes 2*i and 2*i+1
    
    # Build implication graph for 2-SAT
    # Each variable has two nodes: 2*v and 2*v+1 for v and ~v
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    def var(toggle, active):
        # toggle is from 1 to n
        # active: True means we want it active (1), False means dormant (0)
        if active:
            return 2 * toggle
        else:
            return 2 * toggle + 1
            
    def negate(node):
        if node % 2 == 0:  # even, so it's a positive variable
            return node + 1
        else:  # odd, so it's a negated variable
            return node - 1
            
    # For each clause (a OR b), we add edges (~a -> b) and (~b -> a)
    for _ in range(m):
        a = int(data[idx])
        idx += 1
        b = int(data[idx])
        idx += 1
        
        # a represents toggle |a| in state (a>0)
        # b represents toggle |b| in state (b>0)
        
        toggle_a = abs(a)
        active_a = a > 0
        toggle_b = abs(b)
        active_b = b > 0
        
        # Clause: (active_a OR active_b) which is equivalent to (~active_a -> active_b) and (~active_b -> active_a)
        # But we need to be careful about the encoding
        
        # Node for toggle_a being in the required state for a
        node_a = var(toggle_a, active_a)
        # Node for toggle_b being in the required state for b
        node_b = var(toggle_b, active_b)
        
        # The clause is satisfied if at least one of them is true
        # So we have implications: (~node_a -> node_b) and (~node_b -> node_a)
        adj[negate(node_a)].append(node_b)
        adj[negate(node_b)].append(node_a)
        rev_adj[node_b].append(negate(node_a))
        rev_adj[node_a].append(negate(node_b))
        
    # Now we do strongly connected components to check 2-SAT satisfiability
    # Kosaraju's algorithm
    
    visited = [False] * (2 * (n + 1))
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
        
    # First DFS on original graph
    for i in range(2, 2*(n+1)):
        if not visited[i]:
            dfs1(i)
            
    # Reset visited for second DFS
    visited = [False] * (2 * (n + 1))
    scc_id = [0] * (2 * (n + 1))
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
                
    # Second DFS on transpose graph in reverse order
    while order:
        u = order.pop()
        if not visited[u]:
            scc_count += 1
            dfs2(u)
            
    # Check if any variable and its negation are in the same SCC
    for i in range(1, n+1):
        node_pos = 2 * i       # x_i
        node_neg = 2 * i + 1   # ~x_i
        if scc_id[node_pos] == scc_id[node_neg]:
            print("NO")
            return
            
    # Otherwise, there exists a solution
    # Assign values based on SCC completion order
    # Variables in later finishing SCCs are assigned True
    assignment = [None] * (n + 1)
    
    for i in range(1, n+1):
        node_pos = 2 * i
        node_neg = 2 * i + 1
        if scc_id[node_pos] > scc_id[node_neg]:
            assignment[i] = 1  # active
        else:
            assignment[i] = 0  # dormant
            
    print("YES")
    print(' '.join(str(assignment[i]) for i in range(1, n+1)))

solve()
```
