We can model this as a **2-SAT** problem.  
Each module has two possible states: Standard (1) or Auxiliary (0).  
Each interlock is a clause of the form `(condition1 OR condition2)`, where each condition is either `x` (module x is Standard) or `¬x` (module x is Auxiliary).

We need to check if there is a satisfying assignment.

We’ll use implication graph and Kosaraju’s algorithm for SCC to solve 2-SAT in O(n + m).

```python
import sys
sys.setrecursionlimit(1 << 25)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # each variable has two nodes: x_i (even) and not x_i (odd)
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def node(var):
        # var > 0: x_var, var < 0: not x_var
        idx = abs(var) - 1
        if var > 0:
            return 2 * idx
        else:
            return 2 * idx + 1

    def neg_node(u):
        return u ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        u = node(a)
        v = node(b)
        # (a OR b) => (¬a -> b) and (¬b -> a)
        g[neg_node(u)].append(v)
        g[neg_node(v)].append(u)
        gr[v].append(neg_node(u))
        gr[u].append(neg_node(v))

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for w in g[u]:
            if not visited[w]:
                dfs1(w)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for w in gr[u]:
            if comp[w] == -1:
                dfs2(w, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return

    # Build assignment: if comp[x] > comp[¬x] then x = True
    assignment = [0] * n
    for i in range(n):
        assignment[i] = 1 if comp[2 * i] > comp[2 * i + 1] else 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
