```python
import sys
from collections import deque

# Increase recursion depth just in case, though we use iterative approach.
sys.setrecursionlimit(200000)

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Indices:
    # Module i (1-based):
    #   Standard (1): index 2*i - 2
    #   Auxiliary (0): index 2*i - 1
    # Total nodes: 2*n
    
    # Graph representation: Adjacency lists
    adj = [[] for _ in range(2 * n)]
    
    def get_node_index(module_idx, state):
        """
        module_idx: 1-based integer
        state: 1 for Standard (positive input), 0 for Auxiliary (negative input)
        Returns 0-based index for the graph
        """
        if state == 1:
            return 2 * (module_idx - 1)
        else:
            return 2 * (module_idx - 1) + 1

    def parse_constraint(val):
        """
        Parses an integer from input.
        Returns (module_index, state)
        module_index is 1-based.
        state is 1 for positive, 0 for negative.
        """
        if val > 0:
            return (val, 1)
        else:
            return (-val, 0)
    
    # Build the implication graph
    # Clause: (a or b)
    # Implications: not a -> b, not b -> a
    for _ in range(m):
        try:
            c1 = int(next(iterator))
            c2 = int(next(iterator))
            
            u_mod, u_state = parse_constraint(c1)
            v_mod, v_state = parse_constraint(c2)
            
            u_node = get_node_index(u_mod, u_state)
            v_node = get_node_index(v_mod, v_state)
            
            # not u -> v
            # not u is same module, opposite state
            not_u_node = u_node ^ 1
            not_v_node = v_node ^ 1
            
            adj[not_u_node].append(v_node)
            adj[not_v_node].append(u_node)
            
        except StopIteration:
            break

    # Kosaraju's Algorithm to find Strongly Connected Components
    
    # 1. First Pass DFS (fill order stack)
    visited = [False] * (2 * n)
    order_stack = []
    
    for i in range(2 * n):
        if not visited[i]:
            # Iterative DFS to avoid recursion limit issues
            stack = [(i, False)] # (node, processed_flag)
            while stack:
                curr, processed = stack.pop()
                if processed:
                    order_stack.append(curr)
                    continue
                if visited[curr]:
                    continue
                visited[curr] = True
                stack.append((curr, True))
                for neighbor in adj[curr]:
                    if not visited[neighbor]:
                        stack.append((neighbor, False))

    # 2. Second Pass DFS (process in reverse order of finishing times on transposed graph)
    # Build Transposed Graph
    trans_adj = [[] for _ in range(2 * n)]
    for u in range(2 * n):
        for v in adj[u]:
            trans_adj[v].append(u)
            
    visited2 = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    current_scc_idx = 0
    
    while order_stack:
        node = order_stack.pop()
        if not visited2[node]:
            # New SCC component
            # DFS on trans_adj
            component_nodes = []
            stack = [node]
            visited2[node] = True
            
            while stack:
                curr = stack.pop()
                component_nodes.append(curr)
                for neighbor in trans_adj[curr]:
                    if not visited2[neighbor]:
                        visited2[neighbor] = True
                        stack.append(neighbor)
            
            for comp_node in component_nodes:
                scc_id[comp_node] = current_scc_idx
            
            current_scc_idx += 1

    # 3. Check for unsatisfiability
    # A variable and its negation must not be in the same SCC
    # Node u and node v=u^1 are logical opposites
    for i in range(n):
        node_true = get_node_index(i + 1, 1)
        node_false = get_node_index(i + 1, 0)
        if scc_id[node_true] == scc_id[node_false]:
            sys.stdout.write("NO\n")
            return

    # 4. Construct Assignment
    # If scc_id[u] > scc_id[v], then u is processed before v in the second pass (topo order)
    # Which means u is "lower" in the topological DAG.
    # We want to set variables reachable from the "higher" nodes to false.
    # So, if scc_id[True_node] > scc_id[False_node], we pick False (0).
    # Else we pick True (1).
    # This simplifies to: assignment[i] = (scc_id[True_node] < scc_id[False_node])
    
    assignment = []
    for i in range(1, n + 1):
        node_true = get_node_index(i, 1)
        node_false = get_node_index(i, 0)
        if scc_id[node_true] < scc_id[node_false]:
            assignment.append(1)
        else:
            assignment.append(0)

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, assignment)) + "\n")

if __name__ == '__main__':
    solve()
```
