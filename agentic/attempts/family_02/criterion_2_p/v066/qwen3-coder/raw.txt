import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # If no interlocks, any assignment works
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build implication graph for 2-SAT
    # Variables: 0 to n-1 for Standard, n to 2n-1 for Auxiliary
    # But we'll use: for variable i (0-indexed), 
    #                2*i represents Standard, 2*i+1 represents Auxiliary
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    for _ in range(m):
        c1, c2 = map(int, sys.stdin.readline().split())
        # Convert to 0-indexed
        mod1 = abs(c1) - 1
        mod2 = abs(c2) - 1
        
        # c1 > 0 means Standard, c1 < 0 means Auxiliary
        # So !Standard = Auxiliary and !Auxiliary = Standard
        # For 2*i = Standard, 2*i+1 = Auxiliary
        # NOT(2*i) = 2*i+1 and NOT(2*i+1) = 2*i
        
        def NOT(x):
            return x ^ 1
        
        # Convert module indices to variable indices
        var1_standard = 2 * mod1
        var1_auxiliary = 2 * mod1 + 1
        var2_standard = 2 * mod2
        var2_auxiliary = 2 * mod2 + 1
        
        # The constraint is (c1 OR c2)
        # If c1 is Standard (positive), then var1_standard is true
        # If c1 is Auxiliary (negative), then var1_auxiliary is true
        # So we need: (var1_standard OR var2_standard) if both positive
        # Or: (var1_standard OR var2_auxiliary) if c1>0, c2<0
        # Or: (var1_auxiliary OR var2_standard) if c1<0, c2>0
        # Or: (var1_auxiliary OR var2_auxiliary) if both negative
        
        # In 2-SAT, (A OR B) is equivalent to (!A -> B) AND (!B -> A)
        
        if c1 > 0 and c2 > 0:  # (Standard1 OR Standard2)
            # (!Standard1 -> Standard2) AND (!Standard2 -> Standard1)
            # (NOT(var1_standard) -> var2_standard) AND (NOT(var2_standard) -> var1_standard)
            graph[NOT(var1_standard)].append(var2_standard)
            graph[NOT(var2_standard)].append(var1_standard)
            reverse_graph[var2_standard].append(NOT(var1_standard))
            reverse_graph[var1_standard].append(NOT(var2_standard))
            
        elif c1 > 0 and c2 < 0:  # (Standard1 OR Auxiliary2)
            # (!Standard1 -> Auxiliary2) AND (!Auxiliary2 -> Standard1)
            graph[NOT(var1_standard)].append(var2_auxiliary)
            graph[NOT(var2_auxiliary)].append(var1_standard)
            reverse_graph[var2_auxiliary].append(NOT(var1_standard))
            reverse_graph[var1_standard].append(NOT(var2_auxiliary))
            
        elif c1 < 0 and c2 > 0:  # (Auxiliary1 OR Standard2)
            # (!Auxiliary1 -> Standard2) AND (!Standard2 -> Auxiliary1)
            graph[NOT(var1_auxiliary)].append(var2_standard)
            graph[NOT(var2_standard)].append(var1_auxiliary)
            reverse_graph[var2_standard].append(NOT(var1_auxiliary))
            reverse_graph[var1_auxiliary].append(NOT(var2_standard))
            
        else:  # c1 < 0 and c2 < 0:  # (Auxiliary1 OR Auxiliary2)
            # (!Auxiliary1 -> Auxiliary2) AND (!Auxiliary2 -> Auxiliary1)
            graph[NOT(var1_auxiliary)].append(var2_auxiliary)
            graph[NOT(var2_auxiliary)].append(var1_auxiliary)
            reverse_graph[var2_auxiliary].append(NOT(var1_auxiliary))
            reverse_graph[var1_auxiliary].append(NOT(var2_auxiliary))
    
    # Kosaraju's algorithm for finding SCCs
    def kosaraju():
        # First DFS to get finishing times
        visited = [False] * (2 * n)
        finish_stack = []
        
        def dfs1(u):
            visited[u] = True
            for v in graph[u]:
                if not visited[v]:
                    dfs1(v)
            finish_stack.append(u)
        
        # Second DFS on transpose graph
        scc_id = [-1] * (2 * n)
        scc_count = 0
        
        def dfs2(u, cid):
            scc_id[u] = cid
            for v in reverse_graph[u]:
                if scc_id[v] == -1:
                    dfs2(v, cid)
        
        # Run first DFS
        for i in range(2 * n):
            if not visited[i]:
                dfs1(i)
        
        # Run second DFS in reverse order of finishing times
        while finish_stack:
            u = finish_stack.pop()
            if scc_id[u] == -1:
                dfs2(u, scc_count)
                scc_count += 1
                
        return scc_id
    
    scc_id = kosaraju()
    
    # Check if satisfiable
    for i in range(n):
        standard_var = 2 * i
        auxiliary_var = 2 * i + 1
        if scc_id[standard_var] == scc_id[auxiliary_var]:
            print("NO")
            return
    
    # Build assignment
    # Create a graph of SCCs
    scc_graph = defaultdict(set)
    scc_nodes = defaultdict(list)
    
    for i in range(2 * n):
        scc_nodes[scc_id[i]].append(i)
    
    for u in range(2 * n):
        for v in graph[u]:
            if scc_id[u] != scc_id[v]:
                scc_graph[scc_id[u]].add(scc_id[v])
    
    # Topological sort of SCCs
    in_degree = defaultdict(int)
    all_sccs = set(scc_id)
    for scc in all_sccs:
        in_degree[scc] = 0
    
    for scc in all_sccs:
        for neighbor in scc_graph[scc]:
            in_degree[neighbor] += 1
    
    queue = deque()
    for scc in all_sccs:
        if in_degree[scc] == 0:
            queue.append(scc)
    
    scc_order = []
    while queue:
        current = queue.popleft()
        scc_order.append(current)
        for neighbor in scc_graph[current]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)
    
    # Assign values
    assignment = [-1] * (2 * n)
    for scc in reversed(scc_order):  # Process in reverse topological order
        for node in scc_nodes[scc]:
            if assignment[node] == -1:
                assignment[node] = 1
                # Also assign the negation to 0
                not_node = node ^ 1
                if assignment[not_node] == -1:
                    assignment[not_node] = 0
    
    # Output result
    print("YES")
    result = []
    for i in range(n):
        # 1 for Standard, 0 for Auxiliary
        if assignment[2 * i] == 1:
            result.append("1")
        else:
            result.append("0")
    print(" ".join(result))

solve()
