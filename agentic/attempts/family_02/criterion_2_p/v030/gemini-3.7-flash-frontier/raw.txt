```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(2 * n)]
    
    def to_literal(x):
        if x > 0:
            return 2 * (x - 1)
        else:
            return 2 * (-x - 1) + 1
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        lit_u = to_literal(u)
        lit_v = to_literal(v)
        
        # (lit_u or lit_v) => (~lit_u -> lit_v) and (~lit_v -> lit_u)
        adj[lit_u ^ 1].append(lit_v)
        adj[lit_v ^ 1].append(lit_u)
        
    # Tarjan's SCC algorithm (iterative)
    num_nodes = 2 * n
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    scc = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stack = []
    
    timer = 0
    scc_cnt = 0
    
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue
        
        # Iterative DFS for Tarjan's
        call_stack = [i]
        edge_idx = [0]
        
        while call_stack:
            u = call_stack[-1]
            if dfn[u] == -1:
                dfn[u] = low[u] = timer
                timer += 1
                stack.append(u)
                in_stack[u] = True
            
            ei = edge_idx[-1]
            if ei < len(adj[u]):
                v = adj[u][ei]
                edge_idx[-1] += 1
                if dfn[v] == -1:
                    call_stack.append(v)
                    edge_idx.append(0)
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]
            else:
                # Finished exploring u
                call_stack.pop()
                edge_idx.pop()
                if call_stack:
                    p = call_stack[-1]
                    if low[u] < low[p]:
                        low[p] = low[u]
                
                if low[u] == dfn[u]:
                    while True:
                        node = stack.pop()
                        in_stack[node] = False
                        scc[node] = scc_cnt
                        if node == u:
                            break
                    scc_cnt += 1

    ans = [0] * n
    for i in range(n):
        if scc[2 * i] == scc[2 * i + 1]:
            print("NO")
            return
        if scc[2 * i] < scc[2 * i + 1]:
            ans[i] = 1
        else:
            ans[i] = 0
            
    print("YES")
    print(*(ans[i] for i in range(n)))

if __name__ == '__main__':
    solve()
```
