import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 1)]
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        
        # Convert to 1-based indices for 2-SAT variables
        # x_i = 1 means valve i is primary (1)
        # x_i = 0 means valve i is secondary (0)
        # Node indices: 2*i-1 for x_i=1, 2*i for x_i=0
        
        # Target u: if u > 0, requires x_u = 1. If u < 0, requires x_|u| = 0.
        # Let target u correspond to boolean b_u.
        # Guideline (u, v): (b_u OR b_v) must be true.
        
        # Map target value to node index
        def get_node(val):
            idx = abs(val)
            if val > 0:
                return 2 * idx - 1  # x_idx = 1
            else:
                return 2 * idx      # x_idx = 0
        
        u_node = get_node(u)
        v_node = get_node(v)
        
        # Add implication: not u -> v
        # not u is the negation of u_node
        # If u_node is odd (2k-1), its negation is 2k (even)
        # If u_node is even (2k), its negation is 2k-1 (odd)
        u_neg = u_node + 1 if u_node % 2 == 1 else u_node - 1
        v_neg = v_node + 1 if v_node % 2 == 1 else v_node - 1
        
        adj[u_neg].append(v_node)
        adj[v_neg].append(u_node)
    
    # Kosaraju's algorithm for SCC
    N = 2 * n
    visited = [False] * (N + 1)
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack.pop()
            if idx == 0:
                if visited[node]:
                    continue
                visited[node] = True
            if idx < len(adj[node]):
                stack.append((node, idx + 1))
                child = adj[node][idx]
                if not visited[child]:
                    stack.append((child, 0))
            else:
                order.append(node)
    
    for i in range(1, N + 1):
        if not visited[i]:
            dfs1(i)
    
    comp_id = [-1] * (N + 1)
    current_comp = 0
    
    # Build reverse graph
    rev_adj = [[] for _ in range(N + 1)]
    for u in range(1, N + 1):
        for v in adj[u]:
            rev_adj[v].append(u)
    
    def dfs2(u, label):
        stack = [u]
        while stack:
            node = stack.pop()
            if comp_id[node] != -1:
                continue
            comp_id[node] = label
            for child in rev_adj[node]:
                if comp_id[child] == -1:
                    stack.append(child)
    
    for u in reversed(order):
        if comp_id[u] == -1:
            dfs2(u, current_comp)
            current_comp += 1
    
    # Check validity and construct assignment
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        node_true = 2 * i - 1
        node_false = 2 * i
        
        if comp_id[node_true] == comp_id[node_false]:
            print("NO")
            return
        
        # If comp_id[true] > comp_id[false], true is processed later in reverse order
        # which means it appears earlier in topological order, so we should pick true
        if comp_id[node_true] > comp_id[node_false]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
