import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    rev_adj = [[] for _ in range(num_nodes)]
    
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx+1])
        idx += 2
        
        # Literal to node index:
        # positive l: 2*(l-1)
        # negative l: 2*(|l|-1) + 1
        if l1 > 0:
            u = 2 * (l1 - 1)
            not_u = u + 1
        else:
            u = 2 * (-l1 - 1) + 1
            not_u = u - 1
            
        if l2 > 0:
            v = 2 * (l2 - 1)
            not_v = v + 1
        else:
            v = 2 * (-l2 - 1) + 1
            not_v = v - 1
            
        # (l1 or l2) <=> (not l1 => l2) and (not l2 => l1)
        adj[not_u].append(v)
        adj[not_v].append(u)
        rev_adj[v].append(not_u)
        rev_adj[u].append(not_v)
        
    # Kosaraju's Algorithm (Iterative)
    # Pass 1: Order by finishing times
    order = []
    visited = [False] * num_nodes
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_idx = [0]
            visited[i] = True
            
            while stack:
                u = stack[-1]
                if edge_idx[-1] < len(adj[u]):
                    nxt = adj[u][edge_idx[-1]]
                    edge_idx[-1] += 1
                    if not visited[nxt]:
                        visited[nxt] = True
                        stack.append(nxt)
                        edge_idx.append(0)
                else:
                    order.append(u)
                    stack.pop()
                    edge_idx.pop()
                    
    # Pass 2: Find SCCs in reversed graph
    scc = [-1] * num_nodes
    scc_count = 0
    
    for i in reversed(order):
        if scc[i] == -1:
            stack = [i]
            scc[i] = scc_count
            while stack:
                u = stack.pop()
                for v in rev_adj[u]:
                    if scc[v] == -1:
                        scc[v] = scc_count
                        stack.append(v)
            scc_count += 1
            
    # Check satisfiability and construct assignment
    ans = [0] * n
    for i in range(n):
        pos_node = 2 * i
        neg_node = 2 * i + 1
        if scc[pos_node] == scc[neg_node]:
            sys.stdout.write("NO\n")
            return
        # If scc[pos] < scc[neg], pos was processed later in DAG of SCCs,
        # meaning pos -> neg is impossible, but neg -> pos is possible.
        # In Kosaraju, smaller SCC id corresponds to earlier visited in rev_adj,
        # which means higher in topological order of G.
        # If scc[pos] < scc[neg], pos -> neg is possible => pos must be 0 (false).
        # Otherwise pos = 1 (true).
        ans[i] = 1 if scc[pos_node] > scc[neg_node] else 0
        
    sys.stdout.write("YES\n" + " ".join(map(str, ans)) + "\n")

if __name__ == '__main__':
    solve()
