import sys

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    def lit(x):
        return 2 * (x - 1) if x > 0 else 2 * (-x - 1) + 1
    
    def neg(l):
        return l ^ 1

    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        lu = lit(u)
        lv = lit(v)
        
        adj[neg(lu)].append(lv)
        adj[neg(lv)].append(lu)
        rev_adj[lv].append(neg(lu))
        rev_adj[lu].append(neg(lv))
        
    num_nodes = 2 * n
    visited = [False] * num_nodes
    order = []
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_idx = [0]
            visited[i] = True
            while stack:
                u = stack[-1]
                if edge_idx[-1] < len(adj[u]):
                    v = adj[u][edge_idx[-1]]
                    edge_idx[-1] += 1
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
                        edge_idx.append(0)
                else:
                    order.append(u)
                    stack.pop()
                    edge_idx.pop()
                    
    comp = [-1] * num_nodes
    comp_id = 0
    
    for u in reversed(order):
        if comp[u] == -1:
            stack = [u]
            comp[u] = comp_id
            while stack:
                curr = stack.pop()
                for v in rev_adj[curr]:
                    if comp[v] == -1:
                        comp[v] = comp_id
                        stack.append(v)
            comp_id += 1
            
    ans = [0] * n
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
        ans[i] = 1 if comp[2 * i] > comp[2 * i + 1] else 0
        
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    main()
