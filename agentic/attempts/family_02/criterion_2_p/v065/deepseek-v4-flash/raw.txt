We can model each dish’s two possible states as boolean variables and each rule as a 2-SAT clause.  
The constraints are large, so we need an efficient linear-time algorithm — Kosaraju’s or Tarjan’s SCC for 2-SAT.

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(1 << 25)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def idx(var):
        # var > 0: x_i true (spicy)
        # var < 0: x_i false (mild)
        if var > 0:
            return (var - 1) * 2
        else:
            return (-var - 1) * 2 + 1

    def neg(i):
        return i ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        ia = idx(a)
        ib = idx(b)
        # clause (a or b) => (!a -> b) and (!b -> a)
        g[neg(ia)].append(ib)
        gr[ib].append(neg(ia))
        g[neg(ib)].append(ia)
        gr[ia].append(neg(ib))

    # Kosaraju
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    print("YES")
    ans = []
    for i in range(n):
        # If comp[2*i] > comp[2*i+1], then true (spicy) is chosen
        ans.append('1' if comp[2*i] > comp[2*i+1] else '0')
    print(' '.join(ans))

if __name__ == "__main__":
    main()
```
