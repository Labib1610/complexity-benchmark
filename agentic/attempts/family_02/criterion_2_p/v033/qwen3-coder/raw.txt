import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each rule (a OR b), we create implications:
    # NOT a -> b and NOT b -> a
    # Variables: 1 to n represent x_i = 1
    #           -1 to -n represent x_i = 0
    # We'll use node ids: for variable v (1-based), 
    # node 2*v represents v=1, node 2*v+1 represents v=0
    # But it's easier to map: var v (1..n) 
    # positive literal v -> node 2*(v-1)     (x_v = 1)
    # negative literal -v -> node 2*(v-1)+1   (x_v = 0)
    # But actually, let's think differently.
    # Let's say for variable x (1-based):
    # x*2   represents x=1
    # x*2+1 represents x=0
    # Then for literal l:
    # if l > 0: node = l*2
    # if l < 0: node = (-l)*2 + 1
    # The complement of node n is n^1
    
    edges = defaultdict(list)
    
    index = 2
    clauses = []
    for i in range(m):
        l1 = int(data[index])
        l2 = int(data[index+1])
        index += 2
        clauses.append((l1, l2))
        
        # Convert literals to nodes
        # l > 0 means variable l should be 1
        # l < 0 means variable -l should be 0
        def lit_to_node(l):
            if l > 0:
                return l * 2  # x_l = 1
            else:
                return (-l) * 2 + 1  # x_{-l} = 0
        
        node1 = lit_to_node(l1)
        node2 = lit_to_node(l2)
        
        # clause (l1 OR l2) is equivalent to (~l1 -> l2) AND (~l2 -> l1)
        # ~l1 is node1^1
        # ~l2 is node2^1
        edges[node1 ^ 1].append(node2)
        edges[node2 ^ 1].append(node1)
    
    # Now we do strongly connected components to check 2-SAT satisfiability
    # Kosaraju's algorithm
    
    # First DFS to get finishing times
    visited = [False] * (2 * n + 2)
    finish_stack = []
    
    def dfs1(u):
        visited[u] = True
        for v in edges[u]:
            if not visited[v]:
                dfs1(v)
        finish_stack.append(u)
    
    for i in range(2, 2*n + 2):
        if not visited[i]:
            dfs1(i)
    
    # Build reverse graph
    rev_edges = defaultdict(list)
    for u in edges:
        for v in edges[u]:
            rev_edges[v].append(u)
    
    # Second DFS in reverse order of finishing times
    visited = [False] * (2 * n + 2)
    scc_id = [-1] * (2 * n + 2)
    scc_count = 0
    
    def dfs2(u, cid):
        visited[u] = True
        scc_id[u] = cid
        for v in rev_edges[u]:
            if not visited[v]:
                dfs2(v, cid)
    
    while finish_stack:
        u = finish_stack.pop()
        if not visited[u]:
            dfs2(u, scc_count)
            scc_count += 1
    
    # Check for contradictions: variable and its negation in same SCC
    for i in range(1, n+1):
        node_true = i * 2
        node_false = i * 2 + 1
        if scc_id[node_true] == scc_id[node_false]:
            print("NO")
            return
    
    # Satisfiable, reconstruct solution
    # Assign values based on topological order of SCCs
    # In Kosaraju, later SCCs in finish order (in reverse) can be set to true
    # Actually, the component with higher finish time in second DFS can be set to true
    
    # Let's re-think: in the finish_stack from first DFS, later nodes should be processed first in reverse
    # But we already used it. Let's just assign based on SCC ids.
    # If scc_id[x=1] > scc_id[x=0], then x=1 is preferred
    # No, that's not right either.
    # Proper way: process SCCs in topological order (reverse of finish order of second DFS)
    # But since we don't have that directly, we can iterate through variables
    # and pick the one whose SCC finishes later in the first DFS (but that's tricky)
    # Easier: build DAG of SCCs and do topological sort
    
    # Let's build the DAG of SCCs
    scc_graph = defaultdict(set)
    scc_nodes = defaultdict(list)
    
    for i in range(2, 2*n + 2):
        scc_nodes[scc_id[i]].append(i)
    
    for u in edges:
        for v in edges[u]:
            if scc_id[u] != scc_id[v]:
                scc_graph[scc_id[u]].add(scc_id[v])
    
    # Topological sort of SCCs using Kahn's algorithm
    in_degree = defaultdict(int)
    for u in scc_graph:
        for v in scc_graph[u]:
            in_degree[v] += 1
            
    # Find SCCs with in-degree 0
    scc_list = list(range(scc_count))
    queue = deque()
    for cid in scc_list:
        if in_degree[cid] == 0:
            queue.append(cid)
            
    topo_order = []
    while queue:
        u = queue.popleft()
        topo_order.append(u)
        for v in scc_graph[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
                
    # Now assign truth values: process SCCs in reverse topological order
    # SCCs that come later can be set to true
    assignment = [None] * (n + 1)
    assigned_scc = [False] * scc_count
    
    for cid in reversed(topo_order):
        for node in scc_nodes[cid]:
            var = (node // 2) 
            if node % 2 == 0:  # x_var = 1
                if assignment[var] is None:
                    assignment[var] = 1
            else:  # x_var = 0
                if assignment[var] is None:
                    assignment[var] = 0
    
    # Handle any unassigned variables (shouldn't happen in connected graph but safe)
    for i in range(1, n+1):
        if assignment[i] is None:
            assignment[i] = 0  # default to 0
            
    print("YES")
    print(" ".join(str(assignment[i]) for i in range(1, n+1)))

if __name__ == "__main__":
    main()
