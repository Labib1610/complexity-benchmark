We can model this as a **2-SAT** problem.  
Each rule is a clause of the form `(A or B)`, where each literal is either a variable or its negation.  

We can solve 2-SAT using implication graph and SCC (Tarjan or Kosaraju).  

Here’s the implementation:  

```python
import sys
sys.setrecursionlimit(1 << 20)

def main():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def node(var):
        # var > 0: positive literal, var < 0: negative literal
        if var > 0:
            return (var - 1) * 2
        else:
            return (-var - 1) * 2 + 1

    def neg(u):
        return u ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        u = node(a)
        v = node(b)
        # (u or v) => (!u -> v) and (!v -> u)
        g[neg(u)].append(v)
        g[neg(v)].append(u)
        rg[v].append(neg(u))
        rg[u].append(neg(v))

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for w in g[u]:
            if not visited[w]:
                dfs1(w)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for w in rg[u]:
            if comp[w] == -1:
                dfs2(w, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return

    print("YES")
    ans = []
    for i in range(n):
        # In Kosaraju, comp id order gives topological order of SCCs.
        # If comp[2i] > comp[2i+1], then 2i is reachable from 2i+1 => set True.
        ans.append('1' if comp[2 * i] > comp[2 * i + 1] else '0')
    print(' '.join(ans))

if __name__ == "__main__":
    main()
```
