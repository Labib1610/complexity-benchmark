```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # 2-SAT with implication graph.
    # Variable i (1-based): node 2*i is "i is OFF (0)", node 2*i+1 is "i is ON (1)".
    # Condition +i means i is ON  -> node 2*i+1.
    # Condition -i means i is OFF -> node 2*i.
    # Rule (a or b): add edges (not a -> b) and (not b -> a).

    N = 2 * (n + 1)
    graph = [[] for _ in range(N)]
    rgraph = [[] for _ in range(N)]

    def node(cond):
        i = abs(cond)
        if cond > 0:
            return 2 * i + 1
        else:
            return 2 * i

    def neg(x):
        return x ^ 1

    for _ in range(m):
        a = int(next(it))
        b = int(next(it))
        na = node(a)
        nb = node(b)
        # not a -> b
        graph[neg(na)].append(nb)
        rgraph[nb].append(neg(na))
        # not b -> a
        graph[neg(nb)].append(na)
        rgraph[na].append(neg(nb))

    # Iterative Kosaraju: first pass DFS on original graph.
    visited = bytearray(N)
    order = []
    stack = []
    for s in range(2, N):
        if not visited[s]:
            stack.append((s, 0))
            visited[s] = 1
            while stack:
                v, idx = stack.pop()
                if idx == 0:
                    order.append(v)
                adj = graph[v]
                progressed = False
                while idx < len(adj):
                    w = adj[idx]
                    idx += 1
                    if not visited[w]:
                        visited[w] = 1
                        stack.append((v, idx))
                        stack.append((w, 0))
                        progressed = True
                        break
                if not progressed:
                    pass

    # Second pass DFS on reversed graph in reverse finishing order.
    visited = bytearray(N)
    comp = [-1] * N
    cid = 0
    for s in reversed(order):
        if not visited[s]:
            stack = [s]
            visited[s] = 1
            comp[s] = cid
            while stack:
                v = stack.pop()
                for w in rgraph[v]:
                    if not visited[w]:
                        visited[w] = 1
                        comp[w] = cid
                        stack.append(w)
            cid += 1

    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        off = 2 * i
        on = 2 * i + 1
        if comp[off] == comp[on]:
            sys.stdout.write("NO\n")
            return
        assignment[i] = 1 if comp[on] > comp[off] else 0

    out = ["YES", " ".join(map(str, assignment[1:]))]
    sys.stdout.write("\n".join(out) + "\n")

solve()
```
