We can model this as a **2-SAT** problem.  
Each scene has two possible states: day (1) or night (0).  
Each requirement is a clause of the form `(A or B)`, which we can convert into implications:

- `(x or y)` → `(¬x → y)` and `(¬y → x)`

We build an implication graph with `2n` nodes:  
- Node `2*i` represents `scene i = 0`  
- Node `2*i+1` represents `scene i = 1`  

Then we run **Tarjan’s SCC** algorithm.  
If a variable and its negation are in the same SCC, it’s unsatisfiable.  
Otherwise, we assign values in reverse topological order of SCCs.

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def node(var):
        # var > 0: scene var = 1
        # var < 0: scene -var = 0
        if var > 0:
            return 2 * (var - 1) + 1
        else:
            return 2 * (-var - 1)

    def neg_node(u):
        return u ^ 1

    for _ in range(m):
        a, b = map(int, input().split())
        u = node(a)
        v = node(b)
        # (a or b) => (¬a -> b) and (¬b -> a)
        g[neg_node(u)].append(v)
        g[neg_node(v)].append(u)
        rg[v].append(neg_node(u))
        rg[u].append(neg_node(v))

    # Kosaraju's SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for w in g[u]:
            if not visited[w]:
                dfs1(w)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for w in rg[u]:
            if comp[w] == -1:
                dfs2(w, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Assign values: choose the SCC with higher id (appears later in topological order)
    ans = [0] * n
    for i in range(n):
        # comp[2*i] is for 0, comp[2*i+1] is for 1
        ans[i] = 1 if comp[2*i+1] > comp[2*i] else 0

    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
