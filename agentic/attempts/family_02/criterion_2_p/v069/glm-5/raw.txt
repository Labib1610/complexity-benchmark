import sys
sys.setrecursionlimit(1 << 25)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    # Build implication graph
    # Node indices: 0 to 2n-1
    # For variable i (1-indexed): true=2i-2, false=2i-1
    adj = [[] for _ in range(2 * n)]
    radj = [[] for _ in range(2 * n)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        # Parse conditions
        # l > 0 means scene l must be day (1)
        # l < 0 means scene |l| must be night (0)
        # So positive -> true, negative -> false
        
        def get_node(scene, val):
            # scene is 1-indexed, val is 0 or 1
            # returns node index
            return 2 * (scene - 1) + (1 - val)
        
        # Condition 1: scene |l1| should have value (1 if l1>0 else 0)
        s1 = abs(l1)
        v1 = 1 if l1 > 0 else 0
        # Condition 2: scene |l2| should have value (1 if l2>0 else 0)
        s2 = abs(l2)
        v2 = 1 if l2 > 0 else 0
        
        # Requirement: at least one of (s1=v1) or (s2=v2) must hold
        # Clause: (s1=v1) OR (s2=v2)
        # Implications: NOT(s1=v1) -> (s2=v2)
        #               NOT(s2=v2) -> (s1=v1)
        
        # NOT(s1=v1) means s1 has value (1-v1)
        # NOT(s2=v2) means s2 has value (1-v2)
        
        n1 = get_node(s1, v1)  # node for (s1=v1)
        n2 = get_node(s2, v2)  # node for (s2=v2)
        not_n1 = n1 ^ 1        # node for NOT(s1=v1)
        not_n2 = n2 ^ 1        # node for NOT(s2=v2)
        
        adj[not_n1].append(n2)
        adj[not_n2].append(n1)
        radj[n2].append(not_n1)
        radj[n1].append(not_n2)
    
    # Kosaraju's algorithm for SCC
    order = []
    visited = [False] * (2 * n)
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                nei = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[nei]:
                    stack.append((nei, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * (2 * n)
    
    def dfs2(u, c):
        stack = [u]
        while stack:
            node = stack.pop()
            if comp[node] == -1:
                comp[node] = c
                for nei in radj[node]:
                    if comp[nei] == -1:
                        stack.append(nei)
    
    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1
    
    # Check if any variable has both true and false in same SCC
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
    
    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # true node is 2i, false node is 2i+1
        # We want the one with higher topological order (higher comp number)
        if comp[2 * i] > comp[2 * i + 1]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(' '.join(map(str, assignment)))

solve()
