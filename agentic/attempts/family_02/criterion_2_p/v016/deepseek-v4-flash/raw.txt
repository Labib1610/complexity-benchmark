We can model this as a **2-SAT** problem.  
Each valve has two possible states: OPEN (True) or CLOSED (False).  
Each protocol is a clause of the form `(c1 OR c2)`, which is exactly a 2-CNF clause.

We can solve 2-SAT using **implication graph + SCC (Kosaraju or Tarjan)**.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    
    # 2-SAT: variable i (0-indexed) -> 2*i = False, 2*i+1 = True
    N = 2 * n
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]
    
    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)
    
    def add_clause(a, b):
        # a or b
        # not a -> b
        # not b -> a
        add_imp(a ^ 1, b)
        add_imp(b ^ 1, a)
    
    for _ in range(m):
        c1, c2 = map(int, input().split())
        # convert to variable indices
        def lit(c):
            var = abs(c) - 1
            return 2 * var + (1 if c > 0 else 0)
        add_clause(lit(c1), lit(c2))
    
    # Kosaraju SCC
    visited = [False] * N
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in gr[u]:
            if comp[v] == -1:
                dfs2(v, cid)
    
    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1
    
    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return
    
    # Build assignment: choose value based on SCC order
    assignment = [False] * n
    for i in range(n):
        # If comp[2*i] > comp[2*i+1], then True is reachable from False -> set True
        assignment[i] = comp[2*i] > comp[2*i+1]
    
    print("YES")
    print(" ".join("1" if assignment[i] else "0" for i in range(n)))

if __name__ == "__main__":
    solve()
```
