import sys

# Increase recursion depth just in case, but iterative is safer
sys.setrecursionlimit(2000000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Literals: for variable i (1-indexed):
    # +i -> 2 * (i - 1)
    # -i -> 2 * (i - 1) + 1
    # Negation of literal x: x ^ 1
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        
        # Convert to 0-indexed literal
        lit_u = (abs(u) - 1) * 2 if u > 0 else (abs(u) - 1) * 2 + 1
        lit_v = (abs(v) - 1) * 2 if v > 0 else (abs(v) - 1) * 2 + 1
        
        # Clause: (u or v) <=> (~u -> v) and (~v -> u)
        adj[lit_u ^ 1].append(lit_v)
        adj[lit_v ^ 1].append(lit_u)
        
    # Tarjan's SCC algorithm (iterative to avoid recursion limit issues)
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    scc = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stack = []
    
    timer = 0
    scc_cnt = 0
    
    # Iterative DFS for Tarjan
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue
        
        call_stack = [(i, 0)]
        dfn[i] = low[i] = timer
        timer += 1
        stack.append(i)
        in_stack[i] = True
        
        while call_stack:
            u, edge_idx = call_stack[-1]
            
            if edge_idx < len(adj[u]):
                call_stack[-1] = (u, edge_idx + 1)
                v = adj[u][edge_idx]
                if dfn[v] == -1:
                    dfn[v] = low[v] = timer
                    timer += 1
                    stack.append(v)
                    in_stack[v] = True
                    call_stack.append((v, 0))
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]
            else:
                call_stack.pop()
                if low[u] == dfn[u]:
                    while True:
                        v = stack.pop()
                        in_stack[v] = False
                        scc[v] = scc_cnt
                        if v == u:
                            break
                    scc_cnt += 1
                if call_stack:
                    p = call_stack[-1][0]
                    if low[u] < low[p]:
                        low[p] = low[u]
                        
    # Check satisfiability
    ans = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        ans[i] = 1 if scc[pos] < scc[neg] else 0
        
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    main()
