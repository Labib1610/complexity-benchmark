import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    iterator = iter(data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Create implication graph
    # Nodes are 2*n: 0..2n-1
    # For artwork i (1-based):
    # Node 2*(i-1) represents Ambient (0)
    # Node 2*(i-1) + 1 represents Spotlight (1)
    # a -> b means if 'a' clause is true, 'b' clause must be true.
    
    adj = [[] for _ in range(2 * n)]
    
    def get_node(artwork_idx, style):
        # artwork_idx is 1-based, style is 0 (Ambient) or 1 (Spotlight)
        return 2 * (artwork_idx - 1) + style
    
    for _ in range(m):
        try:
            u_in = int(next(iterator))
            v_in = int(next(iterator))
        except StopIteration:
            break
        
        # Parse input
        # Positive v -> Spotlight (style 1)
        # Negative v -> Ambient (style 0)
        u_idx = abs(u_in)
        u_style = 1 if u_in > 0 else 0
        
        v_idx = abs(v_in)
        v_style = 1 if v_in > 0 else 0
        
        # Mandate: (u_idx is u_style) OR (v_idx is v_style)
        # Implications: (not u) -> v AND (not v) -> u
        # not u is negation of style
        
        # Node u
        node_u = get_node(u_idx, u_style)
        # Node not u
        node_not_u = get_node(u_idx, 1 - u_style)
        
        # Node v
        node_v = get_node(v_idx, v_style)
        # Node not v
        node_not_v = get_node(v_idx, 1 - v_style)
        
        # Add edges
        adj[node_not_u].append(node_v)
        adj[node_not_v].append(node_u)

    # Find SCCs using Kosaraju's algorithm
    # Pass 1: DFS to get finishing order
    visited = [False] * (2 * n)
    order = []
    
    for i in range(2 * n):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                node, state = stack.pop()
                if state == 0:
                    if visited[node]:
                        continue
                    visited[node] = True
                    stack.append((node, 1))
                    for neighbor in adj[node]:
                        if not visited[neighbor]:
                            stack.append((neighbor, 0))
                else:
                    order.append(node)
                    
    # Pass 2: DFS on reverse graph to find SCCs
    rev_adj = [[] for _ in range(2 * n)]
    for u in range(2 * n):
        for v in adj[u]:
            rev_adj[v].append(u)
            
    scc_id = [-1] * (2 * n)
    current_id = 0
    
    for u in reversed(order):
        if scc_id[u] == -1:
            stack = [u]
            scc_id[u] = current_id
            while stack:
                node = stack.pop()
                for neighbor in rev_adj[node]:
                    if scc_id[neighbor] == -1:
                        scc_id[neighbor] = current_id
                        stack.append(neighbor)
            current_id += 1
            
    # Check for contradiction: if (i, 0) and (i, 1) are in same SCC
    possible = True
    for i in range(n):
        if scc_id[2 * i] == scc_id[2 * i + 1]:
            possible = False
            break
            
    if not possible:
        print("NO")
    else:
        print("YES")
        # Assign values
        # If scc_id[Spotlight] > scc_id[Ambient], assign Spotlight (1)
        # Else assign Ambient (0)
        # Topological order is reversed id order.
        result = []
        for i in range(n):
            if scc_id[2 * i + 1] > scc_id[2 * i]:
                result.append('1')
            else:
                result.append('0')
        print(' '.join(result))

if __name__ == "__main__":
    solve()
