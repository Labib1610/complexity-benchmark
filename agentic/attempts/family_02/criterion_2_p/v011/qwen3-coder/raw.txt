import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    clauses = []
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        clauses.append((l1, l2))
        idx += 2
    
    # Create implication graph for 2-SAT
    # Variables: x1, x2, ..., xn where xi = 1 means spotlight, 0 means ambient
    # For each variable i, we have:
    # node i*2 represents xi (spotlight)
    # node i*2+1 represents not xi (ambient)
    
    # For clause (a or b), we add edges (not a -> b) and (not b -> a)
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    def add_implication(x, y):
        adj[x].append(y)
        rev_adj[y].append(x)
    
    for l1, l2 in clauses:
        # l1 and l2 are literals
        # positive means spotlight, negative means ambient
        # Convert to variable indices and truth values
        if l1 > 0:
            var1 = l1 - 1  # 0-indexed
            val1 = True    # spotlight
        else:
            var1 = -l1 - 1
            val1 = False   # ambient
            
        if l2 > 0:
            var2 = l2 - 1
            val2 = True
        else:
            var2 = -l2 - 1
            val2 = False
            
        # clause is (val1 or val2)
        # which is equivalent to (not val1 -> val2) and (not val2 -> val1)
        
        # node for var1 being spotlight: var1*2
        # node for var1 being ambient: var1*2+1
        # similarly for var2
        
        node1_true = var1 * 2
        node1_false = var1 * 2 + 1
        node2_true = var2 * 2
        node2_false = var2 * 2 + 1
        
        if val1 and val2:
            # (x1 or x2) => (not x1 -> x2) and (not x2 -> x1)
            add_implication(node1_false, node2_true)
            add_implication(node2_false, node1_true)
        elif val1 and not val2:
            # (x1 or not x2) => (not x1 -> not x2) and (x2 -> x1)
            add_implication(node1_false, node2_false)
            add_implication(node2_true, node1_true)
        elif not val1 and val2:
            # (not x1 or x2) => (x1 -> x2) and (not x2 -> not x1)
            add_implication(node1_true, node2_true)
            add_implication(node2_false, node1_false)
        else:
            # (not x1 or not x2) => (x1 -> not x2) and (x2 -> not x1)
            add_implication(node1_true, node2_false)
            add_implication(node2_true, node1_false)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    visited = [False] * (2 * n)
    scc = [-1] * (2 * n)
    scc_id = 0
    
    def dfs2(u):
        visited[u] = True
        scc[u] = scc_id
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_id += 1
    
    # Check if satisfiable
    for i in range(n):
        if scc[i * 2] == scc[i * 2 + 1]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * n
    # Process SCCs in reverse topological order
    for i in range(n):
        if scc[i * 2] > scc[i * 2 + 1]:
            assignment[i] = 1  # spotlight
        else:
            assignment[i] = 0  # ambient
    
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
