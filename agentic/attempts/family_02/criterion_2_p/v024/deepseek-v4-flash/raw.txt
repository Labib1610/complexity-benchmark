We can model this as a **2-SAT** problem.  
Each sensor has two possible states: high (1) or low (0).  
Each requirement is a clause of the form `(A OR B)`, where `A` and `B` are literals (positive or negative).  

We can solve 2-SAT using **implication graph + SCC (Kosaraju or Tarjan)**.

```python
import sys
sys.setrecursionlimit(10**6)
input = sys.stdin.readline

n, m = map(int, input().split())

# 2-SAT: each variable i (0-indexed) has two nodes:
# node 2*i     -> variable i is True (high gain)
# node 2*i + 1 -> variable i is False (low gain)

N = 2 * n
graph = [[] for _ in range(N)]
rev_graph = [[] for _ in range(N)]

def add_imp(u, v):
    graph[u].append(v)
    rev_graph[v].append(u)

def node(var):
    # var > 0: sensor var is high (True)
    # var < 0: sensor -var is low (False)
    if var > 0:
        return 2 * (var - 1)      # True node
    else:
        return 2 * (-var - 1) + 1 # False node

def neg_node(var):
    return node(var) ^ 1

for _ in range(m):
    a, b = map(int, input().split())
    # clause: (a OR b)
    # implications: !a -> b, !b -> a
    add_imp(neg_node(a), node(b))
    add_imp(neg_node(b), node(a))

# Kosaraju SCC
visited = [False] * N
order = []

def dfs1(u):
    visited[u] = True
    for v in graph[u]:
        if not visited[v]:
            dfs1(v)
    order.append(u)

for i in range(N):
    if not visited[i]:
        dfs1(i)

comp = [-1] * N
def dfs2(u, cid):
    comp[u] = cid
    for v in rev_graph[u]:
        if comp[v] == -1:
            dfs2(v, cid)

cid = 0
for u in reversed(order):
    if comp[u] == -1:
        dfs2(u, cid)
        cid += 1

# Check satisfiability
for i in range(n):
    if comp[2*i] == comp[2*i+1]:
        print("NO")
        sys.exit()

print("YES")
res = []
for i in range(n):
    # If True node comes after False node in topological order, assign True
    # In Kosaraju, comp id order is reverse topological.
    # If comp[2*i] > comp[2*i+1], then True is reachable from False => choose True
    res.append(1 if comp[2*i] > comp[2*i+1] else 0)

print(' '.join(map(str, res)))
```
