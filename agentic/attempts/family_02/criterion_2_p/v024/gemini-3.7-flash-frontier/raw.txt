import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Literals:
    # Sensor i (1-based):
    # high-gain (+i): 2 * (i - 1)
    # low-gain (-i):  2 * (i - 1) + 1
    
    def get_lit(x):
        if x > 0:
            return 2 * (x - 1)
        else:
            return 2 * (-x - 1) + 1
            
    def get_neg(lit):
        return lit ^ 1

    adj = [[] for _ in range(2 * n)]
    rev_adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        u = get_lit(l1)
        v = get_lit(l2)
        
        # ~u -> v
        adj[get_neg(u)].append(v)
        rev_adj[v].append(get_neg(u))
        
        # ~v -> u
        adj[get_neg(v)].append(u)
        rev_adj[u].append(get_neg(v))
        
    num_nodes = 2 * n
    
    # Kosaraju's Algorithm (Iterative)
    visited = [False] * num_nodes
    order = []
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [(i, 0)]
            visited[i] = True
            while stack:
                u, edge_idx = stack[-1]
                if edge_idx < len(adj[u]):
                    stack[-1] = (u, edge_idx + 1)
                    nxt = adj[u][edge_idx]
                    if not visited[nxt]:
                        visited[nxt] = True
                        stack.append((nxt, 0))
                else:
                    stack.pop()
                    order.append(u)
                    
    comp = [-1] * num_nodes
    scc_id = 0
    
    for u in reversed(order):
        if comp[u] == -1:
            stack = [u]
            comp[u] = scc_id
            while stack:
                curr = stack.pop()
                for to in rev_adj[curr]:
                    if comp[to] == -1:
                        comp[to] = scc_id
                        stack.append(to)
            scc_id += 1
            
    ans = []
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if comp[pos] == comp[neg]:
            print("NO")
            return
        if comp[pos] > comp[neg]:
            ans.append(1)
        else:
            ans.append(0)
            
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
