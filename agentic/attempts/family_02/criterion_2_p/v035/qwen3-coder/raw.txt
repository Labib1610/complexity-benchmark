import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each rule (l1 or l2), we can represent it as a logical clause
    # We'll use 2-SAT approach
    # Variables: 1..n (positive) and -1..-n (negative)
    # For each variable x, we have nodes: x and -x
    # For each clause (a or b), we add edges: (-a -> b) and (-b -> a)
    
    # Build implication graph
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    idx = 2
    clauses = []
    for i in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        clauses.append((l1, l2))
        # Add implications: (-l1 => l2) and (-l2 => l1)
        graph[-l1].append(l2)
        graph[-l2].append(l1)
        reverse_graph[l2].append(-l1)
        reverse_graph[l1].append(-l2)
    
    # Kosaraju's algorithm for SCC
    visited = set()
    order = []
    
    def dfs1(node):
        if node in visited:
            return
        visited.add(node)
        for neighbor in graph[node]:
            dfs1(neighbor)
        order.append(node)
    
    # Run DFS1 on all nodes
    nodes = set()
    for l1, l2 in clauses:
        nodes.add(l1)
        nodes.add(-l1)
        nodes.add(l2)
        nodes.add(-l2)
    
    for node in nodes:
        dfs1(node)
    
    # Reverse the graph and process in reverse post-order
    visited.clear()
    scc_map = {}
    scc_id = 0
    
    def dfs2(node):
        if node in visited:
            return
        visited.add(node)
        scc_map[node] = scc_id
        for neighbor in reverse_graph[node]:
            dfs2(neighbor)
    
    # Process in reverse order
    for node in reversed(order):
        if node not in visited:
            dfs2(node)
            scc_id += 1
    
    # Check if any variable and its negation are in the same SCC
    for i in range(1, n+1):
        if i in scc_map and -i in scc_map:
            if scc_map[i] == scc_map[-i]:
                print("NO")
                return
    
    # If satisfiable, reconstruct a solution
    # Assign values based on topological order of SCCs
    # Nodes in later SCCs should be assigned True first
    
    # Create mapping from SCC id to nodes
    scc_nodes = defaultdict(list)
    for node, sid in scc_map.items():
        scc_nodes[sid].append(node)
    
    # Sort SCCs in topological order (reverse of finishing times in dfs1)
    scc_order = []
    scc_visited = set()
    
    def get_scc_order(node):
        # Get the SCC id of this node
        sid = scc_map[node]
        if sid in scc_visited:
            return
        scc_visited.add(sid)
        # Visit all SCCs that this SCC points to
        for u in scc_nodes[sid]:
            for v in graph[u]:
                if scc_map[v] != sid:  # Different SCC
                    get_scc_order(v)
        scc_order.append(sid)
    
    visited.clear()
    for node in nodes:
        if node not in visited:
            get_scc_order(node)
    
    # Now assign truth values
    assignment = {}
    # Process SCCs in reverse topological order (last ones first)
    for sid in reversed(scc_order):
        for node in scc_nodes[sid]:
            var = abs(node)
            if var not in assignment:
                # If node is positive literal, set to False (since we're setting negated literals to True)
                # Actually, let's think carefully:
                # In the implication graph, if we have SCCs ordered topologically,
                # we want to assign False to literals in earlier SCCs
                # But we need to be more precise
                
                # Better approach: for each SCC, if it doesn't contain a variable and its negation,
                # we can assign values consistently
                # If a positive literal x is in an SCC that comes later than -x, then x should be True
                
                # Let's just check if the positive literal is in this SCC
                if node > 0:
                    assignment[var] = 0  # Since we process in reverse topo order, later SCCs are processed first
                else:
                    assignment[var] = 1
    
    # Need a better way to determine assignment
    # Let's re-think: in Kosaraju, SCCs are in reverse topological order in 'order' list
    # So when we process them in normal order for assignment, we get topological order
    
    # Reset assignment
    assignment = {}
    # Process SCCs in topological order (reverse of how they were found)
    processed_sccs = set()
    for sid in reversed(range(scc_id)):
        if sid in processed_sccs:
            continue
        processed_sccs.add(sid)
        # For each unassigned variable in this SCC, assign it appropriately
        # Since we're processing in topological order, we set variables in this SCC to satisfy implications
        # But actually, in 2-SAT, if an SCC is processed later, its variables should be true
        
        # Simpler: for each SCC, if it contains a positive literal, set that variable to make it true
        # But we have to be consistent
        
        # Best approach: process SCCs in topological order, and for each variable not yet assigned,
        # if the positive literal is in this SCC, set it to 1, otherwise to 0
        # No, that's not right either.
        
        # Correct approach for 2-SAT reconstruction:
        # Process SCCs in topological order (which is the reverse of the finish order from first DFS)
        # For each SCC, if a variable x has not been assigned yet, 
        # and x is in this SCC, assign x = False
        # if -x is in this SCC, assign x = True
        # But we have to be careful not to contradict
        
        # Actually, the standard approach:
        # Process SCCs in reverse topological order (so last finished SCC first)
        # For each node in the SCC, if its variable is not assigned, assign it to make the literal true
        # i.e., if node is positive literal x, assign x = 1
        # if node is negative literal -x, assign x = 0
        
        for node in scc_nodes[sid]:
            var = abs(node)
            if var not in assignment:
                if node > 0:
                    assignment[var] = 1
                else:
                    assignment[var] = 0
    
    # Handle variables not mentioned in any clause
    for i in range(1, n+1):
        if i not in assignment:
            assignment[i] = 0  # Default value
    
    print("YES")
    result = []
    for i in range(1, n+1):
        result.append(str(assignment[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
