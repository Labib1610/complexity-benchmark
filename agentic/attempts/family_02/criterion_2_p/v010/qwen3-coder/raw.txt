```python
import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    # For each clause (l1 or l2), we create implications
    # l1 or l2 is equivalent to (not l1 -> l2) and (not l2 -> l1)
    
    # We'll use 2-SAT. Each variable x has two nodes: x (mode 1) and x+n (mode 0)
    # But it's easier to map: for variable x (1-based), 
    # x*2 represents x=0, x*2+1 represents x=1
    
    adj = defaultdict(list)
    rev_adj = defaultdict(list)
    
    for _ in range(m):
        l1, l2 = map(int, sys.stdin.readline().split())
        # l1 or l2
        # not l1 -> l2
        # not l2 -> l1
        
        # Convert to 0-based indexing for internal use
        # For a literal k:
        # if k > 0: variable (k-1) should be 1
        # if k < 0: variable (-k-1) should be 0
        
        # not l1 -> l2
        if l1 > 0:
            u = (l1 - 1) * 2      # l1=0
            if l2 > 0:
                v = (l2 - 1) * 2 + 1  # l2=1
            else:
                v = (-l2 - 1) * 2     # l2=0
        else:
            u = (-l1 - 1) * 2 + 1   # l1=1
            if l2 > 0:
                v = (l2 - 1) * 2 + 1  # l2=1
            else:
                v = (-l2 - 1) * 2     # l2=0
        adj[u].append(v)
        rev_adj[v].append(u)
        
        # not l2 -> l1
        if l2 > 0:
            u = (l2 - 1) * 2      # l2=0
            if l1 > 0:
                v = (l1 - 1) * 2 + 1  # l1=1
            else:
                v = (-l1 - 1) * 2     # l1=0
        else:
            u = (-l2 - 1) * 2 + 1   # l2=1
            if l1 > 0:
                v = (l1 - 1) * 2 + 1  # l1=1
            else:
                v = (-l1 - 1) * 2     # l1=0
        adj[u].append(v)
        rev_adj[v].append(u)
    
    # Now we do Kosaraju's algorithm for 2-SAT
    order = []
    visited = [False] * (2 * n)
    
    def dfs1(u):
        visited[u] = True
        for v in adj[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2 * n):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on reverse graph to find SCCs
    visited = [False] * (2 * n)
    scc_id = [-1] * (2 * n)
    scc_count = 0
    
    def dfs2(u):
        visited[u] = True
        scc_id[u] = scc_count
        for v in rev_adj[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            scc_count += 1
    
    # Check if for each variable, x=0 and x=1 are in different SCCs
    for i in range(n):
        if scc_id[i * 2] == scc_id[i * 2 + 1]:
            print("NO")
            return
    
    # Assign values: variables in later SCCs (higher index) are assigned True
    assignment = [0] * n
    for i in range(n):
        if scc_id[i * 2] > scc_id[i * 2 + 1]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
