import sys

# Increase recursion depth just in case, but we'll use iterative algorithms
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # 2*n vertices:
    # for variable i (1-indexed, 1 <= i <= n):
    # positive i -> 2*(i-1)
    # negative i -> 2*(i-1) + 1
    
    def get_lit(l):
        if l > 0:
            return 2 * (l - 1)
        else:
            return 2 * (-l - 1) + 1
            
    def get_neg(v):
        return v ^ 1

    adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        u = get_lit(l1)
        v = get_lit(l2)
        
        # clause: u or v
        # not u -> v
        # not v -> u
        adj[get_neg(u)].append(v)
        adj[get_neg(v)].append(u)
        
    # Tarjan's SCC algorithm (iterative)
    num_nodes = 2 * n
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stack = []
    scc = [-1] * num_nodes
    timer = 0
    scc_cnt = 0
    
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue
        
        # Iterative DFS for Tarjan
        call_stack = [(i, 0)]
        dfn[i] = low[i] = timer
        timer += 1
        stack.append(i)
        in_stack[i] = True
        
        while call_stack:
            u, edge_idx = call_stack[-1]
            if edge_idx < len(adj[u]):
                call_stack[-1] = (u, edge_idx + 1)
                to = adj[u][edge_idx]
                if dfn[to] == -1:
                    dfn[to] = low[to] = timer
                    timer += 1
                    stack.append(to)
                    in_stack[to] = True
                    call_stack.append((to, 0))
                elif in_stack[to]:
                    if dfn[to] < low[u]:
                        low[u] = dfn[to]
            else:
                call_stack.pop()
                if call_stack:
                    parent = call_stack[-1][0]
                    if low[u] < low[parent]:
                        low[parent] = low[u]
                if low[u] == dfn[u]:
                    while True:
                        w = stack.pop()
                        in_stack[w] = False
                        scc[w] = scc_cnt
                        if w == u:
                            break
                    scc_cnt += 1

    assignment = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        if scc[pos] < scc[neg]:
            assignment[i] = 1
        else:
            assignment[i] = 0
            
    print("YES")
    print(*(assignment))

if __name__ == '__main__':
    solve()
