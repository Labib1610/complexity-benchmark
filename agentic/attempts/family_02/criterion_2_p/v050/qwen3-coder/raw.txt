```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["0"] * n))
        return
    
    clauses = []
    idx = 2
    for _ in range(m):
        l1 = int(data[idx])
        l2 = int(data[idx+1])
        idx += 2
        clauses.append((l1, l2))
    
    # Convert to 2-SAT
    # Variables: x1, x2, ..., xn (xi = 1 means array i is active)
    # For each clause (l1, l2), we have (l1 OR l2)
    # If l1 = i (>0), it means xi is True
    # If l1 = -i (<0), it means xi is False, so (NOT xi)
    
    # We'll use implication graph and 2-SAT
    # For (l1 OR l2), it's equivalent to (NOT l1 -> l2) and (NOT l2 -> l1)
    
    # Build implication graph
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    
    def get_node(var):
        # var is +i or -i
        # node is 2*i for x_i, 2*i+1 for NOT x_i
        if var > 0:
            return 2 * var
        else:
            return 2 * (-var) + 1
    
    def get_neg_node(var):
        if var > 0:
            return 2 * var + 1
        else:
            return 2 * (-var)
    
    for l1, l2 in clauses:
        # (l1 OR l2) is equivalent to (NOT l1 -> l2) and (NOT l2 -> l1)
        # NOT l1 -> l2
        graph[get_neg_node(l1)].append(get_node(l2))
        reverse_graph[get_node(l2)].append(get_neg_node(l1))
        
        # NOT l2 -> l1
        graph[get_neg_node(l2)].append(get_node(l1))
        reverse_graph[get_node(l1)].append(get_neg_node(l2))
    
    # Now run Kosaraju's algorithm to check if 2-SAT is satisfiable
    # Nodes are numbered from 2 to 2*n+1
    visited = [False] * (2 * n + 2)
    order = []
    
    def dfs1(u):
        visited[u] = True
        for v in graph[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)
    
    for i in range(2, 2*n + 2):
        if not visited[i]:
            dfs1(i)
    
    # Second DFS on reverse graph
    visited = [False] * (2 * n + 2)
    component = [-1] * (2 * n + 2)
    comp_num = 0
    
    def dfs2(u):
        visited[u] = True
        component[u] = comp_num
        for v in reverse_graph[u]:
            if not visited[v]:
                dfs2(v)
    
    while order:
        u = order.pop()
        if not visited[u]:
            dfs2(u)
            comp_num += 1
    
    # Check if satisfiable
    for i in range(1, n+1):
        node_true = 2 * i
        node_false = 2 * i + 1
        if component[node_true] == component[node_false]:
            print("NO")
            return
    
    # Satisfiable, construct solution
    # In each pair, the one with higher component number is true
    assignment = [0] * (n + 1)
    for i in range(1, n+1):
        node_true = 2 * i
        node_false = 2 * i + 1
        if component[node_true] > component[node_false]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(" ".join(map(str, assignment[1:])))

if __name__ == "__main__":
    main()
```
