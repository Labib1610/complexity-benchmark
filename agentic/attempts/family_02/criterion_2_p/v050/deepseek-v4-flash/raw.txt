We can model this as a **2-SAT** problem.  
Each array has two possible states: active (1) or standby (0).  
Each failure scenario is a clause of the form:  
`(array a is 1) OR (array b is 1)` or with negations depending on the sign.

We’ll build an implication graph and check satisfiability using SCC (Tarjan’s or Kosaraju’s).

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # each variable has two nodes: x_i and not x_i

    # node mapping:
    # variable i (1-indexed) -> node i for True, node i+n for False
    g = [[] for _ in range(N)]
    gr = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        gr[v].append(u)

    for _ in range(m):
        a, b = map(int, input().split())
        # literal: positive => var True, negative => var False
        # For literal l: node = (abs(l)-1) if l>0 else (abs(l)-1 + n)
        def lit_node(l):
            var = abs(l) - 1
            return var if l > 0 else var + n

        u = lit_node(a)
        v = lit_node(b)
        # clause (u or v) => (!u -> v) and (!v -> u)
        add_imp(u ^ n, v)  # not u -> v
        add_imp(v ^ n, u)  # not v -> u

    # Kosaraju SCC
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for w in g[u]:
            if not visited[w]:
                dfs1(w)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for w in gr[u]:
            if comp[w] == -1:
                dfs2(w, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[i] == comp[i + n]:
            print("NO")
            return

    # Build assignment
    assignment = [0] * n
    for i in range(n):
        # If comp[i] > comp[i+n], set True (1), else False (0)
        assignment[i] = 1 if comp[i] > comp[i + n] else 0

    print("YES")
    print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    solve()
```
