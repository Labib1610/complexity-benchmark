import sys
sys.setrecursionlimit(300000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(2 * n + 2)]
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        
        # Clause: (x1 OR x2)
        # x1 = (array |l1| is in mode sign(l1))
        # Variable i: 1 = array i is active (1), 0 = array i is standby (0)
        # Literal l > 0: variable l is true
        # Literal l < 0: variable |l| is false
        
        # Implication: not x1 -> x2
        # not x1: if l1 > 0, then array l1 is 0 (false). If l1 < 0, then array |l1| is 1 (true).
        # x2: if l2 > 0, then array l2 is 1. If l2 < 0, then array |l2| is 0.
        
        def get_nodes(lit):
            # returns (node_for_true, node_for_false)
            # node indices: 1..n for true, n+1..2n for false
            var = abs(lit)
            if lit > 0:
                return (var, n + var)
            else:
                return (n + var, var)
        
        u1, v1 = get_nodes(l1)
        u2, v2 = get_nodes(l2)
        
        # not x1 -> x2: v1 -> u2
        adj[v1].append(u2)
        # not x2 -> x1: v2 -> u1
        adj[v2].append(u1)
    
    # Kosaraju's algorithm for SCC
    visited = [False] * (2 * n + 2)
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            if idx < len(adj[node]):
                neighbor = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[neighbor]:
                    stack.append((neighbor, 0))
            else:
                stack.pop()
                order.append(node)
    
    for i in range(1, 2 * n + 1):
        if not visited[i]:
            dfs1(i)
    
    adj_rev = [[] for _ in range(2 * n + 2)]
    for u in range(1, 2 * n + 1):
        for v in adj[u]:
            adj_rev[v].append(u)
    
    comp_id = [0] * (2 * n + 2)
    current_comp = 0
    
    def dfs2(u, cid):
        stack = [u]
        while stack:
            node = stack.pop()
            if comp_id[node] == 0:
                comp_id[node] = cid
                for neighbor in adj_rev[node]:
                    if comp_id[neighbor] == 0:
                        stack.append(neighbor)
    
    for node in reversed(order):
        if comp_id[node] == 0:
            current_comp += 1
            dfs2(node, current_comp)
    
    # Check for contradiction
    for i in range(1, n + 1):
        if comp_id[i] == comp_id[n + i]:
            print("NO")
            return
    
    # Assign values
    assignment = [0] * (n + 1)
    # Process components in reverse topological order (higher comp_id first)
    # comp_id assigned in order of finishing, so higher id = earlier in topo order
    for i in range(1, n + 1):
        if comp_id[i] > comp_id[n + i]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

solve()
