import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    # 2-SAT: variable i has nodes 2*i (false) and 2*i+1 (true), for i in 1..n
    # We use 1-indexed variables, so total nodes = 2*(n+1)
    size = 2 * (n + 1)
    adj = [[] for _ in range(size)]
    radj = [[] for _ in range(size)]
    
    def node(var, val):
        # var is 1-indexed, val is 0 or 1
        return 2 * var + val
    
    def add_impl(a, va, b, vb):
        u = node(a, va)
        v = node(b, vb)
        adj[u].append(v)
        radj[v].append(u)
    
    for _ in range(m):
        l1 = int(next(it))
        l2 = int(next(it))
        v1 = abs(l1)
        va = 1 if l1 > 0 else 0
        v2 = abs(l2)
        vb = 1 if l2 > 0 else 0
        # (va OR vb): add implications not va -> vb and not vb -> va
        add_impl(v1, 1 - va, v2, vb)
        add_impl(v2, 1 - vb, v1, va)
    
    # Kosaraju's SCC
    visited = bytearray(size)
    order = []
    for i in range(size):
        if not visited[i]:
            stack = [(i, 0)]
            while stack:
                v, idx = stack[-1]
                if idx == 0:
                    if visited[v]:
                        stack.pop()
                        continue
                    visited[v] = 1
                if idx < len(adj[v]):
                    w = adj[v][idx]
                    stack[-1] = (v, idx + 1)
                    if not visited[w]:
                        stack.append((w, 0))
                else:
                    stack.pop()
                    order.append(v)
    
    comp = [-1] * size
    cid = 0
    for i in reversed(order):
        if comp[i] == -1:
            stack = [i]
            comp[i] = cid
            while stack:
                v = stack.pop()
                for w in radj[v]:
                    if comp[w] == -1:
                        comp[w] = cid
                        stack.append(w)
            cid += 1
    
    # Check satisfiability and assign
    assignment = [0] * (n + 1)
    for i in range(1, n + 1):
        f_node = node(i, 0)
        t_node = node(i, 1)
        if comp[f_node] == comp[t_node]:
            print("NO")
            return
        assignment[i] = 1 if comp[t_node] > comp[f_node] else 0
    
    print("YES")
    print(' '.join(map(str, assignment[1:])))

solve()
