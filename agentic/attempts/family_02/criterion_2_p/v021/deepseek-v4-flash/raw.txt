We can model each valve as a boolean variable, and each conduit as a clause that must be satisfied.  
This is exactly the **2-SAT** problem.

We’ll build an implication graph:
- For each clause `(a OR b)`, add implications `(¬a → b)` and `(¬b → a)`.
- Then run **Tarjan’s SCC** algorithm to check for contradictions.

If a variable and its negation are in the same SCC, it’s unsatisfiable.  
Otherwise, we assign values based on the topological order of SCCs.

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    N = 2 * n  # nodes: 0..n-1 for x_i false, n..2n-1 for x_i true

    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def node(var):
        # var > 0: x_var true -> index var-1 + n
        # var < 0: x_|var| false -> index |var|-1
        if var > 0:
            return var - 1 + n
        else:
            return -var - 1

    def add_imp(u, v):
        g[u].append(v)
        rg[v].append(u)

    for _ in range(m):
        a, b = map(int, input().split())
        na = node(-a)
        nb = node(-b)
        add_imp(na, node(b))
        add_imp(nb, node(a))

    # Kosaraju's algorithm
    visited = [False] * N
    order = []

    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in rg[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    for i in range(n):
        if comp[i] == comp[i + n]:
            print("NO")
            return

    print("YES")
    # assignment: if comp[false] > comp[true] then true
    res = [0] * n
    for i in range(n):
        if comp[i] > comp[i + n]:
            res[i] = 1
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
```
