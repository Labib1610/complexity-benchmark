We can model each microservice as a boolean variable, and each policy as a clause of two literals.  
This is exactly the **2-SAT** problem. We can solve it using implication graph and SCC (Tarjan or Kosaraju).

```python
import sys
sys.setrecursionlimit(1 << 20)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    # 2-SAT: variable i (0-based) -> 2*i = False, 2*i+1 = True
    N = 2 * n
    g = [[] for _ in range(N)]
    rg = [[] for _ in range(N)]

    def add_imp(u, v):
        g[u].append(v)
        rg[v].append(u)

    # clause (a or b) -> (~a => b) and (~b => a)
    for _ in range(m):
        a, b = map(int, input().split())
        # convert to 0-based index
        def lit(x):
            var = abs(x) - 1
            return 2 * var + (1 if x > 0 else 0)
        la = lit(a)
        lb = lit(b)
        # ~a => b
        add_imp(la ^ 1, lb)
        # ~b => a
        add_imp(lb ^ 1, la)

    # Kosaraju SCC
    visited = [False] * N
    order = []
    def dfs1(u):
        visited[u] = True
        for v in g[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    for i in range(N):
        if not visited[i]:
            dfs1(i)

    comp = [-1] * N
    def dfs2(u, cid):
        comp[u] = cid
        for v in rg[u]:
            if comp[v] == -1:
                dfs2(v, cid)

    cid = 0
    for u in reversed(order):
        if comp[u] == -1:
            dfs2(u, cid)
            cid += 1

    # Check satisfiability
    for i in range(n):
        if comp[2*i] == comp[2*i+1]:
            print("NO")
            return

    # Build assignment: choose the component with larger id (Kosaraju gives topological order)
    assignment = [False] * n
    for i in range(n):
        # comp[2*i] < comp[2*i+1] means false => true, so assign true
        assignment[i] = comp[2*i] > comp[2*i+1]

    print("YES")
    print(" ".join("1" if assignment[i] else "0" for i in range(n)))

if __name__ == "__main__":
    solve()
```
