import sys
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(2 * n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
        except StopIteration:
            break
        
        nu = -u
        nv = -v
        
        def get_node(x):
            if x > 0: return x
            return n + (-x)
        
        u_node = get_node(u)
        v_node = get_node(v)
        nu_node = get_node(nu)
        nv_node = get_node(nv)
        
        adj[nu_node].append(v_node)
        adj[nv_node].append(u_node)

    N = 2 * n + 1
    visited = [False] * N
    order = []
    
    def dfs1(u):
        stack = [(u, 0)]
        while stack:
            node, idx = stack[-1]
            if not visited[node]:
                visited[node] = True
            
            if idx < len(adj[node]):
                child = adj[node][idx]
                stack[-1] = (node, idx + 1)
                if not visited[child]:
                    stack.append((child, 0))
            else:
                stack.pop()
                order.append(node)

    for i in range(1, 2 * n + 1):
        if not visited[i]:
            dfs1(i)

    adj_rev = [[] for _ in range(N)]
    for u in range(1, N):
        for v in adj[u]:
            adj_rev[v].append(u)

    visited = [False] * N
    comp_id = [-1] * N
    
    current_comp_id = 0
    for u in reversed(order):
        if not visited[u]:
            stack = [u]
            visited[u] = True
            component_nodes = []
            
            while stack:
                node = stack.pop()
                component_nodes.append(node)
                for v in adj_rev[node]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            for node in component_nodes:
                comp_id[node] = current_comp_id
            current_comp_id += 1

    assignment = [0] * (n + 1)
    possible = True
    
    for i in range(1, n + 1):
        neg_i = n + i
        if comp_id[i] == comp_id[neg_i]:
            possible = False
            break
        if comp_id[i] > comp_id[neg_i]:
            assignment[i] = 1
        else:
            assignment[i] = 0

    if not possible:
        sys.stdout.write("NO\n")
    else:
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, assignment[1:])) + "\n")

if __name__ == '__main__':
    solve()
