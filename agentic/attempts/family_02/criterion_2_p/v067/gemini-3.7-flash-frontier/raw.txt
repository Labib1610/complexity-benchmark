import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    adj = [[] for _ in range(2 * n)]
    radj = [[] for _ in range(2 * n)]

    def get_lit(x):
        if x > 0:
            return 2 * (x - 1)
        else:
            return 2 * (-x - 1) + 1

    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2

        lu = get_lit(u)
        lv = get_lit(v)

        not_lu = lu ^ 1
        not_lv = lv ^ 1

        adj[not_lu].append(lv)
        adj[not_lv].append(lu)
        radj[lv].append(not_lu)
        radj[lu].append(not_lv)

    cur_edge = [0] * (2 * n)
    order = []
    visited = [False] * (2 * n)

    for i in range(2 * n):
        if not visited[i]:
            stack = [i]
            visited[i] = True
            while stack:
                u = stack[-1]
                if cur_edge[u] < len(adj[u]):
                    v = adj[u][cur_edge[u]]
                    cur_edge[u] += 1
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
                else:
                    order.append(u)
                    stack.pop()

    comp = [-1] * (2 * n)
    comp_count = 0

    for i in reversed(order):
        if comp[i] == -1:
            stack = [i]
            comp[i] = comp_count
            while stack:
                u = stack.pop()
                for v in radj[u]:
                    if comp[v] == -1:
                        comp[v] = comp_count
                        stack.append(v)
            comp_count += 1

    ans = [0] * n
    for i in range(n):
        if comp[2 * i] == comp[2 * i + 1]:
            print("NO")
            return
        if comp[2 * i] > comp[2 * i + 1]:
            ans[i] = 1
        else:
            ans[i] = 0

    print("YES")
    print(*(ans))


if __name__ == "__main__":
    solve()
