import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(600000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    rev_adj = [[] for _ in range(num_nodes)]
    
    def get_lit(l):
        if l > 0:
            return 2 * (l - 1)
        else:
            return 2 * (-l - 1) + 1
    
    idx = 2
    for _ in range(m):
        l1 = int(input_data[idx])
        l2 = int(input_data[idx + 1])
        idx += 2
        
        u = get_lit(l1)
        v = get_lit(l2)
        
        not_u = u ^ 1
        not_v = v ^ 1
        
        adj[not_u].append(v)
        adj[not_v].append(u)
        rev_adj[v].append(not_u)
        rev_adj[u].append(not_v)
        
    order = []
    visited = [False] * num_nodes
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [(i, 0)]
            visited[i] = True
            while stack:
                node, edge_idx = stack.pop()
                if edge_idx < len(adj[node]):
                    stack.append((node, edge_idx + 1))
                    neighbor = adj[node][edge_idx]
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        stack.append((neighbor, 0))
                else:
                    order.append(node)
                    
    comp = [-1] * num_nodes
    comp_id = 0
    
    for i in reversed(order):
        if comp[i] == -1:
            stack = [i]
            comp[i] = comp_id
            while stack:
                node = stack.pop()
                for neighbor in rev_adj[node]:
                    if comp[neighbor] == -1:
                        comp[neighbor] = comp_id
                        stack.append(neighbor)
            comp_id += 1
            
    ans = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if comp[pos] == comp[neg]:
            print("NO")
            return
        ans[i] = 1 if comp[pos] < comp[neg] else 0
        
    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
