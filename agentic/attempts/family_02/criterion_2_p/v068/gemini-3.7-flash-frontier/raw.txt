```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])

    # 2-SAT:
    # Variables 1 to n.
    # Positive literal +x -> 2 * (x - 1)
    # Negative literal -x -> 2 * (x - 1) + 1
    # Negation of literal v is v ^ 1

    adj = [[] for _ in range(2 * n)]

    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2

        # Convert u and v to literal indices
        if u > 0:
            lit_u = 2 * (u - 1)
        else:
            lit_u = 2 * (-u - 1) + 1

        if v > 0:
            lit_v = 2 * (v - 1)
        else:
            lit_v = 2 * (-v - 1) + 1

        # Clause: lit_u OR lit_v
        # ~lit_u => lit_v
        # ~lit_v => lit_u
        adj[lit_u ^ 1].append(lit_v)
        adj[lit_v ^ 1].append(lit_u)

    num_nodes = 2 * n
    dfn = [-1] * num_nodes
    low = [-1] * num_nodes
    scc = [-1] * num_nodes
    in_stack = [False] * num_nodes
    stack = []
    timer = 0
    scc_count = 0

    # Iterative Tarjan's algorithm
    for i in range(num_nodes):
        if dfn[i] != -1:
            continue

        call_stack = [(i, 0)]
        while call_stack:
            u, edge_idx = call_stack[-1]

            if edge_idx == 0:
                dfn[u] = low[u] = timer
                timer += 1
                stack.append(u)
                in_stack[u] = True

            pushed = False
            neighbors = adj[u]
            while edge_idx < len(neighbors):
                v = neighbors[edge_idx]
                edge_idx += 1
                if dfn[v] == -1:
                    call_stack[-1] = (u, edge_idx)
                    call_stack.append((v, 0))
                    pushed = True
                    break
                elif in_stack[v]:
                    if dfn[v] < low[u]:
                        low[u] = dfn[v]

            if pushed:
                continue

            call_stack.pop()
            if call_stack:
                p_u, _ = call_stack[-1]
                if low[u] < low[p_u]:
                    low[p_u] = low[u]

            if low[u] == dfn[u]:
                while True:
                    node = stack.pop()
                    in_stack[node] = False
                    scc[node] = scc_count
                    if node == u:
                        break
                scc_count += 1

    ans = [0] * n
    for i in range(n):
        pos = 2 * i
        neg = 2 * i + 1
        if scc[pos] == scc[neg]:
            print("NO")
            return
        # Tarjan's produces SCCs in reverse topological order.
        # If scc[pos] < scc[neg], pos appears later in topological order (or neg -> pos),
        # so pos should be True (1).
        ans[i] = 1 if scc[pos] < scc[neg] else 0

    print("YES")
    print(*(ans))

if __name__ == '__main__':
    solve()
```
