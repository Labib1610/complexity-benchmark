import sys
sys.setrecursionlimit(1 << 25)

def solve():
    input = sys.stdin.read().split()
    if not input:
        return
    it = iter(input)
    n = int(next(it))
    m = int(next(it))

    # 2-SAT problem: 2n variables (for each additive i: included(i), excluded(i))
    # included(i) is variable 2*i, excluded(i) is variable 2*i+1
    # For each additive, exactly one must be true
    # For each directive (a, b), at least one must be true
    
    # Graph for 2-SAT: 2n nodes, edges represent implications
    # (a1 OR b1) means (~a1 -> b1) and (~b1 -> a1)
    # We use indices: 
    # positive c (included): node 2*(c-1)
    # negative c (excluded): node 2*(c-1)+1
    
    N = 2 * n
    adj = [[] for _ in range(N)]
    
    for _ in range(m):
        c1 = int(next(it))
        c2 = int(next(it))
        
        # Get node indices for criteria being true
        # positive: included -> even index
        # negative: excluded -> odd index
        def get_node(c):
            idx = abs(c) - 1
            if c > 0:
                return 2 * idx
            else:
                return 2 * idx + 1
        
        a = get_node(c1)
        b = get_node(c2)
        
        # (a OR b) => (~a -> b) and (~b -> a)
        not_a = a ^ 1
        not_b = b ^ 1
        adj[not_a].append(b)
        adj[not_b].append(a)
    
    # Kosaraju's algorithm for SCC
    # Step 1: order by finish time (topological sort)
    visited = [False] * N
    order = []
    
    def dfs1(v):
        stack = [(v, False)]
        while stack:
            node, processed = stack.pop()
            if processed:
                order.append(node)
                continue
            if visited[node]:
                continue
            visited[node] = True
            stack.append((node, True))
            for w in adj[node]:
                if not visited[w]:
                    stack.append((w, False))
    
    for i in range(N):
        if not visited[i]:
            dfs1(i)
    
    # Step 2: process in reverse order, assigning components
    radj = [[] for _ in range(N)]
    for v in range(N):
        for w in adj[v]:
            radj[w].append(v)
    
    comp = [-1] * N
    comp_id = 0
    
    def dfs2(v, cid):
        stack = [v]
        while stack:
            node = stack.pop()
            if comp[node] != -1:
                continue
            comp[node] = cid
            for w in radj[node]:
                if comp[w] == -1:
                    stack.append(w)
    
    for v in reversed(order):
        if comp[v] == -1:
            dfs2(v, comp_id)
            comp_id += 1
    
    # Check if any variable and its negation are in the same SCC
    assignment = [0] * n
    for i in range(n):
        inc = 2 * i
        exc = 2 * i + 1
        if comp[inc] == comp[exc]:
            print("NO")
            return
        # Assign based on component order (higher comp_id means earlier in topo order, assigned True)
        # Actually, comp_id is assigned in order of processing, so lower comp_id is "earlier"
        # We want the one that is "later" in topological order to be True
        # Since we process in reverse finish order, lower comp_id comes later in topo order
        # So we want the one with lower comp_id to be True
        # Wait, let me think again...
        # In Kosaraju, we process nodes in reverse finish order, so first processed SCC has lowest comp_id
        # Lower comp_id means it appears later in the topological order of SCCs
        # In 2-SAT, we want to assign True to the node that appears later (higher in DAG)
        # So lower comp_id -> later in topo order -> should be True
        # Actually, higher comp_id comes earlier in topo order (reverse finish time)
        # Let me reconsider: we iterate order in reverse, so first popped is last finished
        # Actually order has nodes in finish order (topological), so reversed(order) is reverse topological
        # First processed in dfs2 is last in reversed(order), so first in topological order of SCCs
        # So comp_id 0 corresponds to a source SCC in the condensation graph
        # We want to set source SCCs to False, sink SCCs to True in 2-SAT standard approach
        # But wait, different formulations...
        # Let's use: if comp[inc] > comp[exc], then inc is assigned True
        # Because higher comp_id means processed later, which is earlier in reversed order
        # which is later in original topological order
        # Hmm, this is getting confusing. Let's use the standard:
        # Assign comp_id based on processing order in second pass
        # If comp[x] > comp[~x], then x is True (because x's SCC comes before ~x's SCC in reverse order)
        # Since we process in reverse finish order, higher comp_id means earlier in processing, later in original finish
        
        # Standard approach: if comp[node] > comp[not_node], then node is True
        if comp[inc] > comp[exc]:
            assignment[i] = 1
        else:
            assignment[i] = 0
    
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == '__main__':
    solve()
