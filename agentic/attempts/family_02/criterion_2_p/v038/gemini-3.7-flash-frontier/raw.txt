```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # 2n vertices: for variable i (1-indexed):
    # x_i is 2 * (i - 1)
    # ~x_i is 2 * (i - 1) + 1
    # not(u) is u ^ 1
    
    num_nodes = 2 * n
    adj = [[] for _ in range(num_nodes)]
    rev_adj = [[] for _ in range(num_nodes)]
    
    idx = 2
    for _ in range(m):
        c1 = int(input_data[idx])
        c2 = int(input_data[idx+1])
        idx += 2
        
        # Convert c1, c2 to node indices
        if c1 > 0:
            u = 2 * (c1 - 1)
        else:
            u = 2 * (-c1 - 1) + 1
            
        if c2 > 0:
            v = 2 * (c2 - 1)
        else:
            v = 2 * (-c2 - 1) + 1
            
        # Clause is (u or v) => (~u -> v) and (~v -> u)
        not_u = u ^ 1
        not_v = v ^ 1
        
        adj[not_u].append(v)
        adj[not_v].append(u)
        rev_adj[v].append(not_u)
        rev_adj[u].append(not_v)
        
    # Kosaraju's algorithm (Iterative)
    visited = [False] * num_nodes
    order = []
    
    for i in range(num_nodes):
        if not visited[i]:
            stack = [i]
            edge_idx = [0]
            visited[i] = True
            while stack:
                u = stack[-1]
                e = edge_idx[-1]
                if e < len(adj[u]):
                    edge_idx[-1] += 1
                    nxt = adj[u][e]
                    if not visited[nxt]:
                        visited[nxt] = True
                        stack.append(nxt)
                        edge_idx.append(0)
                else:
                    order.append(u)
                    stack.pop()
                    edge_idx.pop()
                    
    scc = [-1] * num_nodes
    current_scc = 0
    
    for i in reversed(order):
        if scc[i] == -1:
            stack = [i]
            scc[i] = current_scc
            while stack:
                u = stack.pop()
                for nxt in rev_adj[u]:
                    if scc[nxt] == -1:
                        scc[nxt] = current_scc
                        stack.append(nxt)
            current_scc += 1
            
    assignment = [0] * n
    for i in range(n):
        pos_var = 2 * i
        neg_var = 2 * i + 1
        if scc[pos_var] == scc[neg_var]:
            print("NO")
            return
        # In Kosaraju, SCCs found earlier in the second pass are downstream (sinks) in the SCC DAG
        # Thus, smaller scc id means downstream.
        # If scc[pos_var] < scc[neg_var], pos_var is downstream, so assign pos_var = True.
        if scc[pos_var] < scc[neg_var]:
            assignment[i] = 1
        else:
            assignment[i] = 0
            
    print("YES")
    print(*(assignment))

if __name__ == '__main__':
    solve()
```
