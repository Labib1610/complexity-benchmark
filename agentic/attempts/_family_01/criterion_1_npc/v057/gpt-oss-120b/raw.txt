```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [-1] * n  # -1 uncolored, else 0,1,2 for phases 1,2,3

    # Precompute degrees
    degree = [len(adj[i]) for i in range(n)]

    # Order vertices for tie-breaking (higher degree first)
    order = sorted(range(n), key=lambda x: -degree[x])

    # Helper to select next vertex: DSATUR (max saturation, then degree)
    def select_vertex():
        best = -1
        best_sat = -1
        best_deg = -1
        for v in range(n):
            if colors[v] != -1:
                continue
            # saturation: number of different colors among neighbors
            neigh_colors = {colors[u] for u in adj[v] if colors[u] != -1}
            sat = len(neigh_colors)
            if sat > best_sat or (sat == best_sat and degree[v] > best_deg):
                best_sat = sat
                best_deg = degree[v]
                best = v
        return best

    def dfs(colored_cnt: int) -> bool:
        if colored_cnt == n:
            return True
        v = select_vertex()
        # try colors 0..2
        for c in range(3):
            conflict = False
            for nb in adj[v]:
                if colors[nb] == c:
                    conflict = True
                    break
            if not conflict:
                colors[v] = c
                if dfs(colored_cnt + 1):
                    return True
                colors[v] = -1
        return False

    possible = dfs(0)
    if not possible:
        print("NO")
    else:
        print("YES")
        # convert to 1..3
        print(' '.join(str(c+1) for c in colors))

if __name__ == "__main__":
    solve()
```
