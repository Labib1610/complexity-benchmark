def solve():
    import sys
    from collections import defaultdict
    
    # Read input
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list representation of the graph
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Special case: no edges
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Try to 3-color the graph using backtracking
    colors = [0] * (n + 1)  # colors[i] represents the color of node i (1-indexed)
    
    def is_safe(node, color):
        """Check if assigning 'color' to 'node' is safe"""
        for neighbor in adj[node]:
            if colors[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        """Try to color nodes from 'node' onwards"""
        if node > n:
            return True
            
        if colors[node] != 0:  # Node already colored
            return backtrack(node + 1)
            
        for color in [1, 2, 3]:
            if is_safe(node, color):
                colors[node] = color
                if backtrack(node + 1):
                    return True
                colors[node] = 0  # Backtrack
                
        return False
    
    # Start backtracking from node 1
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, colors[1:])))
    else:
        print("NO")

solve()
