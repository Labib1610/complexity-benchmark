import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [[] for _ in range(n)]
    adj_mask = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)

    # 3-coloring using DSATUR / Backtracking with MRV and forward checking
    color = [-1] * n
    # mask of banned colors for each vertex: bit 0: color 0, bit 1: color 1, bit 2: color 2
    banned = [0] * n

    # Precompute degrees
    deg = [len(adj[i]) for i in range(n)]

    def backtrack(colored_count):
        if colored_count == n:
            return True
        
        # Select uncolored vertex with maximum saturation (most banned colors),
        # tie-break by uncolored degree
        best_u = -1
        max_sat = -1
        max_deg = -1

        for u in range(n):
            if color[u] == -1:
                b = banned[u]
                sat = (b & 1) + ((b >> 1) & 1) + ((b >> 2) & 1)
                if sat == 3:
                    return False  # No valid color
                if sat > max_sat or (sat == max_sat and deg[u] > max_deg):
                    max_sat = sat
                    max_deg = deg[u]
                    best_u = u

        u = best_u
        for c in range(3):
            if not (banned[u] & (1 << c)):
                color[u] = c
                # Update neighbors
                changed = []
                c_mask = 1 << c
                for v in adj[u]:
                    if color[v] == -1 and not (banned[v] & c_mask):
                        banned[v] |= c_mask
                        changed.append(v)
                
                if backtrack(colored_count + 1):
                    return True
                
                # Undo
                for v in changed:
                    banned[v] &= ~c_mask
                color[u] = -1

        return False

    if backtrack(0):
        print("YES")
        print(*(c + 1 for c in color))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
