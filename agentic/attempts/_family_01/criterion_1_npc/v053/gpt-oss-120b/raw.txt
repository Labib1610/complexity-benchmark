```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    order = sorted(range(n), key=lambda x: -len(adj[x]))
    color = [-1] * n

    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        used = 0
        for nb in adj[v]:
            c = color[nb]
            if c != -1:
                used |= 1 << c
        for c in range(3):
            if not (used >> c) & 1:
                color[v] = c
                if dfs(idx + 1):
                    return True
                color[v] = -1
        return False

    if dfs(0):
        out = ['YES', ' '.join(str(c + 1) for c in color)]
    else:
        out = ['NO']
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
