```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [0] * n  # 0 = uncolored, else 1..3
    # for each vertex, keep mask of forbidden colors from already colored neighbours
    forbidden = [0] * n  # bitmask of colors (bit 0 for color1, etc)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # compute forbidden from already colored neighbours
        forbid = 0
        neigh = adj[v]
        while neigh:
            u = (neigh & -neigh).bit_length() - 1
            neigh &= neigh - 1
            c = colors[u]
            if c:
                forbid |= 1 << (c - 1)
        for col in range(1, 4):
            if not (forbid & (1 << (col - 1))):
                colors[v] = col
                if dfs(idx + 1):
                    return True
                colors[v] = 0
        return False

    possible = dfs(0)
    out = sys.stdout
    if not possible:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(" ".join(str(colors[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
