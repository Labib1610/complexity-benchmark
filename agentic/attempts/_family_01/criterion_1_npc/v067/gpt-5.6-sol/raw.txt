import sys

sys.setrecursionlimit(1000000)


def color_component(vertices, global_adj):
    k = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}
    all_mask = (1 << k) - 1
    adj = [0] * k

    for i, v in enumerate(vertices):
        mask = 0
        neighbors = global_adj[v]
        for u in vertices:
            if (neighbors >> u) & 1:
                mask |= 1 << index[u]
        adj[i] = mask

    def bipartite(mask):
        side_a = 0
        side_b = 0
        unseen = mask

        while unseen:
            root = unseen & -unseen
            unseen ^= root
            side_a |= root
            queue = root

            while queue:
                bit = queue & -queue
                queue ^= bit
                v = bit.bit_length() - 1

                if side_a & bit:
                    if adj[v] & side_a:
                        return None
                    new_vertices = adj[v] & unseen
                    side_b |= new_vertices
                else:
                    if adj[v] & side_b:
                        return None
                    new_vertices = adj[v] & unseen
                    side_a |= new_vertices

                unseen &= ~new_vertices
                queue |= new_vertices

        return side_a, side_b

    direct = bipartite(all_mask)
    if direct is not None:
        return 0, direct[0], direct[1]

    complement_adj = [
        all_mask & ~(adj[i] | (1 << i))
        for i in range(k)
    ]

    answer = None

    def search(independent, possible, excluded):
        nonlocal answer

        if possible == 0 and excluded == 0:
            remaining = all_mask ^ independent
            partition = bipartite(remaining)
            if partition is not None:
                answer = (independent, partition[0], partition[1])
                return True
            return False

        pivot_pool = possible | excluded
        if pivot_pool:
            best_pivot = -1
            best_count = -1
            bits = pivot_pool
            while bits:
                bit = bits & -bits
                bits ^= bit
                v = bit.bit_length() - 1
                count = (possible & complement_adj[v]).bit_count()
                if count > best_count:
                    best_count = count
                    best_pivot = v
            candidates = possible & ~complement_adj[best_pivot] & all_mask
        else:
            candidates = possible

        while candidates:
            bit = candidates & -candidates
            candidates ^= bit
            v = bit.bit_length() - 1

            if search(
                independent | bit,
                possible & complement_adj[v],
                excluded & complement_adj[v]
            ):
                return True

            possible ^= bit
            excluded |= bit

        return False

    if not search(0, all_mask, 0):
        return None
    return answer


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n

    p = 2
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    visited = 0
    components = []

    for start in range(n):
        if (visited >> start) & 1:
            continue

        component = []
        stack = [start]
        visited |= 1 << start

        while stack:
            v = stack.pop()
            component.append(v)
            remaining = adj[v] & ~visited
            while remaining:
                bit = remaining & -remaining
                remaining ^= bit
                u = bit.bit_length() - 1
                visited |= bit
                stack.append(u)

        components.append(component)

    result = [0] * n

    for component in components:
        colored = color_component(component, adj)
        if colored is None:
            print("NO")
            return

        color1, color2, color3 = colored
        for i, v in enumerate(component):
            bit = 1 << i
            if color1 & bit:
                result[v] = 1
            elif color2 & bit:
                result[v] = 2
            else:
                result[v] = 3

    print("YES")
    print(*result)


if __name__ == "__main__":
    main()
