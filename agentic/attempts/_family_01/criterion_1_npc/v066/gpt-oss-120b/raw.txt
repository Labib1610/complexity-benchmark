import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # allowed colors bitmask per vertex, bits 0,1,2 correspond to colors 1,2,3
    allowed = [0b111] * n
    color = [-1] * n
    uncolored = set(range(n))

    # MRV heuristic: vertex with fewest allowed colors (>0)
    def select_vertex():
        v = -1
        best = 4
        for u in uncolored:
            cnt = (allowed[u] & 0b111).bit_count()
            if cnt < best:
                best = cnt
                v = u
                if cnt == 1:
                    break
        return v

    # recursive backtracking
    def dfs() -> bool:
        if not uncolored:
            return True
        v = select_vertex()
        if v == -1:
            return False
        avail = allowed[v] & 0b111
        # try each possible color
        for c in range(3):
            if not (avail >> c) & 1:
                continue
            # assign
            color[v] = c
            uncolored.remove(v)
            changed = []  # list of (vertex, previous_allowed)
            ok = True
            # update neighbors
            nb_mask = adj[v]
            while nb_mask:
                u = (nb_mask & -nb_mask).bit_length() - 1
                nb_mask &= nb_mask - 1
                if u in uncolored:
                    prev = allowed[u]
                    allowed[u] &= ~(1 << c)
                    if allowed[u] == 0:
                        ok = False
                    if allowed[u] != prev:
                        changed.append((u, prev))
            if ok and dfs():
                return True
            # revert
            for u, prev in changed:
                allowed[u] = prev
            color[v] = -1
            uncolored.add(v)
        return False

    possible = dfs()
    if not possible:
        print("NO")
    else:
        print("YES")
        # convert 0/1/2 to 1/2/3
        print(' '.join(str(c + 1) for c in color))

if __name__ == "__main__":
    solve()
