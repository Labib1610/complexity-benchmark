import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    adj_mask = [0] * n
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v and not ((adj_mask[u] >> v) & 1):
            adj[u].append(v)
            adj[v].append(u)
            adj_mask[u] |= (1 << v)
            adj_mask[v] |= (1 << u)
            
    # Backtracking with MRV and Forward Checking
    # Available colors represented as bitmask: bit 0 -> color 1, bit 1 -> color 2, bit 2 -> color 3
    # Initial masks: 7 (binary 111) for all vertices
    
    color = [0] * n  # 0 means unassigned, 1, 2, 3 assigned
    degrees = [len(adj[i]) for i in range(n)]
    
    def backtrack(domains):
        # Find unassigned vertex with min remaining values
        best_v = -1
        min_rem = 4
        max_deg = -1
        
        for i in range(n):
            if color[i] == 0:
                mask = domains[i]
                rem = (mask & 1) + ((mask >> 1) & 1) + ((mask >> 2) & 1)
                if rem == 0:
                    return False
                if rem < min_rem or (rem == min_rem and degrees[i] > max_deg):
                    min_rem = rem
                    best_v = i
                    max_deg = degrees[i]
                    
        if best_v == -1:
            return True
            
        cand_mask = domains[best_v]
        # Try available colors
        for c in (0, 1, 2):
            if (cand_mask >> c) & 1:
                color[best_v] = c + 1
                new_domains = list(domains)
                new_domains[best_v] = (1 << c)
                
                # Propagate
                queue = [best_v]
                valid = True
                while queue and valid:
                    u = queue.pop()
                    u_col = color[u] - 1
                    mask_to_remove = ~(1 << u_col)
                    
                    for v in adj[u]:
                        if (new_domains[v] >> u_col) & 1:
                            new_domains[v] &= mask_to_remove
                            rem_v = (new_domains[v] & 1) + ((new_domains[v] >> 1) & 1) + ((new_domains[v] >> 2) & 1)
                            if rem_v == 0:
                                valid = False
                                break
                            if rem_v == 1 and color[v] == 0:
                                # Determine single remaining color
                                if new_domains[v] == 1:
                                    color[v] = 1
                                elif new_domains[v] == 2:
                                    color[v] = 2
                                else:
                                    color[v] = 3
                                queue.append(v)
                                
                if valid:
                    if backtrack(new_domains):
                        return True
                        
                # Undo assignments done in queue
                color[best_v] = 0
                for i in range(n):
                    if color[i] != 0 and (domains[i] & (domains[i] - 1)) != 0:
                        color[i] = 0
                        
        return False

    initial_domains = [7] * n
    if backtrack(initial_domains):
        print("YES")
        print(*(color[i] if color[i] != 0 else 1 for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
