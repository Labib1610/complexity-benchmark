import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Check if any node has degree >= 3 (cannot be colored with 3 colors)
    for i in range(n):
        if len(adj[i]) >= 3:
            print("NO")
            return
    
    # Find connected components
    visited = [False] * n
    color = [0] * n
    
    for start in range(n):
        if visited[start]:
            continue
        
        # BFS to find component
        component = []
        queue = [start]
        visited[start] = True
        while queue:
            u = queue.pop()
            component.append(u)
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    queue.append(v)
        
        # Check if component is bipartite
        is_bipartite = True
        for u in component:
            for v in adj[u]:
                if v > u:  # Check each edge once
                    for w in adj[u]:
                        if w > v and w in adj[v]:
                            is_bipartite = False
                            break
                    if not is_bipartite:
                        break
            if not is_bipartite:
                break
        
        if not is_bipartite:
            # Component has a triangle, need 3 colors
            # Find the triangle
            triangle = None
            for u in component:
                for v in adj[u]:
                    if v > u:
                        for w in adj[u]:
                            if w > v and w in adj[v]:
                                triangle = (u, v, w)
                                break
                        if triangle:
                            break
                if triangle:
                    break
            
            u, v, w = triangle
            color[u] = 1
            color[v] = 2
            color[w] = 3
            
            # BFS to color rest of component
            q = [u, v, w]
            visited_tri = {u: True, v: True, w: True}
            head = 0
            while head < len(q):
                curr = q[head]
                head += 1
                used = {color[curr]}
                for neighbor in adj[curr]:
                    if neighbor in visited_tri:
                        used.add(color[neighbor])
                avail = {1, 2, 3} - used
                for neighbor in adj[curr]:
                    if neighbor not in visited_tri:
                        visited_tri[neighbor] = True
                        # Find available color for neighbor
                        neighbor_used = set()
                        for nn in adj[neighbor]:
                            if nn in visited_tri:
                                neighbor_used.add(color[nn])
                        neighbor_avail = {1, 2, 3} - neighbor_used
                        if neighbor_avail:
                            color[neighbor] = neighbor_avail.pop()
                        else:
                            print("NO")
                            return
                        q.append(neighbor)
        else:
            # Component is bipartite, use colors 1 and 2
            color[start] = 1
            queue = [start]
            while queue:
                u = queue.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
