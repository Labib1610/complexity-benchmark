```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by descending degree
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for i, v in enumerate(order):
        pos_in_order[v] = i

    colors = [0] * n
    found = False

    # precompute neighbor list in order indices for quick check
    neigh_order = [[pos_in_order[j] for j in range(n) if (adj[i] >> j) & 1] for i in range(n)]

    def dfs(idx: int) -> bool:
        nonlocal found
        if idx == n:
            return True
        v = order[idx]
        # collect used colors by already colored neighbors
        used = 0
        for nb in neigh_order[v]:
            if nb < idx:  # neighbor already colored
                used |= 1 << (colors[order[nb]] - 1)
        for c in (1, 2, 3):
            if not (used & (1 << (c - 1))):
                colors[v] = c
                if dfs(idx + 1):
                    return True
                colors[v] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(str(colors[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
