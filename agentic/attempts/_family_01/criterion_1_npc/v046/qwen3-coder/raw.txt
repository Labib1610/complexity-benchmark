```python
import sys
from collections import defaultdict

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    return n, edges

def solve(n, edges):
    # Build adjacency list
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to 3-color the graph
    color = [-1] * n  # -1 means uncolored
    
    def is_valid(v, c):
        # Check if we can assign color c to vertex v
        for neighbor in adj[v]:
            if color[neighbor] == c:
                return False
        return True
    
    def dfs(v):
        if v == n:
            return True
        
        for c in range(3):  # Colors 0, 1, 2 (representing shifts 1, 2, 3)
            if is_valid(v, c):
                color[v] = c
                if dfs(v + 1):
                    return True
                color[v] = -1  # Backtrack
        
        return False
    
    # Start DFS coloring
    if dfs(0):
        # Convert colors to shifts (1, 2, 3)
        shifts = [str(c + 1) for c in color]
        return "YES\n" + " ".join(shifts)
    else:
        return "NO"

def main():
    n, edges = read_input()
    result = solve(n, edges)
    print(result)

if __name__ == "__main__":
    main()
```
