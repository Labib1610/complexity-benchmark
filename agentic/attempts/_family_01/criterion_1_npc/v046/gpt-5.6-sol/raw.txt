import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj_mask = [0] * n

    p = 2
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u != v:
            adj_mask[u] |= 1 << v
            adj_mask[v] |= 1 << u

    # Remove vertices of degree at most two. Any coloring of the remaining
    # graph can always be extended to these vertices in reverse order.
    active = (1 << n) - 1
    degree = [adj_mask[v].bit_count() for v in range(n)]
    queue = deque(v for v in range(n) if degree[v] <= 2)
    removed_order = []

    while queue:
        v = queue.popleft()
        if not ((active >> v) & 1) or degree[v] > 2:
            continue
        active ^= 1 << v
        removed_order.append(v)

        neighbors = adj_mask[v] & active
        while neighbors:
            bit = neighbors & -neighbors
            u = bit.bit_length() - 1
            neighbors ^= bit
            degree[u] -= 1
            if degree[u] <= 2:
                queue.append(u)

    answer = [-1] * n

    def solve_component(vertices):
        k = len(vertices)
        index = {v: i for i, v in enumerate(vertices)}
        local_adj = [[] for _ in range(k)]

        for i, v in enumerate(vertices):
            bits = adj_mask[v] & active
            while bits:
                bit = bits & -bits
                u = bit.bit_length() - 1
                bits ^= bit
                j = index.get(u)
                if j is not None:
                    local_adj[i].append(j)

        # Fast path for bipartite components.
        bip_color = [-1] * k
        is_bipartite = True
        for start in range(k):
            if bip_color[start] != -1:
                continue
            bip_color[start] = 0
            q = deque([start])
            while q and is_bipartite:
                v = q.popleft()
                for u in local_adj[v]:
                    if bip_color[u] == -1:
                        bip_color[u] = bip_color[v] ^ 1
                        q.append(u)
                    elif bip_color[u] == bip_color[v]:
                        is_bipartite = False
                        break
            if not is_bipartite:
                break

        if is_bipartite:
            for i, v in enumerate(vertices):
                answer[v] = bip_color[i]
            return True

        colors = [-1] * k
        blocked_count = [[0, 0, 0] for _ in range(k)]
        trail = []
        degrees = [len(local_adj[i]) for i in range(k)]

        def available_mask(v):
            counts = blocked_count[v]
            mask = 0
            if counts[0] == 0:
                mask |= 1
            if counts[1] == 0:
                mask |= 2
            if counts[2] == 0:
                mask |= 4
            return mask

        def assign(v, c):
            colors[v] = c
            trail.append(v)
            for u in local_adj[v]:
                if colors[u] == -1:
                    blocked_count[u][c] += 1

        def rollback(checkpoint):
            while len(trail) > checkpoint:
                v = trail.pop()
                c = colors[v]
                colors[v] = -1
                for u in local_adj[v]:
                    if colors[u] == -1:
                        blocked_count[u][c] -= 1

        def dfs():
            base = len(trail)

            # Unit propagation: a vertex with only one available color is forced.
            while True:
                forced_vertex = -1
                forced_color = -1

                for v in range(k):
                    if colors[v] != -1:
                        continue
                    mask = available_mask(v)
                    if mask == 0:
                        rollback(base)
                        return False
                    if mask & (mask - 1) == 0:
                        forced_vertex = v
                        forced_color = mask.bit_length() - 1
                        break

                if forced_vertex == -1:
                    break
                assign(forced_vertex, forced_color)

            best = -1
            best_mask = 0
            best_key = None

            for v in range(k):
                if colors[v] != -1:
                    continue
                mask = available_mask(v)
                domain_size = mask.bit_count()
                uncolored_degree = 0
                for u in local_adj[v]:
                    if colors[u] == -1:
                        uncolored_degree += 1

                key = (3 - domain_size, uncolored_degree, degrees[v])
                if best_key is None or key > best_key:
                    best_key = key
                    best = v
                    best_mask = mask

            if best == -1:
                return True

            choices = []
            mask = best_mask
            while mask:
                bit = mask & -mask
                mask ^= bit
                c = bit.bit_length() - 1
                impact = 0
                for u in local_adj[best]:
                    if colors[u] == -1 and blocked_count[u][c] == 0:
                        impact += 1
                choices.append((impact, c))
            choices.sort()

            checkpoint = len(trail)
            for _, c in choices:
                assign(best, c)
                if dfs():
                    return True
                rollback(checkpoint)

            rollback(base)
            return False

        # Fix the colors of an edge to remove all global color symmetry.
        edge_u = edge_v = -1
        best_edge_score = -1
        for v in range(k):
            for u in local_adj[v]:
                if v < u:
                    score = degrees[v] + degrees[u]
                    if score > best_edge_score:
                        best_edge_score = score
                        edge_u, edge_v = v, u

        assign(edge_u, 0)
        assign(edge_v, 1)

        if not dfs():
            return False

        for i, v in enumerate(vertices):
            answer[v] = colors[i]
        return True

    # Solve every connected component of the remaining core.
    unseen = active
    while unseen:
        bit = unseen & -unseen
        start = bit.bit_length() - 1
        unseen ^= bit

        component = []
        stack = [start]
        while stack:
            v = stack.pop()
            component.append(v)
            neighbors = adj_mask[v] & unseen
            while neighbors:
                b = neighbors & -neighbors
                u = b.bit_length() - 1
                neighbors ^= b
                unseen ^= b
                stack.append(u)

        if not solve_component(component):
            print("NO")
            return

    # Extend the coloring to the removed vertices.
    for v in reversed(removed_order):
        used = 0
        bits = adj_mask[v]
        while bits:
            bit = bits & -bits
            u = bit.bit_length() - 1
            bits ^= bit
            if answer[u] != -1:
                used |= 1 << answer[u]

        for c in range(3):
            if not (used >> c) & 1:
                answer[v] = c
                break

    print("YES")
    print(*(c + 1 for c in answer))

if __name__ == "__main__":
    sys.setrecursionlimit(10000)
    main()
