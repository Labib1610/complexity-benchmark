import sys

def solve():
    data = sys.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    adj = [0]*(n+1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # color: 0 unassigned, 1/2/3 shifts
    color = [0]*(n+1)

    # For each uncolored vertex, try colors 1,2,3. If a color works,
    # assign greedily. This is fine because we only need a proper 3-coloring
    # of the conflict graph; greedy with 3 colors and backtracking when needed.
    # Since n<=40, we can do simple backtracking.
    
    # But greedy without backtracking may fail on 3-colorable graphs.
    # Let's implement backtracking with ordering.
    
    vertices = list(range(1, n+1))
    # Sort by degree descending for better pruning
    vertices.sort(key=lambda x: -bin(adj[x]).count('1'))
    
    def backtrack(i):
        if i == len(vertices):
            return True
        v = vertices[i]
        used = 0
        # gather colors used by neighbors already colored
        # iterate over neighbors
        mask = adj[v]
        b = mask
        while b:
            low = b & -b
            nb = low.bit_length() - 1
            if color[nb]:
                used |= 1 << color[nb]
            b ^= low
        for c in (1,2,3):
            if not (used & (1 << c)):
                color[v] = c
                if backtrack(i+1):
                    return True
                color[v] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(' '.join(str(color[i]) for i in range(1, n+1)))
    else:
        print("NO")

solve()
