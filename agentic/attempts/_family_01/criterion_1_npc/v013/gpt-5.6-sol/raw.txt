import sys

sys.setrecursionlimit(10000)


def solve_component(vertices, global_adj):
    k = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}
    adj = [[] for _ in range(k)]
    adj_bits = [0] * k

    for i, v in enumerate(vertices):
        for w in global_adj[v]:
            if w in index:
                j = index[w]
                adj[i].append(j)
                adj_bits[i] |= 1 << j

    # Bipartite components can be colored immediately.
    bip_color = [-1] * k
    bipartite = True
    for start in range(k):
        if bip_color[start] != -1:
            continue
        bip_color[start] = 0
        queue = [start]
        for x in queue:
            for y in adj[x]:
                if bip_color[y] == -1:
                    bip_color[y] = bip_color[x] ^ 1
                    queue.append(y)
                elif bip_color[y] == bip_color[x]:
                    bipartite = False
                    break
            if not bipartite:
                break
        if not bipartite:
            break

    if bipartite:
        return bip_color

    domains = [0b111] * k
    colors = [-1] * k

    def assign_vertex(dom, col, vertex, color):
        bit = 1 << color
        if not (dom[vertex] & bit):
            return None

        nd = dom.copy()
        nc = col.copy()
        nd[vertex] = bit
        queue = [vertex]

        while queue:
            x = queue.pop()
            mask = nd[x]
            if mask == 0 or (mask & (mask - 1)):
                return None

            cx = mask.bit_length() - 1
            if nc[x] != -1:
                if nc[x] != cx:
                    return None
                continue

            nc[x] = cx
            cbit = 1 << cx

            for y in adj[x]:
                if nc[y] == cx:
                    return None
                if nc[y] == -1 and (nd[y] & cbit):
                    new_mask = nd[y] & ~cbit
                    if new_mask == 0:
                        return None
                    nd[y] = new_mask
                    if not (new_mask & (new_mask - 1)):
                        queue.append(y)

        return nd, nc

    def dfs(dom, col):
        best = -1
        best_size = 4
        best_degree = -1

        for v in range(k):
            if col[v] != -1:
                continue
            size = dom[v].bit_count()
            degree = 0
            for u in adj[v]:
                if col[u] == -1:
                    degree += 1

            if size < best_size or (size == best_size and degree > best_degree):
                best = v
                best_size = size
                best_degree = degree

        if best == -1:
            return col

        used = 0
        for c in col:
            if c != -1:
                used |= 1 << c

        possible = dom[best]
        choices = []

        for c in range(3):
            bit = 1 << c
            if (possible & bit) and (used & bit):
                impact = sum(
                    1 for u in adj[best]
                    if col[u] == -1 and (dom[u] & bit)
                )
                choices.append((impact, c))

        unused_possible = possible & ~used & 0b111
        if unused_possible:
            bit = unused_possible & -unused_possible
            c = bit.bit_length() - 1
            impact = sum(
                1 for u in adj[best]
                if col[u] == -1 and (dom[u] & bit)
            )
            choices.append((impact, c))

        choices.sort()

        for _, c in choices:
            result = assign_vertex(dom, col, best, c)
            if result is None:
                continue
            answer = dfs(*result)
            if answer is not None:
                return answer

        return None

    # If a triangle exists, its colors can be fixed to 1, 2, 3 by symmetry.
    triangle = None
    full_mask = (1 << k) - 1
    for i in range(k):
        higher_i = adj_bits[i] & (full_mask ^ ((1 << (i + 1)) - 1))
        bits = higher_i
        while bits:
            lb = bits & -bits
            j = lb.bit_length() - 1
            common = higher_i & adj_bits[j]
            common &= full_mask ^ ((1 << (j + 1)) - 1)
            if common:
                lb2 = common & -common
                triangle = (i, j, lb2.bit_length() - 1)
                break
            bits ^= lb
        if triangle is not None:
            break

    if triangle is not None:
        for v, c in zip(triangle, (0, 1, 2)):
            result = assign_vertex(domains, colors, v, c)
            if result is None:
                return None
            domains, colors = result
    else:
        start = max(range(k), key=lambda v: len(adj[v]))
        result = assign_vertex(domains, colors, start, 0)
        if result is None:
            return None
        domains, colors = result

    return dfs(domains, colors)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [set() for _ in range(n)]
    pos = 2

    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2

        if u == v:
            print("NO")
            return

        adj[u].add(v)
        adj[v].add(u)

    components = []
    seen = [False] * n

    for start in range(n):
        if seen[start]:
            continue
        seen[start] = True
        component = []
        stack = [start]

        while stack:
            v = stack.pop()
            component.append(v)
            for u in adj[v]:
                if not seen[u]:
                    seen[u] = True
                    stack.append(u)

        components.append(component)

    answer = [0] * n

    for component in components:
        coloring = solve_component(component, adj)
        if coloring is None:
            print("NO")
            return

        for local, vertex in enumerate(component):
            answer[vertex] = coloring[local] + 1

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
