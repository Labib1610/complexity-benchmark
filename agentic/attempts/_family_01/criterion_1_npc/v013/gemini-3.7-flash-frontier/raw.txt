import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_mask = (1 << n) - 1

    def check_bipartite(S):
        color = [-1] * n
        for i in range(n):
            if (S >> i) & 1 and color[i] == -1:
                color[i] = 2
                queue = [i]
                head = 0
                while head < len(queue):
                    u = queue[head]
                    head += 1
                    c = color[u]
                    next_c = 5 - c  # 2 -> 3, 3 -> 2
                    neighbors = adj[u] & S
                    while neighbors:
                        lsb = neighbors & -neighbors
                        v = lsb.bit_length() - 1
                        if color[v] != -1:
                            if color[v] == c:
                                return None
                        else:
                            color[v] = next_c
                            queue.append(v)
                        neighbors ^= lsb
        return color

    ans = None

    def bron_kerbosch(R, P, X):
        nonlocal ans
        if ans is not None:
            return
        if not P and not X:
            rem = all_mask ^ R
            res = check_bipartite(rem)
            if res is not None:
                for i in range(n):
                    if (R >> i) & 1:
                        res[i] = 1
                ans = res
            return

        # Choose pivot u in P | X that minimizes |P & (adj[u] | (1 << u))|
        u_candidates = P | X
        min_cand_size = 45
        best_cand = 0

        temp = u_candidates
        while temp:
            lsb = temp & -temp
            u = lsb.bit_length() - 1
            cand = P & (adj[u] | (1 << u))
            sz = bin(cand).count("1")
            if sz < min_cand_size:
                min_cand_size = sz
                best_cand = cand
                if sz <= 1:
                    break
            temp ^= lsb

        cand = best_cand
        while cand and ans is None:
            lsb = cand & -cand
            v = lsb.bit_length() - 1

            new_P = P & ~adj[v] & ~(1 << v)
            new_X = X & ~adj[v] & ~(1 << v)
            bron_kerbosch(R | (1 << v), new_P, new_X)

            P &= ~(1 << v)
            X |= 1 << v
            cand ^= lsb

    bron_kerbosch(0, all_mask, 0)

    if ans is None:
        print("NO")
    else:
        print("YES")
        print(*(ans[i] for i in range(n)))


if __name__ == "__main__":
    solve()
