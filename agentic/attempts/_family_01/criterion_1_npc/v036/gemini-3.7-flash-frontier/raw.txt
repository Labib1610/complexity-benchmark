import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)

    # Check for self-loops
    for i in range(n):
        # Already handled by u != v, but if multi-edges or anything:
        pass

    color = [-1] * n
    # mask of used colors for each vertex (bit 0: color 0, bit 1: color 1, bit 2: color 2)
    used = [0] * n

    # Precompute degrees
    deg = [bin(adj[i]).count('1') for i in range(n)]

    def backtrack(assigned_count):
        if assigned_count == n:
            return True

        # Find vertex with minimum available colors
        best_v = -1
        min_avail = 4
        max_deg = -1

        for v in range(n):
            if color[v] == -1:
                avail_mask = (~used[v]) & 7
                avail_count = bin(avail_mask).count('1')
                if avail_count == 0:
                    return False
                if avail_count < min_avail:
                    min_avail = avail_count
                    best_v = v
                    max_deg = deg[v]
                    if min_avail == 1:
                        break
                elif avail_count == min_avail:
                    if deg[v] > max_deg:
                        best_v = v
                        max_deg = deg[v]

        v = best_v
        avail_mask = (~used[v]) & 7

        for c in range(3):
            if (avail_mask & (1 << c)):
                color[v] = c
                # Update neighbors' used colors
                neighbors_changed = []
                c_mask = 1 << c
                nbrs = adj[v]
                while nbrs:
                    lsb = nbrs & -nbrs
                    u = (lsb).bit_length() - 1
                    nbrs ^= lsb
                    if not (used[u] & c_mask):
                        used[u] |= c_mask
                        neighbors_changed.append(u)

                if backtrack(assigned_count + 1):
                    return True

                # Undo
                for u in neighbors_changed:
                    used[u] ^= c_mask
                color[v] = -1

        return False

    if backtrack(0):
        print("YES")
        print(*(color[i] + 1 for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
