import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    for i in range(m):
        u = int(input_data[2 + 2 * i]) - 1
        v = int(input_data[3 + 2 * i]) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        
    for i in range(n):
        adj[i] |= 1 << i
        
    color = [0] * n
    components = []
    
    for i in range(n):
        if color[i] == 0:
            comp = []
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in range(n):
                    if (adj[u] >> v) & 1 and color[v] == 0:
                        color[v] = 1
                        stack.append(v)
            components.append(comp)
            
    for comp in components:
        if len(comp) > 3:
            print("NO")
            return
            
    ans = [0] * n
    size1 = []
    size2 = []
    size3 = []
    
    for comp in components:
        sz = len(comp)
        if sz == 1:
            size1.append(comp)
        elif sz == 2:
            size2.append(comp)
        else:
            size3.append(comp)
            
    if len(size2) > len(size1):
        print("NO")
        return
        
    for i in range(len(size2)):
        c2 = size2[i]
        c1 = size1[i]
        ans[c2[0]] = 1
        ans[c2[1]] = 2
        ans[c1[0]] = 3
        
    rem1 = size1[len(size2):]
    if len(rem1) % 3 != 0:
        print("NO")
        return
        
    for i in range(0, len(rem1), 3):
        ans[rem1[i][0]] = 1
        ans[rem1[i+1][0]] = 2
        ans[rem1[i+2][0]] = 3
        
    for c3 in size3:
        ans[c3[0]] = 1
        ans[c3[1]] = 2
        ans[c3[2]] = 3
        
    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == '__main__':
    solve()
