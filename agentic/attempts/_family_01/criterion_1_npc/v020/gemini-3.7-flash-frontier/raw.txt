import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * n  # 0 means unassigned, 1, 2, 3
    # neighbor_color_count[u][c] counts how many neighbors of u have color c
    # c in 1..3
    neighbor_color_count = [[0] * 4 for _ in range(n)]
    sat_count = [0] * n  # number of distinct colors among neighbors
    deg = [len(adj[i]) for i in range(n)]
    
    assigned_count = 0
    
    def get_next_vertex():
        best_v = -1
        max_sat = -1
        max_deg = -1
        
        for i in range(n):
            if color[i] == 0:
                s = sat_count[i]
                if s > max_sat or (s == max_sat and deg[i] > max_deg):
                    max_sat = s
                    max_deg = deg[i]
                    best_v = i
        return best_v, max_sat

    def backtrack(assigned_cnt):
        if assigned_cnt == n:
            return True
        
        v, s = get_next_vertex()
        if s == 3:
            return False
        
        # Try available colors for v
        for c in (1, 2, 3):
            if neighbor_color_count[v][c] == 0:
                color[v] = c
                # Update neighbors
                changed_sat = []
                for u in adj[v]:
                    if neighbor_color_count[u][c] == 0:
                        sat_count[u] += 1
                        changed_sat.append(u)
                    neighbor_color_count[u][c] += 1
                
                if backtrack(assigned_cnt + 1):
                    return True
                
                # Undo
                for u in adj[v]:
                    neighbor_color_count[u][c] -= 1
                for u in changed_sat:
                    sat_count[u] -= 1
                color[v] = 0
                
        return False

    if backtrack(0):
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
