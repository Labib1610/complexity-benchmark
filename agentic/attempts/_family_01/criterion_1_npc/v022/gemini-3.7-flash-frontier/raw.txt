import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    # 3-coloring backtracking with MRV (DSatur-like)
    # Available colors mask for each vertex: bits 0, 1, 2 (value 1, 2, 4)
    # 7 means all 3 colors available.
    
    color = [-1] * n
    available = [7] * n  # mask of available colors (1: color 1, 2: color 2, 4: color 3)
    degree = [len(adj[i]) for i in range(n)]
    
    popcount = [0, 1, 1, 2, 1, 2, 2, 3]

    def backtrack(assigned_count):
        if assigned_count == n:
            return True
        
        # Select unassigned vertex with minimum available colors
        best_v = -1
        min_c = 4
        max_deg = -1
        
        for v in range(n):
            if color[v] == -1:
                c_count = popcount[available[v]]
                if c_count < min_c:
                    min_c = c_count
                    best_v = v
                    max_deg = degree[v]
                elif c_count == min_c:
                    if degree[v] > max_deg:
                        best_v = v
                        max_deg = degree[v]
        
        if min_c == 0:
            return False
        
        v = best_v
        mask = available[v]
        
        for c in (0, 1, 2):
            if (mask & (1 << c)):
                color[v] = c + 1
                
                # Update neighbors
                changed_neighbors = []
                c_mask = ~(1 << c)
                valid = True
                for u in adj[v]:
                    if color[u] == -1:
                        if available[u] & (1 << c):
                            available[u] &= c_mask
                            changed_neighbors.append(u)
                            if available[u] == 0:
                                valid = False
                
                if valid:
                    if backtrack(assigned_count + 1):
                        return True
                
                # Undo changes
                for u in changed_neighbors:
                    available[u] |= (1 << c)
                color[v] = -1
                
        return False

    if backtrack(0):
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
