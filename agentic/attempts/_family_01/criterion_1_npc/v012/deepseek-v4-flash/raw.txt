We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use backtracking with pruning, or a bitmask-based search.

Here’s the implementation:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n  # 0 = uncolored, 1,2,3 = buses

    # Precompute conflict masks for each node
    conflict_mask = [0] * n
    for i in range(n):
        mask = 0
        for j in adj[i]:
            mask |= (1 << j)
        conflict_mask[i] = mask

    # Backtracking with ordering: color most constrained first
    nodes = list(range(n))
    nodes.sort(key=lambda x: len(adj[x]), reverse=True)

    def backtrack(idx):
        if idx == n:
            return True
        node = nodes[idx]
        used = 0
        for c in range(1, 4):
            # check if any neighbor already has this color
            ok = True
            for nb in adj[node]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[node] = c
                if backtrack(idx + 1):
                    return True
                color[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(color[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
