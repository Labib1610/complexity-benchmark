import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)
            
    # Domain bitmask: 1 -> color 1, 2 -> color 2, 4 -> color 3
    # Full domain = 7 (bits 0, 1, 2 set)
    initial_domains = [7] * n
    
    # Precompute bit count for 0..7
    bit_count = [0, 1, 1, 2, 1, 2, 2, 3]
    degrees = [bin(adj[i]).count('1') for i in range(n)]
    
    def propagate(domains):
        # Forward check / Unit propagation
        # Find any variable with domain size 1 that hasn't propagated, or domain size 0
        changed = True
        while changed:
            changed = False
            for i in range(n):
                d = domains[i]
                if d == 0:
                    return False
                if bit_count[d] == 1:
                    # If this single color hasn't been removed from neighbors
                    mask = ~d
                    neighbors = adj[i]
                    temp = neighbors
                    while temp:
                        # lowest set bit
                        lsb = temp & -temp
                        v = lsb.bit_length() - 1
                        temp ^= lsb
                        if domains[v] & d:
                            domains[v] &= mask
                            if domains[v] == 0:
                                return False
                            changed = True
        return True

    def backtrack(domains):
        if not propagate(domains):
            return None
        
        # Check if all assigned
        # Pick unassigned variable with MRV (smallest domain size > 1), tie-break with highest degree
        best_var = -1
        best_size = 4
        best_deg = -1
        
        for i in range(n):
            sz = bit_count[domains[i]]
            if sz > 1:
                if sz < best_size or (sz == best_size and degrees[i] > best_deg):
                    best_size = sz
                    best_deg = degrees[i]
                    best_var = i
                    
        if best_var == -1:
            # All assigned
            return domains
        
        d = domains[best_var]
        for c in [1, 2, 4]:
            if d & c:
                new_domains = list(domains)
                new_domains[best_var] = c
                res = backtrack(new_domains)
                if res is not None:
                    return res
        return None

    res = backtrack(initial_domains)
    if res is None:
        print("NO")
    else:
        print("YES")
        color_map = {1: 1, 2: 2, 4: 3}
        ans = [color_map[res[i]] for i in range(n)]
        print(*(ans))

if __name__ == '__main__':
    solve()
