import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * n
    # forbidden[i] has bit c set if color c is forbidden for vertex i
    forbidden = [0] * n
    degrees = [len(adj[i]) for i in range(n)]
    
    # Pre-calculate bit counts for 0..7
    bit_count = [0, 1, 1, 2, 1, 2, 2, 3]

    def get_best_vertex():
        best_v = -1
        max_forb_count = -1
        max_deg = -1
        for i in range(n):
            if color[i] == -1:
                fc = bit_count[forbidden[i]]
                if fc > max_forb_count or (fc == max_forb_count and degrees[i] > max_deg):
                    max_forb_count = fc
                    max_deg = degrees[i]
                    best_v = i
        return best_v, max_forb_count

    def backtrack(colored_count):
        if colored_count == n:
            return True
        
        v, forb_count = get_best_vertex()
        if forb_count == 3:
            return False
        
        for c in range(3):
            if not (forbidden[v] & (1 << c)):
                color[v] = c
                changed = []
                for neighbor in adj[v]:
                    if color[neighbor] == -1 and not (forbidden[neighbor] & (1 << c)):
                        forbidden[neighbor] |= (1 << c)
                        changed.append(neighbor)
                
                if backtrack(colored_count + 1):
                    return True
                
                # Undo
                for neighbor in changed:
                    forbidden[neighbor] &= ~(1 << c)
                color[v] = -1

        return False

    if backtrack(0):
        print("YES")
        print(*(color[i] + 1 for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
