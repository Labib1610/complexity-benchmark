import sys

sys.setrecursionlimit(1000000)


def solve_component(vertices, global_adj, answer):
    vertices.sort(key=lambda x: global_adj[x].bit_count(), reverse=True)
    k = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}
    adj = [0] * k

    for i, v in enumerate(vertices):
        bits = global_adj[v]
        local = 0
        while bits:
            b = bits & -bits
            bits ^= b
            u = b.bit_length() - 1
            j = index.get(u)
            if j is not None:
                local |= 1 << j
        adj[i] = local

    full = (1 << k) - 1

    def bipartite(rem):
        side0 = 0
        side1 = 0
        unseen = rem

        while unseen:
            root = unseen & -unseen
            unseen ^= root
            side0 |= root
            stack = root

            while stack:
                vb = stack & -stack
                stack ^= vb
                v = vb.bit_length() - 1
                neighbors = adj[v] & rem

                if side0 & vb:
                    if neighbors & side0:
                        return None
                    new_vertices = neighbors & ~(side0 | side1)
                    side1 |= new_vertices
                else:
                    if neighbors & side1:
                        return None
                    new_vertices = neighbors & ~(side0 | side1)
                    side0 |= new_vertices

                unseen &= ~new_vertices
                stack |= new_vertices

        return side0, side1

    initial_bipartition = bipartite(full)
    if initial_bipartition is not None:
        a, b = initial_bipartition
        for i, v in enumerate(vertices):
            answer[v] = 1 if (a >> i) & 1 else 2
        return True

    # A quick greedy DSATUR attempt often finds a coloring immediately.
    colors = [-1] * k
    color_masks = [0, 0, 0]
    uncolored = full
    greedy_ok = True

    while uncolored:
        chosen = -1
        chosen_key = (-1, -1)

        bits = uncolored
        while bits:
            vb = bits & -bits
            bits ^= vb
            v = vb.bit_length() - 1
            used = 0
            if adj[v] & color_masks[0]:
                used |= 1
            if adj[v] & color_masks[1]:
                used |= 2
            if adj[v] & color_masks[2]:
                used |= 4
            key = (used.bit_count(), (adj[v] & uncolored).bit_count())
            if key > chosen_key:
                chosen_key = key
                chosen = v

        used = 0
        if adj[chosen] & color_masks[0]:
            used |= 1
        if adj[chosen] & color_masks[1]:
            used |= 2
        if adj[chosen] & color_masks[2]:
            used |= 4

        available = (~used) & 7
        if available == 0:
            greedy_ok = False
            break

        color_bit = available & -available
        color = color_bit.bit_length() - 1
        colors[chosen] = color
        color_masks[color] |= 1 << chosen
        uncolored &= ~(1 << chosen)

    if greedy_ok:
        for i, v in enumerate(vertices):
            answer[v] = colors[i] + 1
        return True

    # Find an odd cycle. Every valid first color class must intersect it.
    bfs_color = [-1] * k
    parent = [-1] * k
    odd_cycle = None

    for start in range(k):
        if bfs_color[start] != -1:
            continue
        bfs_color[start] = 0
        queue = [start]
        head = 0

        while head < len(queue) and odd_cycle is None:
            u = queue[head]
            head += 1
            bits = adj[u]
            while bits:
                vb = bits & -bits
                bits ^= vb
                v = vb.bit_length() - 1

                if bfs_color[v] == -1:
                    bfs_color[v] = bfs_color[u] ^ 1
                    parent[v] = u
                    queue.append(v)
                elif bfs_color[v] == bfs_color[u]:
                    path_u = []
                    x = u
                    while x != -1:
                        path_u.append(x)
                        x = parent[x]

                    ancestors = set(path_u)
                    path_v = []
                    x = v
                    while x not in ancestors:
                        path_v.append(x)
                        x = parent[x]
                    lca = x

                    cycle = []
                    for node in path_u:
                        cycle.append(node)
                        if node == lca:
                            break
                    cycle.extend(path_v)
                    odd_cycle = cycle
                    break

        if odd_cycle is not None:
            break

    comp_adj = [full ^ adj[i] ^ (1 << i) for i in range(k)]
    found = None

    def bron_kerbosch(r, p, x):
        nonlocal found

        if found is not None:
            return

        if p == 0:
            if x == 0:
                partition = bipartite(full ^ r)
                if partition is not None:
                    found = (r, partition[0], partition[1])
            return

        union = p | x
        pivot = -1
        best_count = -1
        scan = union

        while scan:
            ub = scan & -scan
            scan ^= ub
            u = ub.bit_length() - 1
            count = (p & comp_adj[u]).bit_count()
            if count > best_count:
                best_count = count
                pivot = u

        candidates = p if pivot == -1 else p & ~comp_adj[pivot]

        while candidates and found is None:
            vb = candidates & -candidates
            candidates ^= vb
            v = vb.bit_length() - 1

            bron_kerbosch(
                r | vb,
                p & comp_adj[v],
                x & comp_adj[v]
            )

            p ^= vb
            x |= vb

    earlier = 0
    for forced in odd_cycle:
        forced_bit = 1 << forced

        direct_partition = bipartite(full ^ forced_bit)
        if direct_partition is not None:
            found = (forced_bit, direct_partition[0], direct_partition[1])
            break

        compatible = comp_adj[forced]
        p = compatible & ~earlier
        x = compatible & earlier
        bron_kerbosch(forced_bit, p, x)

        if found is not None:
            break
        earlier |= forced_bit

    if found is None:
        return False

    first, second, third = found
    for i, v in enumerate(vertices):
        bit = 1 << i
        if first & bit:
            answer[v] = 1
        elif second & bit:
            answer[v] = 2
        else:
            answer[v] = 3

    return True


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n
    pos = 2

    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    answer = [0] * n
    unseen = (1 << n) - 1

    while unseen:
        root = unseen & -unseen
        unseen ^= root
        component_mask = root
        stack = root

        while stack:
            vb = stack & -stack
            stack ^= vb
            v = vb.bit_length() - 1
            new_vertices = adj[v] & unseen
            unseen &= ~new_vertices
            component_mask |= new_vertices
            stack |= new_vertices

        vertices = []
        bits = component_mask
        while bits:
            vb = bits & -bits
            bits ^= vb
            vertices.append(vb.bit_length() - 1)

        if not solve_component(vertices, adj, answer):
            print("NO")
            return

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
