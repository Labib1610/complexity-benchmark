import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Find connected components
    visited = [False] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in range(1, n + 1):
                    if (adj[u] >> v) & 1 and not visited[v]:
                        visited[v] = True
                        stack.append(v)
            components.append(comp)
    
    color = [0] * (n + 1)
    
    for comp in components:
        size = len(comp)
        # Map component vertices to 0..size-1
        idx_map = {v: i for i, v in enumerate(comp)}
        # Build local adjacency masks
        local_adj = [0] * size
        for v in comp:
            iv = idx_map[v]
            mask = 0
            for u in comp:
                if (adj[v] >> u) & 1:
                    mask |= 1 << idx_map[u]
            local_adj[iv] = mask
        
        # Try to 2-color the component (bipartite check)
        local_color = [-1] * size
        is_bipartite = True
        for start in range(size):
            if local_color[start] == -1:
                local_color[start] = 0
                stack = [start]
                while stack and is_bipartite:
                    u = stack.pop()
                    for v in range(size):
                        if (local_adj[u] >> v) & 1:
                            if local_color[v] == -1:
                                local_color[v] = local_color[u] ^ 1
                                stack.append(v)
                            elif local_color[v] == local_color[u]:
                                is_bipartite = False
                                break
        
        if is_bipartite:
            # Use colors 1 and 2
            for v in comp:
                color[v] = local_color[idx_map[v]] + 1
        else:
            # Need 3-coloring; try all assignments (size <= 40, but components are small in practice)
            # Use backtracking with bitmasks
            local_color = [0] * size
            def backtrack(pos):
                if pos == size:
                    return True
                used = 0
                for v in range(size):
                    if (local_adj[pos] >> v) & 1 and v < pos:
                        used |= 1 << local_color[v]
                for c in range(3):
                    if not (used & (1 << c)):
                        local_color[pos] = c
                        if backtrack(pos + 1):
                            return True
                return False
            
            if not backtrack(0):
                print("NO")
                return
            
            for v in comp:
                color[v] = local_color[idx_map[v]] + 1
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
