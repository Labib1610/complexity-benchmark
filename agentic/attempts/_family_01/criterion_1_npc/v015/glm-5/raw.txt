import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * n for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u][v] = True
        adj[v][u] = True
    
    # color[i] will be 1, 2, or 3. 0 means uncolored.
    color = [0] * n
    
    # We use a stack for DFS to avoid recursion depth issues
    # Stack elements: (node, current_color)
    
    for start_node in range(n):
        if color[start_node] != 0:
            continue
        
        # Start a new connected component
        stack = [(start_node, 1)]
        
        while stack:
            u, c = stack.pop()
            
            if color[u] == 0:
                color[u] = c
                # Push neighbors with a different color
                # We can cycle 1->2->3->1
                next_c = c % 3 + 1
                for v in range(n):
                    if adj[u][v] and color[v] == 0:
                        stack.append((v, next_c))
            elif color[u] != c:
                # Conflict detected: node u is already colored with a different color
                # than what the current path requires.
                print("NO")
                return
    
    # Verification step
    for i in range(n):
        for j in range(i + 1, n):
            if adj[i][j] and color[i] == color[j]:
                print("NO")
                return
    
    print("YES")
    print(*(c for c in color))

if __name__ == '__main__':
    solve()
