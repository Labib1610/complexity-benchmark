import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


def solve_component(vertices, graph):
    k = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}
    adj = [0] * k

    for i, v in enumerate(vertices):
        mask = graph[v]
        while mask:
            bit = mask & -mask
            u = bit.bit_length() - 1
            j = index.get(u)
            if j is not None:
                adj[i] |= 1 << j
            mask ^= bit

    # First check whether the component is bipartite.
    bip_color = [-1] * k
    bip_color[0] = 0
    q = deque([0])
    bipartite = True

    while q and bipartite:
        v = q.popleft()
        mask = adj[v]
        while mask:
            bit = mask & -mask
            u = bit.bit_length() - 1
            mask ^= bit
            if bip_color[u] == -1:
                bip_color[u] = bip_color[v] ^ 1
                q.append(u)
            elif bip_color[u] == bip_color[v]:
                bipartite = False
                break

    if bipartite:
        return bip_color

    colors = [-1] * k
    neighbor_count = [[0, 0, 0] for _ in range(k)]
    saturation = [0] * k
    color_masks = [0, 0, 0]
    uncolored = (1 << k) - 1
    failed = set()

    def assign(v, c):
        nonlocal uncolored
        colors[v] = c
        bit_v = 1 << v
        color_masks[c] |= bit_v
        uncolored ^= bit_v

        valid = True
        mask = adj[v] & uncolored
        while mask:
            bit = mask & -mask
            u = bit.bit_length() - 1
            mask ^= bit
            if neighbor_count[u][c] == 0:
                saturation[u] |= 1 << c
            neighbor_count[u][c] += 1
            if saturation[u] == 7:
                valid = False
        return valid

    def unassign(v, c):
        nonlocal uncolored
        mask = adj[v] & uncolored
        while mask:
            bit = mask & -mask
            u = bit.bit_length() - 1
            mask ^= bit
            neighbor_count[u][c] -= 1
            if neighbor_count[u][c] == 0:
                saturation[u] &= ~(1 << c)

        bit_v = 1 << v
        uncolored |= bit_v
        color_masks[c] ^= bit_v
        colors[v] = -1

    # Eliminate color-permutation symmetry.
    triangle = None
    best_score = -1
    for a in range(k):
        mask_b = adj[a] & ~((1 << (a + 1)) - 1)
        while mask_b:
            bit_b = mask_b & -mask_b
            b = bit_b.bit_length() - 1
            mask_b ^= bit_b
            common = adj[a] & adj[b]
            if common:
                bit_c = common & -common
                c = bit_c.bit_length() - 1
                score = adj[a].bit_count() + adj[b].bit_count() + adj[c].bit_count()
                if score > best_score:
                    best_score = score
                    triangle = (a, b, c)

    if triangle is not None:
        for v, c in zip(triangle, (0, 1, 2)):
            assign(v, c)
    else:
        best_edge = None
        best_score = -1
        for a in range(k):
            mask = adj[a]
            while mask:
                bit = mask & -mask
                b = bit.bit_length() - 1
                mask ^= bit
                if a < b:
                    score = adj[a].bit_count() + adj[b].bit_count()
                    if score > best_score:
                        best_score = score
                        best_edge = (a, b)
        if best_edge is not None:
            assign(best_edge[0], 0)
            assign(best_edge[1], 1)

    def dfs():
        if uncolored == 0:
            return True

        key = (color_masks[0], color_masks[1], color_masks[2])
        if key in failed:
            return False

        mask = uncolored
        chosen = -1
        best = (-1, -1, -1)

        while mask:
            bit = mask & -mask
            v = bit.bit_length() - 1
            mask ^= bit
            score = (
                saturation[v].bit_count(),
                (adj[v] & uncolored).bit_count(),
                adj[v].bit_count()
            )
            if score > best:
                best = score
                chosen = v

        available = (~saturation[chosen]) & 7
        options = []
        while available:
            bit = available & -available
            c = bit.bit_length() - 1
            available ^= bit

            impact = 0
            mask = adj[chosen] & uncolored
            while mask:
                nb = mask & -mask
                u = nb.bit_length() - 1
                mask ^= nb
                if neighbor_count[u][c] == 0:
                    impact += 1
            options.append((impact, c))

        options.sort()
        for _, c in options:
            valid = assign(chosen, c)
            if valid and dfs():
                return True
            unassign(chosen, c)

        if len(failed) < 500_000:
            failed.add(key)
        return False

    if not dfs():
        return None
    return colors


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    graph = [0] * n
    impossible = False
    p = 2

    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u == v:
            impossible = True
        else:
            graph[u] |= 1 << v
            graph[v] |= 1 << u

    if impossible:
        print("NO")
        return

    # Repeatedly remove vertices of degree at most two. Every coloring of the
    # remaining graph can be extended to these vertices in reverse order.
    active = (1 << n) - 1
    degree = [graph[v].bit_count() for v in range(n)]
    queue = deque(v for v in range(n) if degree[v] <= 2)
    removed = []
    is_removed = [False] * n

    while queue:
        v = queue.popleft()
        if is_removed[v] or not ((active >> v) & 1) or degree[v] > 2:
            continue

        is_removed[v] = True
        active ^= 1 << v
        removed.append(v)

        mask = graph[v] & active
        while mask:
            bit = mask & -mask
            u = bit.bit_length() - 1
            mask ^= bit
            degree[u] -= 1
            if degree[u] <= 2:
                queue.append(u)

    answer = [-1] * n

    # Solve each connected component of the remaining 3-core.
    unseen = active
    while unseen:
        start_bit = unseen & -unseen
        start = start_bit.bit_length() - 1
        component = []
        stack = [start]
        unseen ^= start_bit

        while stack:
            v = stack.pop()
            component.append(v)
            mask = graph[v] & unseen
            while mask:
                bit = mask & -mask
                u = bit.bit_length() - 1
                mask ^= bit
                unseen ^= bit
                stack.append(u)

        local_colors = solve_component(component, graph)
        if local_colors is None:
            print("NO")
            return

        for v, c in zip(component, local_colors):
            answer[v] = c

    # Add removed vertices back.
    for v in reversed(removed):
        forbidden = 0
        mask = graph[v]
        while mask:
            bit = mask & -mask
            u = bit.bit_length() - 1
            mask ^= bit
            if answer[u] != -1:
                forbidden |= 1 << answer[u]

        for c in range(3):
            if not (forbidden & (1 << c)):
                answer[v] = c
                break

        if answer[v] == -1:
            print("NO")
            return

    print("YES")
    print(*(c + 1 for c in answer))


if __name__ == "__main__":
    main()
