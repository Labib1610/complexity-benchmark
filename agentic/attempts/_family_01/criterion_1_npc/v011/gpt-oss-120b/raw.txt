```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    ALL_COLS = 0b111  # bits for colors 0,1,2
    domain = [ALL_COLS] * n
    assign = [-1] * n
    # order by degree for tie-breaking
    degree = [bin(adj[i]).count('1') for i in range(n)]

    # MRV selection
    def select_vertex():
        best = -1
        best_sz = 4
        best_deg = -1
        for v in range(n):
            if assign[v] == -1:
                sz = domain[v].bit_count()
                if sz < best_sz or (sz == best_sz and degree[v] > best_deg):
                    best_sz = sz
                    best_deg = degree[v]
                    best = v
        return best

    sys.setrecursionlimit(10000)
    def dfs():
        if all(a != -1 for a in assign):
            return True
        v = select_vertex()
        mask = domain[v]
        # try colors
        for c in range(3):
            if mask & (1 << c):
                # assign color c to v
                assign[v] = c
                changed = []  # list of (neighbor, previous_mask)
                ok = True
                nb = adj[v]
                while nb:
                    u = (nb & -nb).bit_length() - 1
                    nb &= nb - 1
                    if assign[u] == -1:
                        prev = domain[u]
                        new = prev & ~(1 << c)
                        if new != prev:
                            changed.append((u, prev))
                            domain[u] = new
                            if new == 0:
                                ok = False
                                break
                if ok and dfs():
                    return True
                # revert
                for u, prev in reversed(changed):
                    domain[u] = prev
                assign[v] = -1
        return False

    if dfs():
        print("YES")
        # convert to 1..3
        print(' '.join(str(c+1) for c in assign))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
