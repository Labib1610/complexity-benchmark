import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    color = [-1] * n
    # domain[i] is a bitmask of available colors (1, 2, 4 for colors 1, 2, 3)
    domain = [7] * n
    unassigned = (1 << n) - 1

    def backtrack():
        nonlocal unassigned
        if unassigned == 0:
            return True
        
        # Pick vertex with MRV (minimum remaining values)
        best_u = -1
        min_dom_size = 4
        max_deg = -1
        
        for u in range(n):
            if (unassigned & (1 << u)):
                d = domain[u]
                size = bin(d).count('1')
                if size == 0:
                    return False
                if size < min_dom_size:
                    min_dom_size = size
                    deg = bin(adj[u] & unassigned).count('1')
                    max_deg = deg
                    best_u = u
                elif size == min_dom_size:
                    deg = bin(adj[u] & unassigned).count('1')
                    if deg > max_deg:
                        max_deg = deg
                        best_u = u
        
        u = best_u
        d = domain[u]
        unassigned ^= (1 << u)
        
        for c in (0, 1, 2):
            if d & (1 << c):
                color[u] = c + 1
                # Save affected domains
                affected = []
                valid = True
                mask = 1 << c
                nbrs = adj[u] & unassigned
                
                # Check and update neighbors
                curr = nbrs
                while curr:
                    v = (curr & -curr).bit_length() - 1
                    curr &= curr - 1
                    if domain[v] & mask:
                        domain[v] ^= mask
                        affected.append(v)
                        if domain[v] == 0:
                            valid = False
                            # Don't break immediately so we can revert properly, or break and revert affected
                            break
                
                if valid:
                    if backtrack():
                        return True
                
                # Revert domain changes
                for v in affected:
                    domain[v] |= mask
                color[u] = -1
                
        unassigned |= (1 << u)
        return False

    if backtrack():
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
