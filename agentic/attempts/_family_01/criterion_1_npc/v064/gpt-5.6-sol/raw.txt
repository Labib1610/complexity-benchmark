import sys
from itertools import permutations

def solve_block(vertices, graph):
    k = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}
    adj = [0] * k

    for i, v in enumerate(vertices):
        mask = graph[v]
        for u in vertices:
            if mask >> u & 1:
                adj[i] |= 1 << index[u]

    all_mask = (1 << k) - 1

    def bipartite(mask):
        unseen = mask
        side0 = 0
        side1 = 0

        while unseen:
            start = unseen & -unseen
            unseen ^= start
            side0 |= start
            queue = start

            while queue:
                bit = queue & -queue
                queue ^= bit
                v = bit.bit_length() - 1
                neighbors = adj[v] & mask

                if side0 & bit:
                    if neighbors & side0:
                        return None
                    new_vertices = neighbors & unseen
                    unseen ^= new_vertices
                    side1 |= new_vertices
                else:
                    if neighbors & side1:
                        return None
                    new_vertices = neighbors & unseen
                    unseen ^= new_vertices
                    side0 |= new_vertices

                queue |= new_vertices

        return side0, side1

    initial = bipartite(all_mask)
    if initial is not None:
        side0, side1 = initial
        result = [1] * k
        for i in range(k):
            if side1 >> i & 1:
                result[i] = 2
        return result

    complement_adj = [
        all_mask & ~(adj[i] | (1 << i))
        for i in range(k)
    ]

    answer = None

    def bron_kerbosch(chosen, candidates, excluded):
        nonlocal answer
        if answer is not None:
            return

        if candidates == 0:
            if excluded == 0:
                remainder = all_mask ^ chosen
                partition = bipartite(remainder)
                if partition is not None:
                    side0, side1 = partition
                    colors = [1] * k
                    for i in range(k):
                        bit = 1 << i
                        if side0 & bit:
                            colors[i] = 2
                        elif side1 & bit:
                            colors[i] = 3
                    answer = colors
            return

        possible_pivots = candidates | excluded
        pivot = -1
        best_score = -1
        temp = possible_pivots

        while temp:
            bit = temp & -temp
            temp ^= bit
            u = bit.bit_length() - 1
            score = (candidates & complement_adj[u]).bit_count()
            if score > best_score:
                best_score = score
                pivot = u
                if score == candidates.bit_count():
                    break

        branch_vertices = candidates
        if pivot != -1:
            branch_vertices &= ~complement_adj[pivot]

        while branch_vertices and answer is None:
            bit = branch_vertices & -branch_vertices
            branch_vertices ^= bit
            v = bit.bit_length() - 1

            bron_kerbosch(
                chosen | bit,
                candidates & complement_adj[v],
                excluded & complement_adj[v]
            )

            candidates ^= bit
            excluded |= bit

    bron_kerbosch(0, all_mask, 0)
    return answer


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    graph = [0] * n
    invalid = False
    pos = 2

    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u == v:
            invalid = True
        else:
            graph[u] |= 1 << v
            graph[v] |= 1 << u

    if invalid:
        print("NO")
        return

    # Split the graph into vertex-biconnected blocks.
    discovery = [-1] * n
    low = [0] * n
    edge_stack = []
    blocks = []
    timer = 0

    sys.setrecursionlimit(10000)

    def dfs(u, parent):
        nonlocal timer
        discovery[u] = low[u] = timer
        timer += 1

        neighbors = graph[u]
        while neighbors:
            bit = neighbors & -neighbors
            neighbors ^= bit
            v = bit.bit_length() - 1

            if discovery[v] == -1:
                edge_stack.append((u, v))
                dfs(v, u)
                low[u] = min(low[u], low[v])

                if low[v] >= discovery[u]:
                    block_vertices = set()
                    while True:
                        a, b = edge_stack.pop()
                        block_vertices.add(a)
                        block_vertices.add(b)
                        if a == u and b == v:
                            break
                    blocks.append(list(block_vertices))

            elif v != parent and discovery[v] < discovery[u]:
                edge_stack.append((u, v))
                low[u] = min(low[u], discovery[v])

    for start in range(n):
        if discovery[start] == -1:
            dfs(start, -1)

    block_colorings = []
    for block in blocks:
        coloring = solve_block(block, graph)
        if coloring is None:
            print("NO")
            return
        block_colorings.append(coloring)

    vertex_blocks = [[] for _ in range(n)]
    for block_id, block in enumerate(blocks):
        for v in block:
            vertex_blocks[v].append(block_id)

    colors = [0] * n
    processed = [False] * len(blocks)
    color_permutations = tuple(permutations((1, 2, 3)))

    for root in range(len(blocks)):
        if processed[root]:
            continue

        queue = [root]
        head = 0

        while head < len(queue):
            block_id = queue[head]
            head += 1
            if processed[block_id]:
                continue

            block = blocks[block_id]
            local = block_colorings[block_id]

            chosen_permutation = None
            for perm in color_permutations:
                valid = True
                for i, v in enumerate(block):
                    if colors[v] and colors[v] != perm[local[i] - 1]:
                        valid = False
                        break
                if valid:
                    chosen_permutation = perm
                    break

            if chosen_permutation is None:
                print("NO")
                return

            processed[block_id] = True
            for i, v in enumerate(block):
                colors[v] = chosen_permutation[local[i] - 1]

            for v in block:
                for next_block in vertex_blocks[v]:
                    if not processed[next_block]:
                        queue.append(next_block)

    for i in range(n):
        if colors[i] == 0:
            colors[i] = 1

    print("YES")
    print(*colors)


if __name__ == "__main__":
    main()
