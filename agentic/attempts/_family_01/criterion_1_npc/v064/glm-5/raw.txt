import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                if not dfs(v, 3 - c):
                    return False
            elif color[v] == c:
                return False
        return True
    
    for i in range(n):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
    
    # Check if any component is bipartite (has a valid 2-coloring)
    # If so, we can color that component with 2 colors and use 3rd color for others
    # If all components are complete graphs (odd cycles), we need to check if we have enough colors
    
    # Group vertices by component
    comp_id = [-1] * n
    comps = []
    curr_id = 0
    for i in range(n):
        if comp_id[i] == -1:
            comp = []
            stack = [i]
            comp_id[i] = curr_id
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if comp_id[v] == -1:
                        comp_id[v] = curr_id
                        stack.append(v)
            comps.append(comp)
            curr_id += 1
    
    # Check if any component is bipartite (not a complete graph)
    has_bipartite = False
    for comp in comps:
        if len(comp) >= 3:
            # Check if it's a complete graph
            is_complete = True
            for u in comp:
                if len(adj[u]) != len(comp) - 1:
                    is_complete = False
                    break
            if not is_complete:
                has_bipartite = True
                break
    
    if has_bipartite:
        # We can color the bipartite component with 2 colors
        # and use 3rd color for isolated vertices
        print("YES")
        print(' '.join(map(str, color)))
        return
    
    # All components are either isolated vertices or complete graphs
    # Count complete graphs of each size
    k1 = 0  # isolated vertices
    k2 = 0  # edges (K2)
    k3 = 0  # triangles (K3)
    
    for comp in comps:
        sz = len(comp)
        if sz == 1:
            k1 += 1
        elif sz == 2:
            k2 += 1
        elif sz == 3:
            k3 += 1
        else:
            # Complete graph of size >= 4, impossible with 3 colors
            print("NO")
            return
    
    # Check if we can pack K2s and K3s into 3 colors
    # K3 needs one of each color
    # K2 needs two different colors
    
    # We need: k2 <= 3*k3 + k1
    if k2 > 3 * k3 + k1:
        print("NO")
        return
    
    print("YES")
    
    # Construct the coloring
    result = [0] * n
    
    # Color K3s first
    # Each K3 gets colors 1, 2, 3
    # We track how many K2s are assigned to each color pair
    c12 = 0  # K2s using colors 1,2
    c13 = 0  # K2s using colors 1,3
    c23 = 0  # K2s using colors 2,3
    
    # For each K3, we can assign up to one K2 to each color pair
    # After K3s, remaining K2s use isolated vertices
    
    k2_assigned = 0
    k3_idx = 0
    
    for comp in comps:
        if len(comp) == 3:
            u, v, w = comp
            result[u] = 1
            result[v] = 2
            result[w] = 3
            
            # Assign up to 3 K2s to this K3
            for _ in range(3):
                if k2_assigned >= k2:
                    break
                # Find an unassigned K2
                for comp2 in comps:
                    if len(comp2) == 2:
                        a, b = comp2
                        if result[a] == 0:
                            if c12 < k3_idx + 1:
                                result[a] = 1
                                result[b] = 2
                                c12 += 1
                            elif c13 < k3_idx + 1:
                                result[a] = 1
                                result[b] = 3
                                c13 += 1
                            else:
                                result[a] = 2
                                result[b] = 3
                                c23 += 1
                            k2_assigned += 1
                            break
            k3_idx += 1
    
    # Assign remaining K2s using isolated vertices
    isolated = [u for comp in comps if len(comp) == 1 for u in comp]
    iso_idx = 0
    
    for comp in comps:
        if len(comp) == 2:
            a, b = comp
            if result[a] == 0:
                # Use an isolated vertex to provide the 3rd color
                # We need to pick two colors for this K2
                # Use colors 1 and 2, and the isolated vertex gets color 3
                result[a] = 1
                result[b] = 2
                if iso_idx < len(isolated):
                    result[isolated[iso_idx]] = 3
                    iso_idx += 1
    
    # Assign remaining isolated vertices
    for u in isolated:
        if result[u] == 0:
            result[u] = 1
    
    print(' '.join(map(str, result)))

solve()
