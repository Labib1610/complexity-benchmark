import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)
            
    # Check for self loops
    for i in range(n):
        if (adj[i] >> i) & 1:
            print("NO")
            return

    # Find connected components
    visited = [False] * n
    components = []
    for i in range(n):
        if not visited[i]:
            comp = []
            queue = [i]
            visited[i] = True
            for node in queue:
                comp.append(node)
                for neighbor in range(n):
                    if (adj[node] & (1 << neighbor)) and not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
            components.append(comp)
            
    final_color = [0] * n

    for comp in components:
        # Solve for this component
        comp_mask = 0
        for u in comp:
            comp_mask |= (1 << u)
            
        avail = {u: 7 for u in comp} # bitmask: 1 (color 1), 2 (color 2), 4 (color 3)
        assignment = {}
        
        def backtrack():
            if len(assignment) == len(comp):
                return True
            
            # Find vertex with minimum available colors
            min_c = 4
            best_u = -1
            best_deg = -1
            
            for u in comp:
                if u not in assignment:
                    c_count = bin(avail[u]).count('1')
                    if c_count == 0:
                        return False
                    if c_count < min_c:
                        min_c = c_count
                        best_u = u
                        best_deg = bin(adj[u] & comp_mask).count('1')
                    elif c_count == min_c:
                        deg = bin(adj[u] & comp_mask).count('1')
                        if deg > best_deg:
                            best_u = u
                            best_deg = deg
            
            # Try colors for best_u
            mask = avail[best_u]
            for color_bit, color_val in [(1, 1), (2, 2), (4, 3)]:
                if mask & color_bit:
                    assignment[best_u] = color_val
                    # update avail for neighbors
                    changed = []
                    valid = True
                    for v in comp:
                        if v not in assignment and ((adj[best_u] >> v) & 1):
                            if avail[v] & color_bit:
                                avail[v] ^= color_bit
                                changed.append(v)
                                if avail[v] == 0:
                                    valid = False
                    
                    if valid and backtrack():
                        return True
                    
                    # undo
                    for v in changed:
                        avail[v] |= color_bit
                    del assignment[best_u]
                    
            return False

        if not backtrack():
            print("NO")
            return
        
        for u in comp:
            final_color[u] = assignment[u]

    print("YES")
    print(*(final_color[i] for i in range(n)))

if __name__ == '__main__':
    solve()
