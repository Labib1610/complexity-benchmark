import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(set)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Try all possible 3-colorings using backtracking
    color = [0] * (n + 1)  # color[i] represents the color of person i
    
    def is_valid(person, c):
        for neighbor in adj[person]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(person):
        if person > n:
            return True
            
        for c in range(1, 4):
            if is_valid(person, c):
                color[person] = c
                if backtrack(person + 1):
                    return True
                color[person] = 0
        
        return False
    
    # Optimization: Start with node that has highest degree
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, color[1:])))
    else:
        print("NO")

solve()
