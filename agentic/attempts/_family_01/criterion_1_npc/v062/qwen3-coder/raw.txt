def solve():
    import sys
    from itertools import combinations
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-indexed
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Build adjacency list
    adj = [set() for _ in range(n)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    
    # For each vertex, find its degree
    deg = [len(adj[i]) for i in range(n)]
    
    # Check if there's a vertex with degree > 3
    # If so, we might need to do backtracking; but given n<=40,
    # let's try a greedy approach with backtracking for small cases.
    
    # Actually, since n <= 40 and we're looking for 3-coloring,
    # which is NP-hard, but feasible for small n.
    # Let's implement backtracking with pruning.
    
    color = [-1] * n  # -1 means uncolored
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
        
        # Pruning: if current node has degree > 3, it's harder to color
        # But let's just proceed normally
        
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = -1
        return False
    
    # Optimization: start coloring with nodes of higher degree
    order = list(range(n))
    order.sort(key=lambda x: deg[x], reverse=True)
    
    # Re-index the color array according to new order
    # Actually, let's keep it simple and stick with original indices
    
    if backtrack(0):
        print("YES")
        # Convert back to 1-indexed
        result = [str(color[i]) for i in range(n)]
        print(" ".join(result))
    else:
        print("NO")

solve()
