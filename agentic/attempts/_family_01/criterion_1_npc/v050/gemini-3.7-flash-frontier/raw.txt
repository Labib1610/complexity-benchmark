import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    color = [0] * n
    mask = [7] * n  # bitmask of available colors: 1 (1st bit), 2 (2nd bit), 4 (3rd bit)

    # Find connected components
    visited = [False] * n
    components = []
    for i in range(n):
        if not visited[i]:
            comp = []
            queue = [i]
            visited[i] = True
            for u in queue:
                comp.append(u)
                for v in range(n):
                    if (adj[u] & (1 << v)) and not visited[v]:
                        visited[v] = True
                        queue.append(v)
            components.append(comp)

    for comp in components:

        def backtrack():
            best_v = -1
            min_c = 4
            all_colored = True

            for u in comp:
                if color[u] == 0:
                    all_colored = False
                    c_count = bin(mask[u]).count("1")
                    if c_count < min_c:
                        min_c = c_count
                        best_v = u
                        if min_c <= 1:
                            break

            if all_colored:
                return True
            if min_c == 0:
                return False

            u = best_v
            avail = mask[u]

            for c_idx, c_val in ((1, 1), (2, 2), (3, 4)):
                if avail & c_val:
                    color[u] = c_idx
                    changed = []
                    valid = True
                    for v in comp:
                        if (adj[u] & (1 << v)) and color[v] == 0:
                            if mask[v] & c_val:
                                mask[v] ^= c_val
                                changed.append(v)
                                if mask[v] == 0:
                                    valid = False

                    if valid and backtrack():
                        return True

                    for v in changed:
                        mask[v] |= c_val
                    color[u] = 0

            return False

        if not backtrack():
            print("NO")
            return

    print("YES")
    print(*(color[i] for i in range(n)))


if __name__ == "__main__":
    solve()
