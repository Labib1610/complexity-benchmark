import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n
    pos = 2

    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u == v:
            print("NO")
            return
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_vertices = (1 << n) - 1
    answer = [0] * n

    def bipartition(mask):
        unseen = mask
        left = 0
        right = 0

        while unseen:
            root = unseen & -unseen
            unseen ^= root
            left |= root
            queue = root

            while queue:
                bit = queue & -queue
                queue ^= bit
                v = bit.bit_length() - 1
                neighbors = adj[v] & mask

                if left & bit:
                    if neighbors & left:
                        return None
                    new_vertices = neighbors & unseen
                    unseen ^= new_vertices
                    right |= new_vertices
                    queue |= new_vertices
                else:
                    if neighbors & right:
                        return None
                    new_vertices = neighbors & unseen
                    unseen ^= new_vertices
                    left |= new_vertices
                    queue |= new_vertices

        return left, right

    def solve_component(universe):
        direct = bipartition(universe)
        if direct is not None:
            left, right = direct
            return 0, left, right

        complement = [0] * n
        bits = universe
        while bits:
            bit = bits & -bits
            bits ^= bit
            v = bit.bit_length() - 1
            complement[v] = universe & ~(adj[v] | bit)

        result = None

        def bron_kerbosch(independent, candidates, excluded):
            nonlocal result

            if candidates == 0:
                if excluded == 0:
                    remaining = universe ^ independent
                    partition = bipartition(remaining)
                    if partition is not None:
                        result = (independent, partition[0], partition[1])
                        return True
                return False

            pivot_set = candidates | excluded
            pivot = -1
            best_count = -1
            temp = pivot_set
            while temp:
                bit = temp & -temp
                temp ^= bit
                v = bit.bit_length() - 1
                count = (candidates & complement[v]).bit_count()
                if count > best_count:
                    best_count = count
                    pivot = v

            branch_vertices = candidates
            if pivot != -1:
                branch_vertices &= ~complement[pivot]

            while branch_vertices:
                bit = branch_vertices & -branch_vertices
                branch_vertices ^= bit
                v = bit.bit_length() - 1

                if bron_kerbosch(
                    independent | bit,
                    candidates & complement[v],
                    excluded & complement[v]
                ):
                    return True

                candidates ^= bit
                excluded |= bit

            return False

        bron_kerbosch(0, universe, 0)
        return result

    remaining = all_vertices

    while remaining:
        start = remaining & -remaining
        remaining ^= start
        component = start
        queue = start

        while queue:
            bit = queue & -queue
            queue ^= bit
            v = bit.bit_length() - 1
            new_vertices = adj[v] & remaining
            remaining ^= new_vertices
            component |= new_vertices
            queue |= new_vertices

        solution = solve_component(component)
        if solution is None:
            print("NO")
            return

        first, second, third = solution

        bits = first
        while bits:
            bit = bits & -bits
            bits ^= bit
            answer[bit.bit_length() - 1] = 1

        bits = second
        while bits:
            bit = bits & -bits
            bits ^= bit
            answer[bit.bit_length() - 1] = 2

        bits = third
        while bits:
            bit = bits & -bits
            bits ^= bit
            answer[bit.bit_length() - 1] = 3

    print("YES")
    print(*answer)

if __name__ == "__main__":
    main()
