import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                if not dfs(v, 3 - c):
                    return False
            elif color[v] == c:
                return False
        return True
    
    components = []
    for i in range(n):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
            comp_nodes = [j for j in range(n) if color[j] in (1, 2) and 
                         (color[j] == 1 and j >= i or color[j] == 2 and j >= i)]
            comp_nodes = []
            visited_in_comp = set()
            stack = [i]
            visited_in_comp.add(i)
            while stack:
                u = stack.pop()
                comp_nodes.append(u)
                for v in adj[u]:
                    if v not in visited_in_comp:
                        visited_in_comp.add(v)
                        stack.append(v)
            ones = sum(1 for node in comp_nodes if color[node] == 1)
            twos = len(comp_nodes) - ones
            components.append((ones, twos))
    
    k = len(components)
    INF = 10**9
    dp = [[INF] * (n + 1) for _ in range(k + 1)]
    dp[0][0] = 0
    
    for i in range(1, k + 1):
        a, b = components[i-1]
        for j in range(n + 1):
            if dp[i-1][j] != INF:
                if j + a <= n:
                    dp[i][j + a] = min(dp[i][j + a], dp[i-1][j] + b)
                if j + b <= n:
                    dp[i][j + b] = min(dp[i][j + b], dp[i-1][j] + a)
    
    found = False
    for s1 in range(n + 1):
        s2 = dp[k][s1]
        if s2 != INF:
            s3 = n - s1 - s2
            if s3 >= 0:
                found = True
                break
    
    if not found:
        print("NO")
        return
    
    print("YES")
    
    target_s1 = s1
    s1 = target_s1
    s2 = dp[k][s1]
    s3 = n - s1 - s2
    
    zone = [0] * n
    color = [0] * n
    
    def dfs2(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                dfs2(v, 3 - c)
    
    for i in range(n):
        if color[i] == 0:
            dfs2(i, 1)
    
    assignments = []
    for i in range(n):
        if color[i] == 0:
            comp_nodes = []
            stack = [i]
            color[i] = 1
            comp_nodes.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp_nodes.append(v)
                        stack.append(v)
            ones = [node for node in comp_nodes if color[node] == 1]
            twos = [node for node in comp_nodes if color[node] == 2]
            assignments.append((ones, twos))
    
    dp = [[INF] * (n + 1) for _ in range(k + 1)]
    choice = [[0] * (n + 1) for _ in range(k + 1)]
    dp[0][0] = 0
    
    for i in range(1, k + 1):
        ones, twos = assignments[i-1]
        a, b = len(ones), len(twos)
        for j in range(n + 1):
            if dp[i-1][j] != INF:
                if j + a <= n:
                    if dp[i][j + a] > dp[i-1][j] + b:
                        dp[i][j + a] = dp[i-1][j] + b
                        choice[i][j + a] = 0
                if j + b <= n:
                    if dp[i][j + b] > dp[i-1][j] + a:
                        dp[i][j + b] = dp[i-1][j] + a
                        choice[i][j + a] = 1
    
    s1 = target_s1
    zone_choices = [0] * k
    for i in range(k, 0, -1):
        if choice[i][s1] == 0:
            zone_choices[i-1] = 0
            s1 -= len(assignments[i-1][0])
        else:
            zone_choices[i-1] = 1
            s1 -= len(assignments[i-1][1])
    
    for i in range(k):
        ones, twos = assignments[i]
        if zone_choices[i] == 0:
            for node in ones:
                zone[node] = 1
            for node in twos:
                zone[node] = 2
        else:
            for node in ones:
                zone[node] = 2
            for node in twos:
                zone[node] = 1
    
    for i in range(n):
        if zone[i] == 0:
            zone[i] = 3
    
    print(' '.join(map(str, zone)))

solve()
