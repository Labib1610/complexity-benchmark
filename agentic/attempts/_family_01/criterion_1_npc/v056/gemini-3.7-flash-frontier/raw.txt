import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)
        else:
            # Self loop cannot be 3-colored properly with valid distinct colors
            print("NO")
            return

    # Check for components and solve each component
    visited = [False] * n
    ans = [0] * n

    for i in range(n):
        if visited[i]:
            continue
        
        # Collect component
        component = []
        q = [i]
        visited[i] = True
        for node in q:
            component.append(node)
            mask = adj[node]
            for neighbor in range(n):
                if (mask & (1 << neighbor)) and not visited[neighbor]:
                    visited[neighbor] = True
                    q.append(neighbor)
        
        # Color the component
        # Domain bitmask: 1 -> color 1, 2 -> color 2, 4 -> color 3
        domain = {node: 7 for node in component}
        comp_nodes = set(component)
        
        def backtrack(unassigned):
            if not unassigned:
                return True
            
            # MRV: Choose node with minimum domain size, tie-break by degree
            best_node = -1
            best_size = 4
            best_deg = -1
            
            for node in unassigned:
                d = domain[node]
                sz = (d & 1) + ((d >> 1) & 1) + ((d >> 2) & 1)
                if sz < best_size:
                    best_size = sz
                    best_node = node
                    best_deg = (adj[node] & comp_nodes_mask).bit_count()
                    if sz <= 1:
                        break
                elif sz == best_size:
                    deg = (adj[node] & comp_nodes_mask).bit_count()
                    if deg > best_deg:
                        best_node = node
                        best_deg = deg
            
            if best_size == 0:
                return False
            
            node = best_node
            unassigned.remove(node)
            d = domain[node]
            
            # Try available colors
            for c_idx, c_mask in ((1, 1), (2, 2), (3, 4)):
                if d & c_mask:
                    ans[node] = c_idx
                    # Forward check
                    changed = []
                    valid = True
                    mask = adj[node]
                    for neighbor in unassigned:
                        if (mask & (1 << neighbor)) and (domain[neighbor] & c_mask):
                            domain[neighbor] &= ~c_mask
                            changed.append(neighbor)
                            if domain[neighbor] == 0:
                                valid = False
                    
                    if valid and backtrack(unassigned):
                        return True
                    
                    # Undo forward check
                    for neighbor in changed:
                        domain[neighbor] |= c_mask
            
            unassigned.add(node)
            return False

        comp_nodes_mask = 0
        for node in component:
            comp_nodes_mask |= (1 << node)

        unassigned = set(component)
        if not backtrack(unassigned):
            print("NO")
            return

    print("YES")
    print(*(ans[i] for i in range(n)))

if __name__ == '__main__':
    solve()
