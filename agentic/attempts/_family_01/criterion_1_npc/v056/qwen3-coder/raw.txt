def solve():
    import sys
    import threading

    def main():
        import sys
        sys.setrecursionlimit(1000000)
        input = sys.stdin.read
        data = input().split()
        n, m = int(data[0]), int(data[1])
        edges = []
        index = 2
        for _ in range(m):
            u = int(data[index]) - 1
            v = int(data[index+1]) - 1
            edges.append((u, v))
            index += 2

        if m == 0:
            print("YES")
            print(" ".join(["1"] * n))
            return

        # Build adjacency list
        adj = [[] for _ in range(n)]
        for u, v in edges:
            adj[u].append(v)
            adj[v].append(u)

        # Color the graph with 3 colors using backtracking
        color = [-1] * n

        def is_safe(node, c):
            for neighbor in adj[node]:
                if color[neighbor] == c:
                    return False
            return True

        def graph_coloring(node):
            if node == n:
                return True
            for c in range(3):
                if is_safe(node, c):
                    color[node] = c
                    if graph_coloring(node + 1):
                        return True
                    color[node] = -1
            return False

        if graph_coloring(0):
            print("YES")
            result = [str(c + 1) for c in color]
            print(" ".join(result))
        else:
            print("NO")

    threading.Thread(target=main).start()

solve()
