import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    # avail: bitmask of available colors for each vertex (1: color 0, 2: color 1, 4: color 2)
    # colors: 0, 1, 2 -> 1, 2, 3 in 1-based output
    
    assignment = [-1] * n
    avail = [7] * n  # bits 0, 1, 2 set
    
    # Precompute popcounts for 0..7
    popcnt = [0, 1, 1, 2, 1, 2, 2, 3]
    
    def backtrack():
        # Find unassigned vertex with minimum available colors
        min_c = 4
        best_u = -1
        
        for u in range(n):
            if assignment[u] == -1:
                c = popcnt[avail[u]]
                if c == 0:
                    return False
                if c < min_c:
                    min_c = c
                    best_u = u
                    if c == 1:
                        break  # Can't get better than 1
                        
        if best_u == -1:
            return True
            
        u = best_u
        mask = avail[u]
        
        # Try available colors
        for color_idx in range(3):
            if (mask & (1 << color_idx)):
                assignment[u] = color_idx
                
                # Update neighbors
                changed_neighbors = []
                valid = True
                u_adj = adj[u]
                
                # Check neighbors
                for v in range(n):
                    if (u_adj & (1 << v)) and assignment[v] == -1:
                        if avail[v] & (1 << color_idx):
                            avail[v] ^= (1 << color_idx)
                            changed_neighbors.append(v)
                            if avail[v] == 0:
                                valid = False
                                
                if valid and backtrack():
                    return True
                    
                # Restore
                for v in changed_neighbors:
                    avail[v] |= (1 << color_idx)
                assignment[u] = -1
                
        return False

    if backtrack():
        print("YES")
        print(*(assignment[i] + 1 for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
