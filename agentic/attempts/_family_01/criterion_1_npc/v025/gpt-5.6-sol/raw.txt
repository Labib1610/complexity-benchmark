import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    adj = [0] * n
    invalid = m > 800

    p = 2
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u == v:
            invalid = True
        else:
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    if invalid:
        print("NO")
        return

    colors = [-1] * n
    unseen = (1 << n) - 1
    components = []

    while unseen:
        start_bit = unseen & -unseen
        unseen ^= start_bit
        stack = start_bit
        component = 0

        while stack:
            bit = stack & -stack
            stack ^= bit
            v = bit.bit_length() - 1
            component |= bit
            nxt = adj[v] & unseen
            unseen ^= nxt
            stack |= nxt

        components.append(component)

    def bipartite_color(comp):
        temp = {}
        remaining = comp

        while remaining:
            bit = remaining & -remaining
            root = bit.bit_length() - 1
            remaining ^= bit
            temp[root] = 0
            stack = [root]

            while stack:
                v = stack.pop()
                neighbors = adj[v] & comp
                while neighbors:
                    b = neighbors & -neighbors
                    neighbors ^= b
                    u = b.bit_length() - 1
                    expected = temp[v] ^ 1
                    if u in temp:
                        if temp[u] != expected:
                            return False
                    else:
                        temp[u] = expected
                        remaining &= ~b
                        stack.append(u)

        for v, c in temp.items():
            colors[v] = c
        return True

    def solve_component(comp):
        if bipartite_color(comp):
            return True

        vertices = []
        bits = comp
        while bits:
            b = bits & -bits
            bits ^= b
            vertices.append(b.bit_length() - 1)

        for v in vertices:
            colors[v] = -1

        degree = [0] * n
        for v in vertices:
            degree[v] = (adj[v] & comp).bit_count()

        triangle = None
        best_score = -1
        for u in vertices:
            neighbors = adj[u] & comp
            while neighbors:
                bv = neighbors & -neighbors
                neighbors ^= bv
                v = bv.bit_length() - 1
                if v <= u:
                    continue
                common = adj[u] & adj[v] & comp
                while common:
                    bw = common & -common
                    common ^= bw
                    w = bw.bit_length() - 1
                    if w <= v:
                        continue
                    score = degree[u] + degree[v] + degree[w]
                    if score > best_score:
                        best_score = score
                        triangle = (u, v, w)

        if triangle is not None:
            anchors = [(triangle[0], 0), (triangle[1], 1), (triangle[2], 2)]
            max_used = 2
        else:
            best_edge = None
            best_score = -1
            for u in vertices:
                neighbors = adj[u] & comp
                while neighbors:
                    b = neighbors & -neighbors
                    neighbors ^= b
                    v = b.bit_length() - 1
                    if v > u:
                        score = degree[u] + degree[v]
                        if score > best_score:
                            best_score = score
                            best_edge = (u, v)
            if best_edge is None:
                colors[vertices[0]] = 0
                return True
            anchors = [(best_edge[0], 0), (best_edge[1], 1)]
            max_used = 1

        counts = [[0, 0, 0] for _ in range(n)]
        saturation = [0] * n
        uncolored = comp

        for v, c in anchors:
            bit = 1 << v
            uncolored &= ~bit
            colors[v] = c
            neighbors = adj[v] & uncolored
            while neighbors:
                b = neighbors & -neighbors
                neighbors ^= b
                u = b.bit_length() - 1
                if counts[u][c] == 0:
                    saturation[u] |= 1 << c
                counts[u][c] += 1

        sys.setrecursionlimit(10000)

        def dfs(uncolored_mask, used_max):
            if not uncolored_mask:
                return True

            selected = -1
            selected_key = (-1, -1, -1)
            bits = uncolored_mask

            while bits:
                b = bits & -bits
                bits ^= b
                v = b.bit_length() - 1
                sat = saturation[v].bit_count()
                if sat == 3:
                    return False
                key = (
                    sat,
                    (adj[v] & uncolored_mask).bit_count(),
                    degree[v]
                )
                if key > selected_key:
                    selected_key = key
                    selected = v

            v = selected
            vbit = 1 << v
            next_uncolored = uncolored_mask & ~vbit
            allowed = (~saturation[v]) & 7
            allowed &= (1 << min(3, used_max + 2)) - 1

            options = []
            for c in range(3):
                if allowed & (1 << c):
                    impact = 0
                    neighbors = adj[v] & next_uncolored
                    while neighbors:
                        b = neighbors & -neighbors
                        neighbors ^= b
                        u = b.bit_length() - 1
                        if counts[u][c] == 0:
                            impact += 1
                    options.append((-impact, c))
            options.sort()

            for _, c in options:
                colors[v] = c
                changed = []
                neighbors = adj[v] & next_uncolored

                while neighbors:
                    b = neighbors & -neighbors
                    neighbors ^= b
                    u = b.bit_length() - 1
                    was_zero = counts[u][c] == 0
                    counts[u][c] += 1
                    if was_zero:
                        saturation[u] |= 1 << c
                    changed.append((u, was_zero))

                if dfs(next_uncolored, max(used_max, c)):
                    return True

                for u, was_zero in changed:
                    counts[u][c] -= 1
                    if was_zero:
                        saturation[u] &= ~(1 << c)
                colors[v] = -1

            return False

        return dfs(uncolored, max_used)

    for comp in components:
        if not solve_component(comp):
            print("NO")
            return

    print("YES")
    print(*[c + 1 for c in colors])

if __name__ == "__main__":
    main()
