import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            
    for i in range(n):
        adj[i] = list(set(adj[i]))

    # If any self-loop was present (though problem says u v conflict pair), but if u == v, impossible.
    # Standard conflict pairs have u != v.
    
    color = [-1] * n
    # avail mask: bit 0: color 0, bit 1: color 1, bit 2: color 2
    avail = [7] * n
    
    # Precompute popcount for 0..7
    popcnt = [bin(i).count('1') for i in range(8)]
    
    def backtrack(unassigned_count):
        if unassigned_count == 0:
            return True
        
        # MRV: find unassigned vertex with smallest number of available colors
        best_v = -1
        min_c = 4
        max_deg = -1
        
        for v in range(n):
            if color[v] == -1:
                c = popcnt[avail[v]]
                if c < min_c or (c == min_c and len(adj[v]) > max_deg):
                    min_c = c
                    max_deg = len(adj[v])
                    best_v = v
                    if min_c <= 1:
                        break
        
        if min_c == 0:
            return False
            
        v = best_v
        mask = avail[v]
        
        # Try colors
        for c in range(3):
            if (mask & (1 << c)):
                color[v] = c
                # Forward checking & save changes
                changed = []
                possible = True
                c_mask = ~(1 << c)
                
                for u in adj[v]:
                    if color[u] == -1:
                        if avail[u] & (1 << c):
                            avail[u] &= c_mask
                            changed.append(u)
                            if avail[u] == 0:
                                possible = False
                
                if possible and backtrack(unassigned_count - 1):
                    return True
                
                # Undo
                for u in changed:
                    avail[u] |= (1 << c)
                color[v] = -1
                
        return False

    if backtrack(n):
        print("YES")
        print(*(color[i] + 1 for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
