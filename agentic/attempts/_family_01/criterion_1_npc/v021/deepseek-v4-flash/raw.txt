```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    
    # If m is 0, we can always assign distinct phases
    if m == 0:
        print("YES")
        return
    
    # If n is too small to have m pairs, impossible
    if n < 2:
        print("NO")
        return
    
    # Read all pairs
    pairs = []
    for _ in range(m):
        u, v = map(int, input().split())
        pairs.append((u, v))
    
    # For n <= 40, we can try all possible phase assignments
    # But that's 3^40 which is too much.
    # Key insight: This is a graph coloring problem with 3 colors.
    # We need to check if the graph is 3-colorable.
    # Since n <= 40, we can use backtracking with pruning.
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    for u, v in pairs:
        adj[u].append(v)
        adj[v].append(u)
    
    # Check if graph is 3-colorable using backtracking
    colors = [0] * (n + 1)  # 0 = uncolored, 1, 2, 3 = colors
    
    def is_safe(vertex, color):
        for neighbor in adj[vertex]:
            if colors[neighbor] == color:
                return False
        return True
    
    def backtrack(vertex):
        if vertex > n:
            return True
        
        # Try all 3 colors
        for color in range(1, 4):
            if is_safe(vertex, color):
                colors[vertex] = color
                if backtrack(vertex + 1):
                    return True
                colors[vertex] = 0
        
        return False
    
    if backtrack(1):
        print("YES")
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
