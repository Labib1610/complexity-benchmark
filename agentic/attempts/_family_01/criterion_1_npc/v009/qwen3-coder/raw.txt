def solve():
    n, m = map(int, input().split())
    
    # Create adjacency list
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 means uncolored, 1, 2, 3 are the colors
    color = [0] * (n + 1)
    
    # Try to color each component
    for start in range(1, n + 1):
        if color[start] == 0:
            # BFS to color this component
            queue = [start]
            color[start] = 1
            
            while queue:
                node = queue.pop(0)
                current_color = color[node]
                
                # Try all neighbors
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        # Assign the first available color (not current_color)
                        # We need to pick a color from {1,2,3} that's different
                        next_color = 1
                        while next_color == current_color:
                            next_color += 1
                            if next_color > 3:
                                next_color = 1
                        
                        # But we also need to make sure it's different from all neighbors
                        valid_colors = [1, 2, 3]
                        for nn in adj[neighbor]:
                            if color[nn] != 0 and color[nn] in valid_colors:
                                valid_colors.remove(color[nn])
                        
                        if not valid_colors:
                            print("NO")
                            return
                        
                        color[neighbor] = valid_colors[0]
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # If we get here, we have a valid coloring
    print("YES")
    print(*color[1:])

solve()
