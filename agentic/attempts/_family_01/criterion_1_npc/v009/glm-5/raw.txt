import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Partition vertices into connected components
    visited = [False] * n
    components = []
    for i in range(n):
        if not visited[i]:
            comp_nodes = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp_nodes.append(u)
                # Iterate neighbors
                neighbors = adj[u]
                v = 0
                while neighbors:
                    if neighbors & 1:
                        if not visited[v]:
                            visited[v] = True
                            stack.append(v)
                    neighbors >>= 1
                    v += 1
            components.append(comp_nodes)
    
    assignment = [0] * n
    
    for comp in components:
        k = len(comp)
        # Map local index to global index
        idx_map = {u: i for i, u in enumerate(comp)}
        # Build local adjacency mask
        local_adj = [0] * k
        for i, u in enumerate(comp):
            mask = 0
            neighbors = adj[u]
            v = 0
            while neighbors:
                if neighbors & 1:
                    if v in idx_map:
                        mask |= 1 << idx_map[v]
                neighbors >>= 1
                v += 1
            local_adj[i] = mask
        
        # Try to 3-color this component
        found = False
        # Heuristic: try vertices with higher degree first
        order = sorted(range(k), key=lambda x: bin(local_adj[x]).count('1'), reverse=True)
        
        # Backtracking with pruning
        colors = [0] * k
        
        def backtrack(d):
            nonlocal found
            if found:
                return
            if d == k:
                found = True
                return
            
            u = order[d]
            # Determine colors used by neighbors
            used = 0
            neighbors = local_adj[u]
            v = 0
            while neighbors:
                if neighbors & 1:
                    used |= 1 << colors[v]
                neighbors >>= 1
                v += 1
            
            for c in range(1, 4):
                if not (used & (1 << c)):
                    colors[u] = c
                    backtrack(d + 1)
                    if found:
                        return
                    colors[u] = 0
        
        backtrack(0)
        
        if not found:
            print("NO")
            return
        
        for i, u in enumerate(comp):
            assignment[u] = colors[i]
    
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    solve()
