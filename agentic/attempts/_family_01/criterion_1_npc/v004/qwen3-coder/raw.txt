def solve():
    n, m = map(int, input().split())
    
    # Create adjacency list representation of conflicts
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to color the graph with 3 colors
    # Color 0 means unassigned, colors 1, 2, 3 represent shifts
    color = [0] * (n + 1)
    
    def is_safe(node, c):
        # Check if assigning color c to node is safe
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for c in range(1, 4):
            if is_safe(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        
        return False
    
    # Try to assign shifts to all workers
    if backtrack(1):
        print("YES")
        print(*color[1:])
    else:
        print("NO")

solve()
