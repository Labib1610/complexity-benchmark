import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                if not dfs(v, 3 - c):
                    return False
            elif color[v] == c:
                return False
        return True
    
    for i in range(n):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
    
    # Build graph of vertices with color 1
    # Find connected components and check if bipartite
    g1_adj = [[] for _ in range(n)]
    for u in range(n):
        if color[u] == 1:
            for v in adj[u]:
                if color[v] == 1:
                    g1_adj[u].append(v)
                    g1_adj[v].append(u)
    
    visited = [False] * n
    comp_id = [-1] * n
    components = []
    
    for i in range(n):
        if color[i] == 1 and not visited[i]:
            stack = [i]
            visited[i] = True
            comp = []
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in g1_adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            for u in comp:
                comp_id[u] = len(components)
            components.append(comp)
    
    k = len(components)
    part = [-1] * n
    
    for idx, comp in enumerate(components):
        if len(comp) == 1:
            part[comp[0]] = 0
            continue
        start = comp[0]
        part[start] = 0
        stack = [start]
        is_bipartite = True
        while stack and is_bipartite:
            u = stack.pop()
            for v in g1_adj[u]:
                if part[v] == -1:
                    part[v] = 1 - part[u]
                    stack.append(v)
                elif part[v] == part[u]:
                    is_bipartite = False
                    break
        if not is_bipartite:
            print("NO")
            return
    
    # Assign shifts
    shift = [0] * n
    for i in range(n):
        if color[i] == 2:
            shift[i] = 3
        else:
            if part[i] == 0:
                shift[i] = 1
            else:
                shift[i] = 2
    
    print("YES")
    print(' '.join(map(str, shift)))

solve()
