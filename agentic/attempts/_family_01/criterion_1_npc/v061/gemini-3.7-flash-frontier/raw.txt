import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    adj_mask = [0] * n
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
    
    # Remove duplicate edges
    for i in range(n):
        adj[i] = list(set(adj[i]))

    # Connected components
    visited = [False] * n
    components = []
    for i in range(n):
        if not visited[i]:
            comp = []
            q = [i]
            visited[i] = True
            while q:
                curr = q.pop()
                comp.append(curr)
                for neighbor in adj[curr]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        q.append(neighbor)
            components.append(comp)

    final_assignment = [0] * n

    for comp in components:
        # Solve 3-coloring for this component
        # domains: bitmask for each vertex, 1: color 0, 2: color 1, 4: color 2
        domains = [7] * n
        
        # Symmetry breaking: color first vertex with color 0 (mask 1)
        root = comp[0]
        
        def propagate(dom, queue):
            while queue:
                u = queue.pop()
                c_mask = dom[u]
                if c_mask == 0:
                    return False
                # c_mask has exactly one bit
                for v in adj[u]:
                    if dom[v] & c_mask:
                        dom[v] &= ~c_mask
                        if dom[v] == 0:
                            return False
                        if dom[v] & (dom[v] - 1) == 0: # single bit remaining
                            queue.append(v)
            return True

        def backtrack(dom):
            # Check if all assigned in comp
            best_v = -1
            best_size = 4
            best_deg = -1
            
            for u in comp:
                sz = bin(dom[u]).count('1')
                if sz > 1:
                    if sz < best_size or (sz == best_size and len(adj[u]) > best_deg):
                        best_size = sz
                        best_deg = len(adj[u])
                        best_v = u
            
            if best_v == -1:
                return dom
            
            # Try available colors for best_v
            for c in range(3):
                if dom[best_v] & (1 << c):
                    new_dom = list(dom)
                    new_dom[best_v] = (1 << c)
                    queue = [best_v]
                    if propagate(new_dom, queue):
                        res = backtrack(new_dom)
                        if res is not None:
                            return res
            return None

        domains[root] = 1
        queue = [root]
        if not propagate(domains, queue):
            print("NO")
            return
        
        res = backtrack(domains)
        if res is None:
            print("NO")
            return
        
        for u in comp:
            # Map mask to color 1, 2, 3
            mask = res[u]
            if mask == 1:
                final_assignment[u] = 1
            elif mask == 2:
                final_assignment[u] = 2
            elif mask == 4:
                final_assignment[u] = 3
            else:
                # Should not happen
                final_assignment[u] = 1

    print("YES")
    print(*(final_assignment))

if __name__ == '__main__':
    solve()
