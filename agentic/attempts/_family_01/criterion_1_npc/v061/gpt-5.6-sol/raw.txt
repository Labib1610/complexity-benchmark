import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n
    pos = 2

    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u == v:
            print("NO")
            return
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    all_vertices = (1 << n) - 1

    def bipartite(mask):
        side0 = 0
        side1 = 0
        unassigned = mask

        while unassigned:
            root = unassigned & -unassigned
            side0 |= root
            unassigned ^= root
            queue = root

            while queue:
                bit = queue & -queue
                queue ^= bit
                v = bit.bit_length() - 1
                neighbors = adj[v] & mask

                if bit & side0:
                    if neighbors & side0:
                        return None
                    new_vertices = neighbors & ~(side0 | side1)
                    side1 |= new_vertices
                else:
                    if neighbors & side1:
                        return None
                    new_vertices = neighbors & ~(side0 | side1)
                    side0 |= new_vertices

                unassigned &= ~new_vertices
                queue |= new_vertices

        return side0, side1

    colors = [0] * n

    def solve_component(universe):
        allowed = [0] * n
        bits = universe
        while bits:
            bit = bits & -bits
            bits ^= bit
            v = bit.bit_length() - 1
            allowed[v] = universe & ~adj[v] & ~bit

        answer = None

        def search(chosen, candidates, excluded):
            nonlocal answer

            partition = bipartite(universe ^ chosen)
            if partition is not None:
                answer = (partition[0], partition[1], chosen)
                return True

            if not candidates:
                return False

            if excluded and bipartite(excluded) is None:
                return False

            pivot_set = candidates | excluded
            pivot = -1
            best_count = -1
            q = pivot_set
            while q:
                bit = q & -q
                q ^= bit
                v = bit.bit_length() - 1
                count = (candidates & allowed[v]).bit_count()
                if count > best_count:
                    best_count = count
                    pivot = v

            branch = candidates & ~allowed[pivot]

            while branch:
                bit = branch & -branch
                branch ^= bit
                v = bit.bit_length() - 1

                if search(
                    chosen | bit,
                    candidates & allowed[v],
                    excluded & allowed[v]
                ):
                    return True

                candidates ^= bit
                excluded |= bit

            return False

        if not search(0, universe, 0):
            return False

        side0, side1, side2 = answer
        bits = side0
        while bits:
            bit = bits & -bits
            bits ^= bit
            colors[bit.bit_length() - 1] = 1

        bits = side1
        while bits:
            bit = bits & -bits
            bits ^= bit
            colors[bit.bit_length() - 1] = 2

        bits = side2
        while bits:
            bit = bits & -bits
            bits ^= bit
            colors[bit.bit_length() - 1] = 3

        return True

    remaining = all_vertices
    while remaining:
        root = remaining & -remaining
        component = 0
        frontier = root
        remaining ^= root

        while frontier:
            bit = frontier & -frontier
            frontier ^= bit
            component |= bit
            v = bit.bit_length() - 1
            new_vertices = adj[v] & remaining
            remaining &= ~new_vertices
            frontier |= new_vertices

        if not solve_component(component):
            print("NO")
            return

    print("YES")
    print(*colors)

if __name__ == "__main__":
    main()
