We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use a backtracking approach with pruning, or a meet-in-the-middle / bitmask DP.

Here’s one efficient approach:

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [0] * n
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)

    # Try to color with 3 colors using backtracking
    color = [-1] * n

    def dfs(node):
        if node == n:
            return True
        # Try colors 0,1,2
        for c in range(3):
            ok = True
            for i in range(n):
                if (adj[node] >> i) & 1 and color[i] == c:
                    ok = False
                    break
            if ok:
                color[node] = c
                if dfs(node + 1):
                    return True
                color[node] = -1
        return False

    if dfs(0):
        print("YES")
        print(" ".join(str(c + 1) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
