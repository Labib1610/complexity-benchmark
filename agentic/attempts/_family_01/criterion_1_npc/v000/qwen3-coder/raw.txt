def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Create adjacency list
    adj = [[] for _ in range(n+1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Color the graph with 3 colors
    color = [0] * (n+1)  # 0 means uncolored, 1,2,3 are the three shifts
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def dfs(node):
        if node > n:
            return True
            
        for c in range(1, 4):
            if is_valid(node, c):
                color[node] = c
                if dfs(node + 1):
                    return True
                color[node] = 0
                
        return False
    
    # Try all possible colorings
    if dfs(1):
        print("YES")
        print(*color[1:])
    else:
        print("NO")

solve()
