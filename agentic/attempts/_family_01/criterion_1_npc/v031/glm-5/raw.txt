import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    adj = [0]*(n+1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Find connected components
    comp_id = [0]*(n+1)
    comps = []
    for i in range(1, n+1):
        if comp_id[i] == 0:
            comp = []
            stack = [i]
            comp_id[i] = len(comps)+1
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in range(1, n+1):
                    if (adj[u] >> v) & 1 and comp_id[v] == 0:
                        comp_id[v] = len(comps)+1
                        stack.append(v)
            comps.append(comp)
    
    color = [0]*(n+1)
    
    for comp in comps:
        # Try to 2-color the component (bipartite check)
        # If not bipartite, need 3 colors
        # First check bipartiteness
        c = [0]*(n+1)  # 0 uncolored, 1,2
        is_bip = True
        for s in comp:
            if c[s] == 0:
                c[s] = 1
                stack = [s]
                while stack and is_bip:
                    u = stack.pop()
                    for v in range(1, n+1):
                        if (adj[u] >> v) & 1:
                            if c[v] == 0:
                                c[v] = 3 - c[u]
                                stack.append(v)
                            elif c[v] == c[u]:
                                is_bip = False
                                break
        
        if is_bip:
            # Use colors 1 and 2
            for u in comp:
                color[u] = c[u]
        else:
            # Need 3 colors. Use backtracking on the component.
            comp_set = set(comp)
            comp_list = comp
            # Order by degree descending for better pruning
            comp_list.sort(key=lambda x: -bin(adj[x] & 0).count('1'))
            # Actually sort by degree within component
            comp_list.sort(key=lambda x: -bin(adj[x]).count('1'))
            
            col = [0]*(n+1)
            found = [False]
            
            def backtrack(pos):
                if found[0]:
                    return
                if pos == len(comp_list):
                    found[0] = True
                    return
                u = comp_list[pos]
                used = 0
                for v in comp_list:
                    if (adj[u] >> v) & 1 and col[v] != 0:
                        used |= 1 << col[v]
                for c in (1,2,3):
                    if not (used >> c) & 1:
                        col[u] = c
                        backtrack(pos+1)
                        if found[0]:
                            return
                        col[u] = 0
            
            backtrack(0)
            if not found[0]:
                print("NO")
                return
            for u in comp:
                color[u] = col[u]
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n+1)))

solve()
