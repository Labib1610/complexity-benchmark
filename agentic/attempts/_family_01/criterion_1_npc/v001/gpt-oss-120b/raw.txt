import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n  # bitmask of neighbors
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    colors = [-1] * n  # -1 uncolored, else 0,1,2

    # order vertices by degree descending for initial heuristic
    order = sorted(range(n), key=lambda x: bin(adj[x]).count('1'), reverse=True)

    # Precompute neighbor list for fast iteration
    neigh = [ [j for j in range(n) if (adj[i] >> j) & 1] for i in range(n) ]

    # Choose next vertex: uncolored with most colored neighbors (DSATUR)
    def select_vertex():
        best = -1
        best_sat = -1
        best_deg = -1
        for v in range(n):
            if colors[v] != -1:
                continue
            used = 0
            deg = 0
            for nb in neigh[v]:
                if colors[nb] != -1:
                    used |= 1 << colors[nb]
                deg += 1
            sat = bin(used).count('1')
            if sat > best_sat or (sat == best_sat and deg > best_deg):
                best_sat = sat
                best_deg = deg
                best = v
        return best

    def dfs(colored_cnt: int) -> bool:
        if colored_cnt == n:
            return True
        v = select_vertex()
        # compute forbidden colors
        forbid = 0
        for nb in neigh[v]:
            c = colors[nb]
            if c != -1:
                forbid |= 1 << c
        for col in range(3):
            if not (forbid >> col) & 1:
                colors[v] = col
                if dfs(colored_cnt + 1):
                    return True
                colors[v] = -1
        return False

    possible = dfs(0)
    if not possible:
        print("NO")
    else:
        print("YES")
        # convert to 1..3
        print(' '.join(str(c+1) for c in colors))

if __name__ == "__main__":
    solve()
