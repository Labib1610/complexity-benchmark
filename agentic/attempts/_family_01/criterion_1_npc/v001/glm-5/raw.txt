import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full = (1 << n) - 1
    independent_sets = []
    for v in range(n):
        new_sets = []
        for s in independent_sets:
            if not (adj[v] & s):
                new_sets.append(s | (1 << v))
        independent_sets.extend(new_sets)
        independent_sets.append(1 << v)

    independent_sets.append(0)
    independent_sets = list(set(independent_sets))

    cover = [0] * (1 << n)
    for s in independent_sets:
        cover[s] = s

    for i in range(n):
        bit = 1 << i
        for mask in range(1 << n):
            if mask & bit:
                prev = mask ^ bit
                if cover[prev] != 0 and cover[prev] != 0:
                    if cover[mask] == 0 or bin(cover[prev]).count('1') > bin(cover[mask]).count('1'):
                        cover[mask] = cover[prev]

    for s1 in independent_sets:
        rem1 = full & ~s1
        s2 = cover[rem1]
        if s2 != 0:
            rem2 = rem1 & ~s2
            s3 = cover[rem2]
            if s3 != 0 and (s3 & rem2) == rem2:
                color = [0] * n
                for i in range(n):
                    if s1 & (1 << i):
                        color[i] = 1
                    elif s2 & (1 << i):
                        color[i] = 2
                    else:
                        color[i] = 3
                print("YES")
                print(' '.join(map(str, color)))
                return

    print("NO")

solve()
