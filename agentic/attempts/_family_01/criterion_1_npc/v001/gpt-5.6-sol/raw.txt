import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    def bipartition(vertices):
        side0 = 0
        side1 = 0
        assigned = 0

        while True:
            unseen = vertices & ~assigned
            if not unseen:
                return side0, side1

            root = unseen & -unseen
            side0 |= root
            assigned |= root
            frontier = root
            parity = 0

            while frontier:
                neighbors = 0
                bits = frontier
                while bits:
                    bit = bits & -bits
                    bits ^= bit
                    neighbors |= adj[bit.bit_length() - 1]
                neighbors &= vertices

                same_side = side0 if parity == 0 else side1
                if neighbors & same_side:
                    return None

                new_vertices = neighbors & ~assigned
                if parity == 0:
                    side1 |= new_vertices
                else:
                    side0 |= new_vertices

                assigned |= new_vertices
                frontier = new_vertices
                parity ^= 1

    all_vertices = (1 << n) - 1
    remaining = all_vertices
    components = []

    while remaining:
        root = remaining & -remaining
        component = 0
        frontier = root
        remaining ^= root

        while frontier:
            component |= frontier
            neighbors = 0
            bits = frontier
            while bits:
                bit = bits & -bits
                bits ^= bit
                neighbors |= adj[bit.bit_length() - 1]
            new_vertices = neighbors & remaining
            remaining ^= new_vertices
            frontier = new_vertices

        components.append(component)

    components.sort(key=int.bit_count, reverse=True)
    colors = [0] * n

    for component in components:
        direct = bipartition(component)
        if direct is not None:
            a, b = direct
            bits = a
            while bits:
                bit = bits & -bits
                bits ^= bit
                colors[bit.bit_length() - 1] = 1
            bits = b
            while bits:
                bit = bits & -bits
                bits ^= bit
                colors[bit.bit_length() - 1] = 2
            continue

        comp_adj = [0] * n
        bits = component
        while bits:
            bit = bits & -bits
            bits ^= bit
            v = bit.bit_length() - 1
            comp_adj[v] = component & ~(adj[v] | bit)

        answer = None

        def search(independent, candidates, excluded):
            nonlocal answer
            if answer is not None:
                return True

            forced_remaining = component & ~(independent | candidates)
            parts = bipartition(forced_remaining)
            if parts is None:
                return False

            if candidates == 0:
                if excluded == 0:
                    answer = (independent, parts[0], parts[1])
                    return True
                return False

            union = candidates | excluded
            pivot = -1
            best = -1
            scan = union
            while scan:
                bit = scan & -scan
                scan ^= bit
                v = bit.bit_length() - 1
                score = (candidates & comp_adj[v]).bit_count()
                if score > best:
                    best = score
                    pivot = v

            branch_vertices = candidates & ~comp_adj[pivot]

            while branch_vertices:
                bit = branch_vertices & -branch_vertices
                branch_vertices ^= bit
                v = bit.bit_length() - 1

                if search(
                    independent | bit,
                    candidates & comp_adj[v],
                    excluded & comp_adj[v]
                ):
                    return True

                candidates ^= bit
                excluded |= bit

            return False

        search(0, component, 0)

        if answer is None:
            print("NO")
            return

        independent, a, b = answer

        bits = independent
        while bits:
            bit = bits & -bits
            bits ^= bit
            colors[bit.bit_length() - 1] = 1

        bits = a
        while bits:
            bit = bits & -bits
            bits ^= bit
            colors[bit.bit_length() - 1] = 2

        bits = b
        while bits:
            bit = bits & -bits
            bits ^= bit
            colors[bit.bit_length() - 1] = 3

    print("YES")
    print(*colors)

if __name__ == "__main__":
    main()
