import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    avail = [7] * n
    color = [0] * n
    uncolored_count = n
    popcnt = [0, 1, 1, 2, 1, 2, 2, 3]

    deg = [bin(adj[i]).count("1") for i in range(n)]

    def backtrack():
        nonlocal uncolored_count
        if uncolored_count == 0:
            return True

        best_v = -1
        best_cnt = 4
        best_deg = -1

        for v in range(n):
            if color[v] == 0:
                c_cnt = popcnt[avail[v]]
                if c_cnt == 0:
                    return False
                if c_cnt < best_cnt:
                    best_cnt = c_cnt
                    best_v = v
                    best_deg = deg[v]
                elif c_cnt == best_cnt:
                    if deg[v] > best_deg:
                        best_v = v
                        best_deg = deg[v]

        v = best_v
        v_avail = avail[v]

        for c in (1, 2, 4):
            if v_avail & c:
                color[v] = c
                uncolored_count -= 1

                changed = []
                curr_nbrs = adj[v]
                valid = True
                while curr_nbrs:
                    lsb = curr_nbrs & -curr_nbrs
                    u = lsb.bit_length() - 1
                    curr_nbrs ^= lsb
                    if color[u] == 0:
                        if avail[u] & c:
                            avail[u] ^= c
                            changed.append(u)
                            if avail[u] == 0:
                                valid = False

                if valid and backtrack():
                    return True

                for u in changed:
                    avail[u] |= c
                color[v] = 0
                uncolored_count += 1

        return False

    sys.setrecursionlimit(2000)
    if backtrack():
        print("YES")
        c_map = {1: 1, 2: 2, 4: 3}
        ans = [c_map[color[i]] for i in range(n)]
        print(*(ans))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
