import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Find all maximal independent sets using Bron-Kerbosch with pivoting
    # We only need maximal independent sets, but any independent set works for coloring.
    # Actually for 3-coloring, we need to cover all vertices with 3 independent sets.
    # We'll find maximal independent sets and try to cover remaining with 2 colors (bipartite check).
    
    mis_list = []
    
    def bron_kerbosch(R, P, X):
        if P == 0 and X == 0:
            mis_list.append(R)
            return
        # Pivot selection
        pivot_candidates = P | X
        # Choose pivot with max degree in P
        pivot = -1
        max_deg = -1
        p = pivot_candidates
        while p:
            v = (p & -p).bit_length() - 1
            deg = bin(adj[v] & P).count('1')
            if deg > max_deg:
                max_deg = deg
                pivot = v
            p &= p - 1
        # Candidates: P \ N(pivot)
        candidates = P & ~adj[pivot]
        c = candidates
        while c:
            v = (c & -c).bit_length() - 1
            v_bit = 1 << v
            bron_kerbosch(R | v_bit, P & ~adj[v] & ~v_bit, X & ~adj[v])
            P &= ~v_bit
            X |= v_bit
            c &= c - 1
    
    full = (1 << n) - 1
    bron_kerbosch(0, full, 0)
    
    # For each maximal independent set as color 1, check if remaining is bipartite
    for mis in mis_list:
        remaining = full & ~mis
        # Check bipartite using BFS on remaining subgraph
        color = [-1] * n
        ok = True
        r = remaining
        while r and ok:
            # Find first set bit
            start = (r & -r).bit_length() - 1
            queue = [start]
            color[start] = 0
            while queue and ok:
                u = queue.pop()
                u_bit = 1 << u
                if not (u_bit & remaining):
                    continue
                neighbors = adj[u] & remaining
                nb = neighbors
                while nb:
                    v = (nb & -nb).bit_length() - 1
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        queue.append(v)
                    elif color[v] == color[u]:
                        ok = False
                        break
                    nb &= nb - 1
            # Clear processed component bits from r
            # Actually we need to remove all vertices in this component
            # But simpler: just remove start and let loop continue
            # However vertices added to queue may not be cleared from r
            # Let's just clear all colored vertices from r
            r = remaining
            for i in range(n):
                if color[i] != -1:
                    r &= ~(1 << i)
        if ok:
            # Build assignment
            res = [0] * n
            for i in range(n):
                if (mis >> i) & 1:
                    res[i] = 1
                elif color[i] == 0:
                    res[i] = 2
                else:
                    res[i] = 3
            print("YES")
            print(" ".join(map(str, res)))
            return
    
    print("NO")

solve()
