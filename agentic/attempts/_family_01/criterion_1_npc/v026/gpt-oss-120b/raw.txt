import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    neigh = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        neigh[u].append(v)
        neigh[v].append(u)

    # possible colors bitmask for each vertex, bits 0,1,2 correspond to colors 1,2,3
    poss = [0b111] * n
    color = [-1] * n
    uncolored = set(range(n))

    # order vertices by degree descending for initial heuristic
    order = sorted(range(n), key=lambda x: -len(neigh[x]))

    # select next vertex: the one with smallest number of possible colors (MRV), tie by degree
    def select_vertex():
        v = -1
        best = 4
        best_deg = -1
        for u in uncolored:
            cnt = (poss[u] & 0b111).bit_count()
            if cnt < best or (cnt == best and len(neigh[u]) > best_deg):
                best = cnt
                best_deg = len(neigh[u])
                v = u
        return v

    # recursive backtracking
    def dfs():
        if not uncolored:
            return True
        v = select_vertex()
        mask = poss[v]
        # try each possible color
        for c in range(3):
            if mask & (1 << c):
                # assign color c to v
                color[v] = c
                uncolored.remove(v)
                # record changes to restore later
                changed = []
                ok = True
                for nb in neigh[v]:
                    if nb in uncolored:
                        if poss[nb] & (1 << c):
                            newmask = poss[nb] & ~(1 << c)
                            if newmask == 0:
                                ok = False
                                # no need to continue updating
                                # but still record for proper backtrack
                            changed.append((nb, poss[nb]))
                            poss[nb] = newmask
                if ok and dfs():
                    return True
                # backtrack
                for nb, old in changed:
                    poss[nb] = old
                color[v] = -1
                uncolored.add(v)
        return False

    if dfs():
        out = ['YES', ' '.join(str(c+1) for c in color)]
    else:
        out = ['NO']
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
