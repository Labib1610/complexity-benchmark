import sys

sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    domain = [7] * n
    
    def backtrack():
        min_choices = 4
        best_u = -1
        max_deg = -1
        
        for i in range(n):
            if color[i] == 0:
                d = domain[i]
                cnt = (d & 1) + ((d >> 1) & 1) + ((d >> 2) & 1)
                if cnt == 0:
                    return False
                if cnt < min_choices:
                    min_choices = cnt
                    best_u = i
                    max_deg = len(adj[i])
                elif cnt == min_choices:
                    if len(adj[i]) > max_deg:
                        best_u = i
                        max_deg = len(adj[i])
                        
        if best_u == -1:
            return True
            
        u = best_u
        d = domain[u]
        
        for c in (1, 2, 3):
            bit = 1 << (c - 1)
            if d & bit:
                color[u] = c
                changed = []
                possible = True
                for v in adj[u]:
                    if color[v] == 0:
                        if domain[v] & bit:
                            domain[v] &= ~bit
                            changed.append(v)
                            if domain[v] == 0:
                                possible = False
                
                if possible and backtrack():
                    return True
                
                for v in changed:
                    domain[v] |= bit
                color[u] = 0
                
        return False

    if backtrack():
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
