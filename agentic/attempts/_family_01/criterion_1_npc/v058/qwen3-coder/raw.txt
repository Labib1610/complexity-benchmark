```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Try all 3^n assignments
    for mask in range(3**n):
        assignment = []
        temp_mask = mask
        valid = True
        for i in range(n):
            tier = (temp_mask % 3) + 1
            assignment.append(tier)
            temp_mask //= 3
        
        # Check all edges
        for u, v in edges:
            if abs(assignment[u] - assignment[v]) < 1:
                valid = False
                break
        
        if valid:
            print("YES")
            print(' '.join(map(str, assignment)))
            return
    
    print("NO")

if __name__ == "__main__":
    main()
```
