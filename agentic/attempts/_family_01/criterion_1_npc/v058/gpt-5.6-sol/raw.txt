import sys

sys.setrecursionlimit(10000)


def solve_block(vertices, global_adj):
    k = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}
    adj = [0] * k

    for i, v in enumerate(vertices):
        mask = global_adj[v]
        local_mask = 0
        for j, w in enumerate(vertices):
            if (mask >> w) & 1:
                local_mask |= 1 << j
        adj[i] = local_mask

    full = (1 << k) - 1

    # A K4 makes 3-coloring impossible.
    for u in range(k):
        candidates = adj[u] & ~((1 << (u + 1)) - 1)
        while candidates:
            vb = candidates & -candidates
            candidates ^= vb
            v = vb.bit_length() - 1
            common = adj[u] & adj[v]
            check = common
            while check:
                xb = check & -check
                check ^= xb
                x = xb.bit_length() - 1
                if adj[x] & common:
                    return None

    def bipartite(rem):
        left = 0
        right = 0
        unseen = rem

        while unseen:
            seed = unseen & -unseen
            unseen ^= seed
            left |= seed
            todo = seed

            while todo:
                bit = todo & -todo
                todo ^= bit
                v = bit.bit_length() - 1
                neighbors = adj[v] & rem

                if left & bit:
                    if neighbors & left:
                        return None
                    new_vertices = neighbors & unseen
                    if new_vertices:
                        unseen ^= new_vertices
                        right |= new_vertices
                        todo |= new_vertices
                else:
                    if neighbors & right:
                        return None
                    new_vertices = neighbors & unseen
                    if new_vertices:
                        unseen ^= new_vertices
                        left |= new_vertices
                        todo |= new_vertices

        return left, right

    initial = bipartite(full)
    if initial is not None:
        left, right = initial
        colors = [0] * k
        for i in range(k):
            colors[i] = 1 if (left >> i) & 1 else 2
        return colors

    complement = [full ^ (1 << i) ^ adj[i] for i in range(k)]
    target = (k + 2) // 3
    answer = None

    def search(chosen, possible, excluded, chosen_size):
        nonlocal answer
        if answer is not None:
            return

        if chosen_size + possible.bit_count() < target:
            return

        if chosen_size == target:
            result = bipartite(full ^ chosen)
            if result is not None:
                left, right = result
                answer = (chosen, left, right)
                return

        if possible == 0:
            if excluded == 0 and chosen_size >= target:
                result = bipartite(full ^ chosen)
                if result is not None:
                    left, right = result
                    answer = (chosen, left, right)
            return

        pivot_set = possible | excluded
        pivot = -1
        best_count = -1
        temp = pivot_set
        while temp:
            bit = temp & -temp
            temp ^= bit
            v = bit.bit_length() - 1
            count = (possible & complement[v]).bit_count()
            if count > best_count:
                best_count = count
                pivot = v

        candidates = possible if pivot == -1 else possible & ~complement[pivot]

        while candidates and answer is None:
            bit = candidates & -candidates
            candidates ^= bit
            v = bit.bit_length() - 1

            search(
                chosen | bit,
                possible & complement[v],
                excluded & complement[v],
                chosen_size + 1
            )

            possible ^= bit
            excluded |= bit

            if chosen_size + possible.bit_count() < target:
                break

    search(0, full, 0, 0)

    if answer is None:
        return None

    chosen, left, right = answer
    colors = [0] * k
    for i in range(k):
        bit = 1 << i
        if chosen & bit:
            colors[i] = 3
        elif left & bit:
            colors[i] = 1
        else:
            colors[i] = 2
    return colors


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    graph = [[] for _ in range(n)]
    global_adj = [0] * n

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if not ((global_adj[u] >> v) & 1):
            graph[u].append(v)
            graph[v].append(u)
            global_adj[u] |= 1 << v
            global_adj[v] |= 1 << u

    discovery = [-1] * n
    low = [0] * n
    timer = 0
    edge_stack = []
    blocks = []

    def dfs(u, parent):
        nonlocal timer
        discovery[u] = low[u] = timer
        timer += 1

        for v in graph[u]:
            if discovery[v] == -1:
                edge_stack.append((u, v))
                dfs(v, u)
                low[u] = min(low[u], low[v])

                if low[v] >= discovery[u]:
                    vertices = set()
                    while True:
                        a, b = edge_stack.pop()
                        vertices.add(a)
                        vertices.add(b)
                        if a == u and b == v:
                            break
                    blocks.append(list(vertices))

            elif v != parent and discovery[v] < discovery[u]:
                edge_stack.append((u, v))
                low[u] = min(low[u], discovery[v])

    for start in range(n):
        if discovery[start] == -1:
            dfs(start, -1)

    block_colors = []
    for vertices in blocks:
        colors = solve_block(vertices, global_adj)
        if colors is None:
            print("NO")
            return
        block_colors.append(colors)

    vertex_blocks = [[] for _ in range(n)]
    for block_id, vertices in enumerate(blocks):
        for v in vertices:
            vertex_blocks[v].append(block_id)

    final_colors = [0] * n
    seen_blocks = [False] * len(blocks)

    for root in range(len(blocks)):
        if seen_blocks[root]:
            continue

        seen_blocks[root] = True
        stack = [root]

        while stack:
            block_id = stack.pop()
            vertices = blocks[block_id]
            colors = block_colors[block_id]

            shared = -1
            for i, v in enumerate(vertices):
                if final_colors[v] != 0:
                    shared = i
                    break

            if shared != -1:
                current = colors[shared]
                required = final_colors[vertices[shared]]
                if current != required:
                    for i in range(len(colors)):
                        if colors[i] == current:
                            colors[i] = required
                        elif colors[i] == required:
                            colors[i] = current

            for i, v in enumerate(vertices):
                final_colors[v] = colors[i]

            for v in vertices:
                for next_block in vertex_blocks[v]:
                    if not seen_blocks[next_block]:
                        seen_blocks[next_block] = True
                        stack.append(next_block)

    for v in range(n):
        if final_colors[v] == 0:
            final_colors[v] = 1

    print("YES")
    print(*final_colors)


if __name__ == "__main__":
    main()
