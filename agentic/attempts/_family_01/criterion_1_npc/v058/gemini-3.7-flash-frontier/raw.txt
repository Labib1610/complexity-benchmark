import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Check 2-colorability for connected components or fast path
    # But general 3-coloring solver works for all.
    # DSATUR-like Backtracking with Bitmasks

    color = [0] * n  # 0: uncolored, 1, 2, 3
    avail = [7] * n  # 7 = 0b111 (colors 1, 2, 3 available)

    # Precompute degrees
    deg = [bin(adj[i]).count("1") for i in range(n)]

    def backtrack(uncolored_mask):
        if uncolored_mask == 0:
            return True

        # Find vertex with minimum available colors
        best_u = -1
        min_c = 4
        max_deg = -1

        temp = uncolored_mask
        while temp > 0:
            u = (temp & -temp).bit_length() - 1
            temp &= temp - 1

            c_count = bin(avail[u]).count("1")
            if c_count == 0:
                return False
            if c_count < min_c or (c_count == min_c and deg[u] > max_deg):
                min_c = c_count
                max_deg = deg[u]
                best_u = u
                if min_c == 1:
                    break

        u = best_u
        u_avail = avail[u]
        u_bit = 1 << u
        next_uncolored = uncolored_mask ^ u_bit

        for c in (1, 2, 3):
            c_mask = 1 << (c - 1)
            if u_avail & c_mask:
                color[u] = c

                # Propagate to neighbors
                changed = []
                temp_adj = adj[u] & next_uncolored
                possible = True

                while temp_adj > 0:
                    v = (temp_adj & -temp_adj).bit_length() - 1
                    temp_adj &= temp_adj - 1
                    if avail[v] & c_mask:
                        avail[v] ^= c_mask
                        changed.append(v)
                        if avail[v] == 0:
                            possible = False

                if possible and backtrack(next_uncolored):
                    return True

                # Undo changes
                for v in changed:
                    avail[v] |= c_mask

                color[u] = 0

        return False

    initial_mask = (1 << n) - 1
    if backtrack(initial_mask):
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
