import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)
            
    deg = [bin(adj[i]).count('1') for i in range(n)]
    color = [0] * n
    
    # We will use backtracking with MRV (Minimum Remaining Values)
    # Available colors for each node will be tracked using bitmask:
    # 1: color 1, 2: color 2, 4: color 3
    # Initial domain: 7 (1 | 2 | 4)
    domains = [7] * n
    
    # Pre-check: self-loops
    # But input says pairs u, v; if self loop exists, no solution.
    # The problem says simple graph or interaction between pair.
    
    def backtrack(assigned_count):
        if assigned_count == n:
            return True
        
        # Select unassigned vertex with minimum domain size
        min_size = 4
        best_u = -1
        best_deg = -1
        
        for u in range(n):
            if color[u] == 0:
                d = domains[u]
                size = (d & 1) + ((d >> 1) & 1) + ((d >> 2) & 1)
                if size == 0:
                    return False
                if size < min_size or (size == min_size and deg[u] > best_deg):
                    min_size = size
                    best_u = u
                    best_deg = deg[u]
                    if size == 1:
                        break
        
        u = best_u
        d = domains[u]
        
        for c in (1, 2, 3):
            c_mask = 1 << (c - 1)
            if d & c_mask:
                color[u] = c
                # Forward check: remove c_mask from neighbors
                changed = []
                possible = True
                nbrs = adj[u]
                while nbrs:
                    lsb = nbrs & -nbrs
                    v = (lsb.bit_length() - 1)
                    nbrs ^= lsb
                    if color[v] == 0 and (domains[v] & c_mask):
                        domains[v] ^= c_mask
                        changed.append(v)
                        if domains[v] == 0:
                            possible = False
                
                if possible and backtrack(assigned_count + 1):
                    return True
                
                # Undo
                for v in changed:
                    domains[v] |= c_mask
                color[u] = 0
                
        return False

    if backtrack(0):
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
