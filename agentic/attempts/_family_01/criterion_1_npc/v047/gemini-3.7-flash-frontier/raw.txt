import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
    color = [-1] * n
    allowed = [7] * n  # 7 = 111 in binary (colors 0, 1, 2 allowed)
    
    # Precompute popcount for 0..7
    popcount = [0, 1, 1, 2, 1, 2, 2, 3]

    def backtrack(assigned_count):
        if assigned_count == n:
            return True
        
        # Pick unassigned vertex with minimum available colors (MRV)
        best_v = -1
        min_c = 4
        for i in range(n):
            if color[i] == -1:
                cnt = popcount[allowed[i]]
                if cnt < min_c:
                    min_c = cnt
                    best_v = i
                    if min_c <= 1:
                        break
        
        if min_c == 0:
            return False
        
        v = best_v
        mask = allowed[v]
        
        for c in range(3):
            if (mask & (1 << c)):
                # Try assigning color c to vertex v
                color[v] = c
                
                # Save affected neighbors' allowed masks
                affected = []
                possible = True
                nbrs = adj[v]
                bit_c = 1 << c
                
                # Iterate over neighbors
                temp_nbrs = nbrs
                while temp_nbrs:
                    # extract lowest set bit
                    lsb = temp_nbrs & -temp_nbrs
                    u = (lsb.bit_length() - 1)
                    temp_nbrs ^= lsb
                    
                    if color[u] == -1 and (allowed[u] & bit_c):
                        affected.append(u)
                        allowed[u] ^= bit_c
                        if allowed[u] == 0:
                            possible = False
                
                if possible and backtrack(assigned_count + 1):
                    return True
                
                # Undo changes
                for u in affected:
                    allowed[u] |= bit_c
                color[v] = -1
                
        return False

    if backtrack(0):
        print("YES")
        print(*(c + 1 for c in color))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
