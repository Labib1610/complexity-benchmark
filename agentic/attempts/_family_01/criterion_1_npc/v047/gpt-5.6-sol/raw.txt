import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n
    p = 2
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Decompose the graph into biconnected blocks.
    disc = [-1] * n
    low = [0] * n
    timer = 0
    edge_stack = []
    blocks = []

    sys.setrecursionlimit(10000)

    def tarjan(u, parent):
        nonlocal timer
        disc[u] = low[u] = timer
        timer += 1

        neighbors = adj[u]
        while neighbors:
            bv = neighbors & -neighbors
            neighbors ^= bv
            v = bv.bit_length() - 1

            if disc[v] == -1:
                edge_stack.append((u, v))
                tarjan(v, u)
                low[u] = min(low[u], low[v])

                if low[v] >= disc[u]:
                    mask = 0
                    while True:
                        a, b = edge_stack.pop()
                        mask |= (1 << a) | (1 << b)
                        if a == u and b == v:
                            break
                    blocks.append(mask)
            elif v != parent and disc[v] < disc[u]:
                edge_stack.append((u, v))
                low[u] = min(low[u], disc[v])

    for root in range(n):
        if disc[root] == -1:
            tarjan(root, -1)

    def analyze(rem, allowed):
        color = [-1] * n
        parent = [-1] * n
        best_cycle = 0
        best_available = n + 1
        best_length = n + 1

        unseen = rem
        while unseen:
            bs = unseen & -unseen
            start = bs.bit_length() - 1
            color[start] = 0
            unseen ^= bs
            q = deque([start])

            while q:
                u = q.popleft()
                neighbors = adj[u] & rem
                while neighbors:
                    bv = neighbors & -neighbors
                    neighbors ^= bv
                    v = bv.bit_length() - 1

                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        parent[v] = u
                        unseen &= ~bv
                        q.append(v)
                    elif color[v] == color[u] and u < v:
                        ancestors = 0
                        x = u
                        while x != -1:
                            ancestors |= 1 << x
                            x = parent[x]

                        side_v = 0
                        y = v
                        while not ((ancestors >> y) & 1):
                            side_v |= 1 << y
                            y = parent[y]
                        lca = y

                        cycle = side_v | (1 << lca)
                        x = u
                        while x != lca:
                            cycle |= 1 << x
                            x = parent[x]

                        available_count = (cycle & allowed).bit_count()
                        cycle_length = cycle.bit_count()
                        if (available_count < best_available or
                            (available_count == best_available and
                             cycle_length < best_length)):
                            best_cycle = cycle
                            best_available = available_count
                            best_length = cycle_length
                            if available_count == 0:
                                return best_cycle, None

        if best_cycle:
            return best_cycle, None
        return 0, color

    def solve_block(mask):
        failed = set()
        cache_limit = 400000

        def dfs(rem, allowed):
            key = rem | (allowed << n)
            if key in failed:
                return None

            cycle, bipartite_colors = analyze(rem, allowed)
            if cycle == 0:
                chosen = mask ^ rem
                result = {}
                vertices = mask
                while vertices:
                    bv = vertices & -vertices
                    vertices ^= bv
                    v = bv.bit_length() - 1
                    if chosen & bv:
                        result[v] = 1
                    else:
                        result[v] = bipartite_colors[v] + 2
                return result

            candidates = cycle & allowed
            if not candidates:
                if len(failed) < cache_limit:
                    failed.add(key)
                return None

            order = []
            temp = candidates
            while temp:
                bv = temp & -temp
                temp ^= bv
                v = bv.bit_length() - 1
                score = (adj[v] & allowed).bit_count()
                order.append((score, v))
            order.sort(reverse=True)

            current_allowed = allowed
            for _, v in order:
                bv = 1 << v
                if not (current_allowed & bv):
                    continue

                child_allowed = current_allowed & ~bv & ~adj[v]
                answer = dfs(rem & ~bv, child_allowed)
                if answer is not None:
                    return answer

                current_allowed &= ~bv

            if len(failed) < cache_limit:
                failed.add(key)
            return None

        return dfs(mask, mask)

    block_colorings = []
    for block in blocks:
        coloring = solve_block(block)
        if coloring is None:
            print("NO")
            return
        block_colorings.append(coloring)

    vertex_blocks = [[] for _ in range(n)]
    for i, block in enumerate(blocks):
        vertices = block
        while vertices:
            bv = vertices & -vertices
            vertices ^= bv
            v = bv.bit_length() - 1
            vertex_blocks[v].append(i)

    answer = [0] * n
    visited_blocks = [False] * len(blocks)

    for start in range(len(blocks)):
        if visited_blocks[start]:
            continue

        visited_blocks[start] = True
        for v, c in block_colorings[start].items():
            answer[v] = c

        stack = [start]
        while stack:
            b = stack.pop()
            block = blocks[b]
            vertices = block

            while vertices:
                bv = vertices & -vertices
                vertices ^= bv
                articulation = bv.bit_length() - 1

                for nb in vertex_blocks[articulation]:
                    if visited_blocks[nb]:
                        continue
                    visited_blocks[nb] = True

                    local = block_colorings[nb]
                    local_shared = local[articulation]
                    global_shared = answer[articulation]

                    remaining_local = [c for c in (1, 2, 3) if c != local_shared]
                    remaining_global = [c for c in (1, 2, 3) if c != global_shared]
                    permutation = {
                        local_shared: global_shared,
                        remaining_local[0]: remaining_global[0],
                        remaining_local[1]: remaining_global[1],
                    }

                    for v, c in local.items():
                        answer[v] = permutation[c]
                    stack.append(nb)

    for i in range(n):
        if answer[i] == 0:
            answer[i] = 1

    print("YES")
    print(*answer)

if __name__ == "__main__":
    main()
