import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    assignment = [0] * n
    
    def is_clique(nodes):
        nodes = list(nodes)
        k = len(nodes)
        for i in range(k):
            for j in range(i + 1, k):
                if nodes[j] not in adj[nodes[i]]:
                    return False
        return True
    
    # Find a maximal clique greedily
    clique = []
    candidates = set(range(n))
    while candidates:
        # Pick node with max degree in remaining candidates
        u = max(candidates, key=lambda x: len(adj[x] & candidates))
        clique.append(u)
        candidates = candidates & adj[u]
    
    # Try all 3-colorings of the clique
    k = len(clique)
    for mask in range(3 ** k):
        colors = []
        temp = mask
        valid_clique = True
        for i in range(k):
            c = temp % 3
            temp //= 3
            colors.append(c)
        
        # Check if clique coloring is valid
        for i in range(k):
            for j in range(i + 1, k):
                if colors[i] == colors[j]:
                    valid_clique = False
                    break
            if not valid_clique:
                break
        
        if not valid_clique:
            continue
        
        # Assign clique colors
        for i, u in enumerate(clique):
            assignment[u] = colors[i] + 1
        
        # Color remaining vertices
        success = True
        for u in range(n):
            if assignment[u] != 0:
                continue
            
            forbidden = set()
            for v in adj[u]:
                if assignment[v] != 0:
                    forbidden.add(assignment[v])
            
            found = False
            for c in range(1, 4):
                if c not in forbidden:
                    assignment[u] = c
                    found = True
                    break
            
            if not found:
                success = False
                break
        
        if success:
            print("YES")
            print(' '.join(map(str, assignment)))
            return
    
    print("NO")

if __name__ == "__main__":
    solve()
