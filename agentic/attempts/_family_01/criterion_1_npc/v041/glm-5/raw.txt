```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    def backtrack(pos):
        if pos == n:
            return True
        for c in (1, 2, 3):
            ok = True
            for nb in adj[pos]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[pos] = c
                if backtrack(pos + 1):
                    return True
                color[pos] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(str(x) for x in color))
    else:
        print("NO")

solve()
```
