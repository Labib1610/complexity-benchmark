```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    color = [0] * n
    order = sorted(range(n), key=lambda x: -bin(adj[x]).count('1'))

    def backtrack(pos):
        if pos == n:
            return True
        v = order[pos]
        used = 0
        av = adj[v]
        while av:
            b = av & -av
            u = b.bit_length() - 1
            used |= 1 << color[u]
            av ^= b
        for c in range(1, 4):
            if not (used & (1 << c)):
                color[v] = c
                if backtrack(pos + 1):
                    return True
        color[v] = 0
        return False

    if backtrack(0):
        out = ['YES']
        out.append(' '.join(str(color[i]) for i in range(n)))
        sys.stdout.write('\n'.join(out) + '\n')
    else:
        sys.stdout.write('NO\n')

solve()
```
