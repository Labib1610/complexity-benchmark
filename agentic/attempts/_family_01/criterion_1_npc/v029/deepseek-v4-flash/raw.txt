        adj[v] |= 1 << u

    colors = [-1] * n
    # Try all colorings, with pruning
    def dfs(node):
        if node == n:
            return True
        # Try each color
        for c in range(1, 4):
            ok = True
            mask = adj[node]
            # Check conflict
            for v in range(n):
                if (mask >> v) & 1 and colors[v] == c:
                    ok = False
                    break
            if ok:
                colors[node] = c
                if dfs(node + 1):
                    return True
                colors[node] = -1
        return False
