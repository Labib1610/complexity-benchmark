import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Build complement graph (edges between non-adjacent pairs)
    comp = [0] * (n + 1)
    full = (1 << (n + 1)) - 1
    mask_all = (1 << (n + 1)) - 1
    for i in range(1, n + 1):
        comp[i] = mask_all ^ adj[i] ^ (1 << i)
        # keep only vertices 1..n
        comp[i] &= (1 << (n + 1)) - 1
        comp[i] &= ~1  # remove bit 0
    
    # Find all maximal cliques in complement graph using Bron-Kerbosch with pivoting
    # We need to find if there's a clique cover of size <= 3
    # Equivalently, check if complement graph can be partitioned into <=3 cliques
    # which is same as original graph being 3-colorable
    
    # Use Bron-Kerbosch to enumerate maximal cliques, but we just need 3-coloring
    # Let's do direct 3-coloring via backtracking with bit operations
    
    # Order vertices by degree descending for better pruning
    vertices = list(range(1, n + 1))
    vertices.sort(key=lambda x: -bin(adj[x]).count('1'))
    
    color = [0] * (n + 1)
    
    # For each vertex, track which colors are forbidden by neighbors
    # Use bitmask for available colors
    
    def backtrack(idx):
        if idx == n:
            return True
        v = vertices[idx]
        # Determine forbidden colors
        forbidden = 0
        av = adj[v]
        while av:
            b = av & -av
            u = b.bit_length() - 1
            if color[u]:
                forbidden |= 1 << color[u]
            av ^= b
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            if not (forbidden & (1 << c)):
                color[v] = c
                if backtrack(idx + 1):
                    return True
                color[v] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(' '.join(str(color[i]) for i in range(1, n + 1)))
    else:
        print("NO")

solve()
