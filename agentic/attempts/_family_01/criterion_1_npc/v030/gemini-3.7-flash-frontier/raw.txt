import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    degree = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            if not ((adj[u] >> v) & 1):
                adj[u] |= (1 << v)
                adj[v] |= (1 << u)
                degree[u] += 1
                degree[v] += 1

    # Connected components
    color = [0] * n
    
    # Simple backtracking with MRV
    # domains: mask of available colors (1: 1, 2: 2, 4: 4)
    # Available colors: 1 (bit 0), 2 (bit 1), 3 (bit 2) -> mask: 7
    domains = [7] * n
    
    # Pre-calculate bit counts for masks 0..7
    popcnt = [0, 1, 1, 2, 1, 2, 2, 3]

    def backtrack(assigned_count):
        if assigned_count == n:
            return True
        
        # Find unassigned vertex with minimum domain size
        best_v = -1
        min_dom = 4
        max_deg = -1
        
        for v in range(n):
            if color[v] == 0:
                cnt = popcnt[domains[v]]
                if cnt < min_dom or (cnt == min_dom and degree[v] > max_deg):
                    min_dom = cnt
                    max_deg = degree[v]
                    best_v = v
                    if min_dom == 1:
                        break
        
        if min_dom == 0:
            return False
        
        v = best_v
        dom = domains[v]
        
        # Try colors for vertex v
        for c in (1, 2, 3):
            c_bit = 1 << (c - 1)
            if dom & c_bit:
                color[v] = c
                # Update neighbors
                changed_neighbors = []
                possible = True
                
                # Iterate over neighbors
                nbrs = adj[v]
                while nbrs > 0:
                    u = (nbrs & -nbrs).bit_length() - 1
                    nbrs &= nbrs - 1
                    
                    if color[u] == 0:
                        if domains[u] & c_bit:
                            domains[u] ^= c_bit
                            changed_neighbors.append(u)
                            if domains[u] == 0:
                                possible = False
                
                if possible and backtrack(assigned_count + 1):
                    return True
                
                # Undo changes
                for u in changed_neighbors:
                    domains[u] |= c_bit
                color[v] = 0
                
        return False

    if backtrack(0):
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
