```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    for i in range(m):
        u = int(input_data[2 + 2 * i]) - 1
        v = int(input_data[3 + 2 * i]) - 1
        adj[u].append(v)
        adj[v].append(u)
        
    # 0: unassigned, 1: tier 1, 2: tier 2, 3: tier 3
    color = [0] * n
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                if not dfs(v, 4 - c):
                    return False
            elif color[v] == c:
                return False
        return True

    for i in range(n):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
                
    print("YES")
    print(" ".join(str(x) for x in color))

if __name__ == '__main__':
    solve()
```
