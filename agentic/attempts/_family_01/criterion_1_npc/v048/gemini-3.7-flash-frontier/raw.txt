import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)
            
    color = [-1] * n
    avail = [7] * n  # bitmask of available colors (1, 2, 4)
    
    # Precompute popcounts
    popcnt = [0] * 8
    for i in range(8):
        popcnt[i] = bin(i).count('1')
        
    degree = [bin(adj[i]).count('1') for i in range(n)]
    
    def backtrack(uncolored_count):
        if uncolored_count == 0:
            return True
        
        # Pick uncolored vertex with minimum available colors (MRV), tie-break with max degree
        best_u = -1
        best_avail_cnt = 4
        best_deg = -1
        
        for u in range(n):
            if color[u] == -1:
                cnt = popcnt[avail[u]]
                if cnt < best_avail_cnt or (cnt == best_avail_cnt and degree[u] > best_deg):
                    best_avail_cnt = cnt
                    best_u = u
                    best_deg = degree[u]
                    if cnt <= 1:
                        break
                        
        if best_avail_cnt == 0:
            return False
            
        u = best_u
        u_avail = avail[u]
        
        for c in range(3):
            if (u_avail & (1 << c)):
                color[u] = c
                # Forward checking
                changed = []
                c_mask = ~(1 << c)
                neighbors = adj[u]
                
                # Check neighbors
                possible = True
                curr_adj = neighbors
                while curr_adj:
                    lsb = curr_adj & -curr_adj
                    v = (lsb.bit_length() - 1)
                    curr_adj ^= lsb
                    if color[v] == -1:
                        if avail[v] & (1 << c):
                            avail[v] &= c_mask
                            changed.append(v)
                            if avail[v] == 0:
                                possible = False
                                break
                
                if possible and backtrack(uncolored_count - 1):
                    return True
                
                # Revert
                for v in changed:
                    avail[v] |= (1 << c)
                color[u] = -1
                
        return False

    if backtrack(n):
        print("YES")
        print(*(color[i] + 1 for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
