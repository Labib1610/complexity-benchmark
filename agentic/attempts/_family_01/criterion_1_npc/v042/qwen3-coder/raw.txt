import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list representation of conflicts
    adj = defaultdict(set)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Try to 3-color the graph
    color = [0] * (n + 1)  # color[i] represents the ring assigned to service i
    
    def is_safe(node, c):
        """Check if we can assign color c to node"""
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        """Try to color nodes from 'node' onwards"""
        if node > n:
            return True
            
        # Skip already colored nodes (this shouldn't happen in our logic but just in case)
        if color[node] != 0:
            return backtrack(node + 1)
            
        for c in range(1, 4):  # Try colors 1, 2, 3
            if is_safe(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0  # Backtrack
                
        return False
    
    # Since the graph might not be connected, we need to handle each component
    # We'll do a more systematic approach: try coloring the entire graph
    
    # Start with node 1 and proceed sequentially
    def dfs(node):
        if node > n:
            return True
            
        for c in range(1, 4):
            if is_safe(node, c):
                color[node] = c
                if dfs(node + 1):
                    return True
                color[node] = 0
        return False
    
    if dfs(1):
        print("YES")
        print(" ".join(map(str, color[1:])))
    else:
        print("NO")

solve()
