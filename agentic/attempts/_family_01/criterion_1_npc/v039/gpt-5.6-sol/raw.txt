```python
import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n
    p = 2
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u != v:
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    colors = [-1] * n

    # Remove vertices of degree at most 2. They can always be colored
    # after the remaining graph has been 3-colored.
    alive = (1 << n) - 1
    degree = [adj[v].bit_count() for v in range(n)]
    queue = deque(v for v in range(n) if degree[v] <= 2)
    removed = []

    while queue:
        v = queue.popleft()
        bv = 1 << v
        if not (alive & bv) or degree[v] > 2:
            continue
        alive ^= bv
        removed.append(v)

        x = adj[v] & alive
        while x:
            b = x & -x
            u = b.bit_length() - 1
            x ^= b
            degree[u] -= 1
            if degree[u] == 2:
                queue.append(u)

    def bipartite_color(component):
        remaining = component
        while remaining:
            b = remaining & -remaining
            start = b.bit_length() - 1
            colors[start] = 0
            stack = [start]
            remaining ^= b

            while stack:
                v = stack.pop()
                x = adj[v] & component
                while x:
                    bu = x & -x
                    u = bu.bit_length() - 1
                    x ^= bu
                    if colors[u] == -1:
                        colors[u] = colors[v] ^ 1
                        remaining &= ~bu
                        stack.append(u)
                    elif colors[u] == colors[v]:
                        for w in range(n):
                            if component >> w & 1:
                                colors[w] = -1
                        return False
        return True

    def solve_component(component):
        # Bipartite components need only two colors.
        if bipartite_color(component):
            return True

        # Restore the component to uncolored after the failed bipartite test.
        x = component
        vertices = []
        while x:
            b = x & -x
            v = b.bit_length() - 1
            x ^= b
            vertices.append(v)
            colors[v] = -1

        comp_degree = {v: (adj[v] & component).bit_count() for v in vertices}

        # Fix a triangle to colors 0,1,2 when possible. This removes all
        # color-permutation symmetry and gives strong initial constraints.
        best_triangle = None
        best_score = -1
        for u in vertices:
            candidates_v = adj[u] & component & ~((1 << (u + 1)) - 1)
            while candidates_v:
                bv = candidates_v & -candidates_v
                v = bv.bit_length() - 1
                candidates_v ^= bv
                common = adj[u] & adj[v] & component & ~((1 << (v + 1)) - 1)
                while common:
                    bw = common & -common
                    w = bw.bit_length() - 1
                    common ^= bw
                    score = comp_degree[u] + comp_degree[v] + comp_degree[w]
                    if score > best_score:
                        best_score = score
                        best_triangle = (u, v, w)

        color_sets = [0, 0, 0]
        uncolored = component

        if best_triangle is not None:
            for c, v in enumerate(best_triangle):
                colors[v] = c
                color_sets[c] |= 1 << v
                uncolored &= ~(1 << v)
        else:
            # The component is non-bipartite, hence it has an edge.
            best_edge = None
            best_score = -1
            for u in vertices:
                x = adj[u] & component & ~((1 << (u + 1)) - 1)
                while x:
                    b = x & -x
                    v = b.bit_length() - 1
                    x ^= b
                    score = comp_degree[u] + comp_degree[v]
                    if score > best_score:
                        best_score = score
                        best_edge = (u, v)

            u, v = best_edge
            colors[u] = 0
            colors[v] = 1
            color_sets[0] |= 1 << u
            color_sets[1] |= 1 << v
            uncolored &= ~(1 << u)
            uncolored &= ~(1 << v)

        def dfs(mask):
            if mask == 0:
                return True

            best_v = -1
            best_forbidden = 0
            best_key = (-1, -1, -1)

            x = mask
            while x:
                b = x & -x
                v = b.bit_length() - 1
                x ^= b

                av = adj[v]
                forbidden = (
                    (1 if av & color_sets[0] else 0) |
                    (2 if av & color_sets[1] else 0) |
                    (4 if av & color_sets[2] else 0)
                )
                if forbidden == 7:
                    return False

                key = (
                    forbidden.bit_count(),
                    (av & mask).bit_count(),
                    comp_degree[v]
                )
                if key > best_key:
                    best_key = key
                    best_v = v
                    best_forbidden = forbidden

            available = [c for c in range(3) if not (best_forbidden >> c) & 1]
            next_mask = mask & ~(1 << best_v)

            if len(available) > 1:
                neighbor_mask = adj[best_v] & next_mask

                def impact(c):
                    count = 0
                    y = neighbor_mask
                    while y:
                        b = y & -y
                        w = b.bit_length() - 1
                        y ^= b
                        if not (adj[w] & color_sets[c]):
                            count += 1
                    return count

                available.sort(key=lambda c: (impact(c), -color_sets[c].bit_count()))

            bit_v = 1 << best_v
            for c in available:
                colors[best_v] = c
                color_sets[c] |= bit_v
                if dfs(next_mask):
                    return True
                color_sets[c] ^= bit_v
                colors[best_v] = -1

            return False

        return dfs(uncolored)

    # Solve every connected component of the remaining 3-core independently.
    remaining_core = alive
    while remaining_core:
        start_bit = remaining_core & -remaining_core
        component = 0
        frontier = start_bit
        remaining_core ^= start_bit

        while frontier:
            b = frontier & -frontier
            frontier ^= b
            v = b.bit_length() - 1
            component |= b

            new_vertices = adj[v] & remaining_core
            remaining_core &= ~new_vertices
            frontier |= new_vertices

        if not solve_component(component):
            print("NO")
            return

    # Add peeled vertices back in reverse order.
    for v in reversed(removed):
        forbidden = 0
        x = adj[v]
        while x:
            b = x & -x
            u = b.bit_length() - 1
            x ^= b
            if colors[u] != -1:
                forbidden |= 1 << colors[u]

        for c in range(3):
            if not (forbidden >> c) & 1:
                colors[v] = c
                break

    print("YES")
    print(*[c + 1 for c in colors])

if __name__ == "__main__":
    main()
```
