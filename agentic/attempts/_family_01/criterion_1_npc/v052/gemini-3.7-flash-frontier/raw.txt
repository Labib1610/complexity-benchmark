import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            
    # Remove duplicate edges
    for i in range(n):
        adj[i] = list(set(adj[i]))
        
    color = [-1] * n
    # domains[i] is a bitmask of available colors (0, 1, 2) -> bits 0, 1, 2
    domain = [7] * n  # 7 is binary 111 (colors 0, 1, 2 available)
    
    # Precompute bit count for small integers
    popcount = [0, 1, 1, 2, 1, 2, 2, 3]
    
    degrees = [len(adj[i]) for i in range(n)]
    
    def backtrack(assigned_count):
        if assigned_count == n:
            return True
        
        # Select unassigned vertex with MRV, breaking ties by largest degree
        best_v = -1
        min_dom_size = 4
        max_deg = -1
        
        for i in range(n):
            if color[i] == -1:
                sz = popcount[domain[i]]
                if sz < min_dom_size:
                    min_dom_size = sz
                    best_v = i
                    max_deg = degrees[i]
                    if sz == 0:
                        return False
                    if sz == 1:
                        break
                elif sz == min_dom_size:
                    if degrees[i] > max_deg:
                        best_v = i
                        max_deg = degrees[i]
                        
        if min_dom_size == 0 or best_v == -1:
            return False
            
        v = best_v
        dom = domain[v]
        
        for c in range(3):
            if (dom & (1 << c)):
                color[v] = c
                # Forward checking
                changed = []
                possible = True
                mask = ~(1 << c)
                for u in adj[v]:
                    if color[u] == -1:
                        if domain[u] & (1 << c):
                            domain[u] &= mask
                            changed.append(u)
                            if domain[u] == 0:
                                possible = False
                
                if possible:
                    if backtrack(assigned_count + 1):
                        return True
                
                # Undo domain changes
                for u in changed:
                    domain[u] |= (1 << c)
                color[v] = -1
                
        return False

    if backtrack(0):
        print("YES")
        print(*(c + 1 for c in color))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
