import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n
    invalid = False

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u == v:
            invalid = True
        else:
            adj[u] |= 1 << v
            adj[v] |= 1 << u

    if invalid:
        print("NO")
        return

    all_mask = (1 << n) - 1
    compatible = [
        all_mask & ~(adj[v] | (1 << v))
        for v in range(n)
    ]

    def bipartition(mask):
        left = 0
        right = 0
        unseen = mask

        while unseen:
            root = unseen & -unseen
            unseen ^= root
            left |= root
            frontier = root
            side = 0

            while frontier:
                neighbors = 0
                current = frontier
                while current:
                    bit = current & -current
                    current ^= bit
                    v = bit.bit_length() - 1
                    neighbors |= adj[v]

                neighbors &= mask

                if side == 0:
                    if neighbors & left:
                        return None
                    new_frontier = neighbors & unseen
                    right |= new_frontier
                else:
                    if neighbors & right:
                        return None
                    new_frontier = neighbors & unseen
                    left |= new_frontier

                unseen &= ~new_frontier
                frontier = new_frontier
                side ^= 1

        return left, right

    whole_bipartition = bipartition(all_mask)
    if whole_bipartition is not None:
        left, right = whole_bipartition
        colors = [2 if (left >> i) & 1 else 3 for i in range(n)]
        print("YES")
        print(*colors)
        return

    answer = None

    def search(independent, candidates, excluded):
        nonlocal answer

        forced_remainder = all_mask & ~(independent | candidates)
        parts = bipartition(forced_remainder)
        if parts is None:
            return False

        if candidates == 0:
            if excluded == 0:
                answer = (independent, parts[0], parts[1])
                return True
            return False

        union = candidates | excluded
        pivot = -1
        best = -1
        temp = union
        while temp:
            bit = temp & -temp
            temp ^= bit
            v = bit.bit_length() - 1
            score = (candidates & compatible[v]).bit_count()
            if score > best:
                best = score
                pivot = v

        branch = candidates & ~compatible[pivot] & all_mask

        while branch:
            bit = branch & -branch
            branch ^= bit
            v = bit.bit_length() - 1

            if search(
                independent | bit,
                candidates & compatible[v],
                excluded & compatible[v]
            ):
                return True

            candidates ^= bit
            excluded |= bit

        return False

    search(0, all_mask, 0)

    if answer is None:
        print("NO")
        return

    independent, left, right = answer
    colors = [0] * n
    for i in range(n):
        bit = 1 << i
        if independent & bit:
            colors[i] = 1
        elif left & bit:
            colors[i] = 2
        else:
            colors[i] = 3

    print("YES")
    print(*colors)

if __name__ == "__main__":
    main()
