import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Find connected components
    seen = [False] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if not seen[i]:
            comp = []
            stack = [i]
            seen[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                mask = adj[u]
                while mask:
                    v = (mask & -mask).bit_length() - 1
                    if not seen[v]:
                        seen[v] = True
                        stack.append(v)
                    mask &= mask - 1
            components.append(comp)

    color = [0] * (n + 1)
    comp_info = []  # (comp, bipartite, side0, side1)
    for comp in components:
        start = comp[0]
        color[start] = 1
        stack = [start]
        is_bip = True
        side0 = []
        side1 = []
        while stack:
            u = stack.pop()
            if color[u] == 1:
                side0.append(u)
            else:
                side1.append(u)
            mask = adj[u]
            while mask:
                v = (mask & -mask).bit_length() - 1
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    stack.append(v)
                elif color[v] == color[u]:
                    is_bip = False
                mask &= mask - 1
        comp_info.append((comp, is_bip, side0, side1))

    # Check if any component is not bipartite
    for _, is_bip, _, _ in comp_info:
        if not is_bip:
            print("NO")
            return

    # DP to split components into two groups (slots 1 and 2 vs slot 3)
    # such that no two incompatible ingredients are in the same group.
    # Incompatible ingredients are in different sides of their component,
    # so they are in different groups iff their sides are assigned to different groups.
    # We need: for every edge (u,v), side(u) and side(v) are in different groups.
    # This is automatically satisfied if we assign side0->group A, side1->group B
    # OR side0->group B, side1->group A for each component.
    # But we also need to ensure that across components, there are no edges
    # between ingredients in the same group. Since components are disconnected,
    # there are no edges across components. So any assignment works.
    # Wait, the problem says "no two incompatible ingredients are assigned to the same slot".
    # If we put side0 of comp1 in slot 1 and side0 of comp2 in slot 1, and there is no edge
    # between comp1 and comp2, that's fine.
    # So we just need to 2-color each component and then assign the two colors to two slots,
    # and the third slot is unused? No, we can use all three slots.
    # Actually, we can assign each component's two sides to any two of the three slots,
    # as long as the two sides get different slots. But we need to ensure that across
    # components, ingredients in the same slot are not incompatible. Since there are no
    # edges across components, any assignment of slots to sides works.
    # So we can just assign side0 -> 1, side1 -> 2 for all components, and slot 3 is unused.
    # But wait, the problem allows using all three slots. The above is a valid solution.
    # However, we need to check if the graph is 3-colorable. A graph is 3-colorable iff
    # it is a subgraph of a complete 3-partite graph. But the condition here is weaker:
    # we just need to assign each vertex to one of 3 slots such that adjacent vertices
    # get different slots. This is exactly 3-coloring.
    # The above reasoning is flawed: components are disconnected, so we can color each
    # component independently. A bipartite component can be 2-colored, which is a valid
    # 3-coloring. A non-bipartite component might still be 3-colorable (e.g., a triangle).
    # So we need to actually 3-color the graph.
    # Since n <= 40, we can use backtracking with bitmasks or meet-in-the-middle.
    # But the graph might not be bipartite. Let's do a proper 3-coloring search.
    # Actually, the problem is equivalent to 3-coloring. Let's implement a backtracking
    # 3-coloring with some pruning.
    
    # Reset colors
    color = [0] * (n + 1)
    
    # Order vertices by degree for better pruning
    vertices = list(range(1, n + 1))
    vertices.sort(key=lambda x: bin(adj[x]).count('1'), reverse=True)
    
    # Map vertex to index in vertices for faster access
    # We'll use recursive backtracking with bitmasks for available colors
    
    # Precompute adjacency as sets for faster iteration
    adj_set = [set() for _ in range(n + 1)]
    for u in range(1, n + 1):
        mask = adj[u]
        while mask:
            v = (mask & -mask).bit_length() - 1
            adj_set[u].add(v)
            mask &= mask - 1
    
    # Use iterative deepening or simple backtracking
    # Since n <= 40, and 3 colors, worst case 3^40, but with pruning it should be fast enough
    # for typical cases. However, worst case might be too slow.
    # Let's use a more efficient approach: for each vertex, try colors not used by neighbors.
    
    # We'll use a recursive function with memoization or just backtracking.
    # To speed up, we can use bitmasks for colors used by neighbors.
    
    # Let's try a simple backtracking with forward checking.
    
    color = [0] * (n + 1)
    
    def backtrack(idx):
        if idx == len(vertices):
            return True
        u = vertices[idx]
        used = 0
        for v in adj_set[u]:
            if color[v]:
                used |= 1 << color[v]
        for c in range(1, 4):
            if not (used & (1 << c)):
                color[u] = c
                if backtrack(idx + 1):
                    return True
                color[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(' '.join(str(color[i]) for i in range(1, n + 1)))
    else:
        print("NO")

solve()
