import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # order vertices by degree descending for initial heuristic
    order = sorted(range(n), key=lambda x: -adj[x].bit_count())
    pos_in_order = [0] * n
    for i, v in enumerate(order):
        pos_in_order[v] = i

    colors = [-1] * n
    found = False

    # precompute neighbor list for fast iteration
    neigh = [ [j for j in range(n) if (adj[i] >> j) & 1] for i in range(n) ]

    def dfs(idx: int):
        nonlocal found
        if found:
            return
        if idx == n:
            found = True
            return
        # select next vertex: the uncolored one with most colored neighbors (MRV)
        best = -1
        best_score = -1
        for i in range(n):
            if colors[i] == -1:
                cnt = 0
                for nb in neigh[i]:
                    if colors[nb] != -1:
                        cnt += 1
                if cnt > best_score:
                    best_score = cnt
                    best = i
        v = best
        # compute forbidden colors from already colored neighbors
        forbid = 0
        for nb in neigh[v]:
            c = colors[nb]
            if c != -1:
                forbid |= 1 << c
        for c in range(3):
            if not (forbid >> c) & 1:
                colors[v] = c
                dfs(idx + 1)
                if found:
                    return
                colors[v] = -1

    dfs(0)

    if not found:
        print("NO")
    else:
        print("YES")
        # output colors as 1..3 in original order (ingredients 1..n)
        print(' '.join(str(c + 1) for c in colors))

if __name__ == "__main__":
    solve()
