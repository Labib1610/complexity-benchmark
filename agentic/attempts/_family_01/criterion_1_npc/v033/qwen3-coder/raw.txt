def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(set)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].add(v)
        adj[v].add(u)
    
    # Try to 3-color the graph
    color = [0] * (n + 1)  # 0 means uncolored, 1,2,3 are the colors
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def dfs(node):
        if node > n:
            return True
            
        for c in range(1, 4):
            if is_valid(node, c):
                color[node] = c
                if dfs(node + 1):
                    return True
                color[node] = 0
        
        return False
    
    # Since n <= 40 and we're doing 3-coloring, this might be too slow
    # Let's use a more efficient approach with backtracking optimization
    
    def solve_coloring():
        # Use a more systematic approach
        uncolored = list(range(1, n + 1))
        
        def backtrack(index):
            if index == len(uncolored):
                return True
                
            node = uncolored[index]
            for c in range(1, 4):
                valid = True
                for neighbor in adj[node]:
                    if color[neighbor] == c:
                        valid = False
                        break
                if valid:
                    color[node] = c
                    if backtrack(index + 1):
                        return True
                    color[node] = 0
            return False
            
        return backtrack(0)
    
    if solve_coloring():
        print("YES")
        print(" ".join(str(color[i]) for i in range(1, n + 1)))
    else:
        print("NO")

solve()
