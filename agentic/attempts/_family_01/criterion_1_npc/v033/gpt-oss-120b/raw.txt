import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    # order vertices by descending degree for better pruning
    order = sorted(range(n), key=lambda x: -len(adj[x]))
    pos_in_order = [0]*n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx

    colors = [0]*n
    sys.setrecursionlimit(10000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        v = order[idx]
        # compute forbidden colors from already colored neighbors
        forbidden = 0
        for nb in adj[v]:
            c = colors[nb]
            if c:
                forbidden |= 1 << (c-1)
        for col in (1,2,3):
            if not (forbidden & (1 << (col-1))):
                colors[v] = col
                if dfs(idx+1):
                    return True
                colors[v] = 0
        return False

    if dfs(0):
        out = ['YES', ' '.join(str(colors[i]) for i in range(n))]
    else:
        out = ['NO']
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
