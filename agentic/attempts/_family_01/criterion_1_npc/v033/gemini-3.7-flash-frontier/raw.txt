import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)
            
    # Check self-loops
    for i in range(n):
        # Already handled by u != v, but if input had u == u, 3-coloring impossible
        pass

    # Find connected components
    visited = [False] * n
    components = []
    for i in range(n):
        if not visited[i]:
            comp = []
            queue = [i]
            visited[i] = True
            for node in queue:
                comp.append(node)
                for neighbor in range(n):
                    if (adj[node] & (1 << neighbor)) and not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
            components.append(comp)

    res_color = [0] * n

    for comp in components:
        # Solve 3-coloring for this component
        # avail[u] is a bitmask of available colors from {1, 2, 4}
        avail = {u: 7 for u in comp}
        assigned = {}

        def backtrack():
            if len(assigned) == len(comp):
                return True
            
            # Find unassigned node with minimum available colors (MRV)
            # Tie breaker: maximum unassigned degree
            best_u = -1
            min_c = 4
            max_deg = -1
            
            for u in comp:
                if u not in assigned:
                    c = bin(avail[u]).count('1')
                    if c < min_c:
                        min_c = c
                        best_u = u
                        # compute unassigned degree
                        deg = bin(adj[u] & ~sum(1 << x for x in assigned)).count('1')
                        max_deg = deg
                    elif c == min_c:
                        deg = bin(adj[u] & ~sum(1 << x for x in assigned)).count('1')
                        if deg > max_deg:
                            best_u = u
                            max_deg = deg
            
            if min_c == 0:
                return False
            
            u = best_u
            mask = avail[u]
            
            for color_bit in (1, 2, 4):
                if mask & color_bit:
                    assigned[u] = color_bit
                    # Update neighbors
                    changed = []
                    valid = True
                    for v in comp:
                        if v not in assigned and (adj[u] & (1 << v)):
                            if avail[v] & color_bit:
                                avail[v] ^= color_bit
                                changed.append(v)
                                if avail[v] == 0:
                                    valid = False
                    
                    if valid and backtrack():
                        return True
                    
                    # Undo
                    for v in changed:
                        avail[v] |= color_bit
                    del assigned[u]
                    
            return False

        if not backtrack():
            print("NO")
            return

        for u, col in assigned.items():
            if col == 1:
                res_color[u] = 1
            elif col == 2:
                res_color[u] = 2
            elif col == 4:
                res_color[u] = 3

    print("YES")
    print(*(res_color[i] for i in range(n)))

if __name__ == '__main__':
    solve()
