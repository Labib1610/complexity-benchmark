import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    adj_mask = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v and not (adj_mask[u] & (1 << v)):
            adj[u].append(v)
            adj[v].append(u)
            adj_mask[u] |= (1 << v)
            adj_mask[v] |= (1 << u)

    # If self-loop, impossible (handled if input can have it, though usually simple graph)
    # Check if any vertex has a self loop:
    # (The problem statement says distinct pairs, but let's be careful)

    color = [0] * n
    # Available colors mask for each vertex: bit 0 -> color 1, bit 1 -> color 2, bit 2 -> color 3
    avail = [7] * n
    
    # Precompute degrees
    deg = [len(adj[i]) for i in range(n)]

    def get_best_vertex():
        best_v = -1
        min_choices = 4
        max_deg = -1
        for i in range(n):
            if color[i] == 0:
                choices = (avail[i] & 1) + ((avail[i] >> 1) & 1) + ((avail[i] >> 2) & 1)
                if choices < min_choices:
                    min_choices = choices
                    max_deg = deg[i]
                    best_v = i
                elif choices == min_choices:
                    if deg[i] > max_deg:
                        max_deg = deg[i]
                        best_v = i
        return best_v, min_choices

    def backtrack(assigned_count):
        if assigned_count == n:
            return True
        
        u, num_choices = get_best_vertex()
        if num_choices == 0:
            return False
        
        u_avail = avail[u]
        for c in (1, 2, 3):
            c_bit = 1 << (c - 1)
            if u_avail & c_bit:
                color[u] = c
                changed = []
                for v in adj[u]:
                    if color[v] == 0 and (avail[v] & c_bit):
                        avail[v] &= ~c_bit
                        changed.append(v)
                
                if backtrack(assigned_count + 1):
                    return True
                
                # Undo
                for v in changed:
                    avail[v] |= c_bit
                color[u] = 0

        return False

    if backtrack(0):
        print("YES")
        print(*(color[i] for i in range(n)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
