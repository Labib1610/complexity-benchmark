import sys

sys.setrecursionlimit(1_000_000)


def solve_component(vertices, global_adj, answer):
    vertices.sort(
        key=lambda x: (global_adj[x] & sum(1 << v for v in vertices)).bit_count(),
        reverse=True
    )
    k = len(vertices)
    index = {v: i for i, v in enumerate(vertices)}
    all_mask = (1 << k) - 1

    adj = [0] * k
    for i, v in enumerate(vertices):
        mask = global_adj[v]
        local = 0
        for u in vertices:
            if mask & (1 << u):
                local |= 1 << index[u]
        adj[i] = local

    def bipartite(mask):
        unseen = mask
        color0 = 0
        color1 = 0

        while unseen:
            root = unseen & -unseen
            unseen ^= root
            color0 |= root
            frontier = root
            side = 0

            while frontier:
                neighbors = 0
                cur = frontier
                while cur:
                    bit = cur & -cur
                    cur ^= bit
                    neighbors |= adj[bit.bit_length() - 1]

                same_color = color0 if side == 0 else color1
                if neighbors & same_color:
                    return None

                new_frontier = neighbors & unseen
                unseen &= ~new_frontier
                if side == 0:
                    color1 |= new_frontier
                else:
                    color0 |= new_frontier

                frontier = new_frontier
                side ^= 1

        return color0, color1

    initial = bipartite(all_mask)
    if initial is not None:
        c0, c1 = initial
        for i, v in enumerate(vertices):
            answer[v] = 2 if c0 & (1 << i) else 3
        return True

    complement = [
        all_mask ^ (1 << i) ^ adj[i]
        for i in range(k)
    ]

    found_independent = 0
    found_bipartition = None

    def bron_kerbosch(independent, possible, excluded):
        nonlocal found_independent, found_bipartition

        if possible == 0:
            if excluded == 0:
                partition = bipartite(all_mask ^ independent)
                if partition is not None:
                    found_independent = independent
                    found_bipartition = partition
                    return True
            return False

        union = possible | excluded
        pivot = -1
        best_count = -1
        cur = union
        while cur:
            bit = cur & -cur
            cur ^= bit
            v = bit.bit_length() - 1
            count = (possible & complement[v]).bit_count()
            if count > best_count:
                best_count = count
                pivot = v

        candidates = possible if pivot == -1 else possible & ~complement[pivot]

        while candidates:
            bit = candidates & -candidates
            candidates ^= bit
            v = bit.bit_length() - 1

            if bron_kerbosch(
                independent | bit,
                possible & complement[v],
                excluded & complement[v]
            ):
                return True

            possible ^= bit
            excluded |= bit

        return False

    if not bron_kerbosch(0, all_mask, 0):
        return False

    c0, c1 = found_bipartition
    for i, v in enumerate(vertices):
        bit = 1 << i
        if found_independent & bit:
            answer[v] = 1
        elif c0 & bit:
            answer[v] = 2
        else:
            answer[v] = 3

    return True


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    adj = [0] * n

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if u == v:
            print("NO")
            return
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    active = (1 << n) - 1
    degree = [a.bit_count() for a in adj]
    stack = []
    queue = [i for i in range(n) if degree[i] <= 2]
    queued = [degree[i] <= 2 for i in range(n)]

    head = 0
    while head < len(queue):
        v = queue[head]
        head += 1
        bit_v = 1 << v

        if not (active & bit_v) or degree[v] > 2:
            continue

        stack.append(v)
        active ^= bit_v

        neighbors = adj[v] & active
        while neighbors:
            bit = neighbors & -neighbors
            neighbors ^= bit
            u = bit.bit_length() - 1
            degree[u] -= 1
            if degree[u] <= 2 and not queued[u]:
                queued[u] = True
                queue.append(u)

    answer = [0] * n
    remaining = active

    while remaining:
        root = remaining & -remaining
        component_mask = 0
        frontier = root

        while frontier:
            component_mask |= frontier
            next_frontier = 0
            cur = frontier
            while cur:
                bit = cur & -cur
                cur ^= bit
                v = bit.bit_length() - 1
                next_frontier |= adj[v] & active & ~component_mask
            frontier = next_frontier

        remaining &= ~component_mask
        vertices = []
        cur = component_mask
        while cur:
            bit = cur & -cur
            cur ^= bit
            vertices.append(bit.bit_length() - 1)

        if not solve_component(vertices, adj, answer):
            print("NO")
            return

    for v in reversed(stack):
        used = 0
        neighbors = adj[v]
        while neighbors:
            bit = neighbors & -neighbors
            neighbors ^= bit
            u = bit.bit_length() - 1
            if answer[u]:
                used |= 1 << (answer[u] - 1)

        for color in range(1, 4):
            if not (used & (1 << (color - 1))):
                answer[v] = color
                break

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
