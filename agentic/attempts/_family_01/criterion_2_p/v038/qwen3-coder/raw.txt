```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 means unvisited, 1 and 2 are the two colors
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS for each component
    # We need to process nodes in order to ensure lexicographically smallest solution
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if visited[start]:
            continue
            
        # BFS to check bipartiteness and assign colors
        queue = deque()
        queue.append((start, 1))  # Assign color 1 to the starting node
        color[start] = 1
        visited[start] = True
        
        while queue:
            node, c = queue.popleft()
            next_color = 3 - c  # Switches 1 <-> 2
            
            for neighbor in adj[node]:
                if not visited[neighbor]:
                    visited[neighbor] = True
                    color[neighbor] = next_color
                    queue.append((neighbor, next_color))
                elif color[neighbor] != next_color:
                    # Conflict found, not bipartite
                    print("NO")
                    return
    
    # If we reach here, the graph is bipartite
    # The coloring already ensures lexicographically smallest assignment
    # because we process nodes in order and always try to assign color 1 first
    print("YES")
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append("1")  # Isolated nodes can be assigned to environment 1
        else:
            result.append(str(color[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
