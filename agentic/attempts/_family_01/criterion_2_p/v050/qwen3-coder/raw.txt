```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build graph: for each pair (u,v), they must be in different groups
    # This is a bipartite graph check problem
    adj = defaultdict(list)
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 = unvisited, 1 = group 1, 2 = group 2
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS for all components
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
            
        queue = deque()
        queue.append(start)
        color[start] = 1  # Start with group 1
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            next_color = 2 if current_color == 1 else 1
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = next_color
                    queue.append(v)
                elif color[v] == current_color:
                    print("NO")
                    return
    
    # If we reach here, it's bipartite
    print("YES")
    # To get lexicographically smallest, we prefer group 1 over 2
    # Since we started with color 1 for each component, this is already ensured
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append('1')  # Isolated node can go to group 1
        else:
            result.append(str(color[i]))
    
    print(' '.join(result))

if __name__ == "__main__":
    main()
```
