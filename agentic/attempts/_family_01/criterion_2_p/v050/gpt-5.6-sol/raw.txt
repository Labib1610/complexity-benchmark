```python
import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    graph = [[] for _ in range(n)]

    pos = 2
    for _ in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        graph[u].append(v)
        graph[v].append(u)

    crew = [0] * n
    queue = deque()

    for start in range(n):
        if crew[start] != 0:
            continue

        crew[start] = 1
        queue.append(start)

        while queue:
            u = queue.popleft()
            expected = 3 - crew[u]

            for v in graph[u]:
                if crew[v] == 0:
                    crew[v] = expected
                    queue.append(v)
                elif crew[v] == crew[u]:
                    sys.stdout.write("NO\n")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, crew)) + "\n")

if __name__ == "__main__":
    main()
```
