We can model this as a bipartite graph coloring problem.  
If the graph is not bipartite, no valid assignment exists.  
Otherwise, we can assign values 1 and 2 to the two partitions, and to get the lexicographically smallest vector, we start with node 1 as 1 and propagate.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    from collections import deque
    
    for start in range(n):
        if color[start] != -1:
            continue
        color[start] = 0
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
    
    # Now color is 0/1, map to 1/2
    ans = [1 if c == 0 else 2 for c in color]
    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
