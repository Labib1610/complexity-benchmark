```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-indexed
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Color array: -1 unvisited, 0 color1, 1 color2
    color = [-1] * n
    result = [0] * n
    
    # Check bipartiteness for each connected component
    for start in range(n):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - current_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # If bipartite, construct the result
    for i in range(n):
        # We want lexicographically smallest: prefer 1 over 2
        # color[i] == 0 -> assign 1
        # color[i] == 1 -> assign 2
        result[i] = str(color[i] + 1)
    
    print("YES")
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
