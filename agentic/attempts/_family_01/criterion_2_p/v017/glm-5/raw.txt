import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        
        # BFS to find the connected component and check bipartiteness
        comp = []
        q = [start]
        color[start] = 1
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            comp.append(u)
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    q.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        
        # Now we have a valid bipartite component.
        # We want to assign colors to minimize the sequence lexicographically.
        # The component is connected, so there are exactly two valid colorings.
        # We pick the one that is lexicographically smaller.
        
        # Find the minimum vertex in the component
        min_v = min(comp)
        
        # If the minimum vertex is currently colored 2, flip all colors in the component
        if color[min_v] == 2:
            for u in comp:
                color[u] = 3 - color[u]
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
