```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    stack = []

    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        color[start] = 1
        stack.append(start)
        while stack:
            u = stack.pop()
            cu = color[u]
            nv = 3 - cu
            for v in adj[u]:
                cv = color[v]
                if cv == 0:
                    color[v] = nv
                    stack.append(v)
                elif cv == cu:
                    sys.stdout.write("NO\n")
                    return

    out = ["YES\n", " ".join(str(color[i]) for i in range(1, n + 1)), "\n"]
    sys.stdout.write("".join(out))

solve()
```
