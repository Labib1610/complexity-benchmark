```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        edges.append((u, v))
        index += 2

    # Build adjacency list
    adj = defaultdict(list)
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0=unvisited, 1=color1, 2=color2
    color = [0] * (n + 1)
    
    # Check bipartiteness for each connected component
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS to color the component
        queue = deque()
        queue.append(start)
        color[start] = 1  # Start with color 1
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            neighbor_color = 3 - current_color  # flips 1->2 and 2->1
            
            for neighbor in adj[node]:
                if color[neighbor] == 0:
                    color[neighbor] = neighbor_color
                    queue.append(neighbor)
                elif color[neighbor] != neighbor_color:
                    print("NO")
                    return
    
    # If we reach here, the graph is bipartite
    print("YES")
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append("1")
        else:
            result.append(str(color[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
