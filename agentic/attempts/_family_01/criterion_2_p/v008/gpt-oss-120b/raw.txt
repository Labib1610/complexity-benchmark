import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # provisional colors 0/1
    shift = [0] * (n + 1)           # final shifts 1/2
    visited = [False] * (n + 1)

    for start in range(1, n + 1):
        if visited[start]:
            continue
        # BFS this component
        q = deque([start])
        visited[start] = True
        color[start] = 0
        comp_vertices = [start]
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp_vertices.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        # decide mapping to make smallest vertex have shift 1
        min_v = min(comp_vertices)
        flip = (color[min_v] == 1)   # if its color is 1, we need to flip
        for v in comp_vertices:
            c = color[v]
            if flip:
                c ^= 1
            shift[v] = 1 if c == 0 else 2

    print("YES")
    print(' '.join(str(shift[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
