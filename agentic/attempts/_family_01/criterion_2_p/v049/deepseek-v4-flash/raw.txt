We can model this as a bipartite graph problem.  
If the graph is not bipartite, it’s impossible.  
Otherwise, we assign shifts using BFS/DFS, and to get the lexicographically smallest sequence, we try to assign shift 1 to as many early workers as possible.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
possible = True

for i in range(n):
    if color[i] == -1:
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    # Try to make lexicographically smallest by flipping components if needed
    visited = [False] * n
    for i in range(n):
        if not visited[i]:
            # BFS to collect component
            comp = []
            q = deque([i])
            visited[i] = True
            while q:
                u = q.popleft()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        q.append(v)
            # Check if flipping this component makes lexicographically smaller
            # Count how many in comp have color 0 vs 1
            # We want smallest index in comp to be 0 if possible
            # If color[i] == 1, flip the whole component
            if color[i] == 1:
                for u in comp:
                    color[u] ^= 1

    print("YES")
    print(' '.join(str(color[i] + 1) for i in range(n)))
```
