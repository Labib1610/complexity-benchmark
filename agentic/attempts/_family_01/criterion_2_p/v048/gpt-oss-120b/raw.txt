```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)   # 0 -> phase 1, 1 -> phase 2
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # start new component, assign start color 0 (phase 1) to keep lexicographically minimal
        dq = deque([start])
        color[start] = 0
        while dq:
            u = dq.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    dq.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    # all good
    print("YES")
    phases = [str(color[i] + 1) for i in range(1, n + 1)]
    print(' '.join(phases))

if __name__ == "__main__":
    solve()
```
