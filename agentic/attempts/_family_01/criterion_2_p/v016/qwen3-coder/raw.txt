import sys
from collections import defaultdict, deque

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        ans = [1] * n
        print(' '.join(map(str, ans)))
        return
    
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    color = [0] * (n + 1)  # 0: unvisited, 1: color 1, 2: color 2
    
    # Check if bipartite and get all components
    def bfs(start):
        queue = deque()
        queue.append(start)
        color[start] = 1
        cnt1 = 1
        cnt2 = 0
        nodes = [start]
        
        while queue:
            u = queue.popleft()
            c = color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - c
                    if color[v] == 1:
                        cnt1 += 1
                    else:
                        cnt2 += 1
                    queue.append(v)
                    nodes.append(v)
                elif color[v] == c:
                    return False, [], 0, 0
        return True, nodes, cnt1, cnt2
    
    components = []
    for i in range(1, n + 1):
        if color[i] == 0:
            is_bipartite, nodes, c1, c2 = bfs(i)
            if not is_bipartite:
                print("NO")
                return
            components.append((nodes, c1, c2))
    
    # For each component, we can choose which color is 1 or 2
    # To get lex smallest, we try both options for each component
    # But since we want global lex smallest, it's better to fix the first node in each component to 1
    
    res = [0] * (n + 1)
    for nodes, c1, c2 in components:
        # We can assign colors such that the first node in the component gets 1
        # This will help us achieve lexicographically smallest result
        first_node = nodes[0]
        target_color = 1
        if color[first_node] == 2:
            # Need to flip colors in this component
            for u in nodes:
                color[u] = 3 - color[u]
        
        # Now ensure first node is colored 1
        if color[first_node] != 1:
            for u in nodes:
                color[u] = 3 - color[u]
    
    # Now just build the result
    for i in range(1, n + 1):
        res[i] = color[i]
    
    print("YES")
    print(' '.join(map(str, res[1:])))

solve()
