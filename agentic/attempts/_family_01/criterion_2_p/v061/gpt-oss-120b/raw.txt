```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [ -1 ] * (n + 1)   # 0 -> shift 1, 1 -> shift 2
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # start new component, assign smallest possible (1) to smallest index node
        color[start] = 0
        dq = deque([start])
        while dq:
            u = dq.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    dq.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    # all assigned
    shifts = [str(c + 1) for c in color[1:]]
    print("YES")
    print(' '.join(shifts))

if __name__ == "__main__":
    solve()
```
