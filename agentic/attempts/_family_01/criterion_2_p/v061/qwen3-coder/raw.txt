import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color assignment: 0 means unvisited, 1 and 2 are the two colors
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS
    for start in range(1, n + 1):
        if color[start] == 0:
            queue = deque([start])
            color[start] = 1
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = 3 - color[node]  # Assign opposite color
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If we get here, a valid assignment exists
    # We want the lexicographically smallest assignment
    # That means we prefer assigning 1 to nodes with smaller indices when possible
    
    # For each component, we can choose which color represents shift 1
    # To get lexicographically smallest, we need to do additional processing
    
    # Let's re-assign colors to ensure lexicographically smallest result
    result = [0] * (n + 1)
    
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i] and adj[i]:  # Only process if node has edges
            # Find all nodes in this component
            component = []
            queue = deque([i])
            visited[i] = True
            component.append(i)
            
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
                        component.append(neighbor)
            
            # For this component, decide whether to flip colors
            # We want to minimize the result lexicographically
            # Try both assignments and pick the better one
            
            # Get current coloring for this component
            color1_nodes = [x for x in component if color[x] == 1]
            color2_nodes = [x for x in component if color[x] == 2]
            
            # Choose the assignment that gives lexicographically smaller result
            # We only care about the first position where they differ
            min_node = min(component)
            
            # If color1_nodes has the minimum node, keep as is
            # Otherwise, flip
            if min_node in color1_nodes:
                for node in color1_nodes:
                    result[node] = 1
                for node in color2_nodes:
                    result[node] = 2
            else:
                for node in color1_nodes:
                    result[node] = 2
                for node in color2_nodes:
                    result[node] = 1
    
    # Handle isolated nodes (nodes with no edges)
    for i in range(1, n + 1):
        if result[i] == 0:
            result[i] = 1  # Assign 1 to isolated nodes
    
    print("YES")
    print(" ".join(map(str, result[1:])))

solve()
