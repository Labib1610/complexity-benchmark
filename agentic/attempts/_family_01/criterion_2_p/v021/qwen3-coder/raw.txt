import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index+1])
        index += 2
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # We'll use a BFS-based approach to check bipartiteness and assign directions
    color = [0] * (n + 1)  # 0 means unvisited, 1 and 2 are the two directions
    
    for i in range(1, n + 1):
        if color[i] == 0:
            queue = deque()
            queue.append(i)
            color[i] = 1  # Start with direction 1
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 3 - current_color  # flips 1->2 and 2->1
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] != next_color:
                        print("NO")
                        return
    
    # If we reach here, a valid assignment exists
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(color[i]))
    print(' '.join(result))

if __name__ == "__main__":
    main()
