import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # 0 or 1
    comp = [-1] * (n + 1)           # component id
    comp_vertices = []              # list of lists
    comp_cnt = 0

    from collections import deque
    for v in range(1, n + 1):
        if color[v] != -1:
            continue
        # start new component
        q = deque([v])
        color[v] = 0
        comp[v] = comp_cnt
        verts = [v]
        ok = True
        while q:
            x = q.popleft()
            for nb in adj[x]:
                if color[nb] == -1:
                    color[nb] = color[x] ^ 1
                    comp[nb] = comp_cnt
                    verts.append(nb)
                    q.append(nb)
                elif color[nb] == color[x]:
                    print("NO")
                    return
        comp_vertices.append(verts)
        comp_cnt += 1

    # decide mapping for each component to get lexicographically smallest assignment
    map_to_one = [None] * comp_cnt   # which color (0/1) will be shift 1
    ans = [0] * (n + 1)

    for i in range(1, n + 1):
        cid = comp[i]
        if map_to_one[cid] is None:
            # first occurrence of this component, make its current color become 1
            map_to_one[cid] = color[i]
        # assign shift
        ans[i] = 1 if color[i] == map_to_one[cid] else 2

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
