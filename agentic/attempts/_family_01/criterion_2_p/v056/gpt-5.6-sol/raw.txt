```python
import sys
from collections import deque

def main():
    data = iter(map(int, sys.stdin.buffer.read().split()))
    n = next(data)
    m = next(data)

    graph = [[] for _ in range(n)]
    for _ in range(m):
        u = next(data) - 1
        v = next(data) - 1
        graph[u].append(v)
        graph[v].append(u)

    color = [0] * n

    for start in range(n):
        if color[start] != 0:
            continue

        color[start] = 1
        queue = deque([start])

        while queue:
            u = queue.popleft()
            opposite = 3 - color[u]

            for v in graph[u]:
                if color[v] == 0:
                    color[v] = opposite
                    queue.append(v)
                elif color[v] == color[u]:
                    sys.stdout.write("NO\n")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color)) + "\n")

if __name__ == "__main__":
    main()
```
