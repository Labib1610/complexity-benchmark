import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    # color array: 0 = uncolored, 1 = shift 1, 2 = shift 2
    color = [0] * (n + 1)
    
    possible = True
    
    # Iterate through workers 1 to n to ensure lexicographically smallest assignment
    for i in range(1, n + 1):
        if color[i] == 0:
            # Worker i is uncolored. Start a BFS/DFS from i.
            # To get lexicographically smallest valid assignment, we must color i with 1.
            # This is valid because if i could be colored 2, then swapping colors in its
            # entire connected component would yield a valid assignment with i colored 1,
            # which would be lexicographically smaller.
            color[i] = 1
            
            # BFS queue
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                
                if not possible:
                    break
        
        if not possible:
            break

    if not possible:
        print("NO")
    else:
        print("YES")
        print(*(color[1:n+1]))

if __name__ == '__main__':
    solve()
