```python
import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)          # 0 = uncolored, 1 or 2 otherwise

    for i in range(1, n + 1):
        if color[i]:
            continue
        # start new component, give smallest-index vertex color 1
        color[i] = 1
        q = deque([i])
        while q:
            v = q.popleft()
            cv = color[v]
            for nb in adj[v]:
                if color[nb] == 0:
                    color[nb] = 3 - cv   # flip 1<->2
                    q.append(nb)
                elif color[nb] == cv:
                    sys.stdout.write("NO")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, color[1:])))

if __name__ == "__main__":
    solve()
```
