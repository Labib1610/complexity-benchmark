```python
import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    graph = [[] for _ in range(n)]

    idx = 2
    for _ in range(m):
        u = data[idx] - 1
        v = data[idx + 1] - 1
        idx += 2
        graph[u].append(v)
        graph[v].append(u)

    phase = [0] * n

    for start in range(n):
        if phase[start] != 0:
            continue

        phase[start] = 1
        queue = deque([start])

        while queue:
            u = queue.popleft()
            expected = 3 - phase[u]

            for v in graph[u]:
                if phase[v] == 0:
                    phase[v] = expected
                    queue.append(v)
                elif phase[v] != expected:
                    print("NO")
                    return

    print("YES")
    print(*phase)

if __name__ == "__main__":
    main()
```
