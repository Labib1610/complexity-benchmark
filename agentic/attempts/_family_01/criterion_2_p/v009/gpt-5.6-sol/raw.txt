```python
import sys

def main():
    data = map(int, sys.stdin.buffer.read().split())
    it = iter(data)
    n = next(it)
    m = next(it)

    graph = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        graph[u].append(v)
        graph[v].append(u)

    shift = bytearray(n + 1)

    for start in range(1, n + 1):
        if shift[start]:
            continue

        shift[start] = 1
        stack = [start]

        while stack:
            u = stack.pop()
            required = 3 - shift[u]

            for v in graph[u]:
                if shift[v] == 0:
                    shift[v] = required
                    stack.append(v)
                elif shift[v] != required:
                    sys.stdout.write("NO\n")
                    return

    assignment = " ".join(map(str, shift[1:]))
    sys.stdout.write("YES\n" + assignment + "\n")

if __name__ == "__main__":
    main()
```
