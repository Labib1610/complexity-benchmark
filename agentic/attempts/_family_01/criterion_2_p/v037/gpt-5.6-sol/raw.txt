```python
import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n, m = data[0], data[1]

    graph = [[] for _ in range(n)]
    p = 2
    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        graph[u].append(v)
        graph[v].append(u)

    shift = [0] * n
    queue = deque()

    for start in range(n):
        if shift[start] != 0:
            continue

        shift[start] = 1
        queue.append(start)

        while queue:
            u = queue.popleft()
            opposite = 3 - shift[u]

            for v in graph[u]:
                if shift[v] == 0:
                    shift[v] = opposite
                    queue.append(v)
                elif shift[v] == shift[u]:
                    sys.stdout.write("NO\n")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, shift)) + "\n")

if __name__ == "__main__":
    main()
```
