import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS to check bipartiteness and assign colors
        # We want to minimize color numbers lexicographically.
        # Since we process components in increasing order of vertices,
        # and we always try to assign the smallest available color to the
        # smallest vertex in the component, this should yield the lexicographically
        # smallest valid coloring.
        # However, for a connected component, the coloring is determined up to
        # a flip once we fix the color of one vertex. To minimize lexicographically,
        # we should assign color 1 to the smallest vertex in the component.
        # But wait, the problem says: minimize employee 1's branch, then employee 2's, etc.
        # So for each component, we should assign the smallest vertex color 1.
        # But if the component is not connected to previously colored vertices,
        # we can freely choose. To get lexicographically smallest, we assign 1 to the
        # smallest vertex in the component.
        # Actually, since components are independent, we process them in order of
        # their smallest vertex. The BFS naturally starts from the smallest uncolored vertex.
        # We assign it color 1, then BFS.
        # But is this always optimal? Consider a component where the smallest vertex
        # is connected to others. Assigning it 1 forces the rest. Could assigning it 2
        # and some other vertex 1 lead to a lexicographically smaller coloring?
        # No, because the smallest vertex in the component is the first one we encounter
        # in the global ordering. Making it 1 is always better or equal to making it 2.
        # So the greedy approach of assigning 1 to the start vertex is correct.
        
        color[start] = 1
        q = deque([start])
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    q.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        if not ok:
            print("NO")
            return
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
