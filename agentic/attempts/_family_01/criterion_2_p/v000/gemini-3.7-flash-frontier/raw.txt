```python
import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        
    s = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if s[i] == 0:
            s[i] = 1
            queue = deque([i])
            while queue:
                u = queue.popleft()
                c_next = 3 - s[u]
                for v in adj[u]:
                    if s[v] == 0:
                        s[v] = c_next
                        queue.append(v)
                    elif s[v] != c_next:
                        print("NO")
                        return
                        
    print("YES")
    print(*(s[1:]))

if __name__ == '__main__':
    solve()
```
