import sys
import heapq

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    B = int(next(it))
    
    root_set = set()
    for _ in range(k):
        root_set.add(int(next(it)))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))
        edges.append((u, v, w))
    
    # Prim's algorithm to find minimum spanning forest
    # Each root server must be in a separate component
    # We grow components from each root server simultaneously
    
    in_mst = [False] * (n + 1)
    min_edge = [float('inf')] * (n + 1)
    best_edge = [(-1, -1, -1, -1)] * (n + 1)  # (u, v, w, edge_idx)
    component = [-1] * (n + 1)  # which root's component this node belongs to
    
    pq = []  # (weight, node, edge_idx, parent)
    
    # Initialize with all root servers
    root_list = list(root_set)
    for idx, r in enumerate(root_list):
        in_mst[r] = True
        component[r] = idx
        for v, w, edge_idx in adj[r]:
            if not in_mst[v]:
                if w < min_edge[v]:
                    min_edge[v] = w
                    best_edge[v] = (r, v, w, edge_idx)
                    heapq.heappush(pq, (w, v, edge_idx, r))
    
    mst_edges = []
    total_cost = 0
    
    while pq:
        w, u, edge_idx, parent = heapq.heappop(pq)
        if in_mst[u]:
            continue
        # Check if this is still the best edge for u
        if w != min_edge[u]:
            continue
        
        in_mst[u] = True
        component[u] = component[parent]
        mst_edges.append((parent, u, w, edge_idx))
        total_cost += w
        
        for v, w2, edge_idx2 in adj[u]:
            if not in_mst[v]:
                if w2 < min_edge[v]:
                    min_edge[v] = w2
                    best_edge[v] = (u, v, w2, edge_idx2)
                    heapq.heappush(pq, (w2, v, edge_idx2, u))
    
    # Check if all roots are in different components
    root_components = {}
    valid = True
    for r in root_list:
        if component[r] in root_components:
            valid = False
            break
        root_components[component[r]] = r
    
    if not valid:
        print("NO")
        return
    
    # Calculate cost to isolate roots
    # We need to cut edges that connect different root components
    # Build MST graph
    mst_adj = [[] for _ in range(n + 1)]
    for u, v, w, idx in mst_edges:
        mst_adj[u].append((v, w, idx))
        mst_adj[v].append((u, w, idx))
    
    # For each root, find edges on path to other roots
    # We need to cut minimum edges to separate all roots
    
    # Build a tree structure and find LCA-style approach
    # Actually, let's use a simpler approach: for each pair of roots,
    # find the minimum edge on the path between them
    
    # First, let's build parent pointers and depths
    parent = [(-1, -1, -1)] * (n + 1)  # (parent_node, weight, edge_idx)
    depth = [0] * (n + 1)
    
    # BFS from first root to set up parent pointers
    # But we have a forest, so we need to handle each tree
    
    visited = [False] * (n + 1)
    for r in root_list:
        if visited[r]:
            continue
        stack = [r]
        visited[r] = True
        while stack:
            u = stack.pop()
            for v, w, idx in mst_adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent[v] = (u, w, idx)
                    depth[v] = depth[u] + 1
                    stack.append(v)
    
    # Function to find minimum edge on path between two nodes
    def get_min_on_path(u, v):
        min_w = float('inf')
        min_idx = -1
        # Bring u and v to same depth
        while depth[u] > depth[v]:
            p, w, idx = parent[u]
            if w < min_w:
                min_w = w
                min_idx = idx
            u = p
        while depth[v] > depth[u]:
            p, w, idx = parent[v]
            if w < min_w:
                min_w = w
                min_idx = idx
            v = p
        # Now same depth, move both up
        while u != v:
            p, w, idx = parent[u]
            if w < min_w:
                min_w = w
                min_idx = idx
            u = p
            p, w, idx = parent[v]
            if w < min_w:
                min_w = w
                min_idx = idx
            v = p
        return min_w, min_idx
    
    # Find minimum cut to separate all roots
    # Use greedy approach: repeatedly cut minimum edge between any two roots in same component
    
    # Track which component each node belongs to (by root)
    root_comp = {}
    comp_id = [i for i in range(n + 1)]  # Initially each root is its own component
    
    # Map component to its root
    comp_to_root = {}
    for i, r in enumerate(root_list):
        comp_to_root[i] = r
    
    # Actually, let's use a simpler approach
    # We need to find minimum set of edges to cut to separate all roots
    # This is equivalent to finding minimum cut in a tree
    
    # For each pair of roots, find minimum edge on path
    # Then greedily cut edges
    
    # Build a list of all edges on paths between roots
    edges_to_consider = set()
    root_list_sorted = sorted(root_list)
    for i in range(len(root_list_sorted)):
        for j in range(i + 1, len(root_list_sorted)):
            r1, r2 = root_list_sorted[i], root_list_sorted[j]
            if component[r1] == component[r2]:
                # They are in same tree, find min edge on path
                min_w, min_idx = get_min_on_path(r1, r2)
                if min_idx != -1:
                    edges_to_consider.add((min_w, min_idx, r1, r2))
    
    # Sort edges by weight and greedily cut
    edges_sorted = sorted(edges_to_consider)
    
    # Use union-find to track which roots are separated
    uf_parent = list(range(n + 1))
    def find(x):
        if uf_parent[x] != x:
            uf_parent[x] = find(uf_parent[x])
        return uf_parent[x]
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            uf_parent[px] = py
            return True
        return False
    
    # Initially, all roots in same component (conceptually)
    # We need to separate them all
    # Actually, let's think differently:
    # We need to cut edges so that no two roots are connected
    
    # The minimum cost is the sum of minimum edges needed to separate all pairs
    # This is a minimum spanning tree on the "root graph"
    
    # Let's compute all pairwise minimum edges and run Kruskal
    root_pairs = []
    for i in range(len(root_list_sorted)):
        for j in range(i + 1, len(root_list_sorted)):
            r1, r2 = root_list_sorted[i], root_list_sorted[j]
            if component[r1] == component[r2]:
                min_w, min_idx = get_min_on_path(r1, r2)
                if min_idx != -1:
                    root_pairs.append((min_w, min_idx, i, j))
    
    root_pairs.sort()
    
    # Run Kruskal to find minimum edges to separate all roots
    root_uf = list(range(len(root_list_sorted)))
    def root_find(x):
        if root_uf[x] != x:
            root_uf[x] = root_find(root_uf[x])
        return root_uf[x]
    def root_union(x, y):
        px, py = root_find(x), root_find(y)
        if px != py:
            root_uf[px] = py
            return True
        return False
    
    edges_to_cut = []
    total_cut_cost = 0
    num_components = len(root_list_sorted)
    
    for w, idx, i, j in root_pairs:
        if root_find(i) != root_find(j):
            root_union(i, j)
            edges_to_cut.append(idx)
            total_cut_cost += w
            num_components -= 1
            if num_components == 1:
                break
    
    if num_components > 1:
        # Not all roots can be separated (shouldn't happen if graph is connected)
        print("NO")
        return
    
    if total_cut_cost > B:
        print("NO")
        return
    
    print("YES")
    print(len(edges_to_cut))
    for idx in edges_to_cut:
        u, v, w = edges[idx]
        print(u, v)

if __name__ == "__main__":
    solve()
