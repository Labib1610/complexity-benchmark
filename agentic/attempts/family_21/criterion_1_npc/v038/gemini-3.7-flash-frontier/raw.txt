import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    B = int(next(it))
    
    terminals = [int(next(it)) for _ in range(k)]
    
    orig_edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        orig_edges.append((u, v, w, i))
        
    if k <= 1:
        print("YES")
        print(0)
        return

    # Dinic's Algorithm
    class Dinic:
        def __init__(self, num_nodes):
            self.n = num_nodes
            self.graph = [[] for _ in range(num_nodes)]
            self.edges = []
            
        def add_edge(self, u, v, cap):
            e1 = [v, cap, len(self.graph[v]), 0] # to, cap, rev, flow
            e2 = [u, 0, len(self.graph[u]), 0]
            self.graph[u].append(e1)
            self.graph[v].append(e2)
            self.edges.append((u, len(self.graph[u]) - 1))

        def bfs(self, s, t):
            self.level = [-1] * self.n
            self.level[s] = 0
            q = deque([s])
            while q:
                u = q.popleft()
                for v, cap, rev, flow in self.graph[u]:
                    if cap - flow > 0 and self.level[v] < 0:
                        self.level[v] = self.level[u] + 1
                        q.append(v)
            return self.level[t] >= 0

        def dfs(self, u, t, pushed):
            if pushed == 0 or u == t:
                return pushed
            for cid in range(self.ptr[u], len(self.graph[u])):
                self.ptr[u] = cid
                v, cap, rev, flow = self.graph[u][cid]
                tr = self.graph[v][rev]
                if self.level[u] + 1 != self.level[v] or cap - flow == 0:
                    continue
                push = self.dfs(v, t, min(pushed, cap - flow))
                if push == 0:
                    continue
                self.graph[u][cid][3] += push
                tr[3] -= push
                return push
            return 0

        def max_flow(self, s, t):
            flow = 0
            while self.bfs(s, t):
                self.ptr = [0] * self.n
                while True:
                    pushed = self.dfs(s, t, float('inf'))
                    if pushed == 0:
                        break
                    flow += pushed
            return flow

        def min_cut_reachable(self, s):
            visited = [False] * self.n
            q = deque([s])
            visited[s] = True
            while q:
                u = q.popleft()
                for v, cap, rev, flow in self.graph[u]:
                    if cap - flow > 0 and not visited[v]:
                        visited[v] = True
                        q.append(v)
            return visited

    # Find isolating cuts for 2-approximation (Dahlhaus et al.)
    # Map vertices 1..n to 0..n-1
    # Super-source/sink node index: n, n+1
    
    isolating_cuts = []
    for i, t_node in enumerate(terminals):
        dinic = Dinic(n + 2)
        S = n
        T = n + 1
        dinic.add_edge(S, t_node - 1, float('inf'))
        for j, other_t in enumerate(terminals):
            if i != j:
                dinic.add_edge(other_t - 1, T, float('inf'))
        for u, v, w, _ in orig_edges:
            dinic.add_edge(u - 1, v - 1, w)
            dinic.add_edge(v - 1, u - 1, w)
            
        flow = dinic.max_flow(S, T)
        reachable = dinic.min_cut_reachable(S)
        # reachable is a set of nodes in S-side
        isolating_cuts.append((flow, reachable, i))

    # Dahlhaus 2-approximation: take the best k-1 isolating cuts
    # Sort isolating cuts by cost descending, remove the largest one
    isolating_cuts.sort(key=lambda x: x[0])
    
    # Check 2-approx solution:
    # Assign each node to the terminal isolating cut that contains it, or the leftover terminal
    # The first k-1 isolating cuts are used
    labels = [-1] * n
    for rank in range(k - 1):
        flow, reachable, term_idx = isolating_cuts[rank]
        for v in range(n):
            if reachable[v] and labels[v] == -1:
                labels[v] = term_idx
                
    last_term_idx = isolating_cuts[-1][2]
    for v in range(n):
        if labels[v] == -1:
            labels[v] = last_term_idx

    # Evaluate cut edges
    cut_edges = []
    total_cost = 0
    for u, v, w, idx in orig_edges:
        if labels[u - 1] != labels[v - 1]:
            cut_edges.append((u, v))
            total_cost += w

    if total_cost <= B:
        print("YES")
        print(len(cut_edges))
        for u, v in cut_edges:
            print(f"{u} {v}")
        return

    # If 2-approx > B, run Branch and Bound
    # DSU for contractions
    best_cost = total_cost
    best_cut = cut_edges

    # We can do exact branch and bound by contracting or cutting edges between different terminal components
    # Find shortest path between terminals and branch
    adj = [[] for _ in range(n + 1)]
    for u, v, w, idx in orig_edges:
        adj[u].append((v, w, idx))
        adj[v].append((u, w, idx))

    # Helper to check components
    term_set = set(terminals)
    parent = list(range(n + 1))
    def find(i):
        path = []
        while parent[i] != i:
            path.append(i)
            i = parent[i]
        for p in path:
            parent[p] = i
        return i

    # State: contracted edges, cut edges
    # Standard BFS to find path between any two terminals
    def search(cut_set, current_cost):
        nonlocal best_cost, best_cut
        if current_cost >= best_cost:
            return

        # Find connected components with non-cut edges
        p = list(range(n + 1))
        def f(x):
            path = []
            while p[x] != x:
                path.append(x)
                x = p[x]
            for node in path:
                p[node] = x
            return x
        
        def union(x, y):
            rx, ry = f(x), f(y)
            if rx != ry:
                p[rx] = ry

        for u, v, w, idx in orig_edges:
            if idx not in cut_set:
                union(u, v)

        # Check if any two terminals are connected
        term_roots = {}
        conflict_pair = None
        for t in terminals:
            rt = f(t)
            if rt in term_roots:
                conflict_pair = (term_roots[rt], t)
                break
            term_roots[rt] = t

        if conflict_pair is None:
            if current_cost < best_cost:
                best_cost = current_cost
                best_cut = [(orig_edges[idx][0], orig_edges[idx][1]) for idx in cut_set]
            return

        # BFS to find shortest path in edges not in cut_set between conflict_pair[0] and conflict_pair[1]
        start, target = conflict_pair
        q = deque([start])
        prev = {start: None}
        prev_edge = {}
        while q:
            curr = q.popleft()
            if curr == target:
                break
            for nxt, w, idx in adj[curr]:
                if idx not in cut_set and nxt not in prev:
                    prev[nxt] = curr
                    prev_edge[nxt] = idx
                    q.append(nxt)

        # Reconstruct path edges
        path_edges = []
        curr = target
        while curr != start:
            path_edges.append(prev_edge[curr])
            curr = prev[curr]

        # Branch on cutting each edge along the path
        # Sort path edges by weight descending or ascending
        path_edges.sort(key=lambda idx: orig_edges[idx][2])
        for edge_idx in path_edges:
            cut_set.add(edge_idx)
            search(cut_set, current_cost + orig_edges[edge_idx][2])
            cut_set.remove(edge_idx)

    search(set(), 0)

    if best_cost <= B:
        print("YES")
        print(len(best_cut))
        for u, v in best_cut:
            print(f"{u} {v}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
