import sys
from collections import deque

def read_input():
    n, m, k, B = map(int, sys.stdin.readline().split())
    sovereign = set(map(int, sys.stdin.readline().split()))
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
    return n, m, k, B, sovereign, edges

def find_critical_edges(n, sovereign, edges):

    # Try all possible combinations of cutting edges within budget B
    # But that's exponential
    
    # Let's model this as finding a minimum cut in a modified graph:
    # Create a new graph where we contract all non-sovereign nodes into components
    # But this gets complex
    
    # Simpler practical approach:
    # Use BFS/DFS to check connectivity between sovereign nodes
    # Greedily remove cheapest edges that help separate them
    
    def is_valid_solution(removed_edges):

        temp_adj = [[] for _ in range(n+1)]

            # BFS to find if there's a path between s1 and s2
            def bfs_path_exists():
                if s1 == s2:
                    return False
                
                visited = [False] * (n+1)
                queue = deque([s1])
                visited[s1] = True
                
                while queue:
                    node = queue.popleft()
                    if node == s2:
                        return True
                    
                    for neighbor, _, _ in adj[node]:
                        if not visited[neighbor]:
                            visited[neighbor] = True
                            queue.append(neighbor)

    # Sort edges by weight (ascending)
    indexed_edges = [(w, i, u, v) for i, (u, v, w) in enumerate(edges)]
    indexed_edges.sort()
    
    # Greedily try removing edges
    removed_edges = []
    total_cost = 0

        # Temporarily add this edge to removed set and check if solution becomes valid
        temp_removed = current_removed.copy()
        temp_removed.add(i)

            visited = [False] * (n+1)

            for s in sovereign:

                    component = set()

                    visited[s] = True

                            if not visited[neighbor]:
                                visited[neighbor] = True
                                queue.append(neighbor)
                                component.add(neighbor)
                    
                    components.append(component)
            
            for comp in components:
                if len(comp.intersection(sovereign)) > 1:

            return True
        
        if check_separation():
            print("YES")
            print(0)
            return
        
        # Otherwise try greedy approach
        edge_with_index = [(w, i, u, v) for i, (u, v, w) in enumerate(edges)]
        edge_with_index.sort()
        
        removed = []
        total_cost = 0
        disabled = [False] * m
        
        def is_disconnected():
            temp_adj = [[] for _ in range(n+1)]
            for i, (u, v, w) in enumerate(edges):
                if not disabled[i]:
                    temp_adj[u].append(v)
                    temp_adj[v].append(u)
            
            visited = [False] * (n+1)
            for start_sovereign in sovereign:
                if visited[start_sovereign]:
                    continue
                    
                component = []

                visited[start_sovereign] = True
                component.append(start_sovereign)

                    node = stack.pop()
                    for neighbor in temp_adj[node]:
                        if not visited[neighbor]:

                sovereign_here = [x for x in component if x in sovereign]
                if len(sovereign_here) > 1:

                disabled[i] = True
                if is_disconnected():
