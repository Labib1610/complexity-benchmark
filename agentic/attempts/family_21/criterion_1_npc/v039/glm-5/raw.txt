import sys

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    k = int(data[idx]); idx += 1
    B = int(data[idx]); idx += 1

    sources = []
    for _ in range(k):
        sources.append(int(data[idx]) - 1); idx += 1

    edges = []
    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        w = int(data[idx]); idx += 1
        edges.append((u, v, w))

    if k <= 1:
        sys.stdout.write("YES\n0\n")
        return

    # DSU
    parent = list(range(n))
    rank = [0] * n

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(x, y):
        rx = find(x)
        ry = find(y)
        if rx == ry:
            return False
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
        return True

    # Sort edges by weight ascending
    sorted_edges = sorted(edges, key=lambda e: e[2])

    # Greedy: cap cheapest edges that connect different source components
    capped = []
    total = 0
    for u, v, w in sorted_edges:
        ru = find(u)
        rv = find(v)
        if ru == rv:
            continue
        # Check if merging would connect two different sources
        # Track which sources are in each component
        # We need to know if ru and rv each contain sources
        # Since we only merge when no two sources are in both,
        # we can track source membership
        # But we didn't track that. Let's track it.
        pass

    # Redo with source tracking
    parent = list(range(n))
    rank = [0] * n
    comp_sources = [0] * n
    for s in sources:
        comp_sources[s] = 1  # count of sources in component (we only care >0)

    def find2(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    capped = []
    total = 0
    for u, v, w in sorted_edges:
        ru = find2(u)
        rv = find2(v)
        if ru == rv:
            continue
        if comp_sources[ru] > 0 and comp_sources[rv] > 0:
            # Capping this edge prevents merging two source-containing components
            capped.append((u, v))
            total += w
            if total > B:
                sys.stdout.write("NO\n")
                return
        else:
            # Safe to merge
            if rank[ru] < rank[rv]:
                parent[ru] = rv
                comp_sources[rv] += comp_sources[ru]
            elif rank[ru] > rank[rv]:
                parent[rv] = ru
                comp_sources[ru] += comp_sources[rv]
            else:
                parent[rv] = ru
                comp_sources[ru] += comp_sources[rv]
                rank[ru] += 1

    # Verify all sources are separated
    # After processing, check that no component has >1 source
    # Actually the greedy ensures we never merge two components that both have sources,
    # so each component has at most 1 source. But we need to verify all sources are
    # in separate components. Since we never merge two source-containing components,
    # they remain separate. However, a source could still be connected to another
    # source if we didn't cap enough edges. But the greedy caps every edge that
    # would connect two source-containing components, so they're isolated.

    # But wait: what if total <= B but some sources are still connected?
    # That can't happen because we cap every edge that would connect two
    # source-containing components. The only way two sources remain connected
    # is if there's a path of uncapped edges, but every edge on any path
    # between two source-containing components would be capped when encountered
    # (since at the time of processing, the endpoints would be in different
    # source-containing components). Actually not necessarily: an edge might
    # be processed before both endpoints become source-containing.
    # 
    # Let me think again. The greedy processes edges in increasing weight.
    # When we process edge (u,v), if find(u) and find(v) both contain sources,
    # we cap it. Otherwise we merge. This is exactly Kruskal-like but we
    # refuse to merge two source-containing components.
    # 
    # This ensures that in the final forest, no two sources are in the same
    # tree, because we never merge two trees that both contain sources.
    # 
    # So the solution is valid if total <= B.

    sys.stdout.write("YES\n")
    sys.stdout.write(str(len(capped)) + "\n")
    for u, v in capped:
        sys.stdout.write(f"{u+1} {v+1}\n")

solve()
