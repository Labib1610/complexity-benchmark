import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    B = int(next(it))
    sources = [int(next(it)) - 1 for _ in range(k)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        edges.append((u, v, w))

    # If m is small, brute‑force all subsets.
    if m <= 22:          # 2^22 ≈ 4 million, fine in 5 s
        parent = [0] * n

        def find(x):
            while parent[x] != x:
                parent[x] = parent[parent[x]]
                x = parent[x]
            return x

        def union(a, b):
            ra, rb = find(a), find(b)
            if ra != rb:
                parent[rb] = ra

        best_subset = None
        # iterate subsets in increasing cost order could help, but simple enumeration is enough
        for mask in range(1 << m):
            total = 0
            # quick cost prune
            for i in range(m):
                if mask >> i & 1:
                    total += edges[i][2]
                    if total > B:
                        break
            else:
                # build DSU with edges NOT removed
                for i in range(n):
                    parent[i] = i
                for i in range(m):
                    if not (mask >> i & 1):
                        u, v, _ = edges[i]
                        union(u, v)
                # check sources are in distinct components
                comps = set(find(s) for s in sources)
                if len(comps) == k:
                    best_subset = mask
                    break
        if best_subset is None:
            print("NO")
            return
        # output removed edges
        removed = [i for i in range(m) if best_subset >> i & 1]
        print("YES")
        print(len(removed))
        for i in removed:
            u, v, _ = edges[i]
            print(u + 1, v + 1)
        return

    # Fallback heuristic: cut all edges incident to any source (guarantees separation)
    removed = []
    total = 0
    for idx, (u, v, w) in enumerate(edges):
        if u in sources or v in sources:
            removed.append(idx)
            total += w
    if total <= B:
        print("YES")
        print(len(removed))
        for idx in removed:
            u, v, _ = edges[idx]
            print(u + 1, v + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
