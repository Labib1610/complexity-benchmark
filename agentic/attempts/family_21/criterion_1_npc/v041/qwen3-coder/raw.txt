    # But that's exponential. Instead, let's think of this as a Steiner forest problem variant.

    # We want to remove minimum weight edges such that no two habitats are connected
    # This is exactly the multiway cut problem
    
    # For small k, we can solve this optimally using flow-based methods or enumeration
    # But since we also have budget constraint B, we need to find a feasible solution
    
    # Let's model this as finding a minimum edge set whose removal disconnects all pairs of habitats

    # So we can use approximation or greedy approaches

    # Keep adding edges to the cut until all habitat pairs are disconnected
    # But this might not give optimal result
    
    # Let's instead compute the actual minimum multiway cut
    # For k habitats, multiway cut can be computed by considering all possible assignments

    # Since k could be up to n, let's think differently
    
    # Key insight: we can reduce this to a graph where we add a super-source connected to all habitats

    # 2. Check if cost <= B

    # The goal is to remove minimum weight edges so that each terminal is in a different component

            return True, []
        visited = set()
        queue = deque([(start, [start])])
        visited.add(start)
        
        while queue:
            node, path = queue.popleft()
            for neighbor, _ in adj_list[node]:

                    return True, path + [neighbor]
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, path + [neighbor]))
        return False, []

            edges_in_path.append((u, v))
        return edges_in_path

                        # Remove this edge from the graph

                        total_cost += min_cost
                        changed = True
                        break

                break

                all_separated = False
                break
        if not all_separated:
            break
    
    if all_separated:
        print("YES")
        print(len(removed_edges))
        for u, v in removed_edges:
            print(u, v)
    else:
        print("NO")
