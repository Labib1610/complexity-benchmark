def find_min_cut(graph, n, source, sink):
    # Find minimum cut between source and sink using max flow/min cut
    from collections import defaultdict, deque
    
    # Build adjacency list
    adj = defaultdict(list)
    capacity = defaultdict(int)
    
    for u, v, w in graph:
        adj[u].append(v)
        adj[v].append(u)
        capacity[(u, v)] += w
        capacity[(v, u)] += w
    
    def bfs():
        parent = [-1] * (n + 1)
        visited = [False] * (n + 1)
        q = deque([source])
        visited[source] = True
        
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v] and capacity[(u, v)] > 0:
                    visited[v] = True
                    parent[v] = u
                    q.append(v)
                    if v == sink:
                        return parent
        return None
    
    def edmonds_karp():
        total_flow = 0
        while True:
            parent = bfs()
            if parent is None:
                break
                
            # Find minimum capacity along the path
            path_flow = float('inf')
            s = sink
            while s != source:
                path_flow = min(path_flow, capacity[(parent[s], s)])
                s = parent[s]
            
            # Update capacities
            v = sink
            while v != source:
                u = parent[v]
                capacity[(u, v)] -= path_flow
                capacity[(v, u)] += path_flow
                v = parent[v]
            
            total_flow += path_flow
        
        return total_flow
    
    return edmonds_karp()
