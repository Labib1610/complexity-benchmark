import sys
from collections import defaultdict
import heapq

def find_mst(nodes, edges):
    # Create a graph with only the k compromised nodes and edges between them
    # We need to find the minimum spanning tree that connects all k nodes
    # This is essentially the Steiner tree problem, but we'll use a simpler approach:
    # Since k is small, we can try to find the MST of the complete graph induced by the k nodes
    # where edge weights are shortest path distances between pairs
    
    # First, let's build the full graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # Compute shortest paths between all pairs of compromised nodes
    # Using Dijkstra from each compromised node
    k_list = list(nodes)
    n = max(max(u, v) for u, v, _ in edges) if edges else 0
    
    dist = {}
    # Run Dijkstra from each compromised node
    for start in k_list:
        dist[start] = {}
        d = [float('inf')] * (n + 1)
        d[start] = 0
        pq = [(0, start)]
        visited = [False] * (n + 1)
        
        while pq:
            cur_d, u = heapq.heappop(pq)
            if visited[u]:
                continue
            visited[u] = True
            dist[start][u] = cur_d
            
            for v, w in graph[u]:
                if not visited[v] and cur_d + w < d[v]:
                    d[v] = cur_d + w
                    heapq.heappush(pq, (d[v], v))
    
    # Now create a complete graph with k nodes, with edge weights as shortest paths
    # And find its MST using Kruskal's algorithm
    k_edges = []
    for i in range(len(k_list)):
        for j in range(i+1, len(k_list)):
            u, v = k_list[i], k_list[j]
            if v in dist[u]:
                k_edges.append((dist[u][v], u, v))
    
    # Sort edges by weight
    k_edges.sort()
    
    # Union-Find for Kruskal's algorithm
    parent = {node: node for node in k_list}
    
    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]
    
    def union(x, y):
        px, py = find(x), find(y)
        if px != py:
            parent[px] = py
            return True
        return False
    
    # Kruskal's algorithm
    mst_edges = []
    total_weight = 0
    for w, u, v in k_edges:
        if union(u, v):
            mst_edges.append((w, u, v))
            total_weight += w
            if len(mst_edges) == len(k_list) - 1:
                break
    
    return total_weight, mst_edges, dist

def get_path_edges(start, end, dist, graph):
    # Reconstruct actual path edges from start to end
    if start == end:
        return []
    
    # Run Dijkstra to get predecessors
    n = max(node for node in graph.keys()) if graph else 0
    d = [float('inf')] * (n + 1)
    parent = [-1] * (n + 1)
    d[start] = 0
    pq = [(0, start)]
    visited = [False] * (n + 1)
    
    while pq:
        cur_d, u = heapq.heappop(pq)
        if visited[u]:
            continue
        visited[u] = True
        
        if u == end:
            break
            
        for v, w in graph[u]:
            if not visited[v] and cur_d + w < d[v]:
                d[v] = cur_d + w
                parent[v] = u
                heapq.heappush(pq, (d[v], v))
    
    # Reconstruct path
    path_edges = []
    current = end
    while parent[current] != -1:
        path_edges.append((parent[current], current))
        current = parent[current]
    
    return path_edges

def solve():
    line = sys.stdin.readline().strip()
    if not line:
        return "NO"
    
    n, m, k, B = map(int, line.split())
    compromised = list(map(int, sys.stdin.readline().split()))
    
    if k <= 1:
        return "YES\n0"
    
    edges = []
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
    
    # Build graph
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # For small k, we can compute pairwise shortest paths and then find MST
    k_set = set(compromised)
    k_list = list(k_set)
    
    # Compute shortest paths between all pairs of compromised nodes
    dist_matrix = {}
    pred_matrix = {}
    
    for start in k_list:
        d = [float('inf')] * (n + 1)
        parent = [-1] * (n + 1)
        d[start] = 0
        pq = [(0, start)]
        visited = [False] * (n + 1)
        
        while pq:
            cur_d, u = heapq.heappop(pq)
            if visited[u]:
                continue
            visited[u] = True
            
            for v, w in graph[u]:
                if not visited[v] and cur_d + w < d[v]:
                    d[v] = cur_d + w
                    parent[v] = u
                    heapq.heappush(pq, (d[v], v))
        
        dist_matrix[start] = d
        pred_matrix[start] = parent
    
    # Create complete graph on compromised nodes with shortest path weights
    complete_edges = []
    for i in range(len(k_list)):
        for j in range(i+1, len(k_list)):
            u, v = k_list[i], k_list[j]
            complete_edges.append((dist_matrix[u][v], u, v))
    
    # Check if any distance is infinite
    for _, u, v in complete_edges:
        if dist_matrix[u][v] == float('inf'):
            return "NO"
    
    # Find MST of this complete graph (Kruskal's algorithm)
    complete_edges.sort()
    
    parent_uf = {node: node for node in k_list}
    
    def find_uf(x):
        if parent_uf[x] != x:
            parent_uf[x] = find_uf(parent_uf[x])
        return parent_uf[x]
    
    def union_uf(x, y):
        px, py = find_uf(x), find_uf(y)
        if px != py:
            parent_uf[px] = py
            return True
        return False
    
    mst_weight = 0
    mst_complete_edges = []
    for w, u, v in complete_edges:
        if union_uf(u, v):
            mst_complete_edges.append((w, u, v))
            mst_weight += w
            if len(mst_complete_edges) == len(k_list) - 1:
                break
    
    if mst_weight > B:
        return "NO"
    
    # Reconstruct actual edges used
    used_edges = set()
    for _, u, v in mst_complete_edges:
        # Reconstruct path from u to v
        path = []
        current = v
        while pred_matrix[u][current] != -1:
            prev = pred_matrix[u][current]
            path.append((prev, current))
            current = prev
        
        for edge in path:
            used_edges.add(tuple(sorted(edge)))
    
    result_edges = list(used_edges)
    
    output = ["YES"]
    output.append(str(len(result_edges)))
    for u, v in result_edges:
        output.append(f"{u} {v}")
    
    return "\n".join(output)

print(solve())
