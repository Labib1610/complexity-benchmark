```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    B = int(next(it))
    specials = set(int(next(it)) for _ in range(k))
    edges = []
    total = 0
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        if u in specials or v in specials:
            edges.append((u, v, w))
            total += w
    if total <= B:
        out = ["YES", str(len(edges))]
        out += [f"{u} {v}" for u, v, _ in edges]
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
