def read_input():
    n, m, k, B = map(int, sys.stdin.readline().split())

        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))
        
    return n, m, k, B, anchors, edges, graph

    # Find connected components using BFS
    visited = [False] * (n + 1)

    for i in range(1, n + 1):

            while queue:
                node = queue.popleft()

                node_to_component[node] = len(components)

                        visited[neighbor] = True
                        queue.append(neighbor)
            
            components.append(component)
    
    # Check how many anchors are in each component
    anchor_components = defaultdict(int)
    for anchor in anchors:

            anchor_components[node_to_component[anchor]] += 1
    
    # Count anchors in components with more than one anchor
    violated_count = sum(1 for count in anchor_components.values() if count > 1)
    
    return violated_count == 0
