We can model this as a **minimum cut** problem: we need to separate all primary anchors from each other, which is equivalent to finding a multiway cut.  

Since the number of primary anchors is small (typically), we can use a **minimum s-t cut** approach by trying all possible assignments of anchors to two sides, or use a known reduction:  

- Add a super source connected to one anchor with infinite capacity, and a super sink connected to another anchor with infinite capacity.  
- The minimum cut between them gives the cheapest way to separate those two anchors.  
- We need to separate **all pairs**, so we can iterate over all pairs of anchors and take the minimum cut, then reconstruct the edges.

Here’s the implementation:  

```python
import sys
from collections import deque

input = sys.stdin.readline

class MaxFlow:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.adj[u].append([v, cap, len(self.adj[v])])
        self.adj[v].append([u, 0, len(self.adj[u]) - 1])
    
    def bfs(self, s, t, level):
        for i in range(self.n):
            level[i] = -1
        q = deque([s])
        level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, _ in self.adj[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] >= 0
    
    def dfs(self, u, t, f, level, it):
        if u == t:
            return f
        while it[u] < len(self.adj[u]):
            v, cap, rev = self.adj[u][it[u]]
            if cap > 0 and level[v] == level[u] + 1:
                ret = self.dfs(v, t, min(f, cap), level, it)
                if ret > 0:
                    self.adj[u][it[u]][1] -= ret
                    self.adj[v][rev][1] += ret
                    return ret
            it[u] += 1
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        level = [-1] * self.n
        INF = 10**18
        while self.bfs(s, t, level):
            it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF, level, it)
                if f == 0:
                    break
                flow += f
        return flow

def solve():
    n, m, k, B = map(int, input().split())
    anchors = list(map(int, input().split()))
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        edges.append((u-1, v-1, w))
    
    # Try all pairs of anchors
    best_cost = float('inf')
    best_cut_edges = None
    
    for i in range(k):
        for j in range(i+1, k):
            s = anchors[i] - 1
            t = anchors[j] - 1
            mf = MaxFlow(n)
            for u, v, w in edges:
                mf.add_edge(u, v, w)
                mf.add_edge(v, u, w)
            cost = mf.max_flow(s, t)
            if cost < best_cost:
                best_cost = cost
                # Reconstruct cut edges
                level = [-1] * n
                mf.bfs(s, t, level)
                cut_edges = []
                for u, v, w in edges:
                    if level[u] >= 0 and level[v] < 0:
                        cut_edges.append((u+1, v+1))
                    elif level[v] >= 0 and level[u] < 0:
                        cut_edges.append((u+1, v+1))
                best_cut_edges = cut_edges
    
    if best_cost <= B:
        print("YES")
        print(len(best_cut_edges))
        for u, v in best_cut_edges:
            print(u, v)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
