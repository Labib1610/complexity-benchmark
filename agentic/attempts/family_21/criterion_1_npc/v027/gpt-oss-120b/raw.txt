```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it)); k = int(next(it)); B = int(next(it))
    specials = set(int(next(it)) for _ in range(k))
    edges = []
    total_cost = 0
    blocked = []
    for _ in range(m):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        edges.append((u, v, w))
        if (u in specials) ^ (v in specials) or (u in specials and v in specials):
            # edge incident to at least one special (including between specials)
            total_cost += w
            blocked.append((u, v))
    if total_cost <= B:
        out = ["YES", str(len(blocked))]
        out += [f"{u} {v}" for u, v in blocked]
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
