```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    k = int(next(it))
    B = int(next(it))

    terminals = [int(next(it)) for _ in range(k)]
    term_set = set(terminals)

    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
        adj[u].append((v, w, i))
        adj[v].append((u, w, i))

    if k <= 1:
        print("YES")
        print(0)
        return

    # Dinic's Algorithm for min-cut
    def min_cut_st(source_nodes, sink_nodes):
        # source_nodes: list, sink_nodes: list
        # We create a flow network
        # Nodes: 1..n, Source: 0, Sink: n + 1
        SRC = 0
        SNK = n + 1
        
        # Build residual graph
        head = [-1] * (n + 2)
        to = []
        nxt = []
        cap = []
        flow = []
        edge_id = []

        def add_edge(u, v, c, eid):
            to.append(v)
            cap.append(c)
            flow.append(0)
            edge_id.append(eid)
            nxt.append(head[u])
            head[u] = len(to) - 1

            to.append(u)
            cap.append(0)
            flow.append(0)
            edge_id.append(eid)
            nxt.append(head[v])
            head[v] = len(to) - 1

        INF = 10**15
        for u in source_nodes:
            add_edge(SRC, u, INF, -1)
        for v in sink_nodes:
            add_edge(v, SNK, INF, -1)

        for i, (u, v, w) in enumerate(edges):
            # undirected edge: capacity w in both directions
            to.append(v)
            cap.append(w)
            flow.append(0)
            edge_id.append(i)
            nxt.append(head[u])
            head[u] = len(to) - 1

            to.append(u)
            cap.append(w)
            flow.append(0)
            edge_id.append(i)
            nxt.append(head[v])
            head[v] = len(to) - 1

        level = [-1] * (n + 2)
        ptr = [0] * (n + 2)

        def bfs():
            for i in range(n + 2):
                level[i] = -1
            level[SRC] = 0
            q = deque([SRC])
            while q:
                v = q.popleft()
                e = head[v]
                while e != -1:
                    u = to[e]
                    if cap[e] - flow[e] > 0 and level[u] < 0:
                        level[u] = level[v] + 1
                        q.append(u)
                    e = nxt[e]
            return level[SNK] != -1

        def dfs(v, pushed):
            if pushed == 0:
                return 0
            if v == SNK:
                return pushed
            e = ptr[v]
            while e != -1:
                u = to[e]
                tr = cap[e] - flow[e]
                if level[v] + 1 == level[u] and tr > 0:
                    tr_pushed = dfs(u, min(pushed, tr))
                    if tr_pushed > 0:
                        flow[e] += tr_pushed
                        flow[e ^ 1] -= tr_pushed
                        return tr_pushed
                e = nxt[e]
                ptr[v] = e
            return 0

        tot_flow = 0
        while bfs():
            for i in range(n + 2):
                ptr[i] = head[i]
            while True:
                pushed = dfs(SRC, INF)
                if pushed == 0:
                    break
                tot_flow += pushed

        # Find reachable from SRC in residual graph
        reachable = [False] * (n + 2)
        q = deque([SRC])
        reachable[SRC] = True
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                u = to[e]
                if cap[e] - flow[e] > 0 and not reachable[u]:
                    reachable[u] = True
                    q.append(u)
                e = nxt[e]

        # Cut edges
        cut_edge_indices = set()
        for idx, (u, v, w) in enumerate(edges):
            if reachable[u] != reachable[v]:
                cut_edge_indices.add(idx)

        return tot_flow, cut_edge_indices

    if k == 2:
        flow, cut_edges = min_cut_st([terminals[0]], [terminals[1]])
        if flow <= B:
            print("YES")
            print(len(cut_edges))
            for idx in cut_edges:
                u, v, w = edges[idx]
                print(f"{u} {v}")
        else:
            print("NO")
        return

    # For k >= 3, check isolating cuts heuristic
    isolating_cuts = []
    for i in range(k):
        src = [terminals[i]]
        snk = [terminals[j] for j in range(k) if j != i]
        flow, cut_set = min_cut_st(src, snk)
        isolating_cuts.append(cut_set)

    best_cut = None
    best_weight = float('inf')

    # Try leaving out each terminal i
    for i in range(k):
        union_cut = set()
        for j in range(k):
            if j != i:
                union_cut.update(isolating_cuts[j])
        weight = sum(edges[idx][2] for idx in union_cut)
        if weight < best_weight:
            best_weight = weight
            best_cut = union_cut

    if best_weight <= B:
        print("YES")
        print(len(best_cut))
        for idx in best_cut:
            u, v, w = edges[idx]
            print(f"{u} {v}")
        return

    # If heuristic didn't meet B, attempt Branch and Bound
    # For general competitive programming instances where B is tight:
    # State: mapping of vertices to terminal index 0..k-1
    # Terminals are pre-assigned.
    best_ans = [None]
    best_cost = [best_weight]

    # Quick DSU to check connectivity
    # Branch and bound assigning vertices
    term_idx = {t: i for i, t in enumerate(terminals)}
    assignment = [-1] * (n + 1)
    for i, t in enumerate(terminals):
        assignment[t] = i

    unassigned = [i for i in range(1, n + 1) if assignment[i] == -1]

    # Sort unassigned by degree / connection to terminals
    def get_priority(u):
        return sum(w for v, w, _ in adj[u])

    unassigned.sort(key=get_priority, reverse=True)

    # Simple branch and bound if unassigned vertices count is reasonable
    def bnb(idx, current_cost):
        if current_cost >= best_cost[0] or current_cost > B:
            return
        if idx == len(unassigned):
            best_cost[0] = current_cost
            cut_edges = []
            for e_idx, (u, v, w) in enumerate(edges):
                if assignment[u] != assignment[v]:
                    cut_edges.append(e_idx)
            best_ans[0] = cut_edges
            return

        u = unassigned[idx]

        # Calculate cost increment for assigning u to each terminal
        costs = []
        for c in range(k):
            add_c = 0
            for v, w, _ in adj[u]:
                if assignment[v] != -1 and assignment[v] != c:
                    add_c += w
            costs.append((add_c, c))

        costs.sort()

        for add_c, c in costs:
            if current_cost + add_c >= best_cost[0] or current_cost + add_c > B:
                continue
            assignment[u] = c
            bnb(idx + 1, current_cost + add_c)
            assignment[u] = -1

    # Only run BnB if unassigned is relatively small, or time permits
    if len(unassigned) <= 25:
        # Calculate initial cut among terminals
        init_cost = 0
        for u, v, w in edges:
            if assignment[u] != -1 and assignment[v] != -1 and assignment[u] != assignment[v]:
                init_cost += w
        bnb(0, init_cost)

    if best_ans[0] is not None and best_cost[0] <= B:
        print("YES")
        print(len(best_ans[0]))
        for idx in best_ans[0]:
            u, v, w = edges[idx]
            print(f"{u} {v}")
    elif best_weight <= B:
        print("YES")
        print(len(best_cut))
        for idx in best_cut:
            u, v, w = edges[idx]
            print(f"{u} {v}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
