We need to sever links so that all proprietary servers are in different connected components.  
This is equivalent to finding a **minimum cut** that separates all `k` terminals, and checking if its cost is ≤ `B`.

A standard approach is to use **Gomory-Hu tree** or **minimum k-cut** via repeated min-cut computations.  
Since `k` is small (not given explicitly, but likely small), we can use a simple method:

- Build a flow network from the original graph.
- For each pair of proprietary servers, compute the min-cut between them.
- The minimum k-cut can be found by a greedy algorithm: start with one group containing all terminals, then repeatedly split a group by computing the min-cut between one terminal and the rest of the group, until each terminal is alone.

We’ll implement using Dinic’s algorithm for max flow.

```python
import sys
from collections import deque

input = sys.stdin.readline

class Dinic:
    def __init__(self, n):
        self.n = n
        self.graph = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.graph[u].append([v, cap, len(self.graph[v])])
        self.graph[v].append([u, 0, len(self.graph[u]) - 1])
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, rev in self.graph[u]:
                if cap > 0 and self.level[v] < 0:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] >= 0
    
    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.graph[u])):
            self.it[u] = i
            v, cap, rev = self.graph[u][i]
            if cap > 0 and self.level[u] + 1 == self.level[v]:
                ret = self.dfs(v, t, min(f, cap))
                if ret > 0:
                    self.graph[u][i][1] -= ret
                    self.graph[v][rev][1] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if f == 0:
                    break
                flow += f
        return flow

def solve():
    n, m, k, B = map(int, input().split())
    terminals = list(map(int, input().split()))
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        edges.append((u-1, v-1, w))
    
    # Build a Dinic instance for each min-cut computation
    def min_cut_between(s, t):
        dinic = Dinic(n)
        for u, v, w in edges:
            dinic.add_edge(u, v, w)
            dinic.add_edge(v, u, w)
        return dinic.max_flow(s, t)
    
    # Greedy k-cut: start with all terminals in one group
    groups = [set(terminals)]
    total_cost = 0
    cut_edges = []
    
    # We'll need to find actual edges cut; we can reconstruct from min-cut
    # For simplicity, we'll store the min-cut edges for each split.
    # We'll implement a function to get the cut edges for a given s-t min cut.
    
    def get_cut_edges(s, t):
        dinic = Dinic(n)
        for u, v, w in edges:
            dinic.add_edge(u, v, w)
            dinic.add_edge(v, u, w)
        flow = dinic.max_flow(s, t)
        # BFS from s in residual graph
        visited = [False] * n
        q = deque([s])
        visited[s] = True
        while q:
            u = q.popleft()
            for v, cap, rev in dinic.graph[u]:
                if cap > 0 and not visited[v]:
                    visited[v] = True
                    q.append(v)
        cut = []
        for u, v, w in edges:
            if visited[u] != visited[v]:
                cut.append((u, v))
        return flow, cut
    
    # We'll do recursive splitting
    def split_group(group):
        nonlocal total_cost, cut_edges
        if len(group) == 1:
            return
        # pick first terminal as source, rest as sink
        terminals_list = list(group)
        s = terminals_list[0]
        t = terminals_list[1]  # any other terminal
        cost, cut = get_cut_edges(s, t)
        total_cost += cost
        cut_edges.extend(cut)
        # Now split group into two: those reachable from s in residual graph
        # We need to know which terminals are on s side
        dinic = Dinic(n)
        for u, v, w in edges:
            dinic.add_edge(u, v, w)
            dinic.add_edge(v, u, w)
        dinic.max_flow(s, t)
        visited = [False] * n
        q = deque([s])
        visited[s] = True
        while q:
            u = q.popleft()
            for v, cap, rev in dinic.graph[u]:
                if cap > 0 and not visited[v]:
                    visited[v] = True
                    q.append(v)
        group1 = {x for x in group if visited[x]}
        group2 = group - group1
        split_group(group1)
        split_group(group2)
    
    split_group(set(terminals))
    
    if total_cost <= B:
        print("YES")
        print(len(cut_edges))
        for u, v in cut_edges:
            print(u+1, v+1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
