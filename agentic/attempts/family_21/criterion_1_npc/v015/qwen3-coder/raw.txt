import sys
from collections import defaultdict, deque

def find_min_edge_cut(graph, n, source, sink):
    # Find minimum edge cut between source and sink using BFS-based approach
    # This is a simplified version for unweighted graphs
    # For weighted graphs, we'd need to use max-flow min-cut
    visited = [False] * (n + 1)
    parent = [-1] * (n + 1)
    queue = deque([source])
    visited[source] = True
    
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if not visited[v] and graph[u][v] > 0:  # Edge has capacity
                visited[v] = True
                parent[v] = u
                queue.append(v)
                if v == sink:
                    break
    
    return visited[sink]

def bfs_check_connectivity(graph, n, start, targets, exclude_edges):
    visited = [False] * (n + 1)
    queue = deque([start])
    visited[start] = True
    
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if (u, v) in exclude_edges or (v, u) in exclude_edges:
                continue
            if not visited[v]:
                visited[v] = True
                queue.append(v)
    
    for target in targets:
        if target != start and visited[target]:
            return True
    return False
