import sys

def find_articulation_points_and_bridges(n, edges):
    # Build adjacency list
    adj = defaultdict(list)
    for u, v, w in edges:
        adj[u].append((v, w))
        adj[v].append((u, w))
    
    # Tarjan's algorithm for finding bridges
    disc = [-1] * (n + 1)
    low = [-1] * (n + 1)
    parent = [-1] * (n + 1)
    time = [0]
    bridges = []
    
    def dfs(u):
        disc[u] = low[u] = time[0]
        time[0] += 1
        
        for v, w in adj[u]:
            if disc[v] == -1:  # If v is not visited yet
                parent[v] = u
                dfs(v)
                low[u] = min(low[u], low[v])
                
                # If the lowest vertex reachable from subtree under v is below u in DFS tree, 
                # then u-v is a bridge
                if low[v] > disc[u]:
                    bridges.append((u, v, w))
            elif v != parent[u]:  # Update low value of u for parent function calls
                low[u] = min(low[u], disc[v])
    
    for i in range(1, n + 1):
        if disc[i] == -1:
            dfs(i)
    
    return bridges

def find_min_cut_for_pair(n, edges, s, t):
    # Use Edmonds-Karp algorithm to find minimum cut
    # Build residual graph
    capacity = defaultdict(lambda: defaultdict(int))
    adj = defaultdict(list)
    
    for u, v, w in edges:
        if capacity[u][v] == 0:  # First edge
            adj[u].append(v)
            adj[v].append(u)
        capacity[u][v] += w
        capacity[v][u] += w
    
    def bfs(source, sink, parent):
        visited = [False] * (n + 1)
        queue = deque([source])
        visited[source] = True
        
        while queue:
            u = queue.popleft()
            
            for v in adj[u]:
                if not visited[v] and capacity[u][v] > 0:
                    queue.append(v)
                    visited[v] = True
                    parent[v] = u
                    if v == sink:
                        return True
        return False
    
    def ford_fulkerson(source, sink):
        parent = [-1] * (n + 1)
        max_flow = 0
        
        while bfs(source, sink, parent):
            path_flow = float('inf')
            s_node = sink
            
            while s_node != source:
                path_flow = min(path_flow, capacity[parent[s_node]][s_node])
                s_node = parent[s_node]
            
            v = sink
            while v != source:
                u = parent[v]
                capacity[u][v] -= path_flow
                capacity[v][u] += path_flow
                v = parent[v]
            
            max_flow += path_flow
        
        return max_flow
    
    # Create a copy of capacity for this run
    original_capacity = defaultdict(lambda: defaultdict(int))
    for u in capacity:
        for v in capacity[u]:
            original_capacity[u][v] = capacity[u][v]
    
    flow = ford_fulkerson(s, t)
    
    # Restore capacity
    for u in capacity:
        for v in capacity[u]:
            capacity[u][v] = original_capacity[u][v]
    
    return flow

def find_min_cut_edges(n, edges, s, t):
    # Build residual graph
    capacity = defaultdict(lambda: defaultdict(int))
    adj = defaultdict(list)
    
    for u, v, w in edges:
        if capacity[u][v] == 0:
            adj[u].append(v)
            adj[v].append(u)
        capacity[u][v] += w
        capacity[v][u] += w

                u = parent[v]
                capacity[u][v] -= path_flow
                capacity[v][u] += path_flow
                v = parent[v]
            
            max_flow += path_flow
        
        return max_flow
    
    ford_fulkerson(s, t)
    
    # Find min-cut edges by checking which edges go from reachable to non-reachable
    visited = [False] * (n + 1)
    queue = deque([s])
    visited[s] = True
    
    while queue:
        u = queue.popleft()
        for v in adj[u]:
            if not visited[v] and capacity[u][v] > 0:
                visited[v] = True
                queue.append(v)
    
    cut_edges = []
    for u, v, w in edges:
        # If u is reachable from source but v is not, or vice versa
        if (visited[u] and not visited[v]) or (visited[v] and not visited[u]):
            if (u, v, w) not in cut_edges and (v, u, w) not in cut_edges:
                cut_edges.append((u, v, w))
    
    return cut_edges
