```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); m = int(data[ptr+1]); s = int(data[ptr+2]); t = int(data[ptr+3]); B = int(data[ptr+4])
    ptr += 5

    N = n + 1
    graph = [[] for _ in range(N)]
    to = []
    cap = []
    eu = [0]*m
    ev = [0]*m

    for i in range(m):
        u = int(data[ptr]); v = int(data[ptr+1]); w = int(data[ptr+2])
        ptr += 3
        eu[i] = u; ev[i] = v
        if u == v:
            continue
        graph[u].append(len(to)); to.append(v); cap.append(w)
        graph[v].append(len(to)); to.append(u); cap.append(w)

    if s == t:
        sys.stdout.write("NO\n")
        return

    flow = 0
    while True:
        level = [-1]*N
        level[s] = 0
        q = [s]
        for u in q:
            lu1 = level[u] + 1
            for e in graph[u]:
                if cap[e] > 0:
                    v = to[e]
                    if level[v] < 0:
                        level[v] = lu1
                        q.append(v)
        if level[t] < 0:
            break
        it = [0]*N
        stack = [s]
        path = []
        while stack:
            u = stack[-1]
            if u == t:
                f = min(cap[e] for e in path)
                cut = -1
                for i in range(len(path)):
                    e = path[i]
                    cap[e] -= f
                    cap[e ^ 1] += f
                    if cut < 0 and cap[e] == 0:
                        cut = i
                flow += f
                del stack[cut+1:]
                del path[cut:]
                continue
            gu = graph[u]
            iu = it[u]
            tgt = level[u] + 1
            L = len(gu)
            e = -1
            while iu < L:
                e = gu[iu]
                if cap[e] > 0 and level[to[e]] == tgt:
                    break
                iu += 1
            it[u] = iu
            if iu < L:
                stack.append(to[e])
                path.append(e)
            else:
                level[u] = -1
                stack.pop()
                if path:
                    path.pop()
                    it[stack[-1]] += 1

    if flow > B:
        sys.stdout.write("NO\n")
        return

    S = [False]*N
    S[s] = True
    q = [s]
    for u in q:
        for e in graph[u]:
            if cap[e] > 0:
                v = to[e]
                if not S[v]:
                    S[v] = True
                    q.append(v)

    res = []
    for i in range(m):
        if S[eu[i]] != S[ev[i]]:
            res.append(str(eu[i]) + " " + str(ev[i]))
    out = ["YES", str(len(res))]
    out += res
    sys.stdout.write("\n".join(out) + "\n")

main()
```

Final answer below.Looking at this problem: we need a minimum s-t cut in an undirected graph where edge weights are sealing costs. If the min cut ≤ B, output the cut edges; otherwise NO. Min cut = max flow (each undirected edge becomes two directed arcs of capacity w). After max flow, the set S of residual-reachable vertices from s defines the cut: all original edges with exactly one endpoint in S. Their total weight equals the max flow value (≤ B), and sealing them disconnects s from t.

I verified on the example: max flow = 10 = B, residual S = {1,2,3}, crossing edges are 2-4 and 3-4 — matching the expected output.

Implementation: iterative Dinic (BFS level graph + iterative blocking-flow DFS with current-arc pointers), which avoids recursion depth issues. Zero-weight edges crossing the cut are included (cost 0, but needed for disconnection). Self-loops are ignored (never useful to seal). If s == t, disconnection is impossible → NO.

```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    ptr = 0
    n = int(data[ptr]); m = int(data[ptr+1]); s = int(data[ptr+2]); t = int(data[ptr+3]); B = int(data[ptr+4])
    ptr += 5

    N = n + 1
    graph = [[] for _ in range(N)]
    to = []
    cap = []
    eu = [0] * m
    ev = [0] * m

    for i in range(m):
        u = int(data[ptr]); v = int(data[ptr+1]); w = int(data[ptr+2])
        ptr += 3
        eu[i] = u; ev[i] = v
        if u == v:
            continue
        graph[u].append(len(to)); to.append(v); cap.append(w)
        graph[v].append(len(to)); to.append(u); cap.append(w)

    if s == t:
        sys.stdout.write("NO\n")
        return

    flow = 0
    while True:
        level = [-1] * N
        level[s] = 0
        q = [s]
        for u in q:
            lu1 = level[u] + 1
            for e in graph[u]:
                if cap[e] > 0:
                    v = to[e]
                    if level[v] < 0:
                        level[v] = lu1
                        q.append(v)
        if level[t] < 0:
            break

        it = [0] * N
        stack = [s]
        path = []
        while stack:
            u = stack[-1]
            if u == t:
                f = min(cap[e] for e in path)
                cut = -1
                for i in range(len(path)):
                    e = path[i]
                    cap[e] -= f
                    cap[e ^ 1] += f
                    if cut < 0 and cap[e] == 0:
                        cut = i
                flow += f
                del stack[cut + 1:]
                del path[cut:]
                continue
            gu = graph[u]
            iu = it[u]
            L = len(gu)
            tgt = level[u] + 1
            e = -1
            while iu < L:
                e = gu[iu]
                if cap[e] > 0 and level[to[e]] == tgt:
                    break
                iu += 1
            it[u] = iu
            if iu < L:
                stack.append(to[e])
                path.append(e)
            else:
                level[u] = -1
                stack.pop()
                if path:
                    path.pop()
                    it[stack[-1]] += 1

    if flow > B:
        sys.stdout.write("NO\n")
        return

    S = [False] * N
    S[s] = True
    q = [s]
    for u in q:
        for e in graph[u]:
            if cap[e] > 0:
                v = to[e]
                if not S[v]:
                    S[v] = True
                    q.append(v)

    res = []
    for i in range(m):
        if S[eu[i]] != S[ev[i]]:
            res.append(str(eu[i]) + " " + str(ev[i]))

    out = ["YES", str(len(res))]
    out.extend(res)
    sys.stdout.write("\n".join(out) + "\n")

main()
```
