import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    s = int(next(iterator))
    t = int(next(iterator))
    B = int(next(iterator))
    
    # Dinic's algorithm for min-cut
    # Graph representation
    head = [-1] * (n + 1)
    to = []
    nxt = []
    cap = []
    flow = []
    
    # To keep track of original edges
    orig_edges = []
    
    def add_edge(u, v, w, idx):
        # directed edge u -> v
        nxt.append(head[u])
        head[u] = len(to)
        to.append(v)
        cap.append(w)
        flow.append(0)
        
        # directed edge v -> u
        nxt.append(head[v])
        head[v] = len(to)
        to.append(u)
        cap.append(w)
        flow.append(0)
    
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        orig_edges.append((u, v, w))
        add_edge(u, v, w, i)
        
    level = [-1] * (n + 1)
    ptr = [0] * (n + 1)
    
    def bfs():
        for i in range(1, n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                u = to[e]
                if cap[e] - flow[e] > 0 and level[u] < 0:
                    level[u] = level[v] + 1
                    q.append(u)
                e = nxt[e]
        return level[t] != -1
    
    def dfs(v, pushed):
        if pushed == 0 or v == t:
            return pushed
        e = ptr[v]
        while e != -1:
            u = to[e]
            tr = cap[e] - flow[e]
            if level[v] + 1 == level[u] and tr > 0:
                push = dfs(u, min(pushed, tr))
                if push > 0:
                    flow[e] += push
                    flow[e ^ 1] -= push
                    ptr[v] = e
                    return push
            e = nxt[e]
            ptr[v] = e
        return 0
    
    INF = 10**18
    max_flow = 0
    while bfs():
        for i in range(1, n + 1):
            ptr[i] = head[i]
        while True:
            pushed = dfs(s, INF)
            if pushed == 0:
                break
            max_flow += pushed
            if max_flow > B:
                break
        if max_flow > B:
            break
            
    if max_flow > B:
        print("NO")
        return
        
    # Reachability in residual graph from s
    visited = [False] * (n + 1)
    visited[s] = True
    q = deque([s])
    while q:
        v = q.popleft()
        e = head[v]
        while e != -1:
            u = to[e]
            if cap[e] - flow[e] > 0 and not visited[u]:
                visited[u] = True
                q.append(u)
            e = nxt[e]
            
    cut_edges = []
    for u, v, w in orig_edges:
        if (visited[u] and not visited[v]) or (visited[v] and not visited[u]):
            cut_edges.append((u, v))
            
    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(f"{u} {v}")

if __name__ == '__main__':
    solve()
