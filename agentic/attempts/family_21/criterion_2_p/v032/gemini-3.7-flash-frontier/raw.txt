import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    s = int(data[2]) - 1
    t = int(data[3]) - 1
    B = int(data[4])
    
    idx = 5
    edges_input = []
    
    class Edge:
        __slots__ = ['to', 'cap', 'flow', 'rev']
        def __init__(self, to, cap, rev):
            self.to = to
            self.cap = cap
            self.flow = 0
            self.rev = rev

    adj = [[] for _ in range(n)]
    
    def add_edge(u, v, cap):
        e1 = Edge(v, cap, None)
        e2 = Edge(u, cap, None)
        e1.rev = e2
        e2.rev = e1
        adj[u].append(e1)
        adj[v].append(e2)

    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        w = int(data[idx+2])
        idx += 3
        edges_input.append((u, v, w))
        add_edge(u, v, w)

    # Dinic's Algorithm
    level = [-1] * n
    ptr = [0] * n

    def bfs():
        for i in range(n):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            for edge in adj[v]:
                if edge.cap - edge.flow > 0 and level[edge.to] == -1:
                    level[edge.to] = level[v] + 1
                    q.append(edge.to)
        return level[t] != -1

    def dfs(v, pushed):
        if pushed == 0 or v == t:
            return pushed
        for cid in range(ptr[v], len(adj[v])):
            ptr[v] = cid
            edge = adj[v][cid]
            tr = edge.to
            if level[v] + 1 != level[tr] or edge.cap - edge.flow == 0:
                continue
            tr_pushed = dfs(tr, min(pushed, edge.cap - edge.flow))
            if tr_pushed == 0:
                continue
            edge.flow += tr_pushed
            edge.rev.flow -= tr_pushed
            return tr_pushed
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        for i in range(n):
            ptr[i] = 0
        while True:
            pushed = dfs(s, INF)
            if pushed == 0:
                break
            flow += pushed

    if flow > B:
        print("NO")
        return

    # Find reachable component from s in residual graph
    visited = [False] * n
    q = deque([s])
    visited[s] = True
    while q:
        u = q.popleft()
        for edge in adj[u]:
            if edge.cap - edge.flow > 0 and not visited[edge.to]:
                visited[edge.to] = True
                q.append(edge.to)

    cut_edges = []
    for u, v, w in edges_input:
        if visited[u] != visited[v]:
            cut_edges.append((u + 1, v + 1))

    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(f"{u} {v}")

if __name__ == '__main__':
    solve()
