import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it))
    t = int(next(it))
    B = int(next(it))

    edges = []
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))

    INF = float('inf')
    dist = [INF] * (n + 1)
    dist[s] = 0
    import heapq
    pq = [(0, s)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        if u == t:
            break
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(pq, (nd, v))

    if dist[t] > B:
        print("NO")
        return

    parent = [0] * (n + 1)
    parent_edge = [None] * (n + 1)
    visited = [False] * (n + 1)
    visited[s] = True
    stack = [s]
    while stack:
        u = stack.pop()
        if u == t:
            break
        for v, w in adj[u]:
            if not visited[v] and dist[u] + w == dist[v]:
                visited[v] = True
                parent[v] = u
                parent_edge[v] = (u, v)
                stack.append(v)

    cut_edges = []
    cur = t
    while cur != s:
        p = parent[cur]
        cut_edges.append(parent_edge[cur])
        cur = p

    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(u, v)

solve()
