We need to find the minimum cut (max-flow/min-cut) between `s` and `t` and check if it’s ≤ `B`.

We’ll use a standard max-flow algorithm like Dinic’s to compute the min-cut capacity. Then, if over budget, we print `NO`. If within budget, we output the edges on the `s`-side of the cut.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

class MaxFlow:
    def __init__(self, n):
        self.n = n
        self.adj = [[] for _ in range(n)]
    
    def add_edge(self, u, v, cap):
        self.adj[u].append([v, cap, len(self.adj[v])])
        self.adj[v].append([u, 0, len(self.adj[u]) - 1])
    
    def bfs(self, s, t):
        self.level = [-1] * self.n
        q = deque([s])
        self.level[s] = 0
        while q:
            u = q.popleft()
            for v, cap, _ in self.adj[u]:
                if cap > 0 and self.level[v] == -1:
                    self.level[v] = self.level[u] + 1
                    q.append(v)
        return self.level[t] != -1
    
    def dfs(self, u, t, f):
        if u == t:
            return f
        for i in range(self.it[u], len(self.adj[u])):
            self.it[u] = i
            v, cap, rev = self.adj[u][i]
            if cap > 0 and self.level[v] == self.level[u] + 1:
                ret = self.dfs(v, t, min(f, cap))
                if ret:
                    self.adj[u][i][1] -= ret
                    self.adj[v][rev][1] += ret
                    return ret
        return 0
    
    def max_flow(self, s, t):
        flow = 0
        INF = 10**18
        while self.bfs(s, t):
            self.it = [0] * self.n
            while True:
                f = self.dfs(s, t, INF)
                if not f:
                    break
                flow += f
        return flow
    
    def min_cut(self, s):
        visited = [False] * self.n
        q = deque([s])
        visited[s] = True
        while q:
            u = q.popleft()
            for v, cap, _ in self.adj[u]:
                if cap > 0 and not visited[v]:
                    visited[v] = True
                    q.append(v)
        return visited

def main():
    n, m, s, t, B = map(int, input().split())
    s -= 1
    t -= 1
    mf = MaxFlow(n)
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v, w))
        mf.add_edge(u, v, w)
        mf.add_edge(v, u, w)  # undirected: add capacity both ways
    
    flow = mf.max_flow(s, t)
    if flow > B:
        print("NO")
        return
    
    reachable = mf.min_cut(s)
    
    cut_edges = []
    for u, v, w in edges:
        if reachable[u] != reachable[v]:
            cut_edges.append((u+1, v+1))
    
    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(u, v)

if __name__ == "__main__":
    main()
```
