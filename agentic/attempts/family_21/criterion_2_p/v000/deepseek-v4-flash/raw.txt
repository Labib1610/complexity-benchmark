import sys
import heapq

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it))
    t = int(next(it))
    B = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        edges.append((u, v, w))
        adj[u].append((v, w, idx))
        adj[v].append((u, w, idx))
    
    s -= 1
    t -= 1
    
    INF = 10**18
    
    # Dijkstra from s
    dist_s = [INF] * n
    dist_s[s] = 0
    pq = [(0, s)]
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist_s[u]:
            continue
        for v, w, _ in adj[u]:
            nd = d + w
            if nd < dist_s[v]:
                dist_s[v] = nd
                heapq.heappush(pq, (nd, v))
    
    # Dijkstra from t
    dist_t = [INF] * n
    dist_t[t] = 0
    pq = [(0, t)]
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist_t[u]:
            continue
        for v, w, _ in adj[u]:
            nd = d + w
            if nd < dist_t[v]:
                dist_t[v] = nd
                heapq.heappush(pq, (nd, v))
    
    # Build shortest path DAG from s to t
    dag_adj = [[] for _ in range(n)]
    dag_edges = []
    for idx, (u, v, w) in enumerate(edges):
        if dist_s[u] + w + dist_t[v] == dist_s[t]:
            dag_adj[u].append((v, w, idx))
            dag_edges.append((u, v, w, idx))
        if dist_s[v] + w + dist_t[u] == dist_s[t]:
            dag_adj[v].append((u, w, idx))
            dag_edges.append((v, u, w, idx))
    
    # If s and t already disconnected in original graph?
    if dist_s[t] == INF:
        print("YES")
        print(0)
        return
    
    # We need to find a minimum s-t cut in the DAG (all edges are on shortest paths)
    # Use max flow (Dinic) on the DAG with capacities = edge weights
    # Since we only need to cut edges, vertices are not capacitated.
    
    class Dinic:
        def __init__(self, n):
            self.n = n
            self.adj = [[] for _ in range(n)]
        
        def add_edge(self, u, v, cap):
            self.adj[u].append([v, cap, len(self.adj[v])])
            self.adj[v].append([u, 0, len(self.adj[u]) - 1])
        
        def bfs(self, s, t):
            level = [-1] * self.n
            q = [s]
            level[s] = 0
            for u in q:
                for v, cap, rev in self.adj[u]:
                    if cap > 0 and level[v] == -1:
                        level[v] = level[u] + 1
                        q.append(v)
            self.level = level
            return level[t] != -1
        
        def dfs(self, u, t, f):
            if u == t:
                return f
            for i in range(self.it[u], len(self.adj[u])):
                self.it[u] = i
                v, cap, rev = self.adj[u][i]
                if cap > 0 and self.level[v] == self.level[u] + 1:
                    ret = self.dfs(v, t, min(f, cap))
                    if ret:
                        self.adj[u][i][1] -= ret
                        self.adj[v][rev][1] += ret
                        return ret
            return 0
        
        def max_flow(self, s, t):
            flow = 0
            INF_FLOW = 10**18
            while self.bfs(s, t):
                self.it = [0] * self.n
                while True:
                    f = self.dfs(s, t, INF_FLOW)
                    if not f:
                        break
                    flow += f
            return flow
    
    dinic = Dinic(n)
    for u, v, w, idx in dag_edges:
        dinic.add_edge(u, v, w)
    
    min_cut_cost = dinic.max_flow(s, t)
    
    if min_cut_cost > B:
        print("NO")
        return
    
    # Now find the actual cut edges: after max flow, edges with cap==0 in residual from reachable side
    # BFS from s in residual graph
    reachable = [False] * n
    q = [s]
    reachable[s] = True
    for u in q:
        for v, cap, rev in dinic.adj[u]:
            if cap > 0 and not reachable[v]:
                reachable[v] = True
                q.append(v)
    
    removed = []
    for u, v, w, idx in dag_edges:
        if reachable[u] and not reachable[v]:
            removed.append((u, v))
    
    # The cut edges found should have total cost == min_cut_cost
    # Output
    print("YES")
    print(len(removed))
    for u, v in removed:
        print(u + 1, v + 1)

if __name__ == "__main__":
    solve()
