import sys
sys.setrecursionlimit(1 << 25)

class Dinic:
    __slots__ = ('n', 'g', 'level', 'it')
    def __init__(self, n):
        self.n = n
        self.g = [[] for _ in range(n)]
    def add_edge(self, fr, to, cap):
        fwd = [to, cap, None]
        rev = [fr, 0, fwd]
        fwd[2] = rev
        self.g[fr].append(fwd)
        self.g[to].append(rev)
    def bfs(self, s, t):
        level = [-1]*self.n
        q = [s]
        level[s]=0
        for v in q:
            for to, cap, rev in self.g[v]:
                if cap>0 and level[to]==-1:
                    level[to]=level[v]+1
                    q.append(to)
        self.level=level
        return level[t]!=-1
    def dfs(self, v, t, f):
        if v==t:
            return f
        for i in range(self.it[v], len(self.g[v])):
            self.it[v]=i
            to, cap, rev = self.g[v][i]
            if cap>0 and self.level[v]+1==self.level[to]:
                ret=self.dfs(to, t, min(f, cap))
                if ret:
                    self.g[v][i][1]-=ret
                    rev[1]+=ret
                    return ret
        return 0
    def max_flow(self, s, t):
        flow=0
        INF=10**18
        while self.bfs(s,t):
            self.it=[0]*self.n
            while True:
                f=self.dfs(s,t,INF)
                if not f: break
                flow+=f
        return flow

def solve():
    data=sys.stdin.read().strip().split()
    if not data: 
        return
    it=iter(data)
    n=int(next(it)); m=int(next(it)); s=int(next(it))-1; t=int(next(it))-1; B=int(next(it))
    edges=[]
    dinic=Dinic(n)
    for _ in range(m):
        u=int(next(it))-1; v=int(next(it))-1; w=int(next(it))
        edges.append((u,v,w))
        # undirected -> two directed edges
        dinic.add_edge(u,v,w)
        dinic.add_edge(v,u,w)
    mincut=dinic.max_flow(s,t)
    if mincut> B:
        print("NO")
        return
    # find reachable vertices in residual graph
    from collections import deque
    q=deque([s])
    reachable=[False]*n
    reachable[s]=True
    while q:
        v=q.popleft()
        for to, cap, rev in dinic.g[v]:
            if cap>0 and not reachable[to]:
                reachable[to]=True
                q.append(to)
    cut_edges=[]
    total=0
    for u,v,w in edges:
        if reachable[u]!=reachable[v]:
            cut_edges.append((u+1,v+1))
            total+=w
    # total should equal mincut
    print("YES")
    print(len(cut_edges))
    for u,v in cut_edges:
        print(u, v)

if __name__=="__main__":
    solve()
