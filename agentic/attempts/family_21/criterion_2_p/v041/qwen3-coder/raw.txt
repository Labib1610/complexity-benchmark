import sys
from collections import defaultdict, deque

def read_input():
    line = sys.stdin.readline().strip()
    n, m, s, t, cap = map(int, line.split())
    edges = []
    graph = defaultdict(list)
    for _ in range(m):
        u, v, w = map(int, sys.stdin.readline().split())
        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))
    return n, m, s, t, cap, edges, graph

def bfs_check(graph, n, s, t, blocked_edges):
    visited = [False] * (n + 1)
    queue = deque([s])
    visited[s] = True
    
    while queue:
        node = queue.popleft()
        if node == t:
            return False  # Still connected
        
        for neighbor, _ in graph[node]:
            edge_blocked = (node, neighbor) in blocked_edges or (neighbor, node) in blocked_edges
            if not edge_blocked and not visited[neighbor]:
                visited[neighbor] = True
                queue.append(neighbor)
    
    return True  # Disconnected

def find_min_cut(edges, n, s, t):
    # Ford-Fulkerson implementation for min-cut
    # Build residual graph
    res_graph = defaultdict(lambda: defaultdict(int))
    edge_map = {}  # Maps (u,v) to original index
    
    for i, (u, v, w) in enumerate(edges):
        res_graph[u][v] += w
        res_graph[v][u] += w
        edge_map[(u, v)] = edge_map.get((u, v), []) + [i]
        edge_map[(v, u)] = edge_map.get((v, u), []) + [i]
    
    def bfs_path(res_graph, s, t, parent):
        visited = [False] * (n + 1)
        queue = deque([s])
        visited[s] = True
        
        while queue:
            u = queue.popleft()
            
            for v in res_graph[u]:
                if not visited[v] and res_graph[u][v] > 0:
                    queue.append(v)
                    visited[v] = True
                    parent[v] = u
                    if v == t:
                        return True
        return False
    
    parent = [-1] * (n + 1)
    max_flow = 0
    
    while bfs_path(res_graph, s, t, parent):
        path_flow = float('inf')
        v = t
        while v != s:
            u = parent[v]
            path_flow = min(path_flow, res_graph[u][v])
            v = u
        
        v = t
        while v != s:
            u = parent[v]
            res_graph[u][v] -= path_flow
            res_graph[v][u] += path_flow
            v = u
            
        max_flow += path_flow
    
    # Find min cut
    visited = [False] * (n + 1)
    queue = deque([s])
    visited[s] = True
    
    while queue:
        u = queue.popleft()
        for v in res_graph[u]:
            if not visited[v] and res_graph[u][v] > 0:
                visited[v] = True
                queue.append(v)
    
    min_cut_edges = []
    cut_weight = 0
    used_indices = set()
    
    for i, (u, v, w) in enumerate(edges):
        # Check if this edge crosses the cut
        if visited[u] and not visited[v]:
            if i not in used_indices:
                min_cut_edges.append((u, v, w, i))
                cut_weight += w
                used_indices.add(i)
        elif visited[v] and not visited[u]:
            if i not in used_indices:
                min_cut_edges.append((v, u, w, i))  # Keep consistent direction
                cut_weight += w
                used_indices.add(i)
                
    return cut_weight, min_cut_edges
