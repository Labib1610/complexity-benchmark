import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    s = int(next(iterator))
    t = int(next(iterator))
    reserve = int(next(iterator))
    
    # Store edges to reconstruct cut
    orig_edges = []
    
    # Adjacency list for Dinic's algorithm
    # Edge structure: [to, cap, flow, rev_index]
    graph = [[] for _ in range(n + 1)]
    
    def add_edge(u, v, cap):
        idx_u = len(graph[u])
        idx_v = len(graph[v])
        graph[u].append([v, cap, 0, idx_v])
        graph[v].append([u, cap, 0, idx_u])
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        orig_edges.append((u, v, w))
        add_edge(u, v, w)
        
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for v, cap, flow, rev in graph[u]:
                if cap - flow > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] >= 0

    ptr = [0] * (n + 1)
    
    def dfs(u, pushed):
        if pushed == 0:
            return 0
        if u == t:
            return pushed
        for cid in range(ptr[u], len(graph[u])):
            ptr[u] = cid
            v, cap, flow, rev = graph[u][cid]
            tr = cap - flow
            if level[u] + 1 != level[v] or tr <= 0:
                continue
            push = dfs(v, min(pushed, tr))
            if push == 0:
                continue
            graph[u][cid][2] += push
            graph[v][rev][2] -= push
            return push
        return 0

    max_flow = 0
    INF = 10**18
    while bfs():
        for i in range(n + 1):
            ptr[i] = 0
        while True:
            pushed = dfs(s, INF)
            if pushed == 0:
                break
            max_flow += pushed

    if max_flow > reserve:
        print("NO")
        return

    # Find reachable nodes from s in residual graph
    visited = [False] * (n + 1)
    q = deque([s])
    visited[s] = True
    while q:
        u = q.popleft()
        for v, cap, flow, rev in graph[u]:
            if cap - flow > 0 and not visited[v]:
                visited[v] = True
                q.append(v)

    cut_edges = []
    for u, v, w in orig_edges:
        if visited[u] != visited[v]:
            cut_edges.append((u, v))

    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(f"{u} {v}")

if __name__ == '__main__':
    solve()
