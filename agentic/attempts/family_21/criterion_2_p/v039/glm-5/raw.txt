```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
        s = int(next(iterator))
        t = int(next(iterator))
        B = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        w = int(next(iterator))
        adj[u].append((v, i))
        adj[v].append((u, i))
        edges.append((u, v, w))

    # BFS to find shortest path by number of edges
    parent = [-1] * (n + 1)
    parent_edge_id = [-1] * (n + 1)
    q = deque([s])
    parent[s] = s
    
    found = False
    while q:
        u = q.popleft()
        if u == t:
            found = True
            break
        for v, edge_id in adj[u]:
            if parent[v] == -1:
                parent[v] = u
                parent_edge_id[v] = edge_id
                q.append(v)

    if not found:
        print("YES")
        print(0)
        return

    # Reconstruct path
    path_edges = []
    curr = t
    while curr != s:
        p = parent[curr]
        edge_id = parent_edge_id[curr]
        path_edges.append(edge_id)
        curr = p
    
    # Calculate total cost
    total_cost = 0
    for edge_id in path_edges:
        u, v, w = edges[edge_id]
        total_cost += w
    
    if total_cost <= B:
        print("YES")
        print(len(path_edges))
        for edge_id in path_edges:
            u, v, w = edges[edge_id]
            print(u, v)
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
