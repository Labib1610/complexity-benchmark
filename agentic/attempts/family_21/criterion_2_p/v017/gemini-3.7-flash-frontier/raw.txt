import sys
from collections import deque

def solve():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    s = int(data[2])
    t = int(data[3])
    B = int(data[4])
    
    # Adjacency list for Dinic
    # Each entry in adj[u]: [to, cap, flow, rev_index]
    head = [[] for _ in range(n + 1)]
    edges = []
    
    idx = 5
    for i in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        w = int(data[idx+2])
        idx += 3
        
        edges.append((u, v, w))
        
        # Add undirected edge
        e1 = [v, w, 0, len(head[v])]
        e2 = [u, w, 0, len(head[u])]
        head[u].append(e1)
        head[v].append(e2)
        
    # Dinic's Algorithm
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            u = q.popleft()
            for v, cap, flow, _ in head[u]:
                if cap - flow > 0 and level[v] == -1:
                    level[v] = level[u] + 1
                    q.append(v)
        return level[t] != -1

    ptr = [0] * (n + 1)
    
    def dfs(u, pushed):
        if pushed == 0 or u == t:
            return pushed
        for cid in range(ptr[u], len(head[u])):
            ptr[u] = cid
            edge = head[u][cid]
            v, cap, flow, rev = edge
            tr = cap - flow
            if level[u] + 1 != level[v] or tr == 0:
                continue
            push = dfs(v, min(pushed, tr))
            if push == 0:
                continue
            edge[2] += push
            head[v][rev][2] -= push
            return push
        return 0

    max_flow = 0
    INF = 10**18
    while bfs():
        ptr = [0] * (n + 1)
        while True:
            pushed = dfs(s, INF)
            if pushed == 0:
                break
            max_flow += pushed
            if max_flow > B:
                break
        if max_flow > B:
            break
            
    if max_flow > B:
        print("NO")
        return
        
    # Find reachable nodes from s in residual graph
    visited = [False] * (n + 1)
    visited[s] = True
    q = deque([s])
    while q:
        u = q.popleft()
        for v, cap, flow, _ in head[u]:
            if cap - flow > 0 and not visited[v]:
                visited[v] = True
                q.append(v)
                
    # Collect cut edges
    cut_edges = []
    for u, v, w in edges:
        if visited[u] != visited[v]:
            cut_edges.append((u, v))
            
    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(f"{u} {v}")

if __name__ == '__main__':
    solve()
