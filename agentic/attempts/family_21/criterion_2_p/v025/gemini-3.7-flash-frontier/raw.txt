import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    s = int(next(iterator)) - 1
    t = int(next(iterator)) - 1
    B = int(next(iterator))

    # Graph representation for Dinic's algorithm
    # Each undirected edge will have forward and backward directed edges
    head = [-1] * n
    to = []
    cap = []
    flow = []
    nxt = []
    edge_orig_idx = []

    orig_edges = []

    def add_edge(u, v, w, idx):
        # u -> v
        nxt.append(head[u])
        to.append(v)
        cap.append(w)
        flow.append(0)
        edge_orig_idx.append(idx)
        head[u] = len(to) - 1

        # v -> u
        nxt.append(head[v])
        to.append(u)
        cap.append(w)
        flow.append(0)
        edge_orig_idx.append(idx)
        head[v] = len(to) - 1

    for i in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        w = int(next(iterator))
        orig_edges.append((u + 1, v + 1, w))
        add_edge(u, v, w, i)

    # Dinic's Algorithm
    level = [-1] * n
    ptr = [0] * n

    def bfs():
        for i in range(n):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                if cap[e] - flow[e] > 0 and level[to[e]] == -1:
                    level[to[e]] = level[v] + 1
                    q.append(to[e])
                e = nxt[e]
        return level[t] != -1

    def dfs(v, pushed):
        if pushed == 0 or v == t:
            return pushed
        e = ptr[v]
        while e != -1:
            tr = to[e]
            if level[v] + 1 == level[tr] and cap[e] - flow[e] > 0:
                tr_pushed = dfs(tr, min(pushed, cap[e] - flow[e]))
                if tr_pushed > 0:
                    flow[e] += tr_pushed
                    flow[e ^ 1] -= tr_pushed
                    return tr_pushed
            e = nxt[e]
            ptr[v] = e
        return 0

    total_flow = 0
    INF = 10**18
    while bfs():
        for i in range(n):
            ptr[i] = head[i]
        while True:
            pushed = dfs(s, INF)
            if pushed == 0:
                break
            total_flow += pushed

    if total_flow > B:
        print("NO")
        return

    # Find reachable component from s in residual graph
    visited = [False] * n
    visited[s] = True
    q = deque([s])
    while q:
        v = q.popleft()
        e = head[v]
        while e != -1:
            if cap[e] - flow[e] > 0 and not visited[to[e]]:
                visited[to[e]] = True
                q.append(to[e])
            e = nxt[e]

    # Find edges crossing the cut
    cut_edges = []
    for i in range(m):
        u, v, w = orig_edges[i]
        u_idx = u - 1
        v_idx = v - 1
        if visited[u_idx] != visited[v_idx]:
            cut_edges.append((u, v))

    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(f"{u} {v}")

if __name__ == '__main__':
    solve()
