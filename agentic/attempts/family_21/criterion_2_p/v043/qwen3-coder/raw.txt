import sys
from collections import deque, defaultdict
import heapq

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    B = int(data[idx])
    idx += 1
    
    edges = []
    graph = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((u, v, w))
        graph[u].append((v, w))
        graph[v].append((u, w))
    
    # We need to find the minimum cut between s and t in the graph
    # This is equivalent to finding the minimum s-t cut, which equals the maximum flow from s to t
    # We'll use Edmonds-Karp algorithm for max flow
    
    def bfs_flow(capacity, source, sink, parent):
        visited = [False] * (n + 1)
        queue = deque([source])
        visited[source] = True
        
        while queue:
            u = queue.popleft()
            
            for v in range(1, n+1):
                if not visited[v] and capacity[u][v] > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    if v == sink:
                        return True
        return False
    
    def max_flow(capacity, source, sink):
        parent = [-1] * (n + 1)
        total_flow = 0
        
        while bfs_flow(capacity, source, sink, parent):
            path_flow = float('inf')
            s_node = sink
            
            while s_node != source:
                path_flow = min(path_flow, capacity[parent[s_node]][s_node])
                s_node = parent[s_node]
                
            v_node = sink
            while v_node != source:
                u_node = parent[v_node]
                capacity[u_node][v_node] -= path_flow
                capacity[v_node][u_node] += path_flow
                v_node = parent[v_node]
                
            total_flow += path_flow
            
        return total_flow
    
    # Build capacity matrix
    capacity = [[0] * (n + 1) for _ in range(n + 1)]
    edge_index_map = {}
    
    for i, (u, v, w) in enumerate(edges):
        capacity[u][v] += w
        capacity[v][u] += w
        edge_index_map[(u, v)] = edge_index_map.get((u, v), 0) + w
        edge_index_map[(v, u)] = edge_index_map.get((v, u), 0) + w
    
    # Make a copy for finding the actual cut
    original_capacity = [row[:] for row in capacity]
    
    maxflow = max_flow(capacity, s, t)
    
    if maxflow > B:
        print("NO")
        return
    
    # Now we need to find which edges form the minimum cut
    # Run BFS to find reachable nodes from s in residual graph
    visited_cut = [False] * (n + 1)
    queue_cut = deque([s])
    visited_cut[s] = True
    
    while queue_cut:
        node = queue_cut.popleft()
        for neighbor in range(1, n+1):
            if not visited_cut[neighbor] and capacity[node][neighbor] > 0:
                visited_cut[neighbor] = True
                queue_cut.append(neighbor)
    
    # Find edges crossing the cut
    cut_edges = []
    total_cost = 0
    
    added_edges = set()
    
    for u, v, w in edges:
        # Check if this edge crosses the cut
        if visited_cut[u] and not visited_cut[v]:
            if (u, v) not in added_edges and (v, u) not in added_edges:
                cut_edges.append((u, v))
                total_cost += w
                added_edges.add((u, v))
        elif visited_cut[v] and not visited_cut[u]:
            if (u, v) not in added_edges and (v, u) not in added_edges:
                cut_edges.append((v, u)) # Keep consistent orientation
                total_cost += w
                added_edges.add((v, u))
    
    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(u, v)

if __name__ == "__main__":
    main()
