from collections import deque
import sys

def bfs_check_disconnected(n, edges, s, t):
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # BFS from s to see if we can reach t
    visited = [False] * (n + 1)
    queue = deque([s])
    visited[s] = True
    
    while queue:
        node = queue.popleft()
        if node == t:
            return False  # Still connected
        for neighbor in adj[node]:
            if not visited[neighbor]:
                visited[neighbor] = True
                queue.append(neighbor)
    
    return True  # Disconnected

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    s = int(data[idx])
    idx += 1
    t = int(data[idx])
    idx += 1
    B = int(data[idx])
    idx += 1
    
    edges = []
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        w = int(data[idx])
        idx += 1
        edges.append((u, v, w))
    
    # Try all 2^m subsets
    for mask in range(1 << m):
        total_energy = 0
        suppressed = []
        remaining_edges = []
        
        for i in range(m):
            if mask & (1 << i):
                suppressed.append(i)
                total_energy += edges[i][2]
            else:
                remaining_edges.append((edges[i][0], edges[i][1]))
        
        if total_energy > B:
            continue
            
        # Check if s and t are disconnected in the remaining graph
        if bfs_check_disconnected(n, remaining_edges, s, t):
            print("YES")
            print(len(suppressed))
            for i in suppressed:
                print(edges[i][0], edges[i][1])
            return
    
    print("NO")

if __name__ == "__main__":
    main()
