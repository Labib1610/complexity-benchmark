```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it))
    t = int(next(it))
    B = int(next(it))

    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, len(edges)))
        adj[v].append((u, len(edges)))
        edges.append((u, v, w))

    INF = B + 1
    dist = [INF] * (n + 1)
    dist[s] = 0
    in_queue = [False] * (n + 1)
    parent_edge = [-1] * (n + 1)

    dq = deque([s])
    in_queue[s] = True

    while dq:
        u = dq.popleft()
        in_queue[u] = False
        du = dist[u]
        for v, eid in adj[u]:
            w = edges[eid][2]
            if du + w < dist[v]:
                dist[v] = du + w
                parent_edge[v] = eid
                if not in_queue[v]:
                    in_queue[v] = True
                    dq.append(v)

    if dist[t] > B:
        sys.stdout.write("NO\n")
        return

    chosen = set()
    cur = t
    while cur != s:
        eid = parent_edge[cur]
        chosen.add(eid)
        u, v, w = edges[eid]
        cur = v if cur == u else u

    out = ["YES", str(len(chosen))]
    for eid in chosen:
        u, v, w = edges[eid]
        out.append(f"{u} {v}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
```
