import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it))
    t = int(next(it))
    B = int(next(it))

    edges = []
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))

    INF = float('inf')
    dist = [INF] * (n + 1)
    dist[s] = 0
    visited = [False] * (n + 1)

    for _ in range(n):
        u = -1
        best = INF
        for i in range(1, n + 1):
            if not visited[i] and dist[i] < best:
                best = dist[i]
                u = i
        if u == -1 or u == t:
            break
        visited[u] = True
        for v, w in adj[u]:
            if not visited[v] and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w

    if dist[t] > B:
        sys.stdout.write("NO\n")
        return

    parent = [0] * (n + 1)
    parent_edge = [None] * (n + 1)
    dist2 = [INF] * (n + 1)
    dist2[s] = 0
    visited2 = [False] * (n + 1)

    for _ in range(n):
        u = -1
        best = INF
        for i in range(1, n + 1):
            if not visited2[i] and dist2[i] < best:
                best = dist2[i]
                u = i
        if u == -1:
            break
        visited2[u] = True
        for v, w in adj[u]:
            if not visited2[v] and dist2[u] + w < dist2[v]:
                dist2[v] = dist2[u] + w
                parent[v] = u
                parent_edge[v] = (u, v)

    cut_edges = []
    cur = t
    while cur != s:
        p = parent[cur]
        cut_edges.append((p, cur))
        cur = p

    out = ["YES", str(len(cut_edges))]
    for u, v in cut_edges:
        out.append(f"{u} {v}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
