import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it))
    t = int(next(it))
    B = int(next(it))

    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w, len(edges)))
        adj[v].append((u, w, len(edges)))
        edges.append((u, v, w))

    INF = float('inf')
    dist = [INF] * (n + 1)
    dist[s] = 0
    parent = [(-1, -1)] * (n + 1)

    import heapq
    pq = [(0, s)]
    while pq:
        d, u = heapq.heappop(pq)
        if d != dist[u]:
            continue
        if u == t:
            break
        for v, w, eid in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                parent[v] = (u, eid)
                heapq.heappush(pq, (nd, v))

    if dist[t] > B:
        sys.stdout.write("NO\n")
        return

    path_edges = []
    cur = t
    while cur != s:
        p, eid = parent[cur]
        path_edges.append(eid)
        cur = p

    out = ["YES"]
    out.append(str(len(path_edges)))
    for eid in path_edges:
        u, v, w = edges[eid]
        out.append(f"{u} {v}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
