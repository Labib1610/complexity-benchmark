import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    s = int(input_data[2])
    t = int(input_data[3])
    B = int(input_data[4])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 5
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        adj[u].append((v, i))
        adj[v].append((u, i))
        edges.append((u, v, w))
        
    if s == t:
        print("YES")
        print(0)
        return

    # BFS from s
    dist_s = [-1] * (n + 1)
    q = [s]
    dist_s[s] = 0
    head = 0
    while head < len(q):
        u = q[head]
        head += 1
        for v, edge_idx in adj[u]:
            if dist_s[v] == -1:
                dist_s[v] = dist_s[u] + 1
                q.append(v)
                
    if dist_s[t] == -1:
        print("YES")
        print(0)
        return
        
    # BFS from t
    dist_t = [-1] * (n + 1)
    q = [t]
    dist_t[t] = 0
    head = 0
    while head < len(q):
        u = q[head]
        head += 1
        for v, edge_idx in adj[u]:
            if dist_t[v] == -1:
                dist_t[v] = dist_t[u] + 1
                q.append(v)

    D = dist_s[t]
    
    # Find edges on shortest paths
    sp_edges = []
    for i, (u, v, w) in enumerate(edges):
        if dist_s[u] != -1 and dist_t[v] != -1 and dist_s[u] + dist_t[v] + 1 == D:
            sp_edges.append((w, i))
        elif dist_s[v] != -1 and dist_t[u] != -1 and dist_s[v] + dist_t[u] + 1 == D:
            sp_edges.append((w, i))
            
    sp_edges.sort()
    
    total_cost = 0
    selected_indices = []
    for w, i in sp_edges:
        if total_cost + w <= B:
            total_cost += w
            selected_indices.append(i)
        else:
            break
            
    if total_cost <= B:
        print("YES")
        print(len(selected_indices))
        for i in selected_indices:
            u, v, w = edges[i]
            print(u, v)
    else:
        print("NO")

if __name__ == '__main__':
    solve()
