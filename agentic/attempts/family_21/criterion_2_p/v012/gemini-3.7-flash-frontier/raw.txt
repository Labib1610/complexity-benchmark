import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    s = int(input_data[2])
    t = int(input_data[3])
    B = int(input_data[4])
    
    idx = 5
    edges_info = []
    
    # Dinic's Algorithm
    head = [-1] * (n + 1)
    to = []
    cap = []
    nxt = []
    
    def add_edge(u, v, w):
        # u -> v
        to.append(v)
        cap.append(w)
        nxt.append(head[u])
        head[u] = len(to) - 1
        
        # v -> u
        to.append(u)
        cap.append(w)
        nxt.append(head[v])
        head[v] = len(to) - 1

    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        edges_info.append((u, v, w))
        add_edge(u, v, w)
        
    level = [-1] * (n + 1)
    
    def bfs():
        for i in range(1, n + 1):
            level[i] = -1
        level[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            e = head[v]
            while e != -1:
                u = to[e]
                if cap[e] > 0 and level[u] < 0:
                    level[u] = level[v] + 1
                    q.append(u)
                e = nxt[e]
        return level[t] != -1

    ptr = [0] * (n + 1)
    
    def dfs(v, pushed):
        if pushed == 0:
            return 0
        if v == t:
            return pushed
        e = ptr[v]
        while e != -1:
            u = to[e]
            tr = level[u]
            if tr == level[v] + 1 and cap[e] > 0:
                tr_pushed = dfs(u, min(pushed, cap[e]))
                if tr_pushed > 0:
                    cap[e] -= tr_pushed
                    cap[e ^ 1] += tr_pushed
                    return tr_pushed
            e = nxt[e]
            ptr[v] = e
        return 0

    flow = 0
    INF = 10**18
    while bfs():
        for i in range(1, n + 1):
            ptr[i] = head[i]
        while True:
            pushed = dfs(s, INF)
            if pushed == 0:
                break
            flow += pushed

    if flow > B:
        print("NO")
        return

    # Find reachable component from s in residual graph
    visited = [False] * (n + 1)
    visited[s] = True
    q = deque([s])
    while q:
        v = q.popleft()
        e = head[v]
        while e != -1:
            u = to[e]
            if cap[e] > 0 and not visited[u]:
                visited[u] = True
                q.append(u)
            e = nxt[e]

    cut_edges = []
    for u, v, w in edges_info:
        if (visited[u] and not visited[v]) or (visited[v] and not visited[u]):
            cut_edges.append((u, v))

    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(f"{u} {v}")

if __name__ == '__main__':
    solve()
