import sys
from collections import defaultdict, deque

def find_min_cut(edges, n, s, t):
    # Create adjacency list
    graph = defaultdict(list)
    capacity = {}
    
    for u, v, w in edges:
        graph[u].append(v)
        graph[v].append(u)
        capacity[(u, v)] = capacity.get((u, v), 0) + w
        capacity[(v, u)] = capacity.get((v, u), 0) + w
    
    def bfs(source, sink, parent):
        visited = [False] * (n + 1)
        queue = deque([source])
        visited[source] = True
        
        while queue:
            u = queue.popleft()
            
            for v in graph[u]:
                if not visited[v] and capacity.get((u, v), 0) > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    
                    if v == sink:
                        return True
        return False
    
    def ford_fulkerson(source, sink):
        parent = [-1] * (n + 1)
        max_flow = 0
        
        while bfs(source, sink, parent):
            path_flow = float('inf')
            s_node = sink
            
            while s_node != source:
                path_flow = min(path_flow, capacity[(parent[s_node], s_node)])
                s_node = parent[s_node]
            
            v = sink
            while v != source:
                u = parent[v]
                capacity[(u, v)] -= path_flow
                capacity[(v, u)] += path_flow
                v = parent[v]
            
            max_flow += path_flow
        
        return max_flow
    
    return ford_fulkerson(s, t)

def find_edges_in_min_cut(edges, n, s, t):
    # Create residual graph
    graph = defaultdict(list)
    capacity = {}
    edge_map = {}  # map (u,v) to original edges
    
    for i, (u, v, w) in enumerate(edges):
        graph[u].append(v)
        graph[v].append(u)
        capacity[(u, v)] = capacity.get((u, v), 0) + w
        capacity[(v, u)] = capacity.get((v, u), 0) + w
        edge_map[(u, v)] = edge_map.get((u, v), []) + [(u, v, w)]
        edge_map[(v, u)] = edge_map.get((v, u), []) + [(v, u, w)]
    
    # Ford-Fulkerson with BFS to find min cut
    parent = [-1] * (n + 1)
    
    def bfs(source, sink):
        visited = [False] * (n + 1)
        queue = deque([source])
        visited[source] = True
        parent = [-1] * (n + 1)
        
        while queue:
            u = queue.popleft()
            
            for v in graph[u]:
                if not visited[v] and capacity.get((u, v), 0) > 0:
                    visited[v] = True
                    parent[v] = u
                    queue.append(v)
                    
                    if v == sink:
                        return True, parent
        return False, parent
    
    # Find max flow
    max_flow = 0
    while True:
        found, parent = bfs(s, t)
        if not found:
            break
            
        # Find minimum capacity along the path
        path_flow = float('inf')
        v = t
        while v != s:
            u = parent[v]
            path_flow = min(path_flow, capacity[(u, v)])
            v = parent[v]
        
        # Update capacities
        v = t
        while v != s:
            u = parent[v]
            capacity[(u, v)] -= path_flow
            capacity[(v, u)] += path_flow
            v = parent[v]
        
        max_flow += path_flow
    
    # Find which vertices are reachable from source
    visited = [False] * (n + 1)
    queue = deque([s])
    visited[s] = True
    
    while queue:
        u = queue.popleft()
        for v in graph[u]:
            if not visited[v] and capacity.get((u, v), 0) > 0:
                visited[v] = True
                queue.append(v)
    
    # Find edges crossing the cut
    cut_edges = []
    for u, v, w in edges:
        # Check if u is reachable from s but v is not (or vice versa)
        if visited[u] and not visited[v]:
            cut_edges.append((u, v))
        elif visited[v] and not visited[u]:
            cut_edges.append((v, u))
    
    # But we need to be more careful about multiple edges
    # Let's recompute properly
    cut_edges_final = []
    visited_set = set()
    for i in range(1, n+1):
        if visited[i]:
            visited_set.add(i)
    
    # For each original edge, check if it crosses the cut
    added = set()
    for u, v, w in edges:
        if ((u in visited_set) and (v not in visited_set)) or ((v in visited_set) and (u not in visited_set)):
            if (u, v) not in added and (v, u) not in added:
                cut_edges_final.append((u, v, w))
                added.add((u, v))
                added.add((v, u))
                
    return cut_edges_final, max_flow

def main():
    try:
        line = input().split()
        n, m, s, t, B = map(int, line)
    except EOFError:
        print("NO")
        return
        
    edges = []
    for _ in range(m):
        u, v, w = map(int, input().split())
        edges.append((u, v, w))
    
    # Make a copy for finding actual cut edges
    cut_edges, min_cost = find_edges_in_min_cut(edges, n, s, t)
    
    if min_cost <= B:
        print("YES")
        print(len(cut_edges))
        for u, v, w in cut_edges:
            print(u, v)
    else:
        print("NO")

if __name__ == "__main__":
    main()
