import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it))
    t = int(next(it))
    B = int(next(it))

    edges = []
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        edges.append((u, v, w))
        adj[u].append((v, w))
        adj[v].append((u, w))

    INF = float('inf')
    dist = [INF] * (n + 1)
    dist[s] = 0
    import heapq
    pq = [(0, s)]
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                pq.append((nd, v))

    if dist[t] <= B:
        out = ["YES"]
        cut = []
        for idx, (u, v, w) in enumerate(edges):
            if dist[u] + w == dist[v] or dist[v] + w == dist[u]:
                cut.append((u, v))
        out.append(str(len(cut)))
        for u, v in cut:
            out.append(f"{u} {v}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
