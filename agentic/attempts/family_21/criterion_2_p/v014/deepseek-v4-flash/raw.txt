We need to find a cut separating two given vertices with total edge weight ≤ a specified limit.  
This is exactly the **minimum s-t cut** problem — solvable via max-flow.

If the min-cut cost is greater than the limit, → `NO`.  
Otherwise, we find the edges crossing that cut and output them.

We'll implement using Edmonds–Karp (BFS-based) for maximum flow since N and M are not extremely large, but still efficient enough.

```python
from collections import deque
import sys

def edmonds_karp(n, graph, source, sink):
    flow = 0
    parent = [-1] * n
    capacity = [row[:] for row in graph]
    
    while True:
        parent = [-1] * n
        parent[source] = source
        q = deque([source])
        while q and parent[sink] == -1:
            u = q.popleft()
            for v in range(n):
                if parent[v] == -1 and capacity[u][v] > 0:
                    parent[v] = u
                    q.append(v)
                    if v == sink:
                        break
        if parent[sink] == -1:
            break
        # find bottleneck
        v = sink
        aug = float('inf')
        while v != source:
            u = parent[v]
            aug = min(aug, capacity[u][v])
            v = u
        # update capacities
        v = sink
        while v != source:
            u = parent[v]
            capacity[u][v] -= aug
            capacity[v][u] += aug
            v = u
        flow += aug
    return flow, capacity, parent

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it)) - 1
    t = int(next(it)) - 1
    limit = int(next(it))

    graph = [[0] * n for _ in range(n)]
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        w = int(next(it))
        graph[u][v] += w
        graph[v][u] += w
        edges.append((u, v, w))

    mincut, residual, _ = edmonds_karp(n, graph, s, t)

    if mincut > limit:
        print("NO")
        return

    # find reachable from source in residual graph
    visited = [False] * n
    q = deque([s])
    visited[s] = True
    while q:
        u = q.popleft()
        for v in range(n):
            if not visited[v] and residual[u][v] > 0:
                visited[v] = True
                q.append(v)

    cut_edges = []
    for u, v, w in edges:
        if visited[u] != visited[v]:
            cut_edges.append((u + 1, v + 1))

    print("YES")
    print(len(cut_edges))
    for u, v in cut_edges:
        print(u, v)

if __name__ == "__main__":
    main()
```
