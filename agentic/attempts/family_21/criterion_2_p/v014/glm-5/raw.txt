Let me write a standard fast Dinic:

```python
    INF = float('inf')
    
    def dinic(s, t):
        flow = 0
        while True:
            # BFS to build levels
            level = [-1] * (n+1)
            level[s] = 0
            q = deque([s])
            while q:
                u = q.popleft()
                for eid in graph[u]:
                    if cap[eid] > 0 and level[to[eid]] == -1:
                        level[to[eid]] = level[u] + 1
                        q.append(to[eid])
            if level[t] == -1:
                break
            # DFS
            it = [0] * (n+1)  # iterator index into graph[u]
            def dfs(u, f):
                if u == t:
                    return f
                while it[u] < len(graph[u]):
                    eid = graph[u][it[u]]
                    v = to[eid]
                    if cap[eid] > 0 and level[v] == level[u] + 1:
                        d = dfs(v, min(f, cap[eid]))
                        if d > 0:
                            cap[eid] -= d
                            cap[eid ^ 1] += d
                            return d
                    it[u] += 1
                return 0
            while True:
                f = dfs(s, INF)
                if f == 0:
                    break
                flow += f
        return flow
```
