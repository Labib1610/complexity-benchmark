def find_min_cut(n, m, s, t, edges):
    # Build adjacency list with edge indices
    adj = defaultdict(list)

        adj[v].append((u, w, idx))

                    flow[v] = min(flow[u], w)

                    queue.append(v)
        return 0, parent, parent_edge
