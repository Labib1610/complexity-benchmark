```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    s = int(input_data[2])
    t = int(input_data[3])
    B = int(input_data[4])
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    idx = 5
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        w = int(input_data[idx+2])
        idx += 3
        adj[u].append((v, i))
        adj[v].append((u, i))
        edges.append((u, v, w))
        
    # Binary search for the minimum capacity C
    low = 0
    high = max(w for _, _, w in edges) if edges else 0
    min_C = -1
    
    while low <= high:
        mid = (low + high) // 2
        q = deque([s])
        visited = [False] * (n + 1)
        visited[s] = True
        found = False
        while q:
            u = q.popleft()
            if u == t:
                found = True
                break
            for v, edge_idx in adj[u]:
                if not visited[v] and edges[edge_idx][2] <= mid:
                    visited[v] = True
                    q.append(v)
        if found:
            min_C = mid
            high = mid - 1
        else:
            low = mid + 1
            
    if min_C == -1:
        print("NO")
        return
        
    # BFS to find the shortest path using edges with capacity <= min_C
    q = deque([s])
    visited = [False] * (n + 1)
    visited[s] = True
    parent = [-1] * (n + 1)
    parent_edge = [-1] * (n + 1)
    
    while q:
        u = q.popleft()
        if u == t:
            break
        for v, edge_idx in adj[u]:
            if not visited[v] and edges[edge_idx][2] <= min_C:
                visited[v] = True
                parent[v] = u
                parent_edge[v] = edge_idx
                q.append(v)
                
    # Reconstruct the path
    path_edges = []
    curr = t
    while curr != s:
        edge_idx = parent_edge[curr]
        path_edges.append(edge_idx)
        curr = parent[curr]
        
    total_cost = sum(edges[idx][2] for idx in path_edges)
    
    if total_cost > B:
        print("NO")
        return
        
    print("YES")
    print(len(path_edges))
    for idx in path_edges:
        u, v, _ = edges[idx]
        print(u, v)

if __name__ == '__main__':
    solve()
```
