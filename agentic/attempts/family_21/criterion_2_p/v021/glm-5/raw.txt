import sys
from collections import deque

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    s = int(next(it))
    t = int(next(it))
    B = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        adj[u].append((v, w, len(edges)))
        adj[v].append((u, w, len(edges)))
        edges.append((u, v, w))
    
    # BFS to find shortest path by number of edges
    parent = [None] * (n + 1)
    parent_edge = [None] * (n + 1)
    q = deque([s])
    parent[s] = s
    
    while q and parent[t] is None:
        u = q.popleft()
        for v, w, idx in adj[u]:
            if parent[v] is None:
                parent[v] = u
                parent_edge[v] = idx
                q.append(v)
    
    if parent[t] is None:
        print("YES")
        print(0)
        return
    
    # Reconstruct path
    path_edges = []
    cur = t
    while cur != s:
        idx = parent_edge[cur]
        path_edges.append(idx)
        u, v, w = edges[idx]
        cur = parent[cur]
    
    # Find minimum weight edge on path
    min_w = float('inf')
    min_idx = -1
    for idx in path_edges:
        w = edges[idx][2]
        if w < min_w:
            min_w = w
            min_idx = idx
    
    if min_w <= B:
        u, v, w = edges[min_idx]
        print("YES")
        print(1)
        print(u, v)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
