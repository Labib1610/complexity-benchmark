from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    # Read test runs
    test_runs = []
    agent_to_vehicle = {}
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        test_runs.append((x, y, z))
        agent_to_vehicle[x] = y
    
    # Group test runs by agent
    agent_runs = defaultdict(list)
    for x, y, z in test_runs:
        agent_runs[x].append((x, y, z))
    
    # Check if we have exactly k agents
    if len(agent_runs) != k:
        print("NO")
        return
    
    # For each agent, we know their vehicle, so we just need to pick a sector
    # This becomes a bipartite matching problem: match agents to sectors
    # But since each agent has a fixed vehicle, we're really matching sectors to vehicles
    
    # Actually, let's think differently:
    # Each agent must be used exactly once
    # Each vehicle must be used exactly once  
    # Each sector must be used exactly once
    # And if agent a uses vehicle v, then all deployments involving agent a must use vehicle v
    
    # Since each agent is permanently assigned to one vehicle, we can think of this as:
    # 1. Each agent-vehicle pair is fixed
    # 2. We need to assign exactly one sector to each agent (and thus to their fixed vehicle)
    # 3. All sectors must be used exactly once
    
    # So this is a matching problem: match agents to sectors, respecting their fixed vehicles
    # And ensuring each vehicle is used exactly once
    
    # Let's build a graph where:
    # - Left side: agents (which implicitly determine vehicles)
    # - Right side: sectors
    # - Edge exists if there's a test run with that agent-sector combination
    
    # But we also need to ensure each vehicle is used exactly once
    # This is a constraint on the matching
    
    # Let's approach this as a bipartite matching with additional constraints
    
    # Actually, simpler approach:
    # 1. For each agent, we know their vehicle
    # 2. For each agent, we have a list of sectors they can reach
    # 3. We need to select one sector per agent such that:
    #    - All agents are selected (automatic since we need exactly k)
    #    - All vehicles are used exactly once
    #    - All sectors are used exactly once
    
    # This is equivalent to finding a perfect matching in a tripartite graph
    # But we can model it as a bipartite matching:
    # Match agents to sectors, with the constraint that the vehicles are also matched
    
    # Even simpler:
    # Since each agent is fixed to one vehicle, and we need each vehicle exactly once,
    # we're essentially looking for a bijection between agents and vehicles that respects
    # the fixed assignment, AND a bijection between agents and sectors from available test runs
    
    # Since the assignment is fixed, the agent-vehicle bijection is predetermined
    # We just need to check if there exists a matching of agents to sectors
    
    # Build available sectors for each agent
    available_sectors = defaultdict(set)
    for x, y, z in test_runs:
        available_sectors[x].add(z)
    
    # Check if all agents have at least one sector
    for agent in range(1, k+1):
        if agent not in available_sectors:
            print("NO")
            return
    
    # Now we need to find a matching from agents to sectors
    # This is a standard bipartite matching problem
    
    # Build bipartite graph: agents on left, sectors on right
    # Edge (agent, sector) exists if sector is in available_sectors[agent]
    
    # Use a simple augmenting path algorithm for bipartite matching
    def find_matching():
        # agent -> sector matching
        match_agent = [-1] * (k + 1)  # match_agent[agent] = sector
        match_sector = [-1] * (k + 1)  # match_sector[sector] = agent
        
        def dfs(agent, visited):
            for sector in available_sectors[agent]:
                if visited[sector]:
                    continue
                visited[sector] = True
                
                # If sector is unmatched or we can find augmenting path
                if match_sector[sector] == -1 or dfs(match_sector[sector], visited):
                    match_agent[agent] = sector
                    match_sector[sector] = agent
                    return True
            return False
        
        matched = 0
        for agent in range(1, k+1):
            visited = [False] * (k + 1)
            if dfs(agent, visited):
                matched += 1
        
        if matched != k:
            return None
        
        return match_agent
    
    match_result = find_matching()
    if match_result is None:
        print("NO")
        return
    
    # Now we have a matching of agents to sectors
    # We also know the vehicle for each agent
    print("YES")
    for agent in range(1, k+1):
        sector = match_result[agent]
        vehicle = agent_to_vehicle[agent]
        print(agent, vehicle, sector)

solve()
