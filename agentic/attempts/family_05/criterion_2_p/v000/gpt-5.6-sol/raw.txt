```python
import sys
from collections import deque

def main():
    read = sys.stdin.buffer.readline
    k, t = map(int, read().split())

    adj = [[] for _ in range(k)]
    machine = [0] * k
    pair_x = [-1] * k
    pair_z = [-1] * k
    matched = 0

    for _ in range(t):
        x, y, z = map(int, read().split())
        x -= 1
        z -= 1
        machine[x] = y
        adj[x].append(z)

        if pair_x[x] == -1 and pair_z[z] == -1:
            pair_x[x] = z
            pair_z[z] = x
            matched += 1

    sys.setrecursionlimit(5000)
    INF = k + 1

    while matched < k:
        dist = [INF] * k
        queue = deque()

        for x in range(k):
            if pair_x[x] == -1:
                dist[x] = 0
                queue.append(x)

        shortest = INF

        while queue:
            x = queue.popleft()
            if dist[x] >= shortest:
                continue

            next_dist = dist[x] + 1
            for z in adj[x]:
                other_x = pair_z[z]
                if other_x == -1:
                    shortest = next_dist
                elif dist[other_x] == INF:
                    dist[other_x] = next_dist
                    queue.append(other_x)

        if shortest == INF:
            break

        edge_pos = [0] * k

        def dfs(x):
            edges = adj[x]
            while edge_pos[x] < len(edges):
                z = edges[edge_pos[x]]
                edge_pos[x] += 1
                other_x = pair_z[z]

                if other_x == -1:
                    if dist[x] + 1 != shortest:
                        continue
                elif dist[other_x] != dist[x] + 1 or not dfs(other_x):
                    continue

                pair_x[x] = z
                pair_z[z] = x
                return True

            dist[x] = INF
            return False

        augmented = 0
        for x in range(k):
            if pair_x[x] == -1 and dfs(x):
                augmented += 1

        if augmented == 0:
            break
        matched += augmented

    if matched != k:
        sys.stdout.write("NO\n")
        return

    output = ["YES"]
    output.extend(
        f"{x + 1} {machine[x]} {pair_x[x] + 1}"
        for x in range(k)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
