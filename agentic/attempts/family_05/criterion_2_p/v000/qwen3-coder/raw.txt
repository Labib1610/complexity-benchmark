from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
        
    # Since each specialist is qualified on exactly one machine and all machines are distinct,
    # we can think of this as a bipartite matching problem between specialists and time slots.
    # For each specialist x, we know their machine y, and we have a list of allowed time slots z.
    
    # Create adjacency list for the bipartite graph: specialist -> list of possible time slots
    adj = defaultdict(list)  # specialist -> [time_slots]
    spec_machine = {}  # specialist -> machine
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        adj[x].append(z)
        spec_machine[x] = y
    
    # Check if we have exactly k specialists
    if len(adj) < k:
        print("NO")
        return
    
    # Now we need to find a perfect matching in bipartite graph
    # Left side: specialists (1 to k)
    # Right side: time slots (1 to k)
    # Edges: (x, z) if it's allowed
    
    # Use Hopcroft-Karp algorithm for maximum bipartite matching
    # But for simplicity, let's use a simpler approach since k <= 1200
    
    # match[z] = x means time slot z is matched to specialist x
    match = [-1] * (k + 1)  # 1-indexed
    matched_spec = [-1] * (k + 1)  # matched_spec[x] = z means specialist x is matched to time slot z
    
    def dfs(x, visited):
        for z in adj[x]:
            if visited[z]:
                continue
            visited[z] = True
            
            # If z is unmatched or we can find augmenting path
            if match[z] == -1 or dfs(match[z], visited):
                match[z] = x
                matched_spec[x] = z
                return True
        return False
    
    matching = 0
    for x in range(1, k + 1):
        if x not in adj:
            print("NO")
            return
        visited = [False] * (k + 1)
        if dfs(x, visited):
            matching += 1
    
    if matching != k:
        print("NO")
        return
    
    print("YES")
    for x in range(1, k + 1):
        z = matched_spec[x]
        y = spec_machine[x]
        print(x, y, z)

solve()
