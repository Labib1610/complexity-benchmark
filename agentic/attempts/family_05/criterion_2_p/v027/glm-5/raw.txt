import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    # protein -> receptor (1-based)
    p_to_r = {}
    # receptor -> protein (1-based)
    r_to_p = {}
    
    # For each protein, store list of catalysts (1-based)
    p_to_catalysts = [[] for _ in range(k + 1)]
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        p_to_catalysts[x].append(z)
        if x not in p_to_r:
            p_to_r[x] = y
            r_to_p[y] = x
    
    # Check that every protein 1..k appears at least once
    for x in range(1, k + 1):
        if x not in p_to_r:
            print("NO")
            return
    
    # Build bipartite graph: left = proteins (1..k), right = catalysts (1..k)
    # We need a perfect matching.
    # Since k <= 1200, Hopcroft-Karp is efficient enough.
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        adj[x] = p_to_catalysts[x]
    
    INF = float('inf')
    pair_u = [0] * (k + 1)  # pair_u[u] = matched catalyst (0 if none)
    pair_v = [0] * (k + 1)  # pair_v[v] = matched protein (0 if none)
    dist = [0] * (k + 1)
    
    from collections import deque
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching != k:
        print("NO")
        return
    
    print("YES")
    for u in range(1, k + 1):
        v = pair_u[u]
        r = p_to_r[u]
        print(f"{u} {r} {v}")

solve()
