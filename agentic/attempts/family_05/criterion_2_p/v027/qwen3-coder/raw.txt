from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    # Read complexes
    complexes = []
    protein_to_receptor = {}
    receptor_to_protein = {}
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        complexes.append((x, y, z))
        
        # Verify the constraint: each protein chain only interacts with one receptor site
        if x in protein_to_receptor:
            if protein_to_receptor[x] != y:
                # This shouldn't happen according to problem constraints
                print("NO")
                return
        else:
            protein_to_receptor[x] = y
            
        if y in receptor_to_protein:
            if receptor_to_protein[y] != x:
                # This shouldn't happen according to problem constraints
                print("NO")
                return
        else:
            receptor_to_protein[y] = x
    
    # Build bipartite graph: proteins <-> catalysts
    # An edge exists if there's a complex with that protein and catalyst
    protein_catalyst_edges = defaultdict(set)
    catalyst_protein_edges = defaultdict(set)
    
    for x, y, z in complexes:
        protein_catalyst_edges[x].add(z)
        catalyst_protein_edges[z].add(x)
    
    # Try to find a perfect matching in this bipartite graph
    # Using a simplified Hopcroft-Karp implementation or justHungarian algorithm concept
    
    # For each protein, we need to assign a unique catalyst
    # This is a maximum bipartite matching problem
    
    def dfs(u, match_v, match_u, visited):
        for v in protein_catalyst_edges[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            if match_v[v] == -1 or dfs(match_v[v], match_v, match_u, visited):
                match_v[v] = u
                match_u[u] = v
                return True
        return False
    
    # Initialize matching arrays
    match_u = [-1] * (k + 1)  # match_u[protein] = catalyst
    match_v = [-1] * (k + 1)  # match_v[catalyst] = protein
    
    matching = 0
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(u, match_v, match_u, visited):
            matching += 1
    
    if matching < k:
        print("NO")
        return
    
    # Reconstruct the solution
    print("YES")
    result = []
    for protein in range(1, k + 1):
        catalyst = match_u[protein]
        receptor = protein_to_receptor[protein]
        result.append(f"{protein} {receptor} {catalyst}")
    
    print("\n".join(result))

solve()
