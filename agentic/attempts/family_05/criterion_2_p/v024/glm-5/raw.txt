```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    
    route_to_vehicle = {}
    route_to_drivers = {}
    
    idx = 2
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        route_to_vehicle[x] = y
        route_to_drivers.setdefault(x, []).append(z)
    
    for x in range(1, k+1):
        if x not in route_to_vehicle:
            print("NO")
            return
    
    vehicles = [0] * (k + 1)
    for x in range(1, k+1):
        y = route_to_vehicle[x]
        if vehicles[y] != 0:
            print("NO")
            return
        vehicles[y] = x
    
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k+1):
        for z in route_to_drivers[x]:
            adj[x].append(z)
    
    match_to_route = [0] * (k + 1)
    
    def try_kuhn(v, visited):
        for to in adj[v]:
            if not visited[to]:
                visited[to] = True
                if match_to_route[to] == 0 or try_kuhn(match_to_route[to], visited):
                    match_to_route[to] = v
                    return True
        return False
    
    for v in range(1, k+1):
        visited = [False] * (k + 1)
        if not try_kuhn(v, visited):
            print("NO")
            return
    
    print("YES")
    out = []
    for z in range(1, k+1):
        x = match_to_route[z]
        y = route_to_vehicle[x]
        out.append(f"{x} {y} {z}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
```
