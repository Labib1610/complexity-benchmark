We can model this as a bipartite matching problem.  
Each route is already paired with a fixed vehicle, so we only need to assign drivers to routes such that each driver is used exactly once.

We can solve it using maximum bipartite matching between routes and drivers.  
If the matching size equals `k`, we output the schedule.

Here’s the Python solution:

```python
import sys
from collections import deque

input = sys.stdin.readline

k, t = map(int, input().split())

# route -> vehicle mapping
route_vehicle = [0] * (k + 1)
# adjacency list: route -> list of drivers
adj = [[] for _ in range(k + 1)]

for _ in range(t):
    x, y, z = map(int, input().split())
    route_vehicle[x] = y
    adj[x].append(z)

# Hopcroft-Karp for maximum bipartite matching
match_route = [-1] * (k + 1)  # driver -> route
match_driver = [-1] * (k + 1) # route -> driver
dist = [0] * (k + 1)

def bfs():
    q = deque()
    for r in range(1, k + 1):
        if match_driver[r] == -1:
            dist[r] = 0
            q.append(r)
        else:
            dist[r] = float('inf')
    found = False
    while q:
        r = q.popleft()
        for d in adj[r]:
            r2 = match_route[d]
            if r2 != -1 and dist[r2] == float('inf'):
                dist[r2] = dist[r] + 1
                q.append(r2)
            elif r2 == -1:
                found = True
    return found

def dfs(r):
    for d in adj[r]:
        r2 = match_route[d]
        if r2 == -1 or (dist[r2] == dist[r] + 1 and dfs(r2)):
            match_route[d] = r
            match_driver[r] = d
            return True
    dist[r] = float('inf')
    return False

matching = 0
while bfs():
    for r in range(1, k + 1):
        if match_driver[r] == -1:
            if dfs(r):
                matching += 1

if matching == k:
    print("YES")
    for r in range(1, k + 1):
        print(r, route_vehicle[r], match_driver[r])
else:
    print("NO")
```
