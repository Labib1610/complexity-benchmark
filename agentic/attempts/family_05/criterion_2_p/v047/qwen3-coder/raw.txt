from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    if k == 0:
        print("YES")
        return
    
    assignments = []
    pilot_assignments = defaultdict(list)
    phase_assignments = defaultdict(list)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        assignments.append((x, y, z))
        pilot_assignments[x].append((x, y, z))
        phase_assignments[z].append((x, y, z))
    
    # Create a bipartite graph matching problem
    # Left side: pilots, Right side: phases
    # We need to match each pilot to exactly one phase such that:
    # 1. Each pilot is used exactly once
    # 2. Each phase is used exactly once
    # 3. The assignment is valid (pilot x can operate mech y in phase z)
    # 4. Sum of pilot IDs = k(k+1)/2 (automatically satisfied if we use each pilot once)
    # 5. Sum of phase IDs = k(k+1)/2 (automatically satisfied if we use each phase once)
    
    # Build adjacency list for valid pilot-phase pairs
    graph = defaultdict(list)
    for x, y, z in assignments:
        graph[x].append(z)
    
    # Try to find a perfect matching using Hungarian algorithm approach
    # But since this is a simple case, we can use a greedy approach with backtracking
    
    # For small k, we can try all permutations, but for k up to 1200, we need something better
    # This is a maximum bipartite matching problem
    
    match_pilot = [-1] * (k + 1)  # match_pilot[pilot] = phase
    match_phase = [-1] * (k + 1)  # match_phase[phase] = pilot
    
    def dfs(pilot, visited):
        for phase in graph[pilot]:
            if visited[phase]:
                continue
            visited[phase] = True
            
            # If phase is unmatched or we can find augmenting path
            if match_phase[phase] == -1 or dfs(match_phase[phase], visited):
                match_pilot[pilot] = phase
                match_phase[phase] = pilot
                return True
        return False
    
    # Try to match all pilots
    matched = 0
    for pilot in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(pilot, visited):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    # Verify that we can find the actual assignments
    # We need to make sure that for each pilot-phase pair, there exists an assignment
    result = []
    mech_of_pilot = {}
    
    # First, determine mech for each pilot from input
    for x, y, z in assignments:
        mech_of_pilot[x] = y
    
    # Check if the matching is valid
    valid = True
    for pilot in range(1, k + 1):
        phase = match_pilot[pilot]
        if phase == -1:
            valid = False
            break
        # Check if this pilot-phase combination exists in assignments
        found = False
        for x, y, z in pilot_assignments[pilot]:
            if z == phase:
                result.append((x, y, z))
                found = True
                break
        if not found:
            valid = False
            break
    
    if not valid:
        print("NO")
        return
    
    print("YES")
    for x, y, z in result:
        print(x, y, z)

solve()
