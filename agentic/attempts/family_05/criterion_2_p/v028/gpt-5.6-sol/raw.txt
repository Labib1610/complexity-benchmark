import sys
from array import array
from collections import deque

def main():
    stream = sys.stdin.buffer
    first = stream.readline().split()
    if not first:
        return

    k, t = map(int, first)

    if t < k:
        sys.stdout.write("NO\n")
        return

    adj = [array('H') for _ in range(k + 1)]
    sensor = array('H', [0]) * (k + 1)

    def consume(block):
        parts = block.split()
        for i in range(0, len(parts), 3):
            x = int(parts[i])
            y = int(parts[i + 1])
            z = int(parts[i + 2])
            if sensor[x] == 0:
                sensor[x] = y
            adj[x].append(z)

    carry = b''
    chunk_size = 1 << 22

    while True:
        chunk = stream.read(chunk_size)
        if not chunk:
            if carry:
                consume(carry)
            break

        data = carry + chunk
        cut = data.rfind(b'\n')
        if cut == -1:
            carry = data
        else:
            consume(data[:cut + 1])
            carry = data[cut + 1:]

    for x in range(1, k + 1):
        if not adj[x]:
            sys.stdout.write("NO\n")
            return

    pair_left = [0] * (k + 1)
    pair_right = [0] * (k + 1)
    dist = [0] * (k + 1)
    matched = 0
    inf = k + 1

    sys.setrecursionlimit(k * 3 + 100)

    while True:
        queue = deque()
        for u in range(1, k + 1):
            if pair_left[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = inf

        shortest = inf

        while queue:
            u = queue.popleft()
            next_dist = dist[u] + 1
            if next_dist > shortest:
                continue

            for v in adj[u]:
                other = pair_right[v]
                if other == 0:
                    shortest = next_dist
                elif dist[other] == inf and next_dist < shortest:
                    dist[other] = next_dist
                    queue.append(other)

        if shortest == inf:
            break

        ptr = [0] * (k + 1)

        def dfs(u):
            edges = adj[u]
            while ptr[u] < len(edges):
                v = edges[ptr[u]]
                ptr[u] += 1
                other = pair_right[v]

                if other == 0:
                    if dist[u] + 1 != shortest:
                        continue
                elif dist[other] != dist[u] + 1 or not dfs(other):
                    continue

                pair_left[u] = v
                pair_right[v] = u
                return True

            dist[u] = inf
            return False

        augmented = 0
        for u in range(1, k + 1):
            if pair_left[u] == 0 and dfs(u):
                augmented += 1

        if augmented == 0:
            break
        matched += augmented

    if matched != k:
        sys.stdout.write("NO\n")
        return

    output = ["YES"]
    output.extend(
        f"{x} {sensor[x]} {pair_left[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
