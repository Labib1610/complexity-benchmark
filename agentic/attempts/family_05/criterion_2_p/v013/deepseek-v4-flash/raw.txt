We can model this as a matching problem.  
Since each specialist is tied to exactly one machine, the problem reduces to assigning time slots to specialists such that each specialist gets a distinct time slot from their allowed set.

We can solve it using a maximum bipartite matching between specialists and time slots, where an edge exists if the configuration `(x, y, z)` is present.

Here’s the implementation:

```python
import sys
from collections import defaultdict, deque

def main():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # For each specialist, store allowed time slots
    allowed = [[] for _ in range(k + 1)]
    machine_for_specialist = [0] * (k + 1)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        allowed[x].append(z)
        machine_for_specialist[x] = y
    
    # Hopcroft-Karp for maximum bipartite matching
    # Left: specialists (1..k), Right: time slots (1..k)
    match_slot = [0] * (k + 1)  # slot -> specialist
    match_spec = [0] * (k + 1)  # specialist -> slot
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if match_spec[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in allowed[u]:
                    if dist[match_slot[v]] == float('inf'):
                        dist[match_slot[v]] = dist[u] + 1
                        queue.append(match_slot[v])
        return dist[0] != float('inf')
    
    def dfs(u):
        if u != 0:
            for v in allowed[u]:
                if dist[match_slot[v]] == dist[u] + 1 and dfs(match_slot[v]):
                    match_slot[v] = u
                    match_spec[u] = v
                    return True
            dist[u] = float('inf')
            return False
        return True
    
    dist = [0] * (k + 1)
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if match_spec[u] == 0 and dfs(u):
                matching += 1
    
    if matching != k:
        print("NO")
    else:
        print("YES")
        for spec in range(1, k + 1):
            slot = match_spec[spec]
            machine = machine_for_specialist[spec]
            print(spec, machine, slot)

if __name__ == "__main__":
    main()
```
