from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    if t < k:
        print("NO")
        return
    
    configs = []
    spec_to_configs = defaultdict(list)
    mach_to_configs = defaultdict(list)
    
    idx = 2
    for i in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        idx += 3
        config = (x, y, z)
        configs.append(config)
        spec_to_configs[x].append(i)
        mach_to_configs[y].append(i)
    
    # Check if every specialist has at least one configuration
    if len(spec_to_configs) < k:
        print("NO")
        return
        
    # Check if every machine has at least one configuration
    if len(mach_to_configs) < k:
        print("NO")
        return
    
    # Build bipartite graph: specialists on one side, time slots on the other
    # Edge exists if there's a config (spec, mach, time) for some machine
    
    # Actually, since each specialist is tied to exactly one machine,
    # and no two specialists use the same machine,
    # we can think of this as matching specialists to time slots.
    
    # For each specialist, we have a list of possible time slots
    # via the configurations they appear in.
    
    # This becomes a matching problem: match each specialist to a distinct time slot.
    
    # Let's build a graph where nodes are specialists and time slots
    # and edges represent possible assignments
    
    from collections import deque
    
    # Map each specialist to its machine
    spec_machine = {}
    for i in range(t):
        x, y, _ = configs[i]
        spec_machine[x] = y
    
    # For each specialist, collect available time slots
    spec_times = defaultdict(list)
    for i in range(t):
        x, _, z = configs[i]
        spec_times[x].append((z, i))  # time slot, config index
    
    # Try to find a maximum matching using Ford-Fulkerson or similar
    # But since k<=1200, let's use a simpler approach with DFS
    
    # Hungarian algorithm or max flow would work, but let's try greedy + backtracking
    
    # Actually, this is a bipartite matching:
    # Left side: specialists 1..k
    # Right side: time slots 1..k  
    # Edge (x,z) exists if there's a config (x,*,z)
    
    # Build adjacency for graph
    adj = defaultdict(list)
    for i in range(t):
        x, _, z = configs[i]
        adj[x].append((z, i))
        
    # Try to match each specialist to a unique time slot
    match_spec_to_time = {}      # spec -> time
    match_time_to_spec = {}      # time -> spec
    selected_configs = {}         # spec -> config index
    
    def dfs(x, visited):
        for z, i in adj[x]:
            if z in visited:
                continue
            visited.add(z)
            if z not in match_time_to_spec or dfs(match_time_to_spec[z], visited):
                match_spec_to_time[x] = z
                match_time_to_spec[z] = x
                selected_configs[x] = i
                return True
        return False
                
    matched = 0
    for x in range(1, k+1):
        if x not in adj:
            print("NO")
            return
        visited = set()
        if dfs(x, visited):
            matched += 1
            
    if matched == k:
        print("YES")
        for x in range(1, k+1):
            i = selected_configs[x]
            print(configs[i][0], configs[i][1], configs[i][2])
    else:
        print("NO")

solve()
