```python
import sys
from collections import deque

def main():
    input_buffer = sys.stdin.buffer
    first = input_buffer.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k + 1)]
    machine = [0] * (k + 1)

    for _ in range(t):
        x, y, z = map(int, input_buffer.readline().split())
        adjacency[x].append(z)
        if machine[x] == 0:
            machine[x] = y

    if any(not adjacency[x] for x in range(1, k + 1)):
        sys.stdout.write("NO\n")
        return

    pair_x = [0] * (k + 1)
    pair_z = [0] * (k + 1)
    distance = [0] * (k + 1)
    inf = k + 1
    matched = 0

    sys.setrecursionlimit(10000)

    while matched < k:
        queue = deque()

        for x in range(1, k + 1):
            if pair_x[x] == 0:
                distance[x] = 0
                queue.append(x)
            else:
                distance[x] = inf

        shortest = inf

        while queue:
            x = queue.popleft()
            next_distance = distance[x] + 1

            if next_distance > shortest:
                continue

            for z in adjacency[x]:
                other_x = pair_z[z]
                if other_x == 0:
                    shortest = next_distance
                elif distance[other_x] == inf:
                    distance[other_x] = next_distance
                    queue.append(other_x)

        if shortest == inf:
            break

        def augment(x):
            next_distance = distance[x] + 1

            for z in adjacency[x]:
                other_x = pair_z[z]

                if other_x == 0:
                    if next_distance == shortest:
                        pair_x[x] = z
                        pair_z[z] = x
                        return True
                elif distance[other_x] == next_distance and augment(other_x):
                    pair_x[x] = z
                    pair_z[z] = x
                    return True

            distance[x] = inf
            return False

        for x in range(1, k + 1):
            if pair_x[x] == 0 and augment(x):
                matched += 1

    if matched != k:
        sys.stdout.write("NO\n")
        return

    output = ["YES"]
    output.extend(
        f"{x} {machine[x]} {pair_x[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
