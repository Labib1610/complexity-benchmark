```python
import sys
from collections import deque

def main():
    input_buffer = sys.stdin.buffer
    first = input_buffer.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k + 1)]
    key_for_station = [0] * (k + 1)
    station_for_key = [0] * (k + 1)
    valid = True

    for _ in range(t):
        x, y, z = map(int, input_buffer.readline().split())

        if key_for_station[x] == 0:
            key_for_station[x] = y
        elif key_for_station[x] != y:
            valid = False

        if station_for_key[y] == 0:
            station_for_key[y] = x
        elif station_for_key[y] != x:
            valid = False

        adjacency[x].append(z)

    if not valid:
        print("NO")
        return

    for x in range(1, k + 1):
        if key_for_station[x] == 0 or not adjacency[x]:
            print("NO")
            return

    pair_left = [0] * (k + 1)
    pair_right = [0] * (k + 1)
    distance = [0] * (k + 1)
    infinity = k + 1
    matching_size = 0

    sys.setrecursionlimit(2 * k + 100)

    while True:
        queue = deque()
        for x in range(1, k + 1):
            if pair_left[x] == 0:
                distance[x] = 0
                queue.append(x)
            else:
                distance[x] = infinity

        shortest = infinity

        while queue:
            x = queue.popleft()
            if distance[x] >= shortest:
                continue

            next_distance = distance[x] + 1
            for z in adjacency[x]:
                matched_x = pair_right[z]
                if matched_x == 0:
                    shortest = next_distance
                elif distance[matched_x] == infinity:
                    distance[matched_x] = next_distance
                    queue.append(matched_x)

        if shortest == infinity:
            break

        edge_index = [0] * (k + 1)

        def augment(x):
            edges = adjacency[x]
            while edge_index[x] < len(edges):
                z = edges[edge_index[x]]
                edge_index[x] += 1
                matched_x = pair_right[z]

                if matched_x == 0:
                    if distance[x] + 1 != shortest:
                        continue
                elif distance[matched_x] != distance[x] + 1 or not augment(matched_x):
                    continue

                pair_left[x] = z
                pair_right[z] = x
                return True

            distance[x] = infinity
            return False

        added = 0
        for x in range(1, k + 1):
            if pair_left[x] == 0 and augment(x):
                added += 1

        if added == 0:
            break
        matching_size += added

    if matching_size != k:
        print("NO")
        return

    output = ["YES"]
    output.extend(
        f"{x} {key_for_station[x]} {pair_left[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
