from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    index = 2
    # Build a mapping from emitter to its dedicated filter
    emitter_to_filter = {}
    # Store all triplets
    triplets = []
    # For each filter, store which detectors it can connect to with which emitters
    filter_to_detectors = defaultdict(list)
    
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        index += 3
        triplets.append((x, y, z))
        # Since each emitter is only ever listed with its dedicated filter
        if x not in emitter_to_filter:
            emitter_to_filter[x] = y
        # Verify consistency 
        elif emitter_to_filter[x] != y:
            # This shouldn't happen according to problem statement
            pass
            
        filter_to_detectors[y].append(z)
        
    # Build a bipartite graph: filters on one side, detectors on the other
    # We need to match each of the k filters to a unique detector
    # But we also need to ensure that the emitters connected to these filters are all distinct (which they are by constraint)
    
    # Actually, let's think differently:
    # Each emitter has exactly one filter. So we have a mapping from emitter -> filter
    # We want to choose k triplets such that:
    # - All emitters 1..k are used exactly once => all emitters must be present
    # - All filters 1..k are used exactly once => all filters must be present and matched to emitters
    # - All detectors 1..k are used exactly once
    
    # First check: do we have all emitters 1..k?
    if len(emitter_to_filter) != k:
        print("NO")
        return
        
    # Check that all filters are distinct and cover 1..k
    filters_used = set(emitter_to_filter.values())
    if len(filters_used) != k or filters_used != set(range(1, k+1)):
        print("NO")
        return
        
    # Now we have exactly k emitters, each mapped to a unique filter, covering all filters 1..k
    # We need to match each emitter-filter pair to a unique detector
    
    # Create a mapping from (emitter, filter) to list of possible detectors
    valid_triplets = {}
    for x, y, z in triplets:
        if (x, y) not in valid_triplets:
            valid_triplets[(x, y)] = []
        valid_triplets[(x, y)].append(z)
        
    # For each emitter (which determines the filter), we need to pick one detector
    # This becomes a matching problem: assign to each emitter a detector, such that all detectors 1..k are used once
    
    # Build a graph for maximum bipartite matching: emitters on left, detectors on right
    # Edge exists if there's a triplet (emitter, filter, detector)
    
    # But we already know the filter for each emitter
    # So for emitter i, its filter is emitter_to_filter[i]
    # We just need to find a detector for it
    
    # Let's build adjacency for emitters to detectors
    emitter_det_adj = defaultdict(list)
    for x, y, z in triplets:
        emitter_det_adj[x].append(z)
        
    # Now we need to match emitters 1..k to detectors 1..k such that:
    # - each emitter maps to one of its adjacent detectors
    # - each detector is used exactly once
    
    # This is a standard maximum bipartite matching problem
    
    def dfs(u, match_right, match_left, visited):
        for v in emitter_det_adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match_right[v] == -1 or dfs(match_right[v], match_right, match_left, visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    # match_right[v] = u means detector v is matched to emitter u
    # match_left[u] = v means emitter u is matched to detector v
    match_right = [-1] * (k + 1)
    match_left = [-1] * (k + 1)
    
    result = 0
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(u, match_right, match_left, visited):
            result += 1
            
    if result != k:
        print("NO")
        return
        
    print("YES")
    for u in range(1, k + 1):
        v = match_left[u]
        # u is emitter, emitter_to_filter[u] is filter, v is detector
        print(u, emitter_to_filter[u], v)

solve()
