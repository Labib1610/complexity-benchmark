def solve():
    k, t = map(int, input().split())
    
    # Since each agent has exactly one alias and all aliases are distinct,
    # we can think of this as a bipartite matching problem between agents and frequencies.
    # But we need to be more precise: for each agent x, we have a fixed alias y,
    # and a list of allowed frequencies z.
    
    # Let's reframe: we have k agents, each can use certain frequencies.
    # We want to assign each agent a distinct frequency.
    
    # Build a graph where agents are on left, frequencies on right
    # Edge exists if agent can use that frequency
    graph = defaultdict(list)  # agent -> list of frequencies
    agents = set()
    frequencies = set()
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        graph[x].append(z)
        agents.add(x)
        frequencies.add(z)
    
    # Check if we have exactly k agents and they're numbered 1..k
    if len(agents) != k or agents != set(range(1, k+1)):
        print("NO")
        return
    
    # Now we need to find a perfect matching in bipartite graph
    # Left side: agents 1..k
    # Right side: frequencies 1..k
    # Edges: (agent, freq) if agent can use freq
    
    # Build adjacency list for the bipartite graph
    adj = defaultdict(list)
    for agent in range(1, k+1):
        for freq in graph[agent]:
            adj[agent].append(freq)
    
    # Use Hopcroft-Karp or simpler Ford-Fulkerson for max bipartite matching
    # match[freq] = agent that is matched to freq, or -1 if unmatched
    match = [-1] * (k + 1)  # 1-indexed
    
    def dfs(u, visited):
        for v in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            # If freq v is unmatched or we can find augmenting path
            if match[v] == -1 or dfs(match[v], visited):
                match[v] = u
                return True
        return False
    
    # Try to match each agent
    matching = 0
    for agent in range(1, k+1):
        visited = [False] * (k + 1)
        if dfs(agent, visited):
            matching += 1
    
    if matching != k:
        print("NO")
        return

    # Let's re-read the input to store alias info
    input_lines = []
    for _ in range(t):
        input_lines.append(list(map(int, input().split())))

        freq_options[x].append((y, z))
