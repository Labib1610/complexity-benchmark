from collections import defaultdict

def find_matching(graph, k):
    match_pilot = [-1] * (k + 1)
    match_aircraft = [-1] * (k + 1)
    match_window = [-1] * (k + 1)
    
    def dfs(pilot, visited):
        for aircraft, window in graph[pilot]:
            if window in visited:
                continue
            visited.add(window)
            
            if match_window[window] == -1 or dfs(match_window[window], visited):
                match_pilot[pilot] = (aircraft, window)
                match_aircraft[aircraft] = (pilot, window)
                match_window[window] = pilot
                return True
        return False
    
    matched = 0
    for pilot in range(1, k + 1):
        if dfs(pilot, set()):
            matched += 1
    
    if matched == k:
        result = []
        for pilot in range(1, k + 1):
            aircraft, window = match_pilot[pilot]
            result.append((pilot, aircraft, window))
        return result
    return None

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    graph = defaultdict(list)
    
    idx = 2
    for _ in range(t):
        p = int(data[idx])
        a = int(data[idx + 1])
        w = int(data[idx + 2])
        idx += 3
        graph[p].append((a, w))
    
    result = find_matching(graph, k)
    
    if result is None:
        print("NO")
    else:
        print("YES")
        for p, a, w in result:
            print(p, a, w)

if __name__ == "__main__":
    main()
