```python
import sys
from collections import deque

def solve():
    input_buffer = sys.stdin.buffer
    first = input_buffer.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k)]
    aircraft = [-1] * k

    for _ in range(t):
        p, a, w = map(int, input_buffer.readline().split())
        p -= 1
        w -= 1
        adjacency[p].append(w)
        aircraft[p] = a

    if t < k or any(not edges for edges in adjacency):
        sys.stdout.write("NO\n")
        return

    match_pilot = [-1] * k
    match_window = [-1] * k
    matched = 0

    # Initial greedy matching.
    for pilot in range(k):
        for window in adjacency[pilot]:
            if match_window[window] == -1:
                match_pilot[pilot] = window
                match_window[window] = pilot
                matched += 1
                break

    sys.setrecursionlimit(10000)
    inf = k + 1
    distance = [inf] * k

    while matched < k:
        queue = deque()
        shortest = inf

        for pilot in range(k):
            if match_pilot[pilot] == -1:
                distance[pilot] = 0
                queue.append(pilot)
            else:
                distance[pilot] = inf

        while queue:
            pilot = queue.popleft()
            current_distance = distance[pilot]
            if current_distance >= shortest:
                continue

            for window in adjacency[pilot]:
                next_pilot = match_window[window]
                if next_pilot == -1:
                    shortest = current_distance + 1
                elif distance[next_pilot] == inf:
                    distance[next_pilot] = current_distance + 1
                    queue.append(next_pilot)

        if shortest == inf:
            break

        next_edge = [0] * k

        def augment(pilot):
            edges = adjacency[pilot]
            index = next_edge[pilot]
            pilot_distance = distance[pilot]

            while index < len(edges):
                window = edges[index]
                index += 1
                next_edge[pilot] = index
                next_pilot = match_window[window]

                if next_pilot == -1:
                    if pilot_distance + 1 != shortest:
                        continue
                elif distance[next_pilot] != pilot_distance + 1 or not augment(next_pilot):
                    continue

                match_pilot[pilot] = window
                match_window[window] = pilot
                return True

            distance[pilot] = inf
            return False

        phase_matches = 0
        for pilot in range(k):
            if match_pilot[pilot] == -1 and augment(pilot):
                phase_matches += 1

        if phase_matches == 0:
            break
        matched += phase_matches

    if matched != k:
        sys.stdout.write("NO\n")
        return

    output = ["YES"]
    for pilot in range(k):
        output.append(f"{pilot + 1} {aircraft[pilot]} {match_pilot[pilot] + 1}")
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    solve()
```
