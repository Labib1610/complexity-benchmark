    t = int(next(it))
    
    adj = [[] for _ in range(k + 1)]
    pilot_aircraft = [0] * (k + 1)
    
    for _ in range(t):
        p = int(next(it))
        a = int(next(it))
        w = int(next(it))
        adj[p].append(w)
        pilot_aircraft[p] = a
    
    pairU = [0] * (k + 1)
    pairV = [0] * (k + 1)
    dist = [0] * (k + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pairU[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    # If v is matched, explore its partner
                    if pairV[v] != 0:
                        if dist[pairV[v]] == float('inf'):
                            dist[pairV[v]] = dist[u] + 1
                            queue.append(pairV[v])
                    else:
                        # v is unmatched, so we reached the end of an augmenting path
                        # We represent this by reaching the NIL node (0)
                        if dist[0] == float('inf'):
                            dist[0] = dist[u] + 1
        return dist[0] != float('inf')
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if pairV[v] == 0:
                    # Found an augmenting path
                    pairU[u] = v
                    pairV[v] = u
                    return True
                elif dist[pairV[v]] == dist[u] + 1:
                    if dfs(pairV[v]):
                        pairU[u] = v
                        pairV[v] = u
                        return True
            dist[u] = float('inf')
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pairU[u] == 0:
                if dfs(u):
                    matching += 1
