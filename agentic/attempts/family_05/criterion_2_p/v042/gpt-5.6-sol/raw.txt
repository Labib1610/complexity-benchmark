```python
import sys
from collections import deque

def main():
    input_buffer = sys.stdin.buffer
    first = input_buffer.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k)]
    guide = [-1] * k

    match_x = [-1] * k
    match_z = [-1] * k
    matched = 0

    for _ in range(t):
        x, y, z = map(int, input_buffer.readline().split())
        x -= 1
        y -= 1
        z -= 1
        guide[x] = y
        adjacency[x].append(z)

        if match_x[x] == -1 and match_z[z] == -1:
            match_x[x] = z
            match_z[z] = x
            matched += 1

    if matched < k:
        sys.setrecursionlimit(10000)
        dist = [0] * k
        inf = k + 1

        while matched < k:
            queue = deque()
            for x in range(k):
                if match_x[x] == -1:
                    dist[x] = 0
                    queue.append(x)
                else:
                    dist[x] = inf

            shortest = inf

            while queue:
                x = queue.popleft()
                if dist[x] + 1 > shortest:
                    continue
                next_dist = dist[x] + 1
                for z in adjacency[x]:
                    other_x = match_z[z]
                    if other_x == -1:
                        shortest = next_dist
                    elif dist[other_x] == inf:
                        dist[other_x] = next_dist
                        queue.append(other_x)

            if shortest == inf:
                break

            def dfs(x):
                next_dist = dist[x] + 1
                for z in adjacency[x]:
                    other_x = match_z[z]
                    if other_x == -1:
                        if next_dist != shortest:
                            continue
                    elif dist[other_x] != next_dist or not dfs(other_x):
                        continue

                    match_x[x] = z
                    match_z[z] = x
                    return True

                dist[x] = inf
                return False

            augmented = 0
            for x in range(k):
                if match_x[x] == -1 and dfs(x):
                    augmented += 1

            if augmented == 0:
                break
            matched += augmented

    if matched != k:
        sys.stdout.write("NO\n")
        return

    output = ["YES"]
    for x in range(k):
        output.append(f"{x + 1} {guide[x] + 1} {match_x[x] + 1}")
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
