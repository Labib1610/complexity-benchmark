from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    # Since each enzyme is bonded to exactly one guide RNA, and they're all different,
    # we can think of this as a matching problem where for each enzyme x, there's exactly
    # one guide RNA y that it can use. So we can represent each viable combination
    # as (x, z) where x is the enzyme and z is the cell culture.
    
    # Build a graph where enzymes are on left, cell cultures on right
    # and edges represent viable combinations
    graph = defaultdict(list)  # enzyme -> list of cell cultures
    viable = set()
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        graph[x].append(z)
        viable.add((x, y, z))
    
    # We need to find a perfect matching in a bipartite graph
    # Left side: enzymes (1 to k)
    # Right side: cell cultures (1 to k)
    # Edges: viable combinations
    
    # Use Ford-Fulkerson with BFS (Edmonds-Karp) to find maximum matching
    def bfs_match(match_enzyme, match_cell):
        # BFS to find augmenting path
        queue = deque()
        parent = {}
        visited = set()
        
        # Start with unmatched enzymes
        for enzyme in range(1, k + 1):
            if enzyme not in match_enzyme:
                queue.append(enzyme)
                parent[enzyme] = None
                visited.add(enzyme)
        
        found_path = False
        last_enzyme = None
        last_cell = None
        
        while queue:
            node = queue.popleft()
            
            # If node is an enzyme
            if node <= k:
                enzyme = node
                # Check all adjacent cell cultures
                for cell in graph[enzyme]:
                    if cell in visited:
                        continue
                    visited.add(cell)
                    parent[cell] = enzyme
                    
                    # If cell is unmatched, we found an augmenting path
                    if cell not in match_cell:
                        found_path = True
                        last_enzyme = enzyme
                        last_cell = cell
                        break
                    else:
                        # Continue with the matched enzyme
                        next_enzyme = match_cell[cell]
                        queue.append(next_enzyme)
                        parent[next_enzyme] = cell
                        visited.add(next_enzyme)
            
            if found_path:
                break
        
        if not found_path:
            return False
        
        # Update matching along the path
        current_cell = last_cell
        current_enzyme = last_enzyme
        
        while current_enzyme is not None:
            # Add edge enzyme -> cell
            old_match = match_enzyme.get(current_enzyme)
            match_enzyme[current_enzyme] = current_cell
            match_cell[current_cell] = current_enzyme
            
            # Move to previous node in path
            current_cell = old_match
            current_enzyme = parent.get(current_enzyme)
        
        return True
    
    # Initialize matching
    match_enzyme = {}  # enzyme -> cell
    match_cell = {}    # cell -> enzyme
    
    # Keep finding augmenting paths until no more exist
    while bfs_match(match_enzyme, match_cell):
        pass
    
    # Check if we have a perfect matching
    if len(match_enzyme) != k:
        print("NO")
        return
    
    print("YES")
    
    # Reconstruct the solution
    # We need to output x y z for each trial
    # For each enzyme x, we know its guide RNA y (from the input constraint)
    # and we've matched it to a cell culture z
    
    # First, we need to know which guide RNA each enzyme is bonded to
    enzyme_to_rna = {}
    for x, y, z in viable:
        enzyme_to_rna[x] = y
    
    for enzyme in range(1, k + 1):
        cell = match_enzyme[enzyme]
        rna = enzyme_to_rna[enzyme]
        print(enzyme, rna, cell)

solve()
