We can model this as a bipartite matching problem between satellites and frequencies, since the satellite–slot pairing is fixed.  
We just need to match each satellite to a distinct frequency using only the viable setups.

```python
import sys
from collections import defaultdict, deque

def main():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # Since each satellite has a fixed slot, we can store possible frequencies per satellite
    sat_to_freqs = [[] for _ in range(k + 1)]
    slot_of_sat = [0] * (k + 1)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        sat_to_freqs[x].append(z)
        slot_of_sat[x] = y
    
    # Build bipartite graph: satellites (left) -> frequencies (right)
    # Hopcroft-Karp for maximum matching
    INF = 10**9
    pairU = [0] * (k + 1)  # frequency matched to satellite
    pairV = [0] * (k + 1)  # satellite matched to frequency
    dist = [0] * (k + 1)
    
    def bfs():
        q = deque()
        for u in range(1, k + 1):
            if pairU[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in sat_to_freqs[u]:
                    if dist[pairV[v]] == INF:
                        dist[pairV[v]] = dist[u] + 1
                        q.append(pairV[v])
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in sat_to_freqs[u]:
                if dist[pairV[v]] == dist[u] + 1:
                    if dfs(pairV[v]):
                        pairV[v] = u
                        pairU[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pairU[u] == 0 and dfs(u):
                matching += 1
    
    if matching != k:
        print("NO")
    else:
        print("YES")
        for sat in range(1, k + 1):
            freq = pairU[sat]
            print(sat, slot_of_sat[sat], freq)

if __name__ == "__main__":
    main()
```
