import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # Map satellite to its fixed orbital slot
    sat_to_slot = {}
    # For each satellite, store list of (frequency, setup_index)
    sat_freqs = [[] for _ in range(k + 1)]
    
    setups = []
    for i in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        setups.append((x, y, z))
        sat_to_slot[x] = y
        sat_freqs[x].append((z, i))
    
    # Build bipartite graph: satellites (left) to frequencies (right)
    # We need to find a perfect matching
    
    # Using Hopcroft-Karp algorithm
    # Left side: satellites 1..k
    # Right side: frequencies 1..k
    
    # Adjacency list for left nodes
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        for z, idx in sat_freqs[x]:
            adj[x].append(z)
    
    # Hopcroft-Karp
    INF = float('inf')
    pair_u = [0] * (k + 1)  # pair_u[u] = v matched with u
    pair_v = [0] * (k + 1)  # pair_v[v] = u matched with v
    dist = [0] * (k + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_v[v] == 0:
                        dist[0] = dist[u] + 1
                    elif dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if pair_v[v] == 0 or (dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v])):
                    pair_u[u] = v
                    pair_v[v] = u
                    return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching != k:
        print("NO")
        return
    
    print("YES")
    # For each satellite, find which setup corresponds to the matched frequency
    # Build a map from (satellite, frequency) to setup index
    sf_to_idx = {}
    for x in range(1, k + 1):
        for z, idx in sat_freqs[x]:
            sf_to_idx[(x, z)] = idx
    
    for x in range(1, k + 1):
        z = pair_u[x]
        idx = sf_to_idx[(x, z)]
        sx, sy, sz = setups[idx]
        print(sx, sy, sz)

if __name__ == "__main__":
    solve()
