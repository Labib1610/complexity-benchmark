```python
import sys
from collections import deque

sys.setrecursionlimit(10000)

input_buffer = sys.stdin.buffer
k, t = map(int, input_buffer.readline().split())

adj = [[] for _ in range(k + 1)]
secondary = [0] * (k + 1)

for _ in range(t):
    x, y, z = map(int, input_buffer.readline().split())
    adj[x].append(z)
    secondary[x] = y

pair_x = [0] * (k + 1)
pair_z = [0] * (k + 1)
dist = [0] * (k + 1)
INF = k + 1

def bfs():
    queue = deque()
    for x in range(1, k + 1):
        if pair_x[x] == 0:
            dist[x] = 0
            queue.append(x)
        else:
            dist[x] = INF

    dist[0] = INF

    while queue:
        x = queue.popleft()
        if dist[x] >= dist[0]:
            continue
        next_dist = dist[x] + 1
        for z in adj[x]:
            matched_x = pair_z[z]
            if dist[matched_x] == INF:
                dist[matched_x] = next_dist
                if matched_x != 0:
                    queue.append(matched_x)

    return dist[0] != INF

def dfs(x):
    if x == 0:
        return True

    next_dist = dist[x] + 1
    for z in adj[x]:
        matched_x = pair_z[z]
        if dist[matched_x] == next_dist and dfs(matched_x):
            pair_x[x] = z
            pair_z[z] = x
            return True

    dist[x] = INF
    return False

matching = 0

while bfs():
    for x in range(1, k + 1):
        if pair_x[x] == 0 and dfs(x):
            matching += 1

if matching != k:
    sys.stdout.write("NO\n")
else:
    output = ["YES"]
    output.extend(
        f"{x} {secondary[x]} {pair_x[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))
    sys.stdout.write("\n")
```
