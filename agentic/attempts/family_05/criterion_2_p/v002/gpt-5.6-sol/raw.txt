```python
import sys
from collections import deque

def main():
    input_buffer = sys.stdin.buffer
    k, t = map(int, input_buffer.readline().split())

    adjacency = [[] for _ in range(k)]
    kitchen = [0] * k

    for _ in range(t):
        x, y, z = map(int, input_buffer.readline().split())
        x -= 1
        z -= 1
        adjacency[x].append(z)
        kitchen[x] = y

    if any(not edges for edges in adjacency):
        print("NO")
        return

    chef_match = [-1] * k
    menu_match = [-1] * k
    distance = [0] * k
    inf = k + 1
    matched = 0

    sys.setrecursionlimit(k * 2 + 100)

    while True:
        queue = deque()
        for chef in range(k):
            if chef_match[chef] == -1:
                distance[chef] = 0
                queue.append(chef)
            else:
                distance[chef] = inf

        shortest = inf

        while queue:
            chef = queue.popleft()
            current_distance = distance[chef]

            if current_distance >= shortest:
                continue

            for menu in adjacency[chef]:
                other_chef = menu_match[menu]
                if other_chef == -1:
                    shortest = current_distance + 1
                elif distance[other_chef] == inf:
                    distance[other_chef] = current_distance + 1
                    queue.append(other_chef)

        if shortest == inf:
            break

        def augment(chef):
            next_distance = distance[chef] + 1

            for menu in adjacency[chef]:
                other_chef = menu_match[menu]

                if other_chef == -1:
                    if next_distance != shortest:
                        continue
                elif distance[other_chef] != next_distance or not augment(other_chef):
                    continue

                chef_match[chef] = menu
                menu_match[menu] = chef
                return True

            distance[chef] = inf
            return False

        for chef in range(k):
            if chef_match[chef] == -1 and augment(chef):
                matched += 1

    if matched != k:
        print("NO")
        return

    output = ["YES"]
    for chef in range(k):
        output.append(f"{chef + 1} {kitchen[chef]} {chef_match[chef] + 1}")

    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
