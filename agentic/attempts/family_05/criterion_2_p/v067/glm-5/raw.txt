Looking at this problem:

- Each drone maps to exactly one station (bijection), so the (drone, station) pairing is fixed.
- We need to select exactly k configurations such that each drone, station, and frequency appears exactly once.
- Since each drone maps to exactly one station, selecting one config per drone automatically covers all drones and stations uniquely.
- The remaining constraint: each frequency used exactly once → this is a bipartite matching problem (drones to frequencies).

For each drone, collect the set of available frequencies. Find a perfect matching. If found, output the corresponding (drone, station, frequency) triples.

Using Hopcroft-Karp for efficiency with k up to 1200.

```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    k = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    
    station_of = [0] * (k + 1)
    freqs_of = [[] for _ in range(k + 1)]
    
    for _ in range(t):
        d = int(input_data[idx]); idx += 1
        c = int(input_data[idx]); idx += 1
        f = int(input_data[idx]); idx += 1
        station_of[d] = c
        freqs_of[d].append(f)
    
    # Hopcroft-Karp: match drones (1..k) to frequencies (1..k)
    # Left side: drones, Right side: frequencies
    pair_u = [0] * (k + 1)  # pair_u[d] = matched frequency
    pair_v = [0] * (k + 1)  # pair_v[f] = matched drone
    dist = [0] * (k + 1)
    INF = float('inf')
    
    adj = [None] * (k + 1)
    for d in range(1, k + 1):
        adj[d] = freqs_of[d]
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching == k:
        out = ['YES']
        for d in range(1, k + 1):
            out.append(f'{d} {station_of[d]} {pair_u[d]}')
        sys.stdout.write('\n'.join(out) + '\n')
    else:
        sys.stdout.write('NO\n')

solve()
```
