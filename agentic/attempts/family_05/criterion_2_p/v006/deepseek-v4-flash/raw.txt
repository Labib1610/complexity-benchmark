We can model this as a bipartite matching problem between operators and shifts, since the station is fixed for each operator.

Here’s the solution:

```python
import sys

def main():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # station_for_operator[x] = y
    station_for_operator = [0] * (k + 1)
    # shifts_allowed[x] = set of z for operator x
    shifts_allowed = [set() for _ in range(k + 1)]
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        station_for_operator[x] = y
        shifts_allowed[x].add(z)
    
    # Build bipartite graph: operator -> shift
    adj = [[] for _ in range(k + 1)]
    for op in range(1, k + 1):
        adj[op] = list(shifts_allowed[op])
    
    # match_shift[shift] = operator assigned to this shift
    match_shift = [0] * (k + 1)
    
    def dfs(op, visited):
        for sh in adj[op]:
            if not visited[sh]:
                visited[sh] = True
                if match_shift[sh] == 0 or dfs(match_shift[sh], visited):
                    match_shift[sh] = op
                    return True
        return False
    
    for op in range(1, k + 1):
        visited = [False] * (k + 1)
        if not dfs(op, visited):
            print("NO")
            return
    
    print("YES")
    for sh in range(1, k + 1):
        op = match_shift[sh]
        print(op, station_for_operator[op], sh)

if __name__ == "__main__":
    main()
```
