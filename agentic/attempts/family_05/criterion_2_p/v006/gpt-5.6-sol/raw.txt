import sys
from collections import deque

def main():
    input_buffer = sys.stdin.buffer
    k, t = map(int, input_buffer.readline().split())

    adjacency = [[] for _ in range(k + 1)]
    station = [0] * (k + 1)

    for _ in range(t):
        x, y, z = map(int, input_buffer.readline().split())
        adjacency[x].append(z)
        station[x] = y

    if any(not adjacency[x] for x in range(1, k + 1)):
        print("NO")
        return

    pair_operator = [0] * (k + 1)
    pair_shift = [0] * (k + 1)
    distance = [0] * (k + 1)
    pointer = [0] * (k + 1)
    inf = k + 1

    sys.setrecursionlimit(2 * k + 100)

    def bfs():
        queue = deque()
        for x in range(1, k + 1):
            if pair_operator[x] == 0:
                distance[x] = 0
                queue.append(x)
            else:
                distance[x] = inf

        distance[0] = inf

        while queue:
            x = queue.popleft()
            if distance[x] >= distance[0]:
                continue

            next_distance = distance[x] + 1
            for z in adjacency[x]:
                matched_x = pair_shift[z]
                if distance[matched_x] == inf:
                    distance[matched_x] = next_distance
                    if matched_x != 0:
                        queue.append(matched_x)

        return distance[0] != inf

    def dfs(x):
        if x == 0:
            return True

        edges = adjacency[x]
        while pointer[x] < len(edges):
            z = edges[pointer[x]]
            pointer[x] += 1
            matched_x = pair_shift[z]

            if distance[matched_x] == distance[x] + 1 and dfs(matched_x):
                pair_operator[x] = z
                pair_shift[z] = x
                return True

        distance[x] = inf
        return False

    matching_size = 0

    while bfs():
        pointer = [0] * (k + 1)
        for x in range(1, k + 1):
            if pair_operator[x] == 0 and dfs(x):
                matching_size += 1

    if matching_size != k:
        print("NO")
        return

    output = ["YES"]
    for x in range(1, k + 1):
        output.append(f"{x} {station[x]} {pair_operator[x]}")

    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
