import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    k = int(data[idx]); idx += 1
    t = int(data[idx]); idx += 1
    
    # Build bipartite graph: reagent x -> pressure z, with edge id = trial index
    adj = [[] for _ in range(k + 1)]
    edges = []
    for i in range(t):
        x = int(data[idx]); idx += 1
        y = int(data[idx]); idx += 1
        z = int(data[idx]); idx += 1
        adj[x].append((z, i))
        edges.append((x, y, z))
    
    # Hopcroft-Karp to find perfect matching (reagent -> pressure)
    pairU = [0] * (k + 1)  # pairU[x] = z matched to reagent x
    pairV = [0] * (k + 1)  # pairV[z] = x matched to pressure z
    dist = [0] * (k + 1)
    INF = float('inf')
    NIL = 0
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pairU[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[NIL] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[NIL]:
                for v, _ in adj[u]:
                    if dist[pairV[v]] == INF:
                        dist[pairV[v]] = dist[u] + 1
                        queue.append(pairV[v])
        return dist[NIL] != INF
    
    def dfs(u):
        if u != 0:
            for v, _ in adj[u]:
                if dist[pairV[v]] == dist[u] + 1:
                    if dfs(pairV[v]):
                        pairV[v] = u
                        pairU[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pairU[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching < k:
        print("NO")
        return
    
    # We have perfect matching reagent->pressure. Now verify vessel constraint.
    # Since input is consistent with protocol: reagent x always in same vessel.
    # The matching picks one trial per (reagent, pressure). Check vessels are all distinct.
    vessel_of_reagent = {}
    result = []
    vessels_used = set()
    
    for x in range(1, k + 1):
        z = pairU[x]
        # find the edge (trial) for this (x, z)
        found = False
        for v, eid in adj[x]:
            if v == z:
                _, y, zz = edges[eid]
                result.append((x, y, zz))
                if x in vessel_of_reagent:
                    if vessel_of_reagent[x] != y:
                        print("NO")
                        return
                else:
                    vessel_of_reagent[x] = y
                vessels_used.add(y)
                found = True
                break
        if not found:
            print("NO")
            return
    
    if len(vessels_used) < k:
        print("NO")
        return
    
    print("YES")
    out = []
    for x, y, z in result:
        out.append(f"{x} {y} {z}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
