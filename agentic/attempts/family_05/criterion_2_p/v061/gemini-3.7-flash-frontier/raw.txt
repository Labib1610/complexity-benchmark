```python
import sys

# Increase recursion depth for DFS in Hopcroft-Karp
sys.setrecursionlimit(10000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])

    idx = 2
    adj = [[] for _ in range(k + 1)]
    vessel = [0] * (k + 1)

    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx + 1])
        z = int(input_data[idx + 2])
        idx += 3
        adj[x].append(z)
        vessel[x] = y

    match_L = [0] * (k + 1)
    match_R = [0] * (k + 1)
    dist = [0] * (k + 1)
    INF = 10**9

    def bfs():
        queue = []
        for u in range(1, k + 1):
            if match_L[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            if dist[u] < dist[0]:
                for v in adj[u]:
                    nxt = match_R[v]
                    if dist[nxt] == INF:
                        dist[nxt] = dist[u] + 1
                        queue.append(nxt)
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u]:
                nxt = match_R[v]
                if dist[nxt] == dist[u] + 1:
                    if dfs(nxt):
                        match_R[v] = u
                        match_L[u] = v
                        return True
            dist[u] = INF
            return False
        return True

    matching_size = 0
    # Greedy initialization
    for u in range(1, k + 1):
        for v in adj[u]:
            if match_R[v] == 0:
                match_L[u] = v
                match_R[v] = u
                matching_size += 1
                break

    while bfs():
        for u in range(1, k + 1):
            if match_L[u] == 0:
                if dfs(u):
                    matching_size += 1

    if matching_size == k:
        out = ["YES"]
        for u in range(1, k + 1):
            out.append(f"{u} {vessel[u]} {match_L[u]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")


if __name__ == "__main__":
    solve()
```
