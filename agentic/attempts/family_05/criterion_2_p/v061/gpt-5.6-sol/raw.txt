import sys
from collections import deque

sys.setrecursionlimit(10000)

def main():
    input_buffer = sys.stdin.buffer
    k, t = map(int, input_buffer.readline().split())

    adjacency = [[] for _ in range(k + 1)]
    vessel = [0] * (k + 1)

    pair_left = [0] * (k + 1)
    pair_right = [0] * (k + 1)
    matched = 0

    for _ in range(t):
        x, y, z = map(int, input_buffer.readline().split())
        adjacency[x].append(z)
        vessel[x] = y

        if pair_left[x] == 0 and pair_right[z] == 0:
            pair_left[x] = z
            pair_right[z] = x
            matched += 1

    if t < k:
        print("NO")
        return

    inf = k + 1
    distance = [inf] * (k + 1)

    while matched < k:
        queue = deque()
        shortest = inf

        for x in range(1, k + 1):
            if pair_left[x] == 0:
                distance[x] = 0
                queue.append(x)
            else:
                distance[x] = inf

        while queue:
            x = queue.popleft()
            next_distance = distance[x] + 1
            if next_distance > shortest:
                continue

            for z in adjacency[x]:
                other_x = pair_right[z]
                if other_x == 0:
                    if next_distance < shortest:
                        shortest = next_distance
                elif distance[other_x] == inf and next_distance < shortest:
                    distance[other_x] = next_distance
                    queue.append(other_x)

        if shortest == inf:
            print("NO")
            return

        pointer = [0] * (k + 1)

        def dfs(x):
            edges = adjacency[x]
            while pointer[x] < len(edges):
                z = edges[pointer[x]]
                pointer[x] += 1
                other_x = pair_right[z]

                if other_x == 0:
                    if distance[x] + 1 != shortest:
                        continue
                elif distance[other_x] != distance[x] + 1 or not dfs(other_x):
                    continue

                pair_left[x] = z
                pair_right[z] = x
                return True

            distance[x] = inf
            return False

        augmented = 0
        for x in range(1, k + 1):
            if pair_left[x] == 0 and dfs(x):
                augmented += 1

        if augmented == 0:
            print("NO")
            return

        matched += augmented

    output = ["YES"]
    output.extend(
        f"{x} {vessel[x]} {pair_left[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
