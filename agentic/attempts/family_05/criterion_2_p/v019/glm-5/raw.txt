import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # pilot_to_starfighter[p] = s (the unique starfighter for pilot p)
    pilot_to_starfighter = {}
    # starfighter_to_pilot[s] = p
    starfighter_to_pilot = {}
    
    # missions_for_pilot[p] = set of missions available for pilot p
    missions_for_pilot = [set() for _ in range(k + 1)]
    
    for _ in range(t):
        p = int(next(it))
        s = int(next(it))
        m = int(next(it))
        
        # Record the unique mapping between pilot and starfighter
        if p not in pilot_to_starfighter:
            pilot_to_starfighter[p] = s
            starfighter_to_pilot[s] = p
        else:
            # Problem guarantees consistency, but we rely on the first entry
            pass
            
        missions_for_pilot[p].add(m)
    
    # Check if all pilots 1..k are present in the input
    if len(pilot_to_starfighter) != k:
        print("NO")
        return
    
    # Bipartite matching between pilots (left) and missions (right)
    # Pilot p is matched to mission matchR[m]
    matchR = [0] * (k + 1)  # 1-indexed missions
    
    def bfs():
        dist = {}
        q = deque()
        for p in range(1, k + 1):
            if matchL[p] == 0:
                dist[p] = 0
                q.append(p)
            else:
                dist[p] = -1
        
        found_augmenting_path = False
        while q:
            u = q.popleft()
            for m in missions_for_pilot[u]:
                v = matchR[m]
                if v != 0:
                    if dist.get(v, -1) == -1:
                        dist[v] = dist[u] + 1
                        q.append(v)
                else:
                    # Reached a free mission node, potential to augment
                    found_augmenting_path = True
        return found_augmenting_path

    def dfs(u):
        for m in missions_for_pilot[u]:
            v = matchR[m]
            if v == 0:
                matchR[m] = u
                matchL[u] = m
                return True
            else:
                if dist.get(v, -1) == dist[u] + 1:
                    if dfs(v):
                        matchR[m] = u
                        matchL[u] = m
                        return True
        return False

    matchL = [0] * (k + 1)
    matching_size = 0
    
    # Hopcroft-Karp
    while bfs():
        for p in range(1, k + 1):
            if matchL[p] == 0:
                if dfs(p):
                    matching_size += 1
    
    if matching_size != k:
        print("NO")
        return
    
    print("YES")
    # Output the k deployments
    # matchL[p] gives the mission assigned to pilot p
    for p in range(1, k + 1):
        m = matchL[p]
        s = pilot_to_starfighter[p]
        print(f"{p} {s} {m}")

if __name__ == "__main__":
    solve()
