```python
import sys
from collections import deque

def main():
    readline = sys.stdin.buffer.readline
    k, t = map(int, readline().split())

    adjacency = [[] for _ in range(k + 1)]
    starfighter = [0] * (k + 1)

    for _ in range(t):
        p, s, m = map(int, readline().split())
        starfighter[p] = s
        adjacency[p].append(m)

    if any(not adjacency[p] for p in range(1, k + 1)):
        print("NO")
        return

    pair_pilot = [0] * (k + 1)
    pair_mission = [0] * (k + 1)
    distance = [0] * (k + 1)
    inf = k + 1

    sys.setrecursionlimit(5000)

    def bfs():
        queue = deque()
        for p in range(1, k + 1):
            if pair_pilot[p] == 0:
                distance[p] = 0
                queue.append(p)
            else:
                distance[p] = inf

        distance[0] = inf

        while queue:
            p = queue.popleft()
            if distance[p] >= distance[0]:
                continue

            next_distance = distance[p] + 1
            for mission in adjacency[p]:
                matched_pilot = pair_mission[mission]
                if distance[matched_pilot] == inf:
                    distance[matched_pilot] = next_distance
                    if matched_pilot != 0:
                        queue.append(matched_pilot)

        return distance[0] != inf

    def dfs(p):
        if p == 0:
            return True

        next_distance = distance[p] + 1
        for mission in adjacency[p]:
            matched_pilot = pair_mission[mission]
            if distance[matched_pilot] == next_distance and dfs(matched_pilot):
                pair_pilot[p] = mission
                pair_mission[mission] = p
                return True

        distance[p] = inf
        return False

    matching_size = 0
    while bfs():
        for p in range(1, k + 1):
            if pair_pilot[p] == 0 and dfs(p):
                matching_size += 1

    if matching_size != k:
        print("NO")
        return

    output = ["YES"]
    output.extend(
        f"{p} {starfighter[p]} {pair_pilot[p]}"
        for p in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
