from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    index = 2
    deployments = []
    pilot_to_starfighter = {}
    
    # Read deployments and build pilot->starfighter mapping
    for _ in range(t):
        p = int(data[index])
        s = int(data[index+1])
        m = int(data[index+2])
        deployments.append((p, s, m))
        pilot_to_starfighter[p] = s
        index += 3
    
    # Build bipartite graph: missions on the right side
    # Left side is pilots (which are tied to specific starfighters)
    # So we can think of this as matching pilots/starfighters to missions
    
    # For each mission, keep track of which deployments use it
    mission_deployments = defaultdict(list)
    for i, (p, s, m) in enumerate(deployments):
        mission_deployments[m].append(i)
    
    # For each pilot, keep track of which deployments belong to it
    pilot_deployments = defaultdict(list)
    for i, (p, s, m) in enumerate(deployments):
        pilot_deployments[p].append(i)
    
    # Build a graph for matching: nodes are pilots, edges are possible missions
    # Since each pilot has exactly one starfighter, we just need to match pilots to missions
    
    # Create adjacency list for pilots to possible missions
    pilot_to_missions = defaultdict(list)
    for i, (p, s, m) in enumerate(deployments):
        pilot_to_missions[p].append((m, i))  # mission, deployment_index
    
    # Now we want a maximum bipartite matching between pilots and missions
    # Each pilot can be matched to at most one mission (and vice versa since we need exactly k)
    
    # Implementing Hopcroft-Karp or similar would be overkill; let's use a simpler approach
    # Since constraints are small enough, we can use a greedy + backtracking approach or Hungarian
    
    # Actually, let's model this properly:
    # We have k pilots, each can do certain missions (based on deployments)
    # We want to assign each pilot to exactly one mission, such that all missions are covered
    # This is a perfect bipartite matching problem
    
    # Let's build a graph where left side is pilots, right side is missions
    graph = defaultdict(list)
    for p in range(1, k+1):
        if p in pilot_to_missions:
            for m, idx in pilot_to_missions[p]:
                graph[p].append((m, idx))
    
    # Maximum bipartite matching implementation
    def dfs(u, match_right, match_left, visited):
        for m, idx in graph[u]:
            if m in visited:
                continue
            visited.add(m)
            
            if m not in match_right or dfs(match_right[m], match_right, match_left, visited):
                match_right[m] = u
                match_left[u] = m
                return True
        return False
    
    match_left = {}   # pilot -> mission
    match_right = {}  # mission -> pilot
    
    matching = 0
    for pilot in range(1, k+1):
        visited = set()
        if dfs(pilot, match_right, match_left, visited):
            matching += 1
    
    if matching < k:
        print("NO")
        return
    
    print("YES")
    result_deployments = {}
    # Reconstruct the actual deployments used
    for m, p in match_right.items():
        # Find the deployment index for pilot p and mission m
        for m_val, idx in pilot_to_missions[p]:
            if m_val == m:
                result_deployments[p] = idx
                break
    
    for pilot in range(1, k+1):
        idx = result_deployments[pilot]
        p, s, m = deployments[idx]
        print(p, s, m)

solve()
