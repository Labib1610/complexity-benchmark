```python
import sys
from collections import deque

def main():
    input_data = sys.stdin.buffer
    first = input_data.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k + 1)]
    waveplate = [0] * (k + 1)
    match_source = [0] * (k + 1)
    match_sensor = [0] * (k + 1)
    matched = 0

    for _ in range(t):
        x, y, z = map(int, input_data.readline().split())
        adjacency[x].append(z)
        waveplate[x] = y

        if match_source[x] == 0 and match_sensor[z] == 0:
            match_source[x] = z
            match_sensor[z] = x
            matched += 1

    if any(not adjacency[x] for x in range(1, k + 1)):
        sys.stdout.write("NO\n")
        return

    sys.setrecursionlimit(10000)
    distance = [0] * (k + 1)
    inf = k + 1

    def bfs():
        queue = deque()
        shortest = inf

        for source in range(1, k + 1):
            if match_source[source] == 0:
                distance[source] = 0
                queue.append(source)
            else:
                distance[source] = inf

        while queue:
            source = queue.popleft()
            if distance[source] >= shortest:
                continue

            next_distance = distance[source] + 1
            for sensor in adjacency[source]:
                next_source = match_sensor[sensor]
                if next_source == 0:
                    shortest = next_distance
                elif distance[next_source] == inf:
                    distance[next_source] = next_distance
                    queue.append(next_source)

        return shortest

    def dfs(source, shortest):
        next_distance = distance[source] + 1

        for sensor in adjacency[source]:
            next_source = match_sensor[sensor]

            if next_source == 0:
                if next_distance == shortest:
                    match_source[source] = sensor
                    match_sensor[sensor] = source
                    return True
            elif distance[next_source] == next_distance and dfs(next_source, shortest):
                match_source[source] = sensor
                match_sensor[sensor] = source
                return True

        distance[source] = inf
        return False

    while matched < k:
        shortest = bfs()
        if shortest == inf:
            break

        for source in range(1, k + 1):
            if match_source[source] == 0 and dfs(source, shortest):
                matched += 1

    if matched != k:
        sys.stdout.write("NO\n")
        return

    output = ["YES"]
    output.extend(
        f"{source} {waveplate[source]} {match_source[source]}"
        for source in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
