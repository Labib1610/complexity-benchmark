from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    index = 2
    configs = []
    # We'll build a bipartite graph between guide sensors and spectrometers
    # Since each celestial target is paired with exactly one guide sensor,
    # and no two targets share a guide sensor, we can treat guide sensors
    # as proxies for celestial targets.
    
    # Map each guide sensor to list of (spectrometer, config_index)
    adj = defaultdict(list)
    
    for i in range(t):
        c = int(data[index])
        g = int(data[index+1])
        s = int(data[index+2])
        index += 3
        configs.append((c, g, s))
        adj[g].append((s, i))
    
    # Now we need to match each guide sensor to a unique spectrometer
    # This is a maximum bipartite matching problem
    
    # For each guide sensor, we need to match it to one spectrometer
    # Guide sensors are 1..k, spectrometers are 1..k
    
    # Build graph for bipartite matching
    # Left side: guide sensors (1 to k) 
    # Right side: spectrometers (1 to k)
    
    # adj_list[g] contains list of spectrometers that guide sensor g can connect to
    adj_list = [[] for _ in range(k+1)]
    for g in range(1, k+1):
        for s, idx in adj[g]:
            adj_list[g].append(s)
    
    # Maximum bipartite matching using Hopcroft-Karp or similar
    # We'll use a simpler augmenting path approach
    
    match_guide_to_spec = [0] * (k+1)  # Which spectrometer is matched to guide sensor
    match_spec_to_guide = [0] * (k+1)  # Which guide sensor is matched to spectrometer
    
    def bfs():
        # BFS to find shortest augmenting paths
        dist = [0] * (k+1)
        queue = deque()
        for g in range(1, k+1):
            if match_guide_to_spec[g] == 0:
                dist[g] = 0
                queue.append(g)
            else:
                dist[g] = float('inf')
        dist[0] = float('inf')
        
        while queue:
            g = queue.popleft()
            if dist[g] < dist[0]:
                for s in adj_list[g]:
                    matched_g = match_spec_to_guide[s]
                    if dist[matched_g] == float('inf'):
                        dist[matched_g] = dist[g] + 1
                        queue.append(matched_g)
        return dist[0] != float('inf')
    
    def dfs(g):
        if g != 0:
            for s in adj_list[g]:
                matched_g = match_spec_to_guide[s]
                if dist[matched_g] == dist[g] + 1:
                    if dfs(matched_g):
                        match_guide_to_spec[g] = s
                        match_spec_to_guide[s] = g
                        return True
            dist[g] = float('inf')
            return False
        return True
    
    # Initial matching
    matching = 0
    dist = [0] * (k+1)
    while bfs():
        for g in range(1, k+1):
            if match_guide_to_spec[g] == 0 and dfs(g):
                matching += 1
                
    if matching < k:
        print("NO")
        return
        
    print("YES")
    result = []
    for g in range(1, k+1):
        s = match_guide_to_spec[g]
        # Find the actual config index
        for spec, idx in adj[g]:
            if spec == s:
                c = configs[idx][0]
                result.append(f"{c} {g} {s}")
                break
    print('\n'.join(result))

solve()
