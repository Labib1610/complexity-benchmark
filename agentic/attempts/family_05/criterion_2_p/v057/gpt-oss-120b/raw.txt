```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    adj = [[] for _ in range(k + 1)]          # 1-indexed left side (celestial)
    guide_of = [0] * (k + 1)                  # store the paired guide sensor for each c
    for _ in range(t):
        c = int(next(it)); g = int(next(it)); s = int(next(it))
        adj[c].append(s)
        guide_of[c] = g   # all edges for same c have same g, safe to overwrite

    # Hopcroft–Karp
    INF = 10 ** 9
    pair_u = [0] * (k + 1)   # left -> right
    pair_v = [0] * (k + 1)   # right -> left (spectrometer banks also 1..k)
    dist = [0] * (k + 1)

    def bfs() -> bool:
        q = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_v[v] == 0:
                        if dist[0] == INF:
                            dist[0] = dist[u] + 1
                    elif dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return dist[0] != INF

    def dfs(u: int) -> bool:
        if u == 0:
            return True
        for v in adj[u]:
            pu = pair_v[v]
            if pu == 0 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0 and dfs(u):
                matching += 1

    if matching != k:
        print("NO")
        return

    print("YES")
    out_lines = []
    for c in range(1, k + 1):
        s = pair_u[c]
        g = guide_of[c] if guide_of[c] != 0 else c  # fallback, though should be set
        out_lines.append(f"{c} {g} {s}")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```
