```python
import sys
from array import array

def main():
    readline = sys.stdin.buffer.readline
    k, t = map(int, readline().split())

    adj = [array('H') for _ in range(k)]
    guide = [-1] * k

    for _ in range(t):
        c, g, s = map(int, readline().split())
        c -= 1
        g -= 1
        s -= 1
        guide[c] = g
        adj[c].append(s)

    if t < k or any(g == -1 for g in guide) or any(not edges for edges in adj):
        print("NO")
        return

    sys.setrecursionlimit(5000)

    match_left = [-1] * k
    match_right = [-1] * k
    matched = 0

    while True:
        dist = [-1] * k
        queue = [0] * k
        head = 0
        tail = 0

        for u in range(k):
            if match_left[u] == -1:
                dist[u] = 0
                queue[tail] = u
                tail += 1

        shortest = k + 1

        while head < tail:
            u = queue[head]
            head += 1
            next_dist = dist[u] + 1

            if next_dist > shortest:
                continue

            for v in adj[u]:
                other = match_right[v]
                if other == -1:
                    shortest = next_dist
                elif next_dist < shortest and dist[other] == -1:
                    dist[other] = next_dist
                    queue[tail] = other
                    tail += 1

        if shortest == k + 1:
            break

        ptr = [0] * k

        def dfs(u):
            edges = adj[u]
            i = ptr[u]
            next_dist = dist[u] + 1

            while i < len(edges):
                v = edges[i]
                i += 1
                ptr[u] = i
                other = match_right[v]

                if other == -1:
                    if next_dist != shortest:
                        continue
                elif next_dist >= shortest or dist[other] != next_dist or not dfs(other):
                    continue

                match_left[u] = v
                match_right[v] = u
                return True

            dist[u] = -1
            return False

        augmented = 0
        for u in range(k):
            if match_left[u] == -1 and dfs(u):
                augmented += 1

        if augmented == 0:
            break
        matched += augmented

    if matched != k:
        print("NO")
        return

    output = ["YES"]
    output.extend(
        f"{c + 1} {guide[c] + 1} {match_left[c] + 1}"
        for c in range(k)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
