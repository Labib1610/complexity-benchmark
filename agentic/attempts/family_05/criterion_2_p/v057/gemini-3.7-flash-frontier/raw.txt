```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    adj = [[] for _ in range(k + 1)]
    sensor = [0] * (k + 1)
    
    idx = 2
    for _ in range(t):
        c = int(input_data[idx])
        g = int(input_data[idx+1])
        s = int(input_data[idx+2])
        idx += 3
        adj[c].append(s)
        sensor[c] = g
        
    pair_u = [0] * (k + 1)
    pair_v = [0] * (k + 1)
    dist = [0] * (k + 1)
    
    INF = 10**9
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
                    
    if matching == k:
        out = ["YES"]
        for c in range(1, k + 1):
            s = pair_u[c]
            g = sensor[c]
            out.append(f"{c} {g} {s}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
