import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    # target -> list of (g, s)
    target_options = [[] for _ in range(k + 1)]
    for _ in range(t):
        c = int(data[idx]); g = int(data[idx+1]); s = int(data[idx+2])
        idx += 3
        target_options[c].append((g, s))
    
    # For each target, we must pick exactly one option (g, s).
    # Since c and g are permanently paired, picking an option for target c
    # automatically uses guide sensor g (unique per target) and spectrometer s.
    # We need to choose one option per target such that all spectrometers 1..k are used exactly once.
    # This is a bipartite matching: targets on left, spectrometers on right.
    # Each target c has edges to spectrometers s from its options.
    
    # Kuhn's algorithm for maximum bipartite matching
    match_spectrometer = [-1] * (k + 1)  # which target is matched to spectrometer s
    match_target = [-1] * (k + 1)        # which spectrometer is matched to target c
    # We'll store the chosen (g, s) for each target
    chosen_option = [None] * (k + 1)
    
    # Build adjacency: for each target, unique spectrometers
    target_spec_edges = [[] for _ in range(k + 1)]
    for c in range(1, k + 1):
        seen = set()
        for (g, s) in target_options[c]:
            if s not in seen:
                seen.add(s)
                target_spec_edges[c].append((s, g))
    
    visited = [0] * (k + 1)
    visit_token = 0
    
    def try_kuhn(u):
        for (s, g) in target_spec_edges[u]:
            if visited[s] == visit_token:
                continue
            visited[s] = visit_token
            if match_spectrometer[s] == -1 or try_kuhn(match_spectrometer[s]):
                match_spectrometer[s] = u
                match_target[u] = s
                chosen_option[u] = (g, s)
                return True
        return False
    
    for c in range(1, k + 1):
        visit_token += 1
        if not try_kuhn(c):
            print("NO")
            return
    
    # All matched
    print("YES")
    out_lines = []
    for c in range(1, k + 1):
        g, s = chosen_option[c]
        out_lines.append(f"{c} {g} {s}")
    sys.stdout.write("\n".join(out_lines))

solve()
