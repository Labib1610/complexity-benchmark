import sys
from collections import deque

def main():
    data = sys.stdin.buffer
    first = data.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k + 1)]
    aircraft = [0] * (k + 1)
    route_seen = [False] * (k + 1)

    for _ in range(t):
        x, y, z = map(int, data.readline().split())
        adjacency[x].append(z)
        aircraft[x] = y
        route_seen[z] = True

    if any(not adjacency[x] for x in range(1, k + 1)):
        print("NO")
        return
    if not all(route_seen[1:]):
        print("NO")
        return

    match_pilot = [0] * (k + 1)
    match_route = [0] * (k + 1)
    matched = 0

    # Build an initial greedy matching, prioritizing constrained pilots.
    for x in sorted(range(1, k + 1), key=lambda u: len(adjacency[u])):
        for z in adjacency[x]:
            if match_route[z] == 0:
                match_pilot[x] = z
                match_route[z] = x
                matched += 1
                break

    sys.setrecursionlimit(5000)
    INF = k + 1

    while matched < k:
        dist = [INF] * (k + 1)
        queue = deque()

        for x in range(1, k + 1):
            if match_pilot[x] == 0:
                dist[x] = 0
                queue.append(x)

        shortest = INF

        while queue:
            x = queue.popleft()
            if dist[x] + 1 > shortest:
                continue

            next_dist = dist[x] + 1
            for z in adjacency[x]:
                other = match_route[z]
                if other == 0:
                    shortest = next_dist
                elif next_dist < shortest and dist[other] == INF:
                    dist[other] = next_dist
                    queue.append(other)

        if shortest == INF:
            print("NO")
            return

        edge_index = [0] * (k + 1)

        def augment(x):
            edges = adjacency[x]
            i = edge_index[x]

            while i < len(edges):
                z = edges[i]
                i += 1
                edge_index[x] = i
                other = match_route[z]

                if other == 0:
                    if dist[x] + 1 == shortest:
                        match_pilot[x] = z
                        match_route[z] = x
                        return True
                elif dist[other] == dist[x] + 1 and augment(other):
                    match_pilot[x] = z
                    match_route[z] = x
                    return True

            dist[x] = INF
            return False

        added = 0
        for x in range(1, k + 1):
            if match_pilot[x] == 0 and augment(x):
                added += 1

        if added == 0:
            print("NO")
            return
        matched += added

    output = ["YES"]
    output.extend(
        f"{x} {aircraft[x]} {match_pilot[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
