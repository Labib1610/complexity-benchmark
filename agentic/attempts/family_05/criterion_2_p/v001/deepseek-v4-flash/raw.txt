We can model this as a bipartite matching problem between pilots and routes, since the aircraft is fixed per pilot.  
We need to match each pilot to a distinct route using only allowed combinations.

import sys
from collections import defaultdict, deque

def main():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # Since each pilot has exactly one aircraft, we can map pilot -> route
    allowed = defaultdict(list)
    for _ in range(t):
        x, y, z = map(int, input().split())
        allowed[x].append(z)
    
    # Bipartite matching: pilots (1..k) to routes (1..k)
    match_route = [-1] * (k + 1)  # route -> pilot
    
    def bfs():
        dist = [-1] * (k + 1)
        q = deque()
        for p in range(1, k + 1):
            if match_route[p] == -1:
                dist[p] = 0
                q.append(p)
            else:
                dist[p] = -1
        found = False
        while q:
            p = q.popleft()
            for r in allowed[p]:
                p2 = match_route[r]
                if p2 == -1:
                    found = True
                elif dist[p2] == -1:
                    dist[p2] = dist[p] + 1
                    q.append(p2)
        return found, dist
    
    def dfs(p, dist):
        for r in allowed[p]:
            p2 = match_route[r]
            if p2 == -1 or (dist[p2] == dist[p] + 1 and dfs(p2, dist)):

                return True
