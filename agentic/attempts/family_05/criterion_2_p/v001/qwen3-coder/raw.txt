from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    index = 2
    combinations = []
    pilot_aircraft = {}  # pilot -> aircraft
    
    # Read combinations and build pilot->aircraft mapping
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        combinations.append((x, y, z))
        if x not in pilot_aircraft:
            pilot_aircraft[x] = y
        index += 3
    
    # Build bipartite graph: pilots <-> routes
    # Edge exists if combination (pilot, aircraft, route) is allowed
    graph = defaultdict(list)  # route -> list of pilots that can use it
    pilot_routes = defaultdict(list)  # pilot -> list of routes they can use
    
    for x, y, z in combinations:
        if pilot_aircraft[x] == y:  # Check consistency
            graph[z].append(x)
            pilot_routes[x].append(z)
    
    # Try to find a perfect matching using DFS-based augmenting paths
    match_pilot_to_route = {}  # pilot -> route
    match_route_to_pilot = {}  # route -> pilot
    
    def dfs(pilot, visited_routes):
        for route in pilot_routes[pilot]:
            if route in visited_routes:
                continue
            visited_routes.add(route)
            # If route is unmatched or we can find augmenting path
            if route not in match_route_to_pilot or dfs(match_route_to_pilot[route], visited_routes):
                match_pilot_to_route[pilot] = route
                match_route_to_pilot[route] = pilot
                return True
        return False
    
    # Try to match all pilots
    matched_pilots = 0
    for pilot in range(1, k+1):
        visited_routes = set()
        if dfs(pilot, visited_routes):
            matched_pilots += 1
    
    if matched_pilots != k:
        print("NO")
        return
    
    print("YES")
    for pilot in range(1, k+1):
        route = match_pilot_to_route[pilot]
        aircraft = pilot_aircraft[pilot]
        print(pilot, aircraft, route)

solve()
