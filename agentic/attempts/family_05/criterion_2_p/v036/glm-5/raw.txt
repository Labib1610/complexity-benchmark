import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        k = int(next(iterator))
        t = int(next(iterator))
    except StopIteration:
        return

    # We need to find a perfect matching in a bipartite graph:
    # Left side: directors (1..k)
    # Right side: screening rooms (1..k)
    # Each director is pre-matched with a unique film.
    configs = []
    # adjacency list: adj[d] = list of screening rooms available for director d
    adj = [[] for _ in range(k + 1)]
    # store configs to retrieve them later: config_map[d][s] = (d, f, s)
    config_map = [{} for _ in range(k + 1)]

    for _ in range(t):
        d = int(next(iterator))
        f = int(next(iterator))
        s = int(next(iterator))
        adj[d].append(s)
        config_map[d][s] = (d, f, s)

    # Hopcroft-Karp algorithm
    pair_U = [0] * (k + 1)  # pair_U[d] = s matched to d
    pair_V = [0] * (k + 1)  # pair_V[s] = d matched to s
    dist = [0] * (k + 1)

    INF = 0

    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_U[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_V[v] == 0:
                        # If v is free, we can reach the dummy node 0
                        if dist[0] == INF:
                            dist[0] = dist[u] + 1
                    else:
                        # If v is matched, go to its partner
                        if dist[pair_V[v]] == INF:
                            dist[pair_V[v]] = dist[u] + 1
                            queue.append(pair_V[v])
        return dist[0] != INF

    def dfs(u):
        for v in adj[u]:
            if pair_V[v] == 0:
                if dist[0] == dist[u] + 1:
                    pair_U[u] = v
                    pair_V[v] = u
                    return True
            else:
                u2 = pair_V[v]
                if dist[u2] == dist[u] + 1:
                    if dfs(u2):
                        pair_U[u] = v
                        pair_V[v] = u
                        return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_U[u] == 0:
                if dfs(u):
                    matching += 1

    if matching == k:
        sys.stdout.write("YES\n")
        for d in range(1, k + 1):
            s = pair_U[d]
            # retrieve original config
            # config_map[d][s] must exist
            d_out, f_out, s_out = config_map[d][s]
            sys.stdout.write(f"{d_out} {f_out} {s_out}\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
