We can model this as a bipartite matching problem: each director–film pair is fixed (since each director has exactly one film), so we only need to assign each director to a distinct screening room.

That means we have a bipartite graph between directors and rooms, and we need a perfect matching. Since k can be up to 1200, we can use the Hopcroft–Karp algorithm for efficiency.

```python
import sys
from collections import deque

def hopcroft_karp(graph, n_left, n_right):
    INF = float('inf')
    pair_u = [-1] * (n_left + 1)
    pair_v = [-1] * (n_right + 1)
    dist = [0] * (n_left + 1)

    def bfs():
        q = deque()
        for u in range(1, n_left + 1):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        found = False
        while q:
            u = q.popleft()
            for v in graph[u]:
                pu = pair_v[v]
                if pu != -1 and dist[pu] == INF:
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in graph[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, n_left + 1):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    # director -> film mapping
    director_film = [0] * (k + 1)
    # adjacency: director -> list of rooms
    graph = [[] for _ in range(k + 1)]

    for _ in range(t):
        d = int(next(it))
        f = int(next(it))
        s = int(next(it))
        director_film[d] = f
        graph[d].append(s)

    # Hopcroft-Karp
    matching_size, pair_u = hopcroft_karp(graph, k, k)

    if matching_size != k:
        print("NO")
        return

    print("YES")
    for d in range(1, k + 1):
        s = pair_u[d]
        print(f"{d} {director_film[d]} {s}")

if __name__ == "__main__":
    main()
```
