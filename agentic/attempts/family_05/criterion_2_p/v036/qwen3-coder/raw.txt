from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    # If t < k, impossible since we need exactly k configurations
    if t < k:
        print("NO")
        return
    
    configs = []
    director_films = {}  # director -> film
    film_to_director = {}  # film -> director
    
    # Read configurations and build mappings
    for _ in range(t):
        d, f, s = map(int, input().split())
        configs.append((d, f, s))
        if d not in director_films:
            director_films[d] = f
            film_to_director[f] = d
    
    # Check if we have all directors and films
    if len(director_films) < k or len(film_to_director) < k:
        print("NO")
        return
    
    # Build a bipartite graph: directors <-> screening rooms
    # Edge exists if there's a config with that director and screening room
    adj = defaultdict(list)  # director -> list of screening rooms
    edges = {}  # (director, room) -> list of config indices
    
    for i, (d, f, s) in enumerate(configs):
        if (d, s) not in edges:
            edges[(d, s)] = []
        edges[(d, s)].append(i)
        adj[d].append(s)
    
    # Check if each director appears at least once
    for director in range(1, k+1):
        if director not in adj:
            print("NO")
            return
    
    # Find maximum matching using Hall's theorem approach with Hungarian algorithm concept
    # But since we just need to check if perfect matching exists, we can use a simpler approach
    
    # For this problem, we can model it as a bipartite matching:
    # Left side: directors (1..k)
    # Right side: screening rooms (1..k)
    # Edges exist where there's a valid configuration
    
    # We'll use a greedy approach with backtracking via DFS matching
    
    match_room_to_director = [0] * (k + 1)  # which director is assigned to room
    match_director_to_room = [0] * (k + 1)  # which room is assigned to director
    visited = [False] * (k + 1)
    
    def dfs(director):
        for room in adj[director]:
            if visited[room]:
                continue
            visited[room] = True
            
            # Room is free or we can rematch
            if match_room_to_director[room] == 0 or dfs(match_room_to_director[room]):
                match_room_to_director[room] = director
                match_director_to_room[director] = room
                return True
        return False
    
    # Try to find a perfect matching
    matched = 0
    for director in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(director):
            matched += 1
    
    if matched < k:
        print("NO")
        return
    
    # Reconstruct the solution
    result = []
    used_configs = set()
    
    for director in range(1, k + 1):
        room = match_director_to_room[director]
        film = director_films[director]
        
        # Find the actual config index
        found_config = False
        for i, (d, f, s) in enumerate(configs):
            if d == director and s == room and f == film and i not in used_configs:
                result.append((d, f, s))
                used_configs.add(i)
                found_config = True
                break
        
        if not found_config:
            print("NO")
            return
    
    print("YES")
    for d, f, s in result:
        print(d, f, s)

solve()
