We can model this as a bipartite matching problem.  

- Each `(p, a)` pair appears at most once for a given `a` because different `p` have different `a`.  
- For each `(p, a)`, we have a set of possible `m` values.  
- We need to pick exactly one `m` for each `(p, a)` such that all `m` values 1..k are used exactly once.  

That is equivalent to finding a perfect matching between the `(p,a)` pairs and the `m` values in a bipartite graph, where `(p,a)` is on one side and `m` on the other.  

Since `k ≤ 1200`, we can use Hopcroft–Karp for O(E√V).

Here’s the implementation:

```python
import sys
from collections import deque

def hopcroft_karp(graph, U_count, V_count):
    pair_u = [-1] * (U_count + 1)
    pair_v = [-1] * (V_count + 1)
    dist = [0] * (U_count + 1)

    def bfs():
        q = deque()
        for u in range(1, U_count + 1):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = float('inf')
        found = False
        while q:
            u = q.popleft()
            for v in graph[u]:
                pu = pair_v[v]
                if pu != -1 and dist[pu] == float('inf'):
                    dist[pu] = dist[u] + 1
                    q.append(pu)
                elif pu == -1:
                    found = True
        return found

    def dfs(u):
        for v in graph[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = float('inf')
        return False

    matching = 0
    while bfs():
        for u in range(1, U_count + 1):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u, pair_v

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # Map (p,a) -> list of m
    pair_to_m = {}
    # Map p -> a
    p_to_a = {}
    
    for _ in range(t):
        p = int(next(it))
        a = int(next(it))
        m = int(next(it))
        key = (p, a)
        if key not in pair_to_m:
            pair_to_m[key] = []
        pair_to_m[key].append(m)
        p_to_a[p] = a
    
    # Number of distinct (p,a) pairs
    pairs = list(pair_to_m.keys())
    pair_index = {pair: i + 1 for i, pair in enumerate(pairs)}
    n_pairs = len(pairs)
    
    if n_pairs != k:
        print("NO")
        return
    
    # Check consistency: must have exactly k distinct p and k distinct a
    if len(p_to_a) != k:
        print("NO")
        return
    distinct_a = set(p_to_a.values())
    if len(distinct_a) != k:
        print("NO")
        return
    
    # Build graph: left side = pair index, right side = m value
    graph = [[] for _ in range(n_pairs + 1)]
    for pair, m_list in pair_to_m.items():
        idx = pair_index[pair]
        for m in m_list:
            graph[idx].append(m)
    
    matching_count, pair_u, _ = hopcroft_karp(graph, n_pairs, k)
    
    if matching_count != k:
        print("NO")
        return
    
    print("YES")
    for i in range(1, n_pairs + 1):
        p, a = pairs[i - 1]
        m = pair_u[i]
        print(p, a, m)

if __name__ == "__main__":
    main()
```
