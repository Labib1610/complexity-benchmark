from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    blueprints = []
    ed_to_blueprints = defaultdict(list)
    index = 2
    for _ in range(t):
        e = int(data[index])
        d = int(data[index+1])
        i = int(data[index+2])
        blueprints.append((e, d, i))
        ed_to_blueprints[(e,d)].append(i)
        index += 3
    
    # Since each extraction site is paired with exactly one distillation column,
    # we can map each extraction site to its column
    e_to_d = {}
    for (e,d) in ed_to_blueprints:
        e_to_d[e] = d
    
    # Create a bipartite graph between extraction sites and injection wells
    # An edge exists if there's a blueprint with that (e,d) and well i
    graph = [[] for _ in range(k+1)]
    for (e,d), wells in ed_to_blueprints.items():
        for i in wells:
            graph[e].append(i)
    
    # Use maximum bipartite matching to find if we can match all k sites to k wells
    match_well = [-1] * (k+1)  # which extraction site is matched to each well
    match_site = [-1] * (k+1)  # which well is matched to each extraction site
    
    def dfs(e, visited):
        for i in graph[e]:
            if visited[i]:
                continue
            visited[i] = True
            
            # If well i is unmatched, or we can find an augmenting path
            if match_well[i] == -1 or dfs(match_well[i], visited):
                match_well[i] = e
                match_site[e] = i
                return True
        return False
    
    matching = 0
    for e in range(1, k+1):
        visited = [False] * (k+1)
        if dfs(e, visited):
            matching += 1
    
    if matching < k:
        print("NO")
        return
    
    print("YES")
    for e in range(1, k+1):
        i = match_site[e]
        d = e_to_d[e]
        print(e, d, i)

solve()
