We can model this as a bipartite matching problem.  
Each container is already paired with a fixed RFID tag, so we can reduce the problem to matching containers to vessels.

We have:

- Containers: 1..k  
- Vessels: 1..k  
- For each container, we know which vessels are allowed (from the cleared entries).

We need to assign each container a distinct vessel, i.e., find a perfect matching in a bipartite graph.

We can use **Hopcroft–Karp** for efficiency (k up to 1200, edges up to k²).

```python
import sys
from collections import deque

def hopcroft_karp(n, adj):
    # adj[u] = list of v (0-indexed)
    pair_u = [-1] * n
    pair_v = [-1] * n
    dist = [0] * n

    def bfs():
        q = deque()
        for u in range(n):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = float('inf')
        found = False
        while q:
            u = q.popleft()
            for v in adj[u]:
                pu = pair_v[v]
                if pu == -1:
                    found = True
                elif dist[pu] == float('inf'):
                    dist[pu] = dist[u] + 1
                    q.append(pu)
        return found

    def dfs(u):
        for v in adj[u]:
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = float('inf')
        return False

    matching = 0
    while bfs():
        for u in range(n):
            if pair_u[u] == -1 and dfs(u):
                matching += 1
    return matching, pair_u

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    # container -> tag mapping
    container_tag = [0] * (k + 1)
    # container -> set of allowed vessels
    allowed = [[] for _ in range(k + 1)]

    for _ in range(t):
        u = int(next(it))
        v = int(next(it))
        w = int(next(it))
        container_tag[u] = v
        allowed[u].append(w)

    # Build adjacency for matching: containers 0..k-1, vessels 0..k-1
    adj = [[] for _ in range(k)]
    for u in range(1, k + 1):
        for w in allowed[u]:
            adj[u - 1].append(w - 1)

    matching, pair_u = hopcroft_karp(k, adj)

    if matching != k:
        print("NO")
    else:
        print("YES")
        for u in range(k):
            v = pair_u[u] + 1
            tag = container_tag[u + 1]
            print(f"{u + 1} {tag} {v}")

if __name__ == "__main__":
    main()
```
