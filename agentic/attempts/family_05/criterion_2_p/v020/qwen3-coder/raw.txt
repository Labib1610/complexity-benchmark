from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    # Read entries and group by container (u)
    entries = []
    container_entries = defaultdict(list)
    
    for _ in range(t):
        u, v, w = map(int, input().split())
        entries.append((u, v, w))
        container_entries[u].append((u, v, w))
    
    # Since each container has exactly one RFID tag, and no two containers share the same tag,
    # we can think of this as a bipartite matching problem:
    # Match containers to cargo vessels, where an edge exists if there's a cleared entry
    # with that container and that cargo vessel.
    
    # For each container, we have a list of possible (container, tag, vessel) entries
    # Since the tag is fixed for each container, we just need to choose a vessel
    
    # Build a graph for matching: container -> list of possible vessels
    # Also keep track of the full entry for each container-vessel pair
    vessel_options = defaultdict(list)  # vessel -> list of containers that can use it
    entry_map = {}  # (container, vessel) -> full entry (u, v, w)
    
    for u in range(1, k+1):
        if u not in container_entries:
            print("NO")
            return
            
        for u, v, w in container_entries[u]:
            vessel_options[w].append(u)
            entry_map[(u, w)] = (u, v, w)
    
    # Now we need to match each container to a distinct vessel
    # This is a bipartite matching problem: containers on one side, vessels on the other
    
    # Use Hungarian algorithm or a simpler approach like DFS matching
    # Let's implement a simple DFS-based maximum bipartite matching
    
    match_vessel_to_container = {}  # vessel -> container
    match_container_to_vessel = {}  # container -> vessel
    
    def dfs(container, visited_vessels):
        # For this container, try all its possible vessels
        for u, v, w in container_entries[container]:
            if w not in visited_vessels:
                visited_vessels.add(w)
                # If vessel w is unmatched, or we can find an augmenting path
                if w not in match_vessel_to_container or dfs(match_vessel_to_container[w], visited_vessels):
                    match_vessel_to_container[w] = container
                    match_container_to_vessel[container] = w
                    return True
        return False
    
    # Try to match each container
    matched = 0
    for container in range(1, k+1):
        if container not in container_entries:
            print("NO")
            return
        visited_vessels = set()
        if dfs(container, visited_vessels):
            matched += 1
    
    if matched == k:
        print("YES")
        for container in range(1, k+1):
            vessel = match_container_to_vessel[container]
            # Find the tag for this container
            u, v, w = container_entries[container][0]
            # Actually, we need to find the right entry with this vessel
            for entry in container_entries[container]:
                if entry[2] == vessel:
                    print(entry[0], entry[1], entry[2])
                    break
    else:
        print("NO")

solve()
