from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    # If k is 0, we need 0 crews
    if k == 0:
        print("YES")
        return
    
    idx = 2
    # We'll store for each specialist, which machine they use
    spec_to_machine = {}
    # For each specialist, we store the list of available time slots
    spec_to_times = defaultdict(list)
    # Also store all configurations for reconstruction
    configs = []
    
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        idx += 3
        configs.append((x, y, z))
        
        # Record that specialist x uses machine y
        if x not in spec_to_machine:
            spec_to_machine[x] = y
        else:
            # Check consistency: same specialist can't use different machines
            if spec_to_machine[x] != y:
                # This shouldn't happen according to problem statement
                pass
        
        # Record that specialist x can work at time z
        spec_to_times[x].append(z)
    
    # Now we have a mapping from specialists to machines
    # And for each specialist, a list of time slots they can work in
    
    # Let's build a bipartite graph: specialists <-> time slots
    # An edge exists if the specialist has a configuration with that time slot
    
    # But actually, since each specialist is tied to one machine,
    # we just need to match specialists to time slots
    
    # So this becomes a matching problem: match each specialist to a unique time slot
    
    # Build adjacency list for specialists to their available time slots
    # Then run maximum bipartite matching
    
    # Create a graph for matching
    # Left side: specialists (1 to k)
    # Right side: time slots (1 to k)
    
    # But we need to make sure we're only considering specialists that exist
    # and that have at least one time slot
    
    # Actually, let's think differently:
    # Each specialist must be matched to exactly one time slot
    # Each time slot must be matched to exactly one specialist
    # Edges exist if there's a configuration with that spec and time
    
    # This is a perfect matching problem
    
    # First check if we have exactly k specialists
    if len(spec_to_machine) != k:
        print("NO")
        return
    
    # For each specialist, we know their machine and list of times
    # We want to assign each specialist to one time, such that all times 1..k are used once
    
    # Build adjacency list for matching
    adj = defaultdict(list)
    for spec in range(1, k+1):
        if spec in spec_to_times:
            adj[spec] = spec_to_times[spec]
        else:
            # Specialist exists but has no available time slots
            print("NO")
            return
    
    # Now we do maximum bipartite matching
    # Using Hungarian algorithm / DFS approach
    
    def dfs(u, match_right, match_left, visited):
        for v in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            if match_right[v] == -1 or dfs(match_right[v], match_right, match_left, visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    # match_right[v] = u means time slot v is matched to specialist u
    # match_left[u] = v means specialist u is matched to time slot v
    match_right = [-1] * (k + 1)  # time slots 1..k
    match_left = [-1] * (k + 1)   # specialists 1..k
    
    matched = 0
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(u, match_right, match_left, visited):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    # Reconstruct the solution
    print("YES")
    result = []
    for u in range(1, k + 1):
        v = match_left[u]  # time slot assigned to specialist u
        # Find the machine for specialist u
        m = spec_to_machine[u]
        result.append(f"{u} {m} {v}")
    
    print("\n".join(result))

solve()
