We can model this as a matching problem.  
Since each specialist is tied to exactly one machine)Skip that, we can reduce the problem to matching specialists to time slots using the given configurations.

Here’s the code:

```python
import sys
from collections import defaultdict, deque

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # specialist -> machine mapping (given by protocol)
    spec_machine = {}
    # For each specialist, list of time slots available
    spec_times = defaultdict(list)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        if x not in spec_machine:
            spec_machine[x] = y
        # assume protocol holds, so y is same for same x
        spec_times[x].append(z)
    
    # We need to match each specialist to a distinct time slot
    # Build bipartite graph: specialists (1..k) -> time slots (1..k)
    # Use Hopcroft-Karp or simple DFS matching
    
    match_slot = [-1] * (k + 1)  # slot -> specialist
    match_spec = [-1] * (k + 1)  # specialist -> slot
    
    def bfs():
        queue = deque()
        for u in range(1, k+1):
            if match_spec[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in spec_times[u]:
                    if dist[match_slot[v]] == float('inf'):
                        dist[match_slot[v]] = dist[u] + 1
                        queue.append(match_slot[v])
        return dist[0] != float('inf')
    
    def dfs(u):
        if u != 0:
            for v in spec_times[u]:
                if dist[match_slot[v]] == dist[u] + 1 and dfs(match_slot[v]):
                    match_slot[v] = u
                    match_spec[u] = v
                    return True
            dist[u] = float('inf')
            return False
        return True
    
    dist = [0] * (k + 1)
    matching = 0
    while bfs():
        for u in range(1, k+1):
            if match_spec[u] == -1 and dfs(u):
                matching += 1
    
    if matching != k:
        print("NO")
        return
    
    print("YES")
    for spec in range(1, k+1):
        slot = match_spec[spec]
        machine = spec_machine[spec]
        print(spec, machine, slot)

if __name__ == "__main__":
    solve()
```
