```python
import sys
from array import array

def numbers(data):
    value = 0
    reading = False
    for c in data:
        if 48 <= c <= 57:
            value = value * 10 + c - 48
            reading = True
        elif reading:
            yield value
            value = 0
            reading = False
    if reading:
        yield value

data = sys.stdin.buffer.read()
it = iter(numbers(data))

try:
    k = next(it)
    t = next(it)
except StopIteration:
    sys.exit()

head = array('i', [-1]) * (k + 1)
to = array('H')
nxt = array('i')
machine = [0] * (k + 1)
machine_owner = [0] * (k + 1)
valid = True

for _ in range(t):
    x = next(it)
    y = next(it)
    z = next(it)

    if machine[x] == 0:
        machine[x] = y
    elif machine[x] != y:
        valid = False

    if machine_owner[y] == 0:
        machine_owner[y] = x
    elif machine_owner[y] != x:
        valid = False

    to.append(z)
    nxt.append(head[x])
    head[x] = len(to) - 1

if not valid:
    print("NO")
    sys.exit()

match_left = [0] * (k + 1)
match_right = [0] * (k + 1)
INF = k + 1
matched = 0

sys.setrecursionlimit(10000)

while True:
    dist = [INF] * (k + 1)
    queue = [0] * k
    qh = 0
    qt = 0

    for u in range(1, k + 1):
        if match_left[u] == 0:
            dist[u] = 0
            queue[qt] = u
            qt += 1

    shortest = INF

    while qh < qt:
        u = queue[qh]
        qh += 1

        if dist[u] >= shortest:
            continue

        edge = head[u]
        while edge != -1:
            v = to[edge]
            paired = match_right[v]
            if paired == 0:
                shortest = dist[u] + 1
            elif dist[paired] == INF:
                dist[paired] = dist[u] + 1
                queue[qt] = paired
                qt += 1
            edge = nxt[edge]

    if shortest == INF:
        break

    current = list(head)

    def dfs(u):
        edge = current[u]
        while edge != -1:
            current[u] = nxt[edge]
            v = to[edge]
            paired = match_right[v]

            if paired == 0:
                if dist[u] + 1 == shortest:
                    match_left[u] = v
                    match_right[v] = u
                    return True
            elif dist[paired] == dist[u] + 1 and dist[paired] < shortest:
                if dfs(paired):
                    match_left[u] = v
                    match_right[v] = u
                    return True

            edge = current[u]

        dist[u] = INF
        return False

    augmentations = 0
    for u in range(1, k + 1):
        if match_left[u] == 0 and dfs(u):
            augmentations += 1

    if augmentations == 0:
        break
    matched += augmentations

if matched != k:
    print("NO")
else:
    out = ["YES"]
    out.extend(
        f"{x} {machine[x]} {match_left[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(out))
```
