from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
        
    # Since each pilot is certified for exactly one aircraft and no two pilots share an aircraft,
    # we can think of this as a bipartite matching problem between pilots and flight paths.
    # For each pilot x, we know which aircraft y they're certified for.
    # So we just need to match each pilot with a flight path.
    
    # Build adjacency list: for each pilot, which flight paths are available
    adj = defaultdict(list)  # pilot -> list of flight paths
    pilot_aircraft = {}      # pilot -> aircraft
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        adj[x].append(z)
        pilot_aircraft[x] = y
    
    # Check if we have exactly k pilots
    if len(adj) < k:
        print("NO")
        return
    
    # Now we need to find a perfect matching in bipartite graph:
    # Left side: pilots (1 to k)
    # Right side: flight paths (1 to k)
    # Edges: pilot x can use flight path z if (x, pilot_aircraft[x], z) is in input
    
    # Use Hungarian algorithm / Ford-Fulkerson for maximum bipartite matching
    def bfs_match():
        # BFS to find augmenting paths
        parent = {}
        visited = set()
        queue = deque()
        
        # Find unmatched pilots
        unmatched_pilots = []
        for pilot in range(1, k+1):
            if pilot not in match_pilot_to_path:
                unmatched_pilots.append(pilot)
                
        # Initialize BFS
        for pilot in unmatched_pilots:
            queue.append(('pilot', pilot))
            parent[('pilot', pilot)] = None
            visited.add(('pilot', pilot))
            
        while queue:
            node_type, node_id = queue.popleft()
            
            if node_type == 'pilot':
                pilot = node_id
                # Go to all adjacent flight paths that are not matched to this pilot
                for path in adj[pilot]:
                    if ('path', path) not in visited:
                        visited.add(('path', path))
                        parent[('path', path)] = ('pilot', pilot)
                        
                        # If path is unmatched, we found an augmenting path
                        if path not in match_path_to_pilot:
                            # Reconstruct path and update matching
                            current = ('path', path)
                            path_nodes = []
                            while current is not None:
                                path_nodes.append(current)
                                current = parent[current]
                            
                            # path_nodes has form [('path', p1), ('pilot', pilot1), ..., ('path', unmatched)]
                            # Process in reverse order
                            for i in range(len(path_nodes)-1, -1, -1):
                                if path_nodes[i][0] == 'path':
                                    path_node = path_nodes[i][1]
                                    pilot_node = path_nodes[i-1][1]
                                    match_pilot_to_path[pilot_node] = path_node
                                    match_path_to_pilot[path_node] = pilot_node
                            return True
                        else:
                            # Path is matched to some other pilot, continue BFS
                            matched_pilot = match_path_to_pilot[path]
                            queue.append(('pilot', matched_pilot))
                            parent[('pilot', matched_pilot)] = ('path', path)
                            visited.add(('pilot', matched_pilot))
            else:  # node_type == 'path'
                pass  # We only go from pilot to path in this implementation
                
        return False
    
    # Initialize matching
    match_pilot_to_path = {}
    match_path_to_pilot = {}
    
    # Keep finding augmenting paths until no more exist
    while bfs_match():
        pass
    
    # Check if we have a perfect matching
    if len(match_pilot_to_path) != k:
        print("NO")
        return
    
    # Output the solution
    print("YES")
    for pilot in range(1, k+1):
        path = match_pilot_to_path[pilot]
        aircraft = pilot_aircraft[pilot]
        print(pilot, aircraft, path)

def solve_efficient():
    k, t = map(int, input().split())
    
    # Build mapping from pilot to their certified aircraft and available paths
    pilot_to_aircraft = {}
    pilot_to_paths = defaultdict(list)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        pilot_to_aircraft[x] = y
        pilot_to_paths[x].append(z)
    
    # Check if we have all pilots
    if len(pilot_to_aircraft) != k:
        print("NO")
        return
    
    # Create bipartite graph for matching
    # Left: pilots 1..k
    # Right: paths 1..k
    # Edge exists if pilot can use that path
    
    # Use a simple greedy + backtracking approach or implement proper matching
    # Let's use a standard algorithm for maximum bipartite matching
    
    def dfs(u, match_right, match_left, visited):
        for v in pilot_to_paths[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            if match_right[v] == -1 or dfs(match_right[v], match_right, match_left, visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    # match_right[v] = u means path v is matched to pilot u
    # match_left[u] = v means pilot u is matched to path v
    match_right = [-1] * (k + 1)
    match_left = [-1] * (k + 1)
    
    result = 0
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(u, match_right, match_left, visited):
            result += 1
    
    if result != k:
        print("NO")
        return
    
    print("YES")
    for u in range(1, k + 1):
        v = match_left[u]
        print(u, pilot_to_aircraft[u], v)

solve_efficient()
