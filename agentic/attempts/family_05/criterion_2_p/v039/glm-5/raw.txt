import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # pilot -> aircraft mapping
    pilot_to_aircraft = {}
    # pilot -> list of flight paths
    pilot_paths = {}
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        pilot_to_aircraft[x] = y
        pilot_paths.setdefault(x, []).append(z)
    
    # Check all pilots present
    if len(pilot_to_aircraft) != k:
        print("NO")
        return
    
    # Hopcroft-Karp for maximum bipartite matching
    # Left: pilots 1..k, Right: flight paths 1..k
    INF = float('inf')
    pair_u = [0] * (k + 1)  # pair_u[pilot] = flight path
    pair_v = [0] * (k + 1)  # pair_v[path] = pilot
    dist = [0] * (k + 1)
    
    adj = [[] for _ in range(k + 1)]
    for p in range(1, k + 1):
        if p in pilot_paths:
            adj[p] = pilot_paths[p]
    
    from collections import deque
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching != k:
        print("NO")
        return
    
    print("YES")
    out = []
    for u in range(1, k + 1):
        out.append(f"{u} {pilot_to_aircraft[u]} {pair_u[u]}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
