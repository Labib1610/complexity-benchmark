```python
import sys
from collections import deque

def main():
    input_stream = sys.stdin.buffer
    first = input_stream.readline().split()
    if not first:
        return

    k, t = map(int, first)
    if t < k:
        print("NO")
        return

    adjacency = [[] for _ in range(k)]
    aircraft = [-1] * k
    pilot_match = [-1] * k
    path_match = [-1] * k
    matching = 0

    for _ in range(t):
        x, y, z = map(int, input_stream.readline().split())
        x -= 1
        y -= 1
        z -= 1
        adjacency[x].append(z)
        aircraft[x] = y

        if pilot_match[x] == -1 and path_match[z] == -1:
            pilot_match[x] = z
            path_match[z] = x
            matching += 1

    if matching < k:
        sys.setrecursionlimit(5000)
        inf = k + 1
        distance = [inf] * k

        while True:
            queue = deque()
            for u in range(k):
                if pilot_match[u] == -1:
                    distance[u] = 0
                    queue.append(u)
                else:
                    distance[u] = inf

            shortest = inf

            while queue:
                u = queue.popleft()
                if distance[u] >= shortest:
                    continue
                next_distance = distance[u] + 1
                for v in adjacency[u]:
                    paired_u = path_match[v]
                    if paired_u == -1:
                        shortest = next_distance
                    elif distance[paired_u] == inf:
                        distance[paired_u] = next_distance
                        queue.append(paired_u)

            if shortest == inf:
                break

            position = [0] * k

            def augment(u):
                neighbors = adjacency[u]
                while position[u] < len(neighbors):
                    v = neighbors[position[u]]
                    position[u] += 1
                    paired_u = path_match[v]

                    if paired_u == -1:
                        if distance[u] + 1 != shortest:
                            continue
                    elif distance[paired_u] != distance[u] + 1 or not augment(paired_u):
                        continue

                    pilot_match[u] = v
                    path_match[v] = u
                    return True

                distance[u] = inf
                return False

            added = 0
            for u in range(k):
                if pilot_match[u] == -1 and augment(u):
                    added += 1

            if added == 0:
                break
            matching += added

    if matching != k:
        print("NO")
        return

    output = ["YES"]
    for x in range(k):
        output.append(f"{x + 1} {aircraft[x] + 1} {pilot_match[x] + 1}")
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
