We can model this as a matching problem.  
Each specialist is tied to a unique machine, so we can reduce the problem to matching specialists to time slots, where each specialist has a set of allowed time slots (from the given combinations).  

We need to assign each specialist a distinct time slot — this is a bipartite matching problem between specialists and time slots.  

We can solve it using a standard DFS-based bipartite matching (Kuhn’s algorithm), which is efficient for up to 1200 nodes.

```python
import sys
sys.setrecursionlimit(10000)

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # For each specialist, store allowed time slots
    allowed = [[] for _ in range(k + 1)]
    machine_of = [0] * (k + 1)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        allowed[x].append(z)
        machine_of[x] = y
    
    # Bipartite matching: specialists -> time slots
    match_slot = [0] * (k + 1)  # which specialist is assigned to each slot
    
    def dfs(spec, visited):
        for slot in allowed[spec]:
            if not visited[slot]:
                visited[slot] = True
                if match_slot[slot] == 0 or dfs(match_slot[slot], visited):
                    match_slot[slot] = spec
                    return True
        return False
    
    for spec in range(1, k + 1):
        visited = [False] * (k + 1)
        if not dfs(spec, visited):
            print("NO")
            return
    
    print("YES")
    for slot in range(1, k + 1):
        spec = match_slot[slot]
        print(spec, machine_of[spec], slot)

if __name__ == "__main__":
    solve()
```
