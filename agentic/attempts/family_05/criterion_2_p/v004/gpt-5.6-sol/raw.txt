```python
import sys
from collections import deque

def main():
    input_stream = sys.stdin.buffer
    first = input_stream.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k + 1)]
    machine = [0] * (k + 1)

    for _ in range(t):
        x, y, z = map(int, input_stream.readline().split())
        adjacency[x].append(z)
        if machine[x] == 0:
            machine[x] = y

    if any(machine[x] == 0 for x in range(1, k + 1)):
        print("NO")
        return

    match_specialist = [0] * (k + 1)
    match_slot = [0] * (k + 1)
    distance = [0] * (k + 1)
    infinity = k + 1

    sys.setrecursionlimit(10000)

    def bfs():
        queue = deque()
        for x in range(1, k + 1):
            if match_specialist[x] == 0:
                distance[x] = 0
                queue.append(x)
            else:
                distance[x] = infinity

        distance[0] = infinity

        while queue:
            x = queue.popleft()
            if distance[x] >= distance[0]:
                continue

            next_distance = distance[x] + 1
            for z in adjacency[x]:
                other = match_slot[z]
                if distance[other] == infinity:
                    distance[other] = next_distance
                    if other != 0:
                        queue.append(other)

        return distance[0] != infinity

    def dfs(x):
        if x == 0:
            return True

        next_distance = distance[x] + 1
        for z in adjacency[x]:
            other = match_slot[z]
            if distance[other] == next_distance and dfs(other):
                match_specialist[x] = z
                match_slot[z] = x
                return True

        distance[x] = infinity
        return False

    matched = 0
    while bfs():
        for x in range(1, k + 1):
            if match_specialist[x] == 0 and dfs(x):
                matched += 1

    if matched != k:
        print("NO")
        return

    output = ["YES"]
    output.extend(
        f"{x} {machine[x]} {match_specialist[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
