```python
import sys
from collections import deque

sys.setrecursionlimit(10000)

buffer = sys.stdin.buffer
k, t = map(int, buffer.readline().split())

adj = [[] for _ in range(k + 1)]
archive = [0] * (k + 1)

match_scholar = [0] * (k + 1)
match_manuscript = [0] * (k + 1)
matched = 0

for _ in range(t):
    x, y, z = map(int, buffer.readline().split())
    adj[x].append(z)
    archive[x] = y

    if match_scholar[x] == 0 and match_manuscript[z] == 0:
        match_scholar[x] = z
        match_manuscript[z] = x
        matched += 1

INF = k + 1
dist = [INF] * (k + 1)

while matched < k:
    queue = deque()
    for u in range(1, k + 1):
        if match_scholar[u] == 0:
            dist[u] = 0
            queue.append(u)
        else:
            dist[u] = INF

    shortest = INF

    while queue:
        u = queue.popleft()
        next_dist = dist[u] + 1
        if next_dist > shortest:
            continue

        for v in adj[u]:
            paired = match_manuscript[v]
            if paired == 0:
                shortest = next_dist
            elif next_dist < shortest and dist[paired] == INF:
                dist[paired] = next_dist
                queue.append(paired)

    if shortest == INF:
        break

    def dfs(u):
        next_dist = dist[u] + 1

        for v in adj[u]:
            paired = match_manuscript[v]

            if paired == 0:
                if next_dist != shortest:
                    continue
            elif dist[paired] != next_dist or not dfs(paired):
                continue

            match_scholar[u] = v
            match_manuscript[v] = u
            return True

        dist[u] = INF
        return False

    augmented = 0
    for u in range(1, k + 1):
        if match_scholar[u] == 0 and dfs(u):
            augmented += 1

    if augmented == 0:
        break
    matched += augmented

if matched != k:
    print("NO")
else:
    output = ["YES"]
    output.extend(
        f"{u} {archive[u]} {match_scholar[u]}"
        for u in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))
```
