import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    # operator -> rig
    op_rig = {}
    # operator -> set of zones
    op_zones = {}
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        op_rig[x] = y
        if x not in op_zones:
            op_zones[x] = []
        op_zones[x].append(z)
    
    # Check all operators present
    if len(op_rig) != k:
        print("NO")
        return
    
    # Build bipartite graph: operators 1..k on left, zones 1..k on right
    # We need a perfect matching
    # Use Hopcroft-Karp
    # Operators are 1-indexed, zones are 1-indexed
    adj = [[] for _ in range(k+1)]
    for x in range(1, k+1):
        if x not in op_zones:
            print("NO")
            return
        # deduplicate zones
        zones = list(set(op_zones[x]))
        adj[x] = zones
    
    # Hopcroft-Karp
    INF = 10**9
    pair_u = [0]*(k+1)  # operator -> zone
    pair_v = [0]*(k+1)  # zone -> operator
    dist = [0]*(k+1)
    from collections import deque
    
    def bfs():
        queue = deque()
        found = False
        for u in range(1, k+1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    pu = pair_v[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        queue.append(pu)
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for v in adj[u]:
                pu = pair_v[v]
                if dist[pu] == dist[u] + 1:
                    if dfs(pu):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k+1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching != k:
        print("NO")
        return
    
    print("YES")
    out_lines = []
    for x in range(1, k+1):
        y = op_rig[x]
        z = pair_u[x]
        out_lines.append(f"{x} {y} {z}")
    sys.stdout.write("\n".join(out_lines) + "\n")

solve()
