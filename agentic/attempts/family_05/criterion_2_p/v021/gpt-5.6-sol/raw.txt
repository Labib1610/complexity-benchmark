import sys

sys.setrecursionlimit(5000)

def main():
    read = sys.stdin.buffer.readline
    k, t = map(int, read().split())

    graph = [[] for _ in range(k + 1)]
    reagent = [0] * (k + 1)

    for _ in range(t):
        x, y, z = map(int, read().split())
        graph[x].append(z)
        reagent[x] = y

    if any(not graph[x] for x in range(1, k + 1)):
        print("NO")
        return

    pair_left = [0] * (k + 1)
    pair_right = [0] * (k + 1)
    dist = [0] * (k + 1)
    inf = k + 1
    shortest = inf

    def bfs():
        nonlocal shortest
        queue = [0] * k
        head = 0
        tail = 0

        for u in range(1, k + 1):
            if pair_left[u] == 0:
                dist[u] = 0
                queue[tail] = u
                tail += 1
            else:
                dist[u] = inf

        shortest = inf

        while head < tail:
            u = queue[head]
            head += 1

            if dist[u] + 1 > shortest:
                continue

            next_dist = dist[u] + 1
            for v in graph[u]:
                matched_u = pair_right[v]
                if matched_u == 0:
                    if next_dist < shortest:
                        shortest = next_dist
                elif dist[matched_u] == inf:
                    dist[matched_u] = next_dist
                    queue[tail] = matched_u
                    tail += 1

        return shortest != inf

    def dfs(u):
        next_dist = dist[u] + 1

        for v in graph[u]:
            matched_u = pair_right[v]

            if matched_u == 0:
                if next_dist != shortest:
                    continue
            elif dist[matched_u] != next_dist or not dfs(matched_u):
                continue

            pair_left[u] = v
            pair_right[v] = u
            return True

        dist[u] = inf
        return False

    matching = 0

    while bfs():
        for u in range(1, k + 1):
            if pair_left[u] == 0 and dfs(u):
                matching += 1

    if matching != k:
        print("NO")
        return

    output = ["YES"]
    output.extend(
        f"{x} {reagent[x]} {pair_left[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
