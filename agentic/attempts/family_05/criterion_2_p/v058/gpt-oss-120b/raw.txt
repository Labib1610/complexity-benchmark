```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    f = [0] * (k + 1)          # f[x] = y
    adj = [[] for _ in range(k + 1)]   # edges x -> z
    # store existence for output (x,z) -> y (but y is f[x])
    for _ in range(t):
        x = int(next(it)); y = int(next(it)); z = int(next(it))
        if f[x] == 0:
            f[x] = y
        # else y must equal f[x] by problem guarantee
        adj[x].append(z)

    # Hopcroft–Karp
    NIL = 0
    INF = k + 5
    pair_u = [NIL] * (k + 1)   # left side match (x -> z)
    pair_v = [NIL] * (k + 1)   # right side match (z -> x)
    dist = [0] * (k + 1)

    def bfs() -> bool:
        q = deque()
        for u in range(1, k + 1):
            if pair_u[u] == NIL:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[NIL] = INF
        while q:
            u = q.popleft()
            if dist[u] < dist[NIL]:
                for v in adj[u]:
                    if pair_v[v] == NIL:
                        if dist[NIL] == INF:
                            dist[NIL] = dist[u] + 1
                    elif dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return dist[NIL] != INF

    def dfs(u: int) -> bool:
        if u != NIL:
            for v in adj[u]:
                if pair_v[v] == NIL or (dist[pair_v[v]] == dist[u] + 1 and dfs(pair_v[v])):
                    pair_u[u] = v
                    pair_v[v] = u
                    return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == NIL and dfs(u):
                matching += 1

    if matching != k:
        print("NO")
        return

    print("YES")
    out_lines = []
    for x in range(1, k + 1):
        z = pair_u[x]
        y = f[x]
        out_lines.append(f"{x} {y} {z}")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```
