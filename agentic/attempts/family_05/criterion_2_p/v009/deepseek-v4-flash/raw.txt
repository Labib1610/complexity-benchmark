The above code has a flaw: we need to store the machine for each specialist during input reading. Let's fix that properly.

```python
import sys
from collections import deque

input = sys.stdin.readline

def main():
    k, t = map(int, input().split())
    allowed = [[] for _ in range(k + 1)]
    machine_of = [0] * (k + 1)
    for _ in range(t):
        x, y, z = map(int, input().split())
        allowed[x].append(z)
        machine_of[x] = y  # fixed machine for specialist x
    
    # Hopcroft-Karp
    match_z = [0] * (k + 1)
    match_x = [0] * (k + 1)
    dist = [0] * (k + 1)
    INF = 10**9
    
    def bfs():
        q = deque()
        for x in range(1, k + 1):
            if match_x[x] == 0:
                dist[x] = 0
                q.append(x)
            else:
                dist[x] = INF
        dist[0] = INF
        while q:
            x = q.popleft()
            if dist[x] < dist[0]:
                for z in allowed[x]:
                    if dist[match_z[z]] == INF:
                        dist[match_z[z]] = dist[x] + 1
                        q.append(match_z[z])
        return dist[0] != INF
    
    def dfs(x):
        if x != 0:
            for z in allowed[x]:
                if dist[match_z[z]] == dist[x] + 1 and dfs(match_z[z]):
                    match_z[z] = x
                    match_x[x] = z
                    return True
            dist[x] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for x in range(1, k + 1):
            if match_x[x] == 0 and dfs(x):
                matching += 1
    
    if matching != k:
        print("NO")
    else:
        print("YES")
        for x in range(1, k + 1):
            z = match_x[x]
            y = machine_of[x]
            print(x, y, z)

if __name__ == "__main__":
    main()
```
