```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # For each specialist x, store the list of allowed time slots z
    # The machine y is fixed for each specialist
    spec_to_time = [[] for _ in range(k + 1)]
    spec_to_machine = [0] * (k + 1)
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        spec_to_time[x].append(z)
        spec_to_machine[x] = y
    
    # Bipartite matching: specialists (1..k) to time slots (1..k)
    # match_z[z] = x if time slot z is matched to specialist x
    match_z = [0] * (k + 1)
    match_x = [0] * (k + 1)
    
    def bfs():
        dist = [-1] * (k + 1)
        q = deque()
        for x in range(1, k + 1):
            if match_x[x] == 0:
                dist[x] = 0
                q.append(x)
        found = False
        while q:
            u = q.popleft()
            for v in spec_to_time[u]:
                w = match_z[v]
                if w != 0 and dist[w] == -1:
                    dist[w] = dist[u] + 1
                    q.append(w)
                elif w == 0:
                    found = True
        return found, dist
    
    def dfs(u, dist):
        for v in spec_to_time[u]:
            w = match_z[v]
            if w == 0 or (dist[w] == dist[u] + 1 and dfs(w, dist)):
                match_z[v] = u
                match_x[u] = v
                return True
        return False
    
    # Hopcroft-Karp
    while True:
        found, dist = bfs()
        if not found:
            break
        for x in range(1, k + 1):
            if match_x[x] == 0:
                dfs(x, dist)
    
    # Check if all specialists are matched
    for x in range(1, k + 1):
        if match_x[x] == 0:
            print("NO")
            return
    
    print("YES")
    for x in range(1, k + 1):
        z = match_x[x]
        y = spec_to_machine[x]
        print(f"{x} {y} {z}")

if __name__ == "__main__":
    solve()
```
