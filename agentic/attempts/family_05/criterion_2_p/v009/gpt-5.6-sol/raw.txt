```python
import sys
from collections import deque

def main():
    input_data = sys.stdin.buffer
    first = input_data.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adj = [[] for _ in range(k + 1)]
    machine = [0] * (k + 1)

    pair_x = [0] * (k + 1)
    pair_z = [0] * (k + 1)
    matching = 0

    for _ in range(t):
        x, y, z = map(int, input_data.readline().split())
        adj[x].append(z)
        machine[x] = y

        if pair_x[x] == 0 and pair_z[z] == 0:
            pair_x[x] = z
            pair_z[z] = x
            matching += 1

    if any(not adj[x] for x in range(1, k + 1)):
        print("NO")
        return

    if matching < k:
        inf = k + 1
        dist = [inf] * (k + 1)
        sys.setrecursionlimit(5000)

        def bfs():
            queue = deque()
            for x in range(1, k + 1):
                if pair_x[x] == 0:
                    dist[x] = 0
                    queue.append(x)
                else:
                    dist[x] = inf
            dist[0] = inf

            while queue:
                x = queue.popleft()
                if dist[x] >= dist[0]:
                    continue
                next_dist = dist[x] + 1
                for z in adj[x]:
                    other_x = pair_z[z]
                    if dist[other_x] == inf:
                        dist[other_x] = next_dist
                        if other_x != 0:
                            queue.append(other_x)

            return dist[0] != inf

        def dfs(x):
            if x == 0:
                return True

            next_dist = dist[x] + 1
            for z in adj[x]:
                other_x = pair_z[z]
                if dist[other_x] == next_dist and dfs(other_x):
                    pair_x[x] = z
                    pair_z[z] = x
                    return True

            dist[x] = inf
            return False

        while matching < k and bfs():
            for x in range(1, k + 1):
                if pair_x[x] == 0 and dfs(x):
                    matching += 1

    if matching != k:
        print("NO")
        return

    output = ["YES"]
    output.extend(
        f"{x} {machine[x]} {pair_x[x]}"
        for x in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
