from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    if t < k:
        print("NO")
        return
    
    # Read the allocations
    idx = 2
    allocations = []
    spec_to_mach = {}  # specialist -> machinery
    mach_to_spec = {}  # machinery -> specialist
    spec_allocs = defaultdict(list)  # specialist -> list of allocations
    
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        idx += 3
        allocations.append((x, y, z))
        
        # Safety regulation check
        if x in spec_to_mach:
            if spec_to_mach[x] != y:
                # This violates the constraint
                print("NO")
                return
        else:
            spec_to_mach[x] = y
            
        if y in mach_to_spec:
            if mach_to_spec[y] != x:
                # This violates the constraint
                print("NO")
                return
        else:
            mach_to_spec[y] = x
            
        spec_allocs[x].append((x, y, z))
    
    # Check if we have exactly k specialists and k machinery
    if len(spec_to_mach) != k or len(mach_to_spec) != k:
        print("NO")
        return
        
    # Build a graph for matching: deployment windows -> specialists
    # For each deployment window, which specialists can be scheduled in it
    dep_to_allocs = defaultdict(list)
    for alloc in allocations:
        x, y, z = alloc
        dep_to_allocs[z].append(alloc)
    
    # Try to find a perfect matching using backtracking with pruning
    # We need to match each specialist to exactly one allocation
    # such that each deployment window is used exactly once
    
    # For efficiency, let's model this as a bipartite matching:
    # Left side: specialists (1..k)
    # Right side: deployment windows (1..k)
    # Edge exists if there's an allocation (spec, *, dep)
    
    # But since each specialist has exactly one machinery, and each machinery 
    # is assigned to exactly one specialist, we can simplify:
    # For each specialist, we have a fixed set of possible allocations
    
    # Let's build available allocations per specialist
    spec_options = {}
    for spec in range(1, k+1):
        if spec not in spec_allocs:
            print("NO")
            return
        spec_options[spec] = spec_allocs[spec]
    
    # Now we want to select one allocation per specialist such that
    # all deployment windows are distinct
    
    # This is a matching problem: match specialists to allocations
    # where allocations have distinct deployment windows
    
    # We can model this as finding a system of distinct representatives
    # or a maximum matching with constraints
    
    # Since k <= 1200, we need an efficient approach
    # Let's use a greedy approach with backtracking optimization
    
    # Map deployment window to list of allocations that use it
    dep_usage = defaultdict(list)
    for alloc in allocations:
        dep_usage[alloc[2]].append(alloc)
        
    # Check if every deployment window appears at least once
    # Actually, we don't need this - we just need to be able to select k allocations
    # with distinct deployment windows
    
    # Create mapping from deployment window to allocations
    dep_to_specs = defaultdict(list)  # dep -> [(spec, alloc), ...]
    for alloc in allocations:
        x, y, z = alloc
        dep_to_specs[z].append((x, alloc))
    
    # We want to assign to each specialist exactly one allocation
    # such that all deployment windows are distinct
    
    # This is equivalent to finding a matching in a tripartite graph
    # But we can think of it as: select one allocation per specialist
    # such that all deployment windows among these allocations are distinct
    
    # Hungarian algorithm or similar would work but might be overkill
    # Let's try a DFS/backtracking approach with bitmasks for small k
    # But k can be up to 1200, so we need something more efficient
    
    # Key insight: This is actually a maximum bipartite matching problem
    # Between specialists and deployment windows, with edge if there exists
    # an allocation for that specialist-deployment pair
    
    # Build the bipartite graph
    # Left: specialists (1..k)  
    # Right: deployment windows (1..k)
    # Edge (x,z) exists if there's an allocation (x,*,z)
    
    from collections import deque
    
    # Adjacency list for the bipartite graph
    adj = defaultdict(list)
    for alloc in allocations:
        x, y, z = alloc
        adj[x].append(z)
    
    # Maximum bipartite matching using Ford-Fulkerson with BFS (Edmonds-Karp)
    def bfs_match(pair_u, pair_v, dist):
        Q = deque()
        # Initialize distances
        for u in range(1, k+1):
            if pair_u[u] == 0:
                dist[u] = 0
                Q.append(u)
            else:
                dist[u] = float('inf')
        dist[0] = float('inf')
        
        while Q:
            u = Q.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if dist[pair_v[v]] == float('inf'):
                        dist[pair_v[v]] = dist[u] + 1
                        Q.append(pair_v[v])
        return dist[0] != float('inf')
    
    def dfs_match(u, pair_u, pair_v, dist):
        if u != 0:
            for v in adj[u]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs_match(pair_v[v], pair_u, pair_v, dist):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = float('inf')
            return False
        return True
    
    def hopcroft_karp():
        pair_u = [0] * (k + 1)  # pair_u[u] = v means u is matched to v
        pair_v = [0] * (k + 1)  # pair_v[v] = u means v is matched to u
        dist = [0] * (k + 1)
        
        matching = 0
        while bfs_match(pair_u, pair_v, dist):
            for u in range(1, k+1):
                if pair_u[u] == 0:
                    if dfs_match(u, pair_u, pair_v, dist):
                        matching += 1
        return matching, pair_u, pair_v
    
    # Run matching
    try:
        matching_size, pair_u, pair_v = hopcroft_karp()
    except:
        print("NO")
        return
        
    if matching_size != k:
        print("NO")
        return
    
    # Reconstruct the solution
    result = []
    # For each specialist u, find the allocation that matches it to pair_u[u]
    dep_used = {}  # deployment window -> allocation
    for spec in range(1, k+1):
        dep_win = pair_u[spec]
        if dep_win != 0:
            # Find the allocation for this specialist and deployment window
            found = False
            for alloc in spec_allocs[spec]:
                if alloc[2] == dep_win:
                    result.append(alloc)
                    dep_used[dep_win] = alloc
                    found = True
                    break
            if not found:
                print("NO")
                return
    
    if len(result) != k:
        print("NO")
        return
        
    print("YES")
    for alloc in result:
        print(alloc[0], alloc[1], alloc[2])

solve()
