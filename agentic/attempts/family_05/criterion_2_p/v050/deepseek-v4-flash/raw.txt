We can model this as a bipartite matching problem between specialists and deployment windows, since the machinery is fixed per specialist.

```python
import sys
from collections import defaultdict

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # Since each specialist has exactly one machinery, we can map specialist -> machinery
    spec_machine = {}
    # For each specialist, store the windows they can work in
    spec_windows = defaultdict(list)
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        spec_machine[x] = y
        spec_windows[x].append(z)
    
    # We need to assign each specialist a unique window
    # This is bipartite matching: specialists (1..k) to windows (1..k)
    
    # Build adjacency list for matching
    adj = [[] for _ in range(k+1)]
    for spec, windows in spec_windows.items():
        for w in windows:
            adj[spec].append(w)
    
    # Standard Kuhn's algorithm for maximum bipartite matching
    match_window = [0] * (k+1)  # which specialist is assigned to window w
    
    def dfs(spec, visited):
        for w in adj[spec]:
            if not visited[w]:
                visited[w] = True
                if match_window[w] == 0 or dfs(match_window[w], visited):
                    match_window[w] = spec
                    return True
        return False
    
    match_count = 0
    for spec in range(1, k+1):
        visited = [False] * (k+1)
        if dfs(spec, visited):
            match_count += 1
    
    if match_count != k:
        print("NO")
        return
    
    # Build output
    print("YES")
    for w in range(1, k+1):
        spec = match_window[w]
        print(spec, spec_machine[spec], w)

if __name__ == "__main__":
    solve()
```
