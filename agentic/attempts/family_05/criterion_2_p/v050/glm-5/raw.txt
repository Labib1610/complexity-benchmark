import sys
from collections import defaultdict

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # Group allocations by specialist x
    # Each specialist has a unique machinery y
    # We need to select exactly one allocation per specialist
    # such that all deployment windows z are distinct
    
    allocations_by_x = defaultdict(list)
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        allocations_by_x[x].append((x, y, z))
    
    # Check if all specialists 1..k have at least one allocation
    for x in range(1, k + 1):
        if x not in allocations_by_x:
            print("NO")
            return
    
    # Bipartite matching: specialists (left) to deployment windows (right)
    # Each specialist x can be matched to any z from their allocations
    
    # Build adjacency list: for each specialist x, list of possible z values
    adj = {}
    z_to_alloc = {}  # For each (x, z), store the allocation (x, y, z)
    
    for x in range(1, k + 1):
        z_list = []
        for alloc in allocations_by_x[x]:
            z = alloc[2]
            z_list.append(z)
            z_to_alloc[(x, z)] = alloc
        adj[x] = z_list
    
    # Hopcroft-Karp for maximum bipartite matching
    match_z = {}  # z -> x (which specialist is matched to this window)
    match_x = {}  # x -> z (which window is matched to this specialist)
    
    def bfs():
        dist = {}
        queue = []
        for x in range(1, k + 1):
            if x not in match_x:
                dist[x] = 0
                queue.append(x)
            else:
                dist[x] = float('inf')
        dist[0] = float('inf')
        
        idx = 0
        while idx < len(queue):
            u = queue[idx]
            idx += 1
            if dist[u] < dist[0]:
                for z in adj[u]:
                    v = match_z.get(z, 0)
                    if v not in dist or dist[v] == float('inf'):
                        dist[v] = dist[u] + 1
                        queue.append(v)
        return dist
    
    def dfs(u, dist):
        if u == 0:
            return True
        for z in adj[u]:
            v = match_z.get(z, 0)
            if v in dist and dist[v] == dist[u] + 1:
                if dfs(v, dist):
                    match_z[z] = u
                    match_x[u] = z
                    return True
        dist[u] = float('inf')
        return False
    
    matching_size = 0
    while True:
        dist = bfs()
        if dist.get(0, float('inf')) == float('inf'):
            break
        for x in range(1, k + 1):
            if x not in match_x:
                if dfs(x, dist):
                    matching_size += 1
    
    if matching_size != k:
        print("NO")
        return
    
    print("YES")
    for x in range(1, k + 1):
        z = match_x[x]
        alloc = z_to_alloc[(x, z)]
        print(f"{alloc[0]} {alloc[1]} {alloc[2]}")

if __name__ == "__main__":
    solve()
