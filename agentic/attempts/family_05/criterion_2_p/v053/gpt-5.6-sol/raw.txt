```python
import sys

def main():
    inp = sys.stdin.buffer
    first = inp.readline().split()
    if not first:
        return

    k, t = map(int, first)

    adj = [[] for _ in range(k)]
    tag = [-1] * k
    pair_u = [-1] * k
    pair_w = [-1] * k
    matched = 0

    for _ in range(t):
        u, v, w = map(int, inp.readline().split())
        u -= 1
        w -= 1
        tag[u] = v
        adj[u].append(w)

        if pair_u[u] == -1 and pair_w[w] == -1:
            pair_u[u] = w
            pair_w[w] = u
            matched += 1

    if matched < k and any(not adj[u] for u in range(k)):
        sys.stdout.write("NO\n")
        return

    sys.setrecursionlimit(100000)

    dist = [-1] * k
    ptr = [0] * k

    while matched < k:
        queue = [0] * k
        head = 0
        tail = 0

        for u in range(k):
            if pair_u[u] == -1:
                dist[u] = 0
                queue[tail] = u
                tail += 1
            else:
                dist[u] = -1

        shortest = k + 1

        while head < tail:
            u = queue[head]
            head += 1
            du = dist[u]

            if du >= shortest:
                continue

            for w in adj[u]:
                nxt = pair_w[w]
                if nxt == -1:
                    shortest = du + 1
                elif dist[nxt] == -1:
                    dist[nxt] = du + 1
                    queue[tail] = nxt
                    tail += 1

        if shortest == k + 1:
            break

        ptr[:] = [0] * k

        def dfs(u):
            du = dist[u]
            edges = adj[u]
            i = ptr[u]

            while i < len(edges):
                w = edges[i]
                i += 1
                ptr[u] = i
                nxt = pair_w[w]

                if nxt == -1:
                    if du + 1 == shortest:
                        pair_u[u] = w
                        pair_w[w] = u
                        return True
                elif du + 1 < shortest and dist[nxt] == du + 1 and dfs(nxt):
                    pair_u[u] = w
                    pair_w[w] = u
                    return True

            dist[u] = -1
            return False

        augmented = 0
        for u in range(k):
            if pair_u[u] == -1 and dfs(u):
                augmented += 1

        if augmented == 0:
            break
        matched += augmented

    if matched != k:
        sys.stdout.write("NO\n")
        return

    out = ["YES"]
    out.extend(
        f"{u + 1} {tag[u]} {pair_u[u] + 1}"
        for u in range(k)
    )
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```
