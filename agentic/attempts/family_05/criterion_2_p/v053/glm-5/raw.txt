Looking at this problem, I need to select exactly k entries where each container, RFID tag, and cargo vessel appears exactly once.

Key insight: Each container has exactly one RFID tag, and different containers have different tags. So containers and tags are paired 1-to-1. This means if I pick an entry for container u, the tag is determined. So I just need to pick one entry per container such that all k cargo vessels are covered exactly once.

This reduces to a bipartite matching problem: containers on one side, vessels on the other. For each container u, I have edges to all vessels w that appear in some entry (u, v, w). I need a perfect matching.

Since k ≤ 1200, I'll use Hopcroft-Karp algorithm for efficient bipartite matching.

```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    k = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    
    # For each container, list of (vessel, tag)
    container_vessels = [[] for _ in range(k + 1)]
    
    for _ in range(t):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        w = int(input_data[idx]); idx += 1
        container_vessels[u].append((w, v))
    
    # Check each container has at least one entry
    for u in range(1, k + 1):
        if not container_vessels[u]:
            print("NO")
            return
    
    # Hopcroft-Karp for bipartite matching: containers (1..k) to vessels (1..k)
    # adj[u] = list of vessels
    adj = [[] for _ in range(k + 1)]
    for u in range(1, k + 1):
        seen = set()
        for w, v in container_vessels[u]:
            if w not in seen:
                seen.add(w)
                adj[u].append(w)
    
    INF = float('inf')
    pair_u = [0] * (k + 1)  # pair_u[u] = matched vessel
    pair_v = [0] * (k + 1)  # pair_v[v] = matched container
    dist = [0] * (k + 1)
    
    def bfs():
        queue = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for w in adj[u]:
                    v = w
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF
    
    def dfs(u):
        if u != 0:
            for w in adj[u]:
                v = w
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_v[v] = u
                        pair_u[u] = v
                        return True
            dist[u] = INF
            return False
        return True
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching != k:
        print("NO")
        return
    
    # Build result: for each container, find the entry matching the assigned vessel
    print("YES")
    for u in range(1, k + 1):
        w = pair_u[u]
        # find tag v for this entry
        v = None
        for ww, vv in container_vessels[u]:
            if ww == w:
                v = vv
                break
        print(f"{u} {v} {w}")

solve()
```
