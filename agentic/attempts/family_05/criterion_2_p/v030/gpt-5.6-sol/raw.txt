import sys

def main():
    data = sys.stdin.buffer
    first = data.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adj = [[] for _ in range(k)]
    y_for_x = [-1] * k
    x_for_y = [-1] * k
    valid = True

    for _ in range(t):
        x, y, z = map(int, data.readline().split())
        x -= 1
        y -= 1
        z -= 1

        if y_for_x[x] == -1:
            y_for_x[x] = y
            if x_for_y[y] == -1:
                x_for_y[y] = x
            elif x_for_y[y] != x:
                valid = False
        elif y_for_x[x] != y:
            valid = False

        adj[x].append(z)

    if not valid or any(y == -1 for y in y_for_x):
        print("NO")
        return

    pair_x = [-1] * k
    pair_z = [-1] * k
    matched = 0

    # A greedy initial matching often avoids all augmenting-path phases.
    for x in sorted(range(k), key=lambda u: len(adj[u])):
        for z in adj[x]:
            if pair_z[z] == -1:
                pair_x[x] = z
                pair_z[z] = x
                matched += 1
                break

    sys.setrecursionlimit(5000)
    dist = [0] * k
    INF = 10**9

    while matched < k:
        queue = []
        for x in range(k):
            if pair_x[x] == -1:
                dist[x] = 0
                queue.append(x)
            else:
                dist[x] = INF

        shortest = INF
        head = 0
        while head < len(queue):
            x = queue[head]
            head += 1
            dx = dist[x]

            if dx >= shortest:
                continue

            for z in adj[x]:
                next_x = pair_z[z]
                if next_x == -1:
                    shortest = dx + 1
                elif dist[next_x] == INF:
                    dist[next_x] = dx + 1
                    queue.append(next_x)

        if shortest == INF:
            break

        def dfs(x):
            dx = dist[x]
            for z in adj[x]:
                next_x = pair_z[z]
                if next_x == -1:
                    if dx + 1 != shortest:
                        continue
                elif dist[next_x] != dx + 1 or not dfs(next_x):
                    continue

                pair_x[x] = z
                pair_z[z] = x
                return True

            dist[x] = INF
            return False

        augmented = 0
        for x in range(k):
            if pair_x[x] == -1 and dfs(x):
                augmented += 1

        if augmented == 0:
            break
        matched += augmented

    if matched != k:
        print("NO")
        return

    output = ["YES"]
    for x in range(k):
        output.append(f"{x + 1} {y_for_x[x] + 1} {pair_x[x] + 1}")
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
