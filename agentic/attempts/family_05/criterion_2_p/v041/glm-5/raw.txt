import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        print("NO")
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # captain -> jet
    cap_jet = {}
    # jet -> captain
    jet_cap = {}
    # (captain, destination) -> list of triplet indices
    cap_dest_trip = {}
    triplets = []
    
    for idx in range(t):
        c = int(next(it))
        j = int(next(it))
        d = int(next(it))
        triplets.append((c, j, d))
        cap_dest_trip.setdefault((c, d), []).append(idx)
        if c in cap_jet:
            if cap_jet[c] != j:
                print("NO")
                return
        else:
            cap_jet[c] = j
        if j in jet_cap:
            if jet_cap[j] != c:
                print("NO")
                return
        else:
            jet_cap[j] = c
    
    # Build bipartite graph: captain -> destinations
    # We need a perfect matching of size k.
    # Use Hopcroft-Karp.
    
    # Captains and destinations are 1..k
    # adj[c] = list of destinations available for captain c
    adj = [[] for _ in range(k + 1)]
    for (c, d), lst in cap_dest_trip.items():
        adj[c].append(d)
    
    # Remove duplicate destinations per captain (shouldn't be duplicates anyway)
    for c in range(1, k + 1):
        if adj[c]:
            # keep unique
            adj[c] = list(set(adj[c]))
    
    # Hopcroft-Karp
    INF = 10**9
    pairU = [0] * (k + 1)  # captain -> destination
    pairV = [0] * (k + 1)  # destination -> captain
    dist = [0] * (k + 1)
    
    from collections import deque
    
    def bfs():
        dq = deque()
        for u in range(1, k + 1):
            if pairU[u] == 0:
                dist[u] = 0
                dq.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while dq:
            u = dq.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    pu = pairV[v]
                    if dist[pu] == INF:
                        dist[pu] = dist[u] + 1
                        dq.append(pu)
        return dist[0] != INF
    
    def dfs(u):
        for v in adj[u]:
            pu = pairV[v]
            if dist[pu] == dist[u] + 1:
                if dfs(pu):
                    pairV[v] = u
                    pairU[u] = v
                    return True
        dist[u] = INF
        return False
    
    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pairU[u] == 0:
                if dfs(u):
                    matching += 1
    
    if matching != k:
        print("NO")
        return
    
    # We have a perfect matching: pairU[c] = d for each captain c
    # Now select one triplet for each (c, d)
    out_lines = ["YES"]
    for c in range(1, k + 1):
        d = pairU[c]
        lst = cap_dest_trip.get((c, d))
        if not lst:
            print("NO")
            return
        idx = lst[0]
        tc, tj, td = triplets[idx]
        out_lines.append(f"{tc} {tj} {td}")
    
    sys.stdout.write("\n".join(out_lines) + "\n")

solve()
