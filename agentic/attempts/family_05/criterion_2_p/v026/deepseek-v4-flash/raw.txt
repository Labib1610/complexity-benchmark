We can model this as a matching problem.  
Since each sensor is already locked to a unique processor, we can reduce each channel to a pair (sensor, antenna).  
We need to assign each sensor a distinct antenna, using only the allowed pairs.

This is a bipartite matching problem between sensors and antennas.

```python
import sys

def main():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    
    # Since sensor -> processor is fixed, we can map sensor to its processor
    sensor_to_processor = {}
    allowed = [[] for _ in range(k + 1)]  # antennas allowed for each sensor
    
    for _ in range(t):
        s, p, a = map(int, input().split())
        sensor_to_processor[s] = p
        allowed[s].append(a)
    
    # Bipartite matching: sensors (1..k) to antennas (1..k)
    match_antenna = [0] * (k + 1)  # which sensor is matched to antenna
    
    def dfs(sensor, visited):
        for antenna in allowed[sensor]:
            if not visited[antenna]:
                visited[antenna] = True
                if match_antenna[antenna] == 0 or dfs(match_antenna[antenna], visited):
                    match_antenna[antenna] = sensor
                    return True
        return False
    
    for sensor in range(1, k + 1):
        visited = [False] * (k + 1)
        if not dfs(sensor, visited):
            print("NO")
            return
    
    print("YES")
    for antenna in range(1, k + 1):
        sensor = match_antenna[antenna]
        print(sensor, sensor_to_processor[sensor], antenna)

if __name__ == "__main__":
    main()
```
