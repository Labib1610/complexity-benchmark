```python
import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # Read all channels
    channels = []
    for _ in range(t):
        s = int(next(it))
        p = int(next(it))
        a = int(next(it))
        channels.append((s, p, a))
    
    # Build bipartite graph: left side = sensors (1..k), right side = antennas (1..k)
    # Edge exists if there's a valid channel for sensor s using antenna a
    # We need to find a perfect matching
    
    # adj[s] = list of antennas a such that channel (s, locked_p[s], a) exists
    adj = [[] for _ in range(k + 1)]
    # channel_map[s][a] = (s, p, a) tuple
    channel_map = [[None] * (k + 1) for _ in range(k + 1)]
    
    for s, p, a in channels:
        adj[s].append(a)
        channel_map[s][a] = (s, p, a)
    
    # Hopcroft-Karp for maximum bipartite matching
    # match_left[s] = antenna matched to sensor s (0 if unmatched)
    # match_right[a] = sensor matched to antenna a (0 if unmatched)
    match_left = [0] * (k + 1)
    match_right = [0] * (k + 1)
    dist = [0] * (k + 1)
    NIL = 0
    
    def bfs():
        queue = deque()
        for s in range(1, k + 1):
            if match_left[s] == NIL:
                dist[s] = 0
                queue.append(s)
            else:
                dist[s] = float('inf')
        dist[NIL] = float('inf')
        
        while queue:
            s = queue.popleft()
            if dist[s] < dist[NIL]:
                for a in adj[s]:
                    s2 = match_right[a]
                    if dist[s2] == float('inf'):
                        dist[s2] = dist[s] + 1
                        queue.append(s2)
        return dist[NIL] != float('inf')
    
    def dfs(s):
        if s != NIL:
            for a in adj[s]:
                s2 = match_right[a]
                if dist[s2] == dist[s] + 1 and dfs(s2):
                    match_left[s] = a
                    match_right[a] = s
                    return True
            dist[s] = float('inf')
            return False
        return True
    
    matching = 0
    while bfs():
        for s in range(1, k + 1):
            if match_left[s] == NIL and dfs(s):
                matching += 1
    
    if matching != k:
        print("NO")
        return
    
    print("YES")
    for s in range(1, k + 1):
        a = match_left[s]
        s_val, p_val, a_val = channel_map[s][a]
        print(s_val, p_val, a_val)

solve()
```
