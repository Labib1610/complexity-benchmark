```python
import sys

def main():
    input_buffer = sys.stdin.buffer
    first = input_buffer.readline().split()
    if not first:
        return

    k, t = map(int, first)
    adjacency = [[] for _ in range(k + 1)]
    processor = [0] * (k + 1)

    for _ in range(t):
        s, p, a = map(int, input_buffer.readline().split())
        adjacency[s].append(a)
        processor[s] = p

    if t < k or any(not adjacency[s] for s in range(1, k + 1)):
        print("NO")
        return

    sys.setrecursionlimit(5000)

    match_sensor = [0] * (k + 1)
    match_antenna = [0] * (k + 1)
    distance = [0] * (k + 1)
    matched = 0
    inf = k + 1

    while True:
        queue = [0] * k
        head = 0
        tail = 0

        for s in range(1, k + 1):
            if match_sensor[s] == 0:
                distance[s] = 0
                queue[tail] = s
                tail += 1
            else:
                distance[s] = inf

        shortest = inf

        while head < tail:
            s = queue[head]
            head += 1
            next_distance = distance[s] + 1

            if next_distance > shortest:
                continue

            for a in adjacency[s]:
                other = match_antenna[a]
                if other == 0:
                    if next_distance < shortest:
                        shortest = next_distance
                elif next_distance < shortest and distance[other] == inf:
                    distance[other] = next_distance
                    queue[tail] = other
                    tail += 1

        if shortest == inf:
            break

        edge_index = [0] * (k + 1)

        def augment(s):
            edges = adjacency[s]
            while edge_index[s] < len(edges):
                a = edges[edge_index[s]]
                edge_index[s] += 1
                other = match_antenna[a]

                if other == 0:
                    if distance[s] + 1 == shortest:
                        match_sensor[s] = a
                        match_antenna[a] = s
                        return True
                elif distance[other] == distance[s] + 1 and augment(other):
                    match_sensor[s] = a
                    match_antenna[a] = s
                    return True

            distance[s] = inf
            return False

        added = 0
        for s in range(1, k + 1):
            if match_sensor[s] == 0 and augment(s):
                added += 1

        if added == 0:
            break
        matched += added

    if matched != k:
        print("NO")
        return

    output = ["YES"]
    output.extend(
        f"{s} {processor[s]} {match_sensor[s]}"
        for s in range(1, k + 1)
    )
    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
