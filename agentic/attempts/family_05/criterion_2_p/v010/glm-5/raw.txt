import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # For each specialist, store list of (time, config_index)
    specialist_times = [[] for _ in range(k + 1)]
    configs = []
    
    for idx in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        configs.append((x, y, z))
        specialist_times[x].append((z, idx))
    
    # Sort times for each specialist for greedy matching
    for x in range(1, k + 1):
        specialist_times[x].sort()
    
    # Greedy bipartite matching: specialists to time slots
    match_time = [-1] * (k + 1)  # match_time[time] = specialist
    match_spec = [-1] * (k + 1)  # match_spec[specialist] = time
    
    def dfs(u, visited):
        for z, _ in specialist_times[u]:
            if visited[z]:
                continue
            visited[z] = True
            if match_time[z] == -1 or dfs(match_time[z], visited):
                match_time[z] = u
                match_spec[u] = z
                return True
        return False
    
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if not dfs(u, visited):
            print("NO")
            return
    
    # Build result: for each specialist, find config with their matched time
    result = []
    for x in range(1, k + 1):
        z = match_spec[x]
        found = False
        for tz, idx in specialist_times[x]:
            if tz == z:
                result.append(configs[idx])
                found = True
                break
        if not found:
            print("NO")
            return
    
    print("YES")
    for cfg in result:
        print(f"{cfg[0]} {cfg[1]} {cfg[2]}")

if __name__ == "__main__":
    solve()
