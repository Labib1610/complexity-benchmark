from collections import defaultdict

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    index = 2
    configs = []
    spec_machine = {}  # specialist -> machine
    
    # Read configurations and build mapping
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        configs.append((x, y, z))
        if x not in spec_machine:
            spec_machine[x] = y
        index += 3
    
    # Build bipartite graph: specialists <-> time slots
    # Edge exists if there's a config with that spec and time slot
    adj = defaultdict(list)  # spec -> list of (time_slot, config_index)
    
    for i, (x, y, z) in enumerate(configs):
        adj[x].append((z, i))
    
    # Try to find maximum matching using Hungarian algorithm approach
    # But since it's simpler, we'll use a greedy approach with backtracking
    
    match_spec_to_slot = {}  # spec -> time_slot
    match_slot_to_spec = {}  # time_slot -> spec
    result_indices = []
    
    def dfs(spec, visited_specs):
        for time_slot, cfg_idx in adj[spec]:
            if time_slot in match_slot_to_spec:
                continue  # Time slot already taken
            
            match_spec_to_slot[spec] = time_slot
            match_slot_to_spec[time_slot] = spec
            result_indices.append(cfg_idx)
            
            return True
        
        return False
    
    # Greedy assignment
    specs_assigned = set()
    slots_assigned = set()
    assigned_configs = []
    
    changed = True
    while changed:
        changed = False
        for i, (x, y, z) in enumerate(configs):
            if x in specs_assigned or z in slots_assigned:
                continue
            specs_assigned.add(x)
            slots_assigned.add(z)
            assigned_configs.append(i)
            changed = True
            break
    
    if len(assigned_configs) == k:
        print("YES")
        for idx in assigned_configs:
            x, y, z = configs[idx]
            print(x, y, z)
        return
    
    # Use a more sophisticated approach - build a graph and find maximum matching
    from collections import deque
    
    # Rebuild with proper matching algorithm
    specs = list(range(1, k+1))
    
    # Create adjacency list for the bipartite graph
    graph = defaultdict(list)
    for i, (x, y, z) in enumerate(configs):
        graph[x].append((z, i))  # spec -> [(slot, config_index)]
    
    # Maximum bipartite matching using DFS
    match_spec = {}      # spec -> slot
    match_slot = {}      # slot -> spec
    match_config = {}    # spec -> config_index
    
    def bpm(spec, visited):
        for slot, cfg_idx in graph[spec]:
            if slot in visited:
                continue
            visited.add(slot)
            
            if slot not in match_slot or bpm(match_slot[slot], visited):
                match_slot[slot] = spec
                match_spec[spec] = slot
                match_config[spec] = cfg_idx
                return True
        return False
    
    # Try to match all specialists
    for spec in specs:
        visited = set()
        bpm(spec, visited)
    
    # Check if we matched all k specialists
    if len(match_spec) == k:
        print("YES")
        for spec in match_spec:
            slot = match_spec[spec]
            cfg_idx = match_config[spec]
            x, y, z = configs[cfg_idx]
            print(x, y, z)
    else:
        print("NO")

solve()
