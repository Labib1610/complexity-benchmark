We can model this as a matching problem between drivers and routes, since each driver is already tied to a fixed truck.  
We need to assign each driver a unique route, and each route must be used exactly once.

This becomes a bipartite matching problem:  
- Left side: drivers (1..k)  
- Right side: routes (1..k)  
- Edge exists if there is a dispatch plan (driver, truck, route) for that driver and route.

We can solve it with a standard DFS-based bipartite matching (Kuhn’s algorithm), which is fast enough for k ≤ 1200.

```python
import sys
sys.setrecursionlimit(10000000)

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    # For each driver, store the truck (same for all their plans)
    driver_truck = [0] * (k + 1)
    # adjacency: driver -> list of routes
    adj = [[] for _ in range(k + 1)]
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        driver_truck[x] = y
        adj[x].append(z)
    
    # Matching: route -> driver
    match_route = [0] * (k + 1)
    
    def dfs(driver, visited):
        for route in adj[driver]:
            if not visited[route]:
                visited[route] = True
                if match_route[route] == 0 or dfs(match_route[route], visited):
                    match_route[route] = driver
                    return True
        return False
    
    match_count = 0
    for driver in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(driver, visited):
            match_count += 1
    
    if match_count != k:
        print("NO")
        return
    
    print("YES")
    for route in range(1, k + 1):
        driver = match_route[route]
        truck = driver_truck[driver]
        print(driver, truck, route)

if __name__ == "__main__":
    solve()
```
