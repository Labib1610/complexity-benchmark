from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    # Since each driver is assigned to exactly one truck, and no two drivers share a truck,
    # we can think of this as a bipartite matching problem.
    # One side is drivers (which are tied to their trucks), the other side is routes.
    
    # For each driver, we store the truck they're assigned to and the routes they can take
    driver_truck = {}  # driver -> truck
    driver_routes = defaultdict(list)  # driver -> [routes]
    
    plans = []
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        plans.append((x, y, z))
        if x not in driver_truck:
            driver_truck[x] = y
        # Verify that the truck assignment is consistent
        assert driver_truck[x] == y
        driver_routes[x].append(z)
    
    # Build a bipartite graph: drivers on one side, routes on the other
    # Edges represent valid dispatch plans
    
    # Use a matching algorithm to find maximum matching
    # We'll use a simple implementation of Hopcroft-Karp or just Hungarian algorithm concept
    
    # For each driver, list of routes they can do
    # This is a bipartite matching problem: match drivers to routes
    
    # Build adjacency for drivers to routes
    adj = defaultdict(list)
    for x, y, z in plans:
        adj[x].append(z)
    
    # Try to find a maximum matching using DFS-based algorithm
    match_route = {}  # route -> driver
    match_driver = {}  # driver -> route
    
    def dfs(driver, visited):
        for route in adj[driver]:
            if route in visited:
                continue
            visited.add(route)
            
            # If route is unmatched or we can find augmenting path
            if route not in match_route or dfs(match_route[route], visited):
                match_route[route] = driver
                match_driver[driver] = route
                return True
        return False
    
    matching = 0
    for driver in range(1, k+1):
        if driver in adj:  # If driver has any plans
            if dfs(driver, set()):
                matching += 1
    
    if matching < k:
        print("NO")
        return
    
    print("YES")
    # Output the selected plans
    for driver in range(1, k+1):
        if driver in match_driver:
            route = match_driver[driver]
            truck = driver_truck[driver]
            print(driver, truck, route)

solve()
