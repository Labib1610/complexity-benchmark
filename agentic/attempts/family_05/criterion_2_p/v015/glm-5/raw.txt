import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    plans = []
    idx = 2
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        plans.append((x, y, z))
        idx += 3

    # driver -> truck
    driver_truck = {}
    for x, y, z in plans:
        if x in driver_truck:
            if driver_truck[x] != y:
                print("NO")
                return
        else:
            driver_truck[x] = y

    # truck -> driver
    truck_driver = {}
    for x, y in driver_truck.items():
        if y in truck_driver:
            if truck_driver[y] != x:
                print("NO")
                return
        else:
            truck_driver[y] = x

    # driver -> list of routes
    driver_routes = {}
    for x, y, z in plans:
        driver_routes.setdefault(x, []).append(z)

    # We need to select exactly one route per driver, and all routes distinct.
    # This is a bipartite matching: drivers on left, routes on right.
    # Each driver can only be matched to routes available to them.
    # Since driver-truck is fixed, matching drivers to routes automatically
    # ensures each truck is used exactly once (because driver<->truck is bijection).

    # Build adjacency: driver index 0..k-1, route 1..k
    # But only drivers that appear can be matched. If a driver has no plans, impossible.
    # Also need exactly k distinct routes 1..k.

    # Let's map drivers to 0..k-1
    # We'll do Hopcroft-Karp or simple DFS matching. k <= 1200, t <= k*k.
    # DFS matching O(V*E) = O(k * t) = O(k^3) worst = 1200^3 ~ 1.7e9, too slow.
    # Use Hopcroft-Karp.

    # adjacency list: for each driver (0-indexed), list of routes
    adj = [[] for _ in range(k)]
    drivers_present = [False] * (k + 1)
    for x, y, z in plans:
        adj[x-1].append(z)
        drivers_present[x] = True

    for d in range(1, k+1):
        if not drivers_present[d]:
            print("NO")
            return

    # Remove duplicate routes per driver
    for i in range(k):
        adj[i] = list(set(adj[i]))

    # Hopcroft-Karp
    INF = float('inf')
    pair_u = [0] * (k + 1)  # driver -> route (0 means unmatched)
    pair_v = [0] * (k + 1)  # route -> driver (0 means unmatched)
    dist = [0] * (k + 1)

    from collections import deque

    def bfs():
        queue = deque()
        for u in range(1, k+1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while queue:
            u = queue.popleft()
            if dist[u] < dist[0]:
                for v in adj[u-1]:
                    if dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        queue.append(pair_v[v])
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for v in adj[u-1]:
                if dist[pair_v[v]] == dist[u] + 1:
                    if dfs(pair_v[v]):
                        pair_u[u] = v
                        pair_v[v] = u
                        return True
            dist[u] = INF
            return False
        return True

    matching = 0
    while bfs():
        for u in range(1, k+1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching += 1

    if matching != k:
        print("NO")
        return

    # pair_u[driver] = route
    # driver -> truck
    print("YES")
    out = []
    for d in range(1, k+1):
        r = pair_u[d]
        tr = driver_truck[d]
        out.append(f"{d} {tr} {r}")
    sys.stdout.write("\n".join(out) + "\n")

solve()
