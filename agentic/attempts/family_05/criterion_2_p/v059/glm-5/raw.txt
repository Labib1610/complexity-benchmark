import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # chassis_to_nav[x] = y
    chassis_to_nav = {}
    # schematics_by_chassis[x] = list of z
    schematics_by_chassis = {}
    
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        chassis_to_nav[x] = y
        schematics_by_chassis.setdefault(x, []).append(z)
    
    # We need to select exactly one battery for each chassis 1..k
    # such that all selected batteries are distinct and cover 1..k.
    # This is a bipartite matching problem between chassis and batteries.
    # Chassis are 1..k, batteries are 1..k.
    # Edge from chassis x to battery z exists if (x, y, z) is a schematic.
    
    # Build adjacency list for chassis (1-indexed)
    adj = [[] for _ in range(k + 1)]
    for x in range(1, k + 1):
        if x in schematics_by_chassis:
            # Remove duplicates just in case, though problem says no duplicate schematics
            adj[x] = list(set(schematics_by_chassis[x]))
    
    # Hopcroft-Karp or simple DFS matching since k <= 1200
    match_chassis = [0] * (k + 1)  # match_chassis[x] = z
    match_battery = [0] * (k + 1)  # match_battery[z] = x
    
    def dfs(x, visited):
        for z in adj[x]:
            if visited[z]:
                continue
            visited[z] = True
            if match_battery[z] == 0 or dfs(match_battery[z], visited):
                match_chassis[x] = z
                match_battery[z] = x
                return True
        return False
    
    match_count = 0
    for x in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(x, visited):
            match_count += 1
    
    if match_count != k:
        sys.stdout.write("NO\n")
        return
    
    sys.stdout.write("YES\n")
    for x in range(1, k + 1):
        y = chassis_to_nav[x]
        z = match_chassis[x]
        sys.stdout.write(f"{x} {y} {z}\n")

solve()
