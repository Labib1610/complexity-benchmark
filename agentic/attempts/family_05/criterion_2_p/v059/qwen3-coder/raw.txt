from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    schematics = []
    chassis_to_nav = {}
    nav_to_chassis = {}
    battery_usage = defaultdict(int)
    
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        idx += 3
        schematics.append((x, y, z))
        
        # Build mapping from chassis to navigation unit (and vice versa)
        if x not in chassis_to_nav:
            chassis_to_nav[x] = y
        if y not in nav_to_chassis:
            nav_to_chassis[y] = x
            
        # Count how many times each battery appears
        battery_usage[z] += 1
    
    # Check if we have exactly k chassis and k navigation units
    if len(chassis_to_nav) != k or len(nav_to_chassis) != k:
        print("NO")
        return
        
    # Build a bipartite graph between chassis and batteries
    # Edge exists if there's a schematic with that chassis-battery combination
    graph = defaultdict(list)  # chassis -> list of batteries
    battery_count = defaultdict(int)
    
    for x, y, z in schematics:
        graph[x].append(z)
        battery_count[z] += 1
    
    # Try to find a perfect matching using Hall's theorem approach
    # Since each chassis must be matched to exactly one battery,
    # and each battery can only be used once (because each nav unit 
    # is paired with exactly one chassis, and we need to use all),
    # this becomes a maximum bipartite matching problem
    
    # Build adjacency list for matching
    adj = defaultdict(list)
    for x, y, z in schematics:
        adj[x].append(z)
    
    # Maximum bipartite matching
    def bpm(u, match_b, seen):
        for v in adj[u]:
            if seen[v]:
                continue
            seen[v] = True
            
            if match_b[v] == -1 or bpm(match_b[v], match_b, seen):
                match_b[v] = u
                return True
        return False
    
    # match_b[battery] = chassis assigned to this battery, or -1 if none
    match_b = defaultdict(lambda: -1)
    
    result = 0
    for u in range(1, k+1):
        seen = defaultdict(bool)
        if bpm(u, match_b, seen):
            result += 1
    
    if result != k:
        print("NO")
        return
    
    # Reconstruct the solution
    print("YES")
    # Create reverse mapping: chassis -> battery
    chassis_to_battery = {}
    for b, c in match_b.items():
        if c != -1:
            chassis_to_battery[c] = b
    
    # For each chassis, get its navigation unit and assigned battery
    for x in range(1, k+1):
        y = chassis_to_nav[x]
        z = chassis_to_battery[x]
        print(x, y, z)

solve()
