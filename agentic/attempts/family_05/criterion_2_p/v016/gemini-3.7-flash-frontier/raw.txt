```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    if t < k:
        print("NO")
        return
    
    adj = [[] for _ in range(k + 1)]
    u_to_v = [0] * (k + 1)
    
    idx = 2
    for _ in range(t):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        w = int(input_data[idx + 2])
        idx += 3
        adj[u].append(w)
        u_to_v[u] = v
        
    pair_u = [0] * (k + 1)
    pair_v = [0] * (k + 1)
    dist = [0] * (k + 1)
    INF = 10**9

    def bfs():
        queue = []
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            if dist[u] < dist[0]:
                for w in adj[u]:
                    v_next = pair_v[w]
                    if dist[v_next] == INF:
                        dist[v_next] = dist[u] + 1
                        queue.append(v_next)
        return dist[0] != INF

    def dfs(u):
        if u != 0:
            for w in adj[u]:
                v_next = pair_v[w]
                if dist[v_next] == dist[u] + 1:
                    if dfs(v_next):
                        pair_v[w] = u
                        pair_u[u] = w
                        return True
            dist[u] = INF
            return False
        return True

    matching_size = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                if dfs(u):
                    matching_size += 1

    if matching_size == k:
        out = ["YES"]
        for u in range(1, k + 1):
            out.append(f"{u} {u_to_v[u]} {pair_u[u]}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
```
