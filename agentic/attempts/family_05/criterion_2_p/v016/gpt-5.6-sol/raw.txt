```python
import sys

def main():
    input_buffer = sys.stdin.buffer
    first = input_buffer.readline().split()
    if not first:
        return

    k, t = map(int, first)

    if t < k:
        sys.stdout.write("NO\n")
        return

    adjacency = [[] for _ in range(k)]
    module_of = [-1] * k

    match_left = [-1] * k
    match_right = [-1] * k
    matched = 0

    for _ in range(t):
        u, v, w = map(int, input_buffer.readline().split())
        u -= 1
        v -= 1
        w -= 1

        adjacency[u].append(w)
        module_of[u] = v

        if match_left[u] == -1 and match_right[w] == -1:
            match_left[u] = w
            match_right[w] = u
            matched += 1

    if matched < k:
        sys.setrecursionlimit(10000)
        INF = k + 1

        while True:
            dist = [INF] * k
            queue = [0] * k
            head = 0
            tail = 0

            for u in range(k):
                if match_left[u] == -1:
                    dist[u] = 0
                    queue[tail] = u
                    tail += 1

            shortest = INF

            while head < tail:
                u = queue[head]
                head += 1
                du = dist[u]

                if du >= shortest:
                    continue

                for w in adjacency[u]:
                    next_u = match_right[w]
                    if next_u == -1:
                        shortest = du + 1
                    elif dist[next_u] == INF:
                        dist[next_u] = du + 1
                        queue[tail] = next_u
                        tail += 1

            if shortest == INF:
                break

            def dfs(u):
                du = dist[u]
                for w in adjacency[u]:
                    next_u = match_right[w]

                    if next_u == -1:
                        if du + 1 != shortest:
                            continue
                    else:
                        if dist[next_u] != du + 1 or not dfs(next_u):
                            continue

                    match_left[u] = w
                    match_right[w] = u
                    return True

                dist[u] = INF
                return False

            augmented = 0
            for u in range(k):
                if match_left[u] == -1 and dfs(u):
                    augmented += 1

            if augmented == 0:
                break
            matched += augmented

    if matched != k:
        sys.stdout.write("NO\n")
        return

    output = ["YES"]
    for u in range(k):
        output.append(f"{u + 1} {module_of[u] + 1} {match_left[u] + 1}")

    sys.stdout.write("\n".join(output))

if __name__ == "__main__":
    main()
```
