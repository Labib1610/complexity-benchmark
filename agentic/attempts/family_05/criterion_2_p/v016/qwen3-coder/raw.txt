from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    configs = []
    transmitter_to_configs = defaultdict(list)
    crypto_to_configs = defaultdict(list)
    freq_to_configs = defaultdict(list)
    
    for _ in range(t):
        u, v, w = map(int, input().split())
        config = (u, v, w)
        configs.append(config)
        transmitter_to_configs[u].append(len(configs) - 1)
        crypto_to_configs[v].append(len(configs) - 1)
        freq_to_configs[w].append(len(configs) - 1)
    
    # Build bipartite graph for matching
    # Left side: transmitters (1 to k)
    # Right side: crypto modules (1 to k)
    # Edges exist if there's at least one configuration with that (transmitter, crypto) pair
    
    # Since each transmitter is hardwired to exactly one crypto module,
    # we first determine which crypto module each transmitter is connected to
    transmitter_crypto_pairs = {}
    for i, (u, v, w) in enumerate(configs):
        if u not in transmitter_crypto_pairs:
            transmitter_crypto_pairs[u] = v
        elif transmitter_crypto_pairs[u] != v:
            print("NO")
            return
    
    # Verify that all k transmitters are present
    if len(transmitter_crypto_pairs) != k:
        print("NO")
        return
    
    # Create mapping from transmitter to its only crypto module
    valid_configs_for_matching = []
    config_index_to_actual_config = {}
    
    for i, (u, v, w) in enumerate(configs):
        if u in transmitter_crypto_pairs and transmitter_crypto_pairs[u] == v:
            valid_configs_for_matching.append((u, w, i))  # (transmitter, frequency, config_index)
            config_index_to_actual_config[i] = (u, v, w)
    
    # Now we need to match transmitters to frequencies
    # This is a bipartite matching problem: transmitters <-> frequencies
    # We can use a simple approach since k <= 1200
    
    # For each transmitter, collect available frequencies
    transmitter_freq_options = defaultdict(list)
    freq_transmitter_options = defaultdict(list)
    
    for u, w, config_idx in valid_configs_for_matching:
        transmitter_freq_options[u].append((w, config_idx))
        freq_transmitter_options[w].append((u, config_idx))
    
    # Check if any frequency appears in more than k configs
    for w, trans_list in freq_transmitter_options.items():
        if len(trans_list) > k:
            print("NO")
            return
    
    # Use greedy matching or Hungarian algorithm approach
    # But since constraints are specific, let's model as max flow or bipartite matching
    
    # Build adjacency for bipartite matching
    # Left nodes: transmitters 1..k
    # Right nodes: frequencies 1..k
    adj = defaultdict(list)
    freq_count = defaultdict(int)
    
    for u, w, config_idx in valid_configs_for_matching:
        adj[u].append((w, config_idx))
        freq_count[w] += 1
    
    # Check necessary condition: each frequency used at most k times
    for freq, count in freq_count.items():
        if count > k:
            print("NO")
            return
    
    # Try to find a perfect matching using a greedy approach with backtracking
    # Actually, this is finding a perfect matching in a bipartite graph
    # where left side is transmitters, right side is frequencies
    
    matched_transmitter = {}  # transmitter -> (frequency, config_index)
    matched_frequency = {}    # frequency -> (transmitter, config_index)
    used_configs = set()
    
    def dfs(u, visited):
        for w, config_idx in adj[u]:
            if w in visited:
                continue
            visited.add(w)
            
            # If frequency w is not matched yet, or we can find augmenting path
            if w not in matched_frequency:
                matched_transmitter[u] = (w, config_idx)
                matched_frequency[w] = (u, config_idx)
                return True
            else:
                # Try to rematch
                old_u, old_config_idx = matched_frequency[w]
                if dfs(old_u, visited):
                    matched_transmitter[u] = (w, config_idx)
                    matched_frequency[w] = (u, config_idx)
                    return True
        return False
    
    # Try to match all transmitters
    for u in range(1, k+1):
        if u not in adj:
            print("NO")
            return
        visited = set()
        if not dfs(u, visited):
            print("NO")
            return
    
    # Verify that we have exactly k matches
    if len(matched_transmitter) != k:
        print("NO")
        return
        
    print("YES")
    for u in range(1, k+1):
        w, config_idx = matched_transmitter[u]
        u_orig, v_orig, w_orig = config_index_to_actual_config[config_idx]
        print(u_orig, v_orig, w_orig)

solve()
