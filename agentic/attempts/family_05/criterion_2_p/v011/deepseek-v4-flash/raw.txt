We can model this as a bipartite matching problem between API endpoints (or shards) and IP addresses, since the endpoint–shard pairing is fixed.

Here’s the Python solution:

```python
import sys

def main():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    # endpoint -> shard mapping (fixed)
    e_to_d = [0] * (k + 1)
    # For each IP, list of endpoints that can use it
    ip_to_endpoints = [[] for _ in range(k + 1)]
    
    for _ in range(t):
        e = int(next(it))
        d = int(next(it))
        i = int(next(it))
        e_to_d[e] = d
        ip_to_endpoints[i].append(e)
    
    # We need to match each endpoint to a unique IP
    match_ip = [0] * (k + 1)  # which endpoint is assigned to each IP
    match_e = [0] * (k + 1)   # which IP is assigned to each endpoint
    
    def dfs(e, visited):
        for ip in ip_to_endpoints[e]:
            if visited[ip]:
                continue
            visited[ip] = True
            if match_ip[ip] == 0 or dfs(match_ip[ip], visited):
                match_ip[ip] = e
                match_e[e] = ip
                return True
        return False
    
    for e in range(1, k + 1):
        visited = [False] * (k + 1)
        if not dfs(e, visited):
            print("NO")
            return
    
    print("YES")
    for e in range(1, k + 1):
        d = e_to_d[e]
        i = match_e[e]
        print(f"{e} {d} {i}")

if __name__ == "__main__":
    main()
```
