from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    triads = []
    source_to_module = {}
    module_to_source = {}
    
    for _ in range(t):
        x, y, z = map(int, input().split())
        triads.append((x, y, z))
        
        # Verify the constraint: each source maps to exactly one module
        if x in source_to_module:
            if source_to_module[x] != y:
                # This shouldn't happen according to problem statement
                print("NO")
                return
        else:
            source_to_module[x] = y
            
        if y in module_to_source:
            if module_to_source[y] != x:
                # This shouldn't happen according to problem statement
                print("NO")
                return
        else:
            module_to_source[y] = x
    
    # If we don't have exactly k sources or k modules, impossible
    if len(source_to_module) != k or len(module_to_source) != k:
        print("NO")
        return
    
    # Build a graph for matching: sources to channels
    # For each source x, we know its module y
    # So we just need to match x with a channel z such that (x,y,z) exists
    
    # Group triads by source
    source_triads = defaultdict(list)
    for x, y, z in triads:
        source_triads[x].append(z)
    
    # Build bipartite graph: left side is sources (1 to k), right side is channels (1 to k)
    # Edge exists if there's a triad (source, module, channel)
    
    # For Hungarian algorithm or max flow, we can model this as max bipartite matching
    # But since we need a perfect matching (k edges), we can use a simpler approach
    
    # Build adjacency for sources to channels
    adj = defaultdict(list)
    for x, y, z in triads:
        adj[x].append(z)
    
    # Try to find a perfect matching using DFS-based algorithm
    match_channel_to_source = [-1] * (k + 1)  # 1-indexed
    match_source_to_channel = [-1] * (k + 1)  # 1-indexed
    
    def dfs(u, visited):
        for v in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            
            if match_channel_to_source[v] == -1 or dfs(match_channel_to_source[v], visited):
                match_channel_to_source[v] = u
                match_source_to_channel[u] = v
                return True
        return False
    
    matched = 0
    for u in range(1, k + 1):
        visited = [False] * (k + 1)
        if dfs(u, visited):
            matched += 1
    
    if matched != k:
        print("NO")
        return
    
    print("YES")
    for u in range(1, k + 1):
        channel = match_source_to_channel[u]
        module = source_to_module[u]
        print(u, module, channel)

solve()
