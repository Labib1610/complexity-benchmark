```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    # mapping sensor -> relay (must be consistent)
    relay_of = [0] * (k + 1)
    used_relays = [0] * (k + 1)
    adj = [[] for _ in range(k + 1)]   # edges from sensor to frequency

    for _ in range(t):
        u = int(next(it)); v = int(next(it)); w = int(next(it))
        if relay_of[u] == 0:
            relay_of[u] = v
            used_relays[v] = u
        # else: consistency guaranteed by statement
        adj[u].append(w)

    # quick check: every sensor must have at least one edge, otherwise impossible
    for u in range(1, k + 1):
        if not adj[u]:
            print("NO")
            return

    # Hopcroft–Karp
    INF = 10 ** 9
    pair_u = [0] * (k + 1)   # sensor -> frequency
    pair_v = [0] * (k + 1)   # frequency -> sensor
    dist = [0] * (k + 1)

    def bfs() -> bool:
        q = deque()
        for u in range(1, k + 1):
            if pair_u[u] == 0:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = INF
        dist[0] = INF
        while q:
            u = q.popleft()
            if dist[u] < dist[0]:
                for v in adj[u]:
                    if pair_v[v] == 0:
                        if dist[0] == INF:
                            dist[0] = dist[u] + 1
                    elif dist[pair_v[v]] == INF:
                        dist[pair_v[v]] = dist[u] + 1
                        q.append(pair_v[v])
        return dist[0] != INF

    def dfs(u: int) -> bool:
        if u == 0:
            return True
        for v in adj[u]:
            pu = pair_v[v]
            if pu == 0 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                return True
        dist[u] = INF
        return False

    matching = 0
    while bfs():
        for u in range(1, k + 1):
            if pair_u[u] == 0 and dfs(u):
                matching += 1

    if matching != k:
        print("NO")
        return

    # construct answer
    print("YES")
    out_lines = []
    for u in range(1, k + 1):
        v = relay_of[u]
        w = pair_u[u]
        out_lines.append(f"{u} {v} {w}")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
```
