from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    configs = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        configs.append((x, y, z))
        index += 3
    
    # We need to find a subset of k configurations such that:
    # - All processors 1..k are used exactly once
    # - All sensors 1..k are used exactly once
    # - All transmitters 1..k are used exactly once
    
    # This is essentially a matching problem. We can model it as:
    # Find a perfect matching in a 3-dimensional matching problem.
    
    # Since k <= 22, we can try backtracking or other approaches.
    # However, since t <= k^3, and k <= 22, t can be up to 10648 which is large.
    # But in practice, we're looking for a subset of size k from t configurations.
    
    # Let's try a backtracking approach:
    # For each configuration, either include it or not.
    # But this is 2^t which can be too large.
    
    # Alternative: since we need exactly k configurations, and each component 
    # must be used exactly once, we can think of this as finding a permutation
    # matching between processors, sensors, and transmitters.
    
    # Another way: since k <= 22, we can try all permutations of processors,
    # and for each permutation, check if there exists a matching.
    # But even k! can be too large for k=22.
    
    # Let's think differently. This is a tripartite matching problem.
    # We can reduce it to a maximum bipartite matching by creating a graph
    # where one side is configurations and the other side is components.
    # But that might not directly help.
    
    # Let's try a more direct approach:
    # Try all subsets of size k from the configurations and check if valid.
    # C(t, k) can be huge.
    
    # Given the constraints, let's implement a backtracking solution that prunes early.
    
    from collections import defaultdict
    
    # Map each component to the list of configurations that use it
    proc_to_configs = defaultdict(list)
    sens_to_configs = defaultdict(list)
    tran_to_configs = defaultdict(list)
    
    for i, (x, y, z) in enumerate(configs):
        proc_to_configs[x].append(i)
        sens_to_configs[y].append(i)
        tran_to_configs[z].append(i)
    
    # Backtracking with pruning
    selected = [False] * t
    used_proc = [False] * (k + 1)
    used_sens = [False] * (k + 1)
    used_tran = [False] * (k + 1)
    result = []
    
    def backtrack(idx):
        if len(result) == k:
            return True
        
        if idx == t:
            return False
            
        # Pruning: if even if we select all remaining configs, we won't have enough
        if len(result) + (t - idx) < k:
            return False
            
        # Try including current config
        x, y, z = configs[idx]
        if not used_proc[x] and not used_sens[y] and not used_tran[z]:
            selected[idx] = True
            used_proc[x] = True
            used_sens[y] = True
            used_tran[z] = True
            result.append((x, y, z))
            
            if backtrack(idx + 1):
                return True
                
            result.pop()
            used_proc[x] = False
            used_sens[y] = False
            used_tran[z] = False
            selected[idx] = False
            
        # Try not including current config
        if backtrack(idx + 1):
            return True
            
        return False
    
    # Special case when k is small, we can try all combinations
    # But for larger k, this backtracking should work well with pruning
    
    if backtrack(0):
        print("YES")
        for x, y, z in result:
            print(x, y, z)
    else:
        print("NO")

solve()
