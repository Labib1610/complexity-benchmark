import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        idx += 3
        triples.append((x, y, z))
    
    # Items:
    # 0 .. k-1: processor x
    # k .. 2k-1: sensor y
    # 2k .. 3k-1: transmitter z
    
    total_items = 3 * k
    triple_mask = []
    item_triples = [[] for _ in range(total_items)]
    
    for tr_idx, (x, y, z) in enumerate(triples):
        i1 = x
        i2 = k + y
        i3 = 2 * k + z
        mask = (1 << i1) | (1 << i2) | (1 << i3)
        triple_mask.append(mask)
        item_triples[i1].append(tr_idx)
        item_triples[i2].append(tr_idx)
        item_triples[i3].append(tr_idx)
        
    solution = []
    
    def search(used_mask, depth):
        if depth == k:
            return True
        
        # Pick the uncovered item with the minimum number of valid candidate triples
        best_item = -1
        best_candidates = None
        min_candidates = 999999
        
        for i in range(total_items):
            if not ((used_mask >> i) & 1):
                candidates = [j for j in item_triples[i] if not (triple_mask[j] & used_mask)]
                cnt = len(candidates)
                if cnt == 0:
                    return False
                if cnt < min_candidates:
                    min_candidates = cnt
                    best_candidates = candidates
                    best_item = i
                    if cnt == 1:
                        break
        
        for tr_idx in best_candidates:
            solution.append(tr_idx)
            if search(used_mask | triple_mask[tr_idx], depth + 1):
                return True
            solution.pop()
            
        return False

    if search(0, 0):
        print("YES")
        out = []
        for tr_idx in solution:
            x, y, z = triples[tr_idx]
            out.append(f"{x+1} {y+1} {z+1}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
