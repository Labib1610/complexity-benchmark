import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    rows = []
    col_rows = [[] for _ in range(3 * k)]

    pos = 2
    for rid in range(t):
        a = data[pos] - 1
        b = data[pos + 1] - 1
        c = data[pos + 2] - 1
        pos += 3
        cols = (a, k + b, 2 * k + c)
        rows.append(cols)
        col_rows[a].append(rid)
        col_rows[k + b].append(rid)
        col_rows[2 * k + c].append(rid)

    if k == 0:
        print("YES")
        return

    counts = [len(x) for x in col_rows]
    if any(x == 0 for x in counts):
        print("NO")
        return

    active = bytearray(b"\x01") * t
    full_mask = (1 << (3 * k)) - 1
    chosen = []
    failed = set()
    cache_limit = 1_000_000
    nodes = 0

    def has_perfect_matching(adj, left_mask, right_mask):
        left = []
        x = left_mask
        while x:
            bit = x & -x
            u = bit.bit_length() - 1
            choices = adj[u] & right_mask
            if not choices:
                return False
            left.append((choices.bit_count(), u))
            x ^= bit

        left.sort()
        match = [-1] * k

        def augment(u, seen):
            bits = adj[u] & right_mask & ~seen[0]
            while bits:
                bit = bits & -bits
                bits ^= bit
                v = bit.bit_length() - 1
                seen[0] |= bit
                if match[v] == -1 or augment(match[v], seen):
                    match[v] = u
                    return True
            return False

        for _, u in left:
            seen = [0]
            if not augment(u, seen):
                return False
        return True

    def projection_feasible(covered):
        available = ((1 << k) - 1)
        ma = available & ~(covered & available)
        mb = available & ~((covered >> k) & available)
        mc = available & ~((covered >> (2 * k)) & available)

        ab = [0] * k
        ac = [0] * k
        bc = [0] * k

        for rid, flag in enumerate(active):
            if flag:
                x, y, z = rows[rid]
                a = x
                b = y - k
                c = z - 2 * k
                ab[a] |= 1 << b
                ac[a] |= 1 << c
                bc[b] |= 1 << c

        return (
            has_perfect_matching(ab, ma, mb)
            and has_perfect_matching(ac, ma, mc)
            and has_perfect_matching(bc, mb, mc)
        )

    def remember_failure(state):
        if len(failed) < cache_limit:
            failed.add(state)

    sys.setrecursionlimit(10000)

    def dfs(covered, depth):
        nonlocal nodes
        nodes += 1

        if covered == full_mask:
            return True
        if covered in failed:
            return False

        best_col = -1
        best_count = t + 1
        uncovered = full_mask ^ covered

        bits = uncovered
        while bits:
            bit = bits & -bits
            col = bit.bit_length() - 1
            cnt = counts[col]
            if cnt == 0:
                remember_failure(covered)
                return False
            if cnt < best_count:
                best_count = cnt
                best_col = col
                if cnt == 1:
                    break
            bits ^= bit

        if best_count > 2 and (depth == 0 or depth % 2 == 0):
            if not projection_feasible(covered):
                remember_failure(covered)
                return False

        candidates = [rid for rid in col_rows[best_col] if active[rid]]
        if len(candidates) > 1:
            candidates.sort(
                key=lambda rid: sum(
                    counts[c] for c in rows[rid] if c != best_col
                )
            )

        for rid in candidates:
            if not active[rid]:
                continue

            cols = rows[rid]
            row_mask = (1 << cols[0]) | (1 << cols[1]) | (1 << cols[2])
            if covered & row_mask:
                continue

            removed = []
            for col in cols:
                for other in col_rows[col]:
                    if active[other]:
                        active[other] = 0
                        removed.append(other)
                        x, y, z = rows[other]
                        counts[x] -= 1
                        counts[y] -= 1
                        counts[z] -= 1

            chosen.append(rid)
            if dfs(covered | row_mask, depth + 1):
                return True
            chosen.pop()

            for other in reversed(removed):
                active[other] = 1
                x, y, z = rows[other]
                counts[x] += 1
                counts[y] += 1
                counts[z] += 1

        remember_failure(covered)
        return False

    if dfs(0, 0):
        out = ["YES"]
        for rid in chosen:
            a, b, c = rows[rid]
            out.append(f"{a + 1} {b - k + 1} {c - 2 * k + 1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    main()
