import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    triplets = []
    idx = 2
    for _ in range(t):
        m = int(input_data[idx]) - 1
        p = int(input_data[idx+1]) - 1
        s = int(input_data[idx+2]) - 1
        idx += 3
        triplets.append((m, p, s))
    
    # Store triplets by component
    by_m = [[] for _ in range(k)]
    by_p = [[] for _ in range(k)]
    by_s = [[] for _ in range(k)]
    
    for i, (m, p, s) in enumerate(triplets):
        by_m[m].append(i)
        by_p[p].append(i)
        by_s[s].append(i)
    
    # Quick checks
    for i in range(k):
        if not by_m[i] or not by_p[i] or not by_s[i]:
            print("NO")
            return
            
    used_m = [False] * k
    used_p = [False] * k
    used_s = [False] * k
    
    selected = []
    
    def search(depth):
        if depth == k:
            return True
        
        # Find item with minimum remaining options
        best_count = float('inf')
        best_type = -1
        best_val = -1
        best_options = None
        
        # Check remaining m
        for m in range(k):
            if not used_m[m]:
                opts = [i for i in by_m[m] if not used_p[triplets[i][1]] and not used_s[triplets[i][2]]]
                c = len(opts)
                if c == 0:
                    return False
                if c < best_count:
                    best_count = c
                    best_options = opts
                    if c == 1:
                        break
        
        if best_count > 1:
            # Check remaining p
            for p in range(k):
                if not used_p[p]:
                    opts = [i for i in by_p[p] if not used_m[triplets[i][0]] and not used_s[triplets[i][2]]]
                    c = len(opts)
                    if c == 0:
                        return False
                    if c < best_count:
                        best_count = c
                        best_options = opts
                        if c == 1:
                            break
                            
        if best_count > 1:
            # Check remaining s
            for s in range(k):
                if not used_s[s]:
                    opts = [i for i in by_s[s] if not used_m[triplets[i][0]] and not used_p[triplets[i][1]]]
                    c = len(opts)
                    if c == 0:
                        return False
                    if c < best_count:
                        best_count = c
                        best_options = opts
                        if c == 1:
                            break

        for idx_trip in best_options:
            m, p, s = triplets[idx_trip]
            used_m[m] = True
            used_p[p] = True
            used_s[s] = True
            selected.append(idx_trip)
            
            if search(depth + 1):
                return True
                
            selected.pop()
            used_m[m] = False
            used_p[p] = False
            used_s[s] = False
            
        return False

    if search(0):
        print("YES")
        for idx_trip in selected:
            m, p, s = triplets[idx_trip]
            print(f"{m + 1} {p + 1} {s + 1}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
