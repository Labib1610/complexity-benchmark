import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))

    # DLX structures
    n_cols = 3 * k  # columns: X0..Xk-1, Y0..Yk-1, Z0..Zk-1
    # header node 0
    size = [0] * (n_cols + 1)
    left = [0] * (n_cols + 1)
    right = [0] * (n_cols + 1)
    up = [i for i in range(n_cols + 1)]
    down = [i for i in range(n_cols + 1)]
    col = [0] * (n_cols + 1)   # column of node
    row_id = [0] * (n_cols + 1)  # which triple this node belongs to

    # init header links
    for i in range(n_cols + 1):
        left[i] = i - 1
        right[i] = i + 1
    left[0] = n_cols
    right[n_cols] = 0

    node_cnt = n_cols

    # add rows
    for idx, (x, y, z) in enumerate(triples):
        cols = [x, k + y, 2 * k + z]  # column indices (1..n_cols)
        first = None
        for c in cols:
            node_cnt += 1
            col.append(c + 1)          # shift by 1 because column 0 is header
            row_id.append(idx)
            # vertical links
            up.append(up[c + 1])
            down.append(c + 1)
            down[up[c + 1]] = node_cnt
            up[c + 1] = node_cnt
            size[c + 1] += 1
            # horizontal links
            if first is None:
                left.append(node_cnt)
                right.append(node_cnt)
                first = node_cnt
            else:
                left.append(right[first - 1])
                right.append(first)
                right[left[first - 1]] = node_cnt
                left[first] = node_cnt
        # adjust first node's left/right to close the row
        # already handled in loop

    solution_rows = []

    def cover(c):
        # remove column header
        right[left[c]] = right[c]
        left[right[c]] = left[c]
        # remove rows
        i = down[c]
        while i != c:
            j = right[i]
            while j != i:
                down[up[j]] = down[j]
                up[down[j]] = up[j]
                size[col[j]] -= 1
                j = right[j]
            i = down[i]

    def uncover(c):
        i = up[c]
        while i != c:
            j = left[i]
            while j != i:
                size[col[j]] += 1
                down[up[j]] = j
                up[down[j]] = j
                j = left[j]
            i = up[i]
        right[left[c]] = c
        left[right[c]] = c

    def search():
        if right[0] == 0:
            return True
        # choose column with minimal size
        c = right[0]
        min_s = size[c]
        j = right[c]
        while j != 0:
            if size[j] < min_s:
                c = j
                min_s = size[j]
                if min_s == 0:
                    break
            j = right[j]
        if min_s == 0:
            return False
        cover(c)
        r = down[c]
        while r != c:
            solution_rows.append(row_id[r])
            j = right[r]
            while j != r:
                cover(col[j])
                j = right[j]
            if search():
                return True
            # backtrack
            j = left[r]
            while j != r:
                uncover(col[j])
                j = left[j]
            solution_rows.pop()
            r = down[r]
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for idx in solution_rows:
            x, y, z = triples[idx]
            out.append(f"{x+1} {y+1} {z+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
