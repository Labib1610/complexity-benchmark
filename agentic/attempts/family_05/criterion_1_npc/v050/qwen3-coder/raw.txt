from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    specs = []
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        specs.append((x, y, z))
        idx += 3
    
    # Create mappings for quick lookup
    spec_set = set(specs)
    
    # Generate all permutations for each dimension
    perm_x = list(permutations(range(1, k+1)))
    perm_y = list(permutations(range(1, k+1)))
    perm_z = list(permutations(range(1, k+1)))
    
    # Try all combinations of permutations
    for px in perm_x:
        for py in perm_y:
            for pz in perm_z:
                # Check if this combination forms valid specs
                valid = True
                result = []
                for i in range(k):
                    x = px[i]
                    y = py[i]
                    z = pz[i]
                    if (x, y, z) not in spec_set:
                        valid = False
                        break
                    result.append((x, y, z))
                
                if valid:
                    print("YES")
                    for spec in result:
                        print(spec[0], spec[1], spec[2])
                    return
    
    print("NO")

solve()
