import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    rows = []
    for _ in range(t):
        e = int(next(it)) - 1
        s = int(next(it)) - 1
        p = int(next(it)) - 1
        rows.append((e, s, p))

    # Exact cover with DLX
    n_cols = 3 * k  # enzymes 0..k-1, substrates k..2k-1, profiles 2k..3k-1

    # Build column headers
    size = [0] * n_cols
    col_nodes = [[] for _ in range(n_cols)]
    for idx, (e, s, p) in enumerate(rows):
        cols = [e, k + s, 2 * k + p]
        for c in cols:
            col_nodes[c].append(idx)

    # DLX structures
    max_nodes = t * 3 + n_cols + 1
    U = [0] * max_nodes
    D = [0] * max_nodes
    L = [0] * max_nodes
    R = [0] * max_nodes
    C = [0] * max_nodes  # column of node
    row_id = [0] * max_nodes

    header = 0
    # init header links
    for i in range(n_cols + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = D[i] = i
    R[n_cols] = header
    L[header] = n_cols

    node_cnt = n_cols + 1

    # add rows
    for r, (e, s, p) in enumerate(rows):
        cols = [e, k + s, 2 * k + p]
        first = None
        for c in cols:
            node = node_cnt
            node_cnt += 1
            C[node] = c + 1          # column headers are 1..n_cols
            row_id[node] = r
            # vertical links
            U[node] = U[c + 1]
            D[node] = c + 1
            D[U[c + 1]] = node
            U[c + 1] = node
            size[c] += 1
            # horizontal links
            if first is None:
                first = node
                L[node] = R[node] = node
            else:
                L[node] = L[first]
                R[node] = first
                R[L[first]] = node
                L[first] = node

    solution = []

    def cover(col):
        # remove column header
        R[L[col]] = R[col]
        L[R[col]] = L[col]
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j] - 1] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                size[C[j] - 1] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        R[L[col]] = col
        L[R[col]] = col

    sys.setrecursionlimit(10000)

    def search():
        if R[header] == header:
            return True
        # choose column with minimal size
        c = R[header]
        min_sz = size[C[c] - 1]
        col = c
        j = R[c]
        while j != header:
            if size[C[j] - 1] < min_sz:
                min_sz = size[C[j] - 1]
                col = j
                if min_sz == 0:
                    break
            j = R[j]
        if min_sz == 0:
            return False
        cover(col)
        r = D[col]
        while r != col:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            if search():
                return True
            # backtrack
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution.pop()
            r = D[r]
        uncover(col)
        return False

    if search():
        out = ["YES"]
        for idx in solution:
            e, s, p = rows[idx]
            out.append(f"{e+1} {s+1} {p+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
