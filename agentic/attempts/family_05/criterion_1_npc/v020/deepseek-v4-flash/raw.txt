We can model this as a 3D matching problem, but since k ≤ 22, we can use a backtracking approach with pruning, or reduce it to a bipartite matching after fixing one dimension.

A common trick: fix the enzyme order, then try to match substrates and thermal profiles using bipartite matching for each enzyme. We can do DFS over enzymes, and for each enzyme, try all possible (substrate, profile) pairs from the catalog, then check if the remaining can be matched.

We’ll implement a recursive backtracking with bitmasks for substrates and profiles, and use memoization to avoid repeated states.

```python
import sys
sys.setrecursionlimit(10000000)

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    combos = [[] for _ in range(k)]
    for _ in range(t):
        e, s, p = map(int, input().split())
        combos[e-1].append((s-1, p-1))
    
    # Precompute for each enzyme the list of (s, p) pairs
    # We'll do DFS over enzymes, tracking used substrates and profiles via bitmasks
    full_mask = (1 << k) - 1
    memo = {}
    
    def dfs(e_idx, s_mask, p_mask):
        if e_idx == k:
            return True if s_mask == full_mask and p_mask == full_mask else False
        key = (e_idx, s_mask, p_mask)
        if key in memo:
            return memo[key]
        for s, p in combos[e_idx]:
            if not (s_mask & (1 << s)) and not (p_mask & (1 << p)):
                if dfs(e_idx + 1, s_mask | (1 << s), p_mask | (1 << p)):
                    memo[key] = True
                    return True
        memo[key] = False
        return False
    
    if dfs(0, 0, 0):
        print("YES")
        # Reconstruct solution
        s_mask = 0
        p_mask = 0
        chosen = []
        for e_idx in range(k):
            for s, p in combos[e_idx]:
                if not (s_mask & (1 << s)) and not (p_mask & (1 << p)):
                    if dfs(e_idx + 1, s_mask | (1 << s), p_mask | (1 << p)):
                        chosen.append((e_idx+1, s+1, p+1))
                        s_mask |= (1 << s)
                        p_mask |= (1 << p)
                        break
        for e, s, p in chosen:
            print(e, s, p)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
