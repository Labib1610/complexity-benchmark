```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    adj = [[] for _ in range(k)]
    idx = 2
    for _ in range(t):
        d = int(input_data[idx]) - 1
        f = int(input_data[idx+1]) - 1
        m = int(input_data[idx+2]) - 1
        idx += 3
        adj[d].append((f, m))
        
    match_fm = [[-1] * k for _ in range(k)]
    match_d = [-1] * k
    
    def dfs(u, visited):
        for v, w in adj[u]:
            if not visited[v][w]:
                visited[v][w] = True
                if match_fm[v][w] == -1 or dfs(match_fm[v][w], visited):
                    match_fm[v][w] = u
                    match_d[u] = (v, w)
                    return True
        return False

    for u in range(k):
        visited = [[False] * k for _ in range(k)]
        if not dfs(u, visited):
            print("NO")
            return
            
    print("YES")
    for u in range(k):
        v, w = match_d[u]
        print(u + 1, v + 1, w + 1)

if __name__ == '__main__':
    solve()
```
