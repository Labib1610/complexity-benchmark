from itertools import permutations

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    combinations = []
    for _ in range(t):
        d, f, m = map(int, input().split())
        combinations.append((d, f, m))
    
    # Try all permutations of k combinations
    for perm in permutations(combinations, k):
        # Check if all demographics, formulations, and facilities are unique
        demographics = set()
        formulations = set()
        facilities = set()
        
        valid = True
        for d, f, m in perm:
            if d in demographics or f in formulations or m in facilities:
                valid = False
                break
            demographics.add(d)
            formulations.add(f)
            facilities.add(m)
        
        if valid and len(demographics) == k and len(formulations) == k and len(facilities) == k:
            print("YES")
            for d, f, m in perm:
                print(d, f, m)
            return
    
    print("NO")

solve()
