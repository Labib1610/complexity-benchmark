from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    triples = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        triples.append((x, y, z))
        index += 3
    
    # Try all permutations of the triples
    for perm in permutations(triples):
        if len(perm) < k:
            continue
            
        # Take first k triples from permutation
        selected = perm[:k]
        
        # Check if all specialists, machines, and time slots are unique
        specialists = set()
        machines = set()
        time_slots = set()
        
        valid = True
        for x, y, z in selected:
            if x in specialists or y in machines or z in time_slots:
                valid = False
                break
            specialists.add(x)
            machines.add(y)
            time_slots.add(z)
        
        if valid and len(specialists) == k and len(machines) == k and len(time_slots) == k:
            print("YES")
            for x, y, z in selected:
                print(x, y, z)
            return
    
    print("NO")

solve()
