import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    idx = 2
    blueprints = []
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        idx += 3
        blueprints.append((x, y, z))
        
    num_cols = 3 * k
    
    # DLX Implementation
    # Column indices: 1 to num_cols
    # Root: 0
    # Nodes: 0 is root, 1..num_cols are column headers
    
    max_nodes = num_cols + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    row_of_node = [0] * max_nodes
    col_size = [0] * (num_cols + 1)
    
    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = (i - 1 + num_cols + 1) % (num_cols + 1)
        R[i] = (i + 1) % (num_cols + 1)
        U[i] = i
        D[i] = i
        C[i] = i
        col_size[i] = 0

    node_count = num_cols + 1
    
    for r, (x, y, z) in enumerate(blueprints):
        cols = [x + 1, k + y + 1, 2 * k + z + 1]
        first = -1
        for col in cols:
            node = node_count
            node_count += 1
            
            C[node] = col
            row_of_node[node] = r
            col_size[col] += 1
            
            # Insert into column vertical list (at bottom, before header)
            U[node] = U[col]
            D[node] = col
            D[U[col]] = node
            U[col] = node
            
            # Insert into row horizontal list
            if first == -1:
                first = node
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first]
                R[node] = first
                R[L[first]] = node
                L[first] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                col_size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                col_size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size (MRV heuristic)
        min_size = 10**9
        c = -1
        curr = R[0]
        while curr != 0:
            if col_size[curr] < min_size:
                min_size = col_size[curr]
                c = curr
                if min_size == 0:
                    break
            curr = R[curr]
            
        if min_size == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_of_node[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
                
            if search():
                return True
                
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution.pop()
            r = D[r]
            
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r in solution:
            x, y, z = blueprints[r]
            out.append(f"{x + 1} {y + 1} {z + 1}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    main()
