import sys
import random

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    k, t = data[0], data[1]
    triples = [
        (data[i] - 1, data[i + 1] - 1, data[i + 2] - 1)
        for i in range(2, 3 * t + 2, 3)
    ]

    if t == 0:
        print("NO")
        return

    random.Random(712367821).shuffle(triples)

    item_inc = [0] * (3 * k)
    pair_sc = [0] * (k * k)
    pair_sf = [0] * (k * k)
    pair_cf = [0] * (k * k)

    for e, (s, c, f) in enumerate(triples):
        bit = 1 << e
        item_inc[s] |= bit
        item_inc[k + c] |= bit
        item_inc[2 * k + f] |= bit
        pair_sc[s * k + c] |= bit
        pair_sf[s * k + f] |= bit
        pair_cf[c * k + f] |= bit

    if any(x == 0 for x in item_inc):
        print("NO")
        return

    conflict = [0] * t
    for e, (s, c, f) in enumerate(triples):
        conflict[e] = item_inc[s] | item_inc[k + c] | item_inc[2 * k + f]

    full_mask = (1 << k) - 1
    all_edges = (1 << t) - 1

    def projection_has_matching(mask_a, mask_b, pair_masks, available):
        adjacency = []
        aa = mask_a
        while aa:
            abit = aa & -aa
            a = abit.bit_length() - 1
            aa ^= abit

            neighbors = 0
            bb = mask_b
            base = a * k
            while bb:
                bbit = bb & -bb
                b = bbit.bit_length() - 1
                bb ^= bbit
                if available & pair_masks[base + b]:
                    neighbors |= bbit

            if neighbors == 0:
                return False
            adjacency.append(neighbors)

        adjacency.sort(key=int.bit_count)
        match = [-1] * k

        def augment(index, seen):
            choices = adjacency[index] & ~seen
            while choices:
                bbit = choices & -choices
                choices ^= bbit
                b = bbit.bit_length() - 1
                previous = match[b]
                if previous == -1 or augment(previous, seen | bbit):
                    match[b] = index
                    return True
            return False

        for i in range(len(adjacency)):
            if not augment(i, 0):
                return False
        return True

    def necessary(ms, mc, mf, available):
        return (
            projection_has_matching(ms, mc,pair_sc, available)
            and projection_has_matching(ms, mf, pair_sf, available)
            and projection_has_matching(mc, mf, pair_cf, available)
        )

    if not necessary(full_mask, full_mask, full_mask, all_edges):
        print("NO")
        return

    sys.setrecursionlimit(10000)
    failed = set()
    memo_limit = 800000
    answer = []

    def dfs(ms, mc, mf, available, depth):
        if ms == 0:
            return True

        state = ms | (mc << k) | (mf << (2 * k))
        if state in failed:
            return False

        best_candidates = None
        best_count = t + 1

        x = ms
        while x:
            bit = x & -x
            idx = bit.bit_length() - 1
            candidates = available & item_inc[idx]
            count = candidates.bit_count()
            if count == 0:
                if len(failed) < memo_limit:
                    failed.add(state)
                return False
            if count < best_count:
                best_count = count
                best_candidates = candidates
                if count == 1:
                    break
            x ^= bit

        if best_count != 1:
            x = mc
            while x:
                bit = x & -x
                idx = bit.bit_length() - 1
                candidates = available & item_inc[k + idx]
                count = candidates.bit_count()
                if count == 0:
                    if len(failed) < memo_limit:
                        failed.add(state)
                    return False
                if count < best_count:
                    best_count = count
                    best_candidates = candidates
                    if count == 1:
                        break
                x ^= bit

        if best_count != 1:
            x = mf
            while x:
                bit = x & -x
                idx = bit.bit_length() - 1
                candidates = available & item_inc[2 * k + idx]
                count = candidates.bit_count()
                if count == 0:
                    if len(failed) < memo_limit:
                        failed.add(state)
                    return False
                if count < best_count:
                    best_count = count
                    best_candidates = candidates
                    if count == 1:
                        break
                x ^= bit

        if depth > 0 and depth % 5 == 0:
            if not necessary(ms, mc, mf, available):
                if len(failed) < memo_limit:
                    failed.add(state)
                return False

        candidate_ids = []
        x = best_candidates
        while x:
            bit = x & -x
            candidate_ids.append(bit.bit_length() - 1)
            x ^= bit

        if len(candidate_ids) <= 12:
            candidate_ids.sort(
                key=lambda e: (available & conflict[e]).bit_count()
            )

        for e in candidate_ids:
            s, c, f = triples[e]
            sb = 1 << s
            cb = 1 << c
            fb = 1 << f

            if not (ms & sb and mc & cb and mf & fb):
                continue

            answer.append(e)
            if dfs(
                ms ^ sb,
                mc ^ cb,
                mf ^ fb,
                available & ~conflict[e],
                depth + 1
            ):
                return True
            answer.pop()

        if len(failed) < memo_limit:
            failed.add(state)
        return False

    if dfs(full_mask, full_mask, full_mask, all_edges, 0):
        out = ["YES"]
        for e in answer:
            s, c, f = triples[e]
            out.append(f"{s + 1} {c + 1} {f + 1}")
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
