import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    if t < k:
        print("NO")
        return

    rows_xyz = []
    row_cols = []
    col_rows = [[] for _ in range(3 * k)]

    p = 2
    for i in range(t):
        x = data[p] - 1
        y = data[p + 1] - 1
        z = data[p + 2] - 1
        p += 3

        cols = (x, k + y, 2 * k + z)
        rows_xyz.append((x, y, z))
        row_cols.append(cols)
        for c in cols:
            col_rows[c].append(i)

    active = bytearray(b"\x01") * t
    counts = [len(col_rows[c]) for c in range(3 * k)]

    if any(v == 0 for v in counts):
        print("NO")
        return

    full_mask = (1 << (3 * k)) - 1
    chosen = []
    failed = set()

    sys.setrecursionlimit(100000)

    def projection_has_matching(covered, left_dim, right_dim):
        left_offset = left_dim * k
        adjacency = []

        for u in range(k):
            col = left_offset + u
            if (covered >> col) & 1:
                continue

            mask = 0
            for r in col_rows[col]:
                if active[r]:
                    v = rows_xyz[r][right_dim]
                    mask |= 1 << v

            if mask == 0:
                return False
            adjacency.append(mask)

        adjacency.sort(key=int.bit_count)
        match = [-1] * k

        def augment(pos, seen):
            bits = adjacency[pos] & ~seen
            while bits:
                bit = bits & -bits
                bits -= bit
                v = bit.bit_length() - 1
                old = match[v]
                if old == -1 or augment(old, seen | bit):
                    match[v] = pos
                    return True
            return False

        for i in range(len(adjacency)):
            if not augment(i, 0):
                return False
        return True

    def search(covered, depth):
        if covered == full_mask:
            return True
        if covered in failed:
            return False

        best_col = -1
        best_count = t + 1

        remaining = full_mask ^ covered
        bits = remaining
        while bits:
            bit = bits & -bits
            bits -= bit
            c = bit.bit_length() - 1
            cnt = counts[c]
            if cnt == 0:
                failed.add(covered)
                return False
            if cnt < best_count:
                best_count = cnt
                best_col = c
                if cnt == 1:
                    break

        if best_count > 1:
            if not projection_has_matching(covered, 0, 1):
                failed.add(covered)
                return False
            if not projection_has_matching(covered, 0, 2):
                failed.add(covered)
                return False
            if not projection_has_matching(covered, 1, 2):
                failed.add(covered)
                return False

        candidates = [r for r in col_rows[best_col] if active[r]]
        if len(candidates) > 1:
            candidates.sort(
                key=lambda r: (
                    counts[row_cols[r][0]]
                    + counts[row_cols[r][1]]
                    + counts[row_cols[r][2]]
                )
            )

        for selected_row in candidates:
            if not active[selected_row]:
                continue

            selected_cols = row_cols[selected_row]
            disabled = []

            for c in selected_cols:
                for r in col_rows[c]:
                    if active[r]:
                        active[r] = 0
                        disabled.append(r)
                        a, b, d = row_cols[r]
                        counts[a] -= 1
                        counts[b] -= 1
                        counts[d] -= 1

            chosen.append(selected_row)
            new_covered = (
                covered
                | (1 << selected_cols[0])
                | (1 << selected_cols[1])
                | (1 << selected_cols[2])
            )

            if search(new_covered, depth + 1):
                return True

            chosen.pop()

            for r in reversed(disabled):
                active[r] = 1
                a, b, d = row_cols[r]
                counts[a] += 1
                counts[b] += 1
                counts[d] += 1

        failed.add(covered)
        return False

    if search(0, 0):
        out = ["YES"]
        for r in chosen:
            x, y, z = rows_xyz[r]
            out.append(f"{x + 1} {y + 1} {z + 1}")
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    main()
