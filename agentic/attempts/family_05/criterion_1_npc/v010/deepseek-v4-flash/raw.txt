We can model this as a 3D matching problem, but since \( k \leq 22 \), we can use a bitmask DP over two of the dimensions and match the third.

We’ll fix specialists and machinery, and for each pair we store the set of possible time windows. Then we do DP over subsets of specialists and machinery, tracking which time windows are used.

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    # For each (x, y) store bitmask of possible z
    edge = [[0] * k for _ in range(k)]
    for _ in range(t):
        x, y, z = map(int, input().split())
        x -= 1
        y -= 1
        z -= 1
        edge[x][y] |= (1 << z)
    
    # DP over subsets of specialists and machinery
    # dp[mask_y] = bitmask of used z for a given mask of machinery
    # We'll iterate over specialists one by one.
    dp = [-1] * (1 << k)
    dp[0] = 0  # no machinery used, no time windows used
    
    # For each specialist, try to assign a machinery and a time
    for x in range(k):
        new_dp = [-1] * (1 << k)
        for mask_y in range(1 << k):
            if dp[mask_y] == -1:
                continue
            used_z = dp[mask_y]
            # Try to assign this specialist to some unused machinery y
            for y in range(k):
                if mask_y & (1 << y):
                    continue
                possible_z = edge[x][y]
                # Remove already used z
                avail = possible_z & ~used_z
                if avail == 0:
                    continue
                # pick the lowest available z
                z = (avail & -avail).bit_length() - 1
                new_mask_y = mask_y | (1 << y)
                new_used_z = used_z | (1 << z)
                if new_dp[new_mask_y] == -1 or bin(new_used_z).count('1') > bin(new_dp[new_mask_y]).count('1'):
                    new_dp[new_mask_y] = new_used_z
        dp = new_dp
    
    # Check if we can use all k specialists
    full_mask = (1 << k) - 1
    if dp[full_mask] == -1:
        print("NO")
        return
    
    # Reconstruct solution
    used_z_mask = dp[full_mask]
    # We need to backtrack: store choices
    # We'll re-run the DP storing parent info
    parent = {}
    dp = [-1] * (1 << k)
    dp[0] = 0
    for x in range(k):
        new_dp = [-1] * (1 << k)
        for mask_y in range(1 << k):
            if dp[mask_y] == -1:
                continue
            used_z = dp[mask_y]
            for y in range(k):
                if mask_y & (1 << y):
                    continue
                possible_z = edge[x][y]
                avail = possible_z & ~used_z
                if avail == 0:
                    continue
                z = (avail & -avail).bit_length() - 1
                new_mask_y = mask_y | (1 << y)
                new_used_z = used_z | (1 << z)
                if new_dp[new_mask_y] == -1 or bin(new_used_z).count('1') > bin(new_dp[new_mask_y]).count('1'):
                    new_dp[new_mask_y] = new_used_z
                    parent[(x, new_mask_y)] = (mask_y, y, z)
        dp = new_dp
    
    # Reconstruct
    mask_y = full_mask
    result = []
    for x in range(k-1, -1, -1):
        prev_mask_y, y, z = parent[(x, mask_y)]
        result.append((x+1, y+1, z+1))
        mask_y = prev_mask_y
    result.reverse()
    
    print("YES")
    for x, y, z in result:
        print(x, y, z)

if __name__ == "__main__":
    solve()
```
