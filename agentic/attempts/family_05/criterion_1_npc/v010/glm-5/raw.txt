import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    blueprints = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        blueprints.append((x - 1, y - 1, z - 1))  # 0-indexed
    
    # We need to find k blueprints such that:
    # - All specialists are distinct (0 to k-1)
    # - All machinery are distinct (0 to k-1)
    # - All time windows are distinct (0 to k-1)
    
    # This is a 3D matching problem. We'll use bipartite matching twice.
    # First, match specialists to (machinery, time) pairs.
    
    # Group blueprints by specialist
    spec_to_blueprints = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(blueprints):
        spec_to_blueprints[x].append((y, z, i))
    
    # For each specialist, we need to choose at most one blueprint
    # The chosen blueprints must have distinct (y, z) pairs
    
    # We'll use bipartite matching: specialists on left, (y, z) pairs on right
    # But (y, z) pairs can be many, so we'll use a different approach
    
    # Alternative: For each specialist, try to match with a unique (y, z)
    # We can use maximum bipartite matching with specialists on left
    # and machinery on right, then check if time windows can also be matched
    
    # Actually, let's use a different approach:
    # 1. First, find a matching between specialists and machinery
    # 2. Then, check if the time windows can also be matched
    
    # But this might not work because the choice of machinery affects time windows
    
    # Let's use a more direct approach: bipartite matching between specialists
    # and machinery, then check if time windows can be assigned
    
    # Build adjacency list for specialist -> machinery
    spec_to_mach = [[] for _ in range(k)]
    for x, y, z in blueprints:
        spec_to_mach[x].append(y)
    
    # Remove duplicates
    for i in range(k):
        spec_to_mach[i] = list(set(spec_to_mach[i]))
    
    # Bipartite matching: specialists to machinery
    def bpm_match(graph, n_left, n_right):
        match_r = [-1] * n_right
        def dfs(u, seen):
            for v in graph[u]:
                if not seen[v]:
                    seen[v] = True
                    if match_r[v] == -1 or dfs(match_r[v], seen):
                        match_r[v] = u
                        return True
            return False
        
        result = 0
        for u in range(n_left):
            seen = [False] * n_right
            if dfs(u, seen):
                result += 1
        return result, match_r
    
    match_count, mach_match = bpm_match(spec_to_mach, k, k)
    if match_count < k:
        print("NO")
        return
    
    # Now we have a matching between specialists and machinery
    # mach_match[m] = s means machinery m is matched to specialist s
    # We need to find blueprints that use these pairs and have distinct time windows
    
    # Build the matching pairs: (specialist, machinery)
    pairs = []
    for m in range(k):
        if mach_match[m] != -1:
            pairs.append((mach_match[m], m))
    
    # Now we need to find blueprints for each pair with distinct time windows
    # This is another bipartite matching: pairs -> time windows
    
    # For each pair (s, m), find all blueprints with that specialist and machinery
    pair_to_times = []
    pair_to_blueprint_idx = []
    for s, m in pairs:
        times = []
        indices = []
        for i, (x, y, z) in enumerate(blueprints):
            if x == s and y == m:
                times.append(z)
                indices.append(i)
        pair_to_times.append(times)
        pair_to_blueprint_idx.append(indices)
    
    # Bipartite matching: pairs -> time windows
    match_count2, time_match = bpm_match(pair_to_times, k, k)
    if match_count2 < k:
        print("NO")
        return
    
    # Extract the result
    result = []
    for t_idx in range(k):
        if time_match[t_idx] != -1:
            pair_idx = time_match[t_idx]
            s, m = pairs[pair_idx]
            blueprint_idx = pair_to_blueprint_idx[pair_idx][pair_to_times[pair_idx].index(t_idx)]
            result.append(blueprints[blueprint_idx])
    
    print("YES")
    for x, y, z in result:
        print(f"{x + 1} {y + 1} {z + 1}")

solve()
