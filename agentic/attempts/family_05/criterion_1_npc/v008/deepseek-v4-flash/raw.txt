We can model this as a 3D matching problem, but since k ≤ 22, we can use a reduction to bipartite matching with bitmask DP.

We can fix the specialists and match machines to time slots using a maximum matching approach, but we need to handle the 3D constraint. A common trick is to iterate over all subsets of specialists and use DP with bitmasks for machines and time slots.

Here’s the implementation:

```python
import sys
from sys import stdin
input = stdin.readline

k, t = map(int, input().split())
allowed = [[] for _ in range(k)]
for _ in range(t):
    x, y, z = map(int, input().split())
    allowed[x-1].append((y-1, z-1))

# dp[mask] = bitmask of time slots used for given machine mask
# We'll do DP over specialists: for each specialist, try all (machine, time) pairs
# But k ≤ 22, so we can do DP over machine mask and time mask.

# We'll use DP: dp[mask_m] = mask_t achievable
dp = [-1] * (1 << k)
dp[0] = 0

# For each specialist, we update
for i in range(k):
    new_dp = [-1] * (1 << k)
    for mask_m in range(1 << k):
        if dp[mask_m] == -1:
            continue
        mask_t = dp[mask_m]
        # Try all allowed combos for specialist i
        for y, z in allowed[i]:
            if (mask_m >> y) & 1:
                continue
            if (mask_t >> z) & 1:
                continue
            new_mask_m = mask_m | (1 << y)
            new_mask_t = mask_t | (1 << z)
            new_dp[new_mask_m] = new_mask_t
    dp = new_dp

# Check if full mask is possible
full_mask = (1 << k) - 1
if dp[full_mask] == -1:
    print("NO")
else:
    print("YES")
    # Reconstruct
    # We need to store choices
    # We'll re-run DP storing parent info
    parent = {}
    dp = [-1] * (1 << k)
    dp[0] = 0
    for i in range(k):
        new_dp = [-1] * (1 << k)
        for mask_m in range(1 << k):
            if dp[mask_m] == -1:
                continue
            mask_t = dp[mask_m]
            for y, z in allowed[i]:
                if (mask_m >> y) & 1:
                    continue
                if (mask_t >> z) & 1:
                    continue
                new_mask_m = mask_m | (1 << y)
                new_mask_t = mask_t | (1 << z)
                if new_dp[new_mask_m] == -1:
                    new_dp[new_mask_m] = new_mask_t
                    parent[(i, new_mask_m)] = (mask_m, y, z)
        dp = new_dp

    # Reconstruct from full
    mask_m = full_mask
    result = []
    for i in range(k-1, -1, -1):
        prev_mask_m, y, z = parent[(i, mask_m)]
        result.append((i+1, y+1, z+1))
        mask_m = prev_mask_m
    for x, y, z in reversed(result):
        print(x, y, z)
```
