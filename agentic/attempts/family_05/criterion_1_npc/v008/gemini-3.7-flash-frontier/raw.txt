import sys

# Increase recursion depth for safety
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    if t < k:
        print("NO")
        return

    triples_x = [[] for _ in range(k)]
    triples_y = [[] for _ in range(k)]
    triples_z = [[] for _ in range(k)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3
        
        triples_x[x].append((y, z))
        triples_y[y].append((x, z))
        triples_z[z].append((x, y))

    # Pre-check if any item has 0 triples
    for i in range(k):
        if not triples_x[i] or not triples_y[i] or not triples_z[i]:
            print("NO")
            return

    chosen = []

    def backtrack(mask_x, mask_y, mask_z, count):
        if count == k:
            return True

        best_type = -1
        best_item = -1
        min_branches = 1000000
        best_candidates = None

        # Check X
        for x in range(k):
            if not (mask_x & (1 << x)):
                valid = []
                for y, z in triples_x[x]:
                    if not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                        valid.append((x, y, z))
                c = len(valid)
                if c == 0:
                    return False
                if c < min_branches:
                    min_branches = c
                    best_candidates = valid
                    best_type = 0
                    best_item = x
                    if c == 1:
                        break

        if min_branches > 1:
            # Check Y
            for y in range(k):
                if not (mask_y & (1 << y)):
                    valid = []
                    for x, z in triples_y[y]:
                        if not (mask_x & (1 << x)) and not (mask_z & (1 << z)):
                            valid.append((x, y, z))
                    c = len(valid)
                    if c == 0:
                        return False
                    if c < min_branches:
                        min_branches = c
                        best_candidates = valid
                        best_type = 1
                        best_item = y
                        if c == 1:
                            break

        if min_branches > 1:
            # Check Z
            for z in range(k):
                if not (mask_z & (1 << z)):
                    valid = []
                    for x, y in triples_z[z]:
                        if not (mask_x & (1 << x)) and not (mask_y & (1 << y)):
                            valid.append((x, y, z))
                    c = len(valid)
                    if c == 0:
                        return False
                    if c < min_branches:
                        min_branches = c
                        best_candidates = valid
                        best_type = 2
                        best_item = z
                        if c == 1:
                            break

        for x, y, z in best_candidates:
            if not (mask_x & (1 << x)) and not (mask_y & (1 << y)) and not (mask_z & (1 << z)):
                chosen.append((x + 1, y + 1, z + 1))
                if backtrack(mask_x | (1 << x), mask_y | (1 << y), mask_z | (1 << z), count + 1):
                    return True
                chosen.pop()

        return False

    if backtrack(0, 0, 0, 0):
        print("YES")
        out = []
        for x, y, z in chosen:
            out.append(f"{x} {y} {z}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
