import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    if t < k:
        print("NO")
        return

    xs = [0] * t
    ys = [0] * t
    zs = [0] * t

    col_x = [0] * k
    col_y = [0] * k
    col_z = [0] * k

    pair_xy = [0] * (k * k)
    pair_xz = [0] * (k * k)
    pair_yz = [0] * (k * k)

    pos = 2
    for i in range(t):
        x = data[pos] - 1
        y = data[pos + 1] - 1
        z = data[pos + 2] - 1
        pos += 3

        xs[i] = x
        ys[i] = y
        zs[i] = z
        bit = 1 << i

        col_x[x] |= bit
        col_y[y] |= bit
        col_z[z] |= bit
        pair_xy[x * k + y] |= bit
        pair_xz[x * k + z] |= bit
        pair_yz[y * k + z] |= bit

    for columns in (col_x, col_y, col_z):
        if any(v == 0 for v in columns):
            print("NO")
            return

    conflicts = [
        col_x[xs[i]] | col_y[ys[i]] | col_z[zs[i]]
        for i in range(t)
    ]

    all_members = (1 << k) - 1
    initial_active = (1 << t) - 1
    selected = []
    failed = set()

    sys.setrecursionlimit(10000)

    def has_perfect_matching(left_mask, right_mask, pair_rows, active):
        adjacency = []
        lm = left_mask

        while lm:
            lb = lm & -lm
            u = lb.bit_length() - 1
            lm ^= lb

            neighbors = 0
            rm = right_mask
            base = u * k
            while rm:
                rb = rm & -rm
                v = rb.bit_length() - 1
                rm ^= rb
                if pair_rows[base + v] & active:
                    neighbors |= rb

            if neighbors == 0:
                return False
            adjacency.append((neighbors.bit_count(), neighbors))

        adjacency.sort()
        match = [-1] * k

        def augment(index, seen):
            candidates = adjacency[index][1] & ~seen
            while candidates:
                bit = candidates & -candidates
                candidates ^= bit
                v = bit.bit_length() - 1
                previous = match[v]
                if previous == -1 or augment(previous, seen | bit):
                    match[v] = index
                    return True
            return False

        for i in range(len(adjacency)):
            if not augment(i, 0):
                return False
        return True

    def pairwise_feasible(mx, my, mz, active):
        ux = all_members ^ mx
        uy = all_members ^ my
        uz = all_members ^ mz

        return (
            has_perfect_matching(ux, uy, pair_xy, active)
            and has_perfect_matching(ux, uz, pair_xz, active)
            and has_perfect_matching(uy, uz, pair_yz, active)
        )

    def dfs(depth, mx, my, mz, active):
        if depth == k:
            return True

        key = mx | (my << k) | (mz << (2 * k))
        if key in failed:
            return False

        best_count = t + 1
        candidates = 0

        for i in range(k):
            if not (mx >> i) & 1:
                available = active & col_x[i]
                count = available.bit_count()
                if count == 0:
                    failed.add(key)
                    return False
                if count < best_count:
                    best_count = count
                    candidates = available

        for i in range(k):
            if not (my >> i) & 1:
                available = active & col_y[i]
                count = available.bit_count()
                if count == 0:
                    failed.add(key)
                    return False
                if count < best_count:
                    best_count = count
                    candidates = available

        for i in range(k):
            if not (mz >> i) & 1:
                available = active & col_z[i]
                count = available.bit_count()
                if count == 0:
                    failed.add(key)
                    return False
                if count < best_count:
                    best_count = count
                    candidates = available

        if best_count > 1 and (depth <= 1 or depth % 4 == 0):
            if not pairwise_feasible(mx, my, mz, active):
                failed.add(key)
                return False

        rows = []
        bits = candidates
        while bits:
            bit = bits & -bits
            bits ^= bit
            rows.append(bit.bit_length() - 1)

        if len(rows) <= 64:
            rows.sort(key=lambda r: (active & conflicts[r]).bit_count())

        for r in rows:
            x = xs[r]
            y = ys[r]
            z = zs[r]

            selected.append(r)
            if dfs(
                depth + 1,
                mx | (1 << x),
                my | (1 << y),
                mz | (1 << z),
                active & ~conflicts[r]
            ):
                return True
            selected.pop()

        failed.add(key)
        return False

    if dfs(0, 0, 0, 0, initial_active):
        output = ["YES"]
        output.extend(
            f"{xs[r] + 1} {ys[r] + 1} {zs[r] + 1}"
            for r in selected
        )
        print("\n".join(output))
    else:
        print("NO")

if __name__ == "__main__":
    main()
