Looking at this problem, I need to find k progressions (x, y, z) such that each recipe 1..k appears exactly once in each of the three phases.

This is a 3-dimensional matching problem. With k ≤ 22, I can use a meet-in-the-middle approach combined with DP over subsets.

**Approach:**
1. Group progressions by their (x, y) pair. For each pair, collect all valid z values.
2. Use DP where state is (mask_x, mask_y) representing which x and y values are used. The value is the mask_z of used z values.
3. For each (x, y) pair, if both x and y are unused, try each z that's unused, and transition to a new state.
4. To keep states manageable, prune dominated states (same mask_x, mask_y but mask_z is subset).

The key insight: if we process (x,y) pairs in a fixed order, the number of set bits in mask_x and mask_y grows predictably, limiting state explosion.

```python
import sys
from collections import defaultdict

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    k = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    
    pair_to_zs = defaultdict(list)
    for _ in range(t):
        x = int(input_data[idx]) - 1; idx += 1
        y = int(input_data[idx]) - 1; idx += 1
        z = int(input_data[idx]) - 1; idx += 1
        pair_to_zs[(x, y)].append(z)
    
    pairs = sorted(pair_to_zs.keys())
    pair_zs = [pair_to_zs[p] for p in pairs]
    n_pairs = len(pairs)
    
    full = (1 << k) - 1
    
    # DP: state (mx, my) -> mz, with parent tracking
    # Use dict for sparse representation
    dp = {(0, 0): (0, None, None, None)}  # (mz, prev_mx, prev_my, choice)
    
    for pi in range(n_pairs):
        x, y = pairs[pi]
        z_list = pair_zs[pi]
        bx, by = 1 << x, 1 << y
        
        new_dp = {}
        for (mx, my), (mz, _, _, _) in dp.items():
            # Option 1: skip this pair
            key = (mx, my)
            if key not in new_dp or mz < new_dp[key][0]:
                new_dp[key] = (mz, mx, my, -1)
            
            # Option 2: use this pair with some z
            if mx & bx or my & by:
                continue
            for z in z_list:
                bz = 1 << z
                if mz & bz:
                    continue
                nmx = mx | bx
                nmy = my | by
                nmz = mz | bz
                key = (nmx, nmy)
                if key not in new_dp or nmz < new_dp[key][0]:
                    new_dp[key] = (nmz, mx, my, z)
        
        # Prune: for same (mx, my), keep only min mz
        # (already done by keeping min)
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    # Find target state (full, full) with mz == full
    target_key = (full, full)
    if target_key not in dp:
        print("NO")
        return
    
    mz = dp[target_key][0]
    if mz != full:
        print("NO")
        return
    
    # Backtrack
    result = []
    cur = target_key
    for pi in range(n_pairs - 1, -1, -1):
        x, y = pairs[pi]
        mz, pmx, pmy, z = dp[cur]
        if z != -1:
            result.append((x + 1, y + 1, z + 1))
        cur = (pmx, pmy)
    
    print("YES")
    for r in result:
        print(*r)

solve()
```

Wait, there's an issue with backtracking - the parent pointers stored in new_dp point to states in the previous dp, but after pruning the previous dp might not have those exact states. Let me fix this by storing the full parent state properly.

Actually, the parent (pmx, pmy) should exist in the previous dp since we only transition from valid states. But the issue is that `dp` gets replaced each iteration, so I need to keep all intermediate dps or store enough info.

Let me reconsider - I'll store parent info that allows reconstruction:

```python
import sys
from collections import defaultdict

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    k = int(input_data[idx]); idx += 1
    t = int(input_data[idx]); idx += 1
    
    pair_to_zs = defaultdict(list)
    for _ in range(t):
        x = int(input_data[idx]) - 1; idx += 1
        y = int(input_data[idx]) - 1; idx += 1
        z = int(input_data[idx]) - 1; idx += 1
        pair_to_zs[(x, y)].append(z)
    
    pairs = sorted(pair_to_zs.keys())
    pair_zs = [pair_to_zs[p] for p in pairs]
    n_pairs = len(pairs)
    
    full = (1 << k) - 1
    
    # dp[(mx,my)] = (mz, parent_key, z_choice)
    dp = {(0, 0): (0, None, -1)}
    all_dps = [dp]
    
    for pi in range(n_pairs):
        x, y = pairs[pi]
        z_list = pair_zs[pi]
        bx, by = 1 << x, 1 << y
        
        new_dp = {}
        for (mx, my), (mz, _, _) in dp.items():
            # Skip
            key = (mx, my)
            if key not in new_dp or mz < new_dp[key][0]:
                new_dp[key] = (mz, (mx, my), -1)
            
            # Use
            if mx & bx or my & by:
                continue
            for z in z_list:
                bz = 1 << z
                if mz & bz:
                    continue
                nmx = mx | bx
                nmy = my | by
                nmz = mz | bz
                key = (nmx, nmy)
                if key not in new_dp or nmz < new_dp[key][0]:
                    new_dp[key] = (nmz, (mx, my), z)
        
        dp = new_dp
        all_dps.append(dp)
        if not dp:
            print("NO")
            return
    
    target_key = (full, full)
    if target_key not in dp or dp[target_key][0] != full:
        print("NO")
        return
    
    result = []
    cur = target_key
    for pi in range(n_pairs, 0, -1):
        mz, parent_key, z = all_dps[pi][cur]
        if z != -1:
            x, y = pairs[pi - 1]
            result.append((x + 1, y + 1, z + 1))
        cur = parent_key
    
    print("YES")
    for r in result:
        print(*r)

solve()
```

This should work correctly. The state space is bounded because at step i, mask_x and mask_y have at most i bits set combined, and pruning keeps only the best mz for each (mx, my) pair.
