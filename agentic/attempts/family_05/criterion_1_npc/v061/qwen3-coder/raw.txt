from collections import defaultdict
import sys

def solve():
    k, t = map(int, input().split())
    
    # Read all transitions
    transitions = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        transitions.append((x, y, z))
    
    # Try all possible subsets of k transitions
    # Since k <= 22 and t <= k^3, t can be up to 10648, but we only need to choose k=22 at most
    # So we can use backtracking or bit manipulation for small k
    
    # For efficiency, we'll model this as a matching problem
    # But given the constraints, we can try backtracking
    
    # We'll use backtracking to find a set of k transitions that covers each recipe exactly once
    def backtrack(index, used_p1, used_p2, used_p3, selected):
        if len(selected) == k:
            # Check if all recipes are used exactly once
            return len(used_p1) == k and len(used_p2) == k and len(used_p3) == k, selected[:]
        
        if index >= t:
            return False, []
        
        # Try not taking current transition
        res, ans = backtrack(index + 1, used_p1, used_p2, used_p3, selected)
        if res:
            return res, ans
        
        # Try taking current transition
        x, y, z = transitions[index]
        if x not in used_p1 and y not in used_p2 and z not in used_p3:
            used_p1.add(x)
            used_p2.add(y)
            used_p3.add(z)
            selected.append(index)
            
            res, ans = backtrack(index + 1, used_p1, used_p2, used_p3, selected)
            if res:
                return res, ans
                
            selected.pop()
            used_p1.remove(x)
            used_p2.remove(y)
            used_p3.remove(z)
        
        return False, []
    
    # Optimization: if t < k, impossible
    if t < k:
        print("NO")
        return
    
    found, indices = backtrack(0, set(), set(), set(), [])
    
    if found:
        print("YES")
        for idx in indices:
            x, y, z = transitions[idx]
            print(x, y, z)
    else:
        print("NO")

solve()
