import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # Build adjacency: for each specialist x, list of (y, z) pairs
    adj = [[] for _ in range(k + 1)]
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        adj[x].append((y, z))
    
    # match_y[y] = x that is matched to y
    match_y = [0] * (k + 1)
    # match_z[z] = x that is matched to z
    match_z = [0] * (k + 1)
    # result[x] = (y, z) for specialist x
    result = [None] * (k + 1)
    
    def dfs(x, visited_y, visited_z):
        for y, z in adj[x]:
            if visited_y[y] or visited_z[z]:
                continue
            visited_y[y] = True
            visited_z[z] = True
            
            my = match_y[y]
            mz = match_z[z]
            
            if my == 0 and mz == 0:
                match_y[y] = x
                match_z[z] = x
                result[x] = (y, z)
                return True
            
            # Case 1: y is matched, z is free (mz == 0)
            if my != 0 and mz == 0:
                # Try to reassign my to another (y', z')
                # Temporarily unmatch
                match_y[y] = 0
                result[my] = None
                match_y[y] = x
                match_z[z] = x
                result[x] = (y, z)
                # Recursively find new match for my
                if dfs(my, visited_y, visited_z):
                    return True
                # Backtrack
                match_y[y] = my
                match_z[z] = 0
                result[x] = None
                result[my] = (y, match_z[y]) if match_z[y] == my else None
                continue
            
            # Case 2: y is free (my == 0), z is matched
            if my == 0 and mz != 0:
                match_z[z] = 0
                result[mz] = None
                match_y[y] = x
                match_z[z] = x
                result[x] = (y, z)
                if dfs(mz, visited_y, visited_z):
                    return True
                # Backtrack
                match_z[z] = mz
                match_y[y] = 0
                result[x] = None
                result[mz] = (match_y[z], z) if match_y[z] == mz else None
                continue
            
            # Case 3: both y and z are matched (to my and mz)
            # my == mz must hold for valid state
            if my != 0 and mz != 0:
                # Temporarily unmatch both from my
                match_y[y] = 0
                match_z[z] = 0
                result[my] = None
                
                match_y[y] = x
                match_z[z] = x
                result[x] = (y, z)
                
                if dfs(my, visited_y, visited_z):
                    return True
                
                # Backtrack
                match_y[y] = my
                match_z[z] = mz
                result[x] = None
                result[my] = (y, z)
                continue
        
        return False
    
    for x in range(1, k + 1):
        visited_y = [False] * (k + 1)
        visited_z = [False] * (k + 1)
        if not dfs(x, visited_y, visited_z):
            print("NO")
            return
    
    print("YES")
    for x in range(1, k + 1):
        y, z = result[x]
        print(x, y, z)

solve()
