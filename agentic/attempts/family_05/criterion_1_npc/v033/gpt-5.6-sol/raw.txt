import sys

sys.setrecursionlimit(1_000_000)


def gf2_consistent(k, triples):
    basis = [0] * (3 * k)
    for x, y, z in triples:
        v = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
        while v:
            p = v.bit_length() - 1
            if basis[p]:
                v ^= basis[p]
            else:
                basis[p] = v
                break

    target = (1 << (3 * k)) - 1
    while target:
        p = target.bit_length() - 1
        if not basis[p]:
            return False
        target ^= basis[p]
    return True


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    triples = []
    pos = 2
    for _ in range(t):
        x, y, z = data[pos] - 1, data[pos + 1] - 1, data[pos + 2] - 1
        pos += 3
        triples.append((x, y, z))

    if t < k or not gf2_consistent(k, triples):
        print("NO")
        return

    ncols = 3 * k
    col_rows = [0] * ncols
    row_conflicts = [0] * t

    pair_xy = [0] * (k * k)
    pair_xz = [0] * (k * k)
    pair_yz = [0] * (k * k)

    for i, (x, y, z) in enumerate(triples):
        bit = 1 << i
        col_rows[x] |= bit
        col_rows[k + y] |= bit
        col_rows[2 * k + z] |= bit
        pair_xy[x * k + y] |= bit
        pair_xz[x * k + z] |= bit
        pair_yz[y * k + z] |= bit

    if any(v == 0 for v in col_rows):
        print("NO")
        return

    for i, (x, y, z) in enumerate(triples):
        row_conflicts[i] = col_rows[x] | col_rows[k + y] | col_rows[2 * k + z]

    full_part = (1 << k) - 1
    full_cols = (1 << ncols) - 1
    all_rows = (1 << t) - 1

    failed = set()
    answer = []
    nodes = 0

    def has_matching(left_mask, right_mask, pair_rows, active):
        match = [-1] * k
        left_vertices = []
        lm = left_mask
        while lm:
            b = lm & -lm
            left_vertices.append(b.bit_length() - 1)
            lm ^= b

        adjacency = {}
        for u in left_vertices:
            nbr = 0
            rm = right_mask
            base = u * k
            while rm:
                b = rm & -rm
                v = b.bit_length() - 1
                if pair_rows[base + v] & active:
                    nbr |= b
                rm ^= b
            if nbr == 0:
                return False
            adjacency[u] = nbr

        left_vertices.sort(key=lambda u: adjacency[u].bit_count())

        def augment(u, seen):
            nbr = adjacency[u] & ~seen
            while nbr:
                b = nbr & -nbr
                nbr ^= b
                v = b.bit_length() - 1
                owner = match[v]
                if owner == -1 or augment(owner, seen | b):
                    match[v] = u
                    return True
            return False

        for u in left_vertices:
            if not augment(u, 0):
                return False
        return True

    def hall_check(cols, active):
        xm = cols & full_part
        ym = (cols >> k) & full_part
        zm = (cols >> (2 * k)) & full_part
        return (
            has_matching(xm, ym, pair_xy, active)
            and has_matching(xm, zm, pair_xz, active)
            and has_matching(ym, zm, pair_yz, active)
        )

    def dfs(cols, active, remaining):
        nonlocal nodes
        nodes += 1

        if cols == 0:
            return True
        if cols in failed:
            return False

        best_candidates = 0
        best_count = t + 1

        cm = cols
        while cm:
            cb = cm & -cm
            c = cb.bit_length() - 1
            candidates = col_rows[c] & active
            count = candidates.bit_count()
            if count == 0:
                failed.add(cols)
                return False
            if count < best_count:
                best_count = count
                best_candidates = candidates
                if count == 1:
                    break
            cm ^= cb

        depth = k - remaining
        if remaining <= 10 or depth % 3 == 0:
            if not hall_check(cols, active):
                failed.add(cols)
                return False

        candidates_list = []
        bits = best_candidates
        while bits:
            rb = bits & -bits
            bits ^= rb
            r = rb.bit_length() - 1
            score = (row_conflicts[r] & active).bit_count()
            candidates_list.append((score, r))

        candidates_list.sort()

        for _, r in candidates_list:
            x, y, z = triples[r]
            used_cols = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
            if (cols & used_cols) != used_cols:
                continue

            answer.append(r)
            if dfs(cols ^ used_cols, active & ~row_conflicts[r], remaining - 1):
                return True
            answer.pop()

        if len(failed) < 1_500_000:
            failed.add(cols)
        return False

    if dfs(full_cols, all_rows, k):
        out = ["YES"]
        for r in answer:
            x, y, z = triples[r]
            out.append(f"{x + 1} {y + 1} {z + 1}")
        sys.stdout.write("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    main()
