We can model this as a 3D matching problem, but since k ≤ 22, we can use a backtracking approach with pruning, or reduce it to a bipartite matching after fixing one dimension.

A practical approach:  
- Fix the data feed order.  
- For each feed, try to match a core and memory bank that are still free.  
- Use recursion with pruning and early exit.

Here’s the code:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    # Store pathways per feed
    pathways = [[] for _ in range(k+1)]
    for _ in range(t):
        f = int(next(it))
        c = int(next(it))
        m = int(next(it))
        pathways[f].append((c, m))
    
    # Sort each list for deterministic order
    for i in range(1, k+1):
        pathways[i].sort()
    
    used_core = [False] * (k+1)
    used_mem = [False] * (k+1)
    result = []
    
    def dfs(feed):
        if feed > k:
            return True
        for c, m in pathways[feed]:
            if not used_core[c] and not used_mem[m]:
                used_core[c] = True
                used_mem[m] = True
                result.append((feed, c, m))
                if dfs(feed+1):
                    return True
                result.pop()
                used_core[c] = False
                used_mem[m] = False
        return False
    
    if dfs(1):
        print("YES")
        for f, c, m in result:
            print(f, c, m)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
