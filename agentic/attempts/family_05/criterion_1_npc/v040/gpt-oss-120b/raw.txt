import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))

    # columns: 0..k-1 feeds, k..2k-1 cores, 2k..3k-1 memories
    n_cols = 3 * k

    # DLX structures
    max_nodes = t * 3 + n_cols + 1
    U = [0] * max_nodes
    D = [0] * max_nodes
    L = [0] * max_nodes
    R = [0] * max_nodes
    C = [0] * max_nodes  # column header for each node
    S = [0] * (n_cols + 1)  # size of each column, header 0 is root

    # header 0 is root
    for i in range(n_cols + 1):
        U[i] = D[i] = i
        L[i] = i - 1
        R[i] = i + 1
    L[0] = n_cols
    R[n_cols] = 0

    row_head = []  # store first node index of each row
    node_cnt = n_cols + 1

    for _ in range(t):
        f = int(next(it)) - 1
        c = int(next(it)) - 1
        m = int(next(it)) - 1
        cols = [f, k + c, 2 * k + m]
        first = -1
        prev = -1
        for col in cols:
            col += 1  # shift because column headers are 1..n_cols
            idx = node_cnt
            node_cnt += 1
            C[idx] = col
            # vertical links
            U[idx] = U[col]
            D[idx] = col
            D[U[col]] = idx
            U[col] = idx
            S[col] += 1
            # horizontal links
            if first == -1:
                first = idx
                L[idx] = R[idx] = idx
            else:
                L[idx] = prev
                R[idx] = first
                R[prev] = idx
                L[first] = idx
            prev = idx
        row_head.append(first)

    solution = []
    ans_rows = []

    def cover(col):
        # remove column header
        L[R[col]] = L[col]
        R[L[col]] = R[col]
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[col]] = col
        R[L[col]] = col

    def search(k_depth):
        if R[0] == 0:  # all columns covered
            return True
        # choose column with minimal size
        c = R[0]
        min_s = S[c]
        j = R[c]
        while j != 0:
            if S[j] < min_s:
                min_s = S[j]
                c = j
                if min_s == 0:
                    break
            j = R[j]
        if min_s == 0:
            return False
        cover(c)
        r = D[c]
        while r != c:
            solution.append(r)
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            if search(k_depth + 1):
                return True
            # backtrack
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution.pop()
            r = D[r]
        uncover(c)
        return False

    if search(0):
        sys.stdout.write("YES\n")
        for node in solution:
            # retrieve the row index from node
            # we can map node to its row by scanning row_head
            # faster: store row id in an extra array when creating nodes
            pass
    else:
        sys.stdout.write("NO\n")
        return

    # Build mapping from node to row index
    node_to_row = {}
    for rid, head in enumerate(row_head):
        i = head
        while True:
            node_to_row[i] = rid
            i = R[i]
            if i == head:
                break

    used_rows = set()
    out_lines = []
    for node in solution:
        rid = node_to_row[node]
        if rid in used_rows:
            continue
        used_rows.add(rid)
        f, c, m = None, None, None
        i = row_head[rid]
        while True:
            col = C[i] - 1
            if col < k:
                f = col + 1
            elif col < 2 * k:
                c = col - k + 1
            else:
                m = col - 2 * k + 1
            i = R[i]
            if i == row_head[rid]:
                break
        out_lines.append(f"{f} {c} {m}")
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
