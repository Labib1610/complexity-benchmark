from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    pathways = []
    index = 2
    for _ in range(t):
        f = int(data[index])
        c = int(data[index+1])
        m = int(data[index+2])
        pathways.append((f, c, m))
        index += 3
    
    # Create sets for quick lookup
    pathway_set = set(pathways)
    
    # Try all permutations of data feeds
    for perm in permutations(range(1, k+1)):
        # For each permutation of data feeds, try to assign processing cores and memory banks
        used_cores = [False] * (k+1)
        used_mems = [False] * (k+1)
        selected = []
        valid = True
        
        for i in range(k):
            f = perm[i]
            found = False
            # Try to find a valid (c, m) pair for this f
            for c in range(1, k+1):
                if used_cores[c]:
                    continue
                for m in range(1, k+1):
                    if used_mems[m]:
                        continue
                    if (f, c, m) in pathway_set:
                        selected.append((f, c, m))
                        used_cores[c] = True
                        used_mems[m] = True
                        found = True
                        break
                if found:
                    break
            if not found:
                valid = False
                break
                
        if valid:
            print("YES")
            for pathway in selected:
                print(pathway[0], pathway[1], pathway[2])
            return
            
    print("NO")

solve()
