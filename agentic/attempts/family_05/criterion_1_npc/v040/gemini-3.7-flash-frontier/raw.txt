import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        f = int(input_data[idx]) - 1
        c = int(input_data[idx + 1]) - 1
        m = int(input_data[idx + 2]) - 1
        idx += 3
        triples.append((f, c, m))

    num_cols = 3 * k

    MAX_NODES = num_cols + 1 + 3 * len(triples) + 10
    L = [0] * MAX_NODES
    R = [0] * MAX_NODES
    U = [0] * MAX_NODES
    D = [0] * MAX_NODES
    C = [0] * MAX_NODES
    ROW = [0] * MAX_NODES
    S = [0] * (num_cols + 1)

    for i in range(num_cols + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
    L[0] = num_cols
    R[num_cols] = 0

    node_cnt = num_cols + 1

    for row_idx, (f, c, m) in enumerate(triples):
        cols = [f + 1, k + c + 1, 2 * k + m + 1]
        first_node = -1
        for col in cols:
            curr = node_cnt
            node_cnt += 1
            C[curr] = col
            ROW[curr] = row_idx
            S[col] += 1

            U[curr] = U[col]
            D[curr] = col
            D[U[col]] = curr
            U[col] = curr

            if first_node == -1:
                first_node = curr
                L[curr] = curr
                R[curr] = curr
            else:
                L[curr] = L[first_node]
                R[curr] = first_node
                R[L[first_node]] = curr
                L[first_node] = curr

    def cover(col):
        L[R[col]] = L[col]
        R[L[col]] = R[col]
        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[col]] = col
        R[L[col]] = col

    ans = []

    def search():
        if R[0] == 0:
            return True

        c = R[0]
        min_s = S[c]
        curr = R[0]
        while curr != 0:
            if S[curr] < min_s:
                min_s = S[curr]
                c = curr
            curr = R[curr]

        if min_s == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            ans.append(ROW[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            ans.pop()
            r = D[r]

        uncover(c)
        return False

    if search():
        print("YES")
        for row_idx in ans:
            f, c, m = triples[row_idx]
            print(f"{f + 1} {c + 1} {m + 1}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
