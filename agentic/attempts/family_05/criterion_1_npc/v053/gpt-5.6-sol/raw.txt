import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    xs = [0] * t
    ys = [0] * t
    zs = [0] * t

    column_rows = [0] * (3 * k)

    xy_support = [0] * (k * k)  # possible z values
    xz_support = [0] * (k * k)  # possible y values
    yz_support = [0] * (k * k)  # possible x values

    p = 2
    for i in range(t):
        x = data[p] - 1
        y = data[p + 1] - 1
        z = data[p + 2] - 1
        p += 3

        xs[i] = x
        ys[i] = y
        zs[i] = z

        bit = 1 << i
        column_rows[x] |= bit
        column_rows[k + y] |= bit
        column_rows[2 * k + z] |= bit

        xy_support[x * k + y] |= 1 << z
        xz_support[x * k + z] |= 1 << y
        yz_support[y * k + z] |= 1 << x

    for rows in column_rows:
        if rows == 0:
            print("NO")
            return

    conflicts = [0] * t
    for i in range(t):
        conflicts[i] = (
            column_rows[xs[i]]
            | column_rows[k + ys[i]]
            | column_rows[2 * k + zs[i]]
        )

    full_values = (1 << k) - 1
    full_rows = (1 << t) - 1

    def has_perfect_projection(rem_a, rem_b, rem_c, support):
        adjacency = []
        a_bits = rem_a

        while a_bits:
            a_bit = a_bits & -a_bits
            a_bits -= a_bit
            a = a_bit.bit_length() - 1

            choices = 0
            b_bits = rem_b
            base = a * k
            while b_bits:
                b_bit = b_bits & -b_bits
                b_bits -= b_bit
                b = b_bit.bit_length() - 1
                if support[base + b] & rem_c:
                    choices |= b_bit

            if choices == 0:
                return False
            adjacency.append(choices)

        adjacency.sort(key=int.bit_count)
        matched_to = [-1] * k
        seen = [0] * k
        token = 0

        def augment(left_index):
            nonlocal token
            bits = adjacency[left_index]
            while bits:
                bit = bits & -bits
                bits -= bit
                b = bit.bit_length() - 1
                if seen[b] == token:
                    continue
                seen[b] = token
                other = matched_to[b]
                if other == -1 or augment(other):
                    matched_to[b] = left_index
                    return True
            return False

        for i in range(len(adjacency)):
            token += 1
            if not augment(i):
                return False
        return True

    failed = set()
    memo_limit = 700000
    answer = []

    sys.setrecursionlimit(100000)

    def search(mx, my, mz, active, depth):
        if depth == k:
            return True

        key = mx | (my << k) | (mz << (2 * k))
        if key in failed:
            return False

        rem_x = full_values ^ mx
        rem_y = full_values ^ my
        rem_z = full_values ^ mz

        counts = [0] * (3 * k)
        best_rows = 0
        best_count = t + 1

        bits = rem_x
        while bits:
            bit = bits & -bits
            bits -= bit
            c = bit.bit_length() - 1
            rows = column_rows[c] & active
            count = rows.bit_count()
            if count == 0:
                if len(failed) < memo_limit:
                    failed.add(key)
                return False
            counts[c] = count
            if count < best_count:
                best_count = count
                best_rows = rows

        bits = rem_y
        while bits:
            bit = bits & -bits
            bits -= bit
            y = bit.bit_length() - 1
            c = k + y
            rows = column_rows[c] & active
            count = rows.bit_count()
            if count == 0:
                if len(failed) < memo_limit:
                    failed.add(key)
                return False
            counts[c] = count
            if count < best_count:
                best_count = count
                best_rows = rows

        bits = rem_z
        while bits:
            bit = bits & -bits
            bits -= bit
            z = bit.bit_length() - 1
            c = 2 * k + z
            rows = column_rows[c] & active
            count = rows.bit_count()
            if count == 0:
                if len(failed) < memo_limit:
                    failed.add(key)
                return False
            counts[c] = count
            if count < best_count:
                best_count = count
                best_rows = rows

        if depth % 4 == 0:
            if not has_perfect_projection(rem_x, rem_y, rem_z, xy_support):
                if len(failed) < memo_limit:
                    failed.add(key)
                return False
            if not has_perfect_projection(rem_x, rem_z, rem_y, xz_support):
                if len(failed) < memo_limit:
                    failed.add(key)
                return False
            if not has_perfect_projection(rem_y, rem_z, rem_x, yz_support):
                if len(failed) < memo_limit:
                    failed.add(key)
                return False

        candidates = []
        rows = best_rows
        while rows:
            bit = rows & -rows
            rows -= bit
            row = bit.bit_length() - 1
            score = (
                counts[xs[row]]
                + counts[k + ys[row]]
                + counts[2 * k + zs[row]]
            )
            candidates.append((score, row))

        candidates.sort()

        for _, row in candidates:
            bx = 1 << xs[row]
            by = 1 << ys[row]
            bz = 1 << zs[row]

            answer.append(row)
            if search(
                mx | bx,
                my | by,
                mz | bz,
                active & ~conflicts[row],
                depth + 1
            ):
                return True
            answer.pop()

        if len(failed) < memo_limit:
            failed.add(key)
        return False

    if search(0, 0, 0, full_rows, 0):
        out = ["YES"]
        for row in answer:
            out.append(f"{xs[row] + 1} {ys[row] + 1} {zs[row] + 1}")
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    main()
