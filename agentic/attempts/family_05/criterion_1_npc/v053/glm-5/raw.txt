import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    # profiles_by_freq[f] = list of (y, z)
    profiles_by_freq = [[] for _ in range(k + 1)]
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        profiles_by_freq[x].append((y, z))
    
    # Check each frequency has at least one profile
    for f in range(1, k + 1):
        if not profiles_by_freq[f]:
            print("NO")
            return
    
    # DP with bitmask: dp[mask] = (prev_mask, y, z) or None
    # mask represents used voltages (bit y-1 set means voltage y used)
    # We process frequencies 1..k, tracking used voltages and used phases
    # State: (used_voltage_mask, used_phase_mask) -> prev_state info
    # Since k <= 22, masks fit in int. But 2^22 * 2^22 is too large.
    # We use dictionary-based DP: dp is dict mapping (vmask, pmask) to (prev_vmask, prev_pmask, y, z)
    
    dp = {}
    dp[(0, 0)] = None
    
    for f in range(1, k + 1):
        new_dp = {}
        for (vmask, pmask), _ in dp.items():
            for (y, z) in profiles_by_freq[f]:
                ybit = 1 << (y - 1)
                zbit = 1 << (z - 1)
                if (vmask & ybit) == 0 and (pmask & zbit) == 0:
                    nvmask = vmask | ybit
                    npmask = pmask | zbit
                    key = (nvmask, npmask)
                    if key not in new_dp:
                        new_dp[key] = (vmask, pmask, y, z)
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    # Find any valid final state (all bits set)
    full_mask = (1 << k) - 1
    final_key = None
    for key in dp:
        if key[0] == full_mask and key[1] == full_mask:
            final_key = key
            break
    
    if final_key is None:
        print("NO")
        return
    
    # Reconstruct
    result = []
    cur = final_key
    for f in range(k, 0, -1):
        prev_vmask, prev_pmask, y, z = dp[cur]
        result.append((f, y, z))
        cur = (prev_vmask, prev_pmask)
    
    result.reverse()
    print("YES")
    for x, y, z in result:
        print(x, y, z)

solve()
