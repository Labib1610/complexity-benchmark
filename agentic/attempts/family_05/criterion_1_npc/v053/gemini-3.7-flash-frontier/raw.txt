import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    if t < k:
        print("NO")
        return
    
    # Read triples
    # 0-indexed: x in [0, k-1], y in [0, k-1], z in [0, k-1]
    adj_x = [[] for _ in range(k)]
    adj_y = [[] for _ in range(k)]
    adj_z = [[] for _ in range(k)]
    
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        idx += 3
        adj_x[x].append((y, z))
        adj_y[y].append((x, z))
        adj_z[z].append((x, y))
        
    # Quick sanity check: every x, y, z must have at least one option
    for i in range(k):
        if not adj_x[i] or not adj_y[i] or not adj_z[i]:
            print("NO")
            return

    chosen = [None] * k  # chosen[x] = (y, z)
    
    # Backtracking with MRV and forward checking
    # used_y: bitmask of used y
    # used_z: bitmask of used z
    # used_x: bitmask of used x
    
    def backtrack(used_x, used_y, used_z, count):
        if count == k:
            return True
        
        # Choose the unused x with minimum valid options
        best_x = -1
        min_options = 1000000
        best_valid_pairs = None
        
        # Check all unused x
        for x in range(k):
            if not (used_x & (1 << x)):
                valid_pairs = []
                for y, z in adj_x[x]:
                    if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                        valid_pairs.append((y, z))
                cnt = len(valid_pairs)
                if cnt == 0:
                    return False
                if cnt < min_options:
                    min_options = cnt
                    best_x = x
                    best_valid_pairs = valid_pairs
                    if min_options == 1:
                        break
        
        # Forward check: check if any unused y or unused z has 0 options
        # To make it fast, we can check y and z degrees if needed, but min_options already prunes well.
        for y in range(k):
            if not (used_y & (1 << y)):
                has_any = False
                for x, z in adj_y[y]:
                    if not (used_x & (1 << x)) and not (used_z & (1 << z)):
                        has_any = True
                        break
                if not has_any:
                    return False
                    
        for z in range(k):
            if not (used_z & (1 << z)):
                has_any = False
                for x, y in adj_z[z]:
                    if not (used_x & (1 << x)) and not (used_y & (1 << y)):
                        has_any = True
                        break
                if not has_any:
                    return False

        # Branch on best_x
        for y, z in best_valid_pairs:
            chosen[best_x] = (y, z)
            if backtrack(used_x | (1 << best_x), used_y | (1 << y), used_z | (1 << z), count + 1):
                return True
            chosen[best_x] = None
            
        return False

    if backtrack(0, 0, 0, 0):
        print("YES")
        out = []
        for x in range(k):
            y, z = chosen[x]
            out.append(f"{x+1} {y+1} {z+1}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
