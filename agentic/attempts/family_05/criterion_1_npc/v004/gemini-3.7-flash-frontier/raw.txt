import sys

# Increase recursion depth just in case
sys.setrecursionlimit(50000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    paths = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        paths.append((x, y, z))
    
    num_cols = 3 * k
    # DLX structures
    # Columns 1..num_cols, 0 is root
    max_nodes = num_cols + 1 + 3 * t + 10
    
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    S = [0] * (num_cols + 1)
    row_id = [0] * max_nodes
    
    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = (i - 1) % (num_cols + 1)
        R[i] = (i + 1) % (num_cols + 1)
        U[i] = i
        D[i] = i
        C[i] = i
        S[i] = 0
        
    node_count = num_cols + 1
    
    for r_idx, (x, y, z) in enumerate(paths):
        # 1-based column indices:
        # specialist x: x
        # machine y: k + y
        # time slot z: 2k + z
        cols = [x, k + y, 2 * k + z]
        first_node = node_count
        
        for j, c in enumerate(cols):
            cur = node_count
            node_count += 1
            
            C[cur] = c
            row_id[cur] = r_idx
            S[c] += 1
            
            # Vertical link: insert above column header c
            U[cur] = U[c]
            D[cur] = c
            D[U[c]] = cur
            U[c] = cur
            
            # Horizontal link
            if j == 0:
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = node_count - 2
                R[cur] = first_node
                R[node_count - 2] = cur
                L[first_node] = cur

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size
        min_s = float('inf')
        c = 0
        cur = R[0]
        while cur != 0:
            if S[cur] < min_s:
                min_s = S[cur]
                c = cur
                if min_s == 0:
                    break
            cur = R[cur]
            
        if min_s == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            
            if search():
                return True
            
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution.pop()
            r = D[r]
            
        uncover(c)
        return False

    if search():
        print("YES")
        for r_idx in solution:
            x, y, z = paths[r_idx]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
