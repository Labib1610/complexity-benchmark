from collections import defaultdict, deque

def solve():
    k, t = map(int, input().split())
    
    # If no paths, impossible unless k=0 (but k>=1)
    if t == 0:
        print("NO")
        return
        
    paths = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        paths.append((x, y, z))
    
    # Build a graph for matching: specialists -> machines -> time slots
    # This is a tripartite matching problem.
    # We can model it as a bipartite matching between specialists and (machine,time) pairs
    # or use a flow network.
    
    # Let's use a flow network approach with Ford-Fulkerson (or rather, we'll model it as bipartite matching)
    # Actually, let's think of it as a 3D matching problem.
    
    # We can reduce this to a bipartite matching:
    # Left side: specialists (1..k)
    # Right side: (machine, time) pairs that appear in allowed paths
    # Edge from specialist x to (y,z) if (x,y,z) is allowed
    
    # But we want a perfect matching that covers all specialists, machines, and time slots.
    
    # Let's try a different approach: model as bipartite matching between:
    # Left: specialists (1..k)
    # Right: machines (1..k)
    # But this doesn't capture the time slot constraint.
    
    # Better approach: model as maximum flow.
    # Nodes: source, specialists, machines, time slots, sink
    # Edges:
    # source -> each specialist (capacity 1)
    # each specialist -> allowed (x,y,z) paths (capacity 1) 
    # each path connects to its machine and time slot
    # Actually, let's think of paths as nodes too.
    
    # Nodes:
    # 0: source
    # 1..k: specialists
    # k+1..2k: machines  
    # 2k+1..3k: time slots
    # 3k+1..3k+t: paths
    # 3k+t+1: sink
    
    # Edges:
    # source (0) -> each specialist x (capacity 1)
    # each specialist x -> path node p if path p uses specialist x (capacity 1)
    # each path node p -> machine y if path p uses machine y (capacity 1)
    # each path node p -> time slot z if path p uses time slot z (capacity 1)
    # each machine y -> sink (capacity 1)
    # each time slot z -> sink (capacity 1)
    
    # But this is overcomplicated. Let's think of it as a 3D matching.
    # We want to select k paths such that all x,y,z coordinates are distinct.
    
    # This is equivalent to finding a matching in a 3-uniform 3-partite hypergraph.
    # But let's reduce to bipartite matching.
    
    # Create a bipartite graph:
    # Left side: specialists 1..k
    # Right side: pairs (y,z) that appear in some allowed path
    # Edge from x to (y,z) if (x,y,z) is an allowed path
    
    # But we also need that each (y,z) pair is used at most once.
    # And we need exactly k edges.
    
    # Let's enumerate all (y,z) pairs that appear in paths
    yz_pairs = set()
    path_dict = {}  # (x,y,z) -> index
    for i, (x,y,z) in enumerate(paths):
        yz_pairs.add((y,z))
        path_dict[(x,y,z)] = i
    
    yz_list = list(yz_pairs)
    yz_index = {yz: i for i, yz in enumerate(yz_list)}
    
    # Now create bipartite graph
    # Left: specialists 0..k-1 (0-indexed)
    # Right: yz pairs 0..len(yz_list)-1
    adj = [[] for _ in range(k)]
    for x, y, z in paths:
        yz = (y, z)
        adj[x-1].append(yz_index[yz])
    
    # Now find maximum matching in this bipartite graph
    # And check if size is k and if corresponding y,z are all distinct
    
    def dfs(u, match_right, match_left, visited):
        for v in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match_right[v] == -1 or dfs(match_right[v], match_right, match_left, visited):
                match_right[v] = u
                match_left[u] = v
                return True
        return False
    
    match_right = [-1] * len(yz_list)
    match_left = [-1] * k
    
    matching = 0
    for u in range(k):
        visited = [False] * len(yz_list)
        if dfs(u, match_right, match_left, visited):
            matching += 1
    
    if matching < k:
        print("NO")
        return
    
    # Check if the y,z values in the matching are all distinct
    used_y = set()
    used_z = set()
    result_paths = []
    
    for x_idx in range(k):
        yz_idx = match_left[x_idx]
        if yz_idx == -1:
            print("NO")
            return
        y, z = yz_list[yz_idx]
        if y in used_y or z in used_z:
            print("NO")
            return
        used_y.add(y)
        used_z.add(z)
        x = x_idx + 1
        result_paths.append((x, y, z))
    
    if len(used_y) != k or len(used_z) != k:
        print("NO")
        return
        
    print("YES")
    for x, y, z in result_paths:
        print(x, y, z)

solve()
