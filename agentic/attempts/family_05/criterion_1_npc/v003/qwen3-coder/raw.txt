from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    configs = []
    index = 2
    for _ in range(t):
        p1 = int(data[index])
        p2 = int(data[index+1])
        p3 = int(data[index+2])
        configs.append((p1, p2, p3))
        index += 3
    
    # We need to choose exactly k configurations such that each value from 1 to k appears exactly once in each parameter position
    # This is equivalent to finding a perfect matching in a tripartite hypergraph
    
    # Try all combinations of k configurations from the t available
    for combo in combinations(range(t), k):
        # Check if this combination satisfies the condition
        p1_vals = set()
        p2_vals = set()
        p3_vals = set()
        
        valid = True
        selected_configs = []
        
        for i in combo:
            p1, p2, p3 = configs[i]
            if p1 in p1_vals or p2 in p2_vals or p3 in p3_vals:
                valid = False
                break
            p1_vals.add(p1)
            p2_vals.add(p2)
            p3_vals.add(p3)
            selected_configs.append((p1, p2, p3))
        
        if valid and len(p1_vals) == k and len(p2_vals) == k and len(p3_vals) == k:
            print("YES")
            for cfg in selected_configs:
                print(cfg[0], cfg[1], cfg[2])
            return
    
    print("NO")

solve()
