import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    commands = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        commands.append((x-1, y-1, z-1))  # 0-indexed
    
    # Build adjacency: for each (a,b) pair, list of c values available
    ab_to_c = {}
    for a, b, c in commands:
        ab_to_c.setdefault((a, b), []).append(c)
    
    # For each a, collect all (b,c) pairs
    a_to_bc = {}
    for a, b, c in commands:
        a_to_bc.setdefault(a, []).append((b, c))
    
    # Try to find a perfect matching using backtracking
    # match_b[b] = a matched to b, match_c[c] = a matched to c
    match_b = [-1] * k
    match_c = [-1] * k
    
    def backtrack(a):
        if a == k:
            return True
        
        # Get all (b,c) pairs for this a
        if a not in a_to_bc:
            return False
        
        bc_pairs = a_to_bc[a]
        
        for b, c in bc_pairs:
            if match_b[b] == -1 and match_c[c] == -1:
                match_b[b] = a
                match_c[c] = a
                if backtrack(a + 1):
                    return True
                match_b[b] = -1
                match_c[c] = -1
        
        return False
    
    if backtrack(0):
        print("YES")
        # Reconstruct the solution
        # For each a, find which (b,c) was chosen
        for a in range(k):
            for b, c in a_to_bc.get(a, []):
                if match_b[b] == a and match_c[c] == a:
                    print(f"{a+1} {b+1} {c+1}")
                    break
    else:
        print("NO")

solve()
