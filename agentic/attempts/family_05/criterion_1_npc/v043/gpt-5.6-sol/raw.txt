import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    k, t = data[0], data[1]
    triples = []
    pos = 2

    basis = [0] * (3 * k + 1)

    for _ in range(t):
        x = data[pos] - 1
        y = data[pos + 1] - 1
        z = data[pos + 2] - 1
        pos += 3
        triples.append((x, y, z))

        value = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z)) | (1 << (3 * k))
        while value:
            p = value.bit_length() - 1
            if basis[p]:
                value ^= basis[p]
            else:
                basis[p] = value
                break

    if t < k:
        print("NO")
        return

    target = (1 << (3 * k)) - 1
    if k & 1:
        target |= 1 << (3 * k)

    value = target
    while value:
        p = value.bit_length() - 1
        if basis[p]:
            value ^= basis[p]
        else:
            print("NO")
            return

    columns = {i: set() for i in range(3 * k)}
    rows = []
    xs = [0] * t
    ys = [0] * t
    zs = [0] * t

    for rid, (x, y, z) in enumerate(triples):
        row = (x, k + y, 2 * k + z)
        rows.append(row)
        xs[rid] = x
        ys[rid] = y
        zs[rid] = z
        columns[row[0]].add(rid)
        columns[row[1]].add(rid)
        columns[row[2]].add(rid)

    failed = set()
    answer = []
    full_mask_limit = 1_200_000

    def select_row(rid):
        removed = []
        for col in rows[rid]:
            current = columns[col]
            for other in current:
                for other_col in rows[other]:
                    if other_col != col:
                        columns[other_col].remove(other)
            removed.append(columns.pop(col))
        return removed

    def restore_row(rid, removed):
        for col in reversed(rows[rid]):
            restored = removed.pop()
            columns[col] = restored
            for other in restored:
                for other_col in rows[other]:
                    if other_col != col:
                        columns[other_col].add(other)

    def has_perfect_matching(neighbors):
        n = len(neighbors)
        if n == 0:
            return True
        neighbors.sort(key=int.bit_count)
        if neighbors[0] == 0:
            return False

        owner = [-1] * k

        sys.setrecursionlimit(10000)

        for left in range(n):
            seen = 0

            def augment(u):
                nonlocal seen
                available = neighbors[u] & ~seen
                while available:
                    bit = available & -available
                    available ^= bit
                    seen |= bit
                    right = bit.bit_length() - 1
                    previous = owner[right]
                    if previous == -1 or augment(previous):
                        owner[right] = u
                        return True
                return False

            if not augment(left):
                return False
        return True

    def projection_check():
        ab = [0] * k
        ac = [0] * k
        bc = [0] * k

        for a in range(k):
            col = a
            active = columns.get(col)
            if active is None:
                continue
            for rid in active:
                y = ys[rid]
                z = zs[rid]
                ab[a] |= 1 << y
                ac[a] |= 1 << z
                bc[y] |= 1 << z

        ab_rows = [ab[a] for a in range(k) if a in columns]
        if not has_perfect_matching(ab_rows):
            return False

        ac_rows = [ac[a] for a in range(k) if a in columns]
        if not has_perfect_matching(ac_rows):
            return False

        bc_rows = [bc[b] for b in range(k) if k + b in columns]
        return has_perfect_matching(bc_rows)

    def search(mask_a, mask_b, mask_c):
        if not columns:
            return True

        state = mask_a | (mask_b << k) | (mask_c << (2 * k))
        if state in failed:
            return False

        chosen_col = min(columns, key=lambda c: len(columns[c]))
        candidates_set = columns[chosen_col]

        if not candidates_set:
            if len(failed) < full_mask_limit:
                failed.add(state)
            return False

        remaining = k - len(answer)
        if remaining > 1 and not projection_check():
            if len(failed) < full_mask_limit:
                failed.add(state)
            return False

        candidates = list(candidates_set)
        candidates.sort(
            key=lambda r: sum(len(columns[c]) for c in rows[r])
        )

        for rid in candidates:
            x = xs[rid]
            y = ys[rid]
            z = zs[rid]

            removed = select_row(rid)
            answer.append(rid)

            if search(mask_a | (1 << x),
                      mask_b | (1 << y),
                      mask_c | (1 << z)):
                return True

            answer.pop()
            restore_row(rid, removed)

        if len(failed) < full_mask_limit:
            failed.add(state)
        return False

    if search(0, 0, 0):
        out = ["YES"]
        for rid in answer:
            x, y, z = triples[rid]
            out.append(f"{x + 1} {y + 1} {z + 1}")
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    main()
