import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    reactions = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        reactions.append((x - 1, y - 1, z - 1))
    
    # Group reactions by base fluid (x)
    by_x = [[] for _ in range(k)]
    for idx, (x, y, z) in enumerate(reactions):
        by_x[x].append((y, z, idx))
    
    # DP with bitmask
    # dp[mask_y][mask_z] = (prev_mask_y, prev_mask_z, reaction_index) or -1 if unreachable
    size = 1 << k
    dp = [[-1] * size for _ in range(size)]
    dp[0][0] = (-2, -2, -2)  # Sentinel for start
    
    for x in range(k):
        dp_new = [[-1] * size for _ in range(size)]
        for mask_y in range(size):
            for mask_z in range(size):
                if dp[mask_y][mask_z] == -1:
                    continue
                # Option 1: don't use any reaction for this x (only valid if we can't find one)
                # Actually, we must use exactly one reaction per x, so we need to find a valid reaction
                found = False
                for y, z, idx in by_x[x]:
                    if (mask_y >> y) & 1 == 0 and (mask_z >> z) & 1 == 0:
                        new_mask_y = mask_y | (1 << y)
                        new_mask_z = mask_z | (1 << z)
                        if dp_new[new_mask_y][new_mask_z] == -1:
                            dp_new[new_mask_y][new_mask_z] = (mask_y, mask_z, idx)
                            found = True
                if not found and by_x[x]:
                    # No valid reaction for this x, this path is dead
                    pass
        dp = dp_new
    
    full_mask = (1 << k) - 1
    if dp[full_mask][full_mask] == -1:
        print("NO")
        return
    
    # Reconstruct the solution
    result = []
    mask_y, mask_z = full_mask, full_mask
    for x in range(k - 1, -1, -1):
        prev_y, prev_z, idx = dp[mask_y][mask_z]
        result.append(reactions[idx])
        mask_y, mask_z = prev_y, prev_z
    
    print("YES")
    for x, y, z in result:
        print(x + 1, y + 1, z + 1)

solve()
