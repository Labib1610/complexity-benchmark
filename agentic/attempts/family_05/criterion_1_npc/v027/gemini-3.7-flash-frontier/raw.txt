import sys


def solve():
    sys.setrecursionlimit(1000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3
        triples.append((x, y, z))

    M = 3 * k
    max_nodes = M + 1 + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    sz = [0] * (M + 1)

    for i in range(M + 1):
        L[i] = (i - 1) % (M + 1)
        R[i] = (i + 1) % (M + 1)
        U[i] = i
        D[i] = i
        sz[i] = 0

    node_cnt = M

    for r_idx, (x, y, z) in enumerate(triples):
        cols_in_row = [x + 1, k + y + 1, 2 * k + z + 1]
        first = node_cnt + 1
        for c in cols_in_row:
            node_cnt += 1
            cur = node_cnt
            col[cur] = c
            row[cur] = r_idx
            sz[c] += 1

            U[cur] = U[c]
            D[cur] = c
            D[U[c]] = cur
            U[c] = cur

        L[first] = node_cnt
        R[node_cnt] = first
        for j in range(first, node_cnt):
            R[j] = j + 1
            L[j + 1] = j

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        c = R[0]
        min_sz = sz[c]
        cur = R[c]
        while cur != 0:
            if sz[cur] < min_sz:
                min_sz = sz[cur]
                c = cur
            cur = R[cur]

        if min_sz == 0:
            return False

        cover(c)
        i = D[c]
        while i != c:
            ans.append(row[i])
            j = R[i]
            while j != i:
                cover(col[j])
                j = R[j]

            if search():
                return True

            ans.pop()
            j = L[i]
            while j != i:
                uncover(col[j])
                j = L[j]
            i = D[i]

        uncover(c)
        return False

    if search():
        print("YES")
        for r_idx in ans:
            x, y, z = triples[r_idx]
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
