import sys

sys.setrecursionlimit(1000000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    raise SystemExit

k, t = data[0], data[1]
xs = [0] * t
ys = [0] * t
zs = [0] * t

cx = [0] * k
cy = [0] * k
cz = [0] * k

pxy = [0] * (k * k)
pxz = [0] * (k * k)
pyz = [0] * (k * k)

edge_vec = [0] * t

pos = 2
for i in range(t):
    x = data[pos] - 1
    y = data[pos + 1] - 1
    z = data[pos + 2] - 1
    pos += 3

    xs[i] = x
    ys[i] = y
    zs[i] = z

    bit = 1 << i
    cx[x] |= bit
    cy[y] |= bit
    cz[z] |= bit
    pxy[x * k + y] |= bit
    pxz[x * k + z] |= bit
    pyz[y * k + z] |= bit

    edge_vec[i] = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z)) | (1 << (3 * k))

if t == 0:
    print("NO")
    raise SystemExit

conflict = [0] * t
for i in range(t):
    conflict[i] = cx[xs[i]] | cy[ys[i]] | cz[zs[i]]

full = (1 << k) - 1
all_edges = (1 << t) - 1
path = []
failed = set()
MEMO_LIMIT = 300000


def iter_bits(mask):
    while mask:
        b = mask & -mask
        yield b.bit_length() - 1
        mask ^= b


def linear_feasible(avail, rx, ry, rz):
    remaining = rx.bit_count()
    target = rx | (ry << k) | (rz << (2 * k))
    if remaining & 1:
        target |= 1 << (3 * k)

    if target == 0:
        return True

    basis = [0] * (3 * k + 1)
    m = avail
    while m:
        b = m & -m
        e = b.bit_length() - 1
        m ^= b

        v = edge_vec[e]
        for p in range(3 * k, -1, -1):
            if (v >> p) & 1:
                if basis[p]:
                    v ^= basis[p]
                else:
                    basis[p] = v
                    if (target >> p) & 1:
                        target ^= v
                    break

        if target == 0:
            return True

    return False


def has_perfect_matching(adj, left_mask, right_mask):
    left = list(iter_bits(left_mask))
    left.sort(key=lambda u: (adj[u] & right_mask).bit_count())
    match = [-1] * k

    def augment(u, seen):
        choices = adj[u] & right_mask & ~seen
        while choices:
            b = choices & -choices
            choices ^= b
            v = b.bit_length() - 1
            old = match[v]
            if old == -1:
                match[v] = u
                return True
            if augment(old, seen | b):
                match[v] = u
                return True
        return False

    for u in left:
        if not augment(u, 0):
            return False
    return True


def pairwise_feasible(avail, rx, ry, rz):
    lx = list(iter_bits(rx))
    ly = list(iter_bits(ry))
    lz = list(iter_bits(rz))

    adj_xy = [0] * k
    adj_xz = [0] * k
    adj_yz = [0] * k

    for x in lx:
        base = x * k
        mask = 0
        for y in ly:
            if pxy[base + y] & avail:
                mask |= 1 << y
        adj_xy[x] = mask

        mask = 0
        for z in lz:
            if pxz[base + z] & avail:
                mask |= 1 << z
        adj_xz[x] = mask

    for y in ly:
        base = y * k
        mask = 0
        for z in lz:
            if pyz[base + z] & avail:
                mask |= 1 << z
        adj_yz[y] = mask

    return (
        has_perfect_matching(adj_xy, rx, ry)
        and has_perfect_matching(adj_xz, rx, rz)
        and has_perfect_matching(adj_yz, ry, rz)
    )


def solve(rx, ry, rz, avail):
    if rx == 0:
        return True

    state = (rx, ry, rz)
    if state in failed:
        return False

    best_mask = 0
    best_count = t + 1

    dx = [0] * k
    dy = [0] * k
    dz = [0] * k

    m = rx
    while m:
        b = m & -m
        i = b.bit_length() - 1
        m ^= b
        c = (cx[i] & avail).bit_count()
        dx[i] = c
        if c == 0:
            if len(failed) < MEMO_LIMIT:
                failed.add(state)
            return False
        if c < best_count:
            best_count = c
            best_mask = cx[i] & avail

    m = ry
    while m:
        b = m & -m
        i = b.bit_length() - 1
        m ^= b
        c = (cy[i] & avail).bit_count()
        dy[i] = c
        if c == 0:
            if len(failed) < MEMO_LIMIT:
                failed.add(state)
            return False
        if c < best_count:
            best_count = c
            best_mask = cy[i] & avail

    m = rz
    while m:
        b = m & -m
        i = b.bit_length() - 1
        m ^= b
        c = (cz[i] & avail).bit_count()
        dz[i] = c
        if c == 0:
            if len(failed) < MEMO_LIMIT:
                failed.add(state)
            return False
        if c < best_count:
            best_count = c
            best_mask = cz[i] & avail

    if not linear_feasible(avail, rx, ry, rz):
        if len(failed) < MEMO_LIMIT:
            failed.add(state)
        return False

    if best_count > 1 and not pairwise_feasible(avail, rx, ry, rz):
        if len(failed) < MEMO_LIMIT:
            failed.add(state)
        return False

    candidates = []
    m = best_mask
    while m:
        b = m & -m
        e = b.bit_length() - 1
        m ^= b
        score = dx[xs[e]] + dy[ys[e]] + dz[zs[e]]
        candidates.append((score, e))
    candidates.sort()

    for _, e in candidates:
        x = xs[e]
        y = ys[e]
        z = zs[e]
        path.append(e)
        if solve(
            rx & ~(1 << x),
            ry & ~(1 << y),
            rz & ~(1 << z),
            avail & ~conflict[e]
        ):
            return True
        path.pop()

    if len(failed) < MEMO_LIMIT:
        failed.add(state)
    return False


if solve(full, full, full, all_edges):
    out = ["YES"]
    for e in path:
        out.append(f"{xs[e] + 1} {ys[e] + 1} {zs[e] + 1}")
    sys.stdout.write("\n".join(out))
else:
    print("NO")
