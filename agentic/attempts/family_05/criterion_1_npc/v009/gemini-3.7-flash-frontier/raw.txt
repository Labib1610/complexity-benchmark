import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3
        triples.append((x, y, z))

    if t < k:
        print("NO")
        return

    num_elements = 3 * k
    element_masks = [0] * num_elements
    triple_conflict_masks = [0] * t

    for i, (x, y, z) in enumerate(triples):
        m = 1 << i
        element_masks[x] |= m
        element_masks[k + y] |= m
        element_masks[2 * k + z] |= m

    for i, (x, y, z) in enumerate(triples):
        triple_conflict_masks[i] = (
            element_masks[x] | element_masks[k + y] | element_masks[2 * k + z]
        )

    all_triples_mask = (1 << t) - 1
    solution = []

    def search(available_triples, uncovered_elements_mask):
        if uncovered_elements_mask == 0:
            return True

        # Find uncovered element with minimum remaining choices (MRV)
        best_c = -1
        min_choices = float("inf")
        best_cand = 0

        temp = uncovered_elements_mask
        while temp:
            lsb = temp & -temp
            c = lsb.bit_length() - 1
            temp ^= lsb

            cand = available_triples & element_masks[c]
            cnt = cand.bit_count()
            if cnt == 0:
                return False
            if cnt < min_choices:
                min_choices = cnt
                best_c = c
                best_cand = cand
                if min_choices == 1:
                    break

        cand = best_cand
        while cand:
            lsb = cand & -cand
            i = lsb.bit_length() - 1
            cand ^= lsb

            solution.append(i)
            x, y, z = triples[i]
            covered = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
            if search(
                available_triples & ~triple_conflict_masks[i],
                uncovered_elements_mask ^ covered,
            ):
                return True
            solution.pop()

        return False

    initial_uncovered = (1 << num_elements) - 1
    if search(all_triples_mask, initial_uncovered):
        print("YES")
        out = []
        for i in solution:
            x, y, z = triples[i]
            out.append(f"{x + 1} {y + 1} {z + 1}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
