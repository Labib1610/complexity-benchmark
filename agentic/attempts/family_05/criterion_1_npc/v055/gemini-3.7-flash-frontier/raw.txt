import sys

# Increase recursion depth just in case
sys.setrecursionlimit(50000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx + 1])
        z = int(input_data[idx + 2])
        idx += 3
        triples.append((x, y, z))

    num_cols = 3 * k
    # 0 is root, 1..num_cols are column headers
    max_nodes = num_cols + 1 + 3 * t + 10

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row_orig = [None] * max_nodes
    sz = [0] * (num_cols + 1)

    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = (i - 1 + num_cols + 1) % (num_cols + 1)
        R[i] = (i + 1) % (num_cols + 1)
        U[i] = i
        D[i] = i
        col[i] = i

    node_cnt = num_cols

    for orig in triples:
        x, y, z = orig
        # Columns: surgeon x (1..k), room y (k+1..2k), slot z (2k+1..3k)
        cols_in_row = [x, y + k, z + 2 * k]
        first_node = node_cnt + 1
        for c in cols_in_row:
            node_cnt += 1
            col[node_cnt] = c
            row_orig[node_cnt] = orig
            sz[c] += 1

            # Insert vertically at the bottom of column c
            U[node_cnt] = U[c]
            D[node_cnt] = c
            D[U[c]] = node_cnt
            U[c] = node_cnt

        # Connect horizontally
        for i in range(3):
            curr = first_node + i
            prev = first_node + (i - 1) % 3
            nxt = first_node + (i + 1) % 3
            L[curr] = prev
            R[curr] = nxt

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with minimum size (MRV heuristic)
        c = R[0]
        min_sz = sz[c]
        curr = R[c]
        while curr != 0:
            if sz[curr] < min_sz:
                min_sz = sz[curr]
                c = curr
            curr = R[curr]

        if min_sz == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_orig[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            solution.pop()
            r = D[r]

        uncover(c)
        return False

    if search():
        print("YES")
        for x, y, z in solution:
            print(f"{x} {y} {z}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
