```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    if t < k:
        print("NO")
        return

    # Columns:
    # 0 to k-1: Chassis
    # k to 2k-1: Power
    # 2k to 3k-1: Logic

    num_cols = 3 * k
    row_cols = []
    col_rows = [set() for _ in range(num_cols)]

    idx = 2
    for r in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3

        c1 = x
        c2 = k + y
        c3 = 2 * k + z

        row_cols.append((c1, c2, c3, x + 1, y + 1, z + 1))
        col_rows[c1].add(r)
        col_rows[c2].add(r)
        col_rows[c3].add(r)

    # If any item cannot be covered, immediately output NO
    for c in range(num_cols):
        if len(col_rows[c]) == 0:
            print("NO")
            return

    active_cols = set(range(num_cols))
    solution = []

    def cover(c):
        active_cols.remove(c)
        removed_rows = []
        for r in col_rows[c]:
            c1, c2, c3, _, _, _ = row_cols[r]
            for other_c in (c1, c2, c3):
                if other_c != c:
                    col_rows[other_c].remove(r)
            removed_rows.append(r)
        return removed_rows

    def uncover(c, removed_rows):
        active_cols.add(c)
        for r in removed_rows:
            c1, c2, c3, _, _, _ = row_cols[r]
            for other_c in (c1, c2, c3):
                if other_c != c:
                    col_rows[other_c].add(r)

    def search():
        if not active_cols:
            return True

        # Pick column with minimum remaining rows
        min_len = float("inf")
        best_col = -1
        for c in active_cols:
            sz = len(col_rows[c])
            if sz < min_len:
                min_len = sz
                best_col = c
                if sz <= 1:
                    break

        if min_len == 0:
            return False

        # Try all rows covering best_col
        candidates = list(col_rows[best_col])
        for r in candidates:
            c1, c2, c3, x, y, z = row_cols[r]
            solution.append((x, y, z))

            # Cover all 3 columns
            rem1 = cover(c1)
            rem2 = cover(c2)
            rem3 = cover(c3)

            if search():
                return True

            uncover(c3, rem3)
            uncover(c2, rem2)
            uncover(c1, rem1)
            solution.pop()

        return False

    sys.setrecursionlimit(2000)
    if search():
        print("YES")
        for x, y, z in solution:
            print(f"{x} {y} {z}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
