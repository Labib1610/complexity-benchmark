import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    groupings = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        groupings.append((x-1, y-1, z-1))
    
    # We need to find k groupings such that:
    # - Each chassis 0..k-1 is used exactly once
    # - Each power unit 0..k-1 is used exactly once
    # - Each logic unit 0..k-1 is used exactly once
    
    # This is a 3D matching problem. With k <= 22, we can use meet-in-the-middle.
    # Split into two halves: first k//2 groupings and remaining k - k//2 groupings.
    
    n1 = k // 2
    n2 = k - n1
    
    # Chassis 0..n1-1 go to first half, n1..k-1 go to second half
    # Similarly for power units and logic units
    
    # For first half: we need to match chassis 0..n1-1, power 0..n1-1, logic 0..n1-1
    # For second half: we need to match chassis n1..k-1, power n1..k-1, logic n1..k-1
    
    # But this partition might not work if groupings cross the boundary.
    # Instead, let's use a different approach: enumerate all valid partial matchings.
    
    # Actually, let's use a simpler meet-in-the-middle:
    # For first half: enumerate all ways to select n1 groupings covering first half of each type
    # For second half: enumerate all ways to select n2 groupings covering second half of each type
    # Then combine them.
    
    # But the partition of indices matters. Let's try partitioning by chassis:
    # First half: chassis 0..n1-1
    # Second half: chassis n1..k-1
    
    # For first half, we need to find n1 groupings with chassis in [0, n1-1]
    # such that power units are distinct and logic units are distinct.
    # We don't care which power/logic units are used, as long as they're distinct.
    
    # Hmm, this is getting complicated. Let me think again.
    
    # Alternative: Use the standard meet-in-the-middle for 3D matching.
    # For each half, enumerate all valid partial matchings.
    # A partial matching for first half is a set of n1 groupings where:
    # - Chassis are distinct and in some subset S_c of size n1
    # - Power units are distinct and in some subset S_p of size n1
    # - Logic units are distinct and in some subset S_l of size n1
    
    # For the two halves to combine, we need:
    # S_c1 ∪ S_c2 = {0..k-1}, S_c1 ∩ S_c2 = ∅
    # Similarly for power and logic.
    
    # This means S_c2 = complement of S_c1, etc.
    
    # So for each half, we enumerate all valid partial matchings and store them
    # keyed by (S_c, S_p, S_l) where each is a bitmask.
    
    # Then for each matching in first half with (S_c1, S_p1, S_l1),
    # we look for a matching in second half with (complement(S_c1), complement(S_p1), complement(S_l1)).
    
    # Let's implement this.
    
    # Filter groupings for each half based on chassis
    groupings1 = [(x, y, z) for (x, y, z) in groupings if x < n1]
    groupings2 = [(x, y, z) for (x, y, z) in groupings if x >= n1]
    
    # Enumerate all valid partial matchings for first half
    # We need to select exactly n1 groupings
    # Each chassis in [0, n1-1] used exactly once
    # Power units distinct, logic units distinct
    
    def enumerate_matchings(grps, n, target_chassis_start, target_chassis_count):
        # grps: list of (x, y, z) where x is in [target_chassis_start, target_chassis_start + target_chassis_count - 1]
        # We need to select exactly target_chassis_count groupings
        # Each chassis in the range used exactly once
        # Power units distinct, logic units distinct
        
        # Returns: dict mapping (chassis_mask, power_mask, logic_mask) to list of groupings
        # chassis_mask: bitmask of chassis used (relative to 0)
        # power_mask: bitmask of power units used (relative to 0)
        # logic_mask: bitmask of logic units used (relative to 0)
        
        result = {}
        
        # DFS with pruning
        grps_count = len(grps)
        
        # Sort groupings by chassis for better pruning
        chassis_to_grps = {}
        for i, (x, y, z) in enumerate(grps):
            if x not in chassis_to_grps:
                chassis_to_grps[x] = []
            chassis_to_grps[x].append((i, y, z))
        
        chassis_list = sorted(chassis_to_grps.keys())
        
        # We need to use exactly the chassis in [target_chassis_start, target_chassis_start + target_chassis_count - 1]
        required_chassis = set(range(target_chassis_start, target_chassis_start + target_chassis_count))
        
        if set(chassis_list) != required_chassis:
            return result  # Can't cover all required chassis
        
        # DFS
        def dfs(chassis_idx, used_power, used_logic, selected):
            if chassis_idx == len(chassis_list):
                # All chassis assigned
                chassis_mask = 0
                for x, y, z in selected:
                    chassis_mask |= (1 << x)
                power_mask = 0
                for x, y, z in selected:
                    power_mask |= (1 << y)
                logic_mask = 0
                for x, y, z in selected:
                    logic_mask |= (1 << z)
                result[(chassis_mask, power_mask, logic_mask)] = selected[:]
                return
            
            x = chassis_list[chassis_idx]
            for i, y, z in chassis_to_grps[x]:
                if (used_power >> y) & 1:
                    continue
                if (used_logic >> z) & 1:
                    continue
                selected.append(grps[i])
                dfs(chassis_idx + 1, used_power | (1 << y), used_logic | (1 << z), selected)
                selected.pop()
        
        dfs(0, 0, 0, [])
        return result
    
    matchings1 = enumerate_matchings(groupings1, n1, 0, n1)
    matchings2 = enumerate_matchings(groupings2, n2, n1, n2)
    
    # For each matching in matchings1, look for complementary matching in matchings2
    full_mask = (1 << k) - 1
    
    for (cm1, pm1, lm1), sel1 in matchings1.items():
        cm2_target = full_mask ^ cm1
        pm2_target = full_mask ^ pm1
        lm2_target = full_mask ^ lm1
        
        key2 = (cm2_target, pm2_target, lm2_target)
        if key2 in matchings2:
            sel2 = matchings2[key2]
            # Found a solution
            print("YES")
            for x, y, z in sel1:
                print(f"{x+1} {y+1} {z+1}")
            for x, y, z in sel2:
                print(f"{x+1} {y+1} {z+1}")
            return
    
    print("NO")

solve()
