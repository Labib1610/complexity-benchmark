We need to select exactly k triples (x, y, z) such that each x, y, z from 1..k appears exactly once. This is a 3D matching problem, which is NP-hard in general, but k ≤ 22 allows a backtracking with pruning or bitmask DP.

We can solve it with recursive backtracking + bitmask state for used x, y, z. Since k ≤ 22, we can use bitmasks (integers) for used sets.

We'll sort the triples to help pruning, and use recursion that picks the next unused array ID with the fewest options.

Here’s the implementation:

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))
    
    # Group by x for faster lookup
    by_x = [[] for _ in range(k)]
    for x, y, z in triples:
        by_x[x].append((y, z))
    
    # Sort each list for deterministic order
    for i in range(k):
        by_x[i].sort()
    
    used_y = 0
    used_z = 0
    selected = []
    
    # Precompute for each x the list of (y,z) pairs
    # We'll do backtracking with pruning: choose the x with fewest available options
    full_mask = (1 << k) - 1
    
    def backtrack(used_x_mask, used_y_mask, used_z_mask, count):
        if count == k:
            return True
        # Find the x with fewest available options
        best_x = -1
        best_options = None
        for x in range(k):
            if not (used_x_mask >> x) & 1:
                options = []
                for y, z in by_x[x]:
                    if not (used_y_mask >> y) & 1 and not (used_z_mask >> z) & 1:
                        options.append((y, z))
                if best_x == -1 or len(options) < len(best_options):
                    best_x = x
                    best_options = options
                    if len(options) == 0:
                        break
        if best_x == -1 or len(best_options) == 0:
            return False
        
        x = best_x
        for y, z in best_options:
            # choose
            selected.append((x, y, z))
            if backtrack(used_x_mask | (1 << x),
                         used_y_mask | (1 << y),
                         used_z_mask | (1 << z),
                         count + 1):
                return True
            selected.pop()
        return False
    
    if backtrack(0, 0, 0, 0):
        print("YES")
        for x, y, z in selected:
            print(x + 1, y + 1, z + 1)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
