from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    linkages = []
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        linkages.append((x, y, z))
        idx += 3
    
    # We need to find a permutation of k linkages such that:
    # - Each array ID appears exactly once
    # - Each target ID appears exactly once
    # - Each band ID appears exactly once
    
    # This is essentially finding a perfect matching in a 3D assignment problem.
    # Since k <= 22, we can't do brute force on all permutations of linkages.
    # But since we're looking for k linkages out of at most k^3, and k is small,
    # we can try backtracking approach.
    
    # More specifically, this is like finding a Latin cube or a 3D matching.
    
    # Let's model this as a bipartite matching where we match arrays to (target, band) pairs.
    # But actually, it's a tripartite matching.
    
    # For small k, let's try backtracking.
    
    def backtrack(pos, used_array, used_target, used_band, selected):
        if pos == k:
            return True
            
        for i in range(t):
            x, y, z = linkages[i]
            if not (used_array & (1 << x)) and not (used_target & (1 << y)) and not (used_band & (1 << z)):
                selected[pos] = (x, y, z)
                if backtrack(pos + 1, used_array | (1 << x), used_target | (1 << y), used_band | (1 << z), selected):
                    return True
        return False
    
    # Optimization: if t < k, impossible
    if t < k:
        print("NO")
        return
        
    selected = [None] * k
    if backtrack(0, 0, 0, 0, selected):
        print("YES")
        for triple in selected:
            print(triple[0], triple[1], triple[2])
    else:
        print("NO")

solve()
