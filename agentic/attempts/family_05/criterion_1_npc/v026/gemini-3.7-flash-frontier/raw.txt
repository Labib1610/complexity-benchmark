```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    triples = []
    masks = []
    triples_of = [[] for _ in range(3 * k)]

    idx = 2
    for j in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3

        triples.append((x + 1, y + 1, z + 1))
        m = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
        masks.append(m)

        triples_of[x].append(j)
        triples_of[k + y].append(j)
        triples_of[2 * k + z].append(j)

    target_mask = (1 << (3 * k)) - 1
    selected = []

    def search(used):
        if used == target_mask:
            return True

        min_count = 10**9
        best_candidates = None

        for e in range(3 * k):
            if not (used & (1 << e)):
                candidates = [j for j in triples_of[e] if not (masks[j] & used)]
                cnt = len(candidates)
                if cnt == 0:
                    return False
                if cnt < min_count:
                    min_count = cnt
                    best_candidates = candidates
                    if min_count == 1:
                        break

        for j in best_candidates:
            selected.append(j)
            if search(used | masks[j]):
                return True
            selected.pop()

        return False

    if search(0):
        print("YES")
        out = []
        for j in selected:
            x, y, z = triples[j]
            out.append(f"{x} {y} {z}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
