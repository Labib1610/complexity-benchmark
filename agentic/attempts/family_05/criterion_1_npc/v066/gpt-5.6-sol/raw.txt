```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    ncol = 3 * k
    triples = []
    row_masks = []

    parent = list(range(ncol))
    size = [1] * ncol

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        a = find(a)
        b = find(b)
        if a == b:
            return
        if size[a] < size[b]:
            a, b = b, a
        parent[b] = a
        size[a] += size[b]

    p = 2
    for _ in range(t):
        x, y, z = data[p], data[p + 1], data[p + 2]
        p += 3
        a = x - 1
        b = k + y - 1
        c = 2 * k + z - 1
        triples.append((x, y, z))
        row_masks.append((1 << a) | (1 << b) | (1 << c))
        union(a, b)
        union(a, c)

    # Every connected component must contain equally many vertices
    # from each of the three parts.
    component_counts = {}
    for v in range(ncol):
        r = find(v)
        counts = component_counts.setdefault(r, [0, 0, 0])
        counts[v // k] += 1

    for a, b, c in component_counts.values():
        if a != b or b != c:
            print("NO")
            return

    full_mask = (1 << ncol) - 1

    # Necessary feasibility test over GF(2).
    def gf2_feasible(used_mask):
        basis = [0] * ncol
        for rm in row_masks:
            if rm & used_mask:
                continue
            value = rm
            while value:
                bit = value.bit_length() - 1
                if basis[bit]:
                    value ^= basis[bit]
                else:
                    basis[bit] = value
                    break

        value = full_mask ^ used_mask
        while value:
            bit = value.bit_length() - 1
            if not basis[bit]:
                return False
            value ^= basis[bit]
        return True

    if not gf2_feasible(0):
        print("NO")
        return

    # Dancing Links structure.
    L = list(range(ncol + 1))
    R = list(range(ncol + 1))
    U = list(range(ncol + 1))
    D = list(range(ncol + 1))
    C = list(range(ncol + 1))
    S = [0] * (ncol + 1)
    RID = [-1] * (ncol + 1)

    if ncol:
        L[0] = ncol
        R[0] = 1
        for c in range(1, ncol + 1):
            L[c] = c - 1
            R[c] = c + 1 if c < ncol else 0

    for row_id, (x, y, z) in enumerate(triples):
        columns = (x, k + y, 2 * k + z)
        nodes = []

        for col in columns:
            idx = len(C)
            C.append(col)
            S.append(0)
            RID.append(row_id)
            L.append(idx)
            R.append(idx)

            up = U[col]
            U.append(up)
            D.append(col)
            D[up] = idx
            U[col] = idx
            S[col] += 1
            nodes.append(idx)

        for i in range(3):
            L[nodes[i]] = nodes[(i - 1) % 3]
            R[nodes[i]] = nodes[(i + 1) % 3]

    sys.setrecursionlimit(10000)
    answer = []
    failed = set()
    memo_limit = 700000

    def cover(c):
        R[L[c]] = R[c]
        L[R[c]] = L[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                D[U[j]] = D[j]
                U[D[j]] = U[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                D[U[j]] = j
                U[D[j]] = j
                j = L[j]
            i = U[i]
        R[L[c]] = c
        L[R[c]] = c

    def remember_failure(state):
        if len(failed) < memo_limit:
            failed.add(state)

    def search(depth, state):
        if R[0] == 0:
            return True
        if state in failed:
            return False

        c = R[0]
        best_size = S[c]
        j = R[c]
        while j != 0:
            if S[j] < best_size:
                c = j
                best_size = S[j]
                if best_size == 0:
                    break
            j = R[j]

        if best_size == 0:
            remember_failure(state)
            return False

        if depth and depth <= 15 and depth % 5 == 0:
            if not gf2_feasible(state):
                remember_failure(state)
                return False

        cover(c)

        candidates = []
        r = D[c]
        while r != c:
            a = R[r]
            b = R[a]
            candidates.append((S[C[a]] + S[C[b]], r))
            r = D[r]
        candidates.sort()

        for _, r in candidates:
            row_id = RID[r]
            answer.append(row_id)

            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search(depth + 1, state | row_masks[row_id]):
                return True

            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]

            answer.pop()

        uncover(c)
        remember_failure(state)
        return False

    if search(0, 0):
        out = ["YES"]
        out.extend("{} {} {}".format(*triples[row_id]) for row_id in answer)
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
