import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    # profiles[i] = list of (y, z) for specialist i (0-indexed)
    profiles = [[] for _ in range(k)]
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        profiles[x].append((y, z))
    
    # For each specialist, sort profiles by facility for tie-breaking
    for i in range(k):
        profiles[i].sort()
    
    # State: (specialist_index, facilities_used_bitmask, dossiers_used_bitmask)
    # We use BFS/DFS with memoization of visited states
    
    # Since k <= 22, bitmask fits in integer
    # Total states: k * 2^k * 2^k which is too large for k=22
    # But we can use meet-in-the-middle or clever pruning
    
    # Actually, let's think about this differently.
    # We need a perfect matching in a 3-partite 3-uniform hypergraph.
    # This is NP-hard in general, but k <= 22 is small enough for backtracking with pruning.
    
    # Let's try backtracking with bitmask DP
    # State: (specialist_index, facilities_mask, dossiers_mask)
    # We process specialists 0, 1, 2, ..., k-1 in order
    # For each specialist, try each of its profiles
    
    # To make this efficient, we memoize failed states
    visited = set()
    
    # Precompute: for each specialist, list of (y, z)
    # Use iterative deepening or simple DFS
    
    # Let's try a recursive DFS with memoization of failed states
    # But the state space is huge. Let's think about pruning.
    
    # Key insight: if at any point, some facility or dossier has no remaining specialist
    # that can cover it, we can prune.
    
    # Let's precompute for each facility, which specialists have a profile using it
    # Similarly for each dossier
    
    facility_specialists = [set() for _ in range(k)]
    dossier_specialists = [set() for _ in range(k)]
    for i in range(k):
        for (y, z) in profiles[i]:
            facility_specialists[y].add(i)
            dossier_specialists[z].add(i)
    
    result = [None] * k
    
    def dfs(idx, fac_mask, dos_mask):
        if idx == k:
            return True
        
        state = (idx, fac_mask, dos_mask)
        if state in visited:
            return False
        
        # Pruning: check if all remaining facilities and dossiers can still be covered
        remaining_specialists = set(range(idx, k))
        for f in range(k):
            if not (fac_mask >> f) & 1:
                if not (facility_specialists[f] & remaining_specialists):
                    visited.add(state)
                    return False
        for d in range(k):
            if not (dos_mask >> d) & 1:
                if not (dossier_specialists[d] & remaining_specialists):
                    visited.add(state)
                    return False
        
        for (y, z) in profiles[idx]:
            if not ((fac_mask >> y) & 1) and not ((dos_mask >> z) & 1):
                result[idx] = (y, z)
                if dfs(idx + 1, fac_mask | (1 << y), dos_mask | (1 << z)):
                    return True
                result[idx] = None
        
        visited.add(state)
        return False
    
    if dfs(0, 0, 0):
        print("YES")
        for i in range(k):
            y, z = result[i]
            print(f"{i+1} {y+1} {z+1}")
    else:
        print("NO")

solve()
