from collections import defaultdict

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
        
    profiles = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        profiles.append((x, y, z))
    
    # Try to find a perfect matching using backtracking
    # We need to match each specialist (1 to k) with a unique facility and dossier
    
    # Preprocess: for each specialist, store which profiles they can be part of
    spec_profiles = defaultdict(list)
    for i, (x, y, z) in enumerate(profiles):
        spec_profiles[x].append(i)
    
    # Try all possible assignments using backtracking
    used_facilities = [False] * (k + 1)
    used_dossiers = [False] * (k + 1)
    selected_profiles = []
    found = [False]  # Use list to pass by reference
    
    def backtrack(spec):
        if spec > k:
            found[0] = True
            return
            
        # Try each profile for this specialist
        for profile_idx in spec_profiles[spec]:
            x, y, z = profiles[profile_idx]
            if not used_facilities[y] and not used_dossiers[z]:
                used_facilities[y] = True
                used_dossiers[z] = True
                selected_profiles.append(profile_idx)
                
                backtrack(spec + 1)
                
                if found[0]:
                    return
                    
                selected_profiles.pop()
                used_facilities[y] = False
                used_dossiers[z] = False
    
    backtrack(1)
    
    if found[0]:
        print("YES")
        for idx in selected_profiles:
            print(profiles[idx][0], profiles[idx][1], profiles[idx][2])
    else:
        print("NO")

solve()
