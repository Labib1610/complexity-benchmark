import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3
        triples.append((x, y, z))

    if t < k:
        print("NO")
        return

    num_cols = 3 * k

    # DLX structures
    max_nodes = 1 + num_cols + 3 * t + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    sz = [0] * (num_cols + 1)

    for i in range(num_cols + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
    L[0] = num_cols
    R[num_cols] = 0

    node_cnt = num_cols

    for r_idx, (x, y, z) in enumerate(triples):
        cols = [x + 1, k + y + 1, 2 * k + z + 1]
        first = node_cnt + 1
        for c in cols:
            node_cnt += 1
            cur = node_cnt
            col[cur] = c
            row[cur] = r_idx
            sz[c] += 1

            # Insert into column c
            U[cur] = U[c]
            D[cur] = c
            D[U[c]] = cur
            U[c] = cur

            # Insert into row
            if cur == first:
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = L[first]
                R[cur] = first
                R[L[first]] = cur
                L[first] = cur

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        c = R[0]
        min_sz = sz[c]
        curr = R[0]
        while curr != 0:
            if sz[curr] < min_sz:
                min_sz = sz[curr]
                c = curr
                if min_sz == 0:
                    break
            curr = R[curr]

        if min_sz == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            ans.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]

            if search():
                return True

            ans.pop()
            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            r = D[r]

        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r_idx in ans:
            x, y, z = triples[r_idx]
            out.append(f"{x + 1} {y + 1} {z + 1}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
