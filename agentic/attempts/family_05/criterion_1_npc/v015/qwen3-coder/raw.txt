from collections import defaultdict
import sys

def find_matching(k, combinations):
    # Build a graph where each compute node points to possible (image, shard) pairs
    graph = defaultdict(list)
    for x, y, z in combinations:
        graph[x].append((y, z))
    
    # Try to find a perfect matching using backtracking
    match_image_shard = {}  # image -> shard
    match_shard_image = {}  # shard -> image
    result = []
    used_compute = [False] * (k + 1)
    
    def dfs(compute_node):
        if used_compute[compute_node]:
            return False
        used_compute[compute_node] = True
        
        for image, shard in graph[compute_node]:
            # Check if this image-shard pair is available
            if image not in match_image_shard or match_image_shard[image] == shard:
                # Try to augment the matching
                if shard not in match_shard_image or dfs(match_shard_image[shard]):
                    match_image_shard[image] = shard
                    match_shard_image[shard] = image
                    result.append((compute_node, image, shard))
                    return True
        return False
    
    # Try to assign each compute node
    assignment = {}
    used_images = set()
    used_shards = set()
    
    def backtrack(node):
        if node > k:
            return len(assignment) == k
        # Try each valid combination for this compute node
        for image, shard in graph[node]:
            if image not in used_images and shard not in used_shards:
                assignment[node] = (image, shard)
                used_images.add(image)
                used_shards.add(shard)
                if backtrack(node + 1):
                    return True
                del assignment[node]
                used_images.remove(image)
                used_shards.remove(shard)
        return False
    
    if backtrack(1):
        return True, [(node, assignment[node][0], assignment[node][1]) for node in assignment]
    else:
        return False, []

def main():
    input_lines = sys.stdin.read().splitlines()
    k, t = map(int, input_lines[0].split())
    
    combinations = []
    for i in range(1, t + 1):
        x, y, z = map(int, input_lines[i].split())
        combinations.append((x, y, z))
    
    found, assignment = find_matching(k, combinations)
    
    if found:
        print("YES")
        for x, y, z in assignment:
            print(x, y, z)
    else:
        print("NO")

if __name__ == "__main__":
    main()
