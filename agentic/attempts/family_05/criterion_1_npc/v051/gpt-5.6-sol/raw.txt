import sys

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    if t < k:
        print("NO")
        return

    triples = []
    row_columns = []
    column_rows = [0] * (3 * k)

    pos = 2
    for r in range(t):
        x = data[pos] - 1
        y = data[pos + 1] - 1
        z = data[pos + 2] - 1
        pos += 3

        triples.append((x + 1, y + 1, z + 1))
        cols = (x, k + y, 2 * k + z)
        row_columns.append(cols)

        bit = 1 << r
        column_rows[cols[0]] |= bit
        column_rows[cols[1]] |= bit
        column_rows[cols[2]] |= bit

    for mask in column_rows:
        if mask == 0:
            print("NO")
            return

    row_masks = [0] * t
    conflicts = [0] * t
    for r, (a, b, c) in enumerate(row_columns):
        row_masks[r] = (1 << a) | (1 << b) | (1 << c)
        conflicts[r] = column_rows[a] | column_rows[b] | column_rows[c]

    all_columns = (1 << (3 * k)) - 1
    all_rows = (1 << t) - 1
    answer = []
    failed = set()
    memo_limit = 500_000

    def in_binary_span(candidates, target):
        basis = [0] * (3 * k)
        rank = 0

        bits = candidates
        while bits:
            low = bits & -bits
            r = low.bit_length() - 1
            bits ^= low

            v = row_masks[r]
            while v:
                p = v.bit_length() - 1
                if basis[p]:
                    v ^= basis[p]
                else:
                    basis[p] = v
                    rank += 1

                    w = target
                    while w:
                        q = w.bit_length() - 1
                        if basis[q]:
                            w ^= basis[q]
                        else:
                            break
                    if w == 0:
                        return True
                    break

            if rank == 3 * k:
                return True

        w = target
        while w:
            p = w.bit_length() - 1
            if basis[p]:
                w ^= basis[p]
            else:
                return False
        return True

    def dfs(covered, candidates, depth):
        if depth == k:
            return covered == all_columns

        if covered in failed:
            return False

        uncovered = all_columns ^ covered
        best_options = 0
        best_count = t + 1

        cols = uncovered
        while cols:
            low = cols & -cols
            c = low.bit_length() - 1
            cols ^= low

            options = candidates & column_rows[c]
            count = options.bit_count()
            if count == 0:
                if len(failed) < memo_limit:
                    failed.add(covered)
                return False
            if count < best_count:
                best_count = count
                best_options = options
                if count == 1:
                    break

        if not in_binary_span(candidates, uncovered):
            if len(failed) < memo_limit:
                failed.add(covered)
            return False

        option_rows = []
        bits = best_options
        while bits:
            low = bits & -bits
            r = low.bit_length() - 1
            bits ^= low
            option_rows.append(r)

        if len(option_rows) <= 128:
            def score(r):
                a, b, c = row_columns[r]
                return (
                    (candidates & column_rows[a]).bit_count()
                    + (candidates & column_rows[b]).bit_count()
                    + (candidates & column_rows[c]).bit_count()
                )
            option_rows.sort(key=score)

        for r in option_rows:
            answer.append(r)
            if dfs(
                covered | row_masks[r],
                candidates & ~conflicts[r],
                depth + 1
            ):
                return True
            answer.pop()

        if len(failed) < memo_limit:
            failed.add(covered)
        return False

    if dfs(0, all_rows, 0):
        out = ["YES"]
        out.extend("{} {} {}".format(*triples[r]) for r in answer)
        sys.stdout.write("\n".join(out))
    else:
        print("NO")

if __name__ == "__main__":
    sys.setrecursionlimit(1000000)
    solve()
