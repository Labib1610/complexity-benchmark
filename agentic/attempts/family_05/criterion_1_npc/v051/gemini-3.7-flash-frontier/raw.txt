import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    trios = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx+1])
        z = int(input_data[idx+2])
        idx += 3
        trios.append((x, y, z))
    
    num_cols = 3 * k
    
    # DLX structures
    # Root is node 0
    # Column headers are 1 to num_cols
    MAX_NODES = (num_cols + 1) + 3 * t + 5
    
    L = [0] * MAX_NODES
    R = [0] * MAX_NODES
    U = [0] * MAX_NODES
    D = [0] * MAX_NODES
    C = [0] * MAX_NODES
    row_id = [0] * MAX_NODES
    size = [0] * (num_cols + 1)
    
    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        C[i] = i
    L[0] = num_cols
    R[num_cols] = 0
    
    node_count = num_cols
    
    for r_idx, (x, y, z) in enumerate(trios):
        cols = [x, k + y, 2 * k + z]
        first_node = node_count + 1
        for j, c in enumerate(cols):
            node_count += 1
            node = node_count
            C[node] = c
            row_id[node] = r_idx
            size[c] += 1
            
            # Vertical linking
            U[node] = U[c]
            D[node] = c
            D[U[c]] = node
            U[c] = node
            
            # Horizontal linking
            if j == 0:
                L[node] = node
                R[node] = node
            else:
                prev = node - 1
                L[node] = prev
                R[node] = R[prev]
                L[R[prev]] = node
                R[prev] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size
        c = R[0]
        min_size = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_size:
                min_size = size[curr]
                c = curr
            curr = R[curr]
            
        if min_size == 0:
            return False
            
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_id[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
                
            if search():
                return True
                
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution.pop()
            r = D[r]
            
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in solution:
            x, y, z = trios[r_idx]
            out.append(f"{x} {y} {z}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
