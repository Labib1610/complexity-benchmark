import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    # Build list of trios
    trios = []
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        trios.append((x, y, z))
    
    # We need to select exactly k trios such that every linguist, manuscript, algorithm is used exactly once.
    # This is a 3-dimensional matching problem. With k <= 22, we can use DP with bitmask.
    # State: (mask_x, mask_y, mask_z) but that's 2^66 states, too large.
    # Instead, since we need exactly k trios and each uses one of each, we can think of it as:
    # We process trios one by one and keep track of used linguists, manuscripts, algorithms.
    # But 2^22 * 2^22 is too large.
    # 
    # Alternative: Since we need exactly k trios, and each trio uses one x, one y, one z,
    # we can fix the matching between x and y (there are k! possibilities, too many for k=22).
    # 
    # Better approach: Meet in the middle or use the fact that we can reduce to bipartite matching?
    # Actually, 3D matching is NP-hard, but k <= 22 is small enough for DP with bitmask on two dimensions
    # if we iterate over trios and use a dictionary of reachable states.
    # 
    # Let's use DP where state = (mask_y, mask_z) and we assign linguists in order 1..k.
    # For each linguist i, we try all trios (i, y, z) where y and z are not yet used.
    # State space: 2^22 * 2^22 is too large (2^44).
    # 
    # Hmm, but we can use the fact that the number of trios t <= k^3 = 22^3 = 10648.
    # We can do DP over trios: process trios one by one, state = (mask_x, mask_y, mask_z).
    # But that's 2^66 states.
    # 
    # Wait, but we only need to track states that are actually reachable, and we only
    # pick at most k trios. The number of reachable states might be manageable if we use
    # a dictionary and prune.
    # 
    # Actually, a better approach: since we need exactly k trios, the masks for x, y, z
    # all have exactly the same number of bits set (equal to the number of trios selected).
    # So after selecting j trios, each mask has exactly j bits set.
    # 
    # For j bits set in mask_x, j bits in mask_y, j bits in mask_z: the number of such
    # states is C(k,j)^3. Sum over j from 0 to k of C(k,j)^3 = C(3k, k) roughly... 
    # For k=22, C(66,22) is about 8e17, way too large.
    # 
    # Let's think differently. We can reduce 3D matching to a series of 2D matchings
    # using Hall's theorem or network flow? No, 3D matching is NP-hard.
    # 
    # But k=22 is small. Let's try a different DP:
    # Group trios by linguist. For each linguist, we have a set of possible (y, z) pairs.
    # We need to select one (y, z) for each linguist such that all y's are distinct and all z's are distinct.
    # 
    # DP over linguists: state = (mask_y, mask_z). But 2^22 * 2^22 is too large.
    # 
    # However, we can use the meet-in-the-middle approach:
    # Split linguists into two halves. For each half, enumerate all possible (mask_y, mask_z) pairs
    # that can be achieved by selecting one trio for each linguist in that half.
    # Then combine the two halves.
    # 
    # For k=22, each half has 11 linguists. For each half, the number of possible (mask_y, mask_z)
    # states is at most (number of trios for that half) choose 11, but more precisely,
    # it's bounded by the number of ways to assign distinct y's and z's.
    # 
    # Actually, for 11 linguists, each with some (y,z) options, the number of valid (mask_y, mask_z)
    # states is at most C(22,11)^2 ≈ 705432^2 ≈ 5e11, still too large.
    # 
    # Hmm, let me think about this differently. Maybe we can use the structure more.
    # 
    # Actually, wait. Let's reconsider the DP over trios with a dictionary.
    # We process trios one by one. At each step, we can either include the trio or not.
    # State = (mask_x, mask_y, mask_z). But we only keep states where popcount(mask_x) = popcount(mask_y) = popcount(mask_z).
    # And we only need states with popcount <= k.
    # 
    # The key insight: if we process trios grouped by linguist, then mask_x is determined
    # by which linguists we've processed. So we can do DP where we process linguists one by one,
    # and for each linguist, we try all their trios.
    # 
    # State = (mask_y, mask_z) after processing some linguists.
    # But 2^22 * 2^22 is still too large for a full table.
    # 
    # However, using a dictionary (sparse DP), the number of reachable states might be manageable.
    # After processing j linguists, mask_y and mask_z each have exactly j bits set.
    # The number of such states is C(k,j)^2. The maximum is at j=11: C(22,11)^2 ≈ 5e11. Too large.
    # 
    # Let me try a different approach: reduce to bipartite matching using the following idea.
    # For each pair (y, z), check if there exists a linguist x such that (x, y, z) is a valid trio.
    # If so, we have a "super-edge" (y, z) that can be assigned to some linguist.
    # But we need to ensure that each linguist is used exactly once, which adds another constraint.
    # 
    # Actually, let's think about it as: we need a perfect matching in a 3-partite 3-uniform hypergraph.
    # This is NP-hard in general, but k=22 might be small enough for exponential algorithms.
    # 
    # One approach: Fix the assignment of linguists to manuscripts (a permutation), then check
    # if the remaining algorithms can be matched. But k! is too large for k=22.
    # 
    # Another approach: Use inclusion-exclusion or the permanent of a matrix.
    # 
    # Actually, let's try the sparse DP approach and see if it's fast enough in practice.
    # We process linguists 1 to k. For each linguist, we have a set of (y, z) pairs.
    # DP state: (mask_y, mask_z) -> list of assignments (or just one assignment for reconstruction).
    # 
    # The number of reachable states after processing j linguists is at most C(k,j)^2, but in practice
    # it might be much smaller because not all combinations are reachable.
    # 
    # For k=22, the worst case might be too large, but let's try with optimizations.
    # 
    # Actually, we can optimize: instead of tracking both mask_y and mask_z separately,
    # we can note that after processing all k linguists, we need mask_y = (1<<k)-1 and mask_z = (1<<k)-1.
    # 
    # Let me try a different formulation: think of it as assigning each linguist a (y, z) pair.
    # We can model this as: first assign each linguist a manuscript (bijection), then assign each
    # (linguist, manuscript) pair an algorithm. But this doesn't simplify things.
    # 
    # Let me just try the sparse DP and hope it's fast enough.
    
    # Group trios by linguist
    linguist_trios = [[] for _ in range(k+1)]
    for x, y, z in trios:
        linguist_trios[x].append((y-1, z-1))  # 0-indexed
    
    # Check each linguist has at least one trio
    for i in range(1, k+1):
        if not linguist_trios[i]:
            print("NO")
            return
    
    # DP: process linguists 1 to k
    # State: (mask_y, mask_z) -> assignment list
    # Use dictionary for sparse representation
    
    # To optimize, we can use the fact that mask_y and mask_z have the same popcount.
    # We can encode the state as a single integer: mask_y * (1<<k) + mask_z, but that's 2^44.
    # Better to use a tuple (mask_y, mask_z) as key.
    
    # Let's try with a dictionary. Start with empty state.
    # dp: dict mapping (mask_y, mask_z) -> list of (y, z) assignments (for reconstruction)
    
    # To save memory, we only need to store one path to each state.
    
    dp = {(0, 0): []}
    
    for i in range(1, k+1):
        new_dp = {}
        for (my, mz), assignments in dp.items():
            for y, z in linguist_trios[i]:
                if (my >> y) & 1 or (mz >> z) & 1:
                    continue
                new_my = my | (1 << y)
                new_mz = mz | (1 << z)
                new_key = (new_my, new_mz)
                if new_key not in new_dp:
                    new_dp[new_key] = assignments + [(i, y+1, z+1)]
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    full_mask = (1 << k) - 1
    key = (full_mask, full_mask)
    if key in dp:
        print("YES")
        for x, y, z in dp[key]:
            print(x, y, z)
    else:
        print("NO")

solve()
