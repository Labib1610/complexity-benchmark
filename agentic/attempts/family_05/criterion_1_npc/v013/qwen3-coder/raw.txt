from itertools import combinations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    profiles = []
    index = 2
    for _ in range(t):
        x = int(data[index])
        y = int(data[index+1])
        z = int(data[index+2])
        profiles.append((x, y, z))
        index += 3
    
    # We need to find a subset of k profiles such that all operators, terminals, and shifts are unique
    # This is essentially a matching problem where we pick k triples with no repeated elements in any position
    
    # Try all combinations of k profiles from the t available
    from itertools import combinations
    
    for combo in combinations(range(t), k):
        operators = set()
        terminals = set()
        shifts = set()
        valid = True
        
        for i in combo:
            x, y, z = profiles[i]
            if x in operators or y in terminals or z in shifts:
                valid = False
                break
            operators.add(x)
            terminals.add(y)
            shifts.add(z)
        
        if valid and len(operators) == k and len(terminals) == k and len(shifts) == k:
            print("YES")
            for i in combo:
                x, y, z = profiles[i]
                print(x, y, z)
            return
    
    print("NO")

solve()
