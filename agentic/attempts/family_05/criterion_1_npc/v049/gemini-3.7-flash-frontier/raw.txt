import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    if t < k:
        print("NO")
        return
        
    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx+1]) - 1
        z = int(input_data[idx+2]) - 1
        idx += 3
        triples.append((x, y, z))
    
    # Store options for each element
    # x in [0, k-1], y in [0, k-1], z in [0, k-1]
    by_x = [[] for _ in range(k)]
    by_y = [[] for _ in range(k)]
    by_z = [[] for _ in range(k)]
    
    for x, y, z in triples:
        by_x[x].append((x, y, z))
        by_y[y].append((x, y, z))
        by_z[z].append((x, y, z))
        
    # Check if any element has degree 0
    for i in range(k):
        if not by_x[i] or not by_y[i] or not by_z[i]:
            print("NO")
            return

    used_x = 0
    used_y = 0
    used_z = 0
    solution = []
    
    # DLX-style MRV backtracking
    def search(depth):
        nonlocal used_x, used_y, used_z
        if depth == k:
            return True
        
        # Find item with minimum remaining valid triples
        best_count = 1000000
        best_item_type = -1
        best_item_val = -1
        best_candidates = None
        
        # Check operators
        for x in range(k):
            if not (used_x & (1 << x)):
                cnt = 0
                cands = []
                for tr in by_x[x]:
                    if not (used_y & (1 << tr[1])) and not (used_z & (1 << tr[2])):
                        cnt += 1
                        cands.append(tr)
                if cnt == 0:
                    return False
                if cnt < best_count:
                    best_count = cnt
                    best_candidates = cands
                    if cnt == 1:
                        break
        
        if best_count > 1:
            # Check terminals
            for y in range(k):
                if not (used_y & (1 << y)):
                    cnt = 0
                    cands = []
                    for tr in by_y[y]:
                        if not (used_x & (1 << tr[0])) and not (used_z & (1 << tr[2])):
                            cnt += 1
                            cands.append(tr)
                    if cnt == 0:
                        return False
                    if cnt < best_count:
                        best_count = cnt
                        best_candidates = cands
                        if cnt == 1:
                            break

        if best_count > 1:
            # Check shifts
            for z in range(k):
                if not (used_z & (1 << z)):
                    cnt = 0
                    cands = []
                    for tr in by_z[z]:
                        if not (used_x & (1 << tr[0])) and not (used_y & (1 << tr[1])):
                            cnt += 1
                            cands.append(tr)
                    if cnt == 0:
                        return False
                    if cnt < best_count:
                        best_count = cnt
                        best_candidates = cands
                        if cnt == 1:
                            break
        
        for x, y, z in best_candidates:
            if (used_x & (1 << x)) or (used_y & (1 << y)) or (used_z & (1 << z)):
                continue
            used_x |= (1 << x)
            used_y |= (1 << y)
            used_z |= (1 << z)
            solution.append((x + 1, y + 1, z + 1))
            
            if search(depth + 1):
                return True
                
            solution.pop()
            used_x &= ~(1 << x)
            used_y &= ~(1 << y)
            used_z &= ~(1 << z)
            
        return False

    if search(0):
        print("YES")
        for x, y, z in solution:
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == '__main__':
    solve()
