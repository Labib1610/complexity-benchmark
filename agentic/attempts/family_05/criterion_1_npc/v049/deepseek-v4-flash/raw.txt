We need to find a perfect matching in a 3-partite hypergraph where each edge is a triple (operator, terminal, shift), and we need to cover all k operators, k terminals, and k shifts exactly once.

Since k ≤ 22, we can use a bitmask DP approach over two of the three dimensions, say operators and terminals, and check if we can cover all shifts.

We can represent each profile as a bitmask for operator and terminal, and store which shift it uses. Then we do DP over operator-terminal pairs.

Here’s the implementation:

```python
import sys

def solve():
    input = sys.stdin.readline
    k, t = map(int, input().split())
    profiles = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        x -= 1
        y -= 1
        z -= 1
        profiles.append((x, y, z))
    
    # dp[mask_operators][mask_terminals] = mask_shifts or None
    # We'll store as dict for sparse DP
    dp = {}
    dp[(0, 0)] = 0  # (op_mask, term_mask) -> shift_mask
    
    # For each profile, try to add it
    for x, y, z in profiles:
        new_dp = dict(dp)
        for (om, tm), sm in dp.items():
            if (om >> x) & 1 or (tm >> y) & 1 or (sm >> z) & 1:
                continue
            new_om = om | (1 << x)
            new_tm = tm | (1 << y)
            new_sm = sm | (1 << z)
            key = (new_om, new_tm)
            if key not in new_dp or new_dp[key] != new_sm:
                # We need to store the path, so we'll store parent info separately
                new_dp[key] = new_sm
        dp = new_dp
    
    full_mask = (1 << k) - 1
    if (full_mask, full_mask) in dp and dp[(full_mask, full_mask)] == full_mask:
        print("YES")
        # Reconstruct
        # We need to store parent for reconstruction
        # Let's redo with parent tracking
        dp = {}
        parent = {}
        dp[(0, 0)] = 0
        for idx, (x, y, z) in enumerate(profiles):
            new_dp = dict(dp)
            for (om, tm), sm in dp.items():
                if (om >> x) & 1 or (tm >> y) & 1 or (sm >> z) & 1:
                    continue
                new_om = om | (1 << x)
                new_tm = tm | (1 << y)
                new_sm = sm | (1 << z)
                key = (new_om, new_tm)
                if key not in new_dp:
                    new_dp[key] = new_sm
                    parent[key] = ((om, tm), idx)
            dp = new_dp
        
        # Reconstruct
        cur = (full_mask, full_mask)
        chosen = []
        while cur != (0, 0):
            prev, idx = parent[cur]
            x, y, z = profiles[idx]
            chosen.append((x+1, y+1, z+1))
            cur = prev
        for x, y, z in chosen:
            print(x, y, z)
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
