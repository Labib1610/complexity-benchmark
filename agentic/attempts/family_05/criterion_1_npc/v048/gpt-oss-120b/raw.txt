```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    rows = []
    # column indexing: chassis 0..k-1, core k..2k-1, payload 2k..3k-1
    for _ in range(t):
        c = int(next(it)) - 1
        e = int(next(it)) - 1
        g = int(next(it)) - 1
        mask = (1 << c) | (1 << (k + e)) | (1 << (2 * k + g))
        rows.append((mask, c, e, g))
    total_cols = 3 * k
    full_mask = (1 << total_cols) - 1

    # build column -> list of row indices
    col_rows = [[] for _ in range(total_cols)]
    for idx, (mask, _, _, _) in enumerate(rows):
        # each mask has exactly three bits
        m = mask
        while m:
            lsb = m & -m
            col = (lsb.bit_length() - 1)
            col_rows[col].append(idx)
            m ^= lsb

    # order columns by degree (fewest rows first) for heuristic
    col_order = sorted(range(total_cols), key=lambda c: len(col_rows[c]))

    sys.setrecursionlimit(10000)
    solution = []

    # cache for quick conflict check
    row_masks = [mask for mask, *_ in rows]

    def dfs(used_mask: int) -> bool:
        if used_mask == full_mask:
            return True
        # choose uncovered column with minimal candidates
        # iterate col_order to find first uncovered
        best_col = -1
        best_len = 10**9
        for col in col_order:
            if not (used_mask >> col) & 1:
                # count viable rows
                cnt = 0
                for r in col_rows[col]:
                    if row_masks[r] & used_mask == 0:
                        cnt += 1
                if cnt == 0:
                    return False
                if cnt < best_len:
                    best_len = cnt
                    best_col = col
                # early break if cnt ==1
                if cnt == 1:
                    break
        # try each viable row for best_col
        for r in col_rows[best_col]:
            rm = row_masks[r]
            if rm & used_mask:
                continue
            solution.append(r)
            if dfs(used_mask | rm):
                return True
            solution.pop()
        return False

    if dfs(0):
        out = ["YES"]
        for idx in solution:
            _, c, e, g = rows[idx]
            out.append(f"{c+1} {e+1} {g+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
