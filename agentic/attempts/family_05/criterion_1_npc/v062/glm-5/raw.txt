import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    
    pathways = []
    for _ in range(t):
        x = int(next(it))
        y = int(next(it))
        z = int(next(it))
        pathways.append((x, y, z))
    
    # We need to select k pathways such that:
    # - All x are distinct (cover 1..k)
    # - All y are distinct (cover 1..k)
    # - All z are distinct (cover 1..k)
    
    # Group pathways by source x
    by_x = [[] for _ in range(k + 1)]
    for idx, (x, y, z) in enumerate(pathways):
        by_x[x].append((y, z, idx))
    
    # State: (used_y_mask, used_z_mask)
    # We process sources x from 1 to k
    # DP: for each state, store the list of pathway indices used
    
    # Since k <= 22, masks fit in 32-bit integers
    # DP maps (used_y_mask, used_z_mask) -> list of pathway indices
    
    dp = {}
    dp[(0, 0)] = []
    
    for x in range(1, k + 1):
        new_dp = {}
        for (used_y, used_z), path_indices in dp.items():
            for y, z, idx in by_x[x]:
                if (used_y & (1 << y)) == 0 and (used_z & (1 << z)) == 0:
                    new_y = used_y | (1 << y)
                    new_z = used_z | (1 << z)
                    new_state = (new_y, new_z)
                    if new_state not in new_dp:
                        new_dp[new_state] = path_indices + [idx]
        dp = new_dp
        if not dp:
            print("NO")
            return
    
    if dp:
        # Any remaining state is valid (all sources covered)
        final_indices = next(iter(dp.values()))
        print("YES")
        for idx in final_indices:
            x, y, z = pathways[idx]
            print(f"{x} {y} {z}")
    else:
        print("NO")

if __name__ == "__main__":
    solve()
