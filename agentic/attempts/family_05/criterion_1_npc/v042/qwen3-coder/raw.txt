from collections import defaultdict
import sys

def find_matching(k, combinations):
    # Create a graph for bipartite matching
    # We'll use a backtracking approach with pruning for efficiency
    
    # Build adjacency list for combinations
    adj = defaultdict(list)
    for i, (p1, p2, p3) in enumerate(combinations):
        adj[p1].append(i)
    
    # Try to find a perfect matching using backtracking
    used_indices = [False] * len(combinations)
    selected = []
    tower_used = [False] * (k + 1)
    key_used = [False] * (k + 1)
    link_used = [False] * (k + 1)
    
    def backtrack(pos):
        if pos == k:
            return True
            
        # Try each combination for position pos
        for i in range(len(combinations)):
            if used_indices[i]:
                continue
                
            p1, p2, p3 = combinations[i]
            
            # Check if this combination can be used
            if tower_used[p1] or key_used[p2] or link_used[p3]:
                continue
            
            # Use this combination
            used_indices[i] = True
            selected.append((p1, p2, p3))
            tower_used[p1] = True
            key_used[p2] = True
            link_used[p3] = True
            
            # Recurse
            if backtrack(pos + 1):
                return True
                
            # Backtrack
            used_indices[i] = False
            selected.pop()
            tower_used[p1] = False
            key_used[p2] = False
            link_used[p3] = False
            
        return False
    
    # Optimization: sort combinations to try more constrained options first
    # Count how many times each component appears
    tower_count = [0] * (k + 1)
    key_count = [0] * (k + 1)
    link_count = [0] * (k + 1)
    
    for p1, p2, p3 in combinations:
        tower_count[p1] += 1
        key_count[p2] += 1
        link_count[p3] += 1
    
    # Sort combinations by the minimum count of their components (most constrained first)
    combinations.sort(key=lambda x: min(tower_count[x[0]], key_count[x[1]], link_count[x[2]]))
    
    if backtrack(0):
        return selected
    else:
        return None

def main():
    input_lines = []
    for line in sys.stdin:
        input_lines.append(line.strip())
    
    k, t = map(int, input_lines[0].split())
    combinations = []
    
    for i in range(1, t + 1):
        p1, p2, p3 = map(int, input_lines[i].split())
        combinations.append((p1, p2, p3))
    
    result = find_matching(k, combinations)
    
    if result is not None:
        print("YES")
        for p1, p2, p3 in result:
            print(p1, p2, p3)
    else:
        print("NO")

if __name__ == "__main__":
    main()
