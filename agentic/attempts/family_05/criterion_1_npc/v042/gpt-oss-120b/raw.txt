```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    edges = []
    by_tower = [[] for _ in range(k + 1)]
    for idx in range(t):
        a = int(next(it)); b = int(next(it)); c = int(next(it))
        edges.append((a, b, c))
        by_tower[a].append((b, c, idx))

    used_tower = [False] * (k + 1)
    used_key = [False] * (k + 1)
    used_link = [False] * (k + 1)
    solution = [None] * k
    found = False

    # precompute list of towers
    towers = list(range(1, k + 1))

    def select_tower():
        # choose unused tower with fewest viable edges
        best = None
        best_cnt = None
        for tw in towers:
            if used_tower[tw]:
                continue
            cnt = 0
            for key, link, _ in by_tower[tw]:
                if not used_key[key] and not used_link[link]:
                    cnt += 1
            if cnt == 0:
                return tw, 0  # dead end quickly
            if best_cnt is None or cnt < best_cnt:
                best_cnt = cnt
                best = tw
        return best, best_cnt

    def dfs(depth: int) -> bool:
        nonlocal found
        if depth == k:
            return True
        tw, cnt = select_tower()
        if cnt == 0:
            return False
        used_tower[tw] = True
        for key, link, idx in by_tower[tw]:
            if used_key[key] or used_link[link]:
                continue
            used_key[key] = used_link[link] = True
            solution[depth] = edges[idx]
            if dfs(depth + 1):
                return True
            used_key[key] = used_link[link] = False
        used_tower[tw] = False
        return False

    if dfs(0):
        out = ["YES"]
        for a, b, c in solution:
            out.append(f"{a} {b} {c}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
```
