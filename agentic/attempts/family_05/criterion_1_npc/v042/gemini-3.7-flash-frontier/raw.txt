import sys

# Increase recursion depth just in case
sys.setrecursionlimit(20000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    if t < k:
        print("NO")
        return
    
    # Each option has 3 items:
    # 1. Tower p1 (0 .. k-1)
    # 2. Key p2 (k .. 2k-1)
    # 3. Link p3 (2k .. 3k-1)
    
    num_items = 3 * k
    options = []
    
    idx = 2
    for _ in range(t):
        p1 = int(input_data[idx]) - 1
        p2 = int(input_data[idx+1]) - 1
        p3 = int(input_data[idx+2]) - 1
        idx += 3
        options.append((p1, k + p2, 2 * k + p3))
    
    # Pre-filter options to be unique (problem says no combination listed more than once)
    # DLX implementation
    # We build the Dancing Links structure
    
    # Column headers: 0 to num_items (0 is root)
    # Items: 1 to num_items
    
    max_nodes = num_items + 1 + len(options) * 3 + 10
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    sz = [0] * (num_items + 1)
    
    # Initialize column headers
    for i in range(num_items + 1):
        L[i] = (i - 1 + num_items + 1) % (num_items + 1)
        R[i] = (i + 1) % (num_items + 1)
        U[i] = i
        D[i] = i
        col[i] = i
        sz[i] = 0
        
    node_count = num_items + 1
    
    for r_idx, opt in enumerate(options):
        first_node = -1
        for item in opt:
            c = item + 1  # 1-indexed item
            n = node_count
            node_count += 1
            
            col[n] = c
            row[n] = r_idx
            sz[c] += 1
            
            # Insert into column c
            U[n] = U[c]
            D[n] = c
            D[U[c]] = n
            U[c] = n
            
            # Insert into row
            if first_node == -1:
                first_node = n
                L[n] = n
                R[n] = n
            else:
                L[n] = L[first_node]
                R[n] = first_node
                R[L[first_node]] = n
                L[first_node] = n

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                sz[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                sz[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum size
        c = R[0]
        min_sz = sz[c]
        curr = R[c]
        while curr != 0:
            if sz[curr] < min_sz:
                min_sz = sz[curr]
                c = curr
            curr = R[curr]
        
        if min_sz == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]
            
            if search():
                return True
            
            solution.pop()
            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            r = D[r]
        
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r_idx in solution:
            p1, p2, p3 = options[r_idx]
            out.append(f"{p1 + 1} {p2 - k + 1} {p3 - 2 * k + 1}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
