import sys

sys.setrecursionlimit(1_000_000)


class DSU:
    def __init__(self, n):
        self.parent = list(range(n))
        self.size = [1] * n

    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a, b):
        a = self.find(a)
        b = self.find(b)
        if a == b:
            return
        if self.size[a] < self.size[b]:
            a, b = b, a
        self.parent[b] = a
        self.size[a] += self.size[b]


def solve_component(xs, ys, zs, component_edges):
    n = len(xs)

    x_index = {v: i for i, v in enumerate(xs)}
    y_index = {v: i for i, v in enumerate(ys)}
    z_index = {v: i for i, v in enumerate(zs)}

    edges = []
    for x, y, z in component_edges:
        edges.append((x_index[x], y_index[y], z_index[z], (x, y, z)))

    m = len(edges)
    adj = [0] * (3 * n)

    for eid, (x, y, z, _) in enumerate(edges):
        bit = 1 << eid
        adj[x] |= bit
        adj[n + y] |= bit
        adj[2 * n + z] |= bit

    if any(mask == 0 for mask in adj):
        return None

    conflicts = [0] * m
    vertex_masks = [0] * m
    for eid, (x, y, z, _) in enumerate(edges):
        conflicts[eid] = adj[x] | adj[n + y] | adj[2 * n + z]
        vertex_masks[eid] = (1 << x) | (1 << (n + y)) | (1 << (2 * n + z))

    xy = [[0] * n for _ in range(n)]
    xz = [[0] * n for _ in range(n)]
    yz = [[0] * n for _ in range(n)]

    for i in range(n):
        ai = adj[i]
        for j in range(n):
            xy[i][j] = ai & adj[n + j]
            xz[i][j] = ai & adj[2 * n + j]

    for i in range(n):
        ai = adj[n + i]
        for j in range(n):
            yz[i][j] = ai & adj[2 * n + j]

    def matching_possible(active, row_mask, col_mask, pair_masks):
        options = []
        rb = row_mask
        while rb:
            lsb = rb & -rb
            r = lsb.bit_length() - 1
            rb ^= lsb

            available = 0
            cb = col_mask
            while cb:
                clsb = cb & -cb
                c = clsb.bit_length() - 1
                cb ^= clsb
                if active & pair_masks[r][c]:
                    available |= clsb

            if available == 0:
                return False
            options.append((available.bit_count(), r, available))

        options.sort()
        match = [-1] * n

        def augment(r, opts, seen):
            bits = opts & ~seen[0]
            while bits:
                bit = bits & -bits
                bits ^= bit
                c = bit.bit_length() - 1
                seen[0] |= bit
                previous = match[c]
                if previous == -1 or augment(previous[0], previous[1], seen):
                    match[c] = (r, opts)
                    return True
            return False

        for _, r, opts in options:
            if not augment(r, opts, [0]):
                return False
        return True

    full_group = (1 << n) - 1
    all_uncovered = (1 << (3 * n)) - 1
    all_active = (1 << m) - 1

    path = []
    failed = set()
    memo_limit = 600_000

    def dfs(active, uncovered, depth):
        if uncovered == 0:
            return True

        if uncovered in failed:
            return False

        ux = uncovered & full_group
        uy = (uncovered >> n) & full_group
        uz = (uncovered >> (2 * n)) & full_group

        remaining = ux.bit_count()
        if remaining >= 4 and depth % 3 == 0:
            if not matching_possible(active, ux, uy, xy):
                if len(failed) < memo_limit:
                    failed.add(uncovered)
                return False
            if not matching_possible(active, ux, uz, xz):
                if len(failed) < memo_limit:
                    failed.add(uncovered)
                return False
            if not matching_possible(active, uy, uz, yz):
                if len(failed) < memo_limit:
                    failed.add(uncovered)
                return False

        best_candidates = 0
        best_count = m + 1

        bits = uncovered
        while bits:
            bit = bits & -bits
            bits ^= bit
            v = bit.bit_length() - 1
            candidates = active & adj[v]
            count = candidates.bit_count()

            if count == 0:
                if len(failed) < memo_limit:
                    failed.add(uncovered)
                return False

            if count < best_count:
                best_count = count
                best_candidates = candidates
                if count == 1:
                    break

        candidate_ids = []
        bits = best_candidates
        while bits:
            bit = bits & -bits
            bits ^= bit
            candidate_ids.append(bit.bit_length() - 1)

        if 1 < len(candidate_ids) <= 32:
            scored = []
            for eid in candidate_ids:
                x, y, z, _ = edges[eid]
                score = (
                    (active & adj[x]).bit_count()
                    + (active & adj[n + y]).bit_count()
                    + (active & adj[2 * n + z]).bit_count()
                )
                scored.append((score, eid))
            scored.sort()
            candidate_ids = [eid for _, eid in scored]

        for eid in candidate_ids:
            path.append(eid)
            if dfs(
                active & ~conflicts[eid],
                uncovered & ~vertex_masks[eid],
                depth + 1
            ):
                return True
            path.pop()

        if len(failed) < memo_limit:
            failed.add(uncovered)
        return False

    if not dfs(all_active, all_uncovered, 0):
        return None

    return [edges[eid][3] for eid in path]


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    triples = []
    dsu = DSU(3 * k)

    p = 2
    for _ in range(t):
        x = data[p] - 1
        y = data[p + 1] - 1
        z = data[p + 2] - 1
        p += 3

        triples.append((x, y, z))
        dsu.union(x, k + y)
        dsu.union(x, 2 * k + z)

    vertices_by_root = {}
    for v in range(3 * k):
        root = dsu.find(v)
        vertices_by_root.setdefault(root, []).append(v)

    edges_by_root = {}
    for edge in triples:
        root = dsu.find(edge[0])
        edges_by_root.setdefault(root, []).append(edge)

    components = []

    for root, vertices in vertices_by_root.items():
        xs = []
        ys = []
        zs = []

        for v in vertices:
            if v < k:
                xs.append(v)
            elif v < 2 * k:
                ys.append(v - k)
            else:
                zs.append(v - 2 * k)

        if not (len(xs) == len(ys) == len(zs)):
            print("NO")
            return

        component_edges = edges_by_root.get(root, [])
        if not component_edges:
            print("NO")
            return

        components.append((xs, ys, zs, component_edges))

    components.sort(key=lambda c: (len(c[3]) / len(c[0]), len(c[0])))

    answer = []
    for xs, ys, zs, component_edges in components:
        result = solve_component(xs, ys, zs, component_edges)
        if result is None:
            print("NO")
            return
        answer.extend(result)

    if len(answer) != k:
        print("NO")
        return

    out = ["YES"]
    for x, y, z in answer:
        out.append(f"{x + 1} {y + 1} {z + 1}")
    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()
