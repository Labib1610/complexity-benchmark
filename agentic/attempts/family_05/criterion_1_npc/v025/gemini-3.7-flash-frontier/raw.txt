import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    if t < k:
        print("NO")
        return

    triples_by_x = [[] for _ in range(k)]
    triples_by_y = [[] for _ in range(k)]
    triples_by_z = [[] for _ in range(k)]

    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3
        triples_by_x[x].append((y, z))
        triples_by_y[y].append((x, z))
        triples_by_z[z].append((x, y))

    # Quick check: every item must have at least one triple
    for i in range(k):
        if (
            not triples_by_x[i]
            or not triples_by_y[i]
            or not triples_by_z[i]
        ):
            print("NO")
            return

    solution = []

    def backtrack(used_x, used_y, used_z, count):
        if count == k:
            return True

        # Find the constraint with minimum remaining choices (MRV)
        best_type = -1
        best_val = -1
        min_choices = 10**9
        best_candidates = []

        # Check X
        for x in range(k):
            if not (used_x & (1 << x)):
                valid = []
                for y, z in triples_by_x[x]:
                    if not (used_y & (1 << y)) and not (used_z & (1 << z)):
                        valid.append((x, y, z))
                c = len(valid)
                if c == 0:
                    return False
                if c < min_choices:
                    min_choices = c
                    best_candidates = valid
                    if min_choices == 1:
                        break

        if min_choices > 1:
            # Check Y
            for y in range(k):
                if not (used_y & (1 << y)):
                    valid = []
                    for x, z in triples_by_y[y]:
                        if not (used_x & (1 << x)) and not (used_z & (1 << z)):
                            valid.append((x, y, z))
                    c = len(valid)
                    if c == 0:
                        return False
                    if c < min_choices:
                        min_choices = c
                        best_candidates = valid
                        if min_choices == 1:
                            break

        if min_choices > 1:
            # Check Z
            for z in range(k):
                if not (used_z & (1 << z)):
                    valid = []
                    for x, y in triples_by_z[z]:
                        if not (used_x & (1 << x)) and not (used_y & (1 << y)):
                            valid.append((x, y, z))
                    c = len(valid)
                    if c == 0:
                        return False
                    if c < min_choices:
                        min_choices = c
                        best_candidates = valid
                        if min_choices == 1:
                            break

        for x, y, z in best_candidates:
            solution.append((x + 1, y + 1, z + 1))
            if backtrack(
                used_x | (1 << x),
                used_y | (1 << y),
                used_z | (1 << z),
                count + 1,
            ):
                return True
            solution.pop()

        return False

    if backtrack(0, 0, 0, 0):
        print("YES")
        for x, y, z in solution:
            print(f"{x} {y} {z}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
