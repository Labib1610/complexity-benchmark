import sys

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

k, t = data[0], data[1]
xs = [0] * t
ys = [0] * t
zs = [0] * t
inc = [0] * (3 * k)

pos = 2
for i in range(t):
    x = data[pos] - 1
    y = data[pos + 1] - 1
    z = data[pos + 2] - 1
    pos += 3
    xs[i] = x
    ys[i] = y
    zs[i] = z
    bit = 1 << i
    inc[x] |= bit
    inc[k + y] |= bit
    inc[2 * k + z] |= bit

full_resources = (1 << k) - 1
all_rows = (1 << t) - 1


def gf2_certificate_impossible(avail, mx, my, mz):
    basis = [0] * (3 * k + 1)
    bits = avail

    while bits:
        lb = bits & -bits
        r = lb.bit_length() - 1
        bits ^= lb

        v = (
            (1 << xs[r])
            | (1 << (k + ys[r]))
            | (1 << (2 * k + zs[r]))
            | (1 << (3 * k))
        )

        while v:
            p = v.bit_length() - 1
            if basis[p]:
                v ^= basis[p]
            else:
                basis[p] = v
                break

    target = 0
    u = full_resources ^ mx
    while u:
        lb = u & -u
        target |= lb
        u ^= lb

    u = full_resources ^ my
    while u:
        lb = u & -u
        target |= 1 << (k + lb.bit_length() - 1)
        u ^= lb

    u = full_resources ^ mz
    while u:
        lb = u & -u
        target |= 1 << (2 * k + lb.bit_length() - 1)
        u ^= lb

    remaining = k - mx.bit_count()
    if remaining & 1:
        target |= 1 << (3 * k)

    while target:
        p = target.bit_length() - 1
        if not basis[p]:
            return True
        target ^= basis[p]

    return False


def modular_certificate_impossible(p):
    n = 3 * k + 1
    basis = [None] * n
    rank = 0

    for r in range(t):
        v = [0] * n
        v[xs[r]] = 1
        v[k + ys[r]] = 1
        v[2 * k + zs[r]] = 1
        v[3 * k] = p - 1

        for j in range(n):
            a = v[j]
            if a == 0:
                continue
            b = basis[j]
            if b is None:
                inv = pow(a, p - 2, p)
                for q in range(j, n):
                    v[q] = (v[q] * inv) % p
                basis[j] = v
                rank += 1
                break

            for q in range(j, n):
                v[q] = (v[q] - a * b[q]) % p

        if rank == n:
            return False

    target = [1] * (3 * k) + [(-k) % p]
    for j in range(n):
        a = target[j]
        if a == 0:
            continue
        b = basis[j]
        if b is None:
            return True
        for q in range(j, n):
            target[q] = (target[q] - a * b[q]) % p

    return False


if t == 0:
    print("NO")
    sys.exit()

if gf2_certificate_impossible(all_rows, 0, 0, 0):
    print("NO")
    sys.exit()

for prime in (3, 5, 7):
    if modular_certificate_impossible(prime):
        print("NO")
        sys.exit()


def has_perfect_matching(adj, left_mask, right_mask):
    match = [-1] * k

    left_vertices = []
    u = left_mask
    while u:
        lb = u & -u
        a = lb.bit_length() - 1
        u ^= lb
        degree = (adj[a] & right_mask).bit_count()
        if degree == 0:
            return False
        left_vertices.append((degree, a))

    left_vertices.sort()

    def augment(a, seen):
        options = adj[a] & right_mask & ~seen[0]
        while options:
            lb = options & -options
            b = lb.bit_length() - 1
            options ^= lb
            seen[0] |= lb
            if match[b] == -1 or augment(match[b], seen):
                match[b] = a
                return True
        return False

    for _, a in left_vertices:
        seen = [0]
        if not augment(a, seen):
            return False
    return True


def hall_check(avail, mx, my, mz):
    ux = full_resources ^ mx
    uy = full_resources ^ my
    uz = full_resources ^ mz

    adj_xy = [0] * k
    adj_xz = [0] * k
    adj_yz = [0] * k

    bits = avail
    while bits:
        lb = bits & -bits
        r = lb.bit_length() - 1
        bits ^= lb
        x = xs[r]
        y = ys[r]
        z = zs[r]
        adj_xy[x] |= 1 << y
        adj_xz[x] |= 1 << z
        adj_yz[y] |= 1 << z

    return (
        has_perfect_matching(adj_xy, ux, uy)
        and has_perfect_matching(adj_xz, ux, uz)
        and has_perfect_matching(adj_yz, uy, uz)
    )


failed = set()
answer = []


def dfs(avail, mx, my, mz, depth):
    if depth == k:
        return True

    state = mx | (my << k) | (mz << (2 * k))
    if state in failed:
        return False

    best_rows = 0
    best_count = t + 1
    counts = [0] * (3 * k)

    ux = full_resources ^ mx
    uy = full_resources ^ my
    uz = full_resources ^ mz

    u = ux
    while u:
        lb = u & -u
        x = lb.bit_length() - 1
        u ^= lb
        rows = inc[x] & avail
        cnt = rows.bit_count()
        if cnt == 0:
            failed.add(state)
            return False
        counts[x] = cnt
        if cnt < best_count:
            best_count = cnt
            best_rows = rows

    u = uy
    while u:
        lb = u & -u
        y = lb.bit_length() - 1
        u ^= lb
        idx = k + y
        rows = inc[idx] & avail
        cnt = rows.bit_count()
        if cnt == 0:
            failed.add(state)
            return False
        counts[idx] = cnt
        if cnt < best_count:
            best_count = cnt
            best_rows = rows

    u = uz
    while u:
        lb = u & -u
        z = lb.bit_length() - 1
        u ^= lb
        idx = 2 * k + z
        rows = inc[idx] & avail
        cnt = rows.bit_count()
        if cnt == 0:
            failed.add(state)
            return False
        counts[idx] = cnt
        if cnt < best_count:
            best_count = cnt
            best_rows = rows

    remaining = k - depth

    if depth == 0 or (remaining <= 16 and depth % 3 == 0):
        if not hall_check(avail, mx, my, mz):
            failed.add(state)
            return False

    if remaining <= 12 and depth % 2 == 0:
        if gf2_certificate_impossible(avail, mx, my, mz):
            failed.add(state)
            return False

    candidates = []
    bits = best_rows
    while bits:
        lb = bits & -bits
        r = lb.bit_length() - 1
        bits ^= lb
        score = counts[xs[r]] + counts[k + ys[r]] + counts[2 * k + zs[r]]
        candidates.append((score, r))

    candidates.sort()

    for _, r in candidates:
        x = xs[r]
        y = ys[r]
        z = zs[r]

        blocked = inc[x] | inc[k + y] | inc[2 * k + z]
        next_avail = avail & ~blocked

        answer.append(r)
        if dfs(
            next_avail,
            mx | (1 << x),
            my | (1 << y),
            mz | (1 << z),
            depth + 1,
        ):
            return True
        answer.pop()

    failed.add(state)
    return False


if dfs(all_rows, 0, 0, 0, 0):
    out = ["YES"]
    for r in answer:
        out.append(f"{xs[r] + 1} {ys[r] + 1} {zs[r] + 1}")
    print("\n".join(out))
else:
    print("NO")
