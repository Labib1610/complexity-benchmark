An optimal and efficient solution in Python 3 using Knuth's Algorithm X (Dancing Links / DLX) to solve the 3-Dimensional Matching problem.

```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    k = int(input_data[0])
    t = int(input_data[1])
    
    # 3 categories of size k:
    # Satellites: 1 .. k
    # Frequencies: k+1 .. 2k
    # Keys: 2k+1 .. 3k
    num_cols = 3 * k
    
    # DLX structures using flat lists (arrays)
    # Head node is 0. Column headers are 1 .. num_cols.
    # Data nodes start at num_cols + 1.
    max_nodes = num_cols + 1 + 3 * t + 5
    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    C = [0] * max_nodes
    S = [0] * (num_cols + 1)
    row_idx = [0] * max_nodes
    
    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = (i - 1 + num_cols + 1) % (num_cols + 1)
        R[i] = (i + 1) % (num_cols + 1)
        U[i] = i
        D[i] = i
    
    node_cnt = num_cols
    triplets = []
    
    idx = 2
    for r in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx + 1])
        z = int(input_data[idx + 2])
        idx += 3
        
        triplets.append((x, y, z))
        
        cols = [x, k + y, 2 * k + z]
        first_node = None
        
        for c in cols:
            node_cnt += 1
            node = node_cnt
            C[node] = c
            row_idx[node] = r
            S[c] += 1
            
            # Link vertically
            U[node] = U[c]
            D[node] = c
            D[U[c]] = node
            U[c] = node
            
            # Link horizontally
            if first_node is None:
                first_node = node
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first_node]
                R[node] = first_node
                R[L[first_node]] = node
                L[first_node] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    solution = []

    def search():
        if R[0] == 0:
            return True
        
        # Choose column with minimum remaining rows (MRV heuristic)
        min_s = float('inf')
        c = 0
        cur = R[0]
        while cur != 0:
            if S[cur] < min_s:
                min_s = S[cur]
                c = cur
                if min_s == 0:
                    break
            cur = R[cur]
            
        if min_s == 0:
            return False
        
        cover(c)
        r = D[c]
        while r != c:
            solution.append(row_idx[r])
            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]
            
            if search():
                return True
            
            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]
            solution.pop()
            r = D[r]
            
        uncover(c)
        return False

    if search():
        print("YES")
        out = []
        for r in solution:
            x, y, z = triplets[r]
            out.append(f"{x} {y} {z}")
        print("\n".join(out))
    else:
        print("NO")

if __name__ == '__main__':
    sys.setrecursionlimit(2000)
    solve()
```
