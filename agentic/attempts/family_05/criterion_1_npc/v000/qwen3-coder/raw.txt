from itertools import permutations

def solve():
    k, t = map(int, input().split())
    
    # Read all allowed combinations
    allowed = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        allowed.append((x, y, z))
    
    # If there are fewer than k allowed combinations, impossible
    if t < k:
        print("NO")
        return
    
    # Try all combinations of k allowed triples
    from itertools import combinations
    
    # For small k, we can try all combinations
    # But since k <= 22, we need a smarter approach
    
    # Model as 3D matching problem - use backtracking
    # We'll try to assign each specialist to a unique (machine, time) pair
    
    # Create mappings
    # For each specialist x, keep track of allowed (y,z) pairs
    allowed_for_x = [[] for _ in range(k + 1)]
    # For each (y,z) pair, keep track of allowed x's
    allowed_xyz = {}
    
    for x, y, z in allowed:
        allowed_for_x[x].append((y, z))
        if (y, z) not in allowed_xyz:
            allowed_xyz[(y, z)] = []
        allowed_xyz[(y, z)].append(x)
    
    # Try to find a matching using backtracking
    # assignment[i] will store the (y,z) assigned to specialist i
    assignment = [None] * (k + 1)
    # Keep track of used machines and time slots
    used_y = [False] * (k + 1)
    used_z = [False] * (k + 1)
    
    def backtrack(x):
        if x > k:
            return True  # All specialists assigned
        
        # Try each allowed (y,z) for specialist x
        for y, z in allowed_for_x[x]:
            if not used_y[y] and not used_z[z]:
                # Assign this combination
                assignment[x] = (y, z)
                used_y[y] = True
                used_z[z] = True
                
                # Recurse
                if backtrack(x + 1):
                    return True
                
                # Backtrack
                used_y[y] = False
                used_z[z] = False
                assignment[x] = None
        
        return False
    
    # Start backtracking from specialist 1
    if backtrack(1):
        print("YES")
        for x in range(1, k + 1):
            y, z = assignment[x]
            print(x, y, z)
    else:
        print("NO")

solve()
