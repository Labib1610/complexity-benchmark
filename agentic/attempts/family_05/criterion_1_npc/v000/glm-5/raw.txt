import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    triples = []
    for _ in range(t):
        x = int(data[idx]); y = int(data[idx+1]); z = int(data[idx+2])
        idx += 3
        triples.append((x, y, z))
    
    # For each (x, y), collect list of z values
    xy_to_z = {}
    for (x, y, z) in triples:
        key = (x, y)
        if key in xy_to_z:
            xy_to_z[key].append(z)
        else:
            xy_to_z[key] = [z]
    
    # For each x, collect list of y values that have at least one z
    x_to_y = {}
    for (x, y) in xy_to_z:
        if x in x_to_y:
            x_to_y[x].append(y)
        else:
            x_to_y[x] = [y]
    
    # Check feasibility: each x, y, z must appear in at least one triple
    x_set = set()
    y_set = set()
    z_set = set()
    for (x, y, z) in triples:
        x_set.add(x)
        y_set.add(y)
        z_set.add(z)
    if len(x_set) < k or len(y_set) < k or len(z_set) < k:
        print("NO")
        return
    
    # We need to select k triples forming a perfect matching in 3D.
    # Approach: iterate over all permutations of y for x=1..k (k! up to 22! too large)
    # Instead use backtracking with pruning.
    
    # Order x by fewest available y options (heuristic)
    xs = list(range(1, k+1))
    xs.sort(key=lambda xx: len(x_to_y.get(xx, [])))
    
    y_used = [False] * (k + 1)
    z_used = [False] * (k + 1)
    assignment = [None] * (k + 1)  # assignment[x] = (y, z)
    
    def backtrack(i):
        if i == k:
            return True
        x = xs[i]
        ys = x_to_y.get(x, [])
        for y in ys:
            if y_used[y]:
                continue
            zs = xy_to_z[(x, y)]
            for z in zs:
                if z_used[z]:
                    continue
                y_used[y] = True
                z_used[z] = True
                assignment[x] = (y, z)
                if backtrack(i + 1):
                    return True
                y_used[y] = False
                z_used[z] = False
                assignment[x] = None
        return False
    
    if backtrack(0):
        print("YES")
        for x in range(1, k+1):
            y, z = assignment[x]
            print(f"{x} {y} {z}")
    else:
        print("NO")

solve()
