We can model this as a 3D matching problem, but since k ≤ 22, we can use a backtracking approach with pruning, or reduce it to a bipartite matching problem by fixing one dimension and trying all permutations of another.

A practical approach:  
- Fix the assignment of specialists to machines via a perfect matching in a bipartite graph, then check if the time slots can be assigned consistently.

We can do this by trying all permutations of one group (e.g., machines) and using bipartite matching for the third dimension.

Here’s the code:

```python
import sys
from itertools import permutations

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    k = int(next(it))
    t = int(next(it))
    
    allowed = set()
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        allowed.add((x, y, z))
    
    # For each (x, y), store possible z
    possible_z = [[[] for _ in range(k)] for _ in range(k)]
    for x, y, z in allowed:
        possible_z[x][y].append(z)
    
    # Try all permutations of machines assigned to specialists
    # Use backtracking with pruning
    from itertools import permutations
    
    # We'll try to assign each specialist a machine, then check if we can assign time slots
    # Use bipartite matching for time slots given a machine assignment
    
    def try_assignment(machine_assign):
        # machine_assign[x] = y
        # Build bipartite graph between specialists and time slots
        adj = [[] for _ in range(k)]
        for x in range(k):
            y = machine_assign[x]
            for z in possible_z[x][y]:
                adj[x].append(z)
        # Check if perfect matching exists
        match_to = [-1] * k
        def dfs(u, visited):
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    if match_to[v] == -1 or dfs(match_to[v], visited):
                        match_to[v] = u
                        return True
            return False
        
        for u in range(k):
            visited = [False] * k
            if not dfs(u, visited):
                return None
        # Build result
        result = []
        for z in range(k):
            x = match_to[z]
            y = machine_assign[x]
            result.append((x+1, y+1, z+1))
        return result
    
    # Try all permutations of machines for specialists
    # But k up to 22, permutations too many. Use backtracking with pruning.
    
    # Instead, we can fix specialist order and try all permutations of machines
    # But 22! is huge. Need smarter.
    
    # Use recursive backtracking with pruning: assign machines to specialists one by one
    # and check if remaining can still be matched.
    
    # Precompute for each specialist, list of machines that have at least one z
    machine_options = []
    for x in range(k):
        opts = [y for y in range(k) if possible_z[x][y]]
        machine_options.append(opts)
    
    # Sort specialists by fewest options for pruning
    order = sorted(range(k), key=lambda x: len(machine_options[x]))
    
    machine_assign = [-1] * k
    used_machines = [False] * k
    
    # For quick check: given partial assignment, can we still match time slots?
    # We'll do full matching check at leaf.
    
    def backtrack(idx):
        if idx == k:
            # full assignment
            res = try_assignment(machine_assign)
            if res is not None:
                return res
            return None
        
        x = order[idx]
        for y in machine_options[x]:
            if not used_machines[y]:
                used_machines[y] = True
                machine_assign[x] = y
                res = backtrack(idx + 1)
                if res is not None:
                    return res
                used_machines[y] = False
                machine_assign[x] = -1
        return None
    
    res = backtrack(0)
    if res is None:
        print("NO")
    else:
        print("YES")
        for x, y, z in res:
            print(x, y, z)

if __name__ == "__main__":
    solve()
```
