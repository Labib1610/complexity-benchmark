import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    rows = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        rows.append((x, y, z))

    # DLX structures
    n_cols = 3 * k
    # column header nodes: 0..n_cols-1
    size = [0] * n_cols          # number of nodes in column
    col_head = list(range(n_cols))  # self reference for headers
    # doubly linked list of column headers
    left = [n_cols - 1] * (n_cols + 1)
    right = [0] * (n_cols + 1)
    for i in range(n_cols):
        left[i + 1] = i
        right[i] = i + 1
    left[0] = n_cols
    right[n_cols] = 0  # sentinel header at index n_cols

    # node links
    up = []
    down = []
    col = []
    row_id = []   # which original row this node belongs to

    # helper to add a node to column c
    def add_node(c, r):
        idx = len(up)
        up.append(up[col_head[c]] if col_head[c] < len(up) else idx)
        down.append(col_head[c])
        col.append(c)
        row_id.append(r)
        # insert into column list
        if col_head[c] < len(down):
            up[col_head[c]] = idx
        down[up[col_head[c]]] = idx
        size[c] += 1
        return idx

    # Build nodes
    # For each row, create nodes for its three columns
    row_nodes = []  # list of node indices for each row (first node)
    for r_idx, (x, y, z) in enumerate(rows):
        nodes = []
        for c in (x, k + y, 2 * k + z):
            nodes.append(add_node(c, r_idx))
        # link the three nodes horizontally (circular)
        for i in range(3):
            left.append(nodes[(i - 1) % 3])
            right.append(nodes[(i + 1) % 3])
        # store first node index for row
        row_nodes.append(nodes[0])

    # adjust left/right for node indices (they were appended after node creation)
    # Actually we built left/right incorrectly; easier: store horizontal links separately
    # We'll rebuild using separate arrays horiz_left, horiz_right
    horiz_left = [-1] * len(col)
    horiz_right = [-1] * len(col)
    for r_idx, nodes in enumerate(row_nodes):
        n0, n1, n2 = nodes, nodes + 1, nodes + 2
        horiz_left[n0] = n2
        horiz_right[n0] = n1
        horiz_left[n1] = n0
        horiz_right[n1] = n2
        horiz_left[n2] = n1
        horiz_right[n2] = n0

    # cover / uncover operations
    def cover(c):
        # remove column header
        right[left[c]] = right[c]
        left[right[c]] = left[c]
        # for each row i in column c
        i = down[col_head[c]]
        while i != col_head[c]:
            # for each node j in row i
            j = horiz_right[i]
            while j != i:
                # remove node j from its column
                down[up[j]] = down[j]
                up[down[j]] = up[j]
                size[col[j]] -= 1
                j = horiz_right[j]
            i = down[i]

    def uncover(c):
        i = up[col_head[c]]
        while i != col_head[c]:
            j = horiz_left[i]
            while j != i:
                size[col[j]] += 1
                down[up[j]] = j
                up[down[j]] = j
                j = horiz_left[j]
            i = up[i]
        right[left[c]] = c
        left[right[c]] = c

    solution_rows = []

    def search(depth: int) -> bool:
        if right[0] == 0:  # all columns covered
            return True
        # choose column with minimal size
        c = right[0]
        min_sz = size[c]
        cc = c
        j = right[c]
        while j != 0:
            if size[j] < min_sz:
                min_sz = size[j]
                cc = j
                if min_sz == 0:
                    break
            j = right[j]
        c = cc
        if min_sz == 0:
            return False
        cover(c)
        r = down[col_head[c]]
        while r != col_head[c]:
            solution_rows.append(row_id[r])
            # cover other columns in this row
            j = horiz_right[r]
            while j != r:
                cover(col[j])
                j = horiz_right[j]
            if search(depth + 1):
                return True
            # backtrack
            j = horiz_left[r]
            while j != r:
                uncover(col[j])
                j = horiz_left[j]
            solution_rows.pop()
            r = down[r]
        uncover(c)
        return False

    # Initialize column header nodes as self-referential circular lists
    for c in range(n_cols):
        col_head[c] = len(up)  # dummy node index for header
        up.append(col_head[c])
        down.append(col_head[c])
        col.append(c)  # header's column id (unused)
        row_id.append(-1)

    # Rebuild nodes now that headers exist
    # Clear previous structures and rebuild properly
    # Simpler: redo building with proper DLX from scratch

    # ---- Rebuild correctly ----
    up = []
    down = []
    col = []
    row_id = []
    size = [0] * n_cols
    # column headers as nodes 0..n_cols-1
    for c in range(n_cols):
        up.append(c)
        down.append(c)
        col.append(c)
        row_id.append(-1)
    # horizontal links for rows
    horiz_left = []
    horiz_right = []

    def link_node(node_idx, c, r):
        # insert node into column c (at bottom)
        up.append(up[c])
        down.append(c)
        col.append(c)
        row_id.append(r)
        # adjust column links
        down[up[c]] = node_idx
        up[c] = node_idx
        size[c] += 1

    # Build rows
    row_nodes = []
    for r_idx, (x, y, z) in enumerate(rows):
        nodes = []
        for c in (x, k + y, 2 * k + z):
            node_idx = len(up)
            # column insertion
            up.append(up[c])
            down.append(c)
            col.append(c)
            row_id.append(r_idx)
            down[up[c]] = node_idx
            up[c] = node_idx
            size[c] += 1
            nodes.append(node_idx)
        # horizontal links
        for i in range(3):
            horiz_left.append(nodes[(i - 1) % 3])
            horiz_right.append(nodes[(i + 1) % 3])
        row_nodes.append(nodes[0])

    # column header list
    left = [n_cols - 1] * (n_cols + 1)
    right = [0] * (n_cols + 1)
    for i in range(n_cols):
        left[i + 1] = i
        right[i] = i + 1
    left[0] = n_cols
    right[n_cols] = 0

    # cover/uncover using updated structures
    def cover(c):
        right[left[c]] = right[c]
        left[right[c]] = left[c]
        i = down[c]
        while i != c:
            j = horiz_right[i]
            while j != i:
                down[up[j]] = down[j]
                up[down[j]] = up[j]
                size[col[j]] -= 1
                j = horiz_right[j]
            i = down[i]

    def uncover(c):
        i = up[c]
        while i != c:
            j = horiz_left[i]
            while j != i:
                size[col[j]] += 1
                down[up[j]] = j
                up[down[j]] = j
                j = horiz_left[j]
            i = up[i]
        right[left[c]] = c
        left[right[c]] = c

    solution_rows = []

    def search():
        if right[0] == 0:
            return True
        # choose column with minimal size
        c = right[0]
        min_sz = size[c]
        cc = c
        j = right[c]
        while j != 0:
            if size[j] < min_sz:
                min_sz = size[j]
                cc = j
                if min_sz == 0:
                    break
            j = right[j]
        c = cc
        if min_sz == 0:
            return False
        cover(c)
        r = down[c]
        while r != c:
            solution_rows.append(row_id[r])
            j = horiz_right[r]
            while j != r:
                cover(col[j])
                j = horiz_right[j]
            if search():
                return True
            # backtrack
            j = horiz_left[r]
            while j != r:
                uncover(col[j])
                j = horiz_left[j]
            solution_rows.pop()
            r = down[r]
        uncover(c)
        return False

    if search():
        out = ["YES"]
        for rid in solution_rows:
            x, y, z = rows[rid]
            out.append(f"{x+1} {y+1} {z+1}")
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
