import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    k = int(data[0])
    t = int(data[1])
    idx = 2
    configs = []
    for _ in range(t):
        x = int(data[idx]) - 1
        y = int(data[idx+1]) - 1
        z = int(data[idx+2]) - 1
        idx += 3
        configs.append((x, y, z))
    
    # If there are no configurations, only possible if k == 0, but k >= 1
    if t == 0:
        print("NO")
        return
    
    # We need to select k configurations such that:
    # - all x are distinct (0..k-1)
    # - all y are distinct (0..k-1)
    # - all z are distinct (0..k-1)
    # This is a 3-dimensional matching problem.
    # k <= 22, so we can use DP with bitmask over one dimension.
    
    # Let's use DP over z (bitmask of used z), and track used x and y.
    # But we also need to ensure x and y are permutations.
    # Actually, since we need exactly k configurations and all x, y, z distinct,
    # we can think of it as: for each z, we pick a configuration with that z,
    # but actually we need to pick k configs total with all distinct.
    
    # Alternative: DP over bitmask of used x, and for each, track used y and z.
    # But that's 2^22 * something, too large.
    
    # Better: Since we need a perfect matching in a 3-partite 3-uniform hypergraph,
    # and k <= 22, we can use inclusion-exclusion or DP with bitmask.
    
    # Let's use DP where state is (mask_x, mask_y) and we process z one by one.
    # But mask_x and mask_y each up to 2^22, product is 2^44, too large.
    
    # Instead, we can fix the order: process configurations grouped by z.
    # For each z, we either pick one config with that z or not.
    # But we need exactly k configs, so we pick exactly one per z? No, we need all z used,
    # so we must pick exactly one config for each z value (since there are k z values and k configs).
    # Wait, we need all z from 1..k to be used, and we pick k configs, so yes, exactly one per z.
    
    # So we can group configs by z, and for each z, pick exactly one config.
    # Then we need the chosen x's to be a permutation and y's to be a permutation.
    
    # Now DP: process z from 0 to k-1. State: (mask_x, mask_y) of used x and y.
    # But that's 2^22 * 2^22 = 2^44, too large.
    
    # However, note that at step i (having processed z=0..i-1), we have picked i configs,
    # so mask_x and mask_y both have exactly i bits set. The number of such pairs is C(k,i)^2.
    # Sum over i of C(k,i)^2 = C(2k, k) which for k=22 is C(44,22) ~ 2e12, too large.
    
    # We need a different approach. Let's use DP over mask_x only, and for each mask_x,
    # we need to know if there's a valid assignment of y and z.
    # But we also need y to be a permutation.
    
    # Alternative: Since we pick one config per z, we can think of it as:
    # for each z, we have a set of (x,y) pairs. We need to pick one (x,y) per z
    # such that x's are all distinct and y's are all distinct.
    # This is like finding a perfect matching in a bipartite graph where left is z,
    # right is (x,y) pairs, but we need x and y to be permutations.
    
    # Actually, we can model it as: we need a permutation of x and a permutation of y
    # such that for each z, (x_z, y_z, z) is a valid config.
    # So for each z, we have a set of allowed (x,y) pairs.
    # We need to choose for each z an (x,y) such that x's are distinct and y's are distinct.
    
    # This is equivalent to finding a perfect matching in a 3-partite hypergraph,
    # which is NP-hard in general, but k <= 22 allows exponential algorithms.
    
    # Let's use DP with bitmask over x, and for each mask_x, we store the set of
    # possible mask_y that can be achieved. But storing sets is expensive.
    
    # Instead, we can use the fact that we process z one by one, and for each z,
    # we update the DP. The number of reachable (mask_x, mask_y) pairs might be small.
    # But in worst case it's large.
    
    # Given k <= 22 and time limit 5s, we can try a meet-in-the-middle or
    # a backtracking with pruning.
    
    # Let's try a recursive backtracking with pruning:
    # - Order z by number of available configs (fewest first) to prune early.
    # - For each z, try each config, check if x and y are unused.
    
    # This might work for k=22 if the graph is not too dense.
    
    # Let's implement backtracking with bitmask for used x and used y.
    
    # Group configs by z
    by_z = [[] for _ in range(k)]
    for c in configs:
        x, y, z = c
        by_z[z].append((x, y))
    
    # Check that every z has at least one config
    for z in range(k):
        if not by_z[z]:
            print("NO")
            return
    
    # Order z by number of configs (ascending) for better pruning
    z_order = sorted(range(k), key=lambda z: len(by_z[z]))
    
    used_x = 0
    used_y = 0
    assignment = [None] * k  # assignment[z] = (x, y)
    
    # Precompute for each z the list of (x, y)
    z_configs = [by_z[z] for z in z_order]
    
    def backtrack(i, used_x, used_y):
        if i == k:
            return True
        for (x, y) in z_configs[i]:
            bx = 1 << x
            by = 1 << y
            if not (used_x & bx) and not (used_y & by):
                assignment[z_order[i]] = (x, y)
                if backtrack(i+1, used_x | bx, used_y | by):
                    return True
        return False
    
    if backtrack(0, 0, 0):
        print("YES")
        for z in range(k):
            x, y = assignment[z]
            print(f"{x+1} {y+1} {z+1}")
    else:
        print("NO")

solve()
