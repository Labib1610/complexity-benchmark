import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    k = int(input_data[0])
    t = int(input_data[1])

    adj_x = [[] for _ in range(k)]
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3
        adj_x[x].append((y, z))

    # Quick checks
    has_y = [False] * k
    has_z = [False] * k
    for x in range(k):
        if not adj_x[x]:
            print("NO")
            return
        for y, z in adj_x[x]:
            has_y[y] = True
            has_z[z] = True

    if not all(has_y) or not all(has_z):
        print("NO")
        return

    ans = [None] * k
    full_mask = (1 << k) - 1

    def search(mask_x, mask_y, mask_z):
        if mask_x == full_mask:
            return True

        best_x = -1
        best_choices = None
        min_c = 1000000

        for x in range(k):
            if not (mask_x & (1 << x)):
                choices = []
                for y, z in adj_x[x]:
                    if not ((mask_y & (1 << y)) or (mask_z & (1 << z))):
                        choices.append((y, z))
                if not choices:
                    return False
                if len(choices) < min_c:
                    min_c = len(choices)
                    best_x = x
                    best_choices = choices
                    if min_c == 1:
                        break

        for y, z in best_choices:
            ans[best_x] = (y, z)
            if search(
                mask_x | (1 << best_x), mask_y | (1 << y), mask_z | (1 << z)
            ):
                return True

        return False

    if search(0, 0, 0):
        print("YES")
        for x in range(k):
            y, z = ans[x]
            print(f"{x + 1} {y + 1} {z + 1}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
