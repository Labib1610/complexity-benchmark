import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    k = int(next(it))
    t = int(next(it))
    triples = []
    for _ in range(t):
        x = int(next(it)) - 1
        y = int(next(it)) - 1
        z = int(next(it)) - 1
        triples.append((x, y, z))

    n_cols = 3 * k  # columns: tech 0..k-1, rig k..2k-1, win 2k..3k-1

    # Build dancing links structures
    # Node 0 is header
    # column headers are 1..n_cols
    # each node has: left, right, up, down, col, row_id
    max_nodes = 1 + n_cols + 3 * t
    left = [0] * max_nodes
    right = [0] * max_nodes
    up = [0] * max_nodes
    down = [0] * max_nodes
    col = [0] * max_nodes
    row_id = [-1] * max_nodes

    # header links
    for i in range(n_cols + 1):
        left[i] = i - 1
        right[i] = i + 1
        up[i] = down[i] = i
    left[0] = n_cols
    right[n_cols] = 0

    size = [0] * (n_cols + 1)  # column sizes
    node_cnt = n_cols + 1

    # add rows
    for rid, (x, y, z) in enumerate(triples):
        cols = [x, k + y, 2 * k + z]
        first = -1
        for c in cols:
            node = node_cnt
            node_cnt += 1
            col[node] = c + 1  # column header index (1‑based)
            row_id[node] = rid

            # vertical links
            up[node] = up[col[node]]
            down[node] = col[node]
            down[up[node]] = node
            up[col[node]] = node
            size[col[node]] += 1

            # horizontal links
            if first == -1:
                left[node] = right[node] = node
                first = node
            else:
                left[node] = left[first]
                right[node] = first
                right[left[first]] = node
                left[first] = node

    solution_rows = []

    def cover(c):
        # remove column header
        right[left[c]] = right[c]
        left[right[c]] = left[c]
        # remove rows
        i = down[c]
        while i != c:
            j = right[i]
            while j != i:
                down[up[j]] = down[j]
                up[down[j]] = up[j]
                size[col[j]] -= 1
                j = right[j]
            i = down[i]

    def uncover(c):
        i = up[c]
        while i != c:
            j = left[i]
            while j != i:
                size[col[j]] += 1
                down[up[j]] = j
                up[down[j]] = j
                j = left[j]
            i = up[i]
        right[left[c]] = c
        left[right[c]] = c

    def search():
        if right[0] == 0:  # all columns covered
            return True
        # choose column with minimal size
        c = right[0]
        min_sz = size[c]
        cc = c
        j = right[c]
        while j != 0:
            if size[j] < min_sz:
                min_sz = size[j]
                cc = j
                if min_sz == 0:
                    break
            j = right[j]
        if min_sz == 0:
            return False
        cover(cc)
        r = down[cc]
        while r != cc:
            solution_rows.append(row_id[r])
            j = right[r]
            while j != r:
                cover(col[j])
                j = right[j]
            if search():
                return True
            # backtrack
            j = left[r]
            while j != r:
                uncover(col[j])
                j = left[j]
            solution_rows.pop()
            r = down[r]
        uncover(cc)
        return False

    possible = search()
    out = []
    if possible and len(solution_rows) == k:
        out.append("YES")
        for rid in solution_rows:
            x, y, z = triples[rid]
            out.append(f"{x+1} {y+1} {z+1}")
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
