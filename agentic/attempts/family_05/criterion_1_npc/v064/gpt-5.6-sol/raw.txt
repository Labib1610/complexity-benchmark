import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    full = (1 << k) - 1

    triples = []
    adjacency = [[] for _ in range(3 * k)]

    pos = 2
    for idx in range(t):
        x = data[pos] - 1
        y = data[pos + 1] - 1
        z = data[pos + 2] - 1
        pos += 3

        bx = 1 << x
        by = 1 << y
        bz = 1 << z
        row = bx | (by << k) | (bz << (2 * k))
        triples.append((x, y, z, bx, by, bz, row))
        adjacency[x].append(idx)
        adjacency[k + y].append(idx)
        adjacency[2 * k + z].append(idx)

    if k == 0:
        print("YES")
        return

    if t < k:
        print("NO")
        return

    def available(tr, mx, my, mz):
        return not ((tr[3] & mx) or (tr[4] & my) or (tr[5] & mz))

    def xor_feasible(mx, my, mz):
        basis = [0] * (3 * k)
        for tr in triples:
            if (tr[3] & mx) or (tr[4] & my) or (tr[5] & mz):
                continue
            value = tr[6]
            while value:
                p = value.bit_length() - 1
                if basis[p]:
                    value ^= basis[p]
                else:
                    basis[p] = value
                    break

        target = ((full ^ mx)
                  | ((full ^ my) << k)
                  | ((full ^ mz) << (2 * k)))
        while target:
            p = target.bit_length() - 1
            if not basis[p]:
                return False
            target ^= basis[p]
        return True

    def has_perfect_matching(neighbors, source_mask):
        sources = []
        m = source_mask
        while m:
            bit = m & -m
            s = bit.bit_length() - 1
            if neighbors[s] == 0:
                return False
            sources.append(s)
            m ^= bit

        sources.sort(key=lambda s: neighbors[s].bit_count())
        matched = [-1] * k

        def augment(s, seen):
            choices = neighbors[s] & ~seen
            while choices:
                bit = choices & -choices
                v = bit.bit_length() - 1
                choices ^= bit
                old = matched[v]
                if old == -1:
                    matched[v] = s
                    return True
                if augment(old, seen | bit):
                    matched[v] = s
                    return True
            return False

        for s in sources:
            if not augment(s, 0):
                return False
        return True

    def projection_feasible(mx, my, mz):
        xy = [0] * k
        xz = [0] * k
        yz = [0] * k

        for tr in triples:
            x, y, z, bx, by, bz, _ = tr
            if (bx & mx) or (by & my) or (bz & mz):
                continue
            xy[x] |= by
            xz[x] |= bz
            yz[y] |= bz

        if not has_perfect_matching(xy, full ^ mx):
            return False
        if not has_perfect_matching(xz, full ^ mx):
            return False
        if not has_perfect_matching(yz, full ^ my):
            return False
        return True

    failed = set()
    answer = []
    CACHE_LIMIT = 1_500_000

    def dfs(mx, my, mz, depth):
        if mx == full:
            return True

        state = mx | (my << k) | (mz << (2 * k))
        if state in failed:
            return False

        best_candidates = None
        best_count = 10 ** 9

        for resource in range(3 * k):
            if resource < k:
                if mx & (1 << resource):
                    continue
            elif resource < 2 * k:
                if my & (1 << (resource - k)):
                    continue
            else:
                if mz & (1 << (resource - 2 * k)):
                    continue

            candidates = []
            exceeded = False
            for idx in adjacency[resource]:
                tr = triples[idx]
                if available(tr, mx, my, mz):
                    candidates.append(idx)
                    if len(candidates) >= best_count:
                        exceeded = True
                        break

            if exceeded:
                continue
            if not candidates:
                if len(failed) < CACHE_LIMIT:
                    failed.add(state)
                return False

            best_count = len(candidates)
            best_candidates = candidates
            if best_count == 1:
                break

        if best_candidates is None:
            return False

        if best_count > 1 and depth % 3 == 0:
            if not xor_feasible(mx, my, mz):
                if len(failed) < CACHE_LIMIT:
                    failed.add(state)
                return False
            if not projection_feasible(mx, my, mz):
                if len(failed) < CACHE_LIMIT:
                    failed.add(state)
                return False

        if 1 < best_count <= 100:
            scored = []
            for idx in best_candidates:
                tr = triples[idx]
                score = 0
                for resource in (tr[0], k + tr[1], 2 * k + tr[2]):
                    count = 0
                    for other_idx in adjacency[resource]:
                        if available(triples[other_idx], mx, my, mz):
                            count += 1
                    score += count
                scored.append((score, idx))
            scored.sort(reverse=True)
            best_candidates = [idx for _, idx in scored]

        for idx in best_candidates:
            tr = triples[idx]
            x, y, z, bx, by, bz, _ = tr
            if (bx & mx) or (by & my) or (bz & mz):
                continue

            answer.append((x + 1, y + 1, z + 1))
            if dfs(mx | bx, my | by, mz | bz, depth + 1):
                return True
            answer.pop()

        if len(failed) < CACHE_LIMIT:
            failed.add(state)
        return False

    if dfs(0, 0, 0, 0):
        out = ["YES"]
        out.extend(f"{x} {y} {z}" for x, y, z in answer)
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    main()
