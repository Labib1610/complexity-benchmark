import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    triplets = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx + 1])
        z = int(input_data[idx + 2])
        idx += 3
        triplets.append((x, y, z))

    num_cols = 3 * k
    # Column indices: 1 to 3*k
    # 0 is root

    # We will build the DLX matrix
    # Maximum nodes: (num_cols + 1) + 3 * t
    max_nodes = num_cols + 1 + 3 * t + 10

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row = [0] * max_nodes
    size = [0] * (num_cols + 1)

    # Initialize column headers
    for i in range(num_cols + 1):
        L[i] = i - 1
        R[i] = i + 1
        U[i] = i
        D[i] = i
        col[i] = i
        size[i] = 0
    L[0] = num_cols
    R[num_cols] = 0

    node_cnt = num_cols + 1

    for r_idx, (x, y, z) in enumerate(triplets):
        cols_to_cover = [x, k + y, 2 * k + z]
        first_node = node_cnt
        for c in cols_to_cover:
            cur = node_cnt
            node_cnt += 1
            col[cur] = c
            row[cur] = r_idx

            # Vertical link (insert at bottom of column c)
            U[cur] = U[c]
            D[cur] = c
            D[U[c]] = cur
            U[c] = cur
            size[c] += 1

            # Horizontal link
            if cur == first_node:
                L[cur] = cur
                R[cur] = cur
            else:
                L[cur] = L[first_node]
                R[cur] = first_node
                R[L[first_node]] = cur
                L[first_node] = cur

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        # Choose column with min size
        c = R[0]
        min_s = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_s:
                min_s = size[curr]
                c = curr
            curr = R[curr]

        if min_s == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            ans.append(row[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            ans.pop()
            r = D[r]

        uncover(c)
        return False

    if search():
        out = ["YES"]
        for r_idx in ans:
            x, y, z = triplets[r_idx]
            out.append(f"{x} {y} {z}")
        sys.stdout.write("\n".join(out) + "\n")
    else:
        sys.stdout.write("NO\n")


if __name__ == "__main__":
    solve()
