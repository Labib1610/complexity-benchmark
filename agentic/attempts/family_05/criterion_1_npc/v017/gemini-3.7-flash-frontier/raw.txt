import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    edges = []
    edge_masks = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx]) - 1
        y = int(input_data[idx + 1]) - 1
        z = int(input_data[idx + 2]) - 1
        idx += 3
        edges.append((x + 1, y + 1, z + 1))
        m = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
        edge_masks.append(m)

    num_items = 3 * k
    edges_by_item = [[] for _ in range(num_items)]
    for e_idx, (x, y, z) in enumerate(edges):
        edges_by_item[x - 1].append(e_idx)
        edges_by_item[k + y - 1].append(e_idx)
        edges_by_item[2 * k + z - 1].append(e_idx)

    # Convert to tuples for slightly faster iteration
    edges_by_item = [tuple(lst) for lst in edges_by_item]

    TARGET = (1 << num_items) - 1
    sol = []

    def dfs(covered):
        if covered == TARGET:
            return True

        best_edges = None
        min_cnt = 1000000

        rem = TARGET ^ covered
        while rem:
            lsb = rem & -rem
            i = lsb.bit_length() - 1
            rem ^= lsb

            valid = []
            cnt = 0
            for e in edges_by_item[i]:
                if not (edge_masks[e] & covered):
                    cnt += 1
                    valid.append(e)
                    if cnt >= min_cnt:
                        break

            if cnt < min_cnt:
                min_cnt = cnt
                best_edges = valid
                if min_cnt == 0:
                    return False
                if min_cnt == 1:
                    break

        for e in best_edges:
            sol.append(e)
            if dfs(covered | edge_masks[e]):
                return True
            sol.pop()

        return False

    sys.setrecursionlimit(100)
    if dfs(0):
        out = ["YES"]
        for e in sol:
            x, y, z = edges[e]
            out.append(f"{x} {y} {z}")
        print("\n".join(out))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
