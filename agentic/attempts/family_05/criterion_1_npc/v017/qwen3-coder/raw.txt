from itertools import permutations

def solve():
    k, t = map(int, input().split())
    
    if t < k:
        print("NO")
        return
    
    configs = []
    for _ in range(t):
        x, y, z = map(int, input().split())
        configs.append((x, y, z))
    
    # We need to find a permutation of k configurations such that
    # all arms, sensors, and modules are distinct and cover 1..k
    
    # Try all combinations of k configurations from the list
    from itertools import combinations
    
    for combo in combinations(range(t), k):
        selected = [configs[i] for i in combo]
        
        # Check if arms, sensors, modules are all distinct and complete
        arms = [cfg[0] for cfg in selected]
        sensors = [cfg[1] for cfg in selected]
        modules = [cfg[2] for cfg in selected]
        
        if len(set(arms)) == k and len(set(sensors)) == k and len(set(modules)) == k:
            # Check if they form a permutation of 1..k each
            if set(arms) == set(range(1, k+1)) and set(sensors) == set(range(1, k+1)) and set(modules) == set(range(1, k+1)):
                print("YES")
                for cfg in selected:
                    print(cfg[0], cfg[1], cfg[2])
                return
    
    print("NO")

solve()
