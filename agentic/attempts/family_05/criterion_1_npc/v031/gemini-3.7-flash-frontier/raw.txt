```python
import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    k = int(input_data[0])
    t = int(input_data[1])

    triples = []
    idx = 2
    for _ in range(t):
        x = int(input_data[idx])
        y = int(input_data[idx + 1])
        z = int(input_data[idx + 2])
        triples.append((x, y, z))
        idx += 3

    num_cols = 3 * k
    max_nodes = num_cols + 1 + 3 * t + 5

    L = [0] * max_nodes
    R = [0] * max_nodes
    U = [0] * max_nodes
    D = [0] * max_nodes
    col = [0] * max_nodes
    row_id = [0] * max_nodes
    size = [0] * (num_cols + 1)

    for i in range(num_cols + 1):
        L[i] = (i - 1) % (num_cols + 1)
        R[i] = (i + 1) % (num_cols + 1)
        U[i] = i
        D[i] = i

    sz = num_cols + 1

    for r_idx, (x, y, z) in enumerate(triples):
        cols_to_add = (x, k + y, 2 * k + z)
        first = -1
        for c in cols_to_add:
            node = sz
            sz += 1
            col[node] = c
            row_id[node] = r_idx
            size[c] += 1

            D[node] = c
            U[node] = U[c]
            D[U[c]] = node
            U[c] = node

            if first == -1:
                first = node
                L[node] = node
                R[node] = node
            else:
                L[node] = L[first]
                R[node] = first
                R[L[first]] = node
                L[first] = node

    def cover(c):
        L[R[c]] = L[c]
        R[L[c]] = R[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                U[D[j]] = U[j]
                D[U[j]] = D[j]
                size[col[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                size[col[j]] += 1
                U[D[j]] = j
                D[U[j]] = j
                j = L[j]
            i = U[i]
        L[R[c]] = c
        R[L[c]] = c

    ans = []

    def search():
        if R[0] == 0:
            return True

        c = R[0]
        min_size = size[c]
        curr = R[c]
        while curr != 0:
            if size[curr] < min_size:
                min_size = size[curr]
                c = curr
            curr = R[curr]

        if min_size == 0:
            return False

        cover(c)
        r = D[c]
        while r != c:
            ans.append(row_id[r])
            j = R[r]
            while j != r:
                cover(col[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(col[j])
                j = L[j]
            ans.pop()
            r = D[r]

        uncover(c)
        return False

    if search():
        print("YES")
        for r_idx in ans:
            x, y, z = triples[r_idx]
            print(f"{x} {y} {z}")
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
