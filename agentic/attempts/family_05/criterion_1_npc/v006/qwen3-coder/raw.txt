from itertools import permutations

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    k = int(data[0])
    t = int(data[1])
    
    triples = []
    idx = 2
    for _ in range(t):
        x = int(data[idx])
        y = int(data[idx+1])
        z = int(data[idx+2])
        triples.append((x, y, z))
        idx += 3
    
    # Create a mapping from surgeons to their possible assignments
    # We'll use backtracking with pruning
    
    # Convert to 0-indexed for easier handling
    triples_0 = [(x-1, y-1, z-1) for (x, y, z) in triples]
    
    # For each surgeon, we need to pick exactly one triple
    # This is essentially finding a perfect matching in a 3-dimensional structure
    
    # Let's model this as a bipartite matching problem:
    # Left side: surgeons (0 to k-1)
    # Right side: (room, time) pairs 
    # But since we also need each room and each time used exactly once,
    # this becomes a problem of finding a Latin square or a 3D matching
    
    # Since k <= 22, we can try a backtracking approach
    
    # Try all permutations of rooms for surgeons, and for each check if times can be assigned
    # Actually, let's think differently - we want a subset of triples such that:
    # - Each surgeon appears exactly once
    # - Each room appears exactly once  
    # - Each time appears exactly once
    
    # This is equivalent to finding a perfect 3D matching
    
    # Since k is small, we can try backtracking:
    # For each surgeon in order, try assigning them to each of their available triples
    # and ensure no conflicts with rooms/times already used
    
    # Build adjacency for each surgeon
    adj = [[] for _ in range(k)]
    for i, (x, y, z) in enumerate(triples_0):
        adj[x].append((y, z, i))  # For surgeon x, store (room, time, index in triples)
        
    used_room = [False] * k
    used_time = [False] * k
    selected = []  # Stores indices of selected triples
    result = []
    
    def backtrack(surgeon):
        if surgeon == k:
            return True
            
        for (room, time, idx) in adj[surgeon]:
            if not used_room[room] and not used_time[time]:
                used_room[room] = True
                used_time[time] = True
                selected.append(idx)
                
                if backtrack(surgeon + 1):
                    result.append((surgeon+1, room+1, time+1))  # Convert back to 1-indexed
                    return True
                    
                selected.pop()
                used_room[room] = False
                used_time[time] = False
                
        return False
        
    if backtrack(0):
        print("YES")
        for triple in reversed(result):  # We built it backwards
            print(*triple)
    else:
        print("NO")

solve()
