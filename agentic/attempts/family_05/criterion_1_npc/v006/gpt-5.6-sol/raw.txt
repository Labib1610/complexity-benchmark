import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    k, t = data[0], data[1]
    triples = []
    p = 2

    xy = [0] * k
    xz = [0] * k
    yz = [0] * k

    basis = [0] * (3 * k)

    for _ in range(t):
        x, y, z = data[p] - 1, data[p + 1] - 1, data[p + 2] - 1
        p += 3
        triples.append((x, y, z))

        xy[x] |= 1 << y
        xz[x] |= 1 << z
        yz[y] |= 1 << z

        v = (1 << x) | (1 << (k + y)) | (1 << (2 * k + z))
        while v:
            b = v.bit_length() - 1
            if basis[b]:
                v ^= basis[b]
            else:
                basis[b] = v
                break

    def has_perfect_matching(adj):
        order = sorted(range(k), key=lambda i: adj[i].bit_count())
        match = [-1] * k

        def augment(u, seen):
            mask = adj[u] & ~seen
            while mask:
                bit = mask & -mask
                mask -= bit
                v = bit.bit_length() - 1
                owner = match[v]
                if owner == -1 or augment(owner, seen | bit):
                    match[v] = u
                    return True
            return False

        for u in order:
            if not augment(u, 0):
                return False
        return True

    if not (has_perfect_matching(xy) and
            has_perfect_matching(xz) and
            has_perfect_matching(yz)):
        print("NO")
        return

    # Necessary parity condition: the all-ones incidence vector must lie
    # in the GF(2) span of the available triples.
    target = (1 << (3 * k)) - 1
    while target:
        b = target.bit_length() - 1
        if basis[b]:
            target ^= basis[b]
        else:
            print("NO")
            return

    ncol = 3 * k

    L = list(range(ncol + 1))
    R = list(range(ncol + 1))
    U = list(range(ncol + 1))
    D = list(range(ncol + 1))
    C = list(range(ncol + 1))
    S = [0] * (ncol + 1)
    row_id = [-1] * (ncol + 1)

    for c in range(ncol + 1):
        L[c] = c - 1
        R[c] = c + 1
    L[0] = ncol
    R[ncol] = 0

    def add_node(col, rid):
        idx = len(C)
        C.append(col)
        S.append(0)
        row_id.append(rid)
        L.append(idx)
        R.append(idx)

        up = U[col]
        U.append(up)
        D.append(col)
        D[up] = idx
        U[col] = idx
        S[col] += 1
        return idx

    for rid, (x, y, z) in enumerate(triples):
        nodes = [
            add_node(1 + x, rid),
            add_node(1 + k + y, rid),
            add_node(1 + 2 * k + z, rid)
        ]
        for i in range(3):
            L[nodes[i]] = nodes[(i - 1) % 3]
            R[nodes[i]] = nodes[(i + 1) % 3]

    def cover(c):
        R[L[c]] = R[c]
        L[R[c]] = L[c]
        i = D[c]
        while i != c:
            j = R[i]
            while j != i:
                D[U[j]] = D[j]
                U[D[j]] = U[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(c):
        i = U[c]
        while i != c:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                D[U[j]] = j
                U[D[j]] = j
                j = L[j]
            i = U[i]
        R[L[c]] = c
        L[R[c]] = c

    solution = []
    failed = set()
    memo_limit = 650000

    sys.setrecursionlimit(10000)

    def search(mx, my, mz):
        if R[0] == 0:
            return True

        key = mx | (my << k) | (mz << (2 * k))
        if key in failed:
            return False

        c = R[0]
        best = S[c]
        q = R[c]
        while q != 0:
            sq = S[q]
            if sq < best:
                c = q
                best = sq
                if best == 0:
                    break
            q = R[q]

        if best == 0:
            if len(failed) < memo_limit:
                failed.add(key)
            return False

        candidates = []
        i = D[c]
        while i != c:
            score = 0
            j = R[i]
            while j != i:
                score += S[C[j]]
                j = R[j]
            candidates.append((score, i))
            i = D[i]
        candidates.sort()

        cover(c)
        for _, rnode in candidates:
            rid = row_id[rnode]
            x, y, z = triples[rid]

            solution.append(rid)

            j = R[rnode]
            while j != rnode:
                cover(C[j])
                j = R[j]

            if search(mx | (1 << x), my | (1 << y), mz | (1 << z)):
                return True

            j = L[rnode]
            while j != rnode:
                uncover(C[j])
                j = L[j]

            solution.pop()

        uncover(c)

        if len(failed) < memo_limit:
            failed.add(key)
        return False

    if not search(0, 0, 0):
        print("NO")
        return

    out = ["YES"]
    for rid in solution:
        x, y, z = triples[rid]
        out.append(f"{x + 1} {y + 1} {z + 1}")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
