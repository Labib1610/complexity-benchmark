import sys

# Increase recursion depth for deep searches
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj_edges = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj_edges[u].append(i)
        adj_edges[v].append(i)
    
    # Precompute edge neighbors
    edge_neighbors = [[] for _ in range(m)]
    for i in range(m):
        u, v = edges[i]
        nbrs = set()
        for e in adj_edges[u]:
            if e != i:
                nbrs.add(e)
        for e in adj_edges[v]:
            if e != i:
                nbrs.add(e)
        edge_neighbors[i] = list(nbrs)

    # Connected components
    visited_edges = [False] * m
    components = []
    for i in range(m):
        if not visited_edges[i]:
            comp = []
            q = [i]
            visited_edges[i] = True
            for curr in q:
                comp.append(curr)
                for nxt in edge_neighbors[curr]:
                    if not visited_edges[nxt]:
                        visited_edges[nxt] = True
                        q.append(nxt)
            components.append(comp)

    final_colors = [0] * m

    for comp in components:
        # Check if component has bridges or odd cuts if needed, but solver will handle it
        mask = [7] * m  # 7 = 111 in binary (colors 1, 2, 3 available)
        colored = [0] * m
        comp_set = set(comp)
        
        # Backtracking solver with MRV
        step_limit = [2000000]

        def backtrack(unassigned_count):
            if unassigned_count == 0:
                return True
            if step_limit[0] <= 0:
                return False
            step_limit[0] -= 1

            # Choose edge with MRV
            best_e = -1
            min_c = 4
            for e in comp:
                if colored[e] == 0:
                    c_count = (mask[e] & 1) + ((mask[e] >> 1) & 1) + ((mask[e] >> 2) & 1)
                    if c_count < min_c:
                        min_c = c_count
                        best_e = e
                        if min_c <= 1:
                            break
            
            if min_c == 0:
                return False
            
            e = best_e
            m_val = mask[e]
            
            for c in (1, 2, 3):
                c_bit = 1 << (c - 1)
                if m_val & c_bit:
                    colored[e] = c
                    # propagate
                    changed = []
                    valid = True
                    for nxt in edge_neighbors[e]:
                        if colored[nxt] == 0 and (mask[nxt] & c_bit):
                            mask[nxt] ^= c_bit
                            changed.append(nxt)
                            if mask[nxt] == 0:
                                valid = False
                    
                    if valid:
                        if backtrack(unassigned_count - 1):
                            return True
                    
                    # undo
                    for nxt in changed:
                        mask[nxt] |= c_bit
                    colored[e] = 0
            
            return False

        if not backtrack(len(comp)):
            print("NO")
            return
        
        for e in comp:
            final_colors[e] = colored[e]

    print("YES")
    print(*(final_colors[i] for i in range(m)))

if __name__ == '__main__':
    solve()
