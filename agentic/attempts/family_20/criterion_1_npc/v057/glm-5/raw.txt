import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check degree constraint (guaranteed by problem)
    # Build edge index map
    edge_colors = [0] * m
    
    # We use a greedy approach with backtracking
    # Since each vertex has degree 3, we need to assign colors 1,2,3 to incident edges
    
    # Sort edges by some heuristic (e.g., vertices with fewer uncolored edges first)
    # But let's try simple greedy with backtracking
    
    # For efficiency, we'll use iterative approach with stack
    
    # Build adjacency with edge indices
    adj_edges = [[] for _ in range(n + 1)]
    for idx, (u, v) in enumerate(edges):
        adj_edges[u].append((v, idx))
        adj_edges[v].append((u, idx))
    
    # State: edge_colors array
    # We'll process edges one by one
    
    # For each edge, we try colors that don't conflict with either endpoint
    
    def get_used_colors(v):
        used = set()
        for (nei, eidx) in adj_edges[v]:
            if edge_colors[eidx] != 0:
                used.add(edge_colors[eidx])
        return used
    
    stack = [(0, 1)]  # (edge_index, next_color_to_try)
    
    while stack:
        eidx, color = stack[-1]
        
        if eidx >= m:
            # All edges colored
            print("YES")
            print(' '.join(map(str, edge_colors)))
            return
        
        u, v = edges[eidx]
        
        if color > 3:
            # Backtrack
            stack.pop()
            if stack:
                prev_eidx, prev_color = stack[-1]
                edge_colors[prev_eidx] = 0
                stack[-1] = (prev_eidx, prev_color + 1)
            continue
        
        # Check if color is valid for both u and v
        used_u = get_used_colors(u)
        used_v = get_used_colors(v)
        
        if color in used_u or color in used_v:
            # Try next color
            stack[-1] = (eidx, color + 1)
        else:
            # Assign color and move to next edge
            edge_colors[eidx] = color
            stack.append((eidx + 1, 1))
    
    print("NO")

solve()
