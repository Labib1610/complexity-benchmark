import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    # Read edges and build adjacency list
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Try to 3-color the line graph (each edge is a node, adjacent if they share a vertex)
    # This is equivalent to assigning frequencies 1,2,3 to edges such that edges incident 
    # to the same vertex have different frequencies
    
    # This is essentially edge coloring of a cubic graph (3-regular graph)
    # We can model this as a constraint satisfaction problem
    
    # Use backtracking with constraint propagation
    color = [0] * m  # 0 means uncolored, 1,2,3 are the colors
    
    def get_available_colors(edge_idx):
        u, v = edges[edge_idx]
        used = set()
        for neighbor, idx in adj[u]:
            if color[idx] != 0:
                used.add(color[idx])
        for neighbor, idx in adj[v]:
            if color[idx] != 0:
                used.add(color[idx])
        return {1, 2, 3} - used
    
    # Sort edges by some heuristic to improve backtracking performance
    # Degree-based ordering might help, but let's try a simple approach first
    
    def backtrack(edge_idx):
        if edge_idx == m:
            return True
            
        if color[edge_idx] != 0:
            return backtrack(edge_idx + 1)
            
        available = get_available_colors(edge_idx)
        if not available:
            return False
            
        u, v = edges[edge_idx]
        for c in available:
            color[edge_idx] = c
            if backtrack(edge_idx + 1):
                return True
            color[edge_idx] = 0
            
        return False
    
    # Better approach: since this is a 3-regular graph, we can try a greedy method
    # or use the fact that 3-edge-coloring exists for bridgeless cubic graphs
    
    # Let's implement a more systematic approach
    # Build the line graph and try to 3-color it
    
    # Actually, let's think differently. Since each vertex has degree 3,
    # and we need to assign 3 different colors to its 3 edges,
    # this is exactly the edge-coloring problem for a 3-regular graph.
    
    # For 3-regular graphs, edge-coloring with 3 colors is possible iff
    # the graph is bridgeless (Petersen's theorem for bridgeless cubic graphs)
    
    # But checking for bridges is complex. Let's just try a greedy approach with backtracking.
    
    # Optimization: process edges in order, but when choosing color, 
    # pick the one that leaves maximum flexibility for neighbors
    
    def improved_backtrack(edge_idx):
        if edge_idx == m:
            return True
        
        if color[edge_idx] != 0:
            return improved_backtrack(edge_idx + 1)
        
        u, v = edges[edge_idx]
        # Count how many constraints each color would create
        available = get_available_colors(edge_idx)
        
        if not available:
            return False
        
        # Try colors in some order
        for c in sorted(list(available)):
            color[edge_idx] = c
            if improved_backtrack(edge_idx + 1):
                return True
            color[edge_idx] = 0
        
        return False
    
    # Even better: use a SAT solver approach or graph algorithm
    # But let's stick to backtracking with a good variable ordering
    
    # Variable ordering: choose the edge that has the most colored neighbors first
    def select_unassigned_edge():
        best_idx = -1
        max_colored_neighbors = -1
        for i in range(m):
            if color[i] == 0:
                u, v = edges[i]
                count = 0
                for neighbor, idx in adj[u]:
                    if color[idx] != 0:
                        count += 1
                for neighbor, idx in adj[v]:
                    if color[idx] != 0:
                        count += 1
                if count > max_colored_neighbors:
                    max_colored_neighbors = count
                    best_idx = i
        return best_idx
    
    def smart_backtrack():
        # Find the next unassigned edge with most constraints
        edge_idx = -1
        for i in range(m):
            if color[i] == 0:
                edge_idx = i
                break
        
        if edge_idx == -1:
            return True  # All assigned
        
        available = get_available_colors(edge_idx)
        if not available:
            return False
        
        for c in available:
            color[edge_idx] = c
            if smart_backtrack():
                return True
            color[edge_idx] = 0
        
        return False
    
    # For small cases, pure backtracking should work
    # For larger cases, we might need something smarter
    
    # Let's try a greedy approach first
    success = False
    # Try greedy first
    temp_color = [0] * m
    def greedy():
        for i in range(m):
            u, v = edges[i]
            used = set()
            for neighbor, idx in adj[u]:
                if temp_color[idx] != 0:
                    used.add(temp_color[idx])
            for neighbor, idx in adj[v]:
                if temp_color[idx] != 0:
                    used.add(temp_color[idx])
            avail = {1, 2, 3} - used
            if not avail:
                return False
            temp_color[i] = min(avail)
        return True
    
    if greedy():
        color = temp_color
        success = True
    else:
        # Fall back to backtracking
        success = smart_backtrack()
    
    if success:
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
