import sys
from collections import defaultdict

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    return n, m, edges, adj

def validate_solution(n, m, edges, adj, assignment):
    if not assignment:
        return False
    # Check each node has 3 edges in different windows
    node_windows = defaultdict(set)
    for i, (u, v) in enumerate(edges):
        w = assignment[i]
        if w not in [1, 2, 3]:
            return False
        node_windows[u].add(w)
        node_windows[v].add(w)
    
    for u in range(1, n+1):
        if len(node_windows[u]) != 3:
            return False
    
    # Check workload balance
    count = [0, 0, 0]
    for w in assignment:
        count[w-1] += 1
    
    if count[0] != count[1] or count[1] != count[2]:
        return False
        
    target = m // 3
    if count[0] != target:
        return False
        
    return True

def solve():
    n, m, edges, adj = read_input()
    
    # For small cases, try brute force
    if n <= 20:
        def backtrack(edge_idx, assignment):
            if edge_idx == m:
                if validate_solution(n, m, edges, adj, assignment):
                    return assignment[:]
                return None
            
            u, v = edges[edge_idx]
            for w in [1, 2, 3]:
                assignment.append(w)
                # Simple check: don't assign same window to both ends if they have common neighbors
                valid = True
                if valid:
                    result = backtrack(edge_idx + 1, assignment)
                    if result:
                        return result
                assignment.pop()
            
            return None
        
        result = backtrack(0, [])
        if result:
            print("YES")
            print(' '.join(map(str, result)))
        else:
            print("NO")
        return
    
    # For larger cases, use a heuristic approach
    # Try to build a valid assignment by ensuring local constraints
    
    # Since every vertex has degree 3, and we need each edge colored differently at each end,
    # this is essentially edge coloring with 3 colors where each vertex is incident to edges of all 3 colors
    
    # Build adjacency for edges (line graph)
    edge_adj = [[] for _ in range(m)]
    edge_map = {}
    for i, (u, v) in enumerate(edges):
        edge_map[(u, v)] = i
        edge_map[(v, u)] = i
    
    for i in range(m):
        u, v = edges[i]
        for w in adj[u]:
            if w != v:
                j = edge_map.get((u, w), edge_map.get((w, u), -1))
                if j != -1 and j != i:
                    edge_adj[i].append(j)
        for x in adj[v]:
            if x != u:
                j = edge_map.get((v, x), edge_map.get((x, v), -1))
                if j != -1 and j != i:
                    edge_adj[i].append(j)
    
    # Greedy coloring with backtracking fallback
    assignment = [0] * m
    used_at_node = [set() for _ in range(n+1)]
    
    def is_valid_edge_color(edge_idx, color):
        u, v = edges[edge_idx]
        return color not in used_at_node[u] and color not in used_at_node[v]
    
    def dfs(edge_idx):
        if edge_idx == m:
            # Check final balance
            count = [0, 0, 0]
            for c in assignment:
                count[c-1] += 1
            return count[0] == count[1] == count[2] == m // 3
        
        u, v = edges[edge_idx]
        for c in [1, 2, 3]:
            if c not in used_at_node[u] and c not in used_at_node[v]:
                assignment[edge_idx] = c
                used_at_node[u].add(c)
                used_at_node[v].add(c)
                
                if dfs(edge_idx + 1):
                    return True
                    
                used_at_node[u].remove(c)
                used_at_node[v].remove(c)
        
        return False
    
    if dfs(0):
        print("YES")
        print(' '.join(map(str, assignment)))
    else:
        print("NO")

solve()
