import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # ---------- Edmonds' Blossom for maximum matching ----------
    N = n
    g = [[] for _ in range(N)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    match = [-1] * N
    p = [-1] * N
    base = list(range(N))
    q = []
    used = [False] * N
    blossom = [False] * N

    def lca(a, b):
        used_path = [False] * N
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal q, used, p, base, blossom
        used = [False] * N
        p = [-1] * N
        base = list(range(N))
        q = []
        q_head = 0
        q.append(root)
        used[root] = True
        while q_head < len(q):
            v = q[q_head]; q_head += 1
            for to in g[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    curbase = lca(v, to)
                    blossom = [False] * N
                    mark_path(v, curbase, to)
                    mark_path(to, curbase, v)
                    for i in range(N):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        # augmenting
                        v = to
                        while v != -1:
                            pv = p[v]
                            nv = match[pv] if pv != -1 else -1
                            match[v] = pv
                            match[pv] = v
                            v = nv
                        return True
                    else:
                        used[match[to]] = True
                        q.append(match[to])
        return False

    for v in range(N):
        if match[v] == -1:
            find_path(v)

    # ---------- check perfect matching ----------
    if any(mch == -1 for mch in match):
        print("NO")
        return

    # build contracted graph of supernodes (each matched pair -> node)
    pair_id = [-1] * N
    super_cnt = 0
    for v in range(N):
        if pair_id[v] == -1:
            u = match[v]
            pair_id[v] = pair_id[u] = super_cnt
            super_cnt += 1

    # each original edge not in matching becomes edge between supernodes
    super_adj = [[] for _ in range(super_cnt)]
    edge_color = [0] * m
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            edge_color[idx] = 3  # matching edges get color 3
        else:
            a = pair_id[u]
            b = pair_id[v]
            super_adj[a].append((b, idx))
            super_adj[b].append((a, idx))

    # now each supernode has degree 2 -> cycles
    visited = [False] * super_cnt
    for start in range(super_cnt):
        if visited[start]:
            continue
        # traverse cycle
        cur = start
        prev_edge = -1
        color = 1
        while True:
            visited[cur] = True
            # find next edge not the one we came from
            for nb, eidx in super_adj[cur]:
                if eidx != prev_edge:
                    break
            else:
                break
            edge_color[eidx] = color
            color = 2 if color == 1 else 1
            prev_edge = eidx
            cur = nb
            if cur == start:
                break

    print("YES")
    print(' '.join(str(c) for c in edge_color))

if __name__ == "__main__":
    solve()
