import sys

# Increase recursion depth just in case
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj_edges = [[] for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj_edges[u].append(i)
        adj_edges[v].append(i)
    
    # Precompute incident edges for each edge to speed up propagation
    neighbor_edges = [[] for _ in range(m)]
    for i in range(m):
        u, v = edges[i]
        nbrs = set()
        for e in adj_edges[u]:
            if e != i:
                nbrs.add(e)
        for e in adj_edges[v]:
            if e != i:
                nbrs.add(e)
        neighbor_edges[i] = list(nbrs)
        
    color = [0] * m
    avail_mask = [7] * m  # bit 0: color 1, bit 1: color 2, bit 2: color 3 (value 7 = 111 in binary)
    
    # We can process connected components
    visited_nodes = [False] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if not visited_nodes[i]:
            comp_nodes = []
            q = [i]
            visited_nodes[i] = True
            for node in q:
                comp_nodes.append(node)
                for e in adj_edges[node]:
                    u, v = edges[e]
                    nxt = v if u == node else u
                    if not visited_nodes[nxt]:
                        visited_nodes[nxt] = True
                        q.append(nxt)
            components.append(comp_nodes)

    def assign(e, c, changes):
        # Assign color c (1, 2, or 3) to edge e
        bit = 1 << (c - 1)
        color[e] = c
        # For all neighbor edges, remove bit
        for ne in neighbor_edges[e]:
            if color[ne] == 0 and (avail_mask[ne] & bit):
                avail_mask[ne] &= ~bit
                changes.append((ne, bit))

    def backtrack():
        # Find unassigned edge with min possibilities
        best_e = -1
        best_count = 4
        best_mask = 0
        
        for e in range(m):
            if color[e] == 0:
                mask = avail_mask[e]
                cnt = (mask & 1) + ((mask >> 1) & 1) + ((mask >> 2) & 1)
                if cnt == 0:
                    return False
                if cnt < best_count:
                    best_count = cnt
                    best_e = e
                    best_mask = mask
                    if cnt == 1:
                        break
        
        if best_e == -1:
            return True
        
        # Try colors
        for c in (1, 2, 3):
            bit = 1 << (c - 1)
            if best_mask & bit:
                changes = []
                assign(best_e, c, changes)
                if backtrack():
                    return True
                # Undo
                color[best_e] = 0
                for ne, b in changes:
                    avail_mask[ne] |= b
                    
        return False

    # For each component, break symmetry by assigning colors 1, 2, 3 to edges of one vertex
    init_changes = []
    possible = True
    for comp in components:
        v0 = comp[0]
        v0_edges = adj_edges[v0]
        if len(v0_edges) == 3:
            for c_idx, e in enumerate(v0_edges):
                c = c_idx + 1
                if (avail_mask[e] & (1 << (c - 1))) == 0:
                    possible = False
                    break
                assign(e, c, init_changes)
        if not possible:
            break

    if possible and backtrack():
        print("YES")
        print(*(color[i] for i in range(m)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
