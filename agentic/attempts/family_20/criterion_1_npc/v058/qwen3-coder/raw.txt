import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    # Store edges with their indices
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Each node must have exactly 3 edges
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    # Colors for edges (0 means uncolored)
    colors = [0] * m
    
    # Try to 3-color the graph
    # This is essentially checking if the line graph is 3-colorable
    # But since each vertex has degree 3 and we want each edge incident 
    # to a vertex to have different colors, this becomes a specific type of coloring
    
    # For cubic graphs, we can try to find a proper 3-edge-coloring
    # Build adjacency list for edges - two edges are adjacent if they share a vertex
    edge_adj = [[] for _ in range(m)]
    
    # For each vertex, connect all its incident edges
    for v in range(1, n + 1):
        incident_edges = [i for _, i in adj[v]]
        for i in range(len(incident_edges)):
            for j in range(i + 1, len(incident_edges)):
                edge_adj[incident_edges[i]].append(incident_edges[j])
                edge_adj[incident_edges[j]].append(incident_edges[i])
    
    # Try to color edges with 3 colors using BFS
    def bfs_color(start_edge):
        queue = deque([start_edge])
        if colors[start_edge] == 0:
            colors[start_edge] = 1
            
        while queue:
            e = queue.popleft()
            current_color = colors[e]
            
            # Available colors (1, 2, 3) minus neighbors' colors
            used_colors = set()
            for neighbor in edge_adj[e]:
                if colors[neighbor] != 0:
                    used_colors.add(colors[neighbor])
                elif neighbor not in queue:
                    queue.append(neighbor)
            
            # Assign first available color if not colored
            if colors[e] == 0:
                for c in [1, 2, 3]:
                    if c not in used_colors:
                        colors[e] = c
                        break
                else:
                    return False
        
        return True
    
    # Color component by component
    visited = [False] * m
    for i in range(m):
        if colors[i] == 0:
            # Try coloring with color 1 initially
            colors[i] = 1
            # Do BFS to propagate coloring
            queue = deque([i])
            
            while queue:
                e = queue.popleft()
                if visited[e]:
                    continue
                visited[e] = True
                
                current_color = colors[e]
                # Find used colors among neighbors
                for neighbor in edge_adj[e]:
                    if colors[neighbor] == 0:
                        # Determine what colors are forbidden
                        forbidden = set()
                        for n2 in edge_adj[neighbor]:
                            if colors[n2] != 0:
                                forbidden.add(colors[n2])
                        
                        # Pick first available color
                        for c in [1, 2, 3]:
                            if c not in forbidden:
                                colors[neighbor] = c
                                break
                        else:
                            print("NO")
                            return
                        
                        if not visited[neighbor]:
                            queue.append(neighbor)
                    else:
                        # Check for conflict
                        if colors[neighbor] == colors[e]:
                            print("NO")
                            return
    
    # Verify solution
    for v in range(1, n + 1):
        incident_edges = [i for _, i in adj[v]]
        edge_colors = [colors[i] for i in incident_edges]
        if len(set(edge_colors)) != 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, colors)))

solve()
