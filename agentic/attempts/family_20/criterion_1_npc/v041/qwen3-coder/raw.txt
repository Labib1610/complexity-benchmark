import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # The key insight: we need each vertex to have exactly 3 edges,
    # and the product of pressures at each vertex must be 6.
    # The only ways to factor 6 as a product of three numbers from {1,2,3} are:
    # (1,2,3) in some order.
    
    # This means every vertex must be incident to exactly one pressure-1 edge,
    # one pressure-2 edge, and one pressure-3 edge.
    
    # This is equivalent to edge-coloring the graph with colors {1,2,3} such that
    # each vertex is incident to exactly one edge of each color.
    # This is the same as decomposing the graph into perfect matchings.
    
    # Since the graph is 3-regular (every vertex has degree 3) and has n vertices (n even),
    # it's possible to decompose it into 3 perfect matchings if and only if
    # the graph is isomorphic to a disjoint union of triangles... No, that's not right.
    
    # Actually, every 3-regular graph can be decomposed into perfect matchings?
    # Not true in general. But there are special cases where it works.
    
    # Let's think differently. We're looking for a valid assignment.
    # One approach: try to find such a decomposition.
    
    # But let's look at the example:
    # Vertices: 1,2,3,4. Each has degree 3.
    # Edges: 1-2, 1-3, 1-4, 2-3, 2-4, 3-4. This is a complete graph K_4.
    # In K_4, we can indeed decompose edges into 3 perfect matchings:
    # M1 = {1-2, 3-4}, M2 = {1-3, 2-4}, M3 = {1-4, 2-3}
    
    # So the approach:
    # 1. Check if we can decompose the graph into 3 perfect matchings.
    # 2. If yes, assign pressure values accordingly.
    
    # But how to check if a 3-regular graph can be decomposed into 3 perfect matchings?
    # This is equivalent to proper 3-edge-coloring.
    
    # For small graphs, we can try backtracking.
    # For larger graphs, there might be structural properties.
    
    # Let's implement a backtracking solution with constraint propagation.
    
    # Variables: for each edge, a value in {1, 2, 3}
    # Constraints: for each vertex, product of its 3 edges is 6
    
    # The only integer solutions to a*b*c = 6 with a,b,c in {1,2,3} are permutations of (1,2,3).
    # So each vertex must be incident to one edge of each type.
    
    # This is exactly a 3-edge coloring problem.
    
    # Try to build a solution using backtracking with constraint checking.
    
    pressure = [0] * m  # 0 means unassigned
    
    def is_valid(vertex):
        """Check if the constraints for 'vertex' are currently satisfied"""
        assigned_values = []
        unassigned_count = 0
        
        for neighbor, edge_idx in adj[vertex]:
            if pressure[edge_idx] == 0:
                unassigned_count += 1
            else:
                assigned_values.append(pressure[edge_idx])
        
        # If all assigned, check product
        if unassigned_count == 0:
            prod = 1
            for val in assigned_values:
                prod *= val
            return prod == 6
        
        # If partially assigned, check if completion is possible
        current_product = 1
        for val in assigned_values:
            current_product *= val
            
        # We need to assign 'unassigned_count' values from {1,2,3}
        # such that the final product is 6
        
        if current_product > 6:
            return False
            
        if 6 % current_product != 0:
            return False
            
        target_remaining = 6 // current_product
        
        # Can we make 'target_remaining' as a product of 'unassigned_count' values from {1,2,3}?
        # Generate all combinations
        def can_factor(num, count):
            if count == 0:
                return num == 1
            if count == 1:
                return num in [1, 2, 3]
            
            # For count >= 2
            for f in [1, 2, 3]:
                if num % f == 0 and can_factor(num // f, count - 1):
                    return True
            return False
        
        return can_factor(target_remaining, unassigned_count)
    
    def backtrack(edge_idx):
        if edge_idx == m:
            return True
            
        u, v = edges[edge_idx]
        
        for p in [1, 2, 3]:
            pressure[edge_idx] = p
            
            # Check local consistency
            if is_valid(u) and is_valid(v):
                if backtrack(edge_idx + 1):
                    return True
                    
        pressure[edge_idx] = 0
        return False
    
    # Optimization: start with vertices that have more assigned edges
    # Or process edges in a smarter order
    
    # For now, let's just run the backtracking
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, pressure)))
    else:
        print("NO")

solve()
