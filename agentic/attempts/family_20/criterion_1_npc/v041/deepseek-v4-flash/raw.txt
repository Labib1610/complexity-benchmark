We can model this as a graph problem. Each pipe is an edge, and each vat is a vertex of degree 3.  
The product condition means the three numbers at each vertex must be a permutation of (1, 2, 3).  

This is equivalent to a **3-edge-coloring** of a cubic graph, where each vertex sees all three colors exactly once.  
A cubic graph has a proper 3-edge-coloring if and only if it is bridgeless (by the 1-factorization theorem for cubic graphs).  

So the algorithm is:
1. Check if the graph is bridgeless (no bridges).
2. If it has a bridge, output `NO`.
3. Otherwise, find a proper 3-edge-coloring using a standard algorithm (e.g., using BFS/DFS and alternating paths).

Here’s the implementation:  

```python
import sys
from collections import deque

def find_bridges(n, adj, edges):
    visited = [False] * (n + 1)
    tin = [-1] * (n + 1)
    low = [-1] * (n + 1)
    timer = 0
    bridges = set()

    def dfs(v, p_edge):
        nonlocal timer
        visited[v] = True
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridges.add(eid)
    for i in range(1, n + 1):
        if not visited[i]:
            dfs(i, -1)
    return bridges

def edge_color_cubic(n, adj, edges):
    # We'll use a simple greedy + Kempe chain approach
    color = [-1] * len(edges)
    # For each vertex, track used colors
    used_at = [set() for _ in range(n + 1)]

    # We'll color edges one by one using BFS to find alternating paths
    for eid, (u, v) in enumerate(edges):
        if color[eid] != -1:
            continue
        # Find a color not used at u and not used at v
        possible = [c for c in (1, 2, 3) if c not in used_at[u] and c not in used_at[v]]
        if possible:
            c = possible[0]
            color[eid] = c
            used_at[u].add(c)
            used_at[v].add(c)
        else:
            # Need to recolor via Kempe chain
            # Pick a color missing at u and a color missing at v
            missing_u = [c for c in (1, 2, 3) if c not in used_at[u]][0]
            missing_v = [c for c in (1, 2, 3) if c not in used_at[v]][0]
            # Build Kempe chain for missing_u and missing_v starting from u
            # Swap colors along the chain
            chain_color1 = missing_u
            chain_color2 = missing_v
            # BFS/DFS to find component of edges colored with these two colors
            stack = [(u, eid)]
            visited_edges = set()
            visited_vertices = set([u])
            while stack:
                node, peid = stack.pop()
                for to, neid in adj[node]:
                    if neid == peid:
                        continue
                    if color[neid] in (chain_color1, chain_color2):
                        if neid not in visited_edges:
                            visited_edges.add(neid)
                            visited_vertices.add(to)
                            stack.append((to, neid))
            # Swap colors in this component
            for e in visited_edges:
                if color[e] == chain_color1:
                    color[e] = chain_color2
                else:
                    color[e] = chain_color1
            # Now assign the missing color to current edge
            color[eid] = chain_color1
            used_at[u].add(chain_color1)
            used_at[v].add(chain_color1)
            # Update used_at for vertices in the chain
            for e in visited_edges:
                a, b = edges[e]
                used_at[a].discard(chain_color1 if color[e] == chain_color2 else chain_color2)
                used_at[a].add(color[e])
                used_at[b].discard(chain_color1 if color[e] == chain_color2 else chain_color2)
                used_at[b].add(color[e])
    return color

def main():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    adj = [[] for _ in range(n + 1)]
    for eid in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, eid))
        adj[v].append((u, eid))

    bridges = find_bridges(n, adj, edges)
    if bridges:
        print("NO")
        return

    color = edge_color_cubic(n, adj, edges)
    print("YES")
    print(" ".join(map(str, color)))

if __name__ == "__main__":
    main()
```
