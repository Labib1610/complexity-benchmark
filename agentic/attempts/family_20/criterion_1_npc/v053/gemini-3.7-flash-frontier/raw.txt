import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
        
    # Quick check 1: Bridge detection
    # A 3-edge-colorable cubic graph cannot contain any bridges.
    tin = [0] * (n + 1)
    low = [0] * (n + 1)
    timer = 0
    has_bridge = False
    
    def dfs_bridge(u, p_edge):
        nonlocal timer, has_bridge
        timer += 1
        tin[u] = low[u] = timer
        for v, e_idx in adj[u]:
            if e_idx == p_edge:
                continue
            if tin[v]:
                low[u] = min(low[u], tin[v])
            else:
                dfs_bridge(v, e_idx)
                low[u] = min(low[u], low[v])
                if low[v] > tin[u]:
                    has_bridge = True

    for i in range(1, n + 1):
        if not tin[i]:
            dfs_bridge(i, -1)
            
    if has_bridge:
        print("NO")
        return

    # Backtracking with MRV / Kempe-style heuristics
    edge_color = [0] * m
    vertex_used = [0] * (n + 1)  # bitmask of used colors (bits 1, 2, 3)

    # Order edges by a connected traversal (BFS/DFS) to keep branching local
    ordered_edges = []
    visited_edges = [False] * m
    for start_node in range(1, n + 1):
        queue = [start_node]
        for u in queue:
            for v, e_idx in adj[u]:
                if not visited_edges[e_idx]:
                    visited_edges[e_idx] = True
                    ordered_edges.append(e_idx)
                    queue.append(v)

    # Backtracking search
    limit_steps = 2000000
    steps = 0

    def backtrack(idx):
        nonlocal steps
        steps += 1
        if steps > limit_steps:
            return False
        
        if idx == m:
            return True
        
        e_idx = ordered_edges[idx]
        u, v = edges[e_idx]
        
        # Available colors are {1, 2, 3} not used by u and not used by v
        used = vertex_used[u] | vertex_used[v]
        
        # Try available colors
        for c in (1, 2, 3):
            mask = 1 << c
            if not (used & mask):
                edge_color[e_idx] = c
                vertex_used[u] |= mask
                vertex_used[v] |= mask
                
                if backtrack(idx + 1):
                    return True
                
                vertex_used[u] &= ~mask
                vertex_used[v] &= ~mask
                edge_color[e_idx] = 0
                
        return False

    if backtrack(0):
        print("YES")
        print(*(edge_color[i] for i in range(m)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
