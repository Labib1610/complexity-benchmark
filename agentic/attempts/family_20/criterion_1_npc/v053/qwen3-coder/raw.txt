import sys
from collections import defaultdict

def solve():
    n, m = map(int, input().split())
    
    # Store edges with their indices
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Each node must have exactly 3 edges
    for node in adj:
        if len(adj[node]) != 3:
            print("NO")
            return
    
    # Try to 3-color the edges such that each node has edges of colors 1, 2, 3
    # This is equivalent to edge coloring of a 3-regular graph with 3 colors
    
    # We'll use a greedy approach with backtracking if needed
    # But for a 3-regular graph, if it's connected and has a valid edge coloring,
    # we can often find one greedily
    
    color = [0] * m  # 1-based indexing for colors, 0 means uncolored
    
    # For each node, we need to assign colors 1,2,3 to its 3 edges
    # Let's try a systematic approach
    
    # Build adjacency for edges (line graph approach)
    # But simpler: just ensure at each node, all 3 edges have different colors
    
    def is_valid_color(edge_idx, c, node):
        u, v = edges[edge_idx]
        # Check if any adjacent edge to node already has color c
        for neighbor, nei_edge_idx in adj[node]:
            if color[nei_edge_idx] == c:
                return False
        return True
    
    # Try to color edges one by one
    def dfs(edge_idx):
        if edge_idx == m:
            return True
            
        u, v = edges[edge_idx]
        
        # Try colors 1, 2, 3
        for c in range(1, 4):
            # Check if we can assign color c to this edge
            # Both endpoints should not have an edge with color c already
            valid = True
            for node in [u, v]:
                for neighbor, nei_edge_idx in adj[node]:
                    if color[nei_edge_idx] == c:
                        # Check if that edge is incident to our current node
                        if neighbor == node or nei_edge_idx == edge_idx:
                            valid = False
                            break
                if not valid:
                    break
                    
            if valid:
                color[edge_idx] = c
                if dfs(edge_idx + 1):
                    return True
                color[edge_idx] = 0
                
        return False
    
    # More efficient approach:
    # Since each vertex has degree exactly 3, we can do this systematically
    
    color = [0] * m
    
    # For each vertex, we need to assign colors 1,2,3 to its 3 edges
    # Let's process vertices one by one
    
    colored_edges = [False] * m
    
    # Process each node and color its edges
    for node in range(1, n+1):
        edge_indices = [edge_info[1] for edge_info in adj[node]]
        available_colors = {1, 2, 3}
        
        # First, see what colors are already used by colored edges
        for edge_idx in edge_indices:
            if colored_edges[edge_idx]:
                available_colors.discard(color[edge_idx])
        
        # Now assign remaining colors to uncolored edges
        uncolored_edges = [edge_idx for edge_idx in edge_indices if not colored_edges[edge_idx]]
        
        # Assign colors to uncolored edges
        uncolored_colors = list(available_colors)
        if len(uncolored_colors) < len(uncolored_edges):
            print("NO")
            return
            
        for i, edge_idx in enumerate(uncolored_edges):
            if i < len(uncolored_colors):
                color[edge_idx] = uncolored_colors[i]
                colored_edges[edge_idx] = True
            else:
                # This shouldn't happen if the logic is right
                print("NO")
                return
    
    # Verify the solution
    for node in range(1, n+1):
        edge_indices = [edge_info[1] for edge_info in adj[node]]
        colors_here = [color[idx] for idx in edge_indices]
        if len(set(colors_here)) != 3 or 0 in colors_here:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
