import sys

# Increase recursion depth just in case
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj_edges = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj_edges[u].append(i)
        adj_edges[v].append(i)

    # Check for bridges using Tarjan's bridge-finding algorithm
    # A cubic graph with a bridge cannot have a 1-factorization (3-edge-coloring).
    tin = [0] * (n + 1)
    low = [0] * (n + 1)
    timer = 0
    has_bridge = False
    
    def dfs_bridge(u, p_edge):
        nonlocal timer, has_bridge
        timer += 1
        tin[u] = low[u] = timer
        for e_idx in adj_edges[u]:
            if e_idx == p_edge:
                continue
            v = edges[e_idx][0] if edges[e_idx][1] == u else edges[e_idx][1]
            if tin[v] > 0:
                if tin[v] < low[u]:
                    low[u] = tin[v]
            else:
                dfs_bridge(v, e_idx)
                if low[v] < low[u]:
                    low[u] = low[v]
                if low[v] > tin[u]:
                    has_bridge = True

    for i in range(1, n + 1):
        if tin[i] == 0:
            dfs_bridge(i, -1)

    if has_bridge:
        print("NO")
        return

    # Build conflict graph of edges (line graph)
    # Each edge is adjacent to up to 4 other edges
    edge_adj = [[] for _ in range(m)]
    for u in range(1, n + 1):
        inc = adj_edges[u]
        l = len(inc)
        for i in range(l):
            for j in range(i + 1, l):
                edge_adj[inc[i]].append(inc[j])
                edge_adj[inc[j]].append(inc[i])

    # 3-edge coloring via backtracking with MRV and Forward Checking
    # Domain representation: bitmask. 1 -> 1, 2 -> 2, 3 -> 4. Full domain = 7 (binary 111).
    domain = [7] * m
    color = [0] * m
    assigned_count = 0

    # Fast bit operations
    popcnt = [0, 1, 1, 2, 1, 2, 2, 3]

    steps = 0
    max_steps = 200000

    def get_first_color(mask):
        if mask & 1: return 1
        if mask & 2: return 2
        if mask & 4: return 3
        return 0

    def backtrack():
        nonlocal assigned_count, steps
        steps += 1
        if steps > max_steps:
            return False

        if assigned_count == m:
            return True

        # MRV: find unassigned edge with minimum domain size
        best_e = -1
        min_size = 4

        for e in range(m):
            if color[e] == 0:
                sz = popcnt[domain[e]]
                if sz < min_size:
                    min_size = sz
                    best_e = e
                    if min_size <= 1:
                        break

        if min_size == 0:
            return False

        e = best_e
        mask = domain[e]

        # Try colors present in domain[e]
        for c in (1, 2, 3):
            c_bit = 1 << (c - 1)
            if mask & c_bit:
                # Assign color c to edge e
                color[e] = c
                assigned_count += 1
                
                # Forward check
                pruned = []
                conflict = False
                for neighbor in edge_adj[e]:
                    if color[neighbor] == 0 and (domain[neighbor] & c_bit):
                        domain[neighbor] &= ~c_bit
                        pruned.append(neighbor)
                        if domain[neighbor] == 0:
                            conflict = True

                if not conflict:
                    if backtrack():
                        return True

                # Undo forward check
                for neighbor in pruned:
                    domain[neighbor] |= c_bit
                color[e] = 0
                assigned_count -= 1

        return False

    if backtrack():
        print("YES")
        print(*(color[i] for i in range(m)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
