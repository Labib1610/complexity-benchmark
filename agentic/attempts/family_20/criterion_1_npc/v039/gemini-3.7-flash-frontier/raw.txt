import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
        
    def try_coloring():
        color = [0] * m
        # edge_with_color[u][c] is the edge index incident to u with color c (1, 2, or 3)
        edge_with_color = [[-1] * 4 for _ in range(n + 1)]
        
        uncolored = list(range(m))
        random.shuffle(uncolored)
        
        max_steps = 150000
        steps = 0
        
        while uncolored and steps < max_steps:
            steps += 1
            e = uncolored.pop()
            if color[e] != 0:
                continue
            
            u, v = edges[e]
            
            free_u = [c for c in (1, 2, 3) if edge_with_color[u][c] == -1]
            free_v = [c for c in (1, 2, 3) if edge_with_color[v][c] == -1]
            
            # Case 1: common free color
            common = set(free_u) & set(free_v)
            if common:
                c = common.pop()
                color[e] = c
                edge_with_color[u][c] = e
                edge_with_color[v][c] = e
                continue
            
            # Case 2: Kempe chain swap
            colored_successfully = False
            # Shuffle pairs to introduce randomness
            pairs = [(c1, c2) for c1 in free_u for c2 in free_v]
            random.shuffle(pairs)
            
            for c1, c2 in pairs:
                # Trace (c1, c2) path from u starting with color c2
                curr = u
                curr_c = c2
                path = []
                while edge_with_color[curr][curr_c] != -1:
                    e_path = edge_with_color[curr][curr_c]
                    path.append(e_path)
                    eu, ev = edges[e_path]
                    curr = ev if curr == eu else eu
                    curr_c = c1 if curr_c == c2 else c2
                
                if curr != v:
                    # Successfully found a Kempe chain not ending at v
                    # Swap colors along the path
                    for e_p in path:
                        c_old = color[e_p]
                        c_new = c1 if c_old == c2 else c2
                        pu, pv = edges[e_p]
                        edge_with_color[pu][c_old] = -1
                        edge_with_color[pv][c_old] = -1
                        edge_with_color[pu][c_new] = e_p
                        edge_with_color[pv][c_new] = e_p
                        color[e_p] = c_new
                    
                    # Now c2 is free at both u and v
                    color[e] = c2
                    edge_with_color[u][c2] = e
                    edge_with_color[v][c2] = e
                    colored_successfully = True
                    break
            
            if colored_successfully:
                continue
            
            # Case 3: All Kempe chains reach v. Kick a random incident edge.
            uncolored.append(e)
            target = u if random.random() < 0.5 else v
            colored_inc = [edge_idx for edge_idx in adj[target] if color[edge_idx] != 0]
            if colored_inc:
                e_kick = random.choice(colored_inc)
                c_kick = color[e_kick]
                ku, kv = edges[e_kick]
                edge_with_color[ku][c_kick] = -1
                edge_with_color[kv][c_kick] = -1
                color[e_kick] = 0
                uncolored.append(e_kick)
        
        if all(c != 0 for c in color):
            return color
        return None

    # Try several random restarts
    for _ in range(6):
        res = try_coloring()
        if res is not None:
            print("YES")
            print(*(res[i] for i in range(m)))
            return
            
    print("NO")

if __name__ == '__main__':
    solve()
