We can model this as a **3-edge-coloring** problem on a **3-regular graph** (each vertex has degree 3).  
By **Kőnig’s line-coloring theorem**, every bipartite graph can be edge-colored with Δ colors, but here the graph is not necessarily bipartite.  
However, for **cubic graphs**, a 3-edge-coloring exists if and only if the graph is **bridgeless** (by the **Petersen’s theorem**).  
We can check for bridges and if none exist, we can find the 3-edge-coloring using a constructive algorithm.

We’ll implement:
1. Check if the graph is connected and bridgeless.
2. If yes, find a 3-edge-coloring using a known constructive method (e.g., using a perfect matching decomposition or a DFS-based approach).

Here’s the Python solution:

```python
import sys
sys.setrecursionlimit(1000000)

def find_bridges(n, edges):
    g = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        g[u].append((v, idx))
        g[v].append((u, idx))
    visited = [False] * n
    tin = [-1] * n
    low = [-1] * n
    timer = 0
    bridges = set()

    def dfs(v, p_edge):
        nonlocal timer
        visited[v] = True
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in g[v]:
            if eid == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridges.add(eid)
    dfs(0, -1)
    return bridges

def edge_color_3(n, edges):
    m = len(edges)
    g = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        g[u].append((v, idx))
        g[v].append((u, idx))
    color = [-1] * m

    # Use a known constructive algorithm for cubic bridgeless graphs:
    # Find a perfect matching, remove it, the remaining graph is 2-regular (cycles),
    # color the matching with color 1, then color the cycles alternately with 2 and 3.
    # To find a perfect matching in a cubic bridgeless graph, we can use a simple
    # greedy + augmenting path approach (since n <= 2000, it's fine).

    # Build adjacency for matching
    adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    match_edge = [-1] * n  # edge id matched to vertex
    match_vertex = [-1] * n  # matched partner vertex

    def augment(u, visited):
        for v, eid in adj[u]:
            if visited[v]:
                continue
            visited[v] = True
            if match_vertex[v] == -1 or augment(match_vertex[v], visited):
                match_vertex[u] = v
                match_vertex[v] = u
                match_edge[u] = eid
                match_edge[v] = eid
                return True
        return False

    for u in range(n):
        if match_vertex[u] == -1:
            visited = [False] * n
            visited[u] = True
            augment(u, visited)

    # Now match_edge contains the perfect matching edges
    matching_edges = set()
    for u in range(n):
        if match_edge[u] != -1:
            matching_edges.add(match_edge[u])

    # Color matching edges with 1
    for eid in matching_edges:
        color[eid] = 1

    # Remaining edges form disjoint cycles
    remaining_adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        if idx not in matching_edges:
            remaining_adj[u].append((v, idx))
            remaining_adj[v].append((u, idx))

    visited_vertex = [False] * n
    for start in range(n):
        if not visited_vertex[start] and remaining_adj[start]:
            # traverse cycle
            cycle_edges = []
            cur = start
            prev = -1
            while True:
                visited_vertex[cur] = True
                nxt = None
                for to, eid in remaining_adj[cur]:
                    if eid != prev:
                        nxt = (to, eid)
                        break
                if nxt is None:
                    break
                to, eid = nxt
                cycle_edges.append(eid)
                prev = eid
                cur = to
                if cur == start:
                    break
            # color alternately 2,3
            for i, eid in enumerate(cycle_edges):
                color[eid] = 2 if i % 2 == 0 else 3

    return color

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))

    # Check if graph is connected
    g = [[] for _ in range(n)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)
    visited = [False] * n
    stack = [0]
    visited[0] = True
    while stack:
        u = stack.pop()
        for v in g[u]:
            if not visited[v]:
                visited[v] = True
                stack.append(v)
    if not all(visited):
        print("NO")
        return

    # Check for bridges
    bridges = find_bridges(n, edges)
    if bridges:
        print("NO")
        return

    # Find 3-edge-coloring
    color = edge_color_3(n, edges)
    print("YES")
    print(" ".join(str(c) for c in color))

if __name__ == "__main__":
    main()
```
