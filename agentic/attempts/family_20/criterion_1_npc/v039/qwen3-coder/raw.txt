import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list and edge list
    adj = defaultdict(list)
    edges = []
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append((v, i))
        adj[v].append((u, i))
        edges.append((u, v))
    
    # Color array for edges (0 means uncolored)
    color = [0] * m
    
    # Try to 3-color the line graph (edges are nodes, adjacent if they share a vertex)
    # This is equivalent to assigning windows to cables such that edges incident to 
    # each vertex have different colors
    
    # We'll use BFS to color each component
    visited_edge = [False] * m
    
    for start_edge in range(m):
        if visited_edge[start_edge]:
            continue
            
        # BFS from this edge
        queue = deque([start_edge])
        visited_edge[start_edge] = True
        
        # Assign color 1 to the starting edge
        color[start_edge] = 1
        
        while queue:
            edge_idx = queue.popleft()
            u, v = edges[edge_idx]
            
            # For each endpoint of the current edge, look at all incident edges
            for node in [u, v]:
                current_color = color[edge_idx]
                # Try colors 1, 2, 3 in order for neighbors
                used_colors = set()
                
                # Check all edges incident to this node
                for neighbor, neighbor_idx in adj[node]:
                    if visited_edge[neighbor_idx]:
                        used_colors.add(color[neighbor_idx])
                
                # Assign colors to unvisited neighbors
                for neighbor, neighbor_idx in adj[node]:
                    if not visited_edge[neighbor_idx]:
                        # Find a color not used by already colored edges at this node
                        available_colors = {1, 2, 3} - used_colors
                        # But also not conflicting with the current edge if they're adjacent
                        # Actually, we just need to ensure at each vertex, all incident edges have diff colors
                        
                        # Find a color not used yet at this vertex
                        c = 1
                        while c in used_colors:
                            c += 1
                        if c > 3:
                            print("NO")
                            return
                            
                        color[neighbor_idx] = c
                        visited_edge[neighbor_idx] = True
                        queue.append(neighbor_idx)
                        used_colors.add(c)
    
    # Verify the solution
    # For each vertex, check that all incident edges have different colors
    for node in range(n):
        colors_at_node = []
        for neighbor, edge_idx in adj[node]:
            colors_at_node.append(color[edge_idx])
        if len(set(colors_at_node)) != len(colors_at_node):
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
