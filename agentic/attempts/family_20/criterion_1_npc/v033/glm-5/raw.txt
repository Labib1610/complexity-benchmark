import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check degree constraint
    for u in range(1, n + 1):
        if len(adj[u]) != 3:
            print("NO")
            return
    
    # Build edge adjacency: which edges share a vertex
    edge_adj = [[] for _ in range(m)]
    for u in range(1, n + 1):
        incident_edges = [i for (v, i) in adj[u]]
        for i in range(3):
            for j in range(i + 1, 3):
                e1 = incident_edges[i]
                e2 = incident_edges[j]
                edge_adj[e1].append(e2)
                edge_adj[e2].append(e1)
    
    # 3-coloring edges using backtracking with constraint propagation
    color = [0] * m
    
    def get_possible_colors(e):
        used = set()
        for nb in edge_adj[e]:
            if color[nb] != 0:
                used.add(color[nb])
        return [c for c in [1, 2, 3] if c not in used]
    
    # Find uncolored edge with minimum remaining values (MRV heuristic)
    def select_edge():
        best = -1
        best_count = 4
        for e in range(m):
            if color[e] == 0:
                cnt = len(get_possible_colors(e))
                if cnt < best_count:
                    best_count = cnt
                    best = e
                    if cnt == 0:
                        return best  # Early termination
        return best
    
    def solve_recursive():
        e = select_edge()
        if e == -1:
            return True
        possible = get_possible_colors(e)
        if not possible:
            return False
        for c in possible:
            color[e] = c
            if solve_recursive():
                return True
            color[e] = 0
        return False
    
    if solve_recursive():
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
