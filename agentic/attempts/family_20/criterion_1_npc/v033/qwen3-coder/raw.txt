import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Store edges in order
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Try to 3-color the edges such that edges incident to same vertex have different colors
    # This is equivalent to edge coloring of a cubic graph with 3 colors
    
    # We'll use backtracking with constraint propagation
    color = [0] * m  # 0 means uncolored
    
    def is_valid(edge_idx, c):
        u, v = edges[edge_idx]
        # Check if any adjacent edge at u or v already has color c
        for nei_edge in adj[u]:
            if color[nei_edge] == c:
                return False
        for nei_edge in adj[v]:
            if color[nei_edge] == c:
                return False
        return True
    
    def backtrack(edge_idx):
        if edge_idx == m:
            return True
            
        for c in [1, 2, 3]:
            if is_valid(edge_idx, c):
                color[edge_idx] = c
                if backtrack(edge_idx + 1):
                    return True
                color[edge_idx] = 0
        
        return False
    
    # Since this is a cubic graph and we need to 3-edge-color it,
    # we can try a more systematic approach
    
    # Build adjacency for edges (line graph)
    # Two edges are adjacent if they share a vertex
    edge_adj = [[] for _ in range(m)]
    for v in range(1, n+1):
        edge_list = adj[v]
        # All edges incident to v are pairwise adjacent
        for i in range(len(edge_list)):
            for j in range(i+1, len(edge_list)):
                edge_adj[edge_list[i]].append(edge_list[j])
                edge_adj[edge_list[j]].append(edge_list[i])
    
    # Now we need to 3-color the line graph
    # Use greedy + backtracking
    
    def can_color(edge_idx, c):
        for neighbor in edge_adj[edge_idx]:
            if color[neighbor] == c:
                return False
        return True
    
    def solve_coloring():
        # Try to color greedily first
        for i in range(m):
            for c in [1, 2, 3]:
                if can_color(i, c):
                    color[i] = c
                    break
        
        # Check if solution is complete
        if 0 not in color:
            return True
            
        # Otherwise do backtracking
        def bt(idx):
            if idx == m:
                return True
            if color[idx] != 0:
                return bt(idx + 1)
                
            for c in [1, 2, 3]:
                if can_color(idx, c):
                    color[idx] = c
                    if bt(idx + 1):
                        return True
                    color[idx] = 0
            return False
            
        return bt(0)
    
    # For cubic graphs, if a valid 3-edge-coloring exists, it can be found
    # But some cubic graphs are not 3-edge-colorable (like the Petersen graph)
    
    # Let's try a different approach - since every vertex has degree 3,
    # we can model this as finding a proper edge coloring
    
    # More robust approach: try all possibilities with constraint checking
    def dfs(pos):
        if pos == m:
            return True
            
        u, v = edges[pos]
        used_colors = set()
        
        # Check what colors are used by adjacent edges
        for edge_idx in adj[u]:
            if edge_idx < pos and color[edge_idx] != 0:
                used_colors.add(color[edge_idx])
                
        for edge_idx in adj[v]:
            if edge_idx < pos and color[edge_idx] != 0:
                used_colors.add(color[edge_idx])
        
        # Try available colors
        for c in [1, 2, 3]:
            if c not in used_colors:
                color[pos] = c
                if dfs(pos + 1):
                    return True
                color[pos] = 0
                
        return False
    
    if dfs(0):
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
