import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check degree constraint
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    # Edge coloring a 3-regular graph with 3 colors
    # Using Vizing's theorem approach for class I graphs
    # Since n is even and graph is 3-regular, it's class I (by Petersen's theorem for bridgeless graphs)
    
    color = [0] * m
    # edge_at[u][c] = edge index of color c incident to u
    edge_at = [[-1] * 3 for _ in range(n + 1)]
    
    def find_alt_path(u, c1, c2, start_edge):
        """Find alternating path starting from u, avoiding start_edge"""
        path = []
        visited_edges = set()
        current_u = u
        current_c = c1
        while True:
            e = edge_at[current_u][current_c]
            if e == -1 or e == start_edge:
                break
            if e in visited_edges:
                break
            visited_edges.add(e)
            path.append(e)
            u1, v1 = edges[e]
            current_u = v1 if current_u == u1 else u1
            current_c = c2 - current_c  # alternate between c1 and c2
        return path
    
    def swap_colors(path, c1, c2):
        for e in path:
            old_c = color[e]
            new_c = c2 if old_c == c1 else c1
            color[e] = new_c
            u1, v1 = edges[e]
            edge_at[u1][old_c] = -1
            edge_at[v1][old_c] = -1
            edge_at[u1][new_c] = e
            edge_at[v1][new_c] = e
    
    for e in range(m):
        u, v = edges[e]
        
        # Find available colors at u and v
        colors_u = set(range(3))
        colors_v = set(range(3))
        for c in range(3):
            if edge_at[u][c] != -1:
                colors_u.discard(c)
            if edge_at[v][c] != -1:
                colors_v.discard(c)
        
        # Find common available color
        common = colors_u & colors_v
        if common:
            c = common.pop()
            color[e] = c
            edge_at[u][c] = e
            edge_at[v][c] = e
        else:
            # Need to recolor
            c1 = colors_u.pop()
            c2 = colors_v.pop()
            
            # Find alternating path from v using colors c1 and c2
            path = find_alt_path(v, c1, c2, e)
            
            # Swap colors on the path
            swap_colors(path, c1, c2)
            
            # Now c1 should be free at v
            color[e] = c1
            edge_at[u][c1] = e
            edge_at[v][c1] = e
    
    print("YES")
    print(' '.join(str(c + 1) for c in color))

solve()
