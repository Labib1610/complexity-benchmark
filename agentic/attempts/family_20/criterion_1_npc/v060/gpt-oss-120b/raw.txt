import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    g = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        g[u].append((v, idx))
        g[v].append((u, idx))

    # ---------- check bridges ----------
    sys.setrecursionlimit(10000)
    tin = [-1]*n
    low = [0]*n
    timer = 0
    bridge = [False]*m

    def dfs(v, p_e):
        nonlocal timer
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in g[v]:
            if eid == p_e:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridge[eid] = True

    dfs(0, -1)
    if any(b for b in bridge):
        print("NO")
        return

    # ---------- Edmonds blossom for maximum matching ----------
    # implementation adapted from CP-algorithms
    from collections import deque

    N = n
    adj = [[] for _ in range(N)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    match = [-1]*N
    p = [-1]*N
    base = list(range(N))
    q = deque()
    used = [False]*N
    blossom = [False]*N

    def lca(a, b):
        used_path = [False]*N
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal used, p, base, q, blossom
        used = [False]*N
        p = [-1]*N
        for i in range(N):
            base[i] = i
        q.clear()
        q.append(root)
        used[root] = True
        while q:
            v = q.popleft()
            for to in adj[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    curbase = lca(v, to)
                    blossom = [False]*N
                    mark_path(v, curbase, to)
                    mark_path(to, curbase, v)
                    for i in range(N):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        # augmenting
                        cur = to
                        while cur != -1:
                            prev = p[cur]
                            nxt = match[prev] if prev != -1 else -1
                            match[cur] = prev
                            match[prev] = cur
                            cur = nxt
                        return True
                    else:
                        used[match[to]] = True
                        q.append(match[to])
        return False

    for v in range(N):
        if match[v] == -1:
            find_path(v)

    if any(mch == -1 for mch in match):
        print("NO")
        return

    # ---------- build cycles from remaining edges ----------
    color = [0]*m
    # edges in matching get color 3
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            color[idx] = 3

    # build adjacency of non-matching edges
    adj2 = [[] for _ in range(N)]
    for idx, (u, v) in enumerate(edges):
        if color[idx] == 0:
            adj2[u].append((v, idx))
            adj2[v].append((u, idx))

    visited_edge = [False]*m
    for start in range(N):
        for to, eid in adj2[start]:
            if visited_edge[eid]:
                continue
            # traverse the cycle
            cycle = []
            cur_v = start
            cur_e = eid
            while True:
                visited_edge[cur_e] = True
                cycle.append(cur_e)
                # move to other vertex
                nxt_v = edges[cur_e][0] ^ edges[cur_e][1] ^ cur_v
                # find next edge in cycle
                for nxt, neid in adj2[nxt_v]:
                    if not visited_edge[neid]:
                        cur_v = nxt_v
                        cur_e = neid
                        break
                else:
                    break
            # assign alternating colors 1 and 2
            for i, eidx in enumerate(cycle):
                color[eidx] = 1 if i % 2 == 0 else 2

    print("YES")
    print(' '.join(str(c) for c in color))

if __name__ == "__main__":
    solve()
