import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
    
    # Build adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    for i, (u, v) in enumerate(edges):
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Find all cycles using DFS
    visited = [False] * (n + 1)
    cycles = []
    edge_to_cycle = [None] * m
    
    for start in range(1, n + 1):
        if visited[start]:
            continue
        
        # DFS to find cycle
        stack = [(start, None, None)]  # (node, parent, edge_index)
        parent = {start: (None, None)}
        cycle_edges = []
        cycle_nodes = []
        
        while stack:
            node, par, edge_idx = stack.pop()
            
            if visited[node]:
                # Found a cycle
                # Trace back to find the cycle
                cycle_nodes = [node]
                curr = par
                curr_edge = edge_idx
                cycle_edges = [curr_edge]
                while curr != node:
                    cycle_nodes.append(curr)
                    curr, curr_edge = parent[curr]
                    cycle_edges.append(curr_edge)
                break
            
            visited[node] = True
            parent[node] = (par, edge_idx)
            
            for neighbor, e_idx in adj[node]:
                if neighbor == par:
                    continue
                if visited[neighbor]:
                    # Found cycle
                    cycle_nodes = [neighbor]
                    curr = node
                    curr_edge = e_idx
                    cycle_edges = [curr_edge]
                    while curr != neighbor:
                        cycle_nodes.append(curr)
                        curr, curr_edge = parent[curr]
                        cycle_edges.append(curr_edge)
                    break
                stack.append((neighbor, node, e_idx))
            
            if cycle_edges:
                break
        
        if cycle_edges:
            # Remove cycle edges from graph
            for e_idx in cycle_edges:
                u, v = edges[e_idx]
                adj[u] = [(ne, ei) for ne, ei in adj[u] if ei != e_idx]
                adj[v] = [(ne, ei) for ne, ei in adj[v] if ei != e_idx]
            
            cycles.append((cycle_nodes, cycle_edges))
    
    # Assign colors to edges
    color = [0] * m
    
    for cycle_nodes, cycle_edges in cycles:
        k = len(cycle_edges)
        # Alternate colors 1 and 2
        for i, e_idx in enumerate(cycle_edges):
            color[e_idx] = 1 if i % 2 == 0 else 2
    
    # Check if all edges are colored
    for c in color:
        if c == 0:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
