import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append((i, v))
        adj[v].append((i, u))
        
    # Check if all vertices have degree 3
    for v in range(1, n + 1):
        if len(adj[v]) != 3:
            print("NO")
            return

    # vertex_edge_with_col[v][c] = edge index with color c at vertex v, or -1
    vertex_col = [[-1] * 4 for _ in range(n + 1)]
    edge_col = [0] * m

    def set_edge_color(e, c):
        old_c = edge_col[e]
        u, v = edges[e]
        if old_c != 0:
            vertex_col[u][old_c] = -1
            vertex_col[v][old_c] = -1
        edge_col[e] = c
        if c != 0:
            vertex_col[u][c] = e
            vertex_col[v][c] = e

    def get_other(e, u):
        u1, v1 = edges[e]
        return v1 if u1 == u else u1

    # Kempe chain coloring algorithm
    max_steps = 150000
    steps = 0

    for initial_e in range(m):
        curr_e = initial_e
        if edge_col[curr_e] != 0:
            continue
            
        while edge_col[curr_e] == 0:
            steps += 1
            if steps > max_steps:
                print("NO")
                return
            
            u, v = edges[curr_e]
            miss_u = [c for c in (1, 2, 3) if vertex_col[u][c] == -1]
            miss_v = [c for c in (1, 2, 3) if vertex_col[v][c] == -1]
            
            # Case 1: common missing color
            common = set(miss_u) & set(miss_v)
            if common:
                c = common.pop()
                set_edge_color(curr_e, c)
                break
            
            # Case 2: try to find an (a, b) alternating path that doesn't connect u and v
            success = False
            for a in miss_u:
                for b in miss_v:
                    path_edges = []
                    curr_node = u
                    curr_color = a
                    while vertex_col[curr_node][curr_color] != -1:
                        e_next = vertex_col[curr_node][curr_color]
                        path_edges.append(e_next)
                        curr_node = get_other(e_next, curr_node)
                        curr_color = b if curr_color == a else a
                    
                    if curr_node != v:
                        # Flip alternating path
                        for pe in path_edges:
                            old_c = edge_col[pe]
                            new_c = b if old_c == a else a
                            set_edge_color(pe, new_c)
                        set_edge_color(curr_e, b)
                        success = True
                        break
                if success:
                    break
            
            if success:
                break
            
            # Case 3: All paths connect u and v, perform a random kick
            c = random.choice([1, 2, 3])
            e_kick = vertex_col[u][c]
            if e_kick == -1:
                c = miss_u[0]
                e_kick = vertex_col[v][c]
                set_edge_color(e_kick, 0)
                set_edge_color(curr_e, c)
                curr_e = e_kick
            else:
                set_edge_color(e_kick, 0)
                set_edge_color(curr_e, c)
                curr_e = e_kick

    print("YES")
    print(*(edge_col[i] for i in range(m)))

if __name__ == '__main__':
    solve()
