We can model this as a **3-edge-coloring** problem on a **3-regular graph** (each vertex has degree 3).  

By **Kőnig’s line coloring theorem**, every bipartite graph has edge chromatic number equal to its maximum degree. But here the graph is not necessarily bipartite.  

However, for **3-regular graphs**, a 3-edge-coloring exists if and only if the graph is **bridgeless** (by **Tait’s theorem**). If the graph has a bridge, it cannot be 3-edge-colored.  

So the algorithm is:
1. Check if the graph is bridgeless (no edge whose removal disconnects the graph).
2. If it has a bridge → output `NO`.
3. Otherwise, find a 3-edge-coloring using a constructive method (e.g., via **Kempe chains** or using a known algorithm for cubic graphs).

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1000000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    edges = []
    adj = [[] for _ in range(n)]
    for i in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if graph is bridgeless
    tin = [-1] * n
    low = [-1] * n
    timer = [0]
    has_bridge = [False]
    
    def dfs(v, p_edge):
        tin[v] = low[v] = timer[0]
        timer[0] += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge[0] = True
    
    dfs(0, -1)
    
    if has_bridge[0]:
        print("NO")
        return
    
    # Now find a 3-edge-coloring
    # We'll use a known constructive method:
    # Since graph is cubic and bridgeless, it has a perfect matching.
    # Remove a perfect matching -> remaining graph is 2-regular (cycles).
    # Color the cycles alternately with 2 colors, and the matching with the third color.
    
    # Find a perfect matching using simple greedy (since graph is cubic and bridgeless, it exists)
    match = [-1] * n
    used_edge = [False] * m
    for u in range(n):
        if match[u] == -1:
            for v, eid in adj[u]:
                if match[v] == -1:
                    match[u] = v
                    match[v] = u
                    used_edge[eid] = True
                    break
    
    # Now color the remaining edges (cycles) with colors 1 and 2
    color = [0] * m
    for eid in range(m):
        if used_edge[eid]:
            color[eid] = 3
        else:
            color[eid] = 0  # unassigned
    
    # Traverse cycles
    visited = [False] * n
    for start in range(n):
        if not visited[start]:
            # find cycle
            cycle_edges = []
            cur = start
            prev = -1
            while True:
                visited[cur] = True
                # find next edge not used in matching
                nxt = -1
                nxt_eid = -1
                for v, eid in adj[cur]:
                    if eid != prev and not used_edge[eid]:
                        nxt = v
                        nxt_eid = eid
                        break
                if nxt == -1:
                    break
                cycle_edges.append(nxt_eid)
                prev = nxt_eid
                cur = nxt
                if cur == start:
                    break
            # color cycle alternately
            for i, eid in enumerate(cycle_edges):
                color[eid] = 1 if i % 2 == 0 else 2
    
    # Verify
    for u in range(n):
        colors = set()
        for v, eid in adj[u]:
            colors.add(color[eid])
        if len(colors) != 3:
            print("NO")
            return
    
    print("YES")
    print(" ".join(str(c) for c in color))

if __name__ == "__main__":
    solve()
```
