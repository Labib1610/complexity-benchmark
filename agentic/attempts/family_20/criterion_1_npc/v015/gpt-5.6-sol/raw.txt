```python
import sys
from collections import deque
from itertools import permutations

sys.setrecursionlimit(10000)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    if len(data) != 2 + 2 * m:
        print("NO")
        return

    edges = []
    adj = [[] for _ in range(n)]

    p = 2
    valid = True
    for eid in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if not (0 <= u < n and 0 <= v < n) or u == v:
            valid = False
        edges.append((u, v))
        if 0 <= u < n and 0 <= v < n:
            adj[u].append(eid)
            adj[v].append(eid)

    if not valid or any(len(adj[v]) != 3 for v in range(n)):
        print("NO")
        return

    # A bridge makes a cubic graph impossible to 3-edge-color.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False

    def bridge_dfs(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1

        for eid in adj[v]:
            if eid == parent_edge:
                continue
            a, b = edges[eid]
            to = b if a == v else a
            if tin[to] == -1:
                bridge_dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for start in range(n):
        if tin[start] == -1:
            bridge_dfs(start, -1)

    if has_bridge:
        print("NO")
        return

    # Find connected components and test bipartiteness.
    side = [-1] * n
    components = []
    component_is_bipartite = []

    for start in range(n):
        if side[start] != -1:
            continue

        side[start] = 0
        q = deque([start])
        vertices = []
        bipartite = True

        while q:
            v = q.popleft()
            vertices.append(v)
            for eid in adj[v]:
                a, b = edges[eid]
                to = b if a == v else a
                if side[to] == -1:
                    side[to] = side[v] ^ 1
                    q.append(to)
                elif side[to] == side[v]:
                    bipartite = False

        components.append(vertices)
        component_is_bipartite.append(bipartite)

    domains = [7] * m
    non_bipartite_edges = []
    non_bipartite_roots = []

    # Cubic bipartite components can be colored in polynomial time:
    # one perfect matching plus alternating colors on the remaining cycles.
    for vertices, bipartite in zip(components, component_is_bipartite):
        if not bipartite:
            non_bipartite_roots.append(vertices[0])
            for v in vertices:
                for eid in adj[v]:
                    if edges[eid][0] == v:
                        non_bipartite_edges.append(eid)
            continue

        left = [v for v in vertices if side[v] == 0]
        pair_left = [-1] * n
        pair_right = [-1] * n
        matched_edge_left = [-1] * n
        dist = [-1] * n

        def bfs_matching():
            q = deque()
            found = False
            for u in left:
                if pair_left[u] == -1:
                    dist[u] = 0
                    q.append(u)
                else:
                    dist[u] = -1

            while q:
                u = q.popleft()
                for eid in adj[u]:
                    a, b = edges[eid]
                    v = b if a == u else a
                    mate = pair_right[v]
                    if mate == -1:
                        found = True
                    elif dist[mate] == -1:
                        dist[mate] = dist[u] + 1
                        q.append(mate)
            return found

        def dfs_matching(u):
            for eid in adj[u]:
                a, b = edges[eid]
                v = b if a == u else a
                mate = pair_right[v]
                if mate == -1 or (
                    dist[mate] == dist[u] + 1 and dfs_matching(mate)
                ):
                    pair_left[u] = v
                    pair_right[v] = u
                    matched_edge_left[u] = eid
                    return True
            dist[u] = -1
            return False

        matching_size = 0
        while bfs_matching():
            for u in left:
                if pair_left[u] == -1 and dfs_matching(u):
                    matching_size += 1

        if matching_size != len(left):
            print("NO")
            return

        matching_edges = set()
        for u in left:
            eid = matched_edge_left[u]
            if eid < 0:
                print("NO")
                return
            matching_edges.add(eid)
            domains[eid] = 1

        component_edges = []
        for v in vertices:
            for eid in adj[v]:
                if edges[eid][0] == v:
                    component_edges.append(eid)

        for start_edge in component_edges:
            if start_edge in matching_edges or domains[start_edge] != 7:
                continue

            domains[start_edge] = 2
            q = deque([start_edge])

            while q:
                eid = q.popleft()
                color = domains[eid]
                u, v = edges[eid]

                for x in (u, v):
                    other = -1
                    for e2 in adj[x]:
                        if e2 != eid and e2 not in matching_edges:
                            other = e2
                            break

                    required = 4 if color == 2 else 2
                    if domains[other] == 7:
                        domains[other] = required
                        q.append(other)
                    elif domains[other] != required:
                        print("NO")
                        return

    if not non_bipartite_edges:
        answer = [d.bit_length() for d in domains]
        print("YES")
        print(*answer)
        return

    local_permutations = list(permutations((1, 2, 4)))
    trail = []

    def propagate(initial_vertices):
        q = deque(initial_vertices)
        in_queue = [False] * n
        for v in initial_vertices:
            in_queue[v] = True

        while q:
            v = q.popleft()
            in_queue[v] = False
            e0, e1, e2 = adj[v]
            incident = (e0, e1, e2)

            allowed0 = allowed1 = allowed2 = 0
            found = False

            d0, d1, d2 = domains[e0], domains[e1], domains[e2]
            for c0, c1, c2 in local_permutations:
                if (d0 & c0) and (d1 & c1) and (d2 & c2):
                    found = True
                    allowed0 |= c0
                    allowed1 |= c1
                    allowed2 |= c2

            if not found:
                return False

            allowed = (allowed0, allowed1, allowed2)
            for eid, mask in zip(incident, allowed):
                new_domain = domains[eid] & mask
                if new_domain == 0:
                    return False
                if new_domain != domains[eid]:
                    trail.append((eid, domains[eid]))
                    domains[eid] = new_domain
                    a, b = edges[eid]
                    if not in_queue[a]:
                        in_queue[a] = True
                        q.append(a)
                    if not in_queue[b]:
                        in_queue[b] = True
                        q.append(b)

        return True

    initial_vertices = []
    for root in non_bipartite_roots:
        for eid, color in zip(adj[root], (1, 2, 4)):
            domains[eid] = color
        initial_vertices.append(root)

    if not propagate(initial_vertices):
        print("NO")
        return

    trail.clear()

    def rollback(mark):
        while len(trail) > mark:
            eid, old_domain = trail.pop()
            domains[eid] = old_domain

    def search():
        chosen = -1
        best_size = 4

        for eid in non_bipartite_edges:
            d = domains[eid]
            size = d.bit_count()
            if size > 1 and size < best_size:
                best_size = size
                chosen = eid
                if size == 2:
                    break

        if chosen == -1:
            return True

        available = domains[chosen]
        u, v = edges[chosen]

        while available:
            color = available & -available
            available -= color
            mark = len(trail)

            trail.append((chosen, domains[chosen]))
            domains[chosen] = color

            if propagate((u, v)) and search():
                return True

            rollback(mark)

        return False

    if not search():
        print("NO")
        return

    answer = [d.bit_length() for d in domains]
    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
```
