import sys
sys.setrecursionlimit(10000000)

def solve() -> None:
    import sys
    from collections import deque

    n, m = map(int, sys.stdin.readline().split())
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # ---------- Blossom algorithm for maximum matching ----------
    # Implementation based on Edmonds' blossom algorithm (Gabow's version)
    # Returns a list match[v] = matched vertex or -1
    def max_matching():
        N = n
        g = [[] for _ in range(N)]
        for u, v in edges:
            g[u].append(v)
            g[v].append(u)

        match = [-1] * N
        p = [-1] * N
        base = list(range(N))
        q = deque()
        used = [False] * N
        blossom = [False] * N

        def lca(a, b):
            used2 = [False] * N
            while True:
                a = base[a]
                used2[a] = True
                if match[a] == -1:
                    break
                a = p[match[a]]
            while True:
                b = base[b]
                if used2[b]:
                    return b
                b = p[match[b]]

        def mark_path(v, b, x):
            while base[v] != b:
                blossom[base[v]] = blossom[base[match[v]]] = True
                p[v] = x
                x = match[v]
                v = p[match[v]]

        def find_path(root):
            nonlocal used, p, base, q, blossom
            used = [False] * N
            p = [-1] * N
            for i in range(N):
                base[i] = i
            q.clear()
            q.append(root)
            used[root] = True
            while q:
                v = q.popleft()
                for to in g[v]:
                    if base[v] == base[to] or match[v] == to:
                        continue
                    if to == root or (match[to] != -1 and p[match[to]] != -1):
                        curbase = lca(v, to)
                        blossom = [False] * N
                        mark_path(v, curbase, to)
                        mark_path(to, curbase, v)
                        for i in range(N):
                            if blossom[base[i]]:
                                base[i] = curbase
                                if not used[i]:
                                    used[i] = True
                                    q.append(i)
                    elif p[to] == -1:
                        p[to] = v
                        if match[to] == -1:
                            # augmenting path found
                            v = to
                            while v != -1:
                                pv = p[v]
                                ppv = match[pv] if pv != -1 else -1
                                match[v] = pv
                                match[pv] = v
                                v = ppv
                            return True
                        else:
                            used[match[to]] = True
                            q.append(match[to])
            return False

        for v in range(N):
            if match[v] == -1:
                find_path(v)
        return match

    match = max_matching()
    # check perfect matching
    if any(x == -1 for x in match):
        print("NO")
        return

    # Build the 2-factor (remaining edges)
    used_edge = [False] * m
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            used_edge[idx] = True

    # adjacency of remaining graph
    rem_adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        if not used_edge[idx]:
            rem_adj[u].append((v, idx))
            rem_adj[v].append((u, idx))

    # find cycles in remaining 2-regular graph
    color = [0] * m
    for idx in range(m):
        if used_edge[idx]:
            color[idx] = 1  # first matching gets color 1

    visited = [False] * n
    for start in range(n):
        if visited[start]:
            continue
        # traverse cycle
        cur = start
        prev = -1
        cyc_edges = []
        while True:
            visited[cur] = True
            for nxt, eid in rem_adj[cur]:
                if eid not in cyc_edges:
                    cyc_edges.append(eid)
                    break
            # move to next vertex
            nxt = edges[cyc_edges[-1]][0] ^ edges[cyc_edges[-1]][1] ^ cur
            if nxt == start:
                break
            cur = nxt

        # assign alternating colors 2 and 3
        for i, eid in enumerate(cyc_edges):
            color[eid] = 2 if i % 2 == 0 else 3

    # verify validity
    ok = True
    for u in range(n):
        seen = set()
        for v, eid in adj[u]:
            c = color[eid]
            if c in seen:
                ok = False
                break
            seen.add(c)
        if not ok:
            break

    if not ok:
        print("NO")
    else:
        print("YES")
        print(' '.join(str(color[i]) for i in range(m)))

if __name__ == "__main__":
    solve()
