import sys
from sys import setrecursionlimit

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))

    color = [0] * m
    visited = [False] * (n + 1)
    parent_edge = [-1] * (n + 1)
    parent_node = [0] * (n + 1)
    depth = [0] * (n + 1)
    stack = []

    for s in range(1, n + 1):
        if not visited[s]:
            visited[s] = True
            stack.append(s)
            while stack:
                u = stack.pop()
                for v, eid in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        parent_edge[v] = eid
                        parent_node[v] = u
                        depth[v] = depth[u] + 1
                        stack.append(v)

    for u, v, eid in edges:
        if color[eid] == 0:
            if depth[u] < depth[v]:
                u, v = v, u
            cur = u
            while depth[cur] > depth[v]:
                e = parent_edge[cur]
                if color[e] == 0:
                    color[e] = 1
                cur = parent_node[cur]
            while cur != v:
                e1 = parent_edge[cur]
                e2 = parent_edge[v]
                if color[e1] == 0:
                    color[e1] = 1
                if color[e2] == 0:
                    color[e2] = 1
                cur = parent_node[cur]
                v = parent_node[v]

    for u, v, eid in edges:
        if color[eid] == 0:
            color[eid] = 2

    for u in range(1, n + 1):
        cnt = [0, 0, 0, 0]
        for v, eid in adj[u]:
            cnt[color[eid]] += 1
        if cnt[1] > 1 or cnt[2] > 1:
            print("NO")
            return

    for u in range(1, n + 1):
        deg = len(adj[u])
        if deg == 3:
            c1 = color[adj[u][0][1]]
            c2 = color[adj[u][1][1]]
            c3 = color[adj[u][2][1]]
            if c1 == c2 or c2 == c3 or c1 == c3:
                print("NO")
                return

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
