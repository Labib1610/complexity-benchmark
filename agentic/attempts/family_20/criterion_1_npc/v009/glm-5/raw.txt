import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    color = [0] * m
    setrecursionlimit(1 << 25)
    
    def dfs(u, pu, c):
        for v, eid in adj[u]:
            if v == pu:
                continue
            if color[eid] == 0:
                color[eid] = c
                dfs(v, u, c ^ 3)
                c ^= 3
    
    for u in range(1, n + 1):
        for v, eid in adj[u]:
            if color[eid] == 0:
                color[eid] = 1
                dfs(v, u, 2)
    
    for u in range(1, n + 1):
        nbrs = adj[u]
        if len(nbrs) == 3:
            c1 = color[nbrs[0][1]]
            c2 = color[nbrs[1][1]]
            c3 = color[nbrs[2][1]]
            if c1 == c2 or c1 == c3 or c2 == c3:
                print("NO")
                return
        else:
            print("NO")
            return
    
    print("YES")
    print(' '.join(str(c) for c in color))

solve()
