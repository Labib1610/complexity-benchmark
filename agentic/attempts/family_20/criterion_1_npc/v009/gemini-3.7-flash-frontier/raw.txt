import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))

    # Fast bridge check: a cubic graph with a bridge is not 3-edge-colorable
    tin = [0] * (n + 1)
    low = [0] * (n + 1)
    visited = [False] * (n + 1)
    timer = 0
    has_bridge = False

    def dfs_bridge(v, p_edge):
        nonlocal timer, has_bridge
        visited[v] = True
        timer += 1
        tin[v] = low[v] = timer
        for to, edge_idx in adj[v]:
            if edge_idx == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs_bridge(to, edge_idx)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True

    for i in range(1, n + 1):
        if not visited[i]:
            dfs_bridge(i, -1)

    if has_bridge:
        print("NO")
        return

    # CDCL SAT Solver implementation
    # Variables: for edge i (0 to m-1) and color c (0, 1, 2)
    # var_id = i * 3 + c + 1  (1-indexed)
    num_vars = m * 3
    clauses = []

    def get_var(edge_i, color_c):
        return edge_i * 3 + color_c + 1

    # 1. Each edge has at least one color and at most one color
    for i in range(m):
        v0 = get_var(i, 0)
        v1 = get_var(i, 1)
        v2 = get_var(i, 2)
        clauses.append([v0, v1, v2])
        clauses.append([-v0, -v1])
        clauses.append([-v0, -v2])
        clauses.append([-v1, -v2])

    # 2. At each vertex, the 3 incident edges must have different colors
    for u in range(1, n + 1):
        inc = [edge_idx for _, edge_idx in adj[u]]
        if len(inc) == 3:
            e0, e1, e2 = inc[0], inc[1], inc[2]
            for c in range(3):
                vc0 = get_var(e0, c)
                vc1 = get_var(e1, c)
                vc2 = get_var(e2, c)
                clauses.append([-vc0, -vc1])
                clauses.append([-vc0, -vc2])
                clauses.append([-vc1, -vc2])
                clauses.append([vc0, vc1, vc2])

    # CDCL Solver
    class CDCL:
        def __init__(self, num_vars, clauses):
            self.num_vars = num_vars
            self.clauses = [list(c) for c in clauses]
            self.watches = [[] for _ in range(2 * num_vars + 1)]
            self.assignment = [0] * (num_vars + 1)  # 1: True, -1: False, 0: Unassigned
            self.decision_level = [0] * (num_vars + 1)
            self.reason = [None] * (num_vars + 1)
            self.trail = []
            self.trail_lim = []
            self.activity = [0.0] * (num_vars + 1)
            self.var_inc = 1.0
            self.var_decay = 0.95

            for ci, clause in enumerate(self.clauses):
                if len(clause) == 1:
                    continue
                self.watches[self._lit_to_index(clause[0])].append(ci)
                self.watches[self._lit_to_index(clause[1])].append(ci)

        def _lit_to_index(self, lit):
            return lit + self.num_vars if lit < 0 else lit

        def _var(self, lit):
            return abs(lit)

        def _val(self, lit):
            v = self.assignment[abs(lit)]
            return v if lit > 0 else -v

        def assign(self, lit, reason=None):
            var = abs(lit)
            val = 1 if lit > 0 else -1
            self.assignment[var] = val
            self.decision_level[var] = len(self.trail_lim)
            self.reason[var] = reason
            self.trail.append(lit)

        def propagate(self):
            head = len(self.trail) - 1
            # We will process from the current point in trail
            if not hasattr(self, 'prop_head'):
                self.prop_head = 0
            
            while self.prop_head < len(self.trail):
                p = self.trail[self.prop_head]
                self.prop_head += 1
                false_lit = -p
                false_idx = self._lit_to_index(false_lit)
                
                watch_list = self.watches[false_idx]
                self.watches[false_idx] = []
                
                i = 0
                while i < len(watch_list):
                    ci = watch_list[i]
                    c = self.clauses[ci]
                    
                    if c[0] == false_lit:
                        c[0], c[1] = c[1], c[0]
                    
                    if self._val(c[0]) == 1:
                        self.watches[false_idx].append(ci)
                        i += 1
                        continue
                    
                    # Look for a new watch
                    found_new_watch = False
                    for k in range(2, len(c)):
                        if self._val(c[k]) != -1:
                            c[1], c[k] = c[k], c[1]
                            self.watches[self._lit_to_index(c[1])].append(ci)
                            found_new_watch = True
                            break
                    
                    if not found_new_watch:
                        self.watches[false_idx].append(ci)
                        i += 1
                        if self._val(c[0]) == -1:
                            # Conflict
                            # Restore remaining watches
                            while i < len(watch_list):
                                self.watches[false_idx].append(watch_list[i])
                                i += 1
                            return ci
                        else:
                            # Unit propagation
                            self.assign(c[0], ci)
            return None

        def analyze(self, confl_clause_idx):
            confl = self.clauses[confl_clause_idx]
            learnt = []
            seen = [False] * (self.num_vars + 1)
            path_cnt = 0
            p = 0
            bt_level = 0
            cur_level = len(self.trail_lim)

            idx = len(self.trail) - 1
            c = confl
            while True:
                for lit in c:
                    var = abs(lit)
                    if not seen[var] and self.decision_level[var] > 0:
                        seen[var] = True
                        self.bump_var_activity(var)
                        if self.decision_level[var] >= cur_level:
                            path_cnt += 1
                        else:
                            learnt.append(lit)
                            if self.decision_level[var] > bt_level:
                                bt_level = self.decision_level[var]

                while True:
                    p = self.trail[idx]
                    idx -= 1
                    if seen[abs(p)]:
                        break
                seen[abs(p)] = False
                path_cnt -= 1
                if path_cnt <= 0:
                    break
                c = self.clauses[self.reason[abs(p)]]

            learnt.insert(0, -p)
            return learnt, bt_level

        def bump_var_activity(self, var):
            self.activity[var] += self.var_inc
            if self.activity[var] > 1e100:
                for i in range(1, self.num_vars + 1):
                    self.activity[i] *= 1e-100
                self.var_inc *= 1e-100

        def backtrack(self, level):
            if len(self.trail_lim) <= level:
                return
            target_trail_len = self.trail_lim[level]
            while len(self.trail) > target_trail_len:
                lit = self.trail.pop()
                var = abs(lit)
                self.assignment[var] = 0
                self.reason[var] = None
            self.trail_lim = self.trail_lim[:level]
            self.prop_head = min(self.prop_head, len(self.trail))

        def pick_branch_var(self):
            best_var = 0
            best_act = -1
            for var in range(1, self.num_vars + 1):
                if self.assignment[var] == 0:
                    if self.activity[var] > best_act:
                        best_act = self.activity[var]
                        best_var = var
            return best_var

        def solve(self):
            # Unit propagation on initial clauses
            for ci, c in enumerate(self.clauses):
                if len(c) == 1:
                    if self._val(c[0]) == -1:
                        return False
                    elif self._val(c[0]) == 0:
                        self.assign(c[0], ci)
            if self.propagate() is not None:
                return False

            while True:
                confl = self.propagate()
                if confl is not None:
                    if len(self.trail_lim) == 0:
                        return False
                    learnt, bt_level = self.analyze(confl)
                    self.backtrack(bt_level)
                    self.var_inc /= self.var_decay
                    
                    ci = len(self.clauses)
                    self.clauses.append(learnt)
                    if len(learnt) > 1:
                        self.watches[self._lit_to_index(learnt[0])].append(ci)
                        self.watches[self._lit_to_index(learnt[1])].append(ci)
                    self.assign(learnt[0], ci)
                else:
                    var = self.pick_branch_var()
                    if var == 0:
                        return True
                    self.trail_lim.append(len(self.trail))
                    self.assign(var, None)

    solver = CDCL(num_vars, clauses)
    if solver.solve():
        ans = []
        for i in range(m):
            for c in range(3):
                if solver.assignment[get_var(i, c)] == 1:
                    ans.append(c + 1)
                    break
        print("YES")
        print(*(ans))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
