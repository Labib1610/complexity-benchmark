import sys
sys.setrecursionlimit(10000000)

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = [(int(next(it))-1, int(next(it))-1) for _ in range(m)]

    # Build adjacency list with edge indices
    adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # Check degree 3
    for a in adj:
        if len(a) != 3:
            print("NO")
            return

    # Find a perfect matching using blossom algorithm (networkx style)
    # We'll implement Edmonds' blossom algorithm for general graph matching.
    # Since n ≤ 2000, O(n^3) is acceptable.

    # Implementation based on https://github.com/jaehyunp/stanfordacm/blob/master/code/EdmondsMatching.cc
    from collections import deque

    N = n
    g = [[] for _ in range(N)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    match = [-1]*N
    p = [-1]*N
    base = list(range(N))
    q = deque()
    in_blossom = [False]*N
    in_queue = [False]*N
    def lca(a, b):
        used = [False]*N
        while True:
            a = base[a]
            used[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom = base[v]
            in_blossom[blossom] = True
            in_blossom[base[match[blossom]]] = True
            p[v] = children
            children = match[blossom]
            v = p[match[blossom]]

    def blossom_contraction(v, u):
        b = lca(v, u)
        in_blossom[:] = [False]*N
        mark_path(v, b, u)
        mark_path(u, b, v)
        for i in range(N):
            if in_blossom[base[i]]:
                base[i] = b
                if not in_queue[i]:
                    q.append(i)
                    in_queue[i] = True

    def bfs(root):
        nonlocal p, base, in_queue, q
        p = [-1]*N
        in_queue = [False]*N
        base = list(range(N))
        q.clear()
        q.append(root)
        in_queue[root] = True
        while q:
            v = q.popleft()
            for u in g[v]:
                if base[v] == base[u] or match[v] == u:
                    continue
                if u == root or (match[u] != -1 and p[match[u]] != -1):
                    blossom_contraction(v, u)
                elif p[u] == -1:
                    p[u] = v
                    if match[u] == -1:
                        # augmenting path found
                        return u
                    if not in_queue[match[u]]:
                        q.append(match[u])
                        in_queue[match[u]] = True
        return -1

    def augment(path_end):
        v = path_end
        while v != -1:
            pv = p[v]
            w = match[pv] if pv != -1 else -1
            match[v] = pv
            match[pv] = v
            v = w

    for v in range(N):
        if match[v] == -1:
            path_end = bfs(v)
            if path_end != -1:
                augment(path_end)

    # Check if perfect matching found
    if any(mch == -1 for mch in match):
        print("NO")
        return

    # Build remaining graph after removing matching edges
    used_edge = [False]*m
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            used_edge[idx] = True

    # Build adjacency of remaining edges with their indices
    rem_adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        if not used_edge[idx]:
            rem_adj[u].append((v, idx))
            rem_adj[v].append((u, idx))

    # The remaining graph is 2-regular; find cycles and color them alternately 1 and 2
    color = [0]*m
    for idx in range(m):
        if used_edge[idx]:
            color[idx] = 3  # matching edges get color 3

    visited = [False]*n
    for start in range(n):
        if visited[start]:
            continue
        # traverse cycle
        cur = start
        prev = -1
        cycle_edges = []
        while True:
            visited[cur] = True
            # find next edge not coming from prev
            nxt = None
            for (to, eid) in rem_adj[cur]:
                if to != prev:
                    nxt = (to, eid)
                    break
            if nxt is None:
                break
            to, eid = nxt
            cycle_edges.append(eid)
            prev, cur = cur, to
            if cur == start:
                break
        # color cycle edges alternately 1 and 2
        for i, eid in enumerate(cycle_edges):
            color[eid] = 1 if i%2==0 else 2

    # Verify correctness
    ok = True
    for u in range(n):
        cols = set()
        for v, eid in adj[u]:
            cols.add(color[eid])
        if len(cols)!=3:
            ok=False
            break
    if not ok:
        print("NO")
        return

    print("YES")
    print(' '.join(str(c) for c in color))

if __name__ == "__main__":
    main()
