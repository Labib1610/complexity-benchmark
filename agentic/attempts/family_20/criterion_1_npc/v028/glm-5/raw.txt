import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n)]
    for i in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if graph is d-regular with d = 2m/n
    if n == 0:
        print("YES")
        return
    if m * 2 % n != 0:
        print("NO")
        return
    d = m * 2 // n
    for i in range(n):
        if len(adj[i]) != d:
            print("NO")
            return
    
    # Find an Eulerian circuit
    edge_used = [False] * m
    adj_ptr = [0] * n
    # Reorder adjacency for efficient iteration
    adj_lists = [[] for _ in range(n)]
    for u in range(n):
        adj_lists[u] = adj[u][:]
    
    path = []
    stack = [0]
    while stack:
        u = stack[-1]
        while adj_ptr[u] < len(adj_lists[u]):
            v, eid = adj_lists[u][adj_ptr[u]]
            adj_ptr[u] += 1
            if not edge_used[eid]:
                edge_used[eid] = True
                stack.append(v)
                u = v
                break
        else:
            path.append(stack.pop())
    
    # Assign colors based on Eulerian path
    ans = [0] * m
    edge_to_idx = {}
    for i, (u, v) in enumerate(edges):
        if u < v:
            edge_to_idx[(u, v)] = i
        else:
            edge_to_idx[(v, u)] = i
    
    color = 1
    for k in range(len(path) - 1):
        u = path[k]
        v = path[k+1]
        if u < v:
            idx = edge_to_idx[(u, v)]
        else:
            idx = edge_to_idx[(v, u)]
        ans[idx] = color
        color = color % d + 1
    
    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
