```python
import sys
from collections import deque

sys.setrecursionlimit(20000)

def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    if len(data) != 2 + 2 * m:
        print("NO")
        return

    edges = []
    adj = [[] for _ in range(n)]
    p = 2

    for eid in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if not (0 <= u < n and 0 <= v < n) or u == v:
            print("NO")
            return
        edges.append((u, v))
        adj[u].append(eid)
        adj[v].append(eid)

    if m * 2 != 3 * n or any(len(adj[v]) != 3 for v in range(n)):
        print("NO")
        return

    # A cubic graph with a bridge cannot have a proper 3-edge-coloring.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False

    def bridge_dfs(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1

        for eid in adj[v]:
            if eid == parent_edge:
                continue
            a, b = edges[eid]
            to = b if a == v else a

            if tin[to] == -1:
                bridge_dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for v in range(n):
        if tin[v] == -1:
            bridge_dfs(v, -1)

    if has_bridge:
        print("NO")
        return

    # Find connected components and test bipartiteness.
    side = [-1] * n
    comp_id = [-1] * n
    components = []
    component_bipartite = []

    for start in range(n):
        if comp_id[start] != -1:
            continue

        cid = len(components)
        vertices = []
        q = deque([start])
        comp_id[start] = cid
        side[start] = 0
        bipartite = True

        while q:
            v = q.popleft()
            vertices.append(v)
            for eid in adj[v]:
                a, b = edges[eid]
                to = b if a == v else a
                if comp_id[to] == -1:
                    comp_id[to] = cid
                    side[to] = side[v] ^ 1
                    q.append(to)
                elif side[to] == side[v]:
                    bipartite = False

        components.append(vertices)
        component_bipartite.append(bipartite)

    component_edges = [[] for _ in components]
    for eid, (u, _) in enumerate(edges):
        component_edges[comp_id[u]].append(eid)

    answer = [0] * m

    # Bipartite cubic components can be colored in polynomial time:
    # one perfect matching gets color 1, and the remaining even cycles alternate.
    for cid, vertices in enumerate(components):
        if not component_bipartite[cid]:
            continue

        left = [v for v in vertices if side[v] == 0]
        match_right = [-1] * n
        match_left_edge = [-1] * n
        seen = [0] * n
        token = 0

        def augment(u):
            nonlocal token
            for eid in adj[u]:
                a, b = edges[eid]
                v = b if a == u else a
                if seen[v] == token:
                    continue
                seen[v] = token

                old_u = match_right[v]
                if old_u == -1 or augment(old_u):
                    match_right[v] = u
                    match_left_edge[u] = eid
                    return True
            return False

        for u in left:
            token += 1
            if not augment(u):
                print("NO")
                return

        for u in left:
            answer[match_left_edge[u]] = 1

        for start_edge in component_edges[cid]:
            if answer[start_edge] != 0:
                continue

            current_edge = start_edge
            current_vertex = edges[start_edge][0]
            color = 2

            while True:
                answer[current_edge] = color
                a, b = edges[current_edge]
                next_vertex = b if a == current_vertex else a

                next_edge = -1
                for eid in adj[next_vertex]:
                    if answer[eid] == 0:
                        next_edge = eid
                        break

                if next_edge == -1:
                    break

                current_vertex = next_vertex
                current_edge = next_edge
                color = 5 - color  # 2 <-> 3

    # Exact constraint solver for non-bipartite components.
    # Each domain is a three-bit mask for colors 1, 2, 3.
    domains = [7] * m
    non_bip_edges = []

    for cid, vertices in enumerate(components):
        if component_bipartite[cid]:
            for eid in component_edges[cid]:
                domains[eid] = 1 << (answer[eid] - 1)
        else:
            non_bip_edges.extend(component_edges[cid])
            root = vertices[0]
            domains[adj[root][0]] = 1
            domains[adj[root][1]] = 2
            domains[adj[root][2]] = 4

    permutations = (
        (1, 2, 4), (1, 4, 2),
        (2, 1, 4), (2, 4, 1),
        (4, 1, 2), (4, 2, 1)
    )

    trail = []

    def propagate(initial_vertices):
        q = deque(initial_vertices)

        while q:
            v = q.popleft()
            e0, e1, e2 = adj[v]
            d0, d1, d2 = domains[e0], domains[e1], domains[e2]

            s0 = s1 = s2 = 0
            valid = False

            for c0, c1, c2 in permutations:
                if (d0 & c0) and (d1 & c1) and (d2 & c2):
                    valid = True
                    s0 |= c0
                    s1 |= c1
                    s2 |= c2

            if not valid:
                return False

            for eid, support in ((e0, s0), (e1, s1), (e2, s2)):
                new_domain = domains[eid] & support
                if new_domain == 0:
                    return False
                if new_domain != domains[eid]:
                    trail.append((eid, domains[eid]))
                    domains[eid] = new_domain
                    a, b = edges[eid]
                    q.append(a)
                    q.append(b)

        return True

    if not propagate(range(n)):
        print("NO")
        return

    def search():
        chosen = -1
        best_size = 4

        for eid in non_bip_edges:
            size = domains[eid].bit_count()
            if 1 < size < best_size:
                best_size = size
                chosen = eid
                if size == 2:
                    break

        if chosen == -1:
            return True

        original = domains[chosen]
        a, b = edges[chosen]
        mask = original

        while mask:
            color_bit = mask & -mask
            mask -= color_bit
            mark = len(trail)

            trail.append((chosen, domains[chosen]))
            domains[chosen] = color_bit

            if propagate((a, b)) and search():
                return True

            while len(trail) > mark:
                eid, old_domain = trail.pop()
                domains[eid] = old_domain

        return False

    if not search():
        print("NO")
        return

    for eid in non_bip_edges:
        d = domains[eid]
        if d & (d - 1):
            print("NO")
            return
        answer[eid] = d.bit_length()

    print("YES")
    print(*answer)

if __name__ == "__main__":
    solve()
```
