import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    # Store edges and their indices
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Color assignment for each edge (0 means uncolored)
    color = [0] * m
    
    # For each node, we need to assign 3 different colors to its 3 edges
    # This is essentially checking if the graph is 3-edge-colorable
    # Since each vertex has degree 3 and the graph is cubic, this is equivalent
    # to checking if it's a line graph of a cubic graph, which is always 3-edge-colorable
    
    # We'll use a BFS-like approach to color the edges
    # Start from an arbitrary vertex and color its edges with 1, 2, 3
    # Then propagate this constraint to neighboring vertices
    
    visited_nodes = [False] * (n + 1)
    queue = deque()
    
    # Find a starting node
    start_node = 1
    queue.append(start_node)
    visited_nodes[start_node] = True
    
    # Keep track of which colors are used at each node
    node_colors = defaultdict(list)
    
    while queue:
        node = queue.popleft()
        # Get all edges of this node
        node_edges = adj[node]
        
        # If this is the first node we're processing, assign colors 1, 2, 3
        if len(node_colors[node]) == 0:
            for i, (neighbor, edge_idx) in enumerate(node_edges):
                color[edge_idx] = i + 1
                node_colors[node].append(i + 1)
                node_colors[neighbor].append(i + 1)
        
        # For each edge from this node
        for neighbor, edge_idx in node_edges:
            if not visited_nodes[neighbor]:
                queue.append(neighbor)
                visited_nodes[neighbor] = True
                
                # Color the remaining edges of the neighbor
                if len(node_colors[neighbor]) < 3:
                    used_colors = set(node_colors[neighbor])
                    available_colors = [c for c in [1, 2, 3] if c not in used_colors]
                    
                    # Find all uncolored edges of the neighbor
                    for n_neighbor, n_edge_idx in adj[neighbor]:
                        if color[n_edge_idx] == 0 and n_edge_idx != edge_idx:
                            if available_colors:
                                color[n_edge_idx] = available_colors.pop()
                                node_colors[neighbor].append(color[n_edge_idx])
                                node_colors[n_neighbor].append(color[n_edge_idx])
    
    # Verify the solution
    valid = True
    for node in range(1, n + 1):
        colors_at_node = []
        for neighbor, edge_idx in adj[node]:
            if color[edge_idx] != 0:
                colors_at_node.append(color[edge_idx])
        if len(set(colors_at_node)) != 3 or len(colors_at_node) != 3:
            valid = False
            break
    
    if valid:
        print("YES")
        print(' '.join(map(str, color)))
    else:
        # Try a different approach - this is a known problem that can be solved
        # by finding a perfect matching and using it to guide the coloring
        
        # Reset and try a different method
        color = [0] * m
        
        # Build adjacency list for edges (line graph)
        edge_adj = defaultdict(list)
        edge_to_idx = {}
        
        for i, (u, v) in enumerate(edges):
            edge_to_idx[(u, v)] = i
            edge_to_idx[(v, u)] = i
            
        # For each vertex, connect its three edges in the line graph
        for node in range(1, n + 1):
            node_edges = [edge_idx for _, edge_idx in adj[node]]
            if len(node_edges) == 3:
                # Create a triangle in the line graph
                edge_adj[node_edges[0]].append(node_edges[1])
                edge_adj[node_edges[0]].append(node_edges[2])
                edge_adj[node_edges[1]].append(node_edges[0])
                edge_adj[node_edges[1]].append(node_edges[2])
                edge_adj[node_edges[2]].append(node_edges[0])
                edge_adj[node_edges[2]].append(node_edges[1])
        
        # Try to 3-color the line graph
        # This is equivalent to 3-edge-coloring the original graph
        def dfs(edge_idx, colors):
            if edge_idx == m:
                return True
                
            used_colors = set()
            for neighbor in edge_adj[edge_idx]:
                if colors[neighbor] != 0:
                    used_colors.add(colors[neighbor])
            
            for c in [1, 2, 3]:
                if c not in used_colors:
                    colors[edge_idx] = c
                    if dfs(edge_idx + 1, colors):
                        return True
                    colors[edge_idx] = 0
            return False
        
        if dfs(0, color):
            print("YES")
            print(' '.join(map(str, color)))
        else:
            print("NO")

solve()
