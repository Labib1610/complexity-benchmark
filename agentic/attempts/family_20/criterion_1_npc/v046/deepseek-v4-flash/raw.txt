We can model this as a **3-edge-coloring** problem on a **3-regular graph** (each valve has degree 3).  

By **Kőnig’s line coloring theorem**, every bipartite graph has edge chromatic number equal to its maximum degree.  
But here the graph is not necessarily bipartite. However, a **3-regular graph** is 3-edge-colorable if and only if it is **bridgeless** (by **Tait’s theorem**).  
If the graph has a bridge, it cannot be 3-edge-colored.

So the algorithm is:
1. Check if the graph is bridgeless (no bridges).
2. If it has a bridge → output `NO`.
3. Otherwise, find a proper 3-edge-coloring.

We can find a 3-edge-coloring by:
- Finding a **perfect matching** (since it’s 3-regular and bridgeless, it has one).
- Remove that matching → remaining graph is 2-regular (disjoint cycles).
- Color the matching edges with color 1.
- Color each cycle alternately with colors 2 and 3.

We can find a perfect matching using **Blossom algorithm** (general graph matching).  
Since n ≤ 2000, m ≤ 3000, we can implement an efficient blossom algorithm.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1000000)

def find_perfect_matching(n, edges):
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        adj[u].append((v, idx))
        adj[v].append((u, idx))
    
    # Blossom algorithm for maximum matching
    match = [-1] * n
    p = [-1] * n
    base = list(range(n))
    q = []
    used = [False] * n
    blossom = [False] * n
    
    def lca(a, b):
        used_lca = [False] * n
        while True:
            a = base[a]
            used_lca[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_lca[b]:
                return b
            b = p[match[b]]
    
    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]
    
    def find_path(root):
        nonlocal p, base, used, q, blossom
        used = [False] * n
        p = [-1] * n
        for i in range(n):
            base[i] = i
        q = [root]
        used[root] = True
        while q:
            v = q.pop(0)
            for to, _ in adj[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    curbase = lca(v, to)
                    blossom = [False] * n
                    mark_path(v, curbase, to)
                    mark_path(to, curbase, v)
                    for i in range(n):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        return to
                    to = match[to]
                    used[to] = True
                    q.append(to)
        return -1
    
    for i in range(n):
        if match[i] == -1:
            v = find_path(i)
            while v != -1:
                pv = p[v]
                nv = match[pv]
                match[v] = pv
                match[pv] = v
                v = nv
    
    # Check if perfect
    if all(x != -1 for x in match):
        matching_edges = []
        seen = set()
        for u, v, idx in edges:
            if match[u] == v and (u, v) not in seen and (v, u) not in seen:
                matching_edges.append(idx)
                seen.add((u, v))
        return matching_edges
    return None

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v, idx))
        adj[u].append((v, idx))
        adj[v].append((u, idx))
    
    # Check for bridges using DFS
    visited = [False] * n
    tin = [-1] * n
    low = [-1] * n
    timer = 0
    has_bridge = False
    
    def dfs(v, p_edge):
        nonlocal timer, has_bridge
        visited[v] = True
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
    
    dfs(0, -1)
    if has_bridge:
        print("NO")
        return
    
    # Find perfect matching
    matching = find_perfect_matching(n, edges)
    if matching is None:
        print("NO")
        return
    
    color = [0] * m
    # Color matching edges with 1
    for eid in matching:
        color[eid] = 1
    
    # Remaining graph is 2-regular (cycles)
    # Build adjacency for remaining edges
    rem_adj = [[] for _ in range(n)]
    for u, v, idx in edges:
        if color[idx] == 0:
            rem_adj[u].append((v, idx))
            rem_adj[v].append((u, idx))
    
    # Color cycles alternately with 2 and 3
    visited_cycle = [False] * n
    for start in range(n):
        if not visited_cycle[start] and rem_adj[start]:
            # traverse cycle
            cur = start
            prev_edge = -1
            c = 2
            while True:
                visited_cycle[cur] = True
                # pick next edge not prev_edge
                nxt_edge = None
                nxt_node = None
                for to, eid in rem_adj[cur]:
                    if eid != prev_edge:
                        nxt_edge = eid
                        nxt_node = to
                        break
                if nxt_edge is None:
                    break
                color[nxt_edge] = c
                c = 3 if c == 2 else 2
                prev_edge = nxt_edge
                cur = nxt_node
                if cur == start:
                    break
    
    print("YES")
    print(" ".join(str(color[i]) for i in range(m)))

if __name__ == "__main__":
    solve()
```
