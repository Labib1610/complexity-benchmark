import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Each node has exactly 3 edges
    for u in range(1, n + 1):
        if len(adj[u]) != 3:
            print("NO")
            return
    
    # Try to 3-color the line graph (each edge is a node, adjacent if they share a vertex)
    # This is equivalent to edge coloring the original graph with 3 colors
    # Since each vertex has degree 3, this is possible iff the graph is 3-edge-colorable
    
    # We'll model this as a constraint satisfaction problem
    # For each edge, we need to assign a color 1,2,3
    # Adjacent edges (sharing a vertex) must have different colors
    
    # This is essentially checking if the graph is 3-edge-colorable
    # For a 3-regular graph, this is equivalent to finding a proper edge 3-coloring
    
    # Let's use a greedy approach with backtracking
    # But since n <= 2000, we need something efficient
    
    # Key insight: This is equivalent to finding a proper edge coloring of a 3-regular graph
    # Such a graph is always class 1 (edge chromatic number = max degree) iff it's not a snark
    # But we can try a direct approach
    
    # Model as edge coloring - each edge gets color 1,2,3
    # Edges sharing a vertex must have different colors
    
    color = [0] * m  # 0 means uncolored
    
    def conflict(e_idx, c):
        u, v = edges[e_idx]
        for e2_idx in adj[u]:
            if color[e2_idx] == c:
                return True
        for e2_idx in adj[v]:
            if color[e2_idx] == c:
                return True
        return False
    
    # Try to color greedily
    def dfs(e_idx):
        if e_idx == m:
            return True
            
        u, v = edges[e_idx]
        # Try colors 1, 2, 3 in order
        for c in [1, 2, 3]:
            if not conflict(e_idx, c):
                color[e_idx] = c
                if dfs(e_idx + 1):
                    return True
                color[e_idx] = 0  # backtrack
        
        return False
    
    # For large graphs, this might be too slow
    # But let's try a smarter approach
    
    # In a 3-regular graph, we can try to find a perfect matching and recurse
    # But let's stick to the DFS with some optimizations
    
    # Actually, let's think differently
    # This problem is equivalent to edge-coloring a 3-regular graph with 3 colors
    # This is possible if and only if the graph is not a snark
    # But checking if it's a snark is hard
    
    # Let's just implement a smart backtracking
    # We can order edges to improve performance
    
    # Order edges by trying to color those with more colored neighbors first
    order = list(range(m))
    
    def bt(e_idx):
        if e_idx == m:
            return True
            
        edge_idx = order[e_idx]
        u, v = edges[edge_idx]
        
        # Available colors
        used = set()
        for nbr_edge in adj[u]:
            if color[nbr_edge] != 0:
                used.add(color[nbr_edge])
        for nbr_edge in adj[v]:
            if color[nbr_edge] != 0:
                used.add(color[nbr_edge])
                
        for c in [1, 2, 3]:
            if c not in used:
                color[edge_idx] = c
                if bt(e_idx + 1):
                    return True
                color[edge_idx] = 0
                
        return False
    
    # Sort edges by how many of their endpoints already have colored edges
    # This heuristic often helps
    def sort_key(e_idx):
        u, v = edges[e_idx]
        count = 0
        for nbr in adj[u]:
            if nbr != e_idx and color[nbr] != 0:
                count += 1
        for nbr in adj[v]:
            if nbr != e_idx and color[nbr] != 0:
                count += 1
        return -count  # prioritize edges with more colored neighbors
    
    # Simple greedy first
    success = bt(0)
    
    if success:
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
