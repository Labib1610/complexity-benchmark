import sys
sys.setrecursionlimit(10000000)

def solve() -> None:
    import sys
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # Blossom algorithm for maximum matching (Edmonds)
    # Implementation based on https://github.com/jaehyunp/stanfordacm/blob/master/code/EdmondsMatching.cc
    N = n
    g = [[] for _ in range(N)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    match = [-1] * N
    p = [-1] * N
    base = list(range(N))
    q = []
    in_blossom = [False] * N
    in_queue = [False] * N

    def lca(a, b):
        used = [False] * N
        while True:
            a = base[a]
            used[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom = base[v]
            in_blossom[blossom] = in_blossom[base[match[blossom]]] = True
            p[v] = children
            children = match[blossom]
            v = p[match[blossom]]

    def blossom_contraction(v, u):
        b = lca(v, u)
        in_blossom[:] = [False] * N
        mark_path(v, b, u)
        mark_path(u, b, v)
        if base[v] != b:
            p[v] = u
        if base[u] != b:
            p[u] = v
        for i in range(N):
            if in_blossom[base[i]]:
                base[i] = b
                if not in_queue[i]:
                    q.append(i)
                    in_queue[i] = True

    def bfs(root):
        nonlocal q, p, base, in_queue
        p = [-1] * N
        in_queue = [False] * N
        base = list(range(N))
        q = [root]
        in_queue[root] = True
        qi = 0
        while qi < len(q):
            v = q[qi]
            qi += 1
            for to in g[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    blossom_contraction(v, to)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        # augmenting path found
                        cur = to
                        while cur != -1:
                            prev = p[cur]
                            nxt = match[prev] if prev != -1 else -1
                            match[cur] = prev
                            match[prev] = cur
                            cur = nxt
                        return True
                    else:
                        q.append(match[to])
                        in_queue[match[to]] = True
        return False

    for v in range(N):
        if match[v] == -1:
            bfs(v)

    # check if perfect matching
    if any(mch == -1 for mch in match):
        print("NO")
        return

    # Build the 2-factor (remaining edges)
    used_edge = [False] * m
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            used_edge[idx] = True

    # adjacency of remaining graph (each vertex degree 2)
    rem_adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        if not used_edge[idx]:
            rem_adj[u].append((v, idx))
            rem_adj[v].append((u, idx))

    color = [0] * m
    # color matching edges with color 1
    for idx, (u, v) in enumerate(edges):
        if used_edge[idx]:
            color[idx] = 1

    # color cycles alternating 2 and 3
    visited = [False] * n
    for start in range(n):
        if visited[start]:
            continue
        # traverse cycle
        cur = start
        prev_edge = -1
        cycle_edges = []
        while True:
            visited[cur] = True
            for nxt, eid in rem_adj[cur]:
                if eid == prev_edge:
                    continue
                cycle_edges.append(eid)
                prev_edge = eid
                cur = nxt
                break
            else:
                break
            if cur == start:
                break
        # assign colors
        for i, eid in enumerate(cycle_edges):
            color[eid] = 2 if i % 2 == 0 else 3

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
