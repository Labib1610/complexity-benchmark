```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(list)
    
    idx = 2
    for i in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Since m = 3*n/2 and each vertex has the same degree,
    # each vertex has degree 3
    deg = m * 2 // n  # which should be 3
    
    # We need to color the edges such that edges incident to each vertex have distinct colors
    # This is edge coloring with deg colors
    
    # Since each vertex has degree 3, we need 3 colors
    k = deg
    
    # Build adjacency for edges (line graph)
    edge_adj = [[] for _ in range(m)]
    for v in range(n):
        incident = adj[v]
        for i in range(len(incident)):
            for j in range(i+1, len(incident)):
                edge_adj[incident[i]].append(incident[j])
                edge_adj[incident[j]].append(incident[i])
    
    # Now we do greedy edge coloring
    color = [-1] * m  # -1 means uncolored
    
    for e in range(m):
        # Find forbidden colors
        forbidden = [False] * (k + 1)  # indices 1 to k
        for nei in edge_adj[e]:
            if color[nei] != -1:
                forbidden[color[nei]] = True
        
        # Assign the smallest available color
        c = -1
        for i in range(1, k + 1):
            if not forbidden[i]:
                c = i
                break
        
        if c == -1:
            print("NO")
            return
            
        color[e] = c
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
```
