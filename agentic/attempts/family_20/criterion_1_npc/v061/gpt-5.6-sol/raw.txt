import sys
sys.setrecursionlimit(10000)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    if m != 3 * n // 2 or len(data) != 2 + 2 * m:
        print("NO")
        return

    edges = []
    adj = [[] for _ in range(n)]

    pos = 2
    for eid in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if not (0 <= u < v < n):
            print("NO")
            return
        edges.append((u, v))
        adj[u].append(eid)
        adj[v].append(eid)

    if any(len(adj[v]) != 3 for v in range(n)):
        print("NO")
        return

    # Find connected components and bridges.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False
    components_vertices = []

    def dfs(v, parent_edge, vertices):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1
        vertices.append(v)

        for eid in adj[v]:
            if eid == parent_edge:
                continue
            a, b = edges[eid]
            to = b if a == v else a
            if tin[to] == -1:
                dfs(to, eid, vertices)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for start in range(n):
        if tin[start] == -1:
            vertices = []
            dfs(start, -1, vertices)
            components_vertices.append(vertices)

    if has_bridge:
        print("NO")
        return

    component_of = [-1] * n
    for cid, vertices in enumerate(components_vertices):
        for v in vertices:
            component_of[v] = cid

    component_edges = [[] for _ in components_vertices]
    for eid, (u, _) in enumerate(edges):
        component_edges[component_of[u]].append(eid)

    # Domain masks: bit 0 -> color 1, bit 1 -> color 2, bit 2 -> color 3.
    domains = [7] * m
    permutations = (
        (1, 2, 4), (1, 4, 2),
        (2, 1, 4), (2, 4, 1),
        (4, 1, 2), (4, 2, 1)
    )

    def propagate(initial_vertices, trail):
        queue = list(initial_vertices)

        while queue:
            v = queue.pop()
            e0, e1, e2 = adj[v]
            d0, d1, d2 = domains[e0], domains[e1], domains[e2]

            s0 = s1 = s2 = 0
            valid = False

            for p0, p1, p2 in permutations:
                if (d0 & p0) and (d1 & p1) and (d2 & p2):
                    valid = True
                    s0 |= p0
                    s1 |= p1
                    s2 |= p2

            if not valid:
                return False

            for eid, support in ((e0, s0), (e1, s1), (e2, s2)):
                old = domains[eid]
                new = old & support
                if new == 0:
                    return False
                if new != old:
                    trail.append((eid, old))
                    domains[eid] = new
                    a, b = edges[eid]
                    queue.append(a)
                    queue.append(b)

        return True

    initial_trail = []
    initial_vertices = []

    # Color symmetry can be fixed independently in every component.
    for vertices in components_vertices:
        root = vertices[0]
        for eid, color_mask in zip(adj[root], (1, 2, 4)):
            old = domains[eid]
            if not (old & color_mask):
                print("NO")
                return
            if old != color_mask:
                initial_trail.append((eid, old))
                domains[eid] = color_mask
                a, b = edges[eid]
                initial_vertices.append(a)
                initial_vertices.append(b)

    if not propagate(initial_vertices, initial_trail):
        print("NO")
        return

    def solve(edge_ids):
        chosen = -1
        chosen_domain = 0

        # Prefer binary domains.
        for eid in edge_ids:
            d = domains[eid]
            if d & (d - 1):
                if d.bit_count() == 2:
                    chosen = eid
                    chosen_domain = d
                    break
                if chosen == -1:
                    chosen = eid
                    chosen_domain = d

        if chosen == -1:
            return True

        bits = []
        d = chosen_domain
        while d:
            bit = d & -d
            bits.append(bit)
            d -= bit

        for bit in bits:
            checkpoint = []
            old = domains[chosen]
            checkpoint.append((chosen, old))
            domains[chosen] = bit
            u, v = edges[chosen]

            if propagate((u, v), checkpoint) and solve(edge_ids):
                return True

            for eid, previous in reversed(checkpoint):
                domains[eid] = previous

        return False

    for edge_ids in component_edges:
        if not solve(edge_ids):
            print("NO")
            return

    colors = [d.bit_length() for d in domains]
    print("YES")
    print(*colors)

if __name__ == "__main__":
    main()
