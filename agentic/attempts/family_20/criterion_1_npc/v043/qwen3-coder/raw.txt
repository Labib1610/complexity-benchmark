import sys
from collections import defaultdict

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
    return n, m, edges

def solve(n, m, edges):
    # Build adjacency list
    adj = defaultdict(list)
    edge_indices = {}
    for i, (u, v) in enumerate(edges):
        adj[u].append((v, i))
        adj[v].append((u, i))
        edge_indices[(u, v)] = i
        edge_indices[(v, u)] = i
    
    # Since each vertex has degree 3, we can try to 3-color the edges
    # This is equivalent to finding a proper edge coloring with 3 colors
    # For a 3-regular graph, this is possible iff the graph is class 1
    # We'll attempt to construct such a coloring
    
    color = [0] * m  # 0 means uncolored, 1,2,3 are the colors
    
    # Try to color edges greedily
    # For each vertex, ensure its 3 edges have different colors
    
    def get_available_colors(u):
        used = set()
        for v, idx in adj[u]:
            if color[idx] != 0:
                used.add(color[idx])
        return {1, 2, 3} - used
    
    # We'll use a greedy approach with backtracking if needed
    # But let's try a simpler method first: since each node has exactly 3 edges,
    # we just need to assign colors 1,2,3 to them
    
    # For each vertex, we'll assign colors to its incident edges
    # We need to be consistent across shared edges
    
    visited_edges = [False] * m
    
    # Process vertices one by one
    for u in range(1, n+1):
        # Get unvisited edges incident to u
        unvisited = []
        for v, idx in adj[u]:
            if not visited_edges[idx]:
                unvisited.append((v, idx))
        
        # Assign colors 1,2,3 to these edges
        colors_to_assign = [1, 2, 3]
        for i, (v, idx) in enumerate(unvisited):
            if i < len(colors_to_assign):
                color[idx] = colors_to_assign[i]
            else:
                # This shouldn't happen in a valid 3-regular graph processing one vertex at a time
                pass
            visited_edges[idx] = True
    
    # Check if our assignment is valid
    valid = True
    for u in range(1, n+1):
        colors_at_u = set()
        for v, idx in adj[u]:
            colors_at_u.add(color[idx])
        if len(colors_at_u) != 3 or 0 in colors_at_u:
            valid = False
            break
    
    if valid:
        return "YES", color
    else:
        # Try a more systematic approach
        # This is essentially edge coloring of a 3-regular graph
        # We can model this as a SAT problem or use known algorithms
        
        # Let's try a backtracking approach
        color = [0] * m
        
        def is_valid_color(edge_idx, c):
            u, v = edges[edge_idx]
            # Check all edges incident to u
            for w, idx in adj[u]:
                if color[idx] == c:
                    return False
            # Check all edges incident to v
            for w, idx in adj[v]:
                if color[idx] == c:
                    return False
            return True
        
        def backtrack(edge_idx):
            if edge_idx == m:
                return True
            
            for c in [1, 2, 3]:
                if is_valid_color(edge_idx, c):
                    color[edge_idx] = c
                    if backtrack(edge_idx + 1):
                        return True
                    color[edge_idx] = 0
            return False
        
        if backtrack(0):
            return "YES", color
        else:
            return "NO", []

def main():
    n, m, edges = read_input()
    result, coloring = solve(n, m, edges)
    print(result)
    if result == "YES":
        print(' '.join(map(str, coloring)))

if __name__ == "__main__":
    main()
