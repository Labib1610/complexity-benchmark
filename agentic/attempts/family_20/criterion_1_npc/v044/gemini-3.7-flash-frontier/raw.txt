import sys
import random

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
        
    # edge_color[i] = 0 (uncolored), 1, 2, or 3
    edge_color = [0] * m
    # color_edge[u][c] = edge index colored c incident to u, or -1
    color_edge = [[-1] * 4 for _ in range(n + 1)]
    
    def get_missing_colors(u):
        res = []
        for c in (1, 2, 3):
            if color_edge[u][c] == -1:
                res.append(c)
        return res

    uncolored = list(range(m))
    random.seed(42)
    random.shuffle(uncolored)
    
    max_steps = 150000
    steps = 0
    
    while uncolored:
        steps += 1
        if steps > max_steps:
            print("NO")
            return
            
        e = uncolored.pop()
        u, v = edges[e]
        
        miss_u = get_missing_colors(u)
        miss_v = get_missing_colors(v)
        
        # Check common missing color
        common = set(miss_u) & set(miss_v)
        if common:
            c = common.pop()
            edge_color[e] = c
            color_edge[u][c] = e
            color_edge[v][c] = e
            continue
            
        # No common missing color
        c_u = random.choice(miss_u)
        c_v = random.choice(miss_v)
        
        # Trace Kempe chain (c_v, c_u) starting from v
        # We want to flip colors c_v and c_u along the path starting at v
        # Path follows edges colored c_u, then c_v, then c_u, etc.
        path = []
        curr = v
        curr_c = c_u # because c_u is present at v, but missing at u
        
        while True:
            nxt_e = color_edge[curr][curr_c]
            if nxt_e == -1:
                break
            path.append((curr, nxt_e, curr_c))
            curr = edges[nxt_e][0] if edges[nxt_e][1] == curr else edges[nxt_e][1]
            curr_c = c_v if curr_c == c_u else c_u
            
        if curr != u:
            # Kempe chain does not end at u. We can flip it!
            for node, pe, pc in path:
                other_c = c_v if pc == c_u else c_u
                # change pe's color from pc to other_c
                p_u, p_v = edges[pe]
                color_edge[p_u][pc] = -1
                color_edge[p_v][pc] = -1
                edge_color[pe] = other_c
                color_edge[p_u][other_c] = pe
                color_edge[p_v][other_c] = pe
                
            # Now c_u is missing at both u and v!
            edge_color[e] = c_u
            color_edge[u][c_u] = e
            color_edge[v][c_u] = e
        else:
            # Kempe chain ends at u.
            # Pick a random colored edge at u or v to uncolor, then color e
            # Alternatively, uncolor an edge at u with color != c_u
            candidates = [color_edge[u][c] for c in (1, 2, 3) if color_edge[u][c] != -1]
            if not candidates:
                candidates = [color_edge[v][c] for c in (1, 2, 3) if color_edge[v][c] != -1]
            
            evict = random.choice(candidates)
            ev_u, ev_v = edges[evict]
            ev_c = edge_color[evict]
            
            color_edge[ev_u][ev_c] = -1
            color_edge[ev_v][ev_c] = -1
            edge_color[evict] = 0
            
            uncolored.append(evict)
            uncolored.append(e)

    print("YES")
    print(*(edge_color[i] for i in range(m)))

if __name__ == '__main__':
    solve()
