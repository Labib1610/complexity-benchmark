We can model this as a **3-edge-coloring** problem on a **3-regular graph** (each vertex has degree 3).  

A 3-regular graph has a proper 3-edge-coloring if and only if it is **bridgeless** (by **Petersen’s theorem**).  
So the algorithm is:

1. Check if the graph has any **bridges** (edges whose removal disconnects the graph).
2. If there is a bridge → output `NO`.
3. Otherwise, find a proper 3-edge-coloring using a constructive method (e.g., based on finding a perfect matching, removing it, then 2-coloring the remaining cycles).

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(10000)

def find_bridges(n, edges, adj):
    visited = [False] * (n + 1)
    tin = [-1] * (n + 1)
    low = [-1] * (n + 1)
    timer = 0
    bridges = set()

    def dfs(v, p_edge):
        nonlocal timer
        visited[v] = True
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridges.add(eid)
    for i in range(1, n+1):
        if not visited[i]:
            dfs(i, -1)
    return bridges

def solve():
    input_data = sys.stdin.read().strip().split()
    n, m = int(input_data[0]), int(input_data[1])
    edges = []
    adj = [[] for _ in range(n+1)]
    idx = 2
    for eid in range(m):
        u = int(input_data[idx]); v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append((v, eid))
        adj[v].append((u, eid))

    # Step 1: Check for bridges
    bridges = find_bridges(n, edges, adj)
    if bridges:
        print("NO")
        return

    # Step 2: Find a perfect matching (since graph is 3-regular and bridgeless)
    # Use simple greedy matching on edges
    matched = [False] * m
    color = [0] * m
    match_edges = []
    used_vertex = [False] * (n+1)

    # Greedy matching
    for eid, (u, v) in enumerate(edges):
        if not used_vertex[u] and not used_vertex[v]:
            used_vertex[u] = True
            used_vertex[v] = True
            match_edges.append(eid)
            matched[eid] = True
            color[eid] = 1

    # Step 3: Remove matching edges, remaining graph is 2-regular (cycles)
    # Color remaining edges with 2 and 3 alternately along cycles
    remaining_adj = [[] for _ in range(n+1)]
    for eid, (u, v) in enumerate(edges):
        if not matched[eid]:
            remaining_adj[u].append((v, eid))
            remaining_adj[v].append((u, eid))

    visited = [False] * (n+1)
    for start in range(1, n+1):
        if not visited[start] and remaining_adj[start]:
            # traverse cycle
            cur = start
            prev_eid = -1
            while True:
                visited[cur] = True
                # pick next edge not equal to prev_eid
                nxt_v, nxt_eid = remaining_adj[cur][0]
                if nxt_eid == prev_eid:
                    nxt_v, nxt_eid = remaining_adj[cur][1]
                # assign color alternating
                if prev_eid == -1:
                    color[nxt_eid] = 2
                else:
                    color[nxt_eid] = 3 if color[prev_eid] == 2 else 2
                prev_eid = nxt_eid
                cur = nxt_v
                if cur == start:
                    break

    print("YES")
    print(" ".join(str(c) for c in color))

if __name__ == "__main__":
    solve()
```
