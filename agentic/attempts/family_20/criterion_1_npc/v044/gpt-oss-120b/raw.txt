import sys
sys.setrecursionlimit(10000)

def blossom_maximum_matching(n, g):
    """Edmonds' blossom algorithm for general graph.
    n: number of vertices (0..n-1)
    g: adjacency list of sets
    returns match list where match[v] = partner or -1
    """
    INF = n + 5
    match = [-1] * n
    p = [-1] * n
    base = list(range(n))
    q = []
    used = [False] * n
    blossom = [False] * n

    def lca(a, b):
        used_path = [False] * n
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal q, used, p, base, blossom
        used = [False] * n
        p = [-1] * n
        base = list(range(n))
        q = []
        q_head = 0
        q.append(root)
        used[root] = True
        while q_head < len(q):
            v = q[q_head]
            q_head += 1
            for to in g[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    curbase = lca(v, to)
                    blossom = [False] * n
                    mark_path(v, curbase, to)
                    mark_path(to, curbase, v)
                    for i in range(n):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        # augmenting path found
                        v = to
                        while v != -1:
                            pv = p[v]
                            ppv = match[pv] if pv != -1 else -1
                            match[v] = pv
                            match[pv] = v
                            v = ppv
                        return True
                    else:
                        used[match[to]] = True
                        q.append(match[to])
        return False

    for v in range(n):
        if match[v] == -1:
            find_path(v)
    return match

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # build simple adjacency set for blossom
    g = [set() for _ in range(n)]
    for u, v in edges:
        g[u].add(v)
        g[v].add(u)

    match = blossom_maximum_matching(n, g)
    if any(x == -1 for x in match):
        print("NO")
        return

    # assign colors
    color = [0] * m
    matched_edge = {}
    for v in range(n):
        u = match[v]
        if u != -1 and v < u:
            # find edge id
            for to, idx in adj[v]:
                if to == u:
                    matched_edge[(v, u)] = idx
                    matched_edge[(u, v)] = idx
                    color[idx] = 1
                    break

    # remaining edges form cycles, color them 2 and 3 alternately
    visited_edge = [False] * m
    for idx in range(m):
        if color[idx] != 0:
            visited_edge[idx] = True

    for start in range(n):
        for to, eid in adj[start]:
            if visited_edge[eid]:
                continue
            # start a new cycle
            cur = start
            prev = -1
            cur_color = 2
            while True:
                # find the unused edge from cur (not the one we came from)
                for nxt, neid in adj[cur]:
                    if visited_edge[neid]:
                        continue
                    # this is the next edge in the cycle
                    color[neid] = cur_color
                    visited_edge[neid] = True
                    # move forward
                    prev, cur = cur, nxt
                    cur_color = 5 - cur_color  # toggle between 2 and 3
                    break
                else:
                    break
                if cur == start:
                    break

    print("YES")
    print(' '.join(str(c) for c in color))

if __name__ == "__main__":
    main()
