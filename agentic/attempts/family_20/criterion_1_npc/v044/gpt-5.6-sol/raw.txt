import sys
sys.setrecursionlimit(1000000)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)

    eu = [0] * m
    ev = [0] * m
    adj = [[] for _ in range(n)]

    for i in range(m):
        u = next(it) - 1
        v = next(it) - 1
        eu[i] = u
        ev[i] = v
        adj[u].append(i)
        adj[v].append(i)

    def other(e, x):
        return ev[e] if eu[e] == x else eu[e]

    # A cubic graph with a bridge cannot have a proper 3-edge-coloring.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False

    def bridge_dfs(u, parent_edge):
        nonlocal timer, has_bridge
        tin[u] = low[u] = timer
        timer += 1

        for e in adj[u]:
            if e == parent_edge:
                continue
            v = other(e, u)
            if tin[v] == -1:
                bridge_dfs(v, e)
                low[u] = min(low[u], low[v])
                if low[v] > tin[u]:
                    has_bridge = True
            else:
                low[u] = min(low[u], tin[v])

    for s in range(n):
        if tin[s] == -1:
            bridge_dfs(s, -1)

    if has_bridge:
        print("NO")
        return

    # Find connected components and their bipartitions.
    components = []
    side = [-1] * n

    for start in range(n):
        if side[start] != -1:
            continue

        stack = [start]
        side[start] = 0
        vertices = []
        edge_set = set()
        bipartite = True

        while stack:
            u = stack.pop()
            vertices.append(u)
            for e in adj[u]:
                edge_set.add(e)
                v = other(e, u)
                if side[v] == -1:
                    side[v] = side[u] ^ 1
                    stack.append(v)
                elif side[v] == side[u]:
                    bipartite = False

        components.append((vertices, list(edge_set), bipartite))

    color = [0] * m
    used = [0] * n
    trail = []

    def rollback(checkpoint):
        while len(trail) > checkpoint:
            e = trail.pop()
            bit = 1 << (color[e] - 1)
            used[eu[e]] ^= bit
            used[ev[e]] ^= bit
            color[e] = 0

    def propagate(initial):
        stack = list(initial)

        while stack:
            e, c = stack.pop()

            if color[e]:
                if color[e] != c:
                    return False
                continue

            bit = 1 << (c - 1)
            u = eu[e]
            v = ev[e]

            if (used[u] & bit) or (used[v] & bit):
                return False

            color[e] = c
            used[u] |= bit
            used[v] |= bit
            trail.append(e)

            for x in (u, v):
                for f in adj[x]:
                    if color[f]:
                        continue
                    a = eu[f]
                    b = ev[f]
                    domain = 7 & ~used[a] & ~used[b]
                    if domain == 0:
                        return False
                    if domain & (domain - 1) == 0:
                        forced_color = domain.bit_length()
                        stack.append((f, forced_color))

        return True

    def solve_bipartite(vertices, edges):
        left = [v for v in vertices if side[v] == 0]
        inf = 10 ** 9
        match = [-1] * n
        match_edge = [-1] * n
        dist = [0] * n

        from collections import deque

        def bfs():
            q = deque()
            found = False
            for u in left:
                if match[u] == -1:
                    dist[u] = 0
                    q.append(u)
                else:
                    dist[u] = inf

            while q:
                u = q.popleft()
                for e in adj[u]:
                    v = other(e, u)
                    w = match[v]
                    if w == -1:
                        found = True
                    elif dist[w] == inf:
                        dist[w] = dist[u] + 1
                        q.append(w)
            return found

        def dfs(u):
            for e in adj[u]:
                v = other(e, u)
                w = match[v]
                if w == -1 or (dist[w] == dist[u] + 1 and dfs(w)):
                    match[u] = v
                    match[v] = u
                    match_edge[u] = e
                    match_edge[v] = e
                    return True
            dist[u] = inf
            return False

        matched = 0
        while bfs():
            for u in left:
                if match[u] == -1 and dfs(u):
                    matched += 1

        if matched != len(left):
            return False

        checkpoint = len(trail)
        initial = [(match_edge[u], 1) for u in left]
        if not propagate(initial):
            rollback(checkpoint)
            return False

        for e in edges:
            if color[e] == 0:
                cp = len(trail)
                if not propagate([(e, 2)]):
                    rollback(cp)
                    rollback(checkpoint)
                    return False

        return True

    def solve_general(vertices, edges):
        checkpoint_component = len(trail)

        # Global color permutation symmetry allows fixing one vertex.
        root = max(vertices, key=lambda x: sum(
            1 for e in adj[x]
            for f in adj[other(e, x)]
            if other(f, other(e, x)) in vertices
        ))
        root_edges = adj[root]

        if not propagate([
            (root_edges[0], 1),
            (root_edges[1], 2),
            (root_edges[2], 3)
        ]):
            rollback(checkpoint_component)
            return False

        def dfs():
            best_edge = -1
            best_domain = 0
            best_size = 4
            best_score = -1

            for e in edges:
                if color[e]:
                    continue
                u = eu[e]
                v = ev[e]
                domain = 7 & ~used[u] & ~used[v]
                size = domain.bit_count()

                if size == 0:
                    return False

                score = used[u].bit_count() + used[v].bit_count()
                if size < best_size or (size == best_size and score > best_score):
                    best_size = size
                    best_score = score
                    best_edge = e
                    best_domain = domain
                    if size == 1:
                        break

            if best_edge == -1:
                return True

            candidates = []
            d = best_domain
            while d:
                bit = d & -d
                candidates.append(bit.bit_length())
                d -= bit

            for c in candidates:
                cp = len(trail)
                if propagate([(best_edge, c)]) and dfs():
                    return True
                rollback(cp)

            return False

        if dfs():
            return True

        rollback(checkpoint_component)
        return False

    # Bipartite components admit a polynomial construction. Other components
    # are solved by constraint propagation and backtracking.
    components.sort(key=lambda x: (not x[2], len(x[1])))

    for vertices, edges, bipartite in components:
        if bipartite:
            if not solve_bipartite(vertices, edges):
                print("NO")
                return
        else:
            if not solve_general(vertices, edges):
                print("NO")
                return

    print("YES")
    print(*color)

if __name__ == "__main__":
    main()
