import sys
from collections import defaultdict

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    adj = defaultdict(list)
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    return n, m, edges, adj

def solve():
    n, m, edges, adj = read_input()
    
    # Each vertex has exactly 3 edges
    # We need to assign phases 1,2,3 to edges such that 
    # each vertex has edges with distinct phases
    
    # This is equivalent to edge-coloring with 3 colors
    # Since each vertex has degree 3, this is possible iff 
    # the graph is 3-edge-colorable
    
    # For a 3-regular graph, it's 3-edge-colorable iff it's bipartite
    # But that's not always true. Let's try a constructive approach.
    
    # Try to build a valid assignment
    color = [0] * m  # 0 means uncolored
    
    # For each vertex, we need to assign colors 1,2,3 to its 3 edges
    # Let's process vertices one by one
    
    # We can model this as finding a proper edge coloring
    # Since the graph is 3-regular, we can use the fact that
    # it can be decomposed into 3 perfect matchings (if it's class 1)
    
    # Let's try a greedy approach with backtracking
    
    def get_available_colors(edge_idx):
        u, v = edges[edge_idx]
        used_by_u = set()
        used_by_v = set()
        for e_idx in adj[u]:
            if color[e_idx] != 0:
                used_by_u.add(color[e_idx])
        for e_idx in adj[v]:
            if color[e_idx] != 0:
                used_by_v.add(color[e_idx])
        available = set([1, 2, 3]) - used_by_u - used_by_v
        return list(available)
    
    # Try a different approach: since each vertex has exactly 3 edges,
    # and we have 3 colors, we just need to ensure no two edges
    # incident to the same vertex have the same color
    
    # This is essentially edge coloring of a 3-regular graph with 3 colors
    # which is always possible (by Vizing's theorem, chromatic index is 3 or 4,
    # and for 3-regular graphs it's 3 when it has a perfect matching)
    
    # Let's try a greedy algorithm with a specific order
    # Process edges in some order, for each edge pick a color
    # that's not used by its endpoints
    
    def can_color_edge(edge_idx, c):
        u, v = edges[edge_idx]
        for e_idx in adj[u]:
            if color[e_idx] == c:
                return False
        for e_idx in adj[v]:
            if color[e_idx] == c:
                return False
        return True
    
    # Try to color greedily
    def dfs(edge_idx):
        if edge_idx == m:
            return True
            
        u, v = edges[edge_idx]
        for c in [1, 2, 3]:
            if can_color_edge(edge_idx, c):
                color[edge_idx] = c
                if dfs(edge_idx + 1):
                    return True
                color[edge_idx] = 0
        return False
    
    # This might be too slow. Let's think of a better way.
    
    # Since the graph is 3-regular, we can try to find a proper edge coloring
    # using a known algorithm or property
    
    # One approach: Find a perfect matching, remove it, get a 2-regular graph (union of cycles),
    # 2-color each cycle, assign the matching edges the 3rd color
    
    # But let's stick with backtracking but optimize it
    
    # Sort edges by the sum of degrees of their endpoints in the current partial graph
    # Actually, all vertices have degree 3, so that won't help
    
    # Let's just implement DFS with good pruning
    
    def solve_coloring():
        # For each vertex, maintain what colors are used by its edges
        vertex_colors = defaultdict(set)
        
        def backtrack(edge_idx):
            if edge_idx == m:
                return True
                
            u, v = edges[edge_idx]
            used_colors = vertex_colors[u] | vertex_colors[v]
            available_colors = [c for c in [1, 2, 3] if c not in used_colors]
            
            for c in available_colors:
                color[edge_idx] = c
                vertex_colors[u].add(c)
                vertex_colors[v].add(c)
                
                if backtrack(edge_idx + 1):
                    return True
                    
                vertex_colors[u].remove(c)
                vertex_colors[v].remove(c)
                
            color[edge_idx] = 0
            return False
        
        return backtrack(0)
    
    if solve_coloring():
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
