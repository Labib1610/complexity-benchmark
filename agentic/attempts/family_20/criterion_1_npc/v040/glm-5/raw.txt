import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check degree constraint
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    # Build edge adjacency: edges sharing a vertex
    edge_adj = [[] for _ in range(m)]
    for u in range(1, n + 1):
        incident_edges = [idx for (v, idx) in adj[u]]
        for i in range(3):
            for j in range(i + 1, 3):
                e1 = incident_edges[i]
                e2 = incident_edges[j]
                edge_adj[e1].append(e2)
                edge_adj[e2].append(e1)
    
    color = [0] * m
    
    # BFS to 2-color each connected component
    for start in range(m):
        if color[start] == 0:
            queue = deque([start])
            color[start] = 1
            while queue:
                u = queue.popleft()
                for v in edge_adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # Now we have a 2-coloring. We need to assign phases 1, 2, 3.
    # At each vertex, we have 3 incident edges. Two must have the same color (say 1),
    # and one must have the other color (say 2). We assign phase 3 to the singleton,
    # and phases 1 and 2 to the pair.
    
    # For each vertex, determine which color appears once (singleton) and which appears twice (pair)
    # We'll use a union-find to ensure consistency across vertices
    
    parent = list(range(2 * m + 2))
    rank = [0] * (2 * m + 2)
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    def union(x, y):
        rx, ry = find(x), find(y)
        if rx == ry:
            return
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1

    # Let's say edges e1, e2 have color c, and e3 has color 3-c.

    # This means: e1 and e2 are "paired" (both get phase 1 or both get phase 2).
    # e3 is "singleton" (gets phase 3).

    # Node 2*e represents "edge e is phase 1", node 2*e+1 represents "edge e is phase 2".
    # For a pair (e1, e2) at a vertex: they must have same phase (both 1 or both 2).
    # So we union(2*e1, 2*e2) and union(2*e1+1, 2*e2+1).
    
    for v in range(1, n + 1):
        incident = [idx for (u, idx) in adj[v]]
        c0, c1, c2 = color[incident[0]], color[incident[1]], color[incident[2]]
        
        # Find which color appears once
        if c0 == c1:
            # e0 and e1 are pair, e2 is singleton
            e_pair = (incident[0], incident[1])
        elif c0 == c2:
            # e0 and e2 are pair, e1 is singleton
            e_pair = (incident[0], incident[2])
        else:
            # e1 and e2 are pair, e0 is singleton
            e_pair = (incident[1], incident[2])
        
        e1, e2 = e_pair
        union(2 * e1, 2 * e2)
        union(2 * e1 + 1, 2 * e2 + 1)
    
    # Check for contradictions: no edge should have both phase 1 and phase 2
    for e in range(m):
        if find(2 * e) == find(2 * e + 1):
            print("NO")
            return
    
    # Assign phases
    phase = [0] * m
    visited = [False] * (2 * m + 2)
    
    for e in range(m):
        if not visited[2 * e] and not visited[2 * e + 1]:
            root = find(2 * e)
            # Assign phase 1 to all nodes in this component
            stack = [2 * e]
            visited[2 * e] = True
            comp = []
            while stack:
                node = stack.pop()
                comp.append(node)
                # Find neighbors in DSU (this is slow, but we'll optimize)
                # Actually, we need to iterate through all nodes
            # Better: iterate through all nodes and group by root
            # Let's do that instead
    
    # Group nodes by root
    components = {}
    for node in range(2 * m):
        root = find(node)
        if root not in components:
            components[root] = []
        components[root].append(node)
    
    assigned = [False] * m
    for root, nodes in components.items():
        # Check if this component contains both 2*e and 2*e+1 for some e
        # We already checked for contradictions, so this shouldn't happen
        # Assign phase 1 to all nodes in this component
        for node in nodes:
            e = node // 2
            if node % 2 == 0:
                phase[e] = 1
            else:
                phase[e] = 2
            assigned[e] = True

    # Actually, we need to determine which edges are singletons at each vertex
    # But we've already used the pairing info to assign phases 1 and 2
    # The singleton edges should get phase 3
    
    # Wait, I think I'm overcomplicating this. Let me reconsider.
    
    # Actually, the DSU approach tracks which edges get the same phase (1 or 2).
    # But we haven't determined which edges get phase 3.
    
    # Let me re-approach: for each vertex, one edge is singleton (phase 3),

    # If it's a singleton at one endpoint and paired at the other, there's a contradiction.
    # If it's paired at both endpoints, it gets phase 1 or 2 (determined by DSU).
    
    # Let's check for each edge whether it's singleton at both endpoints
    is_singleton = [False] * m
    for v in range(1, n + 1):
        incident = [idx for (u, idx) in adj[v]]

            is_singleton[incident[2]] = True
        elif c0 == c2:
            is_singleton[incident[1]] = True
        else:
            is_singleton[incident[0]] = True

    for e in range(m):

        else:
            singleton_at_u = incident_u[0]

            singleton_at_v = incident_v[2]
        elif cv0 == cv2:
            singleton_at_v = incident_v[1]
        else:
            singleton_at_v = incident_v[0]
        
        if (e == singleton_at_u) != (e == singleton_at_v):
            print("NO")
            return
    
    # Now assign phases
    phase = [0] * m
    for e in range(m):
        if is_singleton[e]:
            phase[e] = 3
    
    # For non-singleton edges, use DSU to assign phase 1 or 2
    # Reset DSU
    parent = list(range(2 * m + 2))
    rank = [0] * (2 * m + 2)
    
    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x
    
    def union(x, y):
        rx, ry = find(x), find(y)
        if rx == ry:

        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
    
    for v in range(1, n + 1):
        incident = [idx for (u, idx) in adj[v]]
        c0, c1, c2 = color[incident[0]], color[incident[1]], color[incident[2]]
        
        if c0 == c1:
            e_pair = (incident[0], incident[1])
        elif c0 == c2:
            e_pair = (incident[0], incident[2])
        else:
            e_pair = (incident[1], incident[2])
        
        e1, e2 = e_pair
        union(2 * e1, 2 * e2)
        union(2 * e1 + 1, 2 * e2 + 1)
    
    # Check for contradictions
    for e in range(m):
        if not is_singleton[e] and find(2 * e) == find(2 * e + 1):
            print("NO")
            return
    
    # Assign phases 1 and 2
    visited_phase = [False] * m
    for e in range(m):
        if not is_singleton[e] and not visited_phase[e]:
            # BFS to find all edges in the same component
            queue = [e]
            visited_phase[e] = True
            component = [e]

                idx += 1
                # Find all edges paired with curr
                for v in range(1, n + 1):
                    incident = [idx2 for (u, idx2) in adj[v]]
                    c0, c1, c2 = color[incident[0]], color[incident[1]], color[incident[2]]
                    if c0 == c1:
                        e1, e2 = incident[0], incident[1]
                    elif c0 == c2:
                        e1, e2 = incident[0], incident[2]
                    else:
                        e1, e2 = incident[1], incident[2]
                    
                    if curr == e1 and not visited_phase[e2]:
                        visited_phase[e2] = True
                        queue.append(e2)
                        component.append(e2)
                    elif curr == e2 and not visited_phase[e1]:
                        visited_phase[e1] = True
                        queue.append(e1)
                        component.append(e1)
            
            # Assign phase 1 to all edges in this component
            for edge in component:
                phase[edge] = 1
    
    print("YES")
    print(' '.join(map(str, phase)))

solve()
