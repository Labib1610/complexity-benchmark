```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # ---- bridge detection (Tarjan) ----
    tin = [-1]*n
    low = [0]*n
    timer = 0
    bridge = [False]*m

    def dfs(v, p_e):
        nonlocal timer
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_e:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridge[eid] = True

    dfs(0, -1)
    if any(b for b in bridge):
        print("NO")
        return

    # ---- Edmonds' Blossom for maximum matching ----
    # implementation based on emaxx.ru / Igor Naverniouk
    N = n
    g = [[] for _ in range(N)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    match = [-1]*N
    p = [-1]*N
    base = list(range(N))
    q = []
    used = [False]*N
    blossom = [False]*N

    def lca(a, b):
        used_path = [False]*N
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal q, used, p, base, blossom
        used = [False]*N
        p = [-1]*N
        base = list(range(N))
        q = []
        q_head = 0
        q.append(root)
        used[root] = True
        while q_head < len(q):
            v = q[q_head]; q_head += 1
            for to in g[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    curbase = lca(v, to)
                    blossom = [False]*N
                    mark_path(v, curbase, to)
                    mark_path(to, curbase, v)
                    for i in range(N):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        # augmenting path found
                        return to
                    to_match = match[to]
                    used[to_match] = True
                    q.append(to_match)
        return -1

    def augment_path(finish):
        v = finish
        while v != -1:
            pv = p[v]
            ppv = match[pv] if pv != -1 else -1
            match[v] = pv
            match[pv] = v
            v = ppv

    for v in range(N):
        if match[v] == -1:
            finish = find_path(v)
            if finish != -1:
                augment_path(finish)

    if any(mch == -1 for mch in match):
        print("NO")
        return

    # ---- assign colors ----
    color = [0]*m
    # color 1 for matching edges
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            color[idx] = 1

    # build remaining graph (edges not colored)
    rem_adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        if color[idx] == 0:
            rem_adj[u].append((v, idx))
            rem_adj[v].append((u, idx))

    visited = [False]*n
    for start in range(n):
        if visited[start]:
            continue
        # traverse each cycle
        if not rem_adj[start]:
            continue
        cur = start
        prev_edge = -1
        cur_color = 2
        while True:
            visited[cur] = True
            # pick next edge not equal to prev_edge
            for to, eid in rem_adj[cur]:
                if eid != prev_edge:
                    next_node = to
                    next_eid = eid
                    break
            color[next_eid] = cur_color
            # move
            prev_edge = next_eid
            cur = next_node
            cur_color = 5 - cur_color  # toggle between 2 and 3
            if cur == start:
                break

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
```
