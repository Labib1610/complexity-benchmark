We can model this as a **3-edge-coloring** problem on a **3-regular graph** (each vertex has degree 3).  

A 3-regular graph has a proper 3-edge-coloring if and only if it is **bridgeless** (by the **Petersen's theorem**).  
So the task reduces to checking if the given graph has any **bridges** (edges whose removal disconnects the graph). If there are no bridges, we can find a 3-edge-coloring.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(1000000)

def find_bridges(n, edges, adj):
    visited = [False] * (n + 1)
    tin = [-1] * (n + 1)
    low = [-1] * (n + 1)
    timer = 0
    bridges = set()

    def dfs(v, p_edge):
        nonlocal timer
        visited[v] = True
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if visited[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    bridges.add(eid)
    for i in range(1, n + 1):
        if not visited[i]:
            dfs(i, -1)
    return bridges

def edge_color_3regular(n, edges, adj):
    # We'll use a simple backtracking with heuristics, but for n<=2000
    # we can use a known constructive method: 
    # Since graph is cubic and bridgeless, it has a perfect matching.
    # Remove perfect matching -> 2-regular graph -> color cycles with 2 colors,
    # then assign third color to matching edges.
    
    # Find perfect matching using simple greedy (works for cubic bridgeless)
    match = [-1] * (n + 1)
    used = [False] * (n + 1)
    for u in range(1, n + 1):
        if match[u] == -1:
            for v, eid in adj[u]:
                if match[v] == -1:
                    match[u] = v
                    match[v] = u
                    used[eid] = True
                    break
    # Now color remaining 2-regular graph (cycles) with colors 1 and 2
    color = [-1] * m
    visited = [False] * (n + 1)
    for start in range(1, n + 1):
        if not visited[start]:
            # traverse cycle
            cycle_nodes = []
            cur = start
            while not visited[cur]:
                visited[cur] = True
                cycle_nodes.append(cur)
                # find next unvisited neighbor not matched
                for nxt, eid in adj[cur]:
                    if not used[eid] and not visited[nxt]:
                        cur = nxt
                        break
                else:
                    break
            # color cycle edges alternately 1,2
            for i in range(len(cycle_nodes)):
                u = cycle_nodes[i]
                v = cycle_nodes[(i + 1) % len(cycle_nodes)]
                for eid, (a, b) in enumerate(edges):
                    if (a == u and b == v) or (a == v and b == u):
                        color[eid] = 1 if i % 2 == 0 else 2
                        break
    # assign color 3 to matching edges
    for eid, (u, v) in enumerate(edges):
        if used[eid]:
            color[eid] = 3
    return color

def main():
    input_data = sys.stdin.read().strip().split()
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for eid in range(m):
        u = int(input_data[idx]); v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append((v, eid))
        adj[v].append((u, eid))

    bridges = find_bridges(n, edges, adj)
    if bridges:
        print("NO")
        return

    color = edge_color_3regular(n, edges, adj)
    print("YES")
    print(" ".join(str(c) for c in color))

if __name__ == "__main__":
    main()
```
