import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)

    # Variables: For each edge i (0-indexed) and color c (0, 1, 2):
    # var index = i * 3 + c + 1  (1-based, 1 to 3m)
    def var_of(edge_idx, color):
        return edge_idx * 3 + color + 1

    clauses = []

    # 1. Each edge has at least one color, and at most one color
    for i in range(m):
        v0 = var_of(i, 0)
        v1 = var_of(i, 1)
        v2 = var_of(i, 2)
        clauses.append([v0, v1, v2])
        clauses.append([-v0, -v1])
        clauses.append([-v0, -v2])
        clauses.append([-v1, -v2])

    # 2. For each vertex, incident edges have distinct colors
    for u in range(1, n + 1):
        inc = adj[u]
        if len(inc) != 3:
            # If not 3-regular, can't be valid
            print("NO")
            return
        e0, e1, e2 = inc
        for c in range(3):
            v0 = var_of(e0, c)
            v1 = var_of(e1, c)
            v2 = var_of(e2, c)
            clauses.append([v0, v1, v2])
            clauses.append([-v0, -v1])
            clauses.append([-v0, -v2])
            clauses.append([-v1, -v2])

    # Symmetry breaking at vertex 1
    e0, e1, e2 = adj[1]
    clauses.append([var_of(e0, 0)])
    clauses.append([var_of(e1, 1)])
    clauses.append([var_of(e2, 2)])

    num_vars = 3 * m

    # CDCL SAT Solver
    # Watches: watches[lit] is a list of clauses where `lit` is watched.
    # lit ranges from -num_vars to +num_vars.
    # Array indexing: pos lit -> lit, neg lit -> num_vars + (-lit)
    def lit_to_idx(lit):
        return lit if lit > 0 else num_vars - lit

    watches = [[] for _ in range(2 * num_vars + 2)]
    
    # Store clauses as list of lists
    clause_db = []
    for c in clauses:
        if len(c) == 1:
            clause_db.append(c)
        else:
            clause_db.append(c)
            watches[lit_to_idx(c[0])].append(len(clause_db) - 1)
            watches[lit_to_idx(c[1])].append(len(clause_db) - 1)

    val = [0] * (num_vars + 1)
    level = [0] * (num_vars + 1)
    reason = [None] * (num_vars + 1)
    trail = []
    trail_lim = []

    activity = [0.0] * (num_vars + 1)
    var_inc = 1.0
    decay = 0.95

    for c in clauses:
        for lit in c:
            activity[abs(lit)] += 1.0

    def get_lit_val(lit):
        v = abs(lit)
        if val[v] == 0:
            return 0
        return 1 if (val[v] == 1 and lit > 0) or (val[v] == -1 and lit < 0) else -1

    def assign_val(lit, r):
        v = abs(lit)
        val[v] = 1 if lit > 0 else -1
        level[v] = len(trail_lim)
        reason[v] = r
        trail.append(v)

    def propagate():
        q_head = propagate.head
        while q_head < len(trail):
            p = trail[q_head]
            q_head += 1
            false_lit = -p if val[p] == 1 else p
            f_idx = lit_to_idx(false_lit)
            
            i = 0
            w_list = watches[f_idx]
            while i < len(w_list):
                c_idx = w_list[i]
                c = clause_db[c_idx]
                if c[0] == false_lit:
                    c[0], c[1] = c[1], c[0]
                
                if get_lit_val(c[0]) == 1:
                    i += 1
                    continue
                
                found_new_watch = False
                for k in range(2, len(c)):
                    if get_lit_val(c[k]) != -1:
                        c[1], c[k] = c[k], c[1]
                        watches[lit_to_idx(c[1])].append(c_idx)
                        w_list[i] = w_list[-1]
                        w_list.pop()
                        found_new_watch = True
                        break
                
                if not found_new_watch:
                    first_val = get_lit_val(c[0])
                    if first_val == -1:
                        propagate.head = q_head
                        return c_idx
                    elif first_val == 0:
                        assign_val(c[0], c_idx)
                    i += 1
        propagate.head = q_head
        return -1

    propagate.head = 0

    # Initial unit propagation
    for c_idx, c in enumerate(clause_db):
        if len(c) == 1:
            lv = get_lit_val(c[0])
            if lv == -1:
                print("NO")
                return
            elif lv == 0:
                assign_val(c[0], None)

    if propagate() != -1:
        print("NO")
        return

    def backtrack(to_level):
        while trail_lim and len(trail_lim) > to_level:
            limit = trail_lim.pop()
            while len(trail) > limit:
                v = trail.pop()
                val[v] = 0
                reason[v] = None
        propagate.head = len(trail)

    import heapq
    heap = [(-activity[i], i) for i in range(1, num_vars + 1)]
    heapq.heapify(heap)

    def pick_branching_var():
        while heap:
            act, v = heapq.heappop(heap)
            if val[v] == 0:
                return v
        return 0

    # Main CDCL loop
    while True:
        conflict_c_idx = propagate()
        if conflict_c_idx != -1:
            if not trail_lim:
                print("NO")
                return
            
            c = clause_db[conflict_c_idx]
            cur_level = len(trail_lim)
            
            seen = set()
            learnt = []
            path_cnt = 0
            idx_in_trail = len(trail) - 1
            p = 0
            
            for lit in c:
                v = abs(lit)
                if level[v] == cur_level:
                    seen.add(v)
                    path_cnt += 1
                elif level[v] > 0:
                    learnt.append(lit)

            while path_cnt > 0:
                while trail[idx_in_trail] not in seen:
                    idx_in_trail -= 1
                p = trail[idx_in_trail]
                seen.remove(p)
                path_cnt -= 1
                idx_in_trail -= 1

                if path_cnt == 0:
                    break

                r_idx = reason[p]
                r_clause = clause_db[r_idx]
                for lit in r_clause:
                    v = abs(lit)
                    if v != p and v not in seen:
                        if level[v] == cur_level:
                            seen.add(v)
                            path_cnt += 1
                        elif level[v] > 0:
                            learnt.append(lit)

            p_lit = p if val[p] == -1 else -p
            learnt = [p_lit] + learnt
            
            backtrack_level = 0
            if len(learnt) > 1:
                max_i = 1
                max_lvl = level[abs(learnt[1])]
                for i in range(2, len(learnt)):
                    lvl = level[abs(learnt[i])]
                    if lvl > max_lvl:
                        max_lvl = lvl
                        max_i = i
                learnt[1], learnt[max_i] = learnt[max_i], learnt[1]
                backtrack_level = max_lvl

            backtrack(backtrack_level)

            new_c_idx = len(clause_db)
            clause_db.append(learnt)
            if len(learnt) > 1:
                watches[lit_to_idx(learnt[0])].append(new_c_idx)
                watches[lit_to_idx(learnt[1])].append(new_c_idx)
            
            assign_val(learnt[0], new_c_idx)

            var_inc /= decay
            for lit in learnt:
                v = abs(lit)
                activity[v] += var_inc
                heapq.heappush(heap, (-activity[v], v))

        else:
            next_var = pick_branching_var()
            if next_var == 0:
                # SAT!
                ans = []
                for i in range(m):
                    for c in range(3):
                        v = var_of(i, c)
                        if val[v] == 1:
                            ans.append(c + 1)
                            break
                print("YES")
                print(*(ans))
                return
            
            trail_lim.append(len(trail))
            assign_val(next_var, None)

if __name__ == '__main__':
    solve()
