import sys
sys.setrecursionlimit(10000)

def solve():
    import sys
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    adj = [[] for _ in range(n)]
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1; v -= 1
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)

    # Blossom algorithm for maximum matching in general graph
    N = n
    g = [[] for _ in range(N)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    match = [-1]*N
    p = [-1]*N
    base = list(range(N))
    q = []
    used = [False]*N
    blossom = [False]*N

    def lca(a, b):
        used_path = [False]*N
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, x):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = x
            x = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal q, used, p, base, blossom
        used = [False]*N
        p = [-1]*N
        base = list(range(N))
        q = []
        q_head = 0
        q.append(root)
        used[root] = True
        while q_head < len(q):
            v = q[q_head]; q_head += 1
            for to in g[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    cur = lca(v, to)
                    blossom = [False]*N
                    mark_path(v, cur, to)
                    mark_path(to, cur, v)
                    for i in range(N):
                        if blossom[base[i]]:
                            base[i] = cur
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        # augmenting
                        v = to
                        while v != -1:
                            pv = p[v]
                            nv = match[pv] if pv != -1 else -1
                            match[v] = pv
                            match[pv] = v
                            v = nv
                        return True
                    else:
                        used[match[to]] = True
                        q.append(match[to])
        return False

    for v in range(N):
        if match[v] == -1:
            find_path(v)

    if any(match[i]==-1 for i in range(N)):
        print("NO")
        return

    # match is perfect matching
    color = [0]*m
    # assign color 1 to matching edges
    edge_id = {}
    for idx, (u,v) in enumerate(edges):
        edge_id[(u,v)] = edge_id[(v,u)] = idx
    for u in range(N):
        v = match[u]
        if u < v:
            idx = edge_id[(u,v)]
            color[idx] = 1

    # remaining edges form cycles, color them 2 and 3 alternating
    visited = [False]*N
    for start in range(N):
        if visited[start]:
            continue
        # traverse component of remaining edges
        cur = start
        prev = -1
        cycle_edges = []
        while True:
            visited[cur] = True
            # find next neighbor not in matching and not prev
            nxt = None
            for to in adj[cur]:
                if (cur,to) in edge_id:
                    idx = edge_id[(cur,to)]
                else:
                    idx = edge_id[(to,cur)]
                if color[idx]==0 and to!=prev:
                    nxt = to
                    eidx = idx
                    break
            if nxt is None:
                break
            cycle_edges.append(eidx)
            prev, cur = cur, nxt
            if cur==start:
                break
        # color alternating 2,3
        for i, eid in enumerate(cycle_edges):
            color[eid] = 2 if i%2==0 else 3

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
