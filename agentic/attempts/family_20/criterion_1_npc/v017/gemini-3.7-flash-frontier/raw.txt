import sys

# Increase recursion depth just in case
sys.setrecursionlimit(20000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj_v = [[] for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj_v[u].append(i)
        adj_v[v].append(i)
        
    # Each edge has domain as a bitmask: 1 (color 1), 2 (color 2), 4 (color 3)
    # Total mask is 7 (1 | 2 | 4)
    # Color values: 1 -> mask 1, 2 -> mask 2, 3 -> mask 4
    
    color = [0] * m  # 0 means unassigned, 1, 2, or 3
    domain = [7] * m
    
    # Precompute incident edges at both endpoints of each edge
    # For edge e=(u,v), incident edges at u are u_e1, u_e2, and at v are v_e1, v_e2
    incident_edges = [[] for _ in range(m)]
    for i in range(m):
        u, v = edges[i]
        inc = []
        for e in adj_v[u]:
            if e != i:
                inc.append(e)
        for e in adj_v[v]:
            if e != i:
                inc.append(e)
        incident_edges[i] = inc

    # Vertices incident to each edge
    # and edges incident to each vertex
    # adj_v[u] has exactly 3 edges (or fewer if disconnected/not 3-regular, but problem says cubic)

    # Unit propagation queue
    def propagate():
        # Repeatedly apply deduction rules until fixpoint or conflict
        changed = True
        while changed:
            changed = False
            # Check edge domains
            for e in range(m):
                d = domain[e]
                if d == 0:
                    return False
                if d == 1 or d == 2 or d == 4:
                    # Single value
                    c = 1 if d == 1 else (2 if d == 2 else 3)
                    if color[e] == 0:
                        color[e] = c
                        changed = True
                        # Remove c's bit from neighbors
                        bit = d
                        for ne in incident_edges[e]:
                            if domain[ne] & bit:
                                domain[ne] &= ~bit
                                if domain[ne] == 0:
                                    return False
                                changed = True
            
            # Check vertex bijection constraints
            for u in range(1, n + 1):
                e_list = adj_v[u]
                # For each color mask (1, 2, 4), count which edges in e_list can take it
                for bit in (1, 2, 4):
                    possible = [e for e in e_list if (domain[e] & bit)]
                    if len(possible) == 0:
                        return False
                    if len(possible) == 1:
                        e = possible[0]
                        if domain[e] != bit:
                            domain[e] = bit
                            changed = True
        return True

    # Backtracking search with state saving
    def search():
        # Propagate initial state / current deductions
        # Find an unassigned edge with minimum domain size (MRV)
        best_e = -1
        min_size = 4
        
        for e in range(m):
            if color[e] == 0:
                sz = bin(domain[e]).count('1')
                if sz < min_size:
                    min_size = sz
                    best_e = e
                    if min_size <= 1:
                        break
        
        if best_e == -1:
            return True  # All edges colored
        
        # Try possible colors for best_e
        possible_masks = []
        for bit in (1, 2, 4):
            if domain[best_e] & bit:
                possible_masks.append(bit)
                
        # Save state
        saved_color = list(color)
        saved_domain = list(domain)
        
        for bit in possible_masks:
            domain[best_e] = bit
            if propagate():
                if search():
                    return True
            # Restore state
            for i in range(m):
                color[i] = saved_color[i]
                domain[i] = saved_domain[i]
                
        return False

    if propagate() and search():
        print("YES")
        print(*(color[i] for i in range(m)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
