```python
import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    debates = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        debates.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Each participant has exactly 3 debates
    # Each debate involves exactly 2 participants
    # We need to assign each debate to one of 3 slots such that
    # for each participant, its 3 debates are in 3 different slots
    
    # This is equivalent to edge coloring of a graph where each vertex has degree 3
    # with 3 colors. This is always possible for cubic graphs (3-regular graphs).
    
    # We can model this as a graph coloring problem on the line graph
    # But since each vertex has degree 3, we can use a greedy approach
    
    # For each vertex, we have 3 edges, and we need to assign each edge a color
    # from {1,2,3} such that all 3 edges have different colors
    
    color = [0] * m  # color[i] = color of debate i (1-based)
    
    # For each vertex, we track which colors are used by its edges
    # Initially, no colors are used
    
    for u in range(1, n + 1):
        edges = adj[u]
        if len(edges) != 3:
            print("NO")
            return
            
        used_colors = [False] * 4  # index 0 unused, 1-3 for colors
        
        # First, see which edges already have colors
        uncolored = []
        for e_idx in edges:
            if color[e_idx] != 0:
                used_colors[color[e_idx]] = True
            else:
                uncolored.append(e_idx)
        
        # Assign available colors to uncolored edges
        c = 1
        for e_idx in uncolored:
            while used_colors[c]:
                c += 1
            color[e_idx] = c
            used_colors[c] = True
    
    # Verify that each vertex has all 3 colors
    for u in range(1, n + 1):
        edges = adj[u]
        colors = set()
        for e_idx in edges:
            colors.add(color[e_idx])
        if len(colors) != 3 or any(c == 0 for c in colors):
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
```
