```python
import sys
import random
import time

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Quick check: 3-regular graph with a bridge is not 3-edge-colorable
    # Check bridges using Tarjan's bridge-finding algorithm
    tin = [0] * (n + 1)
    low = [0] * (n + 1)
    timer = 0
    visited = [False] * (n + 1)
    has_bridge = False

    def dfs_bridge(v, p_edge):
        nonlocal timer, has_bridge
        visited[v] = True
        timer += 1
        tin[v] = low[v] = timer
        for e_idx in adj[v]:
            if e_idx == p_edge:
                continue
            u = edges[e_idx][0] if edges[e_idx][1] == v else edges[e_idx][1]
            if visited[u]:
                low[v] = min(low[v], tin[u])
            else:
                dfs_bridge(u, e_idx)
                low[v] = min(low[v], low[u])
                if low[u] > tin[v]:
                    has_bridge = True

    for i in range(1, n + 1):
        if not visited[i]:
            dfs_bridge(i, -1)

    if has_bridge:
        print("NO")
        return

    # Solve connected components or the whole graph
    start_time = time.time()
    
    # Try Kempe-chain based coloring with restarts
    max_restarts = 15
    for restart in range(max_restarts):
        if time.time() - start_time > 4.5:
            break
            
        color = [-1] * m
        # vertex_color[u][c] = edge index with color c at vertex u, or -1
        vertex_color = [[-1] * 3 for _ in range(n + 1)]
        
        uncolored = list(range(m))
        random.shuffle(uncolored)
        
        steps = 0
        max_steps = m * 50
        
        while uncolored and steps < max_steps:
            steps += 1
            e_idx = uncolored.pop()
            u, v = edges[e_idx]
            
            # Find missing colors at u and v
            missing_u = [c for c in range(3) if vertex_color[u][c] == -1]
            missing_v = [c for c in range(3) if vertex_color[v][c] == -1]
            
            common = set(missing_u) & set(missing_v)
            if common:
                c = random.choice(list(common))
                color[e_idx] = c
                vertex_color[u][c] = e_idx
                vertex_color[v][c] = e_idx
                continue
            
            # Try to find a pair (a, b) such that the (a, b) path from u doesn't reach v
            resolved = False
            pairs = [(a, b) for a in missing_u for b in missing_v]
            random.shuffle(pairs)
            
            for a, b in pairs:
                # Trace (a, b) alternating path starting at u
                # Since 'a' is missing at u, the path from u must start with color 'b' (if present at u)
                # But 'a' is missing at u, so path starting from u with color 'b':
                path_edges = []
                curr = u
                c_curr = b
                while True:
                    next_edge = vertex_color[curr][c_curr]
                    if next_edge == -1:
                        break
                    path_edges.append((next_edge, c_curr))
                    nxt = edges[next_edge][0] if edges[next_edge][1] == curr else edges[next_edge][1]
                    curr = nxt
                    c_curr = a if c_curr == b else b
                
                if curr != v:
                    # Path does not reach v! Swap colors along the path
                    for pe, old_c in path_edges:
                        new_c = a if old_c == b else b
                        color[pe] = new_c
                        pu, pv = edges[pe]
                        vertex_color[pu][old_c] = -1
                        vertex_color[pv][old_c] = -1
                        vertex_color[pu][new_c] = pe
                        vertex_color[pv][new_c] = pe
                    
                    # Now color b is missing at u, and b was already missing at v
                    color[e_idx] = b
                    vertex_color[u][b] = e_idx
                    vertex_color[v][b] = e_idx
                    resolved = True
                    break
            
            if not resolved:
                # Pick a random color c at u, uncolor that edge, and color e_idx with c
                c = random.randint(0, 2)
                evicted = vertex_color[u][c]
                if evicted != -1:
                    eu, ev = edges[evicted]
                    vertex_color[eu][c] = -1
                    vertex_color[ev][c] = -1
                    color[evicted] = -1
                    uncolored.append(evicted)
                
                # If v also had color c, uncolor it too
                evicted_v = vertex_color[v][c]
                if evicted_v != -1 and evicted_v != evicted:
                    eu, ev = edges[evicted_v]
                    vertex_color[eu][c] = -1
                    vertex_color[ev][c] = -1
                    color[evicted_v] = -1
                    uncolored.append(evicted_v)
                
                color[e_idx] = c
                vertex_color[u][c] = e_idx
                vertex_color[v][c] = e_idx

        if not uncolored:
            print("YES")
            print(*(c + 1 for c in color))
            return

    print("NO")

if __name__ == '__main__':
    solve()
```
