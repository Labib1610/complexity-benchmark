import sys
sys.setrecursionlimit(10000)

def blossom_maximum_matching(n, adj):
    # implementation based on emaxx.ru
    INF = n + 5
    match = [0] * (n + 1)
    p = [0] * (n + 1)
    base = list(range(n + 1))
    q = []
    used = [False] * (n + 1)
    blossom = [False] * (n + 1)

    def lca(a, b):
        used_path = [False] * (n + 1)
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == 0:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal q, used, p, base, blossom
        used = [False] * (n + 1)
        p = [0] * (n + 1)
        for i in range(1, n + 1):
            base[i] = i
        q = []
        q_head = 0
        q.append(root)
        used[root] = True
        while q_head < len(q):
            v = q[q_head]
            q_head += 1
            for to in adj[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != 0 and p[match[to]] != 0):
                    cur = lca(v, to)
                    blossom = [False] * (n + 1)
                    mark_path(v, cur, to)
                    mark_path(to, cur, v)
                    for i in range(1, n + 1):
                        if blossom[base[i]]:
                            base[i] = cur
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == 0:
                    p[to] = v
                    if match[to] == 0:
                        # augmenting path found
                        cur = to
                        while cur != 0:
                            prev = p[cur]
                            nxt = match[prev]
                            match[cur] = prev
                            match[prev] = cur
                            cur = nxt
                        return True
                    else:
                        used[match[to]] = True
                        q.append(match[to])
        return False

    for v in range(1, n + 1):
        if match[v] == 0:
            find_path(v)
    return match

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    adj = [[] for _ in range(n + 1)]
    for idx in range(m):
        u = int(next(it)); v = int(next(it))
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)

    # find maximum matching
    match = blossom_maximum_matching(n, adj)

    # check perfect matching
    if any(match[i] == 0 for i in range(1, n + 1)):
        print("NO")
        return

    # assign colors
    color = [0] * m
    matched_edge = {}
    for i, (u, v) in enumerate(edges):
        if match[u] == v:
            color[i] = 1
            matched_edge[(u, v)] = matched_edge[(v, u)] = True

    # build graph of remaining edges (degree 2)
    rem_adj = [[] for _ in range(n + 1)]
    edge_id = {}
    for i, (u, v) in enumerate(edges):
        if color[i] == 0:
            rem_adj[u].append((v, i))
            rem_adj[v].append((u, i))
            edge_id[(u, v)] = edge_id[(v, u)] = i

    visited = [False] * (n + 1)
    for start in range(1, n + 1):
        if visited[start]:
            continue
        # traverse cycle
        cur = start
        prev = -1
        cur_color = 2
        while True:
            visited[cur] = True
            # find next edge not coming from prev
            nxt = None
            nxt_id = None
            for to, eid in rem_adj[cur]:
                if eid == prev:
                    continue
                nxt = to
                nxt_id = eid
                break
            if nxt is None:
                break
            color[nxt_id] = cur_color
            cur_color = 5 - cur_color  # toggle between 2 and 3
            prev = nxt_id
            cur = nxt
            if cur == start:
                break

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
