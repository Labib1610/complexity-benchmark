import sys

sys.setrecursionlimit(10000)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    adj = [[] for _ in range(n)]
    edges = []

    pos = 2
    valid = (m * 2 == 3 * n and len(data) == 2 + 2 * m)

    for eid in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if not (0 <= u < n and 0 <= v < n and u != v):
            valid = False
        edges.append((u, v))
        if 0 <= u < n and 0 <= v < n:
            adj[u].append((v, eid))
            adj[v].append((u, eid))

    if not valid or any(len(adj[v]) != 3 for v in range(n)):
        print("NO")
        return

    # A cubic graph containing a bridge cannot be 3-edge-coloured.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    roots = []
    has_bridge = False

    def dfs(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1

        for to, eid in adj[v]:
            if eid == parent_edge:
                continue
            if tin[to] == -1:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for v in range(n):
        if tin[v] == -1:
            roots.append(v)
            dfs(v, -1)

    if has_bridge:
        print("NO")
        return

    # Exact cover:
    # - one column for every edge (choose exactly one colour);
    # - one column for every (vertex, colour) pair.
    # A row represents assigning a colour to an edge.
    column_count = m + 3 * n

    L = list(range(column_count + 1))
    R = list(range(column_count + 1))
    U = list(range(column_count + 1))
    D = list(range(column_count + 1))
    C = list(range(column_count + 1))
    S = [0] * (column_count + 1)
    row_id = [-1] * (column_count + 1)

    for c in range(column_count + 1):
        L[c] = c - 1
        R[c] = c + 1
    L[0] = column_count
    R[column_count] = 0

    row_node = [-1] * (3 * m)

    def add_node(col, rid):
        idx = len(L)
        L.append(idx)
        R.append(idx)
        U.append(U[col])
        D.append(col)
        C.append(col)
        S.append(0)
        row_id.append(rid)

        D[U[col]] = idx
        U[col] = idx
        S[col] += 1
        return idx

    for eid, (u, v) in enumerate(edges):
        for colour in range(3):
            rid = 3 * eid + colour
            cols = (
                1 + eid,
                1 + m + 3 * u + colour,
                1 + m + 3 * v + colour
            )

            nodes = [add_node(col, rid) for col in cols]
            for i in range(3):
                L[nodes[i]] = nodes[(i - 1) % 3]
                R[nodes[i]] = nodes[(i + 1) % 3]
            row_node[rid] = nodes[0]

    answer = [0] * m

    def cover(col):
        R[L[col]] = R[col]
        L[R[col]] = L[col]

        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                D[U[j]] = D[j]
                U[D[j]] = U[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                D[U[j]] = j
                U[D[j]] = j
                j = L[j]
            i = U[i]

        R[L[col]] = col
        L[R[col]] = col

    # Break the global colour symmetry independently in every connected component.
    for root in roots:
        incident = [eid for _, eid in adj[root]]
        for colour, eid in enumerate(incident):
            rid = 3 * eid + colour
            node = row_node[rid]
            answer[eid] = colour + 1

            cover(C[node])
            j = R[node]
            while j != node:
                cover(C[j])
                j = R[j]

    def search():
        if R[0] == 0:
            return True

        col = R[0]
        best = col
        best_size = S[col]

        col = R[col]
        while col != 0:
            size = S[col]
            if size < best_size:
                best = col
                best_size = size
                if size == 0:
                    break
            col = R[col]

        if best_size == 0:
            return False

        cover(best)
        r = D[best]

        while r != best:
            rid = row_id[r]
            eid = rid // 3
            colour = rid % 3
            answer[eid] = colour + 1

            j = R[r]
            while j != r:
                cover(C[j])
                j = R[j]

            if search():
                return True

            j = L[r]
            while j != r:
                uncover(C[j])
                j = L[j]

            answer[eid] = 0
            r = D[r]

        uncover(best)
        return False

    if search():
        print("YES")
        print(*answer)
    else:
        print("NO")


if __name__ == "__main__":
    main()
