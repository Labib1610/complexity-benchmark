import sys

# Increase recursion depth
sys.setrecursionlimit(100000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    edge_color = [-1] * m
    vertex_colors = [[False] * 4 for _ in range(n + 1)]
    
    # Fix the 3 edges incident to vertex 1 to break color symmetry
    v1_edges = adj[1]
    edge_color[v1_edges[0]] = 1
    vertex_colors[1][1] = True
    u0 = edges[v1_edges[0]][0] if edges[v1_edges[0]][1] == 1 else edges[v1_edges[0]][1]
    vertex_colors[u0][1] = True
    
    edge_color[v1_edges[1]] = 2
    vertex_colors[1][2] = True
    u1 = edges[v1_edges[1]][0] if edges[v1_edges[1]][1] == 1 else edges[v1_edges[1]][1]
    vertex_colors[u1][2] = True
    
    edge_color[v1_edges[2]] = 3
    vertex_colors[1][3] = True
    u2 = edges[v1_edges[2]][0] if edges[v1_edges[2]][1] == 1 else edges[v1_edges[2]][1]
    vertex_colors[u2][3] = True

    # Pre-calculate adjacent edges for each edge
    edge_adj = [[] for _ in range(m)]
    for i in range(m):
        u, v = edges[i]
        for e in adj[u]:
            if e != i:
                edge_adj[i].append(e)
        for e in adj[v]:
            if e != i:
                edge_adj[i].append(e)

    # Backtracking with MRV
    # We can maintain available colors for each edge
    # available colors can be computed quickly from u and v
    
    def get_avail(e):
        u, v = edges[e]
        vc_u = vertex_colors[u]
        vc_v = vertex_colors[v]
        avail = []
        for c in (1, 2, 3):
            if not vc_u[c] and not vc_v[c]:
                avail.append(c)
        return avail

    def backtrack():
        # Find uncolored edge with minimum available colors
        best_e = -1
        min_choices = 4
        best_avail = None
        
        for e in range(m):
            if edge_color[e] == -1:
                avail = get_avail(e)
                k = len(avail)
                if k == 0:
                    return False
                if k < min_choices:
                    min_choices = k
                    best_e = e
                    best_avail = avail
                    if k == 1:
                        break
        
        if best_e == -1:
            return True
        
        u, v = edges[best_e]
        for c in best_avail:
            edge_color[best_e] = c
            vertex_colors[u][c] = True
            vertex_colors[v][c] = True
            
            if backtrack():
                return True
                
            edge_color[best_e] = -1
            vertex_colors[u][c] = False
            vertex_colors[v][c] = False
            
        return False

    if backtrack():
        print("YES")
        print(*(edge_color[i] for i in range(m)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
