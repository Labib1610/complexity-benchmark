import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    import sys
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    g = [[] for _ in range(n)]
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        g[u].append((v, i))
        g[v].append((u, i))

    # ---------- bridge check ----------
    tin = [-1] * n
    low = [0] * n
    timer = 0
    is_bridge = [False] * m

    def dfs(v, p_e):
        nonlocal timer
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in g[v]:
            if eid == p_e:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    is_bridge[eid] = True

    dfs(0, -1)
    if any(is_bridge):
        print("NO")
        return

    # ---------- Edmonds blossom for maximum matching ----------
    # implementation based on https://github.com/jaehyunp/stanfordacm/blob/master/code/EdmondsMatching.cc
    N = n
    adj = [set() for _ in range(N)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)

    match = [-1] * N
    p = [-1] * N
    base = list(range(N))
    q = []
    in_blossom = [False] * N
    in_queue = [False] * N

    def lca(a, b):
        used = [False] * N
        while True:
            a = base[a]
            used[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, child):
        while base[v] != b:
            in_blossom[base[v]] = in_blossom[base[match[v]]] = True
            p[v] = child
            child = match[v]
            v = p[match[v]]

    def blossom_contraction(v, u):
        b = lca(v, u)
        in_blossom[:] = [False] * N
        mark_path(v, b, u)
        mark_path(u, b, v)
        for i in range(N):
            if in_blossom[base[i]]:
                base[i] = b
                if not in_queue[i]:
                    q.append(i)
                    in_queue[i] = True

    def bfs(root):
        nonlocal q, p, base, in_queue
        p[:] = [-1] * N
        in_queue[:] = [False] * N
        base = list(range(N))
        q = [root]
        in_queue[root] = True
        qi = 0
        while qi < len(q):
            v = q[qi]
            qi += 1
            for to in adj[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    blossom_contraction(v, to)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        # augmenting path found
                        cur = to
                        while cur != -1:
                            prev = p[cur]
                            nxt = match[prev] if prev != -1 else -1
                            match[cur] = prev
                            match[prev] = cur
                            cur = nxt
                        return True
                    else:
                        q.append(match[to])
                        in_queue[match[to]] = True
        return False

    for v in range(N):
        if match[v] == -1:
            bfs(v)

    if any(mch == -1 for mch in match):
        print("NO")
        return

    # ---------- build coloring ----------
    color = [0] * m
    # edges of matching get color 1
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            color[idx] = 1

    # remaining edges form a 2-regular graph
    # we need to 2‑color its edges (colors 2 and 3)
    # each component is a cycle; we can traverse and color alternately
    used_edge = [c == 1 for c in color]
    for start in range(n):
        # find a vertex with unused incident edge
        cur = start
        while True:
            # find unused edge
            nxt_edge = None
            for to, eid in g[cur]:
                if not used_edge[eid]:
                    nxt_edge = (to, eid)
                    break
            if nxt_edge is None:
                break
            # traverse the cycle
            col = 2
            v = cur
            while True:
                to, eid = nxt_edge
                color[eid] = col
                used_edge[eid] = True
                col = 5 - col  # toggle between 2 and 3
                v = to
                # find next edge at v
                nxt = None
                for w, ee in g[v]:
                    if not used_edge[ee]:
                        nxt = (w, ee)
                        break
                if nxt is None:
                    break
                nxt_edge = nxt
                if v == cur:
                    break
            # end of one cycle
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
