We can model this as a **3-edge-coloring** problem on a **cubic graph** (every vertex has degree 3).  

By **Kőnig’s line coloring theorem**, every cubic graph is 3-edge-colorable if it is **bridgeless** (no bridges).  
If the graph has a bridge, it may still be 3-edge-colorable, but not always.  
However, for this problem, we can attempt a constructive approach using **BFS/DFS** to assign colors while checking constraints.

Here’s the Python solution:

```python
import sys
sys.setrecursionlimit(10000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))
    
    color = [-1] * m
    
    # Try to color edges using backtracking with constraint propagation
    # Since n <= 2000, m <= 3000, we can use a simple backtracking with pruning.
    
    # We'll use a greedy + backtracking approach.
    # At each vertex, the three incident edges must get distinct colors.
    
    # Precompute edge lists per vertex
    edge_list = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        edge_list[u].append(idx)
        edge_list[v].append(idx)
    
    # Try to assign colors
    def backtrack(eidx):
        if eidx == m:
            return True
        u, v = edges[eidx]
        used_u = {color[i] for i in edge_list[u] if color[i] != -1}
        used_v = {color[i] for i in edge_list[v] if color[i] != -1}
        for c in [1, 2, 3]:
            if c not in used_u and c not in used_v:
                color[eidx] = c
                if backtrack(eidx + 1):
                    return True
                color[eidx] = -1
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
