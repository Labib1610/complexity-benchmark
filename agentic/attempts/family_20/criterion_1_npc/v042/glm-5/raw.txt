import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    color = [0] * m
    visited = [False] * (n + 1)
    parent_edge = [-1] * (n + 1)
    parent_node = [0] * (n + 1)
    depth = [0] * (n + 1)
    
    setrecursionlimit(max(10000, n + 100))
    
    def dfs(root):
        stack = [(root, -1, 0, 0)]
        while stack:
            node, p_edge, p_node, d = stack.pop()
            if visited[node]:
                continue
            visited[node] = True
            parent_edge[node] = p_edge
            parent_node[node] = p_node
            depth[node] = d
            for (nb, eid) in adj[node]:
                if not visited[nb]:
                    stack.append((nb, eid, node, d + 1))
    
    for r in range(1, n + 1):
        if not visited[r]:
            dfs(r)
    
    back_edges = []
    for i, (u, v) in enumerate(edges):
        if parent_edge[u] == i or parent_edge[v] == i:
            continue
        if depth[u] < depth[v]:
            back_edges.append((v, u, i))
        else:
            back_edges.append((u, v, i))
    
    for (u, v, eid) in back_edges:
        a, b = u, v
        while a != b:
            if depth[a] >= depth[b]:
                if color[parent_edge[a]] == 0:
                    color[parent_edge[a]] = 1
                a = parent_node[a]
            else:
                if color[parent_edge[b]] == 0:
                    color[parent_edge[b]] = 1
                b = parent_node[b]
        if color[eid] == 0:
            color[eid] = 1
    
    for i in range(m):
        if color[i] == 0:
            color[i] = 2
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
