import sys

# Increase recursion depth just in case
sys.setrecursionlimit(20000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    edges = []
    adj = [[] for _ in range(n + 1)]

    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)

    # Check if all degrees are 3
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return

    edge_color = [0] * m
    vertex_used = [0] * (n + 1)  # bitmask of used colors: bit 1, 2, 3 (1<<1, 1<<2, 1<<3)

    # Find connected components of edges
    visited_v = [False] * (n + 1)
    components = []

    for i in range(1, n + 1):
        if not visited_v[i]:
            comp_v = []
            q = [i]
            visited_v[i] = True
            for u in q:
                comp_v.append(u)
                for eid in adj[u]:
                    v = edges[eid][0] if edges[eid][1] == u else edges[eid][1]
                    if not visited_v[v]:
                        visited_v[v] = True
                        q.append(v)
            components.append(comp_v)

    # MRV Backtracking solver for a component
    def get_domain(eid):
        u, v = edges[eid]
        used = vertex_used[u] | vertex_used[v]
        # available colors from {1, 2, 3}
        # mask of 1, 2, 3 is (1<<1) | (1<<2) | (1<<3) = 2 | 4 | 8 = 14
        avail = 14 & (~used)
        return avail

    def count_bits(mask):
        return bin(mask).count("1")

    def backtrack(uncolored_edges):
        if not uncolored_edges:
            return True

        # Find edge with minimum domain size
        min_size = 4
        best_e = -1
        best_domain = 0

        # We also check for domain size 0 (conflict) or 1 (forced)
        for e in uncolored_edges:
            dom = get_domain(e)
            sz = count_bits(dom)
            if sz == 0:
                return False
            if sz < min_size:
                min_size = sz
                best_e = e
                best_domain = dom
                if sz == 1:
                    break

        uncolored_edges.remove(best_e)
        u, v = edges[best_e]

        for c in (1, 2, 3):
            if best_domain & (1 << c):
                edge_color[best_e] = c
                vertex_used[u] |= 1 << c
                vertex_used[v] |= 1 << c

                if backtrack(uncolored_edges):
                    return True

                vertex_used[u] &= ~(1 << c)
                vertex_used[v] &= ~(1 << c)
                edge_color[best_e] = 0

        uncolored_edges.add(best_e)
        return False

    for comp_v in components:
        if len(comp_v) % 2 != 0:
            print("NO")
            return

        comp_edges = set()
        for u in comp_v:
            for eid in adj[u]:
                comp_edges.add(eid)

        # Fix the 3 incident edges of the first vertex to break symmetry
        start_v = comp_v[0]
        e0, e1, e2 = adj[start_v]

        edge_color[e0] = 1
        vertex_used[edges[e0][0]] |= 1 << 1
        vertex_used[edges[e0][1]] |= 1 << 1

        edge_color[e1] = 2
        vertex_used[edges[e1][0]] |= 1 << 2
        vertex_used[edges[e1][1]] |= 1 << 2

        edge_color[e2] = 3
        vertex_used[edges[e2][0]] |= 1 << 3
        vertex_used[edges[e2][1]] |= 1 << 3

        comp_edges.remove(e0)
        comp_edges.remove(e1)
        comp_edges.remove(e2)

        if not backtrack(comp_edges):
            print("NO")
            return

    print("YES")
    print(*(edge_color[i] for i in range(m)))


if __name__ == "__main__":
    solve()
