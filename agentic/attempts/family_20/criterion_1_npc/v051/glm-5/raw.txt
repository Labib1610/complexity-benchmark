import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n + 1)]
    matches = []
    
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        matches.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check degree constraint
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    # Find cycles
    visited_edges = [False] * m
    cycles = []
    
    for i in range(m):
        if not visited_edges[i]:
            u, v = matches[i]
            visited_edges[i] = True
            cycle_edges = [i]
            curr = v
            prev = u
            
            while True:
                next_edge = -1
                # Find next edge in cycle
                for neighbor, edge_idx in adj[curr]:
                    if edge_idx != i and not visited_edges[edge_idx]:
                        next_edge = edge_idx
                        break
                
                if next_edge == -1:
                    break
                
                cycle_edges.append(next_edge)
                visited_edges[next_edge] = True
                nu, nv = matches[next_edge]
                if curr == nu:
                    prev = nu
                    curr = nv
                else:
                    prev = nv
                    curr = nu
            
            cycles.append(cycle_edges)
    
    # Assign sessions
    sessions = [0] * m
    for cycle in cycles:
        L = len(cycle)
        for k, edge_idx in enumerate(cycle):
            sessions[edge_idx] = (k % 3) + 1
    
    print("YES")
    print(*(sessions))

if __name__ == "__main__":
    solve()
