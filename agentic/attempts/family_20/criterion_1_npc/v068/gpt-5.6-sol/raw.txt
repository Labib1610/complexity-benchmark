import sys
from collections import deque

sys.setrecursionlimit(1000000)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    if m != 3 * n // 2 or len(data) != 2 + 2 * m:
        print("NO")
        return

    edges = []
    adj = [[] for _ in range(n)]

    pos = 2
    for eid in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        if not (0 <= u < n and 0 <= v < n) or u == v:
            print("NO")
            return
        edges.append((u, v))
        adj[u].append(eid)
        adj[v].append(eid)

    if any(len(adj[v]) != 3 for v in range(n)):
        print("NO")
        return

    # Find connected components and reject bridges.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False
    component_roots = []

    def bridge_dfs(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1

        for eid in adj[v]:
            if eid == parent_edge:
                continue
            a, b = edges[eid]
            to = b if a == v else a
            if tin[to] == -1:
                bridge_dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for v in range(n):
        if tin[v] == -1:
            component_roots.append(v)
            bridge_dfs(v, -1)

    if has_bridge:
        print("NO")
        return

    # Bipartite cubic graphs can be colored directly:
    # one perfect matching plus alternating colors on the remaining cycles.
    side = [-1] * n
    is_bipartite = True

    for start in range(n):
        if side[start] != -1:
            continue
        side[start] = 0
        q = deque([start])
        while q and is_bipartite:
            v = q.popleft()
            for eid in adj[v]:
                a, b = edges[eid]
                to = b if a == v else a
                if side[to] == -1:
                    side[to] = side[v] ^ 1
                    q.append(to)
                elif side[to] == side[v]:
                    is_bipartite = False
                    break

    if is_bipartite:
        left = [v for v in range(n) if side[v] == 0]
        pair_left = [-1] * n
        pair_right = [-1] * n
        matching_edge = [-1] * n
        dist = [-1] * n

        def hk_bfs():
            q = deque()
            found = False
            for u in left:
                if pair_left[u] == -1:
                    dist[u] = 0
                    q.append(u)
                else:
                    dist[u] = -1

            while q:
                u = q.popleft()
                for eid in adj[u]:
                    a, b = edges[eid]
                    v = b if a == u else a
                    next_u = pair_right[v]
                    if next_u == -1:
                        found = True
                    elif dist[next_u] == -1:
                        dist[next_u] = dist[u] + 1
                        q.append(next_u)
            return found

        def hk_dfs(u):
            for eid in adj[u]:
                a, b = edges[eid]
                v = b if a == u else a
                next_u = pair_right[v]
                if next_u == -1 or (
                    dist[next_u] == dist[u] + 1 and hk_dfs(next_u)
                ):
                    pair_left[u] = v
                    pair_right[v] = u
                    matching_edge[u] = eid
                    matching_edge[v] = eid
                    return True
            dist[u] = -1
            return False

        matched = 0
        while hk_bfs():
            for u in left:
                if pair_left[u] == -1 and hk_dfs(u):
                    matched += 1

        if matched == len(left):
            answer = [0] * m
            for v in left:
                answer[matching_edge[v]] = 1

            for start_eid in range(m):
                if answer[start_eid] != 0:
                    continue

                answer[start_eid] = 2
                u, v = edges[start_eid]
                current = v
                previous_edge = start_eid
                next_color = 3

                while True:
                    next_edge = -1
                    for eid in adj[current]:
                        if eid != previous_edge and answer[eid] != 1:
                            next_edge = eid
                            break

                    if next_edge == -1 or answer[next_edge] != 0:
                        break

                    answer[next_edge] = next_color
                    next_color = 5 - next_color
                    a, b = edges[next_edge]
                    current = b if a == current else a
                    previous_edge = next_edge

            print("YES")
            print(*answer)
            return

    # General cubic graph: constraint propagation and backtracking.
    # Domain bits 1, 2, 4 correspond to colors 1, 2, 3.
    domains = [7] * m
    trail = []

    permutations = (
        (1, 2, 4),
        (1, 4, 2),
        (2, 1, 4),
        (2, 4, 1),
        (4, 1, 2),
        (4, 2, 1),
    )

    def propagate(initial_vertices):
        q = deque()
        in_queue = [False] * n

        for v in initial_vertices:
            if not in_queue[v]:
                in_queue[v] = True
                q.append(v)

        while q:
            v = q.popleft()
            in_queue[v] = False
            e0, e1, e2 = adj[v]
            d0, d1, d2 = domains[e0], domains[e1], domains[e2]

            s0 = s1 = s2 = 0
            for c0, c1, c2 in permutations:
                if (d0 & c0) and (d1 & c1) and (d2 & c2):
                    s0 |= c0
                    s1 |= c1
                    s2 |= c2

            if s0 == 0:
                return False

            for eid, support in ((e0, s0), (e1, s1), (e2, s2)):
                new_domain = domains[eid] & support
                if new_domain == 0:
                    return False
                if new_domain != domains[eid]:
                    trail.append((eid, domains[eid]))
                    domains[eid] = new_domain
                    a, b = edges[eid]
                    if not in_queue[a]:
                        in_queue[a] = True
                        q.append(a)
                    if not in_queue[b]:
                        in_queue[b] = True
                        q.append(b)

        return True

    def assign(eid, color_bit):
        if not (domains[eid] & color_bit):
            return False
        if domains[eid] != color_bit:
            trail.append((eid, domains[eid]))
            domains[eid] = color_bit
        u, v = edges[eid]
        return propagate((u, v))

    # Remove independent color-permutation symmetry in every component.
    for root in component_roots:
        for eid, color_bit in zip(adj[root], (1, 2, 4)):
            checkpoint = len(trail)
            if not assign(eid, color_bit):
                while len(trail) > checkpoint:
                    changed_eid, old_domain = trail.pop()
                    domains[changed_eid] = old_domain
                print("NO")
                return

    def search():
        chosen = -1
        best_size = 4

        for eid, domain in enumerate(domains):
            size = domain.bit_count()
            if 1 < size < best_size:
                best_size = size
                chosen = eid
                if size == 2:
                    break

        if chosen == -1:
            return True

        options = domains[chosen]
        while options:
            color_bit = options & -options
            options -= color_bit
            checkpoint = len(trail)

            if assign(chosen, color_bit) and search():
                return True

            while len(trail) > checkpoint:
                eid, old_domain = trail.pop()
                domains[eid] = old_domain

        return False

    if not search():
        print("NO")
        return

    answer = [domain.bit_length() for domain in domains]
    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
