```python
import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Read edges
    edges = []
    degree = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        degree[u] += 1
        degree[v] += 1
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if all degrees are 3
    for i in range(1, n + 1):
        if degree[i] != 3:
            print("NO")
            return
    
    # Try to color the edges with 3 colors such that each vertex has edges of 3 different colors
    # This is essentially edge coloring of a 3-regular graph with 3 colors
    
    colors = [0] * m  # 0 means uncolored, 1, 2, 3 are the three colors
    
    # For each vertex, we need to assign 3 different colors to its 3 edges
    # We can do this by finding a perfect matching in a related graph, but let's try a greedy approach
    
    # Try a systematic approach:
    # Since the graph is 3-regular and we have exactly 3 colors, 
    # we can think of this as orienting edges and assigning colors appropriately
    
    # Let's try to build the coloring directly
    # For each vertex, we'll maintain which colors are already used on its edges
    
    used_colors = [set() for _ in range(n + 1)]  # used_colors[v] = set of colors used on edges incident to v
    
    # Process edges one by one and assign colors greedily
    for i in range(m):
        u, v = edges[i]
        
        # Find a color that is not used by either u or v
        found_color = -1
        for color in range(1, 4):
            if color not in used_colors[u] and color not in used_colors[v]:
                found_color = color
                break
        
        if found_color == -1:
            print("NO")
            return
            
        colors[i] = found_color
        used_colors[u].add(found_color)
        used_colors[v].add(found_color)
    
    print("YES")
    print(' '.join(map(str, colors)))

solve()
```
