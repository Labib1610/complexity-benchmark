import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    idx = 2
    for i in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        edges.append((u, v))
        idx += 2

    # adj_edge[u][c] = edge index of color c incident to u (-1 if none)
    adj_edge = [[-1] * 4 for _ in range(n)]
    edge_color = [0] * m

    def set_color(e, c):
        u, v = edges[e]
        old_c = edge_color[e]
        if old_c != 0:
            adj_edge[u][old_c] = -1
            adj_edge[v][old_c] = -1
        edge_color[e] = c
        if c != 0:
            adj_edge[u][c] = e
            adj_edge[v][c] = e

    uncolored = list(range(m))
    random.shuffle(uncolored)

    max_steps = 150000
    steps = 0

    while uncolored and steps < max_steps:
        steps += 1
        e = uncolored.pop()
        u, v = edges[e]

        # Find free colors at u and v
        free_u = [c for c in (1, 2, 3) if adj_edge[u][c] == -1]
        free_v = [c for c in (1, 2, 3) if adj_edge[v][c] == -1]

        # Common free color
        common = set(free_u) & set(free_v)
        if common:
            c = common.pop()
            set_color(e, c)
            continue

        c1 = random.choice(free_u)
        c2 = random.choice(free_v)

        # Trace (c1, c2)-Kempe chain from u
        path = []
        curr = u
        col = c1
        while True:
            nxt_e = adj_edge[curr][col]
            if nxt_e == -1:
                break
            path.append(nxt_e)
            nu, nv = edges[nxt_e]
            curr = nv if curr == nu else nu
            col = c2 if col == c1 else c1

        if curr != v:
            # Path doesn't reach v: flip colors on path
            for pe in path:
                old_c = edge_color[pe]
                new_c = c2 if old_c == c1 else c1
                set_color(pe, new_c)
            set_color(e, c2)
        else:
            # Perturb: uncolor a random incident edge at u or v
            colored_u = [adj_edge[u][c] for c in (1, 2, 3) if adj_edge[u][c] != -1]
            if colored_u:
                drop_e = random.choice(colored_u)
                set_color(drop_e, 0)
                uncolored.append(drop_e)
            uncolored.append(e)

    if uncolored:
        print("NO")
    else:
        print("YES")
        print(*(edge_color[i] for i in range(m)))

if __name__ == '__main__':
    solve()
