import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n)]
    edges = []
    for i in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if graph is 3-regular
    for i in range(n):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    # Find a perfect matching using augmenting paths
    match = [-1] * n
    edge_match = [-1] * m
    
    def bfs():
        dist = [-1] * n
        q = deque()
        for u in range(n):
            if match[u] == -1:
                dist[u] = 0
                q.append(u)
        found = False
        while q:
            u = q.popleft()
            for v, eid in adj[u]:
                w = match[v]
                if w != -1 and dist[w] == -1:
                    dist[w] = dist[u] + 1
                    q.append(w)
                elif w == -1:
                    found = True
        return dist, found
    
    def dfs(u, dist):
        for v, eid in adj[u]:
            w = match[v]
            if w == -1 or (dist[w] == dist[u] + 1 and dfs(w, dist)):
                match[u] = v
                match[v] = u
                edge_match[eid] = 1
                return True
        dist[u] = -1
        return False
    
    while True:
        dist, found = bfs()
        if not found:
            break
        for u in range(n):
            if match[u] == -1:
                dfs(u, dist)
    
    # Verify perfect matching
    for i in range(n):
        if match[i] == -1:
            print("NO")
            return
    
    # Remove matched edges to get 2-regular graph (cycles)
    adj2 = [[] for _ in range(n)]
    for i, (u, v) in enumerate(edges):
        if edge_match[i] == -1:
            adj2[u].append(v)
            adj2[v].append(u)
    
    # 2-color each cycle
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj2[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        q.append(v)
    
    # Assign shifts
    ans = [0] * m
    for i, (u, v) in enumerate(edges):
        if edge_match[i] == 1:
            ans[i] = 1
        else:
            if color[u] == 0:
                ans[i] = 2
            else:
                ans[i] = 3
    
    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == "__main__":
    solve()
