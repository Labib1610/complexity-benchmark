import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
        
    # Check for bridges (Tarjan's algorithm)
    # A cubic graph with a bridge cannot be 3-edge-colored.
    tin = [0] * (n + 1)
    low = [0] * (n + 1)
    timer = 0
    has_bridge = False
    
    def dfs_bridge(v, p_edge):
        nonlocal timer, has_bridge
        timer += 1
        tin[v] = low[v] = timer
        for to, e_idx in adj[v]:
            if e_idx == p_edge:
                continue
            if tin[to]:
                low[v] = min(low[v], tin[to])
            else:
                dfs_bridge(to, e_idx)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True

    for i in range(1, n + 1):
        if not tin[i]:
            dfs_bridge(i, -1)
            
    if has_bridge:
        print("NO")
        return

    # Backtracking with MRV / Forward Checking
    # color assigned to each edge: -1 if uncolored, else 0, 1, 2
    edge_color = [-1] * m
    
    # forbidden colors for each edge (bitmask: bit c is 1 if color c is used by an adjacent edge)
    forbidden = [0] * m
    
    # Precompute edge adjacency: for each edge, the other 4 adjacent edges
    edge_adj = [[] for _ in range(m)]
    for u in range(1, n + 1):
        incident = [e_idx for _, e_idx in adj[u]]
        for i in range(len(incident)):
            for j in range(i + 1, len(incident)):
                e1, e2 = incident[i], incident[j]
                edge_adj[e1].append(e2)
                edge_adj[e2].append(e1)
                
    # Remove duplicates in edge_adj (if multiple edges between same pair)
    for i in range(m):
        edge_adj[i] = list(set(edge_adj[i]))

    # We use a randomized/heuristic backtracking
    import random
    random.seed(42)

    # Count of unassigned edges
    unassigned_count = m
    
    # Backtracking search
    # To be very fast, maintain domain sizes
    domain_size = [3] * m
    
    # We can also fix the colors of the 3 edges incident to vertex 1 to break symmetry
    first_v_edges = [e_idx for _, e_idx in adj[1]]
    if len(first_v_edges) == 3:
        for c, e_idx in enumerate(first_v_edges):
            edge_color[e_idx] = c
            unassigned_count -= 1
            for nxt in edge_adj[e_idx]:
                if (forbidden[nxt] & (1 << c)) == 0:
                    forbidden[nxt] |= (1 << c)
                    domain_size[nxt] -= 1

    def backtrack():
        nonlocal unassigned_count
        if unassigned_count == 0:
            return True
        
        # MRV: find edge with smallest domain_size > 0 among unassigned
        best_e = -1
        min_dom = 4
        
        for e in range(m):
            if edge_color[e] == -1:
                ds = domain_size[e]
                if ds == 0:
                    return False
                if ds < min_dom:
                    min_dom = ds
                    best_e = e
                    if min_dom == 1:
                        break
                        
        e = best_e
        mask = forbidden[e]
        
        # Try available colors
        # Colors: 0, 1, 2
        for c in range(3):
            if not (mask & (1 << c)):
                # Assign color c to edge e
                edge_color[e] = c
                unassigned_count -= 1
                
                # Propagate to neighbors
                changed = []
                conflict = False
                for nxt in edge_adj[e]:
                    if (forbidden[nxt] & (1 << c)) == 0:
                        forbidden[nxt] |= (1 << c)
                        domain_size[nxt] -= 1
                        changed.append(nxt)
                        if edge_color[nxt] == -1 and domain_size[nxt] == 0:
                            conflict = True
                
                if not conflict:
                    if backtrack():
                        return True
                
                # Undo
                for nxt in changed:
                    forbidden[nxt] &= ~(1 << c)
                    domain_size[nxt] += 1
                    
                edge_color[e] = -1
                unassigned_count += 1
                
        return False

    if backtrack():
        print("YES")
        print(*(edge_color[i] + 1 for i in range(m)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
