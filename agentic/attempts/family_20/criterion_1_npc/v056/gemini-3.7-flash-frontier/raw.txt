import sys


def solve():
    sys.setrecursionlimit(20000)
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    edges = []
    vertex_edges = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        vertex_edges[u].append(i)
        vertex_edges[v].append(i)

    # Build line graph adjacency
    adj_edges = [[] for _ in range(m)]
    for i in range(m):
        u, v = edges[i]
        for e in vertex_edges[u]:
            if e != i:
                adj_edges[i].append(e)
        for e in vertex_edges[v]:
            if e != i:
                adj_edges[i].append(e)

    # Domain masks: bit 0: color 1 (1), bit 1: color 2 (2), bit 2: color 3 (4)
    # mask = 7 initially
    mask = [7] * m
    color = [0] * m
    trail = []

    # Pre-assign connected components to break symmetry
    # Assign edge 0 color 1
    def assign_forced(e, c):
        # assign color c (1, 2, or 4) to edge e
        color[e] = c
        old_m = mask[e]
        mask[e] = c
        trail.append((e, old_m ^ c, 0))  # 0 indicates self-assignment
        for v in adj_edges[e]:
            if mask[v] & c:
                mask[v] ^= c
                trail.append((v, c, 1))
                if mask[v] == 0:
                    return False
        return True

    def propagate():
        # Unit propagation loop
        queue = []
        for i in range(m):
            if color[i] == 0:
                m_i = mask[i]
                if m_i == 0:
                    return False
                if (m_i & (m_i - 1)) == 0:
                    queue.append(i)

        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            c = mask[u]
            if color[u] != 0:
                continue
            color[u] = c
            for v in adj_edges[u]:
                if color[v] == c:
                    return False
                if mask[v] & c:
                    mask[v] ^= c
                    trail.append((v, c, 1))
                    if mask[v] == 0:
                        return False
                    if (mask[v] & (mask[v] - 1)) == 0 and color[v] == 0:
                        queue.append(v)
        return True

    def undo(trail_len):
        while len(trail) > trail_len:
            v, removed_mask, typ = trail.pop()
            mask[v] |= removed_mask
            if typ == 0:
                color[v] = 0
            else:
                if color[v] != 0 and (mask[v] & color[v]) == 0:
                    color[v] = 0

    def get_mrv_variable():
        min_size = 4
        best_var = -1
        for i in range(m):
            if color[i] == 0:
                m_i = mask[i]
                if m_i == 0:
                    return -1, 0
                # popcount of 3-bit mask
                cnt = (m_i & 1) + ((m_i >> 1) & 1) + ((m_i >> 2) & 1)
                if cnt < min_size:
                    min_size = cnt
                    best_var = i
                    if min_size == 1:
                        break
        return best_var, min_size

    def backtrack():
        var, size = get_mrv_variable()
        if var == -1:
            if size == 0:
                return False
            return True  # all colored

        avail_colors = []
        m_var = mask[var]
        for c in (1, 2, 4):
            if m_var & c:
                avail_colors.append(c)

        for c in avail_colors:
            t_len = len(trail)
            if assign_forced(var, c) and propagate():
                if backtrack():
                    return True
            undo(t_len)

        return False

    # Seed edge 0 with color 1 to break symmetry
    assign_forced(0, 1)
    if not propagate() or not backtrack():
        print("NO")
        return

    print("YES")
    c_map = {1: 1, 2: 2, 4: 3}
    res = [str(c_map[color[i]]) for i in range(m)]
    print(" ".join(res))


if __name__ == "__main__":
    solve()
