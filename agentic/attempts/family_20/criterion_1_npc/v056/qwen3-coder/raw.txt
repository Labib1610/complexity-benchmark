import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(10000)
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    adj = defaultdict(list)
    
    index = 2
    for i in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        index += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Each node has exactly 3 edges
    # We need to color the edges with 3 colors such that:
    # 1. Each node has each color exactly once (edges incident to node have different colors)
    # 2. Each color is used exactly m/3 = n/2 times
    
    # This is equivalent to edge coloring where each vertex has degree 3
    # And we want a balanced 3-edge-coloring
    
    # Since each vertex has degree 3, and we have 3 colors, 
    # each vertex must use each color exactly once.
    
    # Build a line graph: each edge becomes a node, adjacent if they share a vertex
    # Then we need to 3-color this line graph such that each color appears exactly n/2 times
    
    # However, let's think differently.
    # Since the graph is 3-regular, it's known that it can be edge-colored with 3 colors
    # But we also need the count constraint.
    
    # Let's try to model this as a CSP or use a heuristic.
    # But given the constraints, maybe we can find a pattern.
    
    # Let's attempt a greedy approach with backtracking
    
    # Actually, since each vertex has degree 3, and we have 3 colors,
    # at each vertex, the 3 edges must get 3 different colors.
    
    # So for each vertex, we just need to assign a permutation of {1,2,3} to its 3 edges
    # But globally, we need each color to appear exactly n/2 times
    
    # Let's model this as an assignment problem
    # For each vertex, we have 3 edges and 3 colors -> 6 possible assignments (permutations)
    # But we need global balance
    
    # Let's try a different approach: since the structure is very constrained,
    # maybe we can build a feasible solution directly
    
    # Let's consider that this is a 3-regular graph on n vertices with m=3n/2 edges
    # It's known that such graphs can be edge-colored with 3 colors
    # But we also need each color to be used exactly n/2 times
    
    # Since m=3n/2 and we want each color used n/2 times, that works out perfectly
    
    # Let's try to construct such a coloring
    
    # We'll use a greedy algorithm with backtracking
    # For each edge, try to assign a color that doesn't conflict with its endpoints
    # and doesn't exceed the color count limit
    
    color_count = [0, 0, 0]  # indexed by 0,1,2 for colors 1,2,3
    edge_colors = [-1] * m
    max_color_count = n // 2
    
    # For each vertex, track which colors are used on its edges
    vertex_colors = [set() for _ in range(n)]
    
    def is_valid_color(edge_idx, color):
        u, v = edges[edge_idx]
        if color in vertex_colors[u] or color in vertex_colors[v]:
            return False
        if color_count[color] >= max_color_count:
            return False
        return True
    
    def assign_color(edge_idx, color):
        u, v = edges[edge_idx]
        edge_colors[edge_idx] = color
        vertex_colors[u].add(color)
        vertex_colors[v].add(color)
        color_count[color] += 1
    
    def unassign_color(edge_idx, color):
        u, v = edges[edge_idx]
        edge_colors[edge_idx] = -1
        vertex_colors[u].remove(color)
        vertex_colors[v].remove(color)
        color_count[color] -= 1
    
    def solve(edge_idx):
        if edge_idx == m:
            return True
            
        for color in range(3):
            if is_valid_color(edge_idx, color):
                assign_color(edge_idx, color)
                if solve(edge_idx + 1):
                    return True
                unassign_color(edge_idx, color)
        
        return False
    
    if solve(0):
        print("YES")
        print(' '.join(str(c + 1) for c in edge_colors))
    else:
        print("NO")

if __name__ == "__main__":
    main()
