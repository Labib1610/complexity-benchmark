```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(20000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
        
    # Check if all vertices have degree 3
    for v in range(1, n + 1):
        if len(adj[v]) != 3:
            print("NO")
            return

    # Formulate as SAT
    # Variables: for edge i and color c in {0, 1, 2}, var = 3 * i + c + 1
    # 1-indexed variables: 1 to 3*m
    num_vars = 3 * m
    clauses = []
    
    def var(e, c):
        return 3 * e + c + 1
    
    # 1. Each edge has at least one color: (var(e, 0) v var(e, 1) v var(e, 2))
    for e in range(m):
        clauses.append([var(e, 0), var(e, 1), var(e, 2)])
        # at most one color per edge
        clauses.append([-var(e, 0), -var(e, 1)])
        clauses.append([-var(e, 0), -var(e, 2)])
        clauses.append([-var(e, 1), -var(e, 2)])
        
    # 2. At each vertex, for each color, at most one incident edge has that color
    for v in range(1, n + 1):
        e1, e2, e3 = adj[v]
        for c in range(3):
            v1, v2, v3 = var(e1, c), var(e2, c), var(e3, c)
            # at most one of v1, v2, v3
            clauses.append([-v1, -v2])
            clauses.append([-v1, -v3])
            clauses.append([-v2, -v3])
            # exactly one: at least one must also hold since degree is 3 and 3 colors
            clauses.append([v1, v2, v3])

    # Simple and fast CDCL SAT solver
    assignment = [0] * (num_vars + 1)
    
    # Adjacency for clauses (2-watched literals)
    # Clause representation: list of ints
    # watches[lit] = list of clause indices where lit is watched
    watches = [[] for _ in range(2 * num_vars + 2)]
    
    def lit_to_idx(lit):
        return lit if lit > 0 else num_vars - lit
    
    clause_list = []
    for c in clauses:
        clause_list.append(list(c))
        
    for c_idx, c in enumerate(clause_list):
        if len(c) >= 2:
            watches[lit_to_idx(c[0])].append(c_idx)
            watches[lit_to_idx(c[1])].append(c_idx)
        elif len(c) == 1:
            watches[lit_to_idx(c[0])].append(c_idx)

    # CDCL structures
    decision_level = 0
    trail = []
    trail_lim = []
    reason = [None] * (num_vars + 1)
    level = [0] * (num_vars + 1)
    
    # Activity for VSIDS
    activity = [0.0] * (num_vars + 1)
    var_inc = 1.0
    var_decay = 0.95
    
    # Initial activity based on clause occurrences
    for c in clause_list:
        for lit in c:
            activity[abs(lit)] += 1.0

    def assign(lit, r):
        v = abs(lit)
        val = 1 if lit > 0 else -1
        assignment[v] = val
        reason[v] = r
        level[v] = decision_level
        trail.append(lit)

    def unassign(lit):
        v = abs(lit)
        assignment[v] = 0
        reason[v] = None
        level[v] = 0

    def lit_val(lit):
        v = abs(lit)
        val = assignment[v]
        return val if lit > 0 else -val

    def propagate():
        q_head = len(trail) - 1
        # Actually propagate from where we left off
        # We need a pointer
        nonlocal prop_head
        while prop_head < len(trail):
            p = trail[prop_head]
            prop_head += 1
            false_lit = -p
            w_list = watches[lit_to_idx(false_lit)]
            i = 0
            while i < len(w_list):
                c_idx = w_list[i]
                c = clause_list[c_idx]
                
                # Make sure false_lit is at c[1]
                if c[0] == false_lit:
                    c[0], c[1] = c[1], c[0]
                    
                # If first literal is True, clause is satisfied
                if lit_val(c[0]) == 1:
                    i += 1
                    continue
                    
                # Look for new literal to watch
                found = False
                for k in range(2, len(c)):
                    if lit_val(c[k]) != -1:
                        c[1], c[k] = c[k], c[1]
                        watches[lit_to_idx(c[1])].append(c_idx)
                        # Remove from w_list
                        w_list[i] = w_list[-1]
                        w_list.pop()
                        found = True
                        break
                        
                if found:
                    continue
                    
                # Clause is unit or conflict
                if lit_val(c[0]) == -1: # Conflict
                    return c_idx
                else: # Unit
                    assign(c[0], c_idx)
                    i += 1
        return -1

    prop_head = 0
    
    # Process initial unit clauses if any
    for c_idx, c in enumerate(clause_list):
        if len(c) == 1:
            val = lit_val(c[0])
            if val == -1:
                print("NO")
                return
            elif val == 0:
                assign(c[0], None)
                
    if propagate() != -1:
        print("NO")
        return

    # Symmetry breaking: color the 3 edges of vertex 1 with 0, 1, 2
    e1, e2, e3 = adj[1]
    # Assign edge e1 color 0, e2 color 1, e3 color 2
    decision_level += 1
    trail_lim.append(len(trail))
    assign(var(e1, 0), None)
    assign(var(e2, 1), None)
    assign(var(e3, 2), None)
    
    if propagate() != -1:
        print("NO")
        return

    # Pick branching variable
    def pick_branch_var():
        best_v = -1
        max_act = -1.0
        for v in range(1, num_vars + 1):
            if assignment[v] == 0:
                if activity[v] > max_act:
                    max_act = activity[v]
                    best_v = v
        return best_v

    # Conflict analysis
    def analyze(conflict_c_idx):
        nonlocal var_inc
        learnt = []
        seen = [False] * (num_vars + 1)
        counter = 0
        p = None
        c_idx = conflict_c_idx
        
        idx = len(trail) - 1
        while True:
            c = clause_list[c_idx]
            for lit in c:
                v = abs(lit)
                if not seen[v]:
                    seen[v] = True
                    activity[v] += var_inc
                    if level[v] == decision_level:
                        counter += 1
                    elif level[v] > 0:
                        learnt.append(lit)
                        
            while True:
                p = trail[idx]
                idx -= 1
                if seen[abs(p)]:
                    break
                    
            seen[abs(p)] = False
            counter -= 1
            if counter == 0:
                learnt.append(-p)
                break
            c_idx = reason[abs(p)]

        # Find backtrack level
        bt_level = 0
        if len(learnt) > 1:
            # find the max level among other literals and put it at index 1
            max_i = 1
            max_l = level[abs(learnt[1])]
            for i in range(2, len(learnt)):
                if level[abs(learnt[i])] > max_l:
                    max_l = level[abs(learnt[i])]
                    max_i = i
            learnt[1], learnt[max_i] = learnt[max_i], learnt[1]
            bt_level = max_l
            
        learnt[0], learnt[-1] = learnt[-1], learnt[0] # The 1-UIP literal is at index 0
        var_inc /= var_decay
        return learnt, bt_level

    # Main CDCL loop
    while True:
        confl = propagate()
        if confl != -1:
            if decision_level == 0:
                print("NO")
                return
            learnt, bt_level = analyze(confl)
            
            # Backtrack to bt_level
            while decision_level > bt_level:
                lim = trail_lim.pop()
                while len(trail) > lim:
                    unassign(trail.pop())
                decision_level -= 1
            prop_head = len(trail)
            
            # Add learnt clause
            c_idx = len(clause_list)
            clause_list.append(learnt)
            if len(learnt) >= 2:
                watches[lit_to_idx(learnt[0])].append(c_idx)
                watches[lit_to_idx(learnt[1])].append(c_idx)
                assign(learnt[0], c_idx)
            else:
                watches[lit_to_idx(learnt[0])].append(c_idx)
                assign(learnt[0], None)
        else:
            # Pick decision variable
            v = pick_branch_var()
            if v == -1:
                # All assigned, solution found!
                print("YES")
                res = []
                for e in range(m):
                    for c in range(3):
                        if assignment[var(e, c)] == 1:
                            res.append(c + 1)
                            break
                print(*(res))
                return
            
            decision_level += 1
            trail_lim.append(len(trail))
            # Heuristic: set variable to True
            assign(v, None)

solve()
```
