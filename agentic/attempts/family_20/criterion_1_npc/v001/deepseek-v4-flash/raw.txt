We can model this as a graph problem. Each junction has degree exactly 3, and we need to 3-color the edges so that no two edges at the same vertex share a color. This is equivalent to edge-coloring a 3-regular graph with 3 colors, which is always possible by König’s line coloring theorem (since the graph is bipartite) — but here the graph may not be bipartite.  

However, for a 3-regular graph, a proper 3-edge-coloring exists if and only if the graph is bridgeless (by the Petersen theorem). If there is a bridge, it’s impossible.  

We can check for bridges using DFS, and if none exist, we can find the 3-edge-coloring via a constructive method (e.g., using a matching-based approach or a known algorithm for cubic graphs).  

Here’s the implementation:

import sys
sys.setrecursionlimit(1000000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    edges = []
    adj = [[] for _ in range(n)]
    for i in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if graph is connected (should be, but just in case)
    visited = [False] * n
    stack = [0]
    visited[0] = True
    while stack:
        u = stack.pop()
        for v, _ in adj[u]:
            if not visited[v]:
                visited[v] = True
                stack.append(v)
    if not all(visited):
        print("NO")
        return
    
    # Find bridges using DFS
    tin = [-1] * n
    low = [-1] * n
    timer = [0]
    is_bridge = [False] * m
    
    def dfs(v, p_edge):
        tin[v] = low[v] = timer[0]
        timer[0] += 1
        for to, eid in adj[v]:
            if eid == p_edge:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    is_bridge[eid] = True
    
    dfs(0, -1)
    
    if any(is_bridge):
        print("NO")
        return
    
    # Now find a proper 3-edge-coloring.
    # Use a known constructive method: find a perfect matching, remove it,
    # the remaining graph is 2-regular (cycles), color those with 2 colors.
    # Repeat for the matching edges with the third color.
    
    color = [-1] * m
    
    # Find a perfect matching in a cubic graph using simple greedy + augmenting
    # Since n <= 2000, we can use a simple algorithm.
    match = [-1] * n
    used = [False] * n
    
    def try_kuhn(v, seen):
        for to, eid in adj[v]:
            if seen[to]:
                continue
            seen[to] = True
            if match[to] == -1 or try_kuhn(match[to], seen):
                match[to] = v
                return True
        return False
    
    for v in range(n):
        seen = [False] * n
        try_kuhn(v, seen)
    
    # match[to] = v means edge (v,to) is in matching
    matching_edges = []
    for v in range(n):
        if match[v] != -1 and v < match[v]:
            # find edge id
            for to, eid in adj[v]:
                if to == match[v]:
                    matching_edges.append(eid)
                    break
    
    # Color matching edges with color 3
    for eid in matching_edges:
        color[eid] = 3
    
    # Remaining graph is 2-regular (disjoint cycles)
    # Color each cycle alternately with colors 1 and 2
    remaining_adj = [[] for _ in range(n)]
    for i, (u, v) in enumerate(edges):
        if color[i] == -1:
            remaining_adj[u].append((v, i))
            remaining_adj[v].append((u, i))
    
    visited_cycle = [False] * n
    for start in range(n):
        if not visited_cycle[start] and remaining_adj[start]:
            # traverse cycle
            cur = start
            prev = -1
            while True:
                visited_cycle[cur] = True
                # pick next edge not going back
                nxt = None
                for to, eid in remaining_adj[cur]:
                    if eid != prev:
                        nxt = (to, eid)
                        break
                if nxt is None:
                    break
                to, eid = nxt
                # assign color alternating
                if color[eid] == -1:
                    # determine color based on position
                    # We'll just assign 1 or 2 alternately
                    # Use a simple flag
                    pass
                prev = eid
                cur = to
                if cur == start:
                    break
    
    # Simpler: color cycles by walking and alternating
    visited_edge = [False] * m
    for start in range(n):
        if not visited_cycle[start] and remaining_adj[start]:
            # start a cycle
            cur = start
            prev_edge = -1
            col = 1
            while True:
                visited_cycle[cur] = True
                # find next edge
                nxt_edge = None
                for to, eid in remaining_adj[cur]:
                    if eid != prev_edge and not visited_edge[eid]:
                        nxt_edge = eid
                        break
                if nxt_edge is None:
                    break
                visited_edge[nxt_edge] = True
                color[nxt_edge] = col
                col = 3 - col  # switch 1<->2
                # move to other endpoint
                u, v = edges[nxt_edge]
                nxt_vertex = v if u == cur else u
                prev_edge = nxt_edge
                cur = nxt_vertex
                if cur == start:
                    break
    
    # Verify
    for u in range(n):
        cols = set()
        for v, eid in adj[u]:
            cols.add(color[eid])
        if len(cols) != 3:
            print("NO")
            return
