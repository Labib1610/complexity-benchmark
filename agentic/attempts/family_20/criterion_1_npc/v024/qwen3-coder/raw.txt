```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    # Build adjacency list and edge list
    adj = defaultdict(list)
    edges = []
    
    for i in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append((v, i))
        adj[v].append((u, i))
        edges.append((u, v))
    
    # Color assignments (1, 2, 3)
    colors = [0] * m  # 0 means uncolored
    
    # For each vertex, we need to color its incident edges with 3 different colors
    # This is equivalent to edge-coloring where each vertex has degree 3
    
    # Check if each vertex has degree exactly 3
    for u in range(n):
        if len(adj[u]) != 3:
            print("NO")
            return
    
    # We'll use BFS to color the edges
    # Start with any vertex and assign colors 1, 2, 3 to its edges
    visited_vertices = [False] * n
    visited_edges = [False] * m
    
    # Process each connected component
    for start_vertex in range(n):
        if visited_vertices[start_vertex]:
            continue
            
        queue = deque([start_vertex])
        visited_vertices[start_vertex] = True
        
        while queue:
            u = queue.popleft()
            
            # Get unvisited neighbors
            unvisited_edges = []
            for v, edge_idx in adj[u]:
                if not visited_edges[edge_idx]:
                    unvisited_edges.append((v, edge_idx))
            
            # If we have unvisited edges from this vertex
            if unvisited_edges:
                # Assign colors to these edges
                if len(unvisited_edges) == 3:
                    # First time visiting this vertex - assign colors 1, 2, 3
                    colors[unvisited_edges[0][1]] = 1
                    colors[unvisited_edges[1][1]] = 2
                    colors[unvisited_edges[2][1]] = 3
                elif len(unvisited_edges) == 2:
                    # Need to determine the third color based on the already colored edge
                    colored_count = 0
                    colored_colors = []
                    uncolored_edges = []
                    
                    for v, edge_idx in adj[u]:
                        if colors[edge_idx] != 0:
                            colored_count += 1
                            colored_colors.append(colors[edge_idx])
                        else:
                            uncolored_edges.append((v, edge_idx))
                    
                    if colored_count == 1:
                        # One edge colored, need to color two more
                        used_color = colored_colors[0]
                        remaining_colors = [c for c in [1, 2, 3] if c != used_color]
                        colors[uncolored_edges[0][1]] = remaining_colors[0]
                        colors[uncolored_edges[1][1]] = remaining_colors[1]
                    else:  # colored_count == 2
                        # Two edges colored, determine the third
                        used_colors = set(colored_colors)
                        missing_color = 6 - sum(colored_colors)  # Since 1+2+3=6
                        colors[uncolored_edges[0][1]] = missing_color
                        
                elif len(unvisited_edges) == 1:
                    # Determine the color based on the two already colored edges
                    colored_colors = []
                    for v, edge_idx in adj[u]:
                        if colors[edge_idx] != 0:
                            colored_colors.append(colors[edge_idx])
                    
                    if len(colored_colors) == 2:
                        used_colors = set(colored_colors)
                        missing_color = 6 - sum(colored_colors)
                        colors[unvisited_edges[0][1]] = missing_color
                
                # Mark edges as visited
                for v, edge_idx in unvisited_edges:
                    visited_edges[edge_idx] = True
                    if not visited_vertices[v]:
                        visited_vertices[v] = True
                        queue.append(v)
    
    # Validate the solution
    # For each vertex, check that its three edges have different colors
    for u in range(n):
        edge_colors = []
        for v, edge_idx in adj[u]:
            edge_colors.append(colors[edge_idx])
        if len(set(edge_colors)) != 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, colors)))

solve()
```
