import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    edges = []
    adj = [[] for _ in range(n+1)]
    idx = 2
    for i in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        edges.append((u,v))
        adj[u].append((v,i))
        adj[v].append((u,i))

    color = [0]*m
    visited = [False]*(n+1)
    parent_edge = [-1]*(n+1)
    parent_node = [0]*(n+1)

    for s in range(1, n+1):
        if visited[s]:
            continue
        visited[s] = True
        stack = [s]
        while stack:
            u = stack.pop()
            for v, eid in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parent_node[v] = u
                    parent_edge[v] = eid
                    stack.append(v)

    # Euler tour on each tree
    euler = []
    seen = [False]*(n+1)
    for s in range(1, n+1):
        if seen[s]:
            continue
        seen[s] = True
        stack = [(s, iter(adj[s]))]
        while stack:
            u, it = stack[-1]
            advanced = False
            for v, eid in it:
                if parent_node[v] == u and not seen[v]:
                    seen[v] = True
                    stack.append((v, iter(adj[v])))
                    advanced = True
                    break
            if not advanced:
                stack.pop()
                if stack:
                    p = stack[-1][0]
                    pe = parent_edge[u]
                    if pe != -1:
                        euler.append((pe, p, u))

    # Process back edges
    for u in range(1, n+1):
        for v, eid in adj[u]:
            if v > u and parent_edge[v] != eid and parent_edge[u] != eid:
                # back edge
                if color[eid] == 0:
                    color[eid] = 1
                    # flip path from u to v
                    x = u
                    while x != v:
                        pe = parent_edge[x]
                        if pe == -1:
                            break
                        color[pe] = 3 - color[pe]
                        x = parent_node[x]

    # Process tree edges in reverse Euler order
    for eid, p, c in reversed(euler):
        if color[eid] == 0:
            color[eid] = 1

    # Verify
    ok = True
    for u in range(1, n+1):
        cols = []
        for v, eid in adj[u]:
            cols.append(color[eid])
        if len(set(cols)) != 3:
            ok = False
            break

    if not ok:
        sys.stdout.write("NO\n")
        return

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color)) + "\n")

solve()
