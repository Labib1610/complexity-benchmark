```python
import sys
import random

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
        
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return

    edge_color = [0] * m
    v_color = [[-1] * 4 for _ in range(n + 1)]
    
    def color_edge(e, c):
        edge_color[e] = c
        u, v = edges[e]
        v_color[u][c] = e
        v_color[v][c] = e
        
    def uncolor_edge(e):
        c = edge_color[e]
        edge_color[e] = 0
        u, v = edges[e]
        v_color[u][c] = -1
        v_color[v][c] = -1

    def get_missing_color(u):
        for c in (1, 2, 3):
            if v_color[u][c] == -1:
                return c
        return -1

    def get_chain(start_node, c_missing, c_next):
        curr = start_node
        c = c_next
        path = []
        while v_color[curr][c] != -1:
            e = v_color[curr][c]
            path.append(e)
            u, v = edges[e]
            curr = v if u == curr else u
            c = c_missing if c == c_next else c_next
        return path, curr

    def flip_chain(path, c1, c2):
        for e in path:
            old_c = edge_color[e]
            new_c = c2 if old_c == c1 else c1
            uncolor_edge(e)
            color_edge(e, new_c)

    queue = list(range(m))
    random.shuffle(queue)
    
    steps = 0
    max_steps = 150000
    
    while queue:
        steps += 1
        if steps > max_steps:
            print("NO")
            return
        
        e = queue.pop()
        u, v = edges[e]
        
        success = False
        for _ in range(30):
            a = get_missing_color(u)
            b = get_missing_color(v)
            if a == b:
                color_edge(e, a)
                success = True
                break
            
            path, end_node = get_chain(u, a, b)
            if end_node != v:
                flip_chain(path, a, b)
                color_edge(e, b)
                success = True
                break
            else:
                if random.random() < 0.5:
                    c = 6 - a - b
                    path2, _ = get_chain(u, a, c)
                    flip_chain(path2, a, c)
                else:
                    c_kick = random.choice([x for x in (1, 2, 3) if x != a])
                    e_kick = v_color[u][c_kick]
                    uncolor_edge(e_kick)
                    color_edge(e, c_kick)
                    queue.append(e_kick)
                    success = True
                    break
                    
        if not success:
            avail = [x for x in (1, 2, 3) if v_color[u][x] != -1]
            if avail:
                c_kick = random.choice(avail)
                e_kick = v_color[u][c_kick]
                uncolor_edge(e_kick)
                color_edge(e, c_kick)
                queue.append(e_kick)
            else:
                queue.append(e)

    print("YES")
    print(*(edge_color[i] for i in range(m)))

if __name__ == '__main__':
    solve()
```
