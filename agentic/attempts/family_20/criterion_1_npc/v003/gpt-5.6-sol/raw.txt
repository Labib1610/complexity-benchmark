import sys

sys.setrecursionlimit(10000)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    it = iter(data)
    n = next(it)
    m = next(it)

    edges = []
    adj = [[] for _ in range(n)]
    for eid in range(m):
        u = next(it) - 1
        v = next(it) - 1
        edges.append((u, v))
        adj[u].append((v, eid))
        adj[v].append((u, eid))

    # A cubic graph with a bridge cannot have a proper 3-edge-coloring.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False
    component_roots = []

    def dfs(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1

        for to, eid in adj[v]:
            if eid == parent_edge:
                continue
            if tin[to] == -1:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for v in range(n):
        if tin[v] == -1:
            component_roots.append(v)
            dfs(v, -1)

    if has_bridge:
        print("NO")
        return

    # Exact cover:
    # - one column for each edge (choose exactly one color),
    # - one column for each (vertex, color) pair.
    #
    # Row (edge uv, color c) covers:
    # edge, (u,c), (v,c).
    column_count = m + 3 * n

    L = [0] * (column_count + 1)
    R = [0] * (column_count + 1)
    U = [0] * (column_count + 1)
    D = [0] * (column_count + 1)
    C = [0] * (column_count + 1)
    S = [0] * (column_count + 1)
    row_id = [-1] * (column_count + 1)

    for c in range(column_count + 1):
        L[c] = c - 1
        R[c] = c + 1
        U[c] = D[c] = c
        C[c] = c

    L[0] = column_count
    R[column_count] = 0

    anchors = [[-1] * 3 for _ in range(m)]

    def add_row(rid, columns):
        first = -1
        previous = -1

        for col in columns:
            node = len(L)
            L.append(node)
            R.append(node)
            U.append(U[col])
            D.append(col)
            C.append(col)
            S.append(0)
            row_id.append(rid)

            D[U[col]] = node
            U[col] = node
            S[col] += 1

            if first == -1:
                first = node
            else:
                R[previous] = node
                L[node] = previous
            previous = node

        R[previous] = first
        L[first] = previous
        return first

    for eid, (u, v) in enumerate(edges):
        edge_col = 1 + eid
        for color in range(3):
            u_col = 1 + m + 3 * u + color
            v_col = 1 + m + 3 * v + color
            anchors[eid][color] = add_row(
                eid * 3 + color,
                (edge_col, u_col, v_col)
            )

    def cover(col):
        R[L[col]] = R[col]
        L[R[col]] = L[col]

        i = D[col]
        while i != col:
            j = R[i]
            while j != i:
                D[U[j]] = D[j]
                U[D[j]] = U[j]
                S[C[j]] -= 1
                j = R[j]
            i = D[i]

    def uncover(col):
        i = U[col]
        while i != col:
            j = L[i]
            while j != i:
                S[C[j]] += 1
                D[U[j]] = j
                U[D[j]] = j
                j = L[j]
            i = U[i]

        R[L[col]] = col
        L[R[col]] = col

    selected = []

    # Remove color-permutation symmetry independently in every component.
    for root in component_roots:
        incident = [eid for _, eid in adj[root]]
        if len(incident) != 3:
            print("NO")
            return

        incident.sort()
        for color, eid in enumerate(incident):
            node = anchors[eid][color]
            selected.append(eid * 3 + color)

            j = node
            while True:
                cover(C[j])
                j = R[j]
                if j == node:
                    break

    def search():
        if R[0] == 0:
            return True

        col = R[0]
        best_size = S[col]
        c = R[col]

        while c != 0:
            if S[c] < best_size:
                col = c
                best_size = S[c]
                if best_size <= 1:
                    break
            c = R[c]

        if best_size == 0:
            return False

        cover(col)
        i = D[col]

        while i != col:
            selected.append(row_id[i])

            j = R[i]
            while j != i:
                cover(C[j])
                j = R[j]

            if search():
                return True

            j = L[i]
            while j != i:
                uncover(C[j])
                j = L[j]

            selected.pop()
            i = D[i]

        uncover(col)
        return False

    if not search():
        print("NO")
        return

    answer = [0] * m
    for rid in selected:
        eid, color = divmod(rid, 3)
        answer[eid] = color + 1

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
