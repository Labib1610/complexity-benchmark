```python
import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    color = [0] * m
    visited = [False] * (n + 1)
    
    setrecursionlimit(4 * n + 100)
    
    for start in range(1, n + 1):
        if visited[start]:
            continue
        stack = [(start, 0, -1, -1)]
        visited[start] = True
        while stack:
            node, par_color, par_edge, _ = stack[-1]
            if stack[-1][3] == -1:
                stack[-1] = (node, par_color, par_edge, 0)
            node, par_color, par_edge, idx = stack[-1]
            if idx < len(adj[node]):
                stack[-1] = (node, par_color, par_edge, idx + 1)
                child, edge_idx = adj[node][idx]
                if edge_idx == par_edge:
                    continue
                if not visited[child]:
                    visited[child] = True
                    stack.append((child, par_color, edge_idx, -1))
                else:
                    if color[edge_idx] == 0:
                        color[edge_idx] = 1 if par_color != 1 else 2
            else:
                stack.pop()
    
    for i in range(m):
        if color[i] == 0:
            color[i] = 1
    
    for u in range(1, n + 1):
        seen = set()
        for v, idx in adj[u]:
            seen.add(color[idx])
        if len(seen) != 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
```
