import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    edges = []
    idx = 2
    for i in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        edges.append((u, v))

    deg = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    for i, (u, v) in enumerate(edges):
        adj[u].append((v, i))
        adj[v].append((u, i))
        deg[u] += 1
        deg[v] += 1

    color = [0] * m
    used = [0] * (n + 1)
    removed = [False] * (n + 1)
    stack = []

    for s in range(1, n + 1):
        if deg[s] == 1:
            stack.append(s)

    while stack:
        u = stack.pop()
        if removed[u] or deg[u] != 1:
            continue
        removed[u] = True
        v = -1
        e = -1
        for (w, eid) in adj[u]:
            if not removed[w]:
                v = w
                e = eid
                break
        if v == -1:
            continue
        c = 1
        while used[u] & (1 << c):
            c += 1
        color[e] = c
        used[u] |= (1 << c)
        used[v] |= (1 << c)
        deg[u] -= 1
        deg[v] -= 1
        if deg[v] == 1:
            stack.append(v)

    # remaining graph is 2-regular: disjoint union of cycles
    for s in range(1, n + 1):
        if removed[s]:
            continue
        u = s
        cycle_nodes = []
        while not removed[u]:
            removed[u] = True
            cycle_nodes.append(u)
            nxt = -1
            for (w, eid) in adj[u]:
                if not removed[w]:
                    nxt = w
                    break
            if nxt == -1:
                break
            u = nxt
        L = len(cycle_nodes)
        if L == 0:
            continue
        if L % 2 == 1:
            sys.stdout.write("NO\n")
            return
        for i in range(L):
            a = cycle_nodes[i]
            b = cycle_nodes[(i + 1) % L]
            found = -1
            for (w, eid) in adj[a]:
                if w == b and color[eid] == 0:
                    found = eid
                    break
            if found == -1:
                for (w, eid) in adj[a]:
                    if w == b:
                        found = eid
                        break
            c = 1 if (i % 2 == 0) else 2
            color[found] = c
            used[a] |= (1 << c)
            used[b] |= (1 << c)

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color)) + "\n")

solve()
