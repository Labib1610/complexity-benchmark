```python
import sys
import math
import heapq
from collections import deque

sys.setrecursionlimit(1_000_000)


def output_coloring(colors):
    print("YES")
    print(*colors)
    sys.exit(0)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    incident = [[] for _ in range(n)]
    edges = []

    p = 2
    valid = (m * 2 + 2 == len(data))
    for eid in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if not (0 <= u < n and 0 <= v < n) or u == v:
            valid = False
        edges.append((u, v))
        if 0 <= u < n and 0 <= v < n:
            incident[u].append(eid)
            incident[v].append(eid)

    if not valid or any(len(incident[v]) != 3 for v in range(n)):
        print("NO")
        return

    # A cubic graph with a bridge cannot be 3-edge-colored.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False

    def bridge_dfs(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1
        for eid in incident[v]:
            if eid == parent_edge:
                continue
            a, b = edges[eid]
            to = b if a == v else a
            if tin[to] == -1:
                bridge_dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for s in range(n):
        if tin[s] == -1:
            bridge_dfs(s, -1)

    if has_bridge:
        print("NO")
        return

    # Bipartite cubic graphs are directly 3-edge-colorable.
    side = [-1] * n
    bipartite = True
    for s in range(n):
        if side[s] != -1:
            continue
        side[s] = 0
        q = deque([s])
        while q:
            v = q.popleft()
            for eid in incident[v]:
                a, b = edges[eid]
                to = b if a == v else a
                if side[to] == -1:
                    side[to] = side[v] ^ 1
                    q.append(to)
                elif side[to] == side[v]:
                    bipartite = False

    if bipartite:
        left = [v for v in range(n) if side[v] == 0]
        match_r = [-1] * n
        match_l = [-1] * n
        dist = [-1] * n

        def bfs_matching():
            q = deque()
            found = False
            for v in left:
                if match_l[v] == -1:
                    dist[v] = 0
                    q.append(v)
                else:
                    dist[v] = -1
            while q:
                v = q.popleft()
                for eid in incident[v]:
                    a, b = edges[eid]
                    r = b if a == v else a
                    old = match_r[r]
                    if old == -1:
                        found = True
                    else:
                        x, y = edges[old]
                        old_l = x if side[x] == 0 else y
                        if dist[old_l] == -1:
                            dist[old_l] = dist[v] + 1
                            q.append(old_l)
            return found

        def dfs_matching(v):
            for eid in incident[v]:
                a, b = edges[eid]
                r = b if a == v else a
                old = match_r[r]
                if old == -1:
                    match_l[v] = eid
                    match_r[r] = eid
                    return True
                x, y = edges[old]
                old_l = x if side[x] == 0 else y
                if dist[old_l] == dist[v] + 1 and dfs_matching(old_l):
                    match_l[v] = eid
                    match_r[r] = eid
                    return True
            dist[v] = -1
            return False

        matched = 0
        while bfs_matching():
            for v in left:
                if match_l[v] == -1 and dfs_matching(v):
                    matched += 1

        if matched == len(left):
            colors = [0] * m
            for v in left:
                colors[match_l[v]] = 1

            q = deque()
            for eid in range(m):
                if colors[eid] == 0:
                    colors[eid] = 2
                    q.append(eid)
                    while q:
                        e = q.popleft()
                        c = colors[e]
                        u, v = edges[e]
                        for x in (u, v):
                            for ne in incident[x]:
                                if ne == e or colors[ne] == 1:
                                    continue
                                wanted = 5 - c
                                if colors[ne] == 0:
                                    colors[ne] = wanted
                                    q.append(ne)
                                elif colors[ne] != wanted:
                                    print("NO")
                                    return
            output_coloring(colors)

    # SAT encoding for general cubic graphs.
    # Variable (edge, color) is true iff that edge has that color.
    num_vars = 3 * m
    clauses = []
    watches = [[] for _ in range(2 * num_vars)]
    watch_a = []
    watch_b = []

    def lit_index(lit):
        v = abs(lit) - 1
        return 2 * v + (1 if lit < 0 else 0)

    def add_clause(clause):
        ci = len(clauses)
        clauses.append(clause)
        if len(clause) >= 2:
            watch_a.append(0)
            watch_b.append(1)
            watches[lit_index(clause[0])].append(ci)
            watches[lit_index(clause[1])].append(ci)
        else:
            watch_a.append(0)
            watch_b.append(0)
        return ci

    def exactly_one(vars_):
        add_clause([v + 1 for v in vars_])
        add_clause([-(vars_[0] + 1), -(vars_[1] + 1)])
        add_clause([-(vars_[0] + 1), -(vars_[2] + 1)])
        add_clause([-(vars_[1] + 1), -(vars_[2] + 1)])

    for eid in range(m):
        exactly_one([3 * eid, 3 * eid + 1, 3 * eid + 2])

    for v in range(n):
        e0, e1, e2 = incident[v]
        for c in range(3):
            exactly_one([3 * e0 + c, 3 * e1 + c, 3 * e2 + c])

    assignment = [-1] * num_vars
    level = [0] * num_vars
    reason = [-1] * num_vars
    phase = [1] * num_vars
    trail = []
    trail_lim = []
    qhead = 0

    activity = [0.0] * num_vars
    activity_inc = 1.0
    activity_decay = 0.95
    heap = [(0.0, v) for v in range(num_vars)]
    heapq.heapify(heap)

    def current_level():
        return len(trail_lim)

    def literal_value(lit):
        val = assignment[abs(lit) - 1]
        if val == -1:
            return -1
        if lit > 0:
            return val
        return val ^ 1

    def enqueue(lit, why):
        v = abs(lit) - 1
        val = 1 if lit > 0 else 0
        if assignment[v] != -1:
            return assignment[v] == val
        assignment[v] = val
        phase[v] = val
        level[v] = current_level()
        reason[v] = why
        trail.append(lit)
        return True

    def propagate():
        nonlocal qhead
        while qhead < len(trail):
            true_lit = trail[qhead]
            qhead += 1
            false_lit = -true_lit
            wi = lit_index(false_lit)
            ws = watches[wi]
            i = 0

            while i < len(ws):
                ci = ws[i]
                clause = clauses[ci]
                wa = watch_a[ci]
                wb = watch_b[ci]

                if clause[wa] == false_lit:
                    false_pos = wa
                    other_pos = wb
                    false_is_a = True
                elif clause[wb] == false_lit:
                    false_pos = wb
                    other_pos = wa
                    false_is_a = False
                else:
                    ws[i] = ws[-1]
                    ws.pop()
                    continue

                other_lit = clause[other_pos]
                if literal_value(other_lit) == 1:
                    i += 1
                    continue

                replacement = -1
                for k, lit in enumerate(clause):
                    if k == other_pos or k == false_pos:
                        continue
                    if literal_value(lit) != 0:
                        replacement = k
                        break

                if replacement != -1:
                    new_lit = clause[replacement]
                    if false_is_a:
                        watch_a[ci] = replacement
                    else:
                        watch_b[ci] = replacement
                    ws[i] = ws[-1]
                    ws.pop()
                    watches[lit_index(new_lit)].append(ci)
                    continue

                other_val = literal_value(other_lit)
                if other_val == 0:
                    return ci
                if not enqueue(other_lit, ci):
                    return ci
                i += 1
        return -1

    def backtrack(target_level):
        nonlocal qhead
        if current_level() <= target_level:
            return
        target = trail_lim[target_level]
        for i in range(len(trail) - 1, target - 1, -1):
            v = abs(trail[i]) - 1
            assignment[v] = -1
            reason[v] = -1
            level[v] = 0
            heapq.heappush(heap, (-activity[v], v))
        del trail[target:]
        del trail_lim[target_level:]
        qhead = len(trail)

    def bump(v):
        nonlocal activity_inc
        activity[v] += activity_inc
        heapq.heappush(heap, (-activity[v], v))

    def analyze(conflict_ci):
        nonlocal activity_inc
        learned = [0]
        seen = [False] * num_vars
        touched = []
        path_count = 0
        idx = len(trail) - 1
        p_lit = 0
        ci = conflict_ci

        while True:
            clause = clauses[ci]
            pv = abs(p_lit) - 1 if p_lit else -1

            for lit in clause:
                v = abs(lit) - 1
                if v == pv or seen[v] or level[v] == 0:
                    continue
                seen[v] = True
                touched.append(v)
                bump(v)
                if level[v] == current_level():
                    path_count += 1
                else:
                    learned.append(lit)

            while True:
                p_lit = trail[idx]
                idx -= 1
                v = abs(p_lit) - 1
                if seen[v]:
                    break

            seen[v] = False
            path_count -= 1
            if path_count == 0:
                learned[0] = -p_lit
                break

            ci = reason[v]
            if ci < 0:
                learned[0] = -p_lit
                break

        for v in touched:
            seen[v] = False

        back_level = 0
        if len(learned) > 1:
            best = 1
            for i in range(2, len(learned)):
                if level[abs(learned[i]) - 1] > level[abs(learned[best]) - 1]:
                    best = i
            learned[1], learned[best] = learned[best], learned[1]
            back_level = level[abs(learned[1]) - 1]

        activity_inc /= activity_decay
        if activity_inc > 1e100:
            for v in range(num_vars):
                activity[v] *= 1e-100
            activity_inc *= 1e-100
            heap.clear()
            for v in range(num_vars):
                if assignment[v] == -1:
                    heapq.heappush(heap, (-activity[v], v))

        return learned, back_level

    def choose_variable():
        while heap:
            neg_act, v = heapq.heappop(heap)
            if assignment[v] == -1 and neg_act == -activity[v]:
                return v
        return -1

    # Fix the color permutation at one vertex.
    root_edges = incident[0]
    root_ok = True
    for c, eid in enumerate(root_edges):
        if not enqueue(3 * eid + c + 1, -1):
            root_ok = False
            break

    if not root_ok or propagate() != -1:
        print("NO")
        return

    conflicts_since_restart = 0
    restart_limit = 100

    while True:
        conflict = propagate()

        if conflict != -1:
            if current_level() == 0:
                print("NO")
                return

            learned, back_level = analyze(conflict)
            backtrack(back_level)
            ci = add_clause(learned)

            if not enqueue(learned[0], ci):
                print("NO")
                return

            conflicts_since_restart += 1
            if conflicts_since_restart >= restart_limit:
                backtrack(0)
                conflicts_since_restart = 0
                restart_limit = int(restart_limit * 1.5) + 1
            continue

        v = choose_variable()
        if v == -1:
            colors = [0] * m
            for eid in range(m):
                found = 0
                for c in range(3):
                    if assignment[3 * eid + c] == 1:
                        found = c + 1
                        break
                if found == 0:
                    print("NO")
                    return
                colors[eid] = found

            for vertex in range(n):
                if len({colors[eid] for eid in incident[vertex]}) != 3:
                    print("NO")
                    return

            output_coloring(colors)

        trail_lim.append(len(trail))
        decision_lit = v + 1 if phase[v] else -(v + 1)
        enqueue(decision_lit, -1)


if __name__ == "__main__":
    main()
```
