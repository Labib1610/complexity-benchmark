import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check if the graph is a 3-regular graph (each vertex has degree 3)
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    # Try to 3-color the edges (assign windows 1, 2, 3 to edges)
    # This is equivalent to edge-coloring a 3-regular graph with 3 colors
    # which is always possible for a 3-regular graph that has a perfect matching
    
    # We'll use a BFS-based approach to color the edges
    color = [0] * m  # 0 means uncolored
    colored = [False] * m
    
    # For each vertex, we need to assign 3 different colors to its 3 edges
    for v in range(1, n + 1):
        incident_edges = adj[v]
        used_colors = set()
        uncolored_edges = []
        
        # Check already colored edges and collect uncolored ones
        for neighbor, edge_idx in incident_edges:
            if colored[edge_idx]:
                used_colors.add(color[edge_idx])
            else:
                uncolored_edges.append(edge_idx)
        
        # Assign remaining colors to uncolored edges
        available_colors = {1, 2, 3} - used_colors
        available_colors = list(available_colors)
        
        if len(available_colors) < len(uncolored_edges):
            print("NO")
            return
            
        for i, edge_idx in enumerate(uncolored_edges):
            if i < len(available_colors):
                color[edge_idx] = available_colors[i]
                colored[edge_idx] = True
            else:
                # This shouldn't happen in a valid 3-regular graph
                print("NO")
                return
    
    # Verify the solution
    for v in range(1, n + 1):
        incident_edges = adj[v]
        colors_at_v = set()
        for neighbor, edge_idx in incident_edges:
            colors_at_v.add(color[edge_idx])
        if len(colors_at_v) != 3 or 0 in colors_at_v:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
