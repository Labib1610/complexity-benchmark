```python
import sys
import heapq

sys.setrecursionlimit(10000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

it = iter(data)
n = next(it)
m = next(it)

edges = []
adj = [[] for _ in range(n)]
for eid in range(m):
    u = next(it) - 1
    v = next(it) - 1
    edges.append((u, v))
    adj[u].append((v, eid))
    adj[v].append((u, eid))

if any(len(adj[v]) != 3 for v in range(n)):
    print("NO")
    sys.exit()

# Find bridges and connected components.
tin = [-1] * n
low = [0] * n
timer = 0
has_bridge = False
component_roots = []

def dfs(v, parent_edge):
    global timer, has_bridge
    tin[v] = low[v] = timer
    timer += 1
    for to, eid in adj[v]:
        if eid == parent_edge:
            continue
        if tin[to] == -1:
            dfs(to, eid)
            low[v] = min(low[v], low[to])
            if low[to] > tin[v]:
                has_bridge = True
        else:
            low[v] = min(low[v], tin[to])

for s in range(n):
    if tin[s] == -1:
        component_roots.append(s)
        dfs(s, -1)

if has_bridge:
    print("NO")
    sys.exit()

# SAT variables: variable (edge, color) is edge * 3 + color + 1.
num_vars = 3 * m
clauses = []

def var(e, c):
    return 3 * e + c + 1

def add_clause(lits):
    clauses.append(list(lits))

# Every edge has exactly one color.
for e in range(m):
    a, b, c = var(e, 0), var(e, 1), var(e, 2)
    add_clause((a, b, c))
    add_clause((-a, -b))
    add_clause((-a, -c))
    add_clause((-b, -c))

# At every vertex, every color appears exactly once.
for v in range(n):
    es = [eid for _, eid in adj[v]]
    for color in range(3):
        a, b, c = (var(es[i], color) for i in range(3))
        add_clause((a, b, c))
        add_clause((-a, -b))
        add_clause((-a, -c))
        add_clause((-b, -c))

# Break color symmetry independently in every connected component.
unit_clause_indices = []
for root in component_roots:
    es = [eid for _, eid in adj[root]]
    for color in range(3):
        unit_clause_indices.append(len(clauses))
        add_clause((var(es[color], color),))

watches = [[] for _ in range(2 * num_vars)]

def lit_index(lit):
    v = abs(lit) - 1
    return 2 * v + (1 if lit < 0 else 0)

for ci, clause in enumerate(clauses):
    if len(clause) >= 2:
        watches[lit_index(clause[0])].append(ci)
        watches[lit_index(clause[1])].append(ci)

assignment = [0] * (num_vars + 1)   # 1=true, -1=false
level = [0] * (num_vars + 1)
reason = [-1] * (num_vars + 1)
phase = [1] * (num_vars + 1)

trail = []
trail_lim = []
qhead = 0

activity = [0.0] * (num_vars + 1)
var_inc = 1.0
var_decay = 0.95
heap = []
stamp = [0] * (num_vars + 1)

def push_heap(v):
    stamp[v] += 1
    heapq.heappush(heap, (-activity[v], stamp[v], v))

for v in range(1, num_vars + 1):
    push_heap(v)

def lit_value(lit):
    a = assignment[abs(lit)]
    if a == 0:
        return 0
    return a if lit > 0 else -a

def enqueue(lit, why):
    v = abs(lit)
    wanted = 1 if lit > 0 else -1
    if assignment[v] != 0:
        return assignment[v] == wanted
    assignment[v] = wanted
    phase[v] = wanted
    level[v] = len(trail_lim)
    reason[v] = why
    trail.append(lit)
    return True

def propagate():
    global qhead
    while qhead < len(trail):
        true_lit = trail[qhead]
        qhead += 1
        false_lit = -true_lit
        wi = lit_index(false_lit)
        ws = watches[wi]
        i = 0

        while i < len(ws):
            ci = ws[i]
            clause = clauses[ci]

            if clause[0] == false_lit:
                clause[0], clause[1] = clause[1], clause[0]

            if clause[1] != false_lit:
                i += 1
                continue

            first = clause[0]
            if lit_value(first) == 1:
                i += 1
                continue

            replacement = -1
            for k in range(2, len(clause)):
                if lit_value(clause[k]) != -1:
                    replacement = k
                    break

            if replacement != -1:
                new_watch = clause[replacement]
                clause[1], clause[replacement] = new_watch, false_lit
                ws[i] = ws[-1]
                ws.pop()
                watches[lit_index(new_watch)].append(ci)
                continue

            if lit_value(first) == -1:
                return ci

            if not enqueue(first, ci):
                return ci
            i += 1

    return -1

def cancel_until(target_level):
    global qhead
    if len(trail_lim) <= target_level:
        return
    cut = trail_lim[target_level]
    for i in range(len(trail) - 1, cut - 1, -1):
        v = abs(trail[i])
        assignment[v] = 0
        level[v] = 0
        reason[v] = -1
        push_heap(v)
    del trail[cut:]
    del trail_lim[target_level:]
    qhead = len(trail)

seen = [False] * (num_vars + 1)

def bump_variable(v):
    global var_inc
    activity[v] += var_inc
    push_heap(v)

def rescale_activities():
    global var_inc
    for v in range(1, num_vars + 1):
        activity[v] *= 1e-100
    var_inc *= 1e-100
    for v in range(1, num_vars + 1):
        push_heap(v)

def analyze(conflict_ci):
    touched = []
    learned = [0]
    path_count = 0
    p = 0
    idx = len(trail) - 1
    ci = conflict_ci
    current_level = len(trail_lim)

    while True:
        clause = clauses[ci]
        pv = abs(p) if p else 0

        for q in clause:
            v = abs(q)
            if v == pv or seen[v] or level[v] == 0:
                continue
            seen[v] = True
            touched.append(v)
            bump_variable(v)
            if level[v] == current_level:
                path_count += 1
            else:
                learned.append(q)

        while idx >= 0 and not seen[abs(trail[idx])]:
            idx -= 1

        p = trail[idx]
        idx -= 1
        seen[abs(p)] = False
        path_count -= 1

        if path_count == 0:
            learned[0] = -p
            break

        ci = reason[abs(p)]
        if ci < 0:
            learned[0] = -p
            break

    for v in touched:
        seen[v] = False

    if len(learned) == 1:
        return learned, 0

    best = 1
    for i in range(2, len(learned)):
        if level[abs(learned[i])] > level[abs(learned[best])]:
            best = i
    learned[1], learned[best] = learned[best], learned[1]
    backtrack_level = level[abs(learned[1])]
    return learned, backtrack_level

def add_learned(clause):
    ci = len(clauses)
    clauses.append(clause)
    if len(clause) >= 2:
        watches[lit_index(clause[0])].append(ci)
        watches[lit_index(clause[1])].append(ci)
    return ci

def choose_variable():
    while heap:
        neg_act, st, v = heapq.heappop(heap)
        if st != stamp[v] or assignment[v] != 0:
            continue
        return v
    for v in range(1, num_vars + 1):
        if assignment[v] == 0:
            return v
    return 0

# Enqueue initial symmetry-breaking units.
for ci in unit_clause_indices:
    if not enqueue(clauses[ci][0], ci):
        print("NO")
        sys.exit()

conflict = propagate()
if conflict != -1:
    print("NO")
    sys.exit()

conflicts = 0
restart_limit = 100
conflicts_since_restart = 0
satisfiable = False

while True:
    conflict = propagate()

    if conflict != -1:
        conflicts += 1
        conflicts_since_restart += 1

        if len(trail_lim) == 0:
            break

        learned, backtrack_level = analyze(conflict)
        var_inc /= var_decay

        if var_inc > 1e100:
            rescale_activities()

        cancel_until(backtrack_level)
        ci = add_learned(learned)

        if not enqueue(learned[0], ci):
            break

        if conflicts_since_restart >= restart_limit:
            cancel_until(0)
            conflicts_since_restart = 0
            restart_limit = int(restart_limit * 1.5) + 1
        continue

    decision_var = choose_variable()
    if decision_var == 0:
        satisfiable = True
        break

    trail_lim.append(len(trail))
    decision_lit = decision_var if phase[decision_var] > 0 else -decision_var
    enqueue(decision_lit, -1)

if not satisfiable:
    print("NO")
else:
    colors = []
    valid = True
    for e in range(m):
        chosen = 0
        for c in range(3):
            if assignment[var(e, c)] == 1:
                chosen = c + 1
                break
        if chosen == 0:
            valid = False
            break
        colors.append(chosen)

    if not valid:
        print("NO")
    else:
        print("YES")
        print(*colors)
```
