import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    g = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        g[u].append((v, idx))
        g[v].append((u, idx))

    # find bridges
    tin = [-1] * n
    low = [0] * n
    timer = 0
    is_bridge = [False] * m

    def dfs(v, p_e):
        nonlocal timer
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in g[v]:
            if eid == p_e:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    is_bridge[eid] = True

    dfs(0, -1)
    if any(b for b in is_bridge):
        print("NO")
        return

    # Find a perfect matching that yields only even cycles after removal.
    # We'll use a simple augmenting‑path algorithm on the auxiliary graph
    # whose vertices are the original vertices and whose edges are the
    # original edges.  The matching is stored in match[v] = (neighbor, edge_id)
    match = [-1] * n
    used_edge = [False] * m

    # initial greedy matching
    for v in range(n):
        if match[v] == -1:
            for to, eid in g[v]:
                if match[to] == -1:
                    match[v] = eid
                    match[to] = eid
                    used_edge[eid] = True
                    break

    # augmenting path search (standard for general graphs, using blossom)
    # For n ≤ 2000 a straightforward Edmonds' blossom implementation is fast enough.
    from collections import deque

    def bfs(root):
        base = list(range(n))
        p = [-1] * n
        q = deque()
        in_q = [False] * n
        in_blossom = [False] * n

        def lca(a, b):
            visited = [False] * n
            while True:
                a = base[a]
                visited[a] = True
                if match[a] == -1:
                    break
                a = p[edges[match[a]][0] ^ edges[match[a]][1] ^ a]
            while True:
                b = base[b]
                if visited[b]:
                    return b
                b = p[edges[match[b]][0] ^ edges[match[b]][1] ^ b]

        def mark_path(v, b, child):
            while base[v] != b:
                in_blossom[base[v]] = in_blossom[base[match[v]]] = True
                p[v] = child
                child = edges[match[v]][0] ^ edges[match[v]][1] ^ v
                v = p[child]

        def blossom_contraction(v, u):
            b = lca(v, u)
            in_blossom[:] = [False] * n
            mark_path(v, b, u)
            mark_path(u, b, v)
            for i in range(n):
                if in_blossom[base[i]]:
                    base[i] = b
                    if not in_q[i]:
                        q.append(i)
                        in_q[i] = True

        def find_augmenting_path():
            q.clear()
            for i in range(n):
                p[i] = -1
                base[i] = i
                in_q[i] = False
            q.append(root)
            in_q[root] = True
            while q:
                v = q.popleft()
                for to, eid in g[v]:
                    if base[v] == base[to] or match[v] == eid:
                        continue
                    if to == root or (match[to] != -1 and p[edges[match[to]][0] ^ edges[match[to]][1] ^ to] != -1):
                        blossom_contraction(v, to)
                    elif p[to] == -1:
                        p[to] = v
                        if match[to] == -1:
                            return to
                        nxt = edges[match[to]][0] ^ edges[match[to]][1] ^ to
                        if not in_q[nxt]:
                            q.append(nxt)
                            in_q[nxt] = True
            return -1

        finish = find_augmenting_path()
        if finish == -1:
            return False
        # augment
        v = finish
        while v != -1:
            pv = p[v]
            ppv = edges[match[pv]][0] ^ edges[match[pv]][1] ^ pv if pv != -1 else -1
            me = -1
            # edge between pv and v
            for to, eid in g[pv]:
                if to == v:
                    me = eid
                    break
            match[pv] = me
            match[v] = me
            used_edge[me] = True
            v = ppv
        return True

    # augment until perfect
    while True:
        free = -1
        for i in range(n):
            if match[i] == -1:
                free = i
                break
        if free == -1:
            break
        bfs(free)

    # now match is a perfect matching
    # remove matching edges, remaining graph is a collection of cycles
    color = [0] * m
    for eid in range(m):
        if used_edge[eid]:
            color[eid] = 1

    visited = [False] * n
    for start in range(n):
        if visited[start]:
            continue
        # traverse cycle
        v = start
        cycle_edges = []
        while True:
            visited[v] = True
            # find a non‑matching edge
            for to, eid in g[v]:
                if color[eid] == 0:
                    cycle_edges.append(eid)
                    v = to
                    break
            else:
                break
            if visited[v]:
                break
        # assign alternating colors 2 and 3
        cur = 2
        for eid in cycle_edges:
            color[eid] = cur
            cur = 5 - cur  # toggle between 2 and 3

    print("YES")
    print(' '.join(str(c) for c in color))

if __name__ == "__main__":
    solve()
