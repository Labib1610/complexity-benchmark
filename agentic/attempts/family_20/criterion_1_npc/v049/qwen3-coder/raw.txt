```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    # Store edges with their indices
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Each node must have exactly 3 edges
    for u in range(1, n + 1):
        if len(adj[u]) != 3:
            print("NO")
            return
    
    # Try to 3-color the edges such that at each vertex, all 3 edges have different colors
    # This is equivalent to finding a proper edge coloring of a 3-regular graph with 3 colors
    # Since each vertex has degree 3 and we have 3 colors, this is possible iff the graph is 3-edge-colorable
    
    # We'll use BFS to try to construct such a coloring
    color = [0] * m  # 0 means uncolored
    colored = [False] * m
    
    # For each vertex, we need to assign 3 different colors to its 3 edges
    # We can model this as trying to find a proper edge coloring
    
    # Start with any vertex and assign colors to its edges
    # Then propagate through the graph
    
    queue = deque()
    visited_nodes = set()
    
    # Process components one by one
    for start_node in range(1, n + 1):
        if start_node in visited_nodes:
            continue
            
        queue.append(start_node)
        visited_nodes.add(start_node)
        
        # Keep track of which colors are used at each node
        node_colors = defaultdict(set)  # node -> set of colors used
        node_edge_colors = defaultdict(dict)  # node -> {edge_index: color}
        
        while queue:
            node = queue.popleft()
            
            # Get all edges of this node
            node_edges = adj[node]
            
            # Check which edges are already colored
            available_colors = {1, 2, 3}
            uncolored_edges = []
            
            for neighbor, edge_idx in node_edges:
                if colored[edge_idx]:
                    c = color[edge_idx]
                    if c in available_colors:
                        available_colors.remove(c)
                    node_colors[node].add(c)
                    node_edge_colors[node][edge_idx] = c
                else:
                    uncolored_edges.append((neighbor, edge_idx))
            
            # Assign remaining colors to uncolored edges
            color_list = list(available_colors)
            if len(color_list) != len(uncolored_edges):
                # Not enough colors - impossible
                print("NO")
                return
                
            for i, (neighbor, edge_idx) in enumerate(uncolored_edges):
                if i >= len(color_list):
                    print("NO")
                    return
                c = color_list[i]
                color[edge_idx] = c
                colored[edge_idx] = True
                node_colors[node].add(c)
                node_edge_colors[node][edge_idx] = c
                
                # Also update the neighbor's info
                if neighbor not in visited_nodes:
                    queue.append(neighbor)
                    visited_nodes.add(neighbor)
                
                # Check if neighbor already has conflicting information
                if neighbor in node_edge_colors:
                    neighbor_colors = set()
                    conflict = False
                    for e_idx, nc in node_edge_colors[neighbor].items():
                        if colored[e_idx] and color[e_idx] != nc:
                            conflict = True
                            break
                        neighbor_colors.add(nc)
                    if conflict:
                        print("NO")
                        return
    
    # Verify the solution
    node_edge_check = defaultdict(list)
    for i, (u, v) in enumerate(edges):
        node_edge_check[u].append(color[i])
        node_edge_check[v].append(color[i])
    
    for node in range(1, n + 1):
        if len(node_edge_check[node]) != 3:
            print("NO")
            return
        if len(set(node_edge_check[node])) != 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

solve()
```
