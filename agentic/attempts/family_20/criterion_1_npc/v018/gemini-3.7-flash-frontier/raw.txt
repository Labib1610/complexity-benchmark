import sys

# Increase recursion depth just in case
sys.setrecursionlimit(20000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    edges_u = [0] * m
    edges_v = [0] * m
    adj = [[] for _ in range(n + 1)]

    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges_u[i] = u
        edges_v[i] = v
        adj[u].append(i)
        adj[v].append(i)

    # Check degrees
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return

    # Find connected components of vertices
    visited = [False] * (n + 1)
    components = []
    for i in range(1, n + 1):
        if not visited[i]:
            comp_nodes = []
            queue = [i]
            visited[i] = True
            for node in queue:
                comp_nodes.append(node)
                for e in adj[node]:
                    neighbor = edges_v[e] if edges_u[e] == node else edges_u[e]
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
            components.append(comp_nodes)

    final_colors = [0] * m

    for comp in components:
        # Get all edges in this component
        comp_edges = set()
        for u in comp:
            for e in adj[u]:
                comp_edges.add(e)
        comp_edges = list(comp_edges)

        # Solver for this component
        # mask: 1 -> color 0 (1), 2 -> color 1 (2), 4 -> color 2 (3)
        mask = [7] * m
        assigned = [False] * m
        assignment = [-1] * m

        # Trail for backtracking
        trail = []

        def set_mask(e, new_m):
            if mask[e] != new_m:
                trail.append((e, mask[e], assigned[e]))
                mask[e] = new_m

        def assign(e, c):
            # c in {0, 1, 2}
            bit = 1 << c
            if (mask[e] & bit) == 0:
                return False
            if not assigned[e]:
                trail.append((e, mask[e], False))
                assigned[e] = True
                mask[e] = bit
                assignment[e] = c

            u = edges_u[e]
            v = edges_v[e]

            # Propagate to other edges at u
            for e_other in adj[u]:
                if e_other != e:
                    if (mask[e_other] & bit) != 0:
                        new_m = mask[e_other] & ~bit
                        if new_m == 0:
                            return False
                        set_mask(e_other, new_m)

            # Propagate to other edges at v
            for e_other in adj[v]:
                if e_other != e:
                    if (mask[e_other] & bit) != 0:
                        new_m = mask[e_other] & ~bit
                        if new_m == 0:
                            return False
                        set_mask(e_other, new_m)

            return True

        def propagate():
            changed = True
            while changed:
                changed = False
                # Rule 1: Any unassigned edge with single available color
                for e in comp_edges:
                    if not assigned[e]:
                        m_val = mask[e]
                        if m_val == 0:
                            return False
                        # Check if single bit
                        if (m_val & (m_val - 1)) == 0:
                            c = (m_val.bit_length()) - 1
                            if not assign(e, c):
                                return False
                            changed = True

                # Rule 2: For each vertex and each color, at least one incident edge must support it
                for u in comp:
                    for c in range(3):
                        bit = 1 << c
                        count = 0
                        last_e = -1
                        has_assigned = False
                        for e in adj[u]:
                            if assigned[e] and assignment[e] == c:
                                has_assigned = True
                                break
                            if (mask[e] & bit) != 0:
                                count += 1
                                last_e = e
                        if has_assigned:
                            continue
                        if count == 0:
                            return False
                        if count == 1:
                            if not assigned[last_e]:
                                if not assign(last_e, c):
                                    return False
                                changed = True
            return True

        def backtrack():
            if not propagate():
                return False

            # Find unassigned edge with minimum available colors
            best_e = -1
            min_cand = 4

            for e in comp_edges:
                if not assigned[e]:
                    cnt = bin(mask[e]).count("1")
                    if cnt < min_cand:
                        min_cand = cnt
                        best_e = e
                        if cnt == 2:
                            break

            if best_e == -1:
                return True

            saved_trail_len = len(trail)
            m_val = mask[best_e]

            for c in range(3):
                if (m_val & (1 << c)) != 0:
                    if assign(best_e, c):
                        if backtrack():
                            return True
                    # Undo trail
                    while len(trail) > saved_trail_len:
                        e_undo, m_undo, a_undo = trail.pop()
                        mask[e_undo] = m_undo
                        assigned[e_undo] = a_undo
                        if not a_undo:
                            assignment[e_undo] = -1

            return False

        # Break symmetry for this component: fix vertex comp[0]'s 3 incident edges
        root_v = comp[0]
        r_edges = adj[root_v]
        ok = True
        for c in range(3):
            if not assign(r_edges[c], c):
                ok = False
                break

        if not ok or not backtrack():
            print("NO")
            return

        for e in comp_edges:
            final_colors[e] = assignment[e] + 1

    print("YES")
    print(*(final_colors[i] for i in range(m)))


if __name__ == "__main__":
    solve()
