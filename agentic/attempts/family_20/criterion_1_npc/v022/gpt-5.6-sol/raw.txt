import sys
import heapq

sys.setrecursionlimit(1000000)


class CDCLSolver:
    def __init__(self, num_vars):
        self.n = num_vars
        self.clauses = []
        self.watches = [[] for _ in range(2 * num_vars)]

        self.assign = [-1] * (num_vars + 1)
        self.level = [0] * (num_vars + 1)
        self.reason = [-1] * (num_vars + 1)
        self.phase = [0] * (num_vars + 1)

        self.trail = []
        self.trail_lim = []
        self.qhead = 0

        self.activity = [0.0] * (num_vars + 1)
        self.var_inc = 1.0
        self.heap = [(0.0, v) for v in range(1, num_vars + 1)]
        heapq.heapify(self.heap)

        self.seen = [False] * (num_vars + 1)

    @staticmethod
    def lit_index(lit):
        v = abs(lit) - 1
        return (v << 1) | (1 if lit < 0 else 0)

    def value(self, lit):
        a = self.assign[abs(lit)]
        if a < 0:
            return -1
        return a if lit > 0 else 1 - a

    def add_clause(self, clause):
        ci = len(self.clauses)
        self.clauses.append(clause)
        if len(clause) >= 2:
            self.watches[self.lit_index(clause[0])].append(ci)
            self.watches[self.lit_index(clause[1])].append(ci)
        return ci

    def enqueue(self, lit, reason):
        v = abs(lit)
        val = 1 if lit > 0 else 0
        old = self.assign[v]
        if old >= 0:
            return old == val
        self.assign[v] = val
        self.phase[v] = val
        self.level[v] = len(self.trail_lim)
        self.reason[v] = reason
        self.trail.append(lit)
        return True

    def propagate(self):
        while self.qhead < len(self.trail):
            p = self.trail[self.qhead]
            self.qhead += 1
            false_lit = -p
            wi = self.lit_index(false_lit)
            ws = self.watches[wi]
            i = 0

            while i < len(ws):
                ci = ws[i]
                clause = self.clauses[ci]

                if clause[0] != false_lit:
                    if clause[1] == false_lit:
                        clause[0], clause[1] = clause[1], clause[0]
                    else:
                        i += 1
                        continue

                other = clause[1]
                if self.value(other) == 1:
                    i += 1
                    continue

                replacement = -1
                for k in range(2, len(clause)):
                    if self.value(clause[k]) != 0:
                        replacement = k
                        break

                if replacement >= 0:
                    new_lit = clause[replacement]
                    clause[0], clause[replacement] = new_lit, false_lit
                    ws[i] = ws[-1]
                    ws.pop()
                    self.watches[self.lit_index(new_lit)].append(ci)
                    continue

                ov = self.value(other)
                if ov == 0:
                    return ci

                if not self.enqueue(other, ci):
                    return ci
                i += 1

        return -1

    def bump_var(self, v):
        self.activity[v] += self.var_inc
        heapq.heappush(self.heap, (-self.activity[v], v))

        if self.activity[v] > 1e100:
            for x in range(1, self.n + 1):
                self.activity[x] *= 1e-100
            self.var_inc *= 1e-100
            self.heap = [
                (-self.activity[x], x)
                for x in range(1, self.n + 1)
                if self.assign[x] < 0
            ]
            heapq.heapify(self.heap)

    def analyze(self, conflict_ci):
        learnt = [0]
        path_count = 0
        p = 0
        idx = len(self.trail) - 1
        clause = self.clauses[conflict_ci]
        touched = []

        while True:
            pv = abs(p) if p else 0
            for q in clause:
                v = abs(q)
                if v == pv or self.seen[v] or self.level[v] == 0:
                    continue
                self.seen[v] = True
                touched.append(v)
                self.bump_var(v)
                if self.level[v] == len(self.trail_lim):
                    path_count += 1
                else:
                    learnt.append(q)

            while idx >= 0 and not self.seen[abs(self.trail[idx])]:
                idx -= 1

            p = self.trail[idx]
            idx -= 1
            self.seen[abs(p)] = False
            path_count -= 1

            if path_count == 0:
                learnt[0] = -p
                break

            ri = self.reason[abs(p)]
            clause = self.clauses[ri]

        for v in touched:
            self.seen[v] = False

        if len(learnt) == 1:
            back_level = 0
        else:
            best = 1
            for i in range(2, len(learnt)):
                if self.level[abs(learnt[i])] > self.level[abs(learnt[best])]:
                    best = i
            learnt[1], learnt[best] = learnt[best], learnt[1]
            back_level = self.level[abs(learnt[1])]

        return learnt, back_level

    def backtrack(self, target_level):
        if len(self.trail_lim) <= target_level:
            return

        start = self.trail_lim[target_level]
        for i in range(len(self.trail) - 1, start - 1, -1):
            v = abs(self.trail[i])
            self.assign[v] = -1
            self.reason[v] = -1
            self.level[v] = 0
            heapq.heappush(self.heap, (-self.activity[v], v))

        del self.trail[start:]
        del self.trail_lim[target_level:]
        self.qhead = len(self.trail)

    def pick_branch_var(self):
        while self.heap:
            _, v = heapq.heappop(self.heap)
            if self.assign[v] < 0:
                return v
        return 0

    def solve(self, units):
        for lit in units:
            if not self.enqueue(lit, -1):
                return False

        if self.propagate() >= 0:
            return False

        conflicts_since_restart = 0
        restart_limit = 100

        while True:
            conflict = self.propagate()

            if conflict >= 0:
                if not self.trail_lim:
                    return False

                learnt, back_level = self.analyze(conflict)
                self.var_inc /= 0.95
                self.backtrack(back_level)

                if len(learnt) == 1:
                    if not self.enqueue(learnt[0], -1):
                        return False
                else:
                    ci = self.add_clause(learnt)
                    if not self.enqueue(learnt[0], ci):
                        return False

                conflicts_since_restart += 1
                if conflicts_since_restart >= restart_limit:
                    self.backtrack(0)
                    conflicts_since_restart = 0
                    restart_limit = min(int(restart_limit * 1.5) + 1, 5000)
                continue

            v = self.pick_branch_var()
            if v == 0:
                return True

            self.trail_lim.append(len(self.trail))
            lit = v if self.phase[v] == 1 else -v
            self.enqueue(lit, -1)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    edges = []
    incident = [[] for _ in range(n)]

    pos = 2
    for eid in range(m):
        u = data[pos] - 1
        v = data[pos + 1] - 1
        pos += 2
        edges.append((u, v))
        incident[u].append(eid)
        incident[v].append(eid)

    # A cubic graph with a bridge cannot have a proper 3-edge-coloring.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False

    def dfs_bridge(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1
        for eid in incident[v]:
            if eid == parent_edge:
                continue
            a, b = edges[eid]
            to = b if a == v else a
            if tin[to] < 0:
                dfs_bridge(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for v in range(n):
        if tin[v] < 0:
            dfs_bridge(v, -1)

    if has_bridge:
        print("NO")
        return

    def var(eid, color):
        return eid * 3 + color + 1

    solver = CDCLSolver(3 * m)

    # Every edge receives exactly one color.
    for eid in range(m):
        x0 = var(eid, 0)
        x1 = var(eid, 1)
        x2 = var(eid, 2)
        solver.add_clause([x0, x1, x2])
        solver.add_clause([-x0, -x1])
        solver.add_clause([-x0, -x2])
        solver.add_clause([-x1, -x2])

    # At each vertex, no two incident edges may use the same color.
    for v in range(n):
        es = incident[v]
        if len(es) != 3:
            print("NO")
            return
        for color in range(3):
            a = var(es[0], color)
            b = var(es[1], color)
            c = var(es[2], color)
            solver.add_clause([-a, -b])
            solver.add_clause([-a, -c])
            solver.add_clause([-b, -c])

    # Break the global color-permutation symmetry in every connected component.
    visited = [False] * n
    units = []

    for start in range(n):
        if visited[start]:
            continue
        stack = [start]
        visited[start] = True
        while stack:
            v = stack.pop()
            for eid in incident[v]:
                a, b = edges[eid]
                to = b if a == v else a
                if not visited[to]:
                    visited[to] = True
                    stack.append(to)

        es = incident[start]
        units.append(var(es[0], 0))
        units.append(var(es[1], 1))
        units.append(var(es[2], 2))

    if not solver.solve(units):
        print("NO")
        return

    answer = []
    for eid in range(m):
        chosen = 0
        for color in range(3):
            if solver.assign[var(eid, color)] == 1:
                chosen = color + 1
                break
        if chosen == 0:
            print("NO")
            return
        answer.append(chosen)

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
