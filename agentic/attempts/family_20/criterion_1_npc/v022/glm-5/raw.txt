Looking at this problem, I need to assign frequencies 1, 2, 3 to edges of a 3-regular graph such that at each vertex, all three incident edges have distinct frequencies.

**Key Insight:** This is equivalent to finding a proper 3-edge-coloring of a cubic (3-regular) graph. By Vizing's theorem, a 3-regular graph needs either 3 or 4 colors. A graph is 3-edge-colorable (class 1) if and only if it's bridgeless (has no cut-edges) — this is a consequence of König's theorem for bipartite graphs and the more general result that a cubic graph is 3-edge-colorable iff it's bridgeless.

**Algorithm:**
1. Check if the graph is bridgeless using DFS
2. If bridgeless, find a proper 3-edge-coloring using a Kempe-chain based approach

For the coloring, I use a greedy approach with Kempe chains: when an edge can't be colored directly, I find a Kempe chain (alternating path between two colors) and swap colors along it to free up the needed color.

```python
import sys
from collections import defaultdict

def solve():
    input_data = sys.stdin.read().split()
    idx = 0
    n = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(input_data[idx]); idx += 1
        v = int(input_data[idx]); idx += 1
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check for bridges using DFS
    disc = [0] * (n + 1)
    low = [0] * (n + 1)
    timer = [1]
    has_bridge = [False]
    
    def dfs(u, pe):
        disc[u] = low[u] = timer[0]
        timer[0] += 1
        for v, e in adj[u]:
            if e == pe:
                continue
            if disc[v] == 0:
                dfs(v, e)
                low[u] = min(low[u], low[v])
                if low[v] > disc[u]:
                    has_bridge[0] = True
            else:
                low[u] = min(low[u], disc[v])
    
    for i in range(1, n + 1):
        if disc[i] == 0:
            dfs(i, -1)
    
    if has_bridge[0]:
        print("NO")
        return
    
    # Find proper 3-edge-coloring using Kempe chains
    color = [0] * m
    edge_idx = defaultdict(dict)
    for i, (u, v) in enumerate(edges):
        edge_idx[u][v] = i
        edge_idx[v][u] = i
    
    def get_color(u, v):
        return color[edge_idx[u][v]]
    
    def set_color(u, v, c):
        color[edge_idx[u][v]] = c
    
    def used_colors(u):
        return {color[e] for _, e in adj[u]}
    
    def find_alt_path(u, c1, c2, exclude_edge):
        """Find alternating c1-c2 path from u, return list of (vertex, edge) or None"""
        path = []
        visited = {u}
        cur = u
        while True:
            found = None
            for v, e in adj[cur]:
                if e == exclude_edge:
                    continue
                if color[e] == c1 and v not in visited:
                    found = (v, e)
                    break
            if found is None:
                break
            v, e = found
            path.append((v, e))
            visited.add(v)
            cur = v
            c1, c2 = c2, c1
        return path
    
    for i, (u, v) in enumerate(edges):
        used_u = used_colors(u)
        used_v = used_colors(v)
        avail_u = {1, 2, 3} - used_u
        avail_v = {1, 2, 3} - used_v
        common = avail_u & avail_v
        
        if common:
            c = min(common)
            color[i] = c
        else:
            # Need Kempe chain manipulation
            c_u = min(avail_u)
            c_v = min(avail_v)
            
            # Find alternating path from u using colors c_v, c_u
            path = find_alt_path(u, c_v, c_u, i)
            
            if path and path[-1][0] != v:
                # Swap colors along path
                for _, e in path:
                    if color[e] == c_v:
                        color[e] = c_u
                    else:
                        color[e] = c_v
                color[i] = c_v
            else:
                # Try the other direction
                path = find_alt_path(v, c_u, c_v, i)
                if path and path[-1][0] != u:
                    for _, e in path:
                        if color[e] == c_u:
                            color[e] = c_v
                        else:
                            color[e] = c_u
                    color[i] = c_u
                else:
                    # Try all color combinations
                    done = False
                    for c1 in [1, 2, 3]:
                        for c2 in [1, 2, 3]:
                            if c1 == c2:
                                continue
                            if c1 not in avail_u or c2 not in avail_v:
                                continue
                            path = find_alt_path(u, c2, c1, i)
                            if path and path[-1][0] != v:
                                for _, e in path:
                                    if color[e] == c2:
                                        color[e] = c1
                                    else:
                                        color[e] = c2
                                color[i] = c2
                                done = True
                                break
                        if done:
                            break
                    if not done:
                        color[i] = c_u  # fallback
    
    # Verify coloring
    for u in range(1, n + 1):
        cols = [color[e] for _, e in adj[u]]
        if len(set(cols)) != 3:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))
```

Actually, for cubic graphs, there's a cleaner approach. Since the graph is bridgeless and 3-regular, it has a perfect matching (by Petersen's theorem). Remove the perfect matching, and the remaining 2-regular graph is a union of cycles, which can be 2-colored. The perfect matching edges get the third color.

```python
        if disc[i] == 0:
            dfs(i, -1)
    
    if has_bridge[0]:
        print("NO")
        return
    
    # Find perfect matching using augmenting paths
    match = [0] * (n + 1)  # match[v] = edge index, or -1 if unmatched
    for i in range(n + 1):
        match[i] = -1
    
    def find_augmenting_path(start):
        """BFS to find augmenting path from unmatched vertex start"""
        parent = {start: (None, None)}  # vertex -> (prev_vertex, edge_used)
        queue = deque([start])
        while queue:
            u = queue.popleft()
            for v, e in adj[u]:
                if v in parent:
                    continue
                if match[v] == -1:
                    # Found augmenting path
                    parent[v] = (u, e)
                    # Reconstruct and augment
                    cur = v
                    while parent[cur][0] is not None:
                        pu, pe = parent[cur]
                        match[pu] = pe
                        match[cur] = pe
                        cur = pu
                    return True
                else:
                    # v is matched, continue through matched edge
                    me = match[v]
                    w = edges[me][0] if edges[me][1] == v else edges[me][1]
                    if w not in parent:
                        parent[v] = (u, e)
                        parent[w] = (v, me)
                        queue.append(w)
        return False
    
    for v in range(1, n + 1):
        if match[v] == -1:
            find_augmenting_path(v)
    
    # Extract matching edges
    matching_edges = set()
    for v in range(1, n + 1):
        if match[v] != -1:
            matching_edges.add(match[v])
    
    # Build graph without matching edges
    adj2 = [[] for _ in range(n + 1)]
    for i, (u, v) in enumerate(edges):
        if i not in matching_edges:
            adj2[u].append((v, i))
            adj2[v].append((u, i))
    
    # Find cycles and 2-color them
    color = [0] * m
    visited = [False] * (n + 1)
    
    for start in range(1, n + 1):
        if visited[start]:
            continue
        # Trace cycle
        cycle = []
        cur = start
        prev = -1
        while not visited[cur]:
            visited[cur] = True
            # Find next neighbor that's not prev
            next_v = None
            next_e = None
            for v, e in adj2[cur]:
                if v != prev:
                    next_v = v
                    next_e = e
                    break
            if next_v is None:
                break
            cycle.append((cur, next_e))
            prev = cur
            cur = next_v
            if cur == start:
                break
        
        # 2-color the cycle
        for idx_c, (vertex, e) in enumerate(cycle):
            color[e] = 1 if idx_c % 2 == 0 else 2
    
    # Color matching edges with 3
    for e in matching_edges:
        color[e] = 3
    
    print("YES")
    print(' '.join(map(str, color)))
```
