import sys
from collections import deque

sys.setrecursionlimit(20000)

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if len(data) < 2:
        return

    n, m = data[0], data[1]
    edges = []
    incident = [[] for _ in range(n)]

    p = 2
    valid = True
    for eid in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        if u == v or not (0 <= u < n and 0 <= v < n):
            valid = False
        edges.append((u, v))
        if 0 <= u < n and 0 <= v < n:
            incident[u].append(eid)
            incident[v].append(eid)

    if not valid or 2 * m != 3 * n or any(len(x) != 3 for x in incident):
        print("NO")
        return

    def other(e, v):
        a, b = edges[e]
        return b if a == v else a

    # A cubic graph containing a bridge cannot be 3-edge-colored.
    tin = [-1] * n
    low = [0] * n
    timer = 0
    has_bridge = False

    def bridge_dfs(v, parent_edge):
        nonlocal timer, has_bridge
        tin[v] = low[v] = timer
        timer += 1
        for e in incident[v]:
            if e == parent_edge:
                continue
            to = other(e, v)
            if tin[to] == -1:
                bridge_dfs(to, e)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    has_bridge = True
            else:
                low[v] = min(low[v], tin[to])

    for s in range(n):
        if tin[s] == -1:
            bridge_dfs(s, -1)

    if has_bridge:
        print("NO")
        return

    # Find connected components and test bipartiteness.
    side = [-1] * n
    components = []
    component_bipartite = []

    for s in range(n):
        if side[s] != -1:
            continue
        q = deque([s])
        side[s] = 0
        vertices = []
        bip = True
        while q:
            v = q.popleft()
            vertices.append(v)
            for e in incident[v]:
                to = other(e, v)
                if side[to] == -1:
                    side[to] = side[v] ^ 1
                    q.append(to)
                elif side[to] == side[v]:
                    bip = False
        components.append(vertices)
        component_bipartite.append(bip)

    preset = [0] * m

    # Cubic bipartite components are always 3-edge-colorable.
    for vertices, bip in zip(components, component_bipartite):
        if not bip:
            continue

        left = [v for v in vertices if side[v] == 0]
        match_right = [-1] * n
        seen = [0] * n
        token = 0

        def augment(u):
            nonlocal token
            for e in incident[u]:
                v = other(e, u)
                if seen[v] == token:
                    continue
                seen[v] = token
                old = match_right[v]
                if old == -1 or augment(other(old, v)):
                    match_right[v] = e
                    return True
            return False

        possible = True
        for u in left:
            token += 1
            if not augment(u):
                possible = False
                break

        if not possible:
            print("NO")
            return

        matched = set()
        for v in vertices:
            if side[v] == 1:
                e = match_right[v]
                if e == -1:
                    print("NO")
                    return
                matched.add(e)
                preset[e] = 1

        residual = [[] for _ in vertices]
        pos = {v: i for i, v in enumerate(vertices)}
        for v in vertices:
            for e in incident[v]:
                if e not in matched:
                    residual[pos[v]].append(e)

        for e in set(e for v in vertices for e in incident[v] if e not in matched):
            if preset[e] != 0:
                continue
            preset[e] = 2
            q = deque(edges[e])
            while q:
                v = q.popleft()
                r = residual[pos[v]]
                e1, e2 = r[0], r[1]
                c1, c2 = preset[e1], preset[e2]
                if c1 and c2:
                    if c1 == c2:
                        print("NO")
                        return
                elif c1:
                    preset[e2] = 6 ^ c1
                    a, b = edges[e2]
                    q.append(a)
                    q.append(b)
                elif c2:
                    preset[e1] = 6 ^ c2
                    a, b = edges[e1]
                    q.append(a)
                    q.append(b)

    # Precompute generalized arc consistency for a 3-way all-different constraint.
    support = [None] * 512
    permutations = (
        (1, 2, 4), (1, 4, 2),
        (2, 1, 4), (2, 4, 1),
        (4, 1, 2), (4, 2, 1)
    )

    for a in range(1, 8):
        for b in range(1, 8):
            for c in range(1, 8):
                sa = sb = sc = 0
                for x, y, z in permutations:
                    if (a & x) and (b & y) and (c & z):
                        sa |= x
                        sb |= y
                        sc |= z
                support[(a << 6) | (b << 3) | c] = (
                    (sa, sb, sc) if sa else None
                )

    domains = [preset[e] if preset[e] else 7 for e in range(m)]
    trail = []
    queue = deque()
    in_queue = [False] * n

    def enqueue(v):
        if not in_queue[v]:
            in_queue[v] = True
            queue.append(v)

    def set_domain(e, value):
        old = domains[e]
        if old == value:
            return
        trail.append((e, old))
        domains[e] = value
        u, v = edges[e]
        enqueue(u)
        enqueue(v)

    # Fix color symmetry independently in each non-bipartite component.
    for vertices, bip in zip(components, component_bipartite):
        if bip:
            continue
        root = vertices[0]
        for e, color in zip(incident[root], (1, 2, 4)):
            set_domain(e, color)

    for v in range(n):
        enqueue(v)

    def clear_queue():
        while queue:
            v = queue.popleft()
            in_queue[v] = False

    def propagate():
        while queue:
            v = queue.popleft()
            in_queue[v] = False
            e0, e1, e2 = incident[v]
            a, b, c = domains[e0], domains[e1], domains[e2]
            s = support[(a << 6) | (b << 3) | c]
            if s is None:
                return False
            if s[0] != a:
                set_domain(e0, s[0])
            if s[1] != b:
                set_domain(e1, s[1])
            if s[2] != c:
                set_domain(e2, s[2])
        return True

    if not propagate():
        print("NO")
        return

    bit_count = [0, 1, 1, 2, 1, 2, 2, 3]

    def search():
        chosen = -1
        best_size = 4

        for e, d in enumerate(domains):
            cnt = bit_count[d]
            if 1 < cnt < best_size:
                best_size = cnt
                chosen = e
                if cnt == 2:
                    break

        if chosen == -1:
            return True

        mask = domains[chosen]
        values = []
        for color in (1, 2, 4):
            if mask & color:
                values.append(color)

        for color in values:
            checkpoint = len(trail)
            set_domain(chosen, color)

            if propagate() and search():
                return True

            clear_queue()
            while len(trail) > checkpoint:
                e, old = trail.pop()
                domains[e] = old

        return False

    if not search():
        print("NO")
        return

    answer = []
    for d in domains:
        if d == 1:
            answer.append("1")
        elif d == 2:
            answer.append("2")
        elif d == 4:
            answer.append("3")
        else:
            print("NO")
            return

    print("YES")
    print(" ".join(answer))

if __name__ == "__main__":
    main()
