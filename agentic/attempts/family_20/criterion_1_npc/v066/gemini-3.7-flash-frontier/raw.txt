import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(200000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[m_idx := 1])
    
    edges = []
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges.append((u, v))
        
    incident = [[] for _ in range(n + 1)]
    for e_idx, (u, v) in enumerate(edges):
        incident[u].append(e_idx)
        incident[v].append(e_idx)
        
    adj = [[] for _ in range(m)]
    for e_idx, (u, v) in enumerate(edges):
        nbrs = set(incident[u] + incident[v])
        nbrs.discard(e_idx)
        adj[e_idx] = list(nbrs)
        
    # Find connected components of edges
    visited_edge = [False] * m
    components = []
    
    for i in range(m):
        if not visited_edge[i]:
            comp = []
            queue = [i]
            visited_edge[i] = True
            for edge in queue:
                comp.append(edge)
                for nbr in adj[edge]:
                    if not visited_edge[nbr]:
                        visited_edge[nbr] = True
                        queue.append(nbr)
            components.append(comp)
            
    POPCOUNT = [0, 1, 1, 2, 1, 2, 2, 3]
    color_res = [0] * m
    
    for comp in components:
        comp_set = set(comp)
        uncolored = set(comp)
        mask = {e: 7 for e in comp}
        color = {e: 0 for e in comp}
        
        def search():
            if not uncolored:
                return True
                
            best_e = -1
            best_cnt = 4
            
            for e in uncolored:
                cnt = POPCOUNT[mask[e]]
                if cnt < best_cnt:
                    best_cnt = cnt
                    best_e = e
                    if cnt == 1:
                        break
                        
            if best_cnt == 0:
                return False
                
            m_val = mask[best_e]
            uncolored.remove(best_e)
            
            # Try available colors
            for c_bit in (1, 2, 4):
                if m_val & c_bit:
                    color[best_e] = c_bit
                    saved = []
                    conflict = False
                    
                    for nbr in adj[best_e]:
                        if nbr in comp_set and color[nbr] == 0:
                            if mask[nbr] & c_bit:
                                saved.append((nbr, mask[nbr]))
                                mask[nbr] &= ~c_bit
                                if mask[nbr] == 0:
                                    conflict = True
                                    
                    if not conflict:
                        if search():
                            return True
                            
                    for nbr, old_m in reversed(saved):
                        mask[nbr] = old_m
                    color[best_e] = 0
                    
            uncolored.add(best_e)
            return False

        if not search():
            print("NO")
            return

        for e in comp:
            c = color[e]
            if c == 1:
                color_res[e] = 1
            elif c == 2:
                color_res[e] = 2
            elif c == 4:
                color_res[e] = 3

    print("YES")
    print(*(color_res[i] for i in range(m)))

if __name__ == '__main__':
    main()
