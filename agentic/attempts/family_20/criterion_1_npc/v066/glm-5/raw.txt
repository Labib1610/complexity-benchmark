import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n + 1)]
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Check degree constraint
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return
    
    # Build adjacency list with edge indices for traversal
    adj_edges = [[] for _ in range(n + 1)]
    for i, (u, v) in enumerate(edges):
        adj_edges[u].append((v, i))
        adj_edges[v].append((u, i))
    
    visited_edge = [False] * m
    visited_node = [False] * (n + 1)
    cycles = []
    
    # Find all cycles
    for i in range(m):
        if not visited_edge[i]:
            # BFS to find cycle
            u, v = edges[i]
            parent = {u: None}
            q = deque([u])
            found = False
            start_node = u
            end_node = None
            meet_edge = None
            
            while q and not found:
                curr = q.popleft()
                for neighbor, eid in adj_edges[curr]:
                    if eid == i:
                        continue
                    if neighbor not in parent:
                        parent[neighbor] = (curr, eid)
                        q.append(neighbor)
                    else:
                        # Found cycle
                        end_node = curr
                        meet_edge = eid
                        found = True
                        break
            
            if not found:
                print("NO")
                return
            
            # Reconstruct cycle
            cycle_edges = [i]
            # Trace back from end_node
            path = []
            curr = end_node
            while curr is not None:
                path.append(curr)
                if parent[curr] is None:
                    curr = None
                else:
                    curr = parent[curr][0]
            
            # Trace from the other side
            curr = edges[meet_edge][0] if edges[meet_edge][1] == end_node else edges[meet_edge][1]
            other_path = []
            while curr is not None:
                other_path.append(curr)
                if parent[curr] is None:
                    curr = None
                else:
                    curr = parent[curr][0]
            
            # Find meeting point
            path_set = set(path)
            meet_point = None
            for node in other_path:
                if node in path_set:
                    meet_point = node
                    break
            
            if meet_point is None:
                print("NO")
                return
            
            # Build cycle edges
            cycle_edges = []
            # Add edge i
            cycle_edges.append(i)
            visited_edge[i] = True
            
            # Add edges from end_node to meet_point
            curr = end_node
            while curr != meet_point:
                p = parent[curr]
                cycle_edges.append(p[1])
                visited_edge[p[1]] = True
                curr = p[0]
            
            # Add edges from the other path
            curr = edges[meet_edge][0] if edges[meet_edge][1] == end_node else edges[meet_edge][1]
            while curr != meet_point:
                p = parent[curr]
                cycle_edges.append(p[1])
                visited_edge[p[1]] = True
                curr = p[0]
            
            cycles.append(cycle_edges)
    
    # Assign colors
    color = [0] * m
    for cycle in cycles:
        for idx, eid in enumerate(cycle):
            color[eid] = (idx % 2) + 1
    
    # Verify
    for u in range(1, n + 1):
        cnt = [0, 0, 0, 0]
        for v, eid in adj_edges[u]:
            cnt[color[eid]] += 1
        if cnt[1] != 1 or cnt[2] != 1 or cnt[3] != 1:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
