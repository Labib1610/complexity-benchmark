import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    import sys
    n, m = map(int, sys.stdin.readline().split())
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # check bridges using DFS
    tin = [-1]*n
    low = [0]*n
    timer = 0
    is_bridge = [False]*m

    def dfs(v, p_e):
        nonlocal timer
        tin[v] = low[v] = timer
        timer += 1
        for to, eid in adj[v]:
            if eid == p_e:
                continue
            if tin[to] != -1:
                low[v] = min(low[v], tin[to])
            else:
                dfs(to, eid)
                low[v] = min(low[v], low[to])
                if low[to] > tin[v]:
                    is_bridge[eid] = True

    dfs(0, -1)
    if any(b for b in is_bridge):
        print("NO")
        return

    # ---- Edmonds' blossom for maximum matching ----
    N = n
    g = [[] for _ in range(N)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    match = [-1]*N
    p = [-1]*N
    base = list(range(N))
    q = []
    used = [False]*N
    blossom = [False]*N

    def lca(a, b):
        used_path = [False]*N
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal q, used, p, base, blossom
        used = [False]*N
        p = [-1]*N
        base = list(range(N))
        q = []
        q_head = 0
        q.append(root)
        used[root] = True
        while q_head < len(q):
            v = q[q_head]
            q_head += 1
            for to in g[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    curbase = lca(v, to)
                    blossom = [False]*N
                    mark_path(v, curbase, to)
                    mark_path(to, curbase, v)
                    for i in range(N):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        return to
                    to_match = match[to]
                    used[to_match] = True
                    q.append(to_match)
        return -1

    def augment_path(finish):
        v = finish
        while v != -1:
            pv = p[v]
            ppv = match[pv] if pv != -1 else -1
            match[v] = pv
            match[pv] = v
            v = ppv

    for v in range(N):
        if match[v] == -1:
            finish = find_path(v)
            if finish != -1:
                augment_path(finish)

    if any(mch == -1 for mch in match):
        print("NO")
        return

    # Build matching set
    in_match = set()
    for v in range(N):
        u = match[v]
        if v < u:
            # find edge id
            for to, eid in adj[v]:
                if to == u:
                    in_match.add(eid)
                    break

    # assign colors
    color = [0]*m
    for eid in in_match:
        color[eid] = 1

    # remaining edges form cycles
    visited_edge = [False]*m
    for eid in in_match:
        visited_edge[eid] = True

    for start in range(m):
        if visited_edge[start]:
            continue
        # traverse cycle
        cycle = []
        cur_e = start
        while not visited_edge[cur_e]:
            visited_edge[cur_e] = True
            cycle.append(cur_e)
            u, v = edges[cur_e]
            # move to next edge at the other vertex not used yet
            nxt = None
            for to, ne in adj[u]:
                if ne != cur_e and not visited_edge[ne]:
                    nxt = ne
                    break
            if nxt is None:
                for to, ne in adj[v]:
                    if ne != cur_e and not visited_edge[ne]:
                        nxt = ne
                        break
            cur_e = nxt if nxt is not None else -1
        # color alternately 2,3
        for i, eid in enumerate(cycle):
            color[eid] = 2 if i % 2 == 0 else 3

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
