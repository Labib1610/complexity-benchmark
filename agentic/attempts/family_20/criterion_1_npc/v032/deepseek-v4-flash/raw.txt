We can model this as a **3-edge-coloring** problem on a **3-regular graph** (since every hub has degree 3 and there are 3 phases).  

A 3-regular graph has a proper 3-edge-coloring if and only if it is **bridgeless** (by **Petersen’s theorem**).  
So the algorithm is:

1. Check if the graph is 3-regular (degree of each vertex = 3).
2. Check if the graph is bridgeless (no edge whose removal disconnects the graph).
3. If both hold, find a proper 3-edge-coloring using a standard algorithm (e.g., via **Kőnig’s line coloring** or a backtracking approach since n ≤ 2000).

We can find the coloring by:
- Using a **greedy + backtracking** approach, but that may be too slow for n=2000.
- Better: Use **Kőnig’s theorem** for bipartite graphs, but here the graph is not necessarily bipartite.
- Since the graph is 3-regular and bridgeless, we can use **Petersen’s theorem** to decompose into a 1-factor and a 2-factor, then color the 2-factor with 2 colors and the 1-factor with the third color.

Steps:
1. Find a perfect matching (1-factor) using **Blossom algorithm** (or since n ≤ 2000, we can use networkx? But we must implement ourselves).
2. Remove the matching edges; the remaining graph is 2-regular (a disjoint union of cycles), which is 2-edge-colorable.
3. Color the cycles alternately with colors 1 and 2, and the matching edges with color 3.

We need to implement a **maximum matching** for general graphs (Blossom).  
Since n ≤ 2000, we can implement an efficient O(n^3) Blossom algorithm.

Let’s write the solution.

```python
import sys
sys.setrecursionlimit(10000)

def find_3_edge_coloring(n, edges):
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        adj[u].append((v, idx))
        adj[v].append((u, idx))
    
    # Check degree
    for i in range(n):
        if len(adj[i]) != 3:
            return None
    
    # Blossom algorithm for maximum matching
    # We'll implement Edmonds' Blossom algorithm
    # Returns a list of matched edges (u,v) or None if no perfect matching
    
    # Implementation of Blossom for general graph
    # Based on standard O(n^3) implementation
    
    n_vertices = n
    # We'll store graph as adjacency list of vertices
    graph = [[] for _ in range(n_vertices)]
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    # Blossom algorithm
    match = [-1] * n_vertices
    p = [-1] * n_vertices
    base = list(range(n_vertices))
    q = []
    used = [False] * n_vertices
    blossom = [False] * n_vertices
    
    def lca(a, b):
        used_lca = [False] * n_vertices
        while True:
            a = base[a]
            used_lca[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_lca[b]:
                return b
            b = p[match[b]]
    
    def mark_path(v, b, children):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = children
            children = match[v]
            v = p[match[v]]
    
    def find_path(root):
        nonlocal used, p, base, q, blossom
        used = [False] * n_vertices
        p = [-1] * n_vertices
        base = list(range(n_vertices))
        q = [root]
        used[root] = True
        for qi in range(len(q)):
            v = q[qi]
            for u in graph[v]:
                if base[v] == base[u] or match[v] == u:
                    continue
                if u == root or (match[u] != -1 and p[match[u]] != -1):
                    curbase = lca(v, u)
                    blossom = [False] * n_vertices
                    mark_path(v, curbase, u)
                    mark_path(u, curbase, v)
                    for i in range(n_vertices):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[u] == -1:
                    p[u] = v
                    if match[u] == -1:
                        return u
                    u = match[u]
                    used[u] = True
                    q.append(u)
        return -1
    
    def augment(v):
        while v != -1:
            pv = p[v]
            nv = match[pv]
            match[v] = pv
            match[pv] = v
            v = nv
    
    for i in range(n_vertices):
        if match[i] == -1:
            v = find_path(i)
            if v != -1:
                augment(v)
    
    # Check perfect matching
    for i in range(n_vertices):
        if match[i] == -1:
            return None
    
    # Build matching edge list
    matching_edges = []
    for u in range(n_vertices):
        v = match[u]
        if u < v:
            matching_edges.append((u, v))
    
    # Remove matching edges, remaining graph is 2-regular (cycles)
    # Build adjacency for remaining graph
    rem_adj = [[] for _ in range(n_vertices)]
    for u, v in edges:
        if match[u] != v:  # not a matching edge
            rem_adj[u].append(v)
            rem_adj[v].append(u)
    
    # Color the cycles with 1 and 2
    color = [-1] * len(edges)
    # Map edge index to color
    edge_index = {}
    for idx, (u, v) in enumerate(edges):
        edge_index[(u, v)] = idx
        edge_index[(v, u)] = idx
    
    # Color matching edges with 3
    for u, v in matching_edges:
        color[edge_index[(u, v)]] = 3
    
    # Color cycles
    visited = [False] * n_vertices
    for i in range(n_vertices):
        if not visited[i] and len(rem_adj[i]) > 0:
            # traverse cycle
            cycle_vertices = []
            cur = i
            prev = -1
            while True:
                visited[cur] = True
                cycle_vertices.append(cur)
                # find next
                nxt = -1
                for nb in rem_adj[cur]:
                    if nb != prev:
                        nxt = nb
                        break
                if nxt == -1:
                    break
                prev, cur = cur, nxt
                if cur == i:
                    break
            # color edges alternately
            for j in range(len(cycle_vertices)):
                u = cycle_vertices[j]
                v = cycle_vertices[(j+1) % len(cycle_vertices)]
                if (u, v) in edge_index:
                    color[edge_index[(u, v)]] = 1 if j % 2 == 0 else 2
                else:
                    color[edge_index[(v, u)]] = 1 if j % 2 == 0 else 2
    
    return color

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
    
    color = find_3_edge_coloring(n, edges)
    if color is None:
        print("NO")
    else:
        print("YES")
        print(" ".join(str(c) for c in color))

if __name__ == "__main__":
    main()
```
