```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])

    edges = []
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)

    # Check if all degrees are 3
    for i in range(1, n + 1):
        if len(adj[i]) != 3:
            print("NO")
            return

    # Edge colors: 0 (unassigned), 1, 2, 3
    color = [0] * m
    # For each edge, a bitmask of available colors: bit 1 (1<<1), bit 2 (1<<2), bit 3 (1<<3)
    # 14 = (1<<1) | (1<<2) | (1<<3) = 2 | 4 | 8 = 14
    ALL_MASK = 14
    domain = [ALL_MASK] * m

    # For each vertex, which colors are used (bitmask)
    vert_used = [0] * (n + 1)

    # Find connected components of edges
    visited_v = [False] * (n + 1)
    components = []
    for start_v in range(1, n + 1):
        if not visited_v[start_v]:
            comp_edges = []
            q = [start_v]
            visited_v[start_v] = True
            comp_verts = [start_v]
            while q:
                curr = q.pop()
                for e_idx in adj[curr]:
                    u, v = edges[e_idx]
                    nxt = v if curr == u else u
                    if not visited_v[nxt]:
                        visited_v[nxt] = True
                        q.append(nxt)
                        comp_verts.append(nxt)
            components.append(comp_verts)

    # History stack for backtracking
    history = []

    def set_color(e, c):
        """Attempts to set edge e to color c. Returns True if consistent, False otherwise."""
        mask = 1 << c
        if not (domain[e] & mask):
            return False

        u, v = edges[e]
        if (vert_used[u] & mask) or (vert_used[v] & mask):
            return False

        # Apply assignment
        changes = []  # will store (edge, old_domain) or ('vert', u/v, old_mask) or ('color', e, old_c)
        changes.append(("color", e, color[e]))
        color[e] = c

        changes.append(("domain", e, domain[e]))
        domain[e] = mask

        changes.append(("vert", u, vert_used[u]))
        vert_used[u] |= mask

        changes.append(("vert", v, vert_used[v]))
        vert_used[v] |= mask

        # Propagate to neighbors
        queue = [e]
        while queue:
            curr_e = queue.pop()
            c_curr = color[curr_e]
            m_curr = 1 << c_curr
            cu, cv = edges[curr_e]

            for vert in (cu, cv):
                for nb_e in adj[vert]:
                    if nb_e == curr_e:
                        continue
                    if domain[nb_e] & m_curr:
                        changes.append(("domain", nb_e, domain[nb_e]))
                        domain[nb_e] &= ~m_curr
                        d = domain[nb_e]
                        if d == 0:
                            # Conflict, rollback
                            rollback(changes)
                            return False
                        # If domain size is 1 and edge is not yet colored
                        if color[nb_e] == 0 and (d & (d - 1)) == 0:
                            # Exactly one color possible
                            c_nb = d.bit_length() - 1
                            nu, nv = edges[nb_e]
                            if (vert_used[nu] & (1 << c_nb)) or (
                                vert_used[nv] & (1 << c_nb)
                            ):
                                rollback(changes)
                                return False
                            changes.append(("color", nb_e, color[nb_e]))
                            color[nb_e] = c_nb
                            changes.append(("vert", nu, vert_used[nu]))
                            vert_used[nu] |= 1 << c_nb
                            changes.append(("vert", nv, vert_used[nv]))
                            vert_used[nv] |= 1 << c_nb
                            queue.append(nb_e)

        history.append(changes)
        return True

    def rollback(changes):
        for item in reversed(changes):
            typ = item[0]
            if typ == "color":
                color[item[1]] = item[2]
            elif typ == "domain":
                domain[item[1]] = item[2]
            elif typ == "vert":
                vert_used[item[1]] = item[2]

    def undo():
        changes = history.pop()
        rollback(changes)

    # Initialize symmetry breaking per component
    for comp in components:
        v0 = comp[0]
        e0, e1, e2 = adj[v0]
        if not set_color(e0, 1):
            print("NO")
            return
        if not set_color(e1, 2):
            print("NO")
            return
        if not set_color(e2, 3):
            print("NO")
            return

    # Backtracking search with MRV (Minimum Remaining Values)
    def backtrack():
        # Find uncolored edge with minimum domain size
        best_e = -1
        min_choices = 4

        for i in range(m):
            if color[i] == 0:
                d = domain[i]
                cnt = (
                    (1 if d & 2 else 0)
                    + (1 if d & 4 else 0)
                    + (1 if d & 8 else 0)
                )
                if cnt < min_choices:
                    min_choices = cnt
                    best_e = i
                    if min_choices <= 2:
                        break

        if best_e == -1:
            return True

        d = domain[best_e]
        for c in (1, 2, 3):
            if d & (1 << c):
                if set_color(best_e, c):
                    if backtrack():
                        return True
                    undo()
        return False

    if backtrack():
        print("YES")
        print(*(color[i] for i in range(m)))
    else:
        print("NO")


if __name__ == "__main__":
    solve()
```
