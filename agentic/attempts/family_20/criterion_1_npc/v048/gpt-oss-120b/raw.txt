import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    edges = []
    adj = [[] for _ in range(n)]
    for idx in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
        adj[u].append((v, idx))
        adj[v].append((u, idx))

    # ---------- Edmonds' blossom for maximum matching ----------
    N = n
    g = [[] for _ in range(N)]
    for u, v in edges:
        g[u].append(v)
        g[v].append(u)

    match = [-1] * N
    p = [-1] * N
    base = list(range(N))
    q = []
    used = [False] * N
    blossom = [False] * N

    def lca(a, b):
        used_path = [False] * N
        while True:
            a = base[a]
            used_path[a] = True
            if match[a] == -1:
                break
            a = p[match[a]]
        while True:
            b = base[b]
            if used_path[b]:
                return b
            b = p[match[b]]

    def mark_path(v, b, x):
        while base[v] != b:
            blossom[base[v]] = blossom[base[match[v]]] = True
            p[v] = x
            x = match[v]
            v = p[match[v]]

    def find_path(root):
        nonlocal q, used, p, base, blossom
        used = [False] * N
        p = [-1] * N
        base = list(range(N))
        q = []
        q_head = 0
        q.append(root)
        used[root] = True
        while q_head < len(q):
            v = q[q_head]
            q_head += 1
            for to in g[v]:
                if base[v] == base[to] or match[v] == to:
                    continue
                if to == root or (match[to] != -1 and p[match[to]] != -1):
                    curbase = lca(v, to)
                    blossom = [False] * N
                    mark_path(v, curbase, to)
                    mark_path(to, curbase, v)
                    for i in range(N):
                        if blossom[base[i]]:
                            base[i] = curbase
                            if not used[i]:
                                used[i] = True
                                q.append(i)
                elif p[to] == -1:
                    p[to] = v
                    if match[to] == -1:
                        return to
                    to_match = match[to]
                    used[to_match] = True
                    q.append(to_match)
        return -1

    def augment_path(finish):
        v = finish
        while v != -1:
            pv = p[v]
            ppv = match[pv] if pv != -1 else -1
            match[v] = pv
            match[pv] = v
            v = ppv

    for v in range(N):
        if match[v] == -1:
            finish = find_path(v)
            if finish != -1:
                augment_path(finish)

    # ---------- Build coloring ----------
    if any(mch == -1 for mch in match):
        print("NO")
        return

    color = [0] * m
    # color matching edges with 1
    for idx, (u, v) in enumerate(edges):
        if match[u] == v:
            color[idx] = 1

    # remaining edges form cycles, color them 2/3
    visited_edge = [False] * m
    for idx in range(m):
        if color[idx] != 0:
            visited_edge[idx] = True

    # adjacency with edge indices
    edge_id = [{} for _ in range(n)]
    for idx, (u, v) in enumerate(edges):
        edge_id[u][v] = idx
        edge_id[v][u] = idx

    for start in range(n):
        for v, eid in adj[start]:
            if visited_edge[eid]:
                continue
            # traverse the cycle
            cycle = []
            cur = start
            prev = -1
            while True:
                # find next edge not visited
                for nxt, neid in adj[cur]:
                    if neid == prev:
                        continue
                    if visited_edge[neid]:
                        continue
                    # this edge is part of the 2-regular subgraph
                    cycle.append(neid)
                    visited_edge[neid] = True
                    prev = neid
                    cur = nxt
                    break
                else:
                    break
                if cur == start:
                    break
            # color alternating 2,3
            for i, eid in enumerate(cycle):
                color[eid] = 2 if i % 2 == 0 else 3

    if any(c == 0 for c in color):
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
