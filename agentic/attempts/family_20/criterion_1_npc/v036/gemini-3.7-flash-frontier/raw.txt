Answering the problem with an efficient implementation of Kempe-chain edge coloring for 3-regular graphs.

```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2
        
    # adj[u][c] = (neighbor, edge_index) colored with color c (c in 0, 1, 2)
    adj = [[None] * 3 for _ in range(n)]
    edge_color = [-1] * m
    
    import random
    rng = random.Random(42)
    
    for e_idx, (u, v) in enumerate(edges):
        steps = 0
        max_steps = 2000
        while steps < max_steps:
            steps += 1
            # Missing colors at u and v
            m_u = [c for c in range(3) if adj[u][c] is None]
            m_v = [c for c in range(3) if adj[v][c] is None]
            
            common = set(m_u) & set(m_v)
            if common:
                c = next(iter(common))
                edge_color[e_idx] = c
                adj[u][c] = (v, e_idx)
                adj[v][c] = (u, e_idx)
                break
            
            c1 = rng.choice(m_u)
            c2 = rng.choice(m_v)
            
            # Trace (c1, c2) path from u
            # Since c1 is missing at u, u has no c1 edge. 
            # Wait, the path from u starts with c2!
            curr = u
            color_to_follow = c2
            path = []
            
            while curr is not None and adj[curr][color_to_follow] is not None:
                nxt, edge_i = adj[curr][color_to_follow]
                path.append((curr, nxt, edge_i, color_to_follow))
                curr = nxt
                color_to_follow = c1 if color_to_follow == c2 else c2
            
            if curr != v:
                # The path does not reach v. Invert colors along the path.
                for n1, n2, edge_i, col in path:
                    new_col = c1 if col == c2 else c2
                    adj[n1][col] = None
                    adj[n2][col] = None
                    adj[n1][new_col] = (n2, edge_i)
                    adj[n2][new_col] = (n1, edge_i)
                    edge_color[edge_i] = new_col
                
                # Now c2 is missing at u, and c2 was missing at v
                edge_color[e_idx] = c2
                adj[u][c2] = (v, e_idx)
                adj[v][c2] = (u, e_idx)
                break
            else:
                # The (c1, c2) path reaches v.
                # Flip the (c1, c3) chain from u, where c3 is the remaining color.
                c3 = 3 - c1 - c2
                curr = u
                color_to_follow = c3
                path = []
                while curr is not None and adj[curr][color_to_follow] is not None:
                    nxt, edge_i = adj[curr][color_to_follow]
                    path.append((curr, nxt, edge_i, color_to_follow))
                    curr = nxt
                    color_to_follow = c1 if color_to_follow == c3 else c3
                
                for n1, n2, edge_i, col in path:
                    new_col = c1 if col == c3 else c3
                    adj[n1][col] = None
                    adj[n2][col] = None
                    adj[n1][new_col] = (n2, edge_i)
                    adj[n2][new_col] = (n1, edge_i)
                    edge_color[edge_i] = new_col
        else:
            print("NO")
            return

    print("YES")
    print(*(c + 1 for c in edge_color))

solve()
```
