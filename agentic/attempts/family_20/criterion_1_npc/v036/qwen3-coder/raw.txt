```python
import sys
from collections import defaultdict

def solve():
    n, m = map(int, input().split())
    
    # Since m = 3n/2 and each node has the same degree
    # Total degree = 2m = 3n
    # So each node has degree 3
    degree = 3
    
    edges = []
    adj = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
        adj[u].append(i)
        adj[v].append(i)
    
    # Try to color the edges with 3 colors such that
    # at each vertex, all incident edges have different colors
    
    # This is edge coloring of a 3-regular graph with 3 colors
    # Since n is even and we have a 3-regular graph, this is always possible
    # We can use a greedy approach or specific construction
    
    # For a 3-regular graph with n even, we can always 3-edge-color it
    # Let's try a greedy approach: for each edge, assign the smallest color
    # that hasn't been used by its adjacent edges at either endpoint
    
    color = [0] * m  # Colors: 1, 2, 3
    used_colors = defaultdict(set)  # used_colors[v] = set of colors used at vertex v
    
    for i in range(m):
        u, v = edges[i]
        
        # Find the smallest available color (1, 2, or 3)
        # that is not used at both u and v
        forbidden = used_colors[u] | used_colors[v]
        c = 1
        while c in forbidden:
            c += 1
            
        color[i] = c
        used_colors[u].add(c)
        used_colors[v].add(c)
    
    # Check if we successfully colored with only 3 colors
    if max(color) <= 3 and min(color) >= 1:
        print("YES")
        print(' '.join(map(str, color)))
    else:
        # This should not happen for valid inputs, but let's handle it
        print("NO")

solve()
```
