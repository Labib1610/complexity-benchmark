import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N1 = int(next(it)); N2 = int(next(it)); M = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(M)]

    deg1 = [0] * N1
    deg2 = [0] * N2
    for u, v in edges:
        deg1[u] += 1
        deg2[v] += 1
    K = max(max(deg1) if N1 else 0, max(deg2) if N2 else 0)

    # for each vertex, color -> edge id
    col_edge1 = [{} for _ in range(N1)]
    col_edge2 = [{} for _ in range(N2)]
    # used colors sets
    used1 = [set() for _ in range(N1)]
    used2 = [set() for _ in range(N2)]

    ans = [0] * M

    from collections import deque

    for idx, (u, v) in enumerate(edges):
        # try to find free color
        free = None
        for c in range(1, K + 1):
            if c not in used1[u] and c not in used2[v]:
                free = c
                break
        if free is not None:
            ans[idx] = free
            used1[u].add(free); col_edge1[u][free] = idx
            used2[v].add(free); col_edge2[v][free] = idx
            continue

        # need to recolor via alternating path
        # pick any color a not in used1[u]
        a = next(c for c in range(1, K + 1) if c not in used1[u])
        # pick any color b not in used2[v]
        b = next(c for c in range(1, K + 1) if c not in used2[v])

        # BFS on vertices of both sides, states are (side, vertex, color)
        # we look for a vertex where color a is free on its opposite side
        # classic alternating path between colors a and b
        parent = dict()  # key: (side, vertex, color) -> previous state
        q = deque()
        # start from u with color a on side 0 (primary)
        q.append((0, u, a))
        parent[(0, u, a)] = None
        end_state = None

        while q and end_state is None:
            side, vert, col = q.popleft()
            if side == 0:  # primary vertex, look at edge with this color if exists
                if col in col_edge1[vert]:
                    e_id = col_edge1[vert][col]
                    _, v2 = edges[e_id]
                    # move to secondary vertex with same color
                    nxt = (1, v2, col)
                    if nxt not in parent:
                        parent[nxt] = (side, vert, col)
                        q.append(nxt)
                else:
                    # free on primary side -> we can stop
                    end_state = (side, vert, col)
                    break
            else:  # secondary side
                if col in col_edge2[vert]:
                    e_id = col_edge2[vert][col]
                    u1, _ = edges[e_id]
                    nxt = (0, u1, col)
                    if nxt not in parent:
                        parent[nxt] = (side, vert, col)
                        q.append(nxt)
                else:
                    end_state = (side, vert, col)
                    break

        # reconstruct path and flip colors a<->b along it
        # we know start is (0,u,a), target is a free spot (could be on either side)
        cur = end_state
        while cur is not None:
            side, vert, col = cur
            prev = parent[cur]
            if prev is not None:
                pside, pvert, pcol = prev
                # edge between pvert and vert has color pcol (=col)
                # we will swap its color to the other one (a<->b)
                # actually along the alternating path we just toggle between a and b,
                # but since we only used two colors a and b, we can just set edge to the other color.
                e_id = col_edge1[pvert][pcol] if pside == 0 else col_edge2[pvert][pcol]
                # remove old assignment
                if pside == 0:
                    used1[pvert].remove(pcol); del col_edge1[pvert][pcol]
                    used2[vert].remove(pcol); del col_edge2[vert][pcol]
                else:
                    used2[pvert].remove(pcol); del col_edge2[pvert][pcol]
                    used1[vert].remove(pcol); del col_edge1[vert][pcol]
                # new color is the other one (a if pcol==b else b)
                newcol = a if pcol == b else b
                # assign
                if pside == 0:
                    used1[pvert].add(newcol); col_edge1[pvert][newcol] = e_id
                    used2[vert].add(newcol); col_edge2[vert][newcol] = e_id
                else:
                    used2[pvert].add(newcol); col_edge2[pvert][newcol] = e_id
                    used1[vert].add(newcol); col_edge1[vert][newcol] = e_id
                ans[e_id] = newcol
            cur = prev

        # now color a is free on u and b is free on v, assign a to current edge
        ans[idx] = a
        used1[u].add(a); col_edge1[u][a] = idx
        used2[v].add(a); col_edge2[v][a] = idx

    out = ['YES', ' '.join(str(x) for x in ans)]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
