We can model this as a bipartite graph edge-coloring problem.  
Since the graph is bipartite, we can color edges with exactly Δ colors, where Δ is the maximum degree.  

We can use a simple greedy algorithm with alternating path recoloring to achieve Δ colors.  

Here’s the implementation:  

```python
import sys
from collections import deque

input = sys.stdin.readline

def solve():
    N1, N2, M = map(int, input().split())
    edges = []
    adj1 = [[] for _ in range(N1)]
    adj2 = [[] for _ in range(N2)]
    
    for i in range(M):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adj1[u].append((v, i))
        adj2[v].append((u, i))
    
    # Find maximum degree
    deg1 = [len(adj1[i]) for i in range(N1)]
    deg2 = [len(adj2[i]) for i in range(N2)]
    K = max(max(deg1) if deg1 else 0, max(deg2) if deg2 else 0)
    
    color = [0] * M
    # color of edge incident to a vertex
    color_of_edge1 = [[-1] * K for _ in range(N1)]
    color_of_edge2 = [[-1] * K for _ in range(N2)]
    
    for idx, (u, v) in enumerate(edges):
        # find free color for u
        c1 = -1
        for c in range(K):
            if color_of_edge1[u][c] == -1:
                c1 = c
                break
        c2 = -1
        for c in range(K):
            if color_of_edge2[v][c] == -1:
                c2 = c
                break
        
        if c1 == c2:
            color[idx] = c1
            color_of_edge1[u][c1] = idx
            color_of_edge2[v][c1] = idx
        else:
            # recolor path
            # start from v with color c1
            current_vertex = v
            current_side = 2
            current_color = c1
            path_edges = []
            while True:
                if current_side == 2:
                    # find edge with color current_color incident to current_vertex
                    eid = color_of_edge2[current_vertex][current_color]
                    if eid == -1:
                        break
                    path_edges.append(eid)
                    # move to other side
                    u2, v2 = edges[eid]
                    current_vertex = u2
                    current_side = 1
                    current_color = c2 if current_color == c1 else c1
                else:
                    eid = color_of_edge1[current_vertex][current_color]
                    if eid == -1:
                        break
                    path_edges.append(eid)
                    u2, v2 = edges[eid]
                    current_vertex = v2
                    current_side = 2
                    current_color = c2 if current_color == c1 else c1
            
            # flip colors along path
            for eid in path_edges:
                old_c = color[eid]
                new_c = c2 if old_c == c1 else c1
                u2, v2 = edges[eid]
                color_of_edge1[u2][old_c] = -1
                color_of_edge2[v2][old_c] = -1
                color[eid] = new_c
                color_of_edge1[u2][new_c] = eid
                color_of_edge2[v2][new_c] = eid
            
            # assign new edge
            color[idx] = c1
            color_of_edge1[u][c1] = idx
            color_of_edge2[v][c1] = idx
    
    print("YES")
    print(" ".join(str(c + 1) for c in color))

if __name__ == "__main__":
    solve()
```
