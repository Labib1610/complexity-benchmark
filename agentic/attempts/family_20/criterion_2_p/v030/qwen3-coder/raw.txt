```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    N1 = int(data[0])
    N2 = int(data[1])
    M = int(data[2])
    
    tasks = []
    primary_count = [0] * (N1 + 1)
    secondary_count = [0] * (N2 + 1)
    
    index = 3
    for i in range(M):
        u = int(data[index])
        v = int(data[index + 1])
        tasks.append((u, v))
        primary_count[u] += 1
        secondary_count[v] += 1
        index += 2
    
    K = max(max(primary_count), max(secondary_count))
    
    # Build adjacency list for tasks: two tasks are adjacent if they share a processor
    adj = [[] for _ in range(M)]
    task_indices_primary = defaultdict(list)
    task_indices_secondary = defaultdict(list)
    
    for i in range(M):
        u, v = tasks[i]
        task_indices_primary[u].append(i)
        task_indices_secondary[v].append(i)
    
    # For each processor, connect all its tasks
    for u, task_list in task_indices_primary.items():
        for i in range(len(task_list)):
            for j in range(i+1, len(task_list)):
                adj[task_list[i]].append(task_list[j])
                adj[task_list[j]].append(task_list[i])
                
    for v, task_list in task_indices_secondary.items():
        for i in range(len(task_list)):
            for j in range(i+1, len(task_list)):
                adj[task_list[i]].append(task_list[j])
                adj[task_list[j]].append(task_list[i])
    
    # Graph coloring using BFS-based greedy approach
    color = [0] * M
    available = [True] * (K + 1)
    
    for i in range(M):
        if color[i] == 0:
            # Mark colors used by neighbors
            for neighbor in adj[i]:
                if color[neighbor] != 0:
                    available[color[neighbor]] = False
            
            # Find the first available color
            for c in range(1, K + 1):
                if available[c]:
                    color[i] = c
                    break
            
            # Reset available array
            for neighbor in adj[i]:
                if color[neighbor] != 0:
                    available[color[neighbor]] = True
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
```
