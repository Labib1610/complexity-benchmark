import sys
from collections import defaultdict

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        N1 = int(next(iterator))
        N2 = int(next(iterator))
        M = int(next(iterator))
    except StopIteration:
        return

    tasks = []
    # Track degrees for each processor
    deg_u = defaultdict(int)
    deg_v = defaultdict(int)
    
    for _ in range(M):
        u = int(next(iterator))
        v = int(next(iterator))
        tasks.append((u, v))
        deg_u[u] += 1
        deg_v[v] += 1
    
    # Determine K: maximum degree
    K = 0
    if deg_u:
        K = max(K, max(deg_u.values()))
    if deg_v:
        K = max(K, max(deg_v.values()))
    
    # Edge case: no tasks
    if M == 0:
        print("YES")
        print("")
        return

    # Build adjacency list for the bipartite graph
    # Nodes 1..N1 are primary, N1+1..N1+N2 are secondary
    adj = defaultdict(list)
    edge_info = [] # (u, v, original_index)
    
    for i, (u, v) in enumerate(tasks):
        adj[u].append((N1 + v, i))
        adj[N1 + v].append((u, i))
        edge_info.append((u, N1 + v, i))
    
    # Hierholzer's algorithm for Eulerian trail
    # We use a stack for DFS and a 'current edge' pointer for each node
    stack = []
    result_slots = [0] * M
    
    # Start at any node with non-zero degree
    start_node = tasks[0][0]
    stack.append(start_node)
    
    # To efficiently get the next edge, we use an index pointer for each node's adjacency list
    current_edge_idx = defaultdict(int)
    
    path_nodes = []
    
    while stack:
        u = stack[-1]
        idx = current_edge_idx[u]
        if idx < len(adj[u]):
            # There is an edge to traverse
            v, edge_orig_idx = adj[u][idx]
            current_edge_idx[u] = idx + 1
            stack.append(v)
        else:
            # No more edges from u, backtrack
            path_nodes.append(stack.pop())
            
    # The path_nodes contains nodes in reverse order of visitation.
    # We need to assign slots to edges.
    # An edge connects path_nodes[i] and path_nodes[i+1].
    # We assign slot (i % K) + 1.
    # We must identify which original task corresponds to this edge.
    
    # To map an unordered pair of nodes to the task index, we can use a dictionary.
    # Since the graph is simple (no parallel edges mentioned, and if there were, 
    # we'd need to handle them, but the problem implies a set of tasks).
    # Actually, the problem doesn't explicitly forbid multiple tasks between same pair.
    # But let's assume simple or handle via list of candidates.
    # Given constraints, a dict of sets or lists is fine.
    
    edge_map = defaultdict(list)
    for idx, (u, v) in enumerate(tasks):
        # Store canonical representation
        n1, n2 = (u, N1 + v) if u < N1 + v else (N1 + v, u)
        edge_map[(n1, n2)].append(idx)
        
    for i in range(len(path_nodes) - 1):
        u_node = path_nodes[i]
        v_node = path_nodes[i+1]
        
        n1, n2 = (u_node, v_node) if u_node < v_node else (v_node, u_node)
        
        # Retrieve and remove one task index for this edge
        # Using pop() from the list of candidates handles potential parallel edges correctly
        candidates = edge_map[(n1, n2)]
        task_idx = candidates.pop()
        
        slot = (i % K) + 1
        result_slots[task_idx] = slot
        
    print("YES")
    print(" ".join(map(str, result_slots)))

if __name__ == '__main__':
    solve()
