import sys

# Increase recursion depth just in case
sys.setrecursionlimit(500000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    Na = int(input_data[0])
    Nw = int(input_data[1])
    m = int(input_data[2])
    
    if m == 0:
        print("YES\n")
        return

    deg_u = [0] * (Na + 1)
    deg_v = [0] * (Nw + 1)
    
    edges = []
    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        deg_u[u] += 1
        deg_v[v] += 1
        edges.append((u, v))
        
    K = max(max(deg_u), max(deg_v))
    if K == 0:
        print("YES")
        print()
        return

    # to_u[u][c] = (v, edge_index)
    # to_v[v][c] = (u, edge_index)
    to_u = [{} for _ in range(Na + 1)]
    to_v = [{} for _ in range(Nw + 1)]
    ans = [0] * m

    for e_idx, (u, v) in enumerate(edges):
        # Find free color for u
        c1 = 1
        while c1 in to_u[u]:
            c1 += 1
            
        # Find free color for v
        c2 = 1
        while c2 in to_v[v]:
            c2 += 1
            
        if c1 == c2:
            to_u[u][c1] = (v, e_idx)
            to_v[v][c1] = (u, e_idx)
            ans[e_idx] = c1
        else:
            # Alternating path of c2, c1 starting at v
            curr_v = v
            curr_c = c2
            alt_c = c1
            path = []
            
            curr_node = v
            is_v_side = True # curr_node is on Omega (v) side
            
            while True:
                if is_v_side:
                    if curr_c in to_v[curr_node]:
                        nxt_u, nxt_e = to_v[curr_node][curr_c]
                        path.append((curr_node, nxt_u, nxt_e, is_v_side, curr_c))
                        curr_node = nxt_u
                        is_v_side = False
                        curr_c, alt_c = alt_c, curr_c
                    else:
                        break
                else:
                    if curr_c in to_u[curr_node]:
                        nxt_v, nxt_e = to_u[curr_node][curr_c]
                        path.append((nxt_v, curr_node, nxt_e, is_v_side, curr_c))
                        curr_node = nxt_v
                        is_v_side = True
                        curr_c, alt_c = alt_c, curr_c
                    else:
                        break
            
            # Flip colors along the path in reverse to avoid collision during update
            for node_v, node_u, edge_i, side, old_c in reversed(path):
                new_c = c1 if old_c == c2 else c2
                del to_v[node_v][old_c]
                del to_u[node_u][old_c]
                to_v[node_v][new_c] = (node_u, edge_i)
                to_u[node_u][new_c] = (node_v, edge_i)
                ans[edge_i] = new_c
                
            to_u[u][c1] = (v, e_idx)
            to_v[v][c1] = (u, e_idx)
            ans[e_idx] = c1

    sys.stdout.write("YES\n" + " ".join(map(str, ans)) + "\n")

if __name__ == '__main__':
    solve()
