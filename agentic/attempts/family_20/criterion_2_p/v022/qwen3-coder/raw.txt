import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N_alpha = int(data[idx])
    idx += 1
    N_omega = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    if m == 0:
        print("YES")
        return

    jobs = []
    alpha_graph = defaultdict(list)
    omega_graph = defaultdict(list)
    
    for i in range(m):
        u = int(data[idx]) - 1
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        jobs.append((u, v))
        alpha_graph[u].append(i)
        omega_graph[v].append(i)
    
    # Build conflict graph: edges between jobs that share at least one resource
    adj = [set() for _ in range(m)]
    for u in alpha_graph:
        nodes = alpha_graph[u]
        for i in range(len(nodes)):
            for j in range(i+1, len(nodes)):
                adj[nodes[i]].add(nodes[j])
                adj[nodes[j]].add(nodes[i])
                
    for v in omega_graph:
        nodes = omega_graph[v]
        for i in range(len(nodes)):
            for j in range(i+1, len(nodes)):
                adj[nodes[i]].add(nodes[j])
                adj[nodes[j]].add(nodes[i])
    
    # Greedy coloring with numbering from 1 to K
    color = [-1] * m
    for i in range(m):
        forbidden = set()
        for neighbor in adj[i]:
            if color[neighbor] != -1:
                forbidden.add(color[neighbor])
        c = 1
        while c in forbidden:
            c += 1
        color[i] = c

    K = max(color)
    print("YES")
    print(' '.join(str(c) for c in color))

if __name__ == "__main__":
    main()
