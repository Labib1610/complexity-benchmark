import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

nL, nR, m = data[0], data[1], data[2]
U = [0] * m
V = [0] * m
degL = [0] * nL
degR = [0] * nR

p = 3
for i in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2
    U[i] = u
    V[i] = v
    degL[u] += 1
    degR[v] += 1

if m == 0:
    sys.stdout.write("YES\n\n")
    sys.exit()

Delta = max(max(degL), max(degR))
answer = [0] * m

pair_u = [-1] * nL
pair_v = [-1] * nR
pair_edge = [-1] * nL
dist = [0] * nL
ptr = [0] * nL

mark = [0] * m
seen = [0] * m
mark_stamp = 0
seen_stamp = 0


def maximum_matching(edges):
    adj = {}

    for eid in edges:
        u = U[eid]
        v = V[eid]
        lst = adj.get(u)
        if lst is None:
            adj[u] = [eid]
        else:
            lst.append(eid)
        pair_v[v] = -1

    active_u = list(adj.keys())
    for u in active_u:
        pair_u[u] = -1
        pair_edge[u] = -1

    for u in active_u:
        for eid in adj[u]:
            v = V[eid]
            if pair_v[v] == -1:
                pair_u[u] = v
                pair_v[v] = u
                pair_edge[u] = eid
                break

    q = deque()

    def bfs():
        q.clear()
        found = False
        for u in active_u:
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = -1

        while q:
            u = q.popleft()
            nd = dist[u] + 1
            for eid in adj[u]:
                pu = pair_v[V[eid]]
                if pu == -1:
                    found = True
                elif dist[pu] == -1:
                    dist[pu] = nd
                    q.append(pu)
        return found

    def dfs(u):
        lst = adj[u]
        i = ptr[u]
        while i < len(lst):
            eid = lst[i]
            ptr[u] = i + 1
            v = V[eid]
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                pair_edge[u] = eid
                return True
            i = ptr[u]
        dist[u] = -1
        return False

    while bfs():
        for u in active_u:
            ptr[u] = 0
        for u in active_u:
            if pair_u[u] == -1:
                dfs(u)

    result = []
    for u in active_u:
        if pair_u[u] != -1:
            result.append(pair_edge[u])
    return result


def balanced_split(edges):
    global seen_stamp
    seen_stamp += 1
    stamp = seen_stamp

    adj = {}
    offset = nL

    for eid in edges:
        a = U[eid]
        b = offset + V[eid]

        la = adj.get(a)
        if la is None:
            adj[a] = [eid]
        else:
            la.append(eid)

        lb = adj.get(b)
        if lb is None:
            adj[b] = [eid]
        else:
            lb.append(eid)

    odd_vertices = [x for x, lst in adj.items() if len(lst) & 1]
    virtual = nL + nR
    virtual_used = [False] * len(odd_vertices)

    if odd_vertices:
        adj[virtual] = []
        for i, x in enumerate(odd_vertices):
            token = -i - 1
            adj[x].append(token)
            adj[virtual].append(token)

    left_part = []
    right_part = []

    def token_used(token):
        if token >= 0:
            return seen[token] == stamp
        return virtual_used[-token - 1]

    def traverse(start):
        stack_v = [start]
        stack_e = [-1]
        circuit = []

        while stack_v:
            x = stack_v[-1]
            lst = adj[x]

            while lst and token_used(lst[-1]):
                lst.pop()

            if not lst:
                stack_v.pop()
                token = stack_e.pop()
                if token != -1:
                    circuit.append(token)
                continue

            token = lst.pop()
            if token >= 0:
                seen[token] = stamp
                if x < nL:
                    y = offset + V[token]
                else:
                    y = U[token]
            else:
                idx = -token - 1
                virtual_used[idx] = True
                y = odd_vertices[idx] if x == virtual else virtual

            stack_v.append(y)
            stack_e.append(token)

        for i, token in enumerate(circuit):
            if token >= 0:
                if i & 1:
                    right_part.append(token)
                else:
                    left_part.append(token)

    if odd_vertices:
        traverse(virtual)

    for x in list(adj.keys()):
        lst = adj[x]
        while lst and token_used(lst[-1]):
            lst.pop()
        if lst:
            traverse(x)

    return left_part, right_part


def color_edges(edges, degree_bound, first_color):
    global mark_stamp

    if not edges:
        return

    if degree_bound == 1:
        for eid in edges:
            answer[eid] = first_color
        return

    if degree_bound & 1:
        matching = maximum_matching(edges)
        matching_color = first_color + degree_bound - 1

        mark_stamp += 1
        stamp = mark_stamp
        for eid in matching:
            mark[eid] = stamp
            answer[eid] = matching_color

        remaining = [eid for eid in edges if mark[eid] != stamp]
        color_edges(remaining, degree_bound - 1, first_color)
    else:
        half = degree_bound // 2
        a, b = balanced_split(edges)
        color_edges(a, half, first_color)
        color_edges(b, half, first_color + half)


color_edges(list(range(m)), Delta, 1)

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answer)))
sys.stdout.write("\n")
