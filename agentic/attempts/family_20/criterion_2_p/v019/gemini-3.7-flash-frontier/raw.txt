An optimal edge coloring of a bipartite graph with maximum degree $D$ can be found using Euler tour splitting. 

When $D$ is even, we can partition the edge set into two subgraphs, each having maximum degree at most $D/2$, by pairing up odd-degree vertices, finding Euler tours in each connected component, and splitting edges based on the direction they are traversed. 

When $D$ is odd, we can reduce $D$ by finding a matching that covers all vertices of maximum degree $D$, assigning that matching a distinct color, and then solving for the remaining graph of maximum degree $D - 1$. Alternatively, by standard regular graph reduction (or standard Hopcroft-Karp maximum bipartite matching on the subgraph of max-degree vertices), or simply reducing odd $D$ by matching.

Here is the complete, efficient Python 3 implementation.

```python
import sys

# Increase recursion depth for deep recursive calls
sys.setrecursionlimit(2000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    it = iter(input_data)
    N_E = int(next(it))
    N_C = int(next(it))
    M = int(next(it))

    if M == 0:
        print("YES")
        print()
        return

    deg_E = [0] * (N_E + 1)
    deg_C = [0] * (N_C + 1)
    edges = []

    for i in range(M):
        u = int(next(it))
        v = int(next(it))
        deg_E[u] += 1
        deg_C[v] += 1
        edges.append((u, v, i))

    D = max(max(deg_E), max(deg_C))
    if D == 0:
        print("YES")
        print()
        return

    color_assignment = [0] * M

    def find_max_deg_matching(edge_list, target_deg):
        # We need a matching covering all vertices with degree == target_deg.
        # We can find a maximum matching in the bipartite graph induced by edge_list.
        adj = {}
        for u, v, idx in edge_list:
            if u not in adj:
                adj[u] = []
            adj[u].append((v, idx))

        pair_u = {}
        pair_v = {}
        dist = {}

        def bfs():
            queue = []
            for u in adj:
                if u not in pair_u:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = float('inf')
            dist[None] = float('inf')
            
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                if dist[u] < dist[None]:
                    for v, _ in adj.get(u, []):
                        next_u = pair_v.get(v, None)
                        if dist.get(next_u, float('inf')) == float('inf'):
                            dist[next_u] = dist[u] + 1
                            queue.append(next_u)
            return dist[None] != float('inf')

        def dfs(u):
            if u is not None:
                for v, idx in adj.get(u, []):
                    next_u = pair_v.get(v, None)
                    if dist.get(next_u, float('inf')) == dist[u] + 1:
                        if dfs(next_u):
                            pair_v[v] = u
                            pair_u[u] = (v, idx)
                            return True
                dist[u] = float('inf')
                return False
            return True

        while bfs():
            for u in list(adj.keys()):
                if u not in pair_u:
                    dfs(u)

        matching_indices = set()
        for u, (v, idx) in pair_u.items():
            matching_indices.add(idx)

        matching_edges = []
        remaining_edges = []
        for e in edge_list:
            if e[2] in matching_indices:
                matching_edges.append(e)
            else:
                remaining_edges.append(e)

        return matching_edges, remaining_edges

    def euler_split(edge_list):
        # Nodes: 1..N_E in Left, 1..N_C in Right
        # Right nodes are offset by N_E
        # Dummy node for odd degree vertices: 0
        adj = {}
        deg = {}
        for u, v, idx in edge_list:
            vr = v + N_E
            adj.setdefault(u, []).append((vr, idx, 0))
            adj.setdefault(vr, []).append((u, idx, 1))
            deg[u] = deg.get(u, 0) + 1
            deg[vr] = deg.get(vr, 0) + 1

        odd_nodes = [node for node, d in deg.items() if d % 2 != 0]
        dummy_idx = -1
        for i in range(0, len(odd_nodes), 2):
            u = odd_nodes[i]
            v = odd_nodes[i+1]
            adj.setdefault(u, []).append((v, dummy_idx, 0))
            adj.setdefault(v, []).append((u, dummy_idx, 1))
            dummy_idx -= 1

        used_edge = set()
        pos = {node: 0 for node in adj}
        e1 = []
        e2 = []

        for start_node in list(adj.keys()):
            stack = [start_node]
            while stack:
                u = stack[-1]
                while pos[u] < len(adj[u]) and adj[u][pos[u]][1] in used_edge:
                    pos[u] += 1
                if pos[u] < len(adj[u]):
                    v, idx, side = adj[u][pos[u]]
                    pos[u] += 1
                    used_edge.add(idx)
                    stack.append(v)
                    if idx >= 0:
                        # u is the source of the directed traversal u -> v
                        if u <= N_E:
                            e1.append((u, v - N_E, idx))
                        else:
                            e2.append((v, u - N_E, idx))
                else:
                    stack.pop()

        return e1, e2

    def color_graph(edge_list, d, color_offset):
        if not edge_list or d == 0:
            return
        if d == 1:
            for _, _, idx in edge_list:
                color_assignment[idx] = color_offset + 1
            return

        if d % 2 == 1:
            matching, remaining = find_max_deg_matching(edge_list, d)
            for _, _, idx in matching:
                color_assignment[idx] = color_offset + d
            color_graph(remaining, d - 1, color_offset)
        else:
            e1, e2 = euler_split(edge_list)
            half = d // 2
            color_graph(e1, half, color_offset)
            color_graph(e2, half, color_offset + half)

    color_graph(edges, D, 0)

    print("YES")
    print(*(color_assignment[i] for i in range(M)))

if __name__ == '__main__':
    solve()
```
