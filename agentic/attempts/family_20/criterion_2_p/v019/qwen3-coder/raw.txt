```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N_E = int(data[idx])
    idx += 1
    N_C = int(data[idx])
    idx += 1
    M = int(data[idx])
    idx += 1
    
    tests = []
    emitter_degree = [0] * (N_E + 1)
    chamber_degree = [0] * (N_C + 1)
    
    for i in range(M):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        tests.append((u, v))
        emitter_degree[u] += 1
        chamber_degree[v] += 1
    
    D = max(max(emitter_degree), max(chamber_degree))
    
    # Build a graph for edge coloring
    # Each test is an edge between emitter and chamber
    # We need to color the edges such that no two edges sharing a vertex have the same color
    # This is edge coloring, and we know it can be done with D colors (since max degree is D)
    
    # For each vertex (emitter or chamber), maintain a list of adjacent edges and their assigned colors
    adj = defaultdict(list)  # adj[vertex] = [(neighbor, edge_id)]
    for i in range(M):
        u, v = tests[i]
        # Shift chamber indices to avoid conflict with emitter indices
        v_shifted = v + N_E
        adj[u].append((v_shifted, i))
        adj[v_shifted].append((u, i))
    
    # Color each edge
    result = [0] * M
    used_colors = defaultdict(set)  # used_colors[vertex] = set of colors used by its edges
    
    for i in range(M):
        u, v = tests[i]
        v_shifted = v + N_E
        # Find a color not used by u or v_shifted
        color = 1
        while color in used_colors[u] or color in used_colors[v_shifted]:
            color += 1
        result[i] = color
        used_colors[u].add(color)
        used_colors[v_shifted].add(color)
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
