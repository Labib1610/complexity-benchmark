Looking at this problem, we need to edge-color a bipartite graph where the number of colors equals the maximum degree (peak load). This is König's theorem - a bipartite graph with maximum degree Δ can be edge-colored with exactly Δ colors.

**Algorithm:**
1. Compute the maximum degree Δ (peak load)
2. For each edge, find a color c such that neither endpoint currently has an edge colored c
3. If both endpoints are free for color c, assign it
4. If only one endpoint is free, we need to do a color swap along an alternating path (Kempe chain)

For the Kempe chain: if lock u has color c but key v is free for c, and key v has color c' but lock u is free for c', we swap colors along the alternating path starting from the edge colored c' at key v, following the path of edges alternating between colors c' and c. Since the graph is bipartite, this path cannot form an odd cycle, so it terminates properly.

```python
import sys
from sys import setrecursionlimit

def solve():
    input_data = sys.stdin.buffer.read().split()
    idx = 0
    nL = int(input_data[idx]); idx += 1
    nR = int(input_data[idx]); idx += 1
    m = int(input_data[idx]); idx += 1
    
    edges = []
    for i in range(m):
        u = int(input_data[idx]) - 1; idx += 1
        v = int(input_data[idx]) - 1; idx += 1
        edges.append((u, v))
    
    degL = [0] * nL
    degR = [0] * nR
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    
    peak = 0
    for d in degL:
        if d > peak: peak = d
    for d in degR:
        if d > peak: peak = d
    
    if m == 0:
        sys.stdout.write("YES\n\n")
        return
    
    # colorL[u][c] = edge index of edge at lock u with color c
    colorL = [dict() for _ in range(nL)]
    colorR = [dict() for _ in range(nR)]
    ans = [0] * m
    
    for i, (u, v) in enumerate(edges):
        # Find a color free at u
        cu = None
        for c in range(1, peak + 1):
            if c not in colorL[u]:
                cu = c
                break
        
        if cu not in colorR[v]:
            ans[i] = cu
            colorL[u][cu] = i
            colorR[v][cu] = i
        else:
            # cu is used at v by some edge; find color free at v
            cv = None
            for c in range(1, peak + 1):
                if c not in colorR[v]:
                    cv = c
                    break
            
            # Kempe chain: swap colors cu and cv along alternating path
            # Start from edge at v with color cv, follow alternating cv/cu path
            stack = []
            cur_v = v
            cur_c = cv
            while cur_c in colorR[cur_v]:
                e_idx = colorR[cur_v][cur_c]
                stack.append(e_idx)
                uu, vv = edges[e_idx]
                cur_v = uu  # move to lock side
                # swap roles
                cur_c = cu if cur_c == cv else cv
                # now look at lock uu for color cur_c
                if cur_c in colorL[uu]:
                    e_idx2 = colorL[uu][cur_c]
                    stack.append(e_idx2)
                    uu2, vv2 = edges[e_idx2]
                    cur_v = vv2
                    cur_c = cu if cur_c == cv else cv
                else:
                    break
            
            # Swap colors along the chain
            for e_idx in stack:
                uu, vv = edges[e_idx]
                old_c = ans[e_idx]
                new_c = cv if old_c == cu else cu
                del colorL[uu][old_c]
                del colorR[vv][old_c]
                ans[e_idx] = new_c
                colorL[uu][new_c] = e_idx
                colorR[vv][new_c] = e_idx
            
            # Now assign color cu to edge i
            ans[i] = cu
            colorL[u][cu] = i
            colorR[v][cu] = i
    
    out = ["YES"]
    out.append(" ".join(map(str, ans)))
    sys.stdout.write("\n".join(out) + "\n")

solve()
```
