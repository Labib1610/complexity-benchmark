import sys

import heapq

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    nL = int(data[0])
    nR = int(data[1])
    m = int(data[2])
    
    if m == 0:
        print("YES")
        return
    
    tests = []
    lock_degree = [0] * (nL + 1)
    key_degree = [0] * (nR + 1)
    
    idx = 3
    for i in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        tests.append((u, v))
        lock_degree[u] += 1
        key_degree[v] += 1
    
    peak_load = max(max(lock_degree), max(key_degree))
    
    # For each lock, maintain a min-heap of available time slots
    lock_slots = [list(range(1, peak_load + 1)) for _ in range(nL + 1)]
    key_slots = [list(range(1, peak_load + 1)) for _ in range(nR + 1)]
    
    # Convert to heaps
    for i in range(1, nL + 1):
        heapq.heapify(lock_slots[i])
    for i in range(1, nR + 1):
        heapq.heapify(key_slots[i])
        
    result = []
    
    for u, v in tests:
        # Find a common available slot
        # We'll pick the smallest available slot that's free for both
        # To do this efficiently, we can peek at the tops and adjust if needed
        
        # But a simpler approach: since we know a solution exists,
        # we can just assign greedily by checking availability
        
        # Let's use a different strategy:
        # Keep track of which slots are used by each lock and key
        # Actually, let's think of it as a graph coloring problem where
        # each test is an edge between a lock and a key, and we color edges
        # such that no two edges sharing a vertex have the same color
        
        # A known algorithm for edge coloring of bipartite graphs:
        # Since the graph is bipartite (locks on one side, keys on other),
        # and the maximum degree is peak_load, we can color it with peak_load colors
        
        # We'll simulate a greedy edge coloring:
        # For each test (edge), assign it the smallest color not used by 
        # edges incident to its endpoints
        
        # But to make it efficient, let's precompute for each vertex the set of colors used
        pass
    
    # Let's reapproach with a cleaner method:
    # Build adjacency for locks and keys to track used slots
    
    lock_used = [set() for _ in range(nL + 1)]
    key_used = [set() for _ in range(nR + 1)]
    
    result = []
    
    for u, v in tests:
        # Find the smallest slot not used by either u or v
        slot = 1
        while slot in lock_used[u] or slot in key_used[v]:
            slot += 1
        result.append(slot)
        lock_used[u].add(slot)
        key_used[v].add(slot)
    
    print("YES")
    print(' '.join(map(str, result)))
