import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    nA, nB, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency lists for ports
    adjA = defaultdict(list)  # adjA[u] = list of (v, idx)
    adjB = defaultdict(list)  # adjB[v] = list of (u, idx)
    
    edges = []
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adjA[u].append((v, i))
        adjB[v].append((u, i))
        edges.append((u, v))
    
    # Find maximum degree
    max_degree = 0
    for u in range(1, nA + 1):
        max_degree = max(max_degree, len(adjA[u]))
    for v in range(1, nB + 1):
        max_degree = max(max_degree, len(adjB[v]))
    
    # Color the edges: each color represents a service window
    # We use a greedy algorithm with priority queues to minimize the number of colors used at each vertex
    
    result = [0] * m  # result[i] = color (1-based) of edge i
    
    # For each vertex, maintain a min-heap of available colors
    # and a set of used colors
    used_colors_A = defaultdict(set)
    used_colors_B = defaultdict(set)
    
    # We process edges in a specific order to ensure we can always color them
    # We'll use a greedy approach: for each edge, assign the smallest available color
    # that is not used by either endpoint
    
    # To make it efficient, we can process edges in any order, but for each edge,
    # we find the minimum color not used by u or v
    
    # A more efficient way: for each vertex, keep track of the next available color
    next_color_A = defaultdict(lambda: 1)
    next_color_B = defaultdict(lambda: 1)
    
    for i in range(m):
        u, v = edges[i]
        
        # Find the first color that is not used by u and not used by v
        cu = used_colors_A[u]
        cv = used_colors_B[v]
        
        c = 1
        while c in cu or c in cv:
            c += 1
            
        result[i] = c
        used_colors_A[u].add(c)
        used_colors_B[v].add(c)
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
