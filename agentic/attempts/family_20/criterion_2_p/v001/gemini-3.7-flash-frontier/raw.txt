import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    nA = int(input_data[0])
    nB = int(input_data[1])
    m = int(input_data[2])

    if m == 0:
        print("YES\n")
        return

    V = nA + nB
    adj = [{} for _ in range(V + 1)]
    first_free = [0] * (V + 1)
    ans = [0] * m

    idx = 3
    for e in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1]) + nA
        idx += 2

        # Find free color for u
        c1 = first_free[u]
        adj_u = adj[u]
        while c1 in adj_u:
            c1 += 1
        first_free[u] = c1

        # Find free color for v
        c2 = first_free[v]
        adj_v = adj[v]
        while c2 in adj_v:
            c2 += 1
        first_free[v] = c2

        if c1 == c2:
            adj_u[c1] = (v, e)
            adj_v[c1] = (u, e)
            ans[e] = c1
        else:
            # Trace alternating path starting at u with color c2
            curr = u
            c_curr = c2
            c_next = c1
            path = []

            while c_curr in adj[curr]:
                nxt, edge_idx = adj[curr][c_curr]
                path.append((curr, nxt, edge_idx, c_curr, c_next))
                curr = nxt
                c_curr, c_next = c_next, c_curr

            # Delete old colors and update first_free lower bound
            for c_u, c_v, edge_idx, c_old, c_new in path:
                del adj[c_u][c_old]
                del adj[c_v][c_old]
                if c_old < first_free[c_u]:
                    first_free[c_u] = c_old
                if c_old < first_free[c_v]:
                    first_free[c_v] = c_old

            # Insert new colors
            for c_u, c_v, edge_idx, c_old, c_new in path:
                adj[c_u][c_new] = (c_v, edge_idx)
                adj[c_v][c_new] = (c_u, edge_idx)
                ans[edge_idx] = c_new

            # Add the new edge (u, v) with color c2
            adj[u][c2] = (v, e)
            adj[v][c2] = (u, e)
            ans[e] = c2

    sys.stdout.write("YES\n" + " ".join(str(c + 1) for c in ans) + "\n")


if __name__ == "__main__":
    solve()
