import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    import sys
    input = sys.stdin.readline
    nL, nR, m = map(int, input().split())
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    degL = [0] * (nL + 1)
    degR = [0] * (nR + 1)
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = max(max(degL), max(degR), 1)

    # adjacency lists with edge indices
    adjL = [[] for _ in range(nL + 1)]
    adjR = [[] for _ in range(nR + 1)]
    for idx, (u, v) in enumerate(edges):
        adjL[u].append(idx)
        adjR[v].append(idx)

    # color of each edge, 0 = uncolored
    col = [0] * m

    # for each vertex, set of colors already used
    usedL = [set() for _ in range(nL + 1)]
    usedR = [set() for _ in range(nR + 1)]

    # helper to find a free color for a vertex
    def free_color(used_set):
        # colors are 1..Delta, we keep a pointer per vertex
        # but here we just scan; total scans are bounded by O(m)
        c = 1
        while c in used_set:
            c += 1
        return c

    # simple greedy with occasional recoloring via alternating path
    for idx, (u, v) in enumerate(edges):
        # try to assign a color not used at either endpoint
        cu = usedL[u]
        cv = usedR[v]
        c = 1
        while c in cu or c in cv:
            c += 1
        if c <= Delta:
            col[idx] = c
            cu.add(c)
            cv.add(c)
            continue

        # need recoloring: find alternating path
        # we will use BFS on the "conflict graph" of colors
        from collections import deque
        # start from u with missing color a (any not in cu)
        a = free_color(cu)
        # BFS state: (vertex, side) side 0 = left lock, 1 = right key, and current color
        q = deque()
        prev = dict()  # (vertex, side) -> (prev_vertex, prev_side, edge_idx, old_color)
        q.append((u, 0, a))
        visited = set()
        visited.add((u, 0, a))
        found = None
        target = None
        while q and found is None:
            vert, side, curc = q.popleft()
            if side == 0:  # left lock
                # look at edge colored curc incident to this lock, if any
                for eidx in adjL[vert]:
                    if col[eidx] == curc:
                        # go to the right key of this edge
                        _, v2 = edges[eidx]
                        state = (v2, 1, curc)
                        if state not in visited:
                            visited.add(state)
                            prev[state] = (vert, side, eidx, curc)
                            q.append(state)
            else:  # right key
                # look for missing color at this key
                if curc not in usedR[vert]:
                    found = (vert, side, curc)
                    break
                # else follow edge with curc
                for eidx in adjR[vert]:
                    if col[eidx] == curc:
                        u2, _ = edges[eidx]
                        state = (u2, 0, curc)
                        if state not in visited:
                            visited.add(state)
                            prev[state] = (vert, side, eidx, curc)
                            q.append(state)
        # reconstruct path and flip colors
        if found is None:
            # should not happen in bipartite graphs
            print("NO")
            return
        # flip along path
        cur = found
        while cur in prev:
            pv, pside, eidx, oldc = prev[cur]
            # recolor edge eidx to the other color
            newc = a if oldc != a else cur[2]
            col[eidx] = newc
            # update used sets
            if pside == 0:
                usedL[pv].discard(oldc)
                usedL[pv].add(newc)
                usedR[edges[eidx][1]].discard(oldc)
                usedR[edges[eidx][1]].add(newc)
            else:
                usedR[pv].discard(oldc)
                usedR[pv].add(newc)
                usedL[edges[eidx][0]].discard(oldc)
                usedL[edges[eidx][0]].add(newc)
            cur = (pv, pside, oldc)
        # finally assign the free color a to original edge
        col[idx] = a
        usedL[u].add(a)
        usedR[v].add(a)

    print("YES")
    print(' '.join(map(str, col)))

if __name__ == "__main__":
    solve()
