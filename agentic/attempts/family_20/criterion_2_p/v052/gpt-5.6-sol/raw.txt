import sys
from array import array
from collections import deque

sys.setrecursionlimit(1_000_000)

data = sys.stdin.buffer.read()
p = 0
ln = len(data)

def next_int():
    global p
    while p < ln and data[p] <= 32:
        p += 1
    x = 0
    while p < ln and data[p] > 32:
        x = x * 10 + data[p] - 48
        p += 1
    return x

nL = next_int()
nR = next_int()
m = next_int()

U = array('i', [0]) * m
V = array('i', [0]) * m
degL0 = array('i', [0]) * nL
degR0 = array('i', [0]) * nR

for i in range(m):
    u = next_int() - 1
    v = next_int() - 1
    U[i] = u
    V[i] = v
    degL0[u] += 1
    degR0[v] += 1

del data

Delta = 0
if nL:
    Delta = max(degL0)
if nR:
    Delta = max(Delta, max(degR0))

answer = array('i', [0]) * m
next_color = 1

degL = array('i', [0]) * nL
degR = array('i', [0]) * nR
posL = array('i', [-1]) * nL
posR = array('i', [-1]) * nR


def covering_matching(edges, bound):
    touchedL = []
    touchedR = []

    for eid in edges:
        u = U[eid]
        v = V[eid]
        if degL[u] == 0:
            touchedL.append(u)
        if degR[v] == 0:
            touchedR.append(v)
        degL[u] += 1
        degR[v] += 1

    reqL = 0
    reqR = 0
    for u in touchedL:
        if degL[u] == bound:
            reqL += 1
    for v in touchedR:
        if degR[v] == bound:
            reqR += 1

    if reqL == 0 and reqR == 0:
        for u in touchedL:
            degL[u] = 0
        for v in touchedR:
            degR[v] = 0
        return None

    a = len(touchedL)
    b = len(touchedR)
    for i, u in enumerate(touchedL):
        posL[u] = i
    for i, v in enumerate(touchedR):
        posR[v] = i

    S = a + b
    T = S + 1
    SS = S + 2
    TT = S + 3
    N = S + 4

    head = array('i', [-1]) * N
    to = array('i')
    nxt = array('i')
    cap = array('i')

    def add_edge(x, y, c):
        idx = len(to)
        to.append(y)
        cap.append(c)
        nxt.append(head[x])
        head[x] = idx

        to.append(x)
        cap.append(0)
        nxt.append(head[y])
        head[y] = idx + 1
        return idx

    for i, u in enumerate(touchedL):
        if degL[u] == bound:
            add_edge(SS, i, 1)
        else:
            add_edge(S, i, 1)

    for j, v in enumerate(touchedR):
        node = a + j
        if degR[v] == bound:
            add_edge(node, TT, 1)
        else:
            add_edge(node, T, 1)

    edge_arc = array('i')
    for eid in edges:
        edge_arc.append(add_edge(posL[U[eid]], a + posR[V[eid]], 1))

    add_edge(T, S, reqL + reqR + 1)
    if reqR:
        add_edge(SS, T, reqR)
    if reqL:
        add_edge(S, TT, reqL)

    level = array('i', [-1]) * N
    it = array('i', [-1]) * N

    def bfs():
        for i in range(N):
            level[i] = -1
        level[SS] = 0
        q = deque([SS])
        while q:
            x = q.popleft()
            e = head[x]
            nl = level[x] + 1
            while e != -1:
                y = to[e]
                if cap[e] and level[y] < 0:
                    level[y] = nl
                    if y == TT:
                        return True
                    q.append(y)
                e = nxt[e]
        return level[TT] >= 0

    def dfs(x, pushed):
        if x == TT:
            return pushed
        e = it[x]
        while e != -1:
            y = to[e]
            if cap[e] and level[y] == level[x] + 1:
                got = dfs(y, pushed if pushed < cap[e] else cap[e])
                if got:
                    cap[e] -= got
                    cap[e ^ 1] += got
                    it[x] = e
                    return got
            e = nxt[e]
            it[x] = e
        return 0

    need = reqL + reqR
    flow = 0
    while flow < need and bfs():
        for i in range(N):
            it[i] = head[i]
        while flow < need:
            got = dfs(SS, need - flow)
            if not got:
                break
            flow += got

    matching = []
    remaining = []
    for i, eid in enumerate(edges):
        if cap[edge_arc[i]] == 0:
            matching.append(eid)
        else:
            remaining.append(eid)

    for u in touchedL:
        degL[u] = 0
        posL[u] = -1
    for v in touchedR:
        degR[v] = 0
        posR[v] = -1

    return matching, remaining


def split_even(edges):
    touchedL = []
    touchedR = []

    for eid in edges:
        u = U[eid]
        v = V[eid]
        if degL[u] == 0:
            touchedL.append(u)
        if degR[v] == 0:
            touchedR.append(v)
        degL[u] += 1
        degR[v] += 1

    oddL = [u for u in touchedL if degL[u] & 1]
    oddR = [v for v in touchedR if degR[v] & 1]

    total_vertices = nL + nR + 2
    dummyL = nL + nR
    dummyR = dummyL + 1

    head = array('i', [-1]) * total_vertices
    eu = array('i')
    ev = array('i')
    original = array('i')
    nxt = array('i')
    arc_eid = array('i')

    def add_aug(x, y, orig):
        eid = len(eu)
        eu.append(x)
        ev.append(y)
        original.append(orig)

        arc_eid.append(eid)
        nxt.append(head[x])
        head[x] = len(arc_eid) - 1

        arc_eid.append(eid)
        nxt.append(head[y])
        head[y] = len(arc_eid) - 1

    active_vertices = []
    for eid in edges:
        x = U[eid]
        y = nL + V[eid]
        add_aug(x, y, eid)

    active_vertices.extend(touchedL)
    active_vertices.extend(nL + v for v in touchedR)

    for u in oddL:
        add_aug(u, dummyR, -1)
    for v in oddR:
        add_aug(dummyL, nL + v, -1)

    if len(oddL) & 1:
        add_aug(dummyL, dummyR, -1)

    if oddL:
        active_vertices.append(dummyR)
    if oddR:
        active_vertices.append(dummyL)

    used = bytearray(len(eu))
    part0 = []
    part1 = []

    for start in active_vertices:
        e0 = head[start]
        while e0 != -1 and used[arc_eid[e0]]:
            e0 = nxt[e0]
        head[start] = e0
        if e0 == -1:
            continue

        stack_v = [start]
        stack_e = []
        tour = []

        while stack_v:
            x = stack_v[-1]
            aidx = head[x]
            while aidx != -1 and used[arc_eid[aidx]]:
                aidx = nxt[aidx]
            head[x] = aidx

            if aidx == -1:
                stack_v.pop()
                if stack_e:
                    tour.append(stack_e.pop())
            else:
                ae = arc_eid[aidx]
                head[x] = nxt[aidx]
                if used[ae]:
                    continue
                used[ae] = 1
                y = ev[ae] if eu[ae] == x else eu[ae]
                stack_v.append(y)
                stack_e.append(ae)

        parity = 0
        for ae in tour:
            orig = original[ae]
            if orig >= 0:
                if parity == 0:
                    part0.append(orig)
                else:
                    part1.append(orig)
            parity ^= 1

    for u in touchedL:
        degL[u] = 0
    for v in touchedR:
        degR[v] = 0

    return part0, part1


def color_graph(edges, bound):
    global next_color

    if not edges:
        return

    if bound <= 1:
        c = next_color
        next_color += 1
        for eid in edges:
            answer[eid] = c
        return

    if bound & 1:
        result = covering_matching(edges, bound)
        if result is None:
            color_graph(edges, bound - 1)
            return

        matching, remaining = result
        if matching:
            c = next_color
            next_color += 1
            for eid in matching:
                answer[eid] = c
        color_graph(remaining, bound - 1)
    else:
        left, right = split_even(edges)
        half = bound // 2
        color_graph(left, half)
        color_graph(right, half)


if m:
    color_graph(list(range(m)), Delta)

out = ["YES\n"]
if m:
    out.append(" ".join(map(str, answer)))
    out.append("\n")
else:
    out.append("\n")

sys.stdout.write("".join(out))
