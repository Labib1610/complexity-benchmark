import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n_L = int(input_data[0])
    n_R = int(input_data[1])
    m = int(input_data[2])
    
    if m == 0:
        print("YES")
        print()
        return

    deg_L = [0] * (n_L + 1)
    deg_R = [0] * (n_R + 1)
    
    edges_input = []
    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        deg_L[u] += 1
        deg_R[v] += 1
        edges_input.append((u, v, i))
    
    Delta = max(max(deg_L), max(deg_R))
    
    if Delta == 0:
        print("YES")
        print()
        return

    # Pack vertices into super-vertices of degree <= Delta
    super_L = []
    cur_super = []
    cur_deg = 0
    super_id_L = [0] * (n_L + 1)
    for u in range(1, n_L + 1):
        d = deg_L[u]
        if d == 0:
            continue
        if cur_deg + d > Delta:
            super_L.append(cur_super)
            cur_super = [u]
            cur_deg = d
        else:
            cur_super.append(u)
            cur_deg += d
    if cur_super:
        super_L.append(cur_super)
        
    for s_id, s_nodes in enumerate(super_L):
        for u in s_nodes:
            super_id_L[u] = s_id

    super_R = []
    cur_super = []
    cur_deg = 0
    super_id_R = [0] * (n_R + 1)
    for v in range(1, n_R + 1):
        d = deg_R[v]
        if d == 0:
            continue
        if cur_deg + d > Delta:
            super_R.append(cur_super)
            cur_super = [v]
            cur_deg = d
        else:
            cur_super.append(v)
            cur_deg += d
    if cur_super:
        super_R.append(cur_super)
        
    for s_id, s_nodes in enumerate(super_R):
        for v in s_nodes:
            super_id_R[v] = s_id

    N_prime = max(len(super_L), len(super_R), 1)

    D_L = [0] * N_prime
    D_R = [0] * N_prime
    
    graph_edges = []
    for u, v, orig_id in edges_input:
        su = super_id_L[u]
        sv = super_id_R[v]
        D_L[su] += 1
        D_R[sv] += 1
        graph_edges.append((su, sv, orig_id))

    # Add dummy edges to make each super-vertex have degree Delta
    i_ptr, j_ptr = 0, 0
    while i_ptr < N_prime and j_ptr < N_prime:
        while i_ptr < N_prime and D_L[i_ptr] == Delta:
            i_ptr += 1
        while j_ptr < N_prime and D_R[j_ptr] == Delta:
            j_ptr += 1
        if i_ptr == N_prime or j_ptr == N_prime:
            break
        add = min(Delta - D_L[i_ptr], Delta - D_R[j_ptr])
        for _ in range(add):
            graph_edges.append((i_ptr, j_ptr, -1))
        D_L[i_ptr] += add
        D_R[j_ptr] += add

    ans = [0] * m

    def split_euler(edges):
        V_total = 2 * N_prime
        adj = [[] for _ in range(V_total)]
        for i, (u, v, orig_id) in enumerate(edges):
            adj[u].append((N_prime + v, i))
            adj[N_prime + v].append((u, i))

        num_edges = len(edges)
        used = [False] * num_edges
        ptr = [0] * V_total

        edges0 = []
        edges1 = []

        for start_node in range(V_total):
            if ptr[start_node] == len(adj[start_node]):
                continue
            stack = [(start_node, -1)]
            circuit_edges = []
            while stack:
                u, _ = stack[-1]
                while ptr[u] < len(adj[u]) and used[adj[u][ptr[u]][1]]:
                    ptr[u] += 1
                if ptr[u] < len(adj[u]):
                    nxt, e_idx = adj[u][ptr[u]]
                    ptr[u] += 1
                    used[e_idx] = True
                    stack.append((nxt, e_idx))
                else:
                    _, e_idx = stack.pop()
                    if e_idx != -1:
                        circuit_edges.append(e_idx)

            for k, e_idx in enumerate(circuit_edges):
                if k % 2 == 0:
                    edges0.append(edges[e_idx])
                else:
                    edges1.append(edges[e_idx])

        return edges0, edges1

    def find_matching(edges):
        adj = [[] for _ in range(N_prime)]
        for i, (u, v, orig_id) in enumerate(edges):
            adj[u].append((v, i))

        match_R = [-1] * N_prime
        match_edge = [-1] * N_prime
        used_R = [False] * N_prime
        unmatched_L = []

        for u in range(N_prime):
            matched = False
            for v, e_idx in adj[u]:
                if not used_R[v]:
                    used_R[v] = True
                    match_R[v] = u
                    match_edge[v] = e_idx
                    matched = True
                    break
            if not matched:
                unmatched_L.append(u)

        visited = [-1] * N_prime
        vis_token = 0
        prev_node = [None] * N_prime

        for u in unmatched_L:
            vis_token += 1
            stack = [u]
            edge_ptr = [0] * N_prime
            visited[u] = vis_token
            found_v = -1
            last_u = -1
            last_e = -1

            while stack:
                curr = stack[-1]
                p = edge_ptr[curr]
                if p < len(adj[curr]):
                    v, e_idx = adj[curr][p]
                    edge_ptr[curr] += 1
                    nxt = match_R[v]
                    if nxt == -1:
                        found_v = v
                        last_u = curr
                        last_e = e_idx
                        break
                    elif visited[nxt] != vis_token:
                        visited[nxt] = vis_token
                        prev_node[nxt] = (curr, v, e_idx)
                        stack.append(nxt)
                else:
                    stack.pop()

            if found_v != -1:
                cur_v = found_v
                cur_u = last_u
                cur_e = last_e
                while True:
                    old_u = match_R[cur_v]
                    match_R[cur_v] = cur_u
                    match_edge[cur_v] = cur_e
                    if cur_u == u:
                        break
                    cur_u, cur_v, cur_e = prev_node[cur_u]

        matched_set = [False] * len(edges)
        matching = []
        remaining = []
        for e_idx in match_edge:
            matched_set[e_idx] = True
            matching.append(edges[e_idx])
        for i, e in enumerate(edges):
            if not matched_set[i]:
                remaining.append(e)

        return matching, remaining

    def solve_rec(edges, deg, color_offset):
        if deg == 0:
            return
        if deg == 1:
            for u, v, orig_id in edges:
                if orig_id != -1:
                    ans[orig_id] = color_offset
            return
        if deg % 2 == 1:
            matching, remaining = find_matching(edges)
            for u, v, orig_id in matching:
                if orig_id != -1:
                    ans[orig_id] = color_offset
            solve_rec(remaining, deg - 1, color_offset + 1)
        else:
            edges0, edges1 = split_euler(edges)
            solve_rec(edges0, deg // 2, color_offset)
            solve_rec(edges1, deg // 2, color_offset + deg // 2)

    solve_rec(graph_edges, Delta, 1)

    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == '__main__':
    solve()
