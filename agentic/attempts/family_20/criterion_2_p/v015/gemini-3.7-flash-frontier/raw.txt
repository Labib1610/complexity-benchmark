import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    nL = int(input_data[0])
    nR = int(input_data[1])
    m = int(input_data[2])

    if m == 0:
        print("YES")
        print()
        return

    deg_L = [0] * (nL + 1)
    deg_R = [0] * (nR + 1)

    edges_u = [0] * m
    edges_v = [0] * m

    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges_u[i] = u
        edges_v[i] = v
        deg_L[u] += 1
        deg_R[v] += 1

    Delta = max(max(deg_L), max(deg_R))

    # Map original vertices to super-vertices of capacity Delta
    map_L = [0] * (nL + 1)
    cur_super_L = 0
    cur_load_L = 0
    for u in range(1, nL + 1):
        if deg_L[u] == 0:
            continue
        if cur_load_L + deg_L[u] > Delta:
            cur_super_L += 1
            cur_load_L = 0
        map_L[u] = cur_super_L
        cur_load_L += deg_L[u]
    num_super_L = cur_super_L + 1 if m > 0 else 0

    map_R = [0] * (nR + 1)
    cur_super_R = 0
    cur_load_R = 0
    for v in range(1, nR + 1):
        if deg_R[v] == 0:
            continue
        if cur_load_R + deg_R[v] > Delta:
            cur_super_R += 1
            cur_load_R = 0
        map_R[v] = cur_super_R
        cur_load_R += deg_R[v]
    num_super_R = cur_super_R + 1 if m > 0 else 0

    N = max(num_super_L, num_super_R, 1)

    deg_sL = [0] * N
    deg_sR = [0] * N

    all_edges_u = []
    all_edges_v = []
    all_edges_id = []

    for i in range(m):
        u = map_L[edges_u[i]]
        v = map_R[edges_v[i]]
        all_edges_u.append(u)
        all_edges_v.append(v)
        all_edges_id.append(i)
        deg_sL[u] += 1
        deg_sR[v] += 1

    # Add dummy edges to make the graph Delta-regular
    ptr_L = 0
    ptr_R = 0
    while ptr_L < N and ptr_R < N:
        while ptr_L < N and deg_sL[ptr_L] == Delta:
            ptr_L += 1
        while ptr_R < N and deg_sR[ptr_R] == Delta:
            ptr_R += 1
        if ptr_L < N and ptr_R < N:
            add = min(Delta - deg_sL[ptr_L], Delta - deg_sR[ptr_R])
            for _ in range(add):
                all_edges_u.append(ptr_L)
                all_edges_v.append(ptr_R)
                all_edges_id.append(-1)
            deg_sL[ptr_L] += add
            deg_sR[ptr_R] += add

    ans = [0] * m

    def hopcroft_karp(cur_edges, cur_D):
        adj = [[] for _ in range(N)]
        for idx_e, (u, v, _) in enumerate(cur_edges):
            adj[u].append((v, idx_e))

        pair_u = [-1] * N
        pair_v = [-1] * N
        edge_for_u = [-1] * N
        dist = [-1] * N

        def bfs():
            queue = []
            for u in range(N):
                if pair_u[u] == -1:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = -1
            res = False
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v, _ in adj[u]:
                    next_u = pair_v[v]
                    if next_u != -1:
                        if dist[next_u] == -1:
                            dist[next_u] = dist[u] + 1
                            queue.append(next_u)
                    else:
                        res = True
            return res

        def dfs(u):
            for v, e_idx in adj[u]:
                next_u = pair_v[v]
                if next_u == -1 or (dist[next_u] == dist[u] + 1 and dfs(next_u)):
                    pair_u[u] = v
                    pair_v[v] = u
                    edge_for_u[u] = e_idx
                    return True
            dist[u] = -1
            return False

        matching = 0
        while bfs():
            for u in range(N):
                if pair_u[u] == -1:
                    if dfs(u):
                        matching += 1

        matched_edge_indices = set(edge_for_u[u] for u in range(N))
        return matched_edge_indices

    def divide_and_conquer(cur_edges, cur_D, color_offset):
        if cur_D == 0:
            return
        if cur_D == 1:
            for u, v, orig_id in cur_edges:
                if orig_id != -1:
                    ans[orig_id] = color_offset + 1
            return

        if cur_D % 2 == 1:
            matched_indices = hopcroft_karp(cur_edges, cur_D)
            matched_edges = []
            remaining_edges = []
            for idx_e, e in enumerate(cur_edges):
                if idx_e in matched_indices:
                    matched_edges.append(e)
                else:
                    remaining_edges.append(e)
            divide_and_conquer(matched_edges, 1, color_offset + cur_D - 1)
            divide_and_conquer(remaining_edges, cur_D - 1, color_offset)
            return

        # cur_D is even: Euler tour decomposition
        total_v = 2 * N
        head = [0] * total_v
        adj = [[] for _ in range(total_v)]
        num_e = len(cur_edges)
        for i, (u, v, orig_id) in enumerate(cur_edges):
            adj[u].append((v + N, i))
            adj[v + N].append((u, i))

        used = [False] * num_e
        edges1 = []
        edges2 = []

        for start_node in range(total_v):
            if head[start_node] < len(adj[start_node]):
                stack = [start_node]
                trail = []
                while stack:
                    u = stack[-1]
                    while head[u] < len(adj[u]) and used[adj[u][head[u]][1]]:
                        head[u] += 1
                    if head[u] < len(adj[u]):
                        nxt_node, e_idx = adj[u][head[u]]
                        used[e_idx] = True
                        head[u] += 1
                        stack.append(nxt_node)
                    else:
                        trail.append(stack.pop())

                for i in range(len(trail) - 1):
                    # We can assign alternating colors along the Euler trail
                    # We need the original edge
                    pass

        # Faster Euler tour edge separation:
        # Instead of vertex trail, we can track edge indices traversed in cycles
        used = [False] * num_e
        head = [0] * total_v

        for start_node in range(total_v):
            while head[start_node] < len(adj[start_node]):
                while head[start_node] < len(adj[start_node]) and used[adj[start_node][head[start_node]][1]]:
                    head[start_node] += 1
                if head[start_node] >= len(adj[start_node]):
                    break

                stack = [start_node]
                edge_stack = []
                cycle_edges = []

                while stack:
                    u = stack[-1]
                    while head[u] < len(adj[u]) and used[adj[u][head[u]][1]]:
                        head[u] += 1
                    if head[u] < len(adj[u]):
                        nxt_node, e_idx = adj[u][head[u]]
                        used[e_idx] = True
                        head[u] += 1
                        stack.append(nxt_node)
                        edge_stack.append(e_idx)
                    else:
                        stack.pop()
                        if edge_stack:
                            cycle_edges.append(edge_stack.pop())

                for i, e_idx in enumerate(cycle_edges):
                    if i % 2 == 0:
                        edges1.append(cur_edges[e_idx])
                    else:
                        edges2.append(cur_edges[e_idx])

        divide_and_conquer(edges1, cur_D // 2, color_offset)
        divide_and_conquer(edges2, cur_D // 2, color_offset + cur_D // 2)

    initial_edges = list(zip(all_edges_u, all_edges_v, all_edges_id))
    divide_and_conquer(initial_edges, Delta, 0)

    print("YES")
    print(*(ans[i] for i in range(m)))


if __name__ == "__main__":
    solve()
