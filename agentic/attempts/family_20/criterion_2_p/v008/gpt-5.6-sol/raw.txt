```python
import sys
from collections import defaultdict, deque

sys.setrecursionlimit(1_000_000)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    nL, nR, m = data[0], data[1], data[2]
    U = [0] * m
    V = [0] * m
    deg_l = [0] * nL
    deg_r = [0] * nR

    p = 3
    for i in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        U[i] = u
        V[i] = v
        deg_l[u] += 1
        deg_r[v] += 1

    if m == 0:
        sys.stdout.write("YES\n\n")
        return

    delta = max(max(deg_l), max(deg_r))
    answer = [0] * m
    pair_l = [-1] * m
    pair_r = [-1] * m
    next_color = 1

    def build_adjacency(edges):
        adj_l = defaultdict(list)
        adj_r = defaultdict(list)
        for e in edges:
            adj_l[U[e]].append(e)
            adj_r[V[e]].append(e)
        return adj_l, adj_r

    def matching_covering_max_degree(adj_l, adj_r, D):
        target_l = [u for u, es in adj_l.items() if len(es) == D]
        target_r_set = {v for v, es in adj_r.items() if len(es) == D}

        match_l = {}
        match_r = {}
        edge_l = {}
        edge_r = {}

        # First find a matching saturating all maximum-degree left vertices.
        while True:
            dist = {}
            q = deque()
            for u in target_l:
                if u not in match_l:
                    dist[u] = 0
                    q.append(u)

            shortest = 10**18
            while q:
                u = q.popleft()
                du = dist[u]
                if du >= shortest:
                    continue
                for e in adj_l[u]:
                    v = V[e]
                    nxt = match_r.get(v)
                    if nxt is None:
                        shortest = du + 1
                    elif nxt not in dist:
                        dist[nxt] = du + 1
                        q.append(nxt)

            if shortest == 10**18:
                break

            def dfs_left(u):
                du = dist[u]
                for e in adj_l[u]:
                    v = V[e]
                    nxt = match_r.get(v)
                    if nxt is None:
                        if du + 1 != shortest:
                            continue
                        match_l[u] = v
                        match_r[v] = u
                        edge_l[u] = e
                        edge_r[v] = e
                        return True
                    if dist.get(nxt) == du + 1 and dfs_left(nxt):
                        match_l[u] = v
                        match_r[v] = u
                        edge_l[u] = e
                        edge_r[v] = e
                        return True
                dist[u] = -1
                return False

            progress = False
            for u in target_l:
                if u not in match_l and dist.get(u, -1) == 0:
                    if dfs_left(u):
                        progress = True
            if not progress:
                break

        # Modify/augment the matching until all maximum-degree right vertices
        # are also saturated, without ever unsaturating a target left vertex.
        while True:
            roots = [v for v in target_r_set if v not in match_r]
            if not roots:
                break

            dist = {}
            q = deque()
            for v in roots:
                dist[v] = 0
                q.append(v)

            shortest = 10**18
            while q:
                v = q.popleft()
                dv = dist[v]
                if dv >= shortest:
                    continue
                for e in adj_r[v]:
                    u = U[e]
                    old = match_l.get(u)
                    if old is None or old not in target_r_set:
                        shortest = dv + 1
                    elif old not in dist:
                        dist[old] = dv + 1
                        q.append(old)

            def dfs_right(v):
                dv = dist[v]
                for e in adj_r[v]:
                    u = U[e]
                    old = match_l.get(u)

                    if old is None:
                        if dv + 1 != shortest:
                            continue
                        match_l[u] = v
                        match_r[v] = u
                        edge_l[u] = e
                        edge_r[v] = e
                        return True

                    if old not in target_r_set:
                        if dv + 1 != shortest:
                            continue
                        match_r.pop(old, None)
                        edge_r.pop(old, None)
                        match_l[u] = v
                        match_r[v] = u
                        edge_l[u] = e
                        edge_r[v] = e
                        return True

                    if dist.get(old) == dv + 1 and dfs_right(old):
                        match_l[u] = v
                        match_r[v] = u
                        edge_l[u] = e
                        edge_r[v] = e
                        return True

                dist[v] = -1
                return False

            progress = False
            for v in roots:
                if v not in match_r and dist.get(v, -1) == 0:
                    if dfs_right(v):
                        progress = True
            if not progress:
                break

        return list(edge_l.values())

    def split_even(edges, adj_l, adj_r):
        for e in edges:
            pair_l[e] = -1
            pair_r[e] = -1

        for es in adj_l.values():
            limit = len(es) - (len(es) & 1)
            for i in range(0, limit, 2):
                a, b = es[i], es[i + 1]
                pair_l[a] = b
                pair_l[b] = a

        for es in adj_r.values():
            limit = len(es) - (len(es) & 1)
            for i in range(0, limit, 2):
                a, b = es[i], es[i + 1]
                pair_r[a] = b
                pair_r[b] = a

        visited = set()
        first = []
        second = []

        def walk(start, start_side):
            e = start
            side = start_side
            parity = 0
            while e != -1 and e not in visited:
                visited.add(e)
                if parity == 0:
                    first.append(e)
                else:
                    second.append(e)
                parity ^= 1

                if side == 0:
                    e = pair_r[e]
                    side = 1
                else:
                    e = pair_l[e]
                    side = 0

        for e in edges:
            if e in visited:
                continue
            if pair_l[e] == -1:
                walk(e, 0)
            elif pair_r[e] == -1:
                walk(e, 1)

        for e in edges:
            if e not in visited:
                walk(e, 0)

        return first, second

    def color_graph(edges, D):
        nonlocal next_color

        if not edges:
            return

        if D <= 1:
            c = next_color
            next_color += 1
            for e in edges:
                answer[e] = c
            return

        adj_l, adj_r = build_adjacency(edges)

        if D & 1:
            matching = matching_covering_max_degree(adj_l, adj_r, D)
            c = next_color
            next_color += 1
            selected = set(matching)
            for e in matching:
                answer[e] = c
            remaining = [e for e in edges if e not in selected]
            color_graph(remaining, D - 1)
        else:
            first, second = split_even(edges, adj_l, adj_r)
            half = D // 2
            color_graph(first, half)
            color_graph(second, half)

    color_graph(list(range(m)), delta)

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, answer)) + "\n")


if __name__ == "__main__":
    main()
```
