    old_c = color[eid]
    if old_c == new_c: return
    u = left_vertex[eid]  # we can store endpoints arrays
    v = right_vertex[eid]

    del color_to_left[u][old_c]
    del color_to_right[v][old_c]

    used_right[v].remove(old_c)

    used_right[v].add(new_c)
