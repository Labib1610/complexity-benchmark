import sys
sys.setrecursionlimit(1 << 25)

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N_in = int(next(it))
    N_out = int(next(it))
    M = int(next(it))
    edges = []
    deg_in = [0] * (N_in + 1)
    deg_out = [0] * (N_out + 1)
    for _ in range(M):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        deg_in[u] += 1
        deg_out[v] += 1
    D = max(max(deg_in), max(deg_out))
    # adjacency: for each vertex store list of (neighbor, edge_id)
    adj_in = [[] for _ in range(N_in + 1)]
    adj_out = [[] for _ in range(N_out + 1)]
    for idx, (u, v) in enumerate(edges):
        adj_in[u].append((v, idx))
        adj_out[v].append((u, idx))

    # for each vertex, map color -> edge id using dict
    col_edge_in = [dict() for _ in range(N_in + 1)]
    col_edge_out = [dict() for _ in range(N_out + 1)]
    edge_color = [0] * M

    # helper to get other endpoint
    def other(eid, side, u_or_v):
        u, v = edges[eid]
        return v if side == 0 else u  # side 0: left u, side1: right v

    # attempt to assign color c to edge eid, possibly recoloring
    def dfs_recolor(eid, c, visited_e):
        if visited_e[eid]:
            return False
        visited_e[eid] = True
        u, v = edges[eid]
        # try to free color c at u
        if c not in col_edge_in[u]:
            # free at u, assign
            assign(eid, c)
            return True
        # else occupied at u by edge e1
        e1 = col_edge_in[u][c]
        # try to move e1 to another color
        for nc in range(1, D + 1):
            if nc == c:
                continue
            if nc not in col_edge_in[u] and nc not in col_edge_out[v]:
                # can directly swap colors
                assign(e1, nc)
                assign(eid, c)
                return True
            # try recursive recolor for e1
            if dfs_recolor(e1, nc, visited_e):
                assign(eid, c)
                return True
        return False

    def assign(eid, c):
        u, v = edges[eid]
        # remove previous assignment if any
        old = edge_color[eid]
        if old:
            col_edge_in[u].pop(old, None)
            col_edge_out[v].pop(old, None)
        edge_color[eid] = c
        col_edge_in[u][c] = eid
        col_edge_out[v][c] = eid

    # main coloring loop
    for eid in range(M):
        u, v = edges[eid]
        # find a free color directly
        for c in range(1, D + 1):
            if c not in col_edge_in[u] and c not in col_edge_out[v]:
                assign(eid, c)
                break
        else:
            # need recoloring
            visited = [False] * M
            for c in range(1, D + 1):
                if dfs_recolor(eid, c, visited):
                    break

    # output
    out = ['YES', '\n']
    out.append(' '.join(str(edge_color[i]) for i in range(M)))
    sys.stdout.write(''.join(out))

if __name__ == "__main__":
    main()
