import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    Nin = int(input_data[0])
    Nout = int(input_data[1])
    M = int(input_data[2])
    
    if M == 0:
        print("YES")
        print()
        return

    deg_in = [0] * (Nin + 1)
    deg_out = [0] * (Nout + 1)
    
    edges_u = [0] * M
    edges_v = [0] * M
    
    idx = 3
    for i in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges_u[i] = u
        edges_v[i] = v
        deg_in[u] += 1
        deg_out[v] += 1
        
    Delta = max(max(deg_in), max(deg_out))
    
    if Delta == 0:
        print("YES")
        print()
        return
    
    # Merge vertices on left side into bins of capacity Delta
    bin_L = [0] * (Nin + 1)
    L_bins = []
    cur_bin_deg = 0
    cur_bin_id = 0
    for u in range(1, Nin + 1):
        d = deg_in[u]
        if d == 0:
            continue
        if cur_bin_deg + d > Delta:
            L_bins.append(cur_bin_deg)
            cur_bin_id += 1
            cur_bin_deg = 0
        bin_L[u] = cur_bin_id
        cur_bin_deg += d
    if cur_bin_deg > 0:
        L_bins.append(cur_bin_deg)
        cur_bin_id += 1
    nL = max(1, cur_bin_id)
    if len(L_bins) == 0:
        L_bins.append(0)

    # Merge vertices on right side into bins of capacity Delta
    bin_R = [0] * (Nout + 1)
    R_bins = []
    cur_bin_deg = 0
    cur_bin_id = 0
    for v in range(1, Nout + 1):
        d = deg_out[v]
        if d == 0:
            continue
        if cur_bin_deg + d > Delta:
            R_bins.append(cur_bin_deg)
            cur_bin_id += 1
            cur_bin_deg = 0
        bin_R[v] = cur_bin_id
        cur_bin_deg += d
    if cur_bin_deg > 0:
        R_bins.append(cur_bin_deg)
        cur_bin_id += 1
    nR = max(1, cur_bin_id)
    if len(R_bins) == 0:
        R_bins.append(0)

    N = max(nL, nR)
    while len(L_bins) < N:
        L_bins.append(0)
    while len(R_bins) < N:
        R_bins.append(0)

    # Build regularized graph
    # Regular edges
    reg_edges = []
    for i in range(M):
        u = bin_L[edges_u[i]]
        v = bin_R[edges_v[i]]
        reg_edges.append((u, v, i))

    # Add dummy edges to make every bin have degree exactly Delta
    dummy_eid = M
    i_ptr = 0
    j_ptr = 0
    while i_ptr < N and j_ptr < N:
        rem_L = Delta - L_bins[i_ptr]
        rem_R = Delta - R_bins[j_ptr]
        if rem_L == 0:
            i_ptr += 1
            continue
        if rem_R == 0:
            j_ptr += 1
            continue
        add = min(rem_L, rem_R)
        for _ in range(add):
            reg_edges.append((i_ptr, j_ptr, dummy_eid))
            dummy_eid += 1
        L_bins[i_ptr] += add
        R_bins[j_ptr] += add

    ans = [0] * M

    def hopcroft_karp(N, adj):
        pair_u = [-1] * N
        pair_v = [-1] * N
        dist = [-1] * N

        def bfs():
            queue = deque()
            for u in range(N):
                if pair_u[u] == -1:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = -1
            found = False
            while queue:
                u = queue.popleft()
                for v, eid in adj[u]:
                    pu = pair_v[v]
                    if pu == -1:
                        found = True
                    elif dist[pu] == -1:
                        dist[pu] = dist[u] + 1
                        queue.append(pu)
            return found

        def dfs(u):
            for v, eid in adj[u]:
                pu = pair_v[v]
                if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                    pair_u[u] = (v, eid)
                    pair_v[v] = u
                    return True
            dist[u] = -1
            return False

        while bfs():
            for u in range(N):
                if pair_u[u] == -1:
                    dfs(u)
        return pair_u

    def color_graph(cur_edges, cur_deg, color_start):
        if cur_deg == 0 or not cur_edges:
            return
        if cur_deg == 1:
            for u, v, eid in cur_edges:
                if eid < M:
                    ans[eid] = color_start
            return

        if cur_deg % 2 == 1:
            adj = [[] for _ in range(N)]
            for u, v, eid in cur_edges:
                adj[u].append((v, eid))
            
            pair_u = hopcroft_karp(N, adj)
            matched_eids = set()
            for u in range(N):
                if pair_u[u] != -1:
                    v, eid = pair_u[u]
                    matched_eids.add(eid)
                    if eid < M:
                        ans[eid] = color_start
            
            remaining_edges = [e for e in cur_edges if e[2] not in matched_eids]
            color_graph(remaining_edges, cur_deg - 1, color_start + 1)
        else:
            # cur_deg is even: Euler tour decomposition
            adj = [[] for _ in range(2 * N)]
            for idx, (u, v, eid) in enumerate(cur_edges):
                adj[u].append((v + N, idx))
                adj[v + N].append((u, idx))

            used = [False] * len(cur_edges)
            head = [0] * (2 * N)
            
            E1 = []
            E2 = []

            for start_node in range(2 * N):
                if head[start_node] >= len(adj[start_node]):
                    continue
                stack = [start_node]
                tour_edges = []
                while stack:
                    u = stack[-1]
                    while head[u] < len(adj[u]) and used[adj[u][head[u]][1]]:
                        head[u] += 1
                    if head[u] < len(adj[u]):
                        v, e_idx = adj[u][head[u]]
                        head[u] += 1
                        used[e_idx] = True
                        stack.append(v)
                        tour_edges.append(e_idx)
                    else:
                        stack.pop()
                
                for k, e_idx in enumerate(tour_edges):
                    if k % 2 == 0:
                        E1.append(cur_edges[e_idx])
                    else:
                        E2.append(cur_edges[e_idx])

            half = cur_deg // 2
            color_graph(E1, half, color_start)
            color_graph(E2, half, color_start + half)

    color_graph(reg_edges, Delta, 1)

    print("YES")
    print(*(ans[i] for i in range(M)))

if __name__ == '__main__':
    sys.setrecursionlimit(2000000)
    solve()
