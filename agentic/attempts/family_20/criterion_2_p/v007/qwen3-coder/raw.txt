```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    nL = int(data[idx]); idx += 1
    nR = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1
    
    edges = []
    degree = [0] * (nL + nR + 1)
    adj = [[] for _ in range(nL + nR + 1)]
    
    for i in range(m):
        u = int(data[idx]) - 1  # Convert to 0-based
        idx += 1
        v = int(data[idx]) + nL - 1  # Convert to 0-based, right nodes start from nL
        idx += 1
        edges.append((u, v))
        degree[u] += 1
        degree[v] += 1
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    max_degree = max(degree) if degree else 0
    
    if max_degree == 0:
        print("YES")
        return
    
    # Color each component
    color = [0] * m
    used_colors = [False] * (max_degree + 1)
    node_color_used = [None] * (nL + nR)
    
    for start in range(nL + nR):
        if degree[start] == 0:
            continue
        if node_color_used[start] is None:
            node_color_used[start] = [False] * (max_degree + 1)
        
        queue = deque()
        queue.append(start)
        visited_nodes = set()
        visited_nodes.add(start)
        
        while queue:
            u = queue.popleft()
            forbidden = node_color_used[u]
            for v, edge_idx in adj[u]:
                if color[edge_idx] == 0:
                    # Find a free color
                    c = 1
                    if node_color_used[v] is None:
                        node_color_used[v] = [False] * (max_degree + 1)
                        visited_nodes.add(v)
                    while forbidden[c] or node_color_used[v][c]:
                        c += 1
                    color[edge_idx] = c
                    forbidden[c] = True
                    node_color_used[v][c] = True
                if v not in visited_nodes:
                    queue.append(v)
                    visited_nodes.add(v)
            
        # Clear the forbidden marks for next component
        for u in visited_nodes:
            if node_color_used[u] is not None:
                for c in range(max_degree + 1):
                    node_color_used[u][c] = False
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
```
