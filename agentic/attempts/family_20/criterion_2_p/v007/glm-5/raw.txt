import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    
    adjL = [[] for _ in range(nL + 1)]
    adjR = [[] for _ in range(nR + 1)]
    
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adjL[u].append(i)
        adjR[v].append(i)
    
    degL = [len(adjL[i]) for i in range(nL + 1)]
    degR = [len(adjR[i]) for i in range(nR + 1)]
    
    max_deg = 0
    for d in degL:
        if d > max_deg: max_deg = d
    for d in degR:
        if d > max_deg: max_deg = d
    
    if max_deg == 0:
        print("YES")
        print("")
        return
    
    # Build bipartite graph for edge coloring
    # Left side: ports 1..nL, Right side: ports nL+1..nL+nR
    # Edge i connects u to nL+v
    N = nL + nR
    graph = [[] for _ in range(N + 1)]
    for i, (u, v) in enumerate(edges):
        graph[u].append((nL + v, i))
        graph[nL + v].append((u, i))
    
    # Vizing's theorem for bipartite graphs: edge chromatic number = max degree
    # We use the standard algorithm
    
    color = [0] * m
    # For each vertex, track which colors are used on incident edges
    used = [set() for _ in range(N + 1)]
    
    # Process edges one by one
    for edge_idx in range(m):
        u, v = edges[edge_idx]
        v_node = nL + v
        
        # Find available color for this edge
        # Color c is available if c not in used[u] and c not in used[v_node]
        c1 = 1
        while c1 in used[u] or c1 in used[v_node]:
            c1 += 1
        
        if c1 <= max_deg:
            # Color c1 is available, use it
            color[edge_idx] = c1
            used[u].add(c1)
            used[v_node].add(c1)
        else:
            # No color available, need to recolor
            # Find c2 such that c2 not in used[u] (missing at u)
            c2 = 1
            while c2 in used[u]:
                c2 += 1
            
            # Find c3 such that c3 not in used[v_node] (missing at v)
            c3 = 1
            while c3 in used[v_node]:
                c3 += 1
            
            # Perform alternating path recoloring starting from v_node with colors c2, c3
            # We alternate between c2 and c3
            # Start from v_node, try to find edge colored c2
            # Then go to its other endpoint, try to find edge colored c3, etc.
            
            path = []
            current = v_node
            colors = (c2, c3)
            idx = 0
            
            while True:
                target_color = colors[idx]
                found = False
                for neighbor, e_idx in graph[current]:
                    if color[e_idx] == target_color:
                        path.append(e_idx)
                        current = neighbor
                        found = True
                        idx = 1 - idx
                        break
                if not found:
                    break
            
            # Recolor: swap colors along the path
            for i, e_idx in enumerate(path):
                old_color = color[e_idx]
                new_color = colors[(i + 1) % 2]
                color[e_idx] = new_color
                u_e, v_e = edges[e_idx]
                used[u_e].remove(old_color)
                used[u_e].add(new_color)
                used[nL + v_e].remove(old_color)
                used[nL + v_e].add(new_color)
            
            # Now color the new edge with c2
            color[edge_idx] = c2
            used[u].add(c2)
            used[v_node].add(c2)
    
    print("YES")
    print(" ".join(map(str, color)))

if __name__ == "__main__":
    solve()
