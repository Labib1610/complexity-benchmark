import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        nG = int(next(iterator))
        nA = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return
    
    tests = []
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        tests.append((u, v))
    
    # Calculate degree of each component
    degG = [0] * (nG + 1)
    degA = [0] * (nA + 1)
    for u, v in tests:
        degG[u] += 1
        degA[v] += 1
    
    # K is the maximum degree
    K = max(max(degG[1:]) if nG > 0 else 0, max(degA[1:]) if nA > 0 else 0)
    
    # Greedy assignment: for each test, pick smallest available phase
    # Use sets to track available phases for each component
    # Since we always pick smallest available, we can use a counter instead
    next_phase_G = [1] * (nG + 1)
    next_phase_A = [1] * (nA + 1)
    
    phases = []
    for u, v in tests:
        p = max(next_phase_G[u], next_phase_A[v])
        phases.append(p)
        next_phase_G[u] = p + 1
        next_phase_A[v] = p + 1
    
    # Verify correctness (optional but good for debugging)
    # For correctness, we need to ensure no component has two tests in same phase
    # The greedy algorithm ensures this by construction
    
    output = ['YES', ' '.join(map(str, phases))]
    sys.stdout.write('\n'.join(output))

if __name__ == "__main__":
    solve()
