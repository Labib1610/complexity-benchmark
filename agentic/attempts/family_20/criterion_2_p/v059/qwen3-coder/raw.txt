import sys
from collections import defaultdict
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n_G, n_A, m = map(int, sys.stdin.readline().split())
    
    # We'll treat this as a graph coloring problem.
    # Each test is an edge between a generator and an analyzer.
    # The constraint is that edges incident to the same vertex must have different colors.
    # This is equivalent to edge-coloring the bipartite graph.
    # By König's theorem, the edge chromatic number of a bipartite graph equals its maximum degree.
    
    # Store the tests
    tests = []
    deg_G = [0] * (n_G + 1)
    deg_A = [0] * (n_A + 1)
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        tests.append((u, v))
        deg_G[u] += 1
        deg_A[v] += 1
    
    K = max(max(deg_G[1:]), max(deg_A[1:])) if m > 0 else 0
    
    # Now we need to assign a color (phase) from 1 to K to each edge such that
    # no two edges incident to the same vertex have the same color.
    
    # For each vertex, maintain a set of used colors
    used_G = [set() for _ in range(n_G + 1)]
    used_A = [set() for _ in range(n_A + 1)]
    
    # For each vertex, maintain a list of edges (for efficient assignment)
    adj_G = [[] for _ in range(n_G + 1)]
    adj_A = [[] for _ in range(n_A + 1)]
    
    for i in range(m):
        u, v = tests[i]
        adj_G[u].append((v, i))
        adj_A[v].append((u, i))
    
    result = [0] * m
    
    # Process each generator
    for u in range(1, n_G + 1):
        # For each test involving this generator
        for v, idx in adj_G[u]:
            # Find the first available color for this edge
            color = 1
            while color in used_G[u] or color in used_A[v]:
                color += 1
            result[idx] = color
            used_G[u].add(color)
            used_A[v].add(color)
    
    print("YES")
    print(*result)

threading.Thread(target=main).start()
