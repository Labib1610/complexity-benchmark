import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    nG = int(data[0])
    nA = int(data[1])
    m = int(data[2])
    
    if m == 0:
        print("YES")
        print()
        return

    # Generators: 1 .. nG
    # Analyzers: nG + 1 .. nG + nA
    total_v = nG + nA
    G = [{} for _ in range(total_v + 1)]
    ans = [0] * m
    
    idx = 3
    for e in range(m):
        u = int(data[idx])
        v = int(data[idx + 1]) + nG
        idx += 2
        
        Gu = G[u]
        Gv = G[v]
        
        # Find a free color at u
        a = 1
        while a in Gu:
            a += 1
            
        # Find a free color at v
        b = 1
        while b in Gv:
            b += 1
            
        if a == b:
            Gu[a] = (v, e)
            Gv[a] = (u, e)
            ans[e] = a
        else:
            # Alternating path starting from v with color a
            cur = v
            c = a
            p_edges = []
            p_nodes = [v]
            
            while c in G[cur]:
                nxt, eid = G[cur][c]
                p_edges.append((eid, c))
                p_nodes.append(nxt)
                cur = nxt
                c = b if c == a else a
            
            # Flip colors along the alternating path
            for i, (eid, old_c) in enumerate(p_edges):
                u_node = p_nodes[i]
                v_node = p_nodes[i + 1]
                new_c = b if old_c == a else a
                ans[eid] = new_c
                del G[u_node][old_c]
                del G[v_node][old_c]
                G[u_node][new_c] = (v_node, eid)
                G[v_node][new_c] = (u_node, eid)
            
            # Add edge (u, v) with color a
            ans[e] = a
            G[u][a] = (v, e)
            G[v][a] = (u, e)
            
    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == '__main__':
    main()
