import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    raise SystemExit

nG, nA, m = data[0], data[1], data[2]
U = [0] * m
V = [0] * m

p = 3
for i in range(m):
    U[i] = data[p] - 1
    V[i] = data[p + 1] - 1
    p += 2

colors = [0] * m


def maximum_matching(adj_l):
    match_l = {}
    match_r = {}

    # Greedy initialization.
    for l, neighbors in adj_l.items():
        for r in neighbors:
            if r not in match_r:
                match_l[l] = r
                match_r[r] = l
                break

    INF = 10**18

    while True:
        dist = {}
        q = deque()
        for l in adj_l:
            if l not in match_l:
                dist[l] = 0
                q.append(l)

        best = INF
        while q:
            l = q.popleft()
            dl = dist[l]
            if dl + 1 > best:
                continue
            for r in adj_l[l]:
                nxt = match_r.get(r)
                if nxt is None:
                    best = dl + 1
                elif dl + 1 < best and nxt not in dist:
                    dist[nxt] = dl + 1
                    q.append(nxt)

        if best == INF:
            break

        def dfs(l):
            dl = dist[l]
            for r in adj_l[l]:
                nxt = match_r.get(r)
                if nxt is None:
                    if dl + 1 == best:
                        match_l[l] = r
                        match_r[r] = l
                        return True
                elif dist.get(nxt, INF) == dl + 1 and dfs(nxt):
                    match_l[l] = r
                    match_r[r] = l
                    return True
            dist[l] = INF
            return False

        changed = False
        for l in list(adj_l.keys()):
            if l not in match_l and dist.get(l, INF) == 0:
                if dfs(l):
                    changed = True
        if not changed:
            break

    return match_l, match_r


def force_required(adj, primary_match, opposite_match, required):
    INF = 10**18

    while True:
        sources = [x for x in required if x not in primary_match]
        if not sources:
            return

        dist = {}
        q = deque()
        for x in sources:
            dist[x] = 0
            q.append(x)

        best = INF
        while q:
            x = q.popleft()
            dx = dist[x]
            if dx + 1 > best:
                continue
            for y in adj.get(x, ()):
                nxt = opposite_match.get(y)
                if nxt is None:
                    best = dx + 1
                elif nxt not in required:
                    best = dx + 1
                elif dx + 1 < best and nxt not in dist:
                    dist[nxt] = dx + 1
                    q.append(nxt)

        if best == INF:
            raise RuntimeError("Required matching does not exist")

        def dfs(x):
            dx = dist[x]
            for y in adj.get(x, ()):
                nxt = opposite_match.get(y)

                if nxt is None:
                    if dx + 1 != best:
                        continue
                    primary_match[x] = y
                    opposite_match[y] = x
                    return True

                if nxt not in required:
                    if dx + 1 != best:
                        continue
                    primary_match.pop(nxt, None)
                    primary_match[x] = y
                    opposite_match[y] = x
                    return True

                if dist.get(nxt, INF) == dx + 1 and dfs(nxt):
                    primary_match[x] = y
                    opposite_match[y] = x
                    return True

            dist[x] = INF
            return False

        progress = False
        for x in sources:
            if x not in primary_match and dist.get(x, INF) == 0:
                if dfs(x):
                    progress = True

        if not progress:
            raise RuntimeError("Required matching adjustment failed")


def extract_required_matching(edges, degree_limit):
    adj_l = {}
    adj_r = {}
    deg_l = {}
    deg_r = {}

    for eid in edges:
        l = U[eid]
        r = V[eid]
        adj_l.setdefault(l, []).append(r)
        adj_r.setdefault(r, []).append(l)
        deg_l[l] = deg_l.get(l, 0) + 1
        deg_r[r] = deg_r.get(r, 0) + 1

    required_l = {x for x, d in deg_l.items() if d == degree_limit}
    required_r = {x for x, d in deg_r.items() if d == degree_limit}

    match_l, match_r = maximum_matching(adj_l)
    force_required(adj_l, match_l, match_r, required_l)
    force_required(adj_r, match_r, match_l, required_r)

    selected_by_left = {}
    for eid in edges:
        l = U[eid]
        if l not in selected_by_left and match_l.get(l) == V[eid]:
            selected_by_left[l] = eid

    return set(selected_by_left.values())


def split_edges(edges):
    adjacency = {}
    degree = {}

    for eid in edges:
        a = U[eid]
        b = nG + V[eid]
        adjacency.setdefault(a, []).append(eid)
        adjacency.setdefault(b, []).append(eid)
        degree[a] = degree.get(a, 0) + 1
        degree[b] = degree.get(b, 0) + 1

    odd_vertices = [x for x, d in degree.items() if d & 1]

    aug_a = []
    aug_b = []
    aug_real = []

    for eid in edges:
        aug_a.append(U[eid])
        aug_b.append(nG + V[eid])
        aug_real.append(eid)

    super_vertex = nG + nA
    if odd_vertices:
        adjacency.setdefault(super_vertex, [])
        for x in odd_vertices:
            idx = len(aug_real)
            aug_a.append(super_vertex)
            aug_b.append(x)
            aug_real.append(-1)
            adjacency[super_vertex].append(idx)
            adjacency[x].append(idx)

    used = bytearray(len(aug_real))
    ptr = {x: len(lst) - 1 for x, lst in adjacency.items()}

    first = []
    second = []

    def run_hierholzer(start):
        stack_v = [start]
        stack_e = []
        circuit = []

        while stack_v:
            x = stack_v[-1]
            lst = adjacency[x]
            i = ptr[x]
            while i >= 0 and used[lst[i]]:
                i -= 1
            ptr[x] = i

            if i < 0:
                stack_v.pop()
                if stack_e:
                    circuit.append(stack_e.pop())
                continue

            ae = lst[i]
            ptr[x] = i - 1
            if used[ae]:
                continue
            used[ae] = 1
            y = aug_b[ae] if aug_a[ae] == x else aug_a[ae]
            stack_v.append(y)
            stack_e.append(ae)

        for pos, ae in enumerate(circuit):
            eid = aug_real[ae]
            if eid >= 0:
                if pos & 1:
                    second.append(eid)
                else:
                    first.append(eid)

    if odd_vertices:
        run_hierholzer(super_vertex)

    for x in degree:
        lst = adjacency[x]
        i = ptr[x]
        while i >= 0 and used[lst[i]]:
            i -= 1
        ptr[x] = i
        if i >= 0:
            run_hierholzer(x)

    return first, second


def solve(edges, offset, capacity):
    if not edges:
        return

    degree = {}
    max_degree = 0
    for eid in edges:
        a = U[eid]
        b = nG + V[eid]
        da = degree.get(a, 0) + 1
        db = degree.get(b, 0) + 1
        degree[a] = da
        degree[b] = db
        if da > max_degree:
            max_degree = da
        if db > max_degree:
            max_degree = db

    if max_degree < capacity:
        capacity = max_degree

    if capacity == 1:
        c = offset + 1
        for eid in edges:
            colors[eid] = c
        return

    if capacity & 1:
        matching_edges = extract_required_matching(edges, capacity)
        matching_color = offset + capacity
        remaining = []
        for eid in edges:
            if eid in matching_edges:
                colors[eid] = matching_color
            else:
                remaining.append(eid)
        solve(remaining, offset, capacity - 1)
    else:
        left, right = split_edges(edges)
        half = capacity // 2
        solve(left, offset, half)
        solve(right, offset + half, half)


if m > 0:
    initial_degree = {}
    K = 0
    for i in range(m):
        a = U[i]
        b = nG + V[i]
        initial_degree[a] = initial_degree.get(a, 0) + 1
        initial_degree[b] = initial_degree.get(b, 0) + 1
        if initial_degree[a] > K:
            K = initial_degree[a]
        if initial_degree[b] > K:
            K = initial_degree[b]

    solve(list(range(m)), 0, K)

out = ["YES", " ".join(map(str, colors))]
sys.stdout.write("\n".join(out))
