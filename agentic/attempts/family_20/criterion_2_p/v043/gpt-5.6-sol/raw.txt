import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

n_left, n_right, m = data[0], data[1], data[2]

U = [0] * m
V = [0] * m
deg_left_total = [0] * n_left
deg_right_total = [0] * n_right

pos = 3
for i in range(m):
    u = data[pos] - 1
    v = data[pos + 1] - 1
    pos += 2
    U[i] = u
    V[i] = v
    deg_left_total[u] += 1
    deg_right_total[v] += 1

D = 0
if deg_left_total:
    D = max(D, max(deg_left_total))
if deg_right_total:
    D = max(D, max(deg_right_total))

if m == 0:
    sys.stdout.write("YES\n\n")
    sys.exit()

answer = [0] * m

n_vertices = n_left + n_right
pending = [-1] * n_vertices
link1 = [-1] * m
link2 = [-1] * m
visited_edge = [0] * m
visit_token = 0

adj_left = [None] * n_left
adj_right = [None] * n_right
local_deg_left = [0] * n_left
local_deg_right = [0] * n_right
pair_left = [-1] * n_left
pair_right = [-1] * n_right
dist = [-1] * n_left
seen_left = [0] * n_left
seen_token = 0
chosen_mark = [0] * m
chosen_token = 0


def split_even(edges):
    global visit_token

    touched_vertices = []

    for e in edges:
        link1[e] = -1
        link2[e] = -1

    for e in edges:
        a = U[e]
        b = n_left + V[e]

        old = pending[a]
        if old == -1:
            pending[a] = e
            touched_vertices.append(a)
        else:
            pending[a] = -1
            if link1[old] == -1:
                link1[old] = e
            else:
                link2[old] = e
            if link1[e] == -1:
                link1[e] = old
            else:
                link2[e] = old

        old = pending[b]
        if old == -1:
            pending[b] = e
            touched_vertices.append(b)
        else:
            pending[b] = -1
            if link1[old] == -1:
                link1[old] = e
            else:
                link2[old] = e
            if link1[e] == -1:
                link1[e] = old
            else:
                link2[e] = old

    for x in touched_vertices:
        pending[x] = -1

    visit_token += 1
    token = visit_token
    first = []
    second = []

    def walk(start):
        prev = -1
        cur = start
        parity = 0
        while cur != -1 and visited_edge[cur] != token:
            visited_edge[cur] = token
            if parity == 0:
                first.append(cur)
            else:
                second.append(cur)
            parity ^= 1

            a = link1[cur]
            b = link2[cur]
            if a != -1 and a != prev and visited_edge[a] != token:
                nxt = a
            elif b != -1 and b != prev and visited_edge[b] != token:
                nxt = b
            else:
                nxt = -1
            prev, cur = cur, nxt

    for e in edges:
        degree = (1 if link1[e] != -1 else 0) + (1 if link2[e] != -1 else 0)
        if degree < 2 and visited_edge[e] != token:
            walk(e)

    for e in edges:
        if visited_edge[e] != token:
            walk(e)

    return first, second


def mandatory_matching(edges, bound):
    global seen_token, chosen_token

    touched_left = []
    touched_right = []

    for e in edges:
        u = U[e]
        v = V[e]

        if adj_left[u] is None:
            adj_left[u] = []
            touched_left.append(u)
        adj_left[u].append(e)
        local_deg_left[u] += 1

        if adj_right[v] is None:
            adj_right[v] = []
            touched_right.append(v)
        adj_right[v].append(e)
        local_deg_right[v] += 1

    required_left = [u for u in touched_left if local_deg_left[u] == bound]
    required_right = [v for v in touched_right if local_deg_right[v] == bound]

    # First saturate all maximum-degree vertices on the left.
    while True:
        q = deque()
        for u in required_left:
            if pair_left[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = -1

        while q:
            u = q.popleft()
            nd = dist[u] + 1
            for e in adj_left[u]:
                old = pair_right[V[e]]
                if old != -1:
                    nu = U[old]
                    if dist[nu] == -1:
                        dist[nu] = nd
                        q.append(nu)

        def dfs_left(u):
            for e in adj_left[u]:
                v = V[e]
                old = pair_right[v]
                if old == -1:
                    pair_left[u] = e
                    pair_right[v] = e
                    return True
                nu = U[old]
                if dist[nu] == dist[u] + 1 and dfs_left(nu):
                    pair_left[u] = e
                    pair_right[v] = e
                    return True
            dist[u] = -1
            return False

        augmented = 0
        for u in required_left:
            if pair_left[u] == -1 and dfs_left(u):
                augmented += 1
        if augmented == 0:
            break

    # Then saturate all maximum-degree right vertices, preserving all
    # already saturated maximum-degree left vertices.
    def place_right(r, token):
        for e in adj_right[r]:
            u = U[e]
            if seen_left[u] == token:
                continue
            seen_left[u] = token

            old = pair_left[u]
            if old == -1:
                pair_left[u] = e
                pair_right[r] = e
                return True

            old_r = V[old]
            if local_deg_right[old_r] != bound:
                pair_right[old_r] = -1
                pair_left[u] = e
                pair_right[r] = e
                return True

            if place_right(old_r, token):
                pair_left[u] = e
                pair_right[r] = e
                return True
        return False

    for r in required_right:
        if pair_right[r] == -1:
            seen_token += 1
            place_right(r, seen_token)

    chosen = []
    chosen_token += 1
    token = chosen_token
    for u in touched_left:
        e = pair_left[u]
        if e != -1 and chosen_mark[e] != token:
            chosen_mark[e] = token
            chosen.append(e)

    for u in touched_left:
        adj_left[u] = None
        local_deg_left[u] = 0
        pair_left[u] = -1
        dist[u] = -1

    for v in touched_right:
        adj_right[v] = None
        local_deg_right[v] = 0
        pair_right[v] = -1

    return chosen, token


stack = [(list(range(m)), D, 1)]

while stack:
    edges, bound, offset = stack.pop()

    if not edges:
        continue

    if bound == 1:
        for e in edges:
            answer[e] = offset
        continue

    if bound % 2 == 0:
        half = bound // 2
        a, b = split_even(edges)
        stack.append((b, half, offset + half))
        stack.append((a, half, offset))
    else:
        chosen, token = mandatory_matching(edges, bound)
        for e in chosen:
            answer[e] = offset

        remaining = [e for e in edges if chosen_mark[e] != token]
        half = (bound - 1) // 2
        a, b = split_even(remaining)
        stack.append((b, half, offset + 1 + half))
        stack.append((a, half, offset + 1))

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answer)))
sys.stdout.write("\n")
