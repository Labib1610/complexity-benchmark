import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N_in = int(next(it))
    N_out = int(next(it))
    M = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(M)]

    # degree calculation
    deg_in = [0] * N_in
    deg_out = [0] * N_out
    for u, v in edges:
        deg_in[u] += 1
        deg_out[v] += 1
    D = max(max(deg_in, default=0), max(deg_out, default=0))

    # adjacency: for each vertex store color -> neighbor vertex
    # vertices are indexed 0..N_in+N_out-1, first part inputs, second part outputs
    total_vertices = N_in + N_out
    used = [{} for _ in range(total_vertices)]   # color -> neighbor vertex
    next_free = [1] * total_vertices            # next candidate color to try

    ans = [0] * M

    # helper to find a free color for a vertex
    def get_free(v):
        c = next_free[v]
        while c in used[v]:
            c += 1
        next_free[v] = c
        return c

    # augmenting recolor using DFS
    def dfs(u, target_color, seen):
        """try to make color target_color free at vertex u, recoloring along a path.
        returns True if successful."""
        if target_color not in used[u]:
            return True
        w = used[u][target_color]          # neighbor vertex currently using target_color
        if w in seen:
            return False
        seen.add(w)
        # try to move the edge (u,w) to another color
        for c in range(1, D + 1):
            if c == target_color:
                continue
            if c not in used[w]:
                # we can recolor (u,w) to c, freeing target_color at u
                used[w][c] = u
                used[u][c] = w
                del used[w][target_color]
                del used[u][target_color]
                return True
            # otherwise try to free c at w recursively
            if dfs(w, c, seen):
                # after recursion, c is free at w, so we can shift
                used[w][c] = u
                used[u][c] = w
                del used[w][target_color]
                del used[u][target_color]
                return True
        return False

    for idx, (u_in, v_out) in enumerate(edges):
        u = u_in
        v = N_in + v_out
        # try to assign a free color directly
        cu = get_free(u)
        cv = get_free(v)
        if cu == cv:
            color = cu
            used[u][color] = v
            used[v][color] = u
            ans[idx] = color
            continue
        # try to use cu if free at v, else cv if free at u
        if cu not in used[v]:
            color = cu
            used[u][color] = v
            used[v][color] = u
            ans[idx] = color
            continue
        if cv not in used[u]:
            color = cv
            used[u][color] = v
            used[v][color] = u
            ans[idx] = color
            continue
        # need recoloring: we will try to free cu at v via augmenting path
        seen = set()
        if dfs(v, cu, seen):
            color = cu
            used[u][color] = v
            used[v][color] = u
            ans[idx] = color
            continue
        # otherwise free cv at u
        seen.clear()
        if dfs(u, cv, seen):
            color = cv
            used[u][color] = v
            used[v][color] = u
            ans[idx] = color
            continue
        # fallback (should not happen)
        color = cu
        used[u][color] = v
        used[v][color] = u
        ans[idx] = color

    out = ['YES', ' '.join(map(str, ans))]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
