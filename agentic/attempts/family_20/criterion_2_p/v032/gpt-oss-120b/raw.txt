import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    import sys
    input = sys.stdin.readline
    nL, nR, m = map(int, input().split())
    edges = [tuple(map(int, input().split())) for _ in range(m)]

    # degree calculation
    degL = [0] * (nL + 1)
    degR = [0] * (nR + 1)
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = max(max(degL), max(degR), 1)

    # adjacency lists for colored edges
    # for each vertex store dict color->neighbor
    colL = [{} for _ in range(nL + 1)]
    colR = [{} for _ in range(nR + 1)]

    ans = [0] * m

    # helper to find a free color for a vertex
    def first_free(col_dict):
        # colors are 1..Delta, dict size <= degree
        # linear scan (Delta up to 4e5, but average small)
        for c in range(1, Delta + 1):
            if c not in col_dict:
                return c
        return 1  # never reached

    for idx, (u, v) in enumerate(edges):
        # try to assign a free color directly
        cu = first_free(colL[u])
        if cu not in colR[v]:
            # free for both
            colL[u][cu] = v
            colR[v][cu] = u
            ans[idx] = cu
            continue

        # need to recolor via alternating path
        # we will try to free color cu at v
        # BFS on vertices, alternating colors
        from collections import deque
        q = deque()
        prev = {}          # vertex -> (prev_vertex, color_used_to_enter)
        start = ('L', u)   # side, id
        q.append(start)
        prev[start] = None
        target = None
        target_color = cu

        while q:
            side, x = q.popleft()
            if side == 'L':
                # we are at left vertex x, we can move via any used color
                for c, nb in colL[x].items():
                    if (side, x, c) in prev:  # already visited this edge
                        continue
                    nxt = ('R', nb)
                    if nxt not in prev:
                        prev[nxt] = (side, x, c)
                        q.append(nxt)
                # also we can try the free color we want to free
                if target_color not in colL[x]:
                    # free at this left vertex -> we have found augmenting path
                    target = (side, x)
                    break
            else:  # side == 'R'
                # at right vertex x
                for c, nb in colR[x].items():
                    if (side, x, c) in prev:
                        continue
                    nxt = ('L', nb)
                    if nxt not in prev:
                        prev[nxt] = (side, x, c)
                        q.append(nxt)
                if target_color not in colR[x]:
                    target = (side, x)
                    break

        # reconstruct path and flip colors
        # target vertex has free target_color
        cur = target
        while prev[cur] is not None:
            pside, pnode, col = prev[cur]
            # edge (pnode, col) is currently colored col
            # we will recolor it to target_color
            if pside == 'L':
                # edge from left pnode to right cur_node
                right = cur[1]
                # remove old color
                del colL[pnode][col]
                del colR[right][col]
                # add new color
                colL[pnode][target_color] = right
                colR[right][target_color] = pnode
            else:
                left = cur[1]
                # edge from left left to right pnode
                del colL[left][col]
                del colR[pnode][col]
                colL[left][target_color] = pnode
                colR[pnode][target_color] = left
            cur = (pside, pnode)

        # finally assign the original edge with target_color
        colL[u][target_color] = v
        colR[v][target_color] = u
        ans[idx] = target_color

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
