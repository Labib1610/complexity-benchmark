```python
import sys
import random
from array import array

data = sys.stdin.buffer.read()
p = 0
ln = len(data)

def read_int():
    global p
    while p < ln and data[p] <= 32:
        p += 1
    x = 0
    while p < ln and data[p] > 32:
        x = x * 10 + data[p] - 48
        p += 1
    return x

n_left = read_int()
n_right = read_int()
m = read_int()
vertex_count = n_left + n_right

left = array('i')
right = array('i')
degree = array('i', [0]) * vertex_count

for _ in range(m):
    u = read_int() - 1
    v = n_left + read_int() - 1
    left.append(u)
    right.append(v)
    degree[u] += 1
    degree[v] += 1

if m == 0:
    sys.stdout.write("YES\n\n")
    raise SystemExit

delta = max(degree)
del data, degree

colors = array('i', [0]) * m
incident = [None] * vertex_count
missing = array('i', [1]) * vertex_count
next_color = array('i', [1]) * vertex_count
available = [None] * vertex_count

def find_missing(v):
    mp = incident[v]
    stack = available[v]
    if stack is not None:
        while stack:
            c = stack.pop()
            if c not in mp:
                return c

    c = next_color[v]
    while c <= delta and c in mp:
        c += 1
    next_color[v] = c + 1
    return c

order = list(range(m))
random.seed(712367821)
random.shuffle(order)

for edge_id in order:
    u = left[edge_id]
    v = right[edge_id]

    mu = incident[u]
    if mu is None:
        mu = {}
        incident[u] = mu

    mv = incident[v]
    if mv is None:
        mv = {}
        incident[v] = mv

    a = missing[u]
    b = missing[v]

    if a != b and a in mv:
        path = []
        cur = v
        expected = a

        while True:
            mp = incident[cur]
            e = mp.get(expected)
            if e is None:
                break
            path.append(e)
            if left[e] == cur:
                cur = right[e]
            else:
                cur = left[e]
            expected = b if expected == a else a

        end_vertex = cur
        old_missing = expected
        new_missing = b if old_missing == a else a

        for e in path:
            old = colors[e]
            new = b if old == a else a
            x = left[e]
            y = right[e]
            mx = incident[x]
            my = incident[y]
            del mx[old]
            del my[old]
            mx[new] = e
            my[new] = e
            colors[e] = new

        missing[v] = a

        if missing[end_vertex] == old_missing:
            missing[end_vertex] = new_missing
        else:
            stack = available[end_vertex]
            if stack is None:
                stack = []
                available[end_vertex] = stack
            stack.append(new_missing)

    mu[a] = edge_id
    mv[a] = edge_id
    colors[edge_id] = a

    if missing[u] == a:
        if len(mu) < delta:
            missing[u] = find_missing(u)
        else:
            missing[u] = 0

    if missing[v] == a:
        if len(mv) < delta:
            missing[v] = find_missing(v)
        else:
            missing[v] = 0

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, colors)))
sys.stdout.write("\n")
```
