```python
import sys
from collections import defaultdict
import heapq

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n_L = int(data[0])
    n_R = int(data[1])
    m = int(data[2])
    
    jobs = []
    proc_jobs = defaultdict(list)
    mem_jobs = defaultdict(list)
    
    idx = 3
    for i in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        jobs.append((u, v))
        proc_jobs[u].append(i)
        mem_jobs[v].append(i)
    
    # Compute Δ (maximum number of jobs sharing any single resource)
    Δ = 0
    for lst in proc_jobs.values():
        Δ = max(Δ, len(lst))
    for lst in mem_jobs.values():
        Δ = max(Δ, len(lst))
    
    # Initialize phase assignment array
    phases = [0] * m
    
    # For each processor and memory bank, maintain a min-heap of used phases
    proc_used = defaultdict(list)  # proc -> heap of phases
    mem_used = defaultdict(list)   # mem -> heap of phases
    
    # Also track which phases are used by each resource
    proc_phase_set = defaultdict(set)
    mem_phase_set = defaultdict(set)
    
    # We'll process jobs in an order such that we can assign phases greedily
    # A good heuristic is to process jobs by the sum of degrees (or just process them as they come)
    # But to ensure correctness with Δ phases, we can use a greedy coloring approach
    
    # Let's create a graph where each job is a node and edges represent conflicts
    # However, since m can be up to 400000, building the full graph is not feasible
    # Instead, we simulate a greedy assignment respecting both resource constraints
    
    # For each job, we'll pick the smallest available phase that's not used by its proc or mem
    
    for i in range(m):
        u, v = jobs[i]
        used_phases = proc_phase_set[u] | mem_phase_set[v]
        # Find the smallest phase in 1..Δ not in used_phases
        p = 1
        while p in used_phases:
            p += 1
        phases[i] = p
        proc_phase_set[u].add(p)
        mem_phase_set[v].add(p)
        
        # Push into heaps to allow releasing when needed (though not necessary here)
        heapq.heappush(proc_used[u], p)
        heapq.heappush(mem_used[v], p)
    
    print("YES")
    print(' '.join(map(str, phases)))

if __name__ == "__main__":
    main()
```
