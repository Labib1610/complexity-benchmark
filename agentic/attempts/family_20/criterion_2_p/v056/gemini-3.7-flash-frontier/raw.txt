import sys

# Increase recursion depth just in case
sys.setrecursionlimit(500000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n_V = int(input_data[0])
    n_D = int(input_data[1])
    m = int(input_data[2])

    if m == 0:
        print("YES")
        print()
        return

    edges_u = [0] * m
    edges_v = [0] * m
    deg_u = [0] * (n_V + 1)
    deg_v = [0] * (n_D + 1)

    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges_u[i] = u
        edges_v[i] = v
        deg_u[u] += 1
        deg_v[v] += 1

    Delta = max(max(deg_u), max(deg_v))

    # Pack vertices into bins of capacity Delta to minimize graph size
    # Left bins
    left_bins = []
    cur_bin = []
    cur_deg = 0
    for u in range(1, n_V + 1):
        if deg_u[u] == 0:
            continue
        if cur_deg + deg_u[u] <= Delta:
            cur_bin.append(u)
            cur_deg += deg_u[u]
        else:
            if cur_bin:
                left_bins.append(cur_bin)
            cur_bin = [u]
            cur_deg = deg_u[u]
    if cur_bin:
        left_bins.append(cur_bin)

    # Right bins
    right_bins = []
    cur_bin = []
    cur_deg = 0
    for v in range(1, n_D + 1):
        if deg_v[v] == 0:
            continue
        if cur_deg + deg_v[v] <= Delta:
            cur_bin.append(v)
            cur_deg += deg_v[v]
        else:
            if cur_bin:
                right_bins.append(cur_bin)
            cur_bin = [v]
            cur_deg = deg_v[v]
    if cur_bin:
        right_bins.append(cur_bin)

    num_bins = max(len(left_bins), len(right_bins))
    while len(left_bins) < num_bins:
        left_bins.append([])
    while len(right_bins) < num_bins:
        right_bins.append([])

    u_to_bin = [0] * (n_V + 1)
    for b_idx, bin_nodes in enumerate(left_bins):
        for u in bin_nodes:
            u_to_bin[u] = b_idx

    v_to_bin = [0] * (n_D + 1)
    for b_idx, bin_nodes in enumerate(right_bins):
        for v in bin_nodes:
            v_to_bin[v] = b_idx

    bin_deg_L = [0] * num_bins
    bin_deg_R = [0] * num_bins

    # Build regularized bipartite graph
    reg_u = [0] * (num_bins * Delta)
    reg_v = [0] * (num_bins * Delta)
    orig_id = [-1] * (num_bins * Delta)

    edge_cnt = 0
    for i in range(m):
        bu = u_to_bin[edges_u[i]]
        bv = v_to_bin[edges_v[i]]
        reg_u[edge_cnt] = bu
        reg_v[edge_cnt] = bv
        orig_id[edge_cnt] = i
        bin_deg_L[bu] += 1
        bin_deg_R[bv] += 1
        edge_cnt += 1

    # Add dummy edges to make every bin regular of degree Delta
    ptr_r = 0
    for bu in range(num_bins):
        while bin_deg_L[bu] < Delta:
            while ptr_r < num_bins and bin_deg_R[ptr_r] == Delta:
                ptr_r += 1
            if ptr_r == num_bins:
                break
            add = min(Delta - bin_deg_L[bu], Delta - bin_deg_R[ptr_r])
            for _ in range(add):
                reg_u[edge_cnt] = bu
                reg_v[edge_cnt] = ptr_r
                orig_id[edge_cnt] = -1
                edge_cnt += 1
            bin_deg_L[bu] += add
            bin_deg_R[ptr_r] += add

    N = num_bins
    ans = [0] * m

    # Hopcroft-Karp for finding a perfect matching in regular bipartite graph
    def find_matching(active_edges, cur_D):
        head_L = [-1] * N
        next_edge = [0] * len(active_edges)
        to_R = [0] * len(active_edges)
        edge_ref = [0] * len(active_edges)

        for i, e_idx in enumerate(active_edges):
            u = reg_u[e_idx]
            v = reg_v[e_idx]
            next_edge[i] = head_L[u]
            to_R[i] = v
            edge_ref[i] = e_idx
            head_L[u] = i

        match_L = [-1] * N
        match_R = [-1] * N
        match_edge_L = [-1] * N

        # Greedy matching first
        for u in range(N):
            curr = head_L[u]
            while curr != -1:
                v = to_R[curr]
                if match_R[v] == -1:
                    match_L[u] = v
                    match_R[v] = u
                    match_edge_L[u] = edge_ref[curr]
                    break
                curr = next_edge[curr]

        dist = [0] * N
        q = [0] * N

        def bfs():
            q_head = 0
            q_tail = 0
            for u in range(N):
                if match_L[u] == -1:
                    dist[u] = 0
                    q[q_tail] = u
                    q_tail += 1
                else:
                    dist[u] = -1
            found_free_R = False
            while q_head < q_tail:
                u = q[q_head]
                q_head += 1
                curr = head_L[u]
                while curr != -1:
                    v = to_R[curr]
                    u2 = match_R[v]
                    if u2 == -1:
                        found_free_R = True
                    elif dist[u2] == -1:
                        dist[u2] = dist[u] + 1
                        q[q_tail] = u2
                        q_tail += 1
                    curr = next_edge[curr]
            return found_free_R

        work_head = [0] * N

        def dfs(u):
            curr = work_head[u]
            while curr != -1:
                work_head[u] = next_edge[curr]
                v = to_R[curr]
                u2 = match_R[v]
                if u2 == -1 or (dist[u2] == dist[u] + 1 and dfs(u2)):
                    match_L[u] = v
                    match_R[v] = u
                    match_edge_L[u] = edge_ref[curr]
                    return True
                curr = work_head[u]
            return False

        while bfs():
            for u in range(N):
                work_head[u] = head_L[u]
            for u in range(N):
                if match_L[u] == -1:
                    dfs(u)

        matching_edges = set()
        for u in range(N):
            matching_edges.add(match_edge_L[u])
        return matching_edges

    def color_regular(active_edges, cur_D, offset):
        if cur_D == 1:
            for e_idx in active_edges:
                oid = orig_id[e_idx]
                if oid != -1:
                    ans[oid] = offset + 1
            return

        if cur_D % 2 == 1:
            matching_set = find_matching(active_edges, cur_D)
            remaining = []
            for e_idx in active_edges:
                if e_idx in matching_set:
                    oid = orig_id[e_idx]
                    if oid != -1:
                        ans[oid] = offset + cur_D
                else:
                    remaining.append(e_idx)
            color_regular(remaining, cur_D - 1, offset)
            return

        # cur_D is even: Euler decomposition
        # Build adjacency list with undirected multi-edges
        # Since it's regular of even degree, every vertex has degree cur_D.
        # We can find Eulerian circuits and alternate edges.
        num_v = 2 * N
        head = [-1] * num_v
        num_e = len(active_edges)
        to_node = [0] * (2 * num_e)
        nxt = [-1] * (2 * num_e)
        e_idx_map = [0] * (2 * num_e)

        for i, e_idx in enumerate(active_edges):
            u = reg_u[e_idx]
            v = reg_v[e_idx] + N
            # 2*i
            to_node[2 * i] = v
            nxt[2 * i] = head[u]
            e_idx_map[2 * i] = e_idx
            head[u] = 2 * i
            # 2*i + 1
            to_node[2 * i + 1] = u
            nxt[2 * i + 1] = head[v]
            e_idx_map[2 * i + 1] = e_idx
            head[v] = 2 * i + 1

        used_edge = [False] * num_e
        cur_edge_ptr = list(head)

        edges_0 = []
        edges_1 = []

        for start_v in range(num_v):
            while cur_edge_ptr[start_v] != -1:
                # Hierholzer's algorithm
                curr = start_v
                # We can trace an Eulerian trail/cycle iteratively
                stack = [curr]
                edge_stack = []
                while stack:
                    v = stack[-1]
                    e = cur_edge_ptr[v]
                    while e != -1 and used_edge[e // 2]:
                        e = nxt[e]
                    cur_edge_ptr[v] = e
                    if e != -1:
                        used_edge[e // 2] = True
                        cur_edge_ptr[v] = nxt[e]
                        nxt_v = to_node[e]
                        stack.append(nxt_v)
                        edge_stack.append(e_idx_map[e])
                    else:
                        stack.pop()
                        if edge_stack:
                            # Add to tour in alternate order
                            # But Hierholzer pops vertices when no edges left.
                            # We alternate colors along the circuit:
                            pass

                # Actually, simpler cycle-tracing:
                pass

        # Let's do standard alternating cycle tracing directly:
        used_edge = [False] * num_e
        cur_edge_ptr = list(head)
        edges_0 = []
        edges_1 = []

        for start_node in range(num_v):
            while cur_edge_ptr[start_node] != -1:
                e = cur_edge_ptr[start_node]
                while e != -1 and used_edge[e // 2]:
                    e = nxt[e]
                cur_edge_ptr[start_node] = e
                if e == -1:
                    break

                # Trace a cycle from start_node
                curr = start_node
                color = 0
                while True:
                    e = cur_edge_ptr[curr]
                    while e != -1 and used_edge[e // 2]:
                        e = nxt[e]
                    cur_edge_ptr[curr] = e
                    if e == -1:
                        break
                    used_edge[e // 2] = True
                    cur_edge_ptr[curr] = nxt[e]
                    real_e = e_idx_map[e]
                    if color == 0:
                        edges_0.append(real_e)
                        color = 1
                    else:
                        edges_1.append(real_e)
                        color = 0
                    curr = to_node[e]
                    if curr == start_node:
                        break

        half = cur_D // 2
        color_regular(edges_0, half, offset)
        color_regular(edges_1, half, offset + half)

    all_edges = list(range(edge_cnt))
    color_regular(all_edges, Delta, 0)

    print("YES")
    print(*(ans[i] for i in range(m)))


if __name__ == "__main__":
    solve()
