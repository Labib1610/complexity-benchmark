```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n_V = int(data[idx])
    idx += 1
    n_D = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    if m == 0:
        print("YES")
        return
    
    transfers = []
    vessel_deg = [0] * (n_V + 1)
    dock_deg = [0] * (n_D + 1)
    
    for i in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        idx += 2
        transfers.append((u, v))
        vessel_deg[u] += 1
        dock_deg[v] += 1
    
    delta = max(max(vessel_deg[1:]), max(dock_deg[1:]))
    
    if delta > min(n_V, n_D):
        print("NO")
        return
    
    # Build graph for each day
    # We'll use a greedy approach: for each transfer, assign it to the earliest day possible
    
    # For each vessel and dock, keep track of which days they're used
    vessel_days = [set() for _ in range(n_V + 1)]
    dock_days = [set() for _ in range(n_D + 1)]
    
    result = [0] * m
    
    # For each day, keep track of how many transfers are scheduled
    day_count = [0] * (delta + 1)  # 1-based indexing
    
    for i in range(m):
        u, v = transfers[i]
        # Find the first day that is available for both u and v and within capacity
        day = 1
        while day <= delta:
            if (day not in vessel_days[u] and 
                day not in dock_days[v] and 
                day_count[day] < n_V and 
                day_count[day] < n_D):
                break
            day += 1
        else:
            print("NO")
            return
        
        result[i] = day
        vessel_days[u].add(day)
        dock_days[v].add(day)
        day_count[day] += 1
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
