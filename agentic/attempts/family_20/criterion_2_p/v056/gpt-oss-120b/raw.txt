import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    nV = int(next(it))
    nD = int(next(it))
    m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    degV = [0] * nV
    degD = [0] * nD
    for u, v in edges:
        degV[u] += 1
        degD[v] += 1
    Delta = 0
    for d in degV:
        if d > Delta:
            Delta = d
    for d in degD:
        if d > Delta:
            Delta = d
    if Delta == 0:
        print("YES")
        print()
        return

    # for each vertex keep color -> neighbor edge id
    colV = [dict() for _ in range(nV)]
    colD = [dict() for _ in range(nD)]
    ans = [0] * m

    # helper to find augmenting path and recolor
    def dfs(u, target_color, seen):
        # try to free target_color on vertex u (vessel side)
        if u in seen:
            return False
        seen.add(u)
        if target_color not in colV[u]:
            return True
        # edge currently using target_color
        eid = colV[u][target_color]
        _, v = edges[eid]
        # try to move this edge to another color on dock side
        for c in range(1, Delta + 1):
            if c == target_color:
                continue
            if c not in colD[v]:
                # recolor edge eid to c
                colV[u].pop(target_color)
                colD[v].pop(target_color)
                colV[u][c] = eid
                colD[v][c] = eid
                ans[eid] = c
                return True
        # try deeper alternating path
        for c in range(1, Delta + 1):
            if c == target_color:
                continue
            if c in colD[v]:
                eid2 = colD[v][c]
                u2, _ = edges[eid2]
                if dfs(u2, c, seen):
                    # shift eid2 to target_color on dock side
                    colV[u2].pop(c)
                    colD[v].pop(c)
                    colV[u2][target_color] = eid2
                    colD[v][target_color] = eid2
                    ans[eid2] = target_color
                    # now free target_color on u
                    colV[u].pop(target_color)
                    colD[v].pop(target_color)
                    colV[u][c] = eid
                    colD[v][c] = eid
                    ans[eid] = c
                    return True
        return False

    for eid, (u, v) in enumerate(edges):
        # try to assign smallest free color
        for c in range(1, Delta + 1):
            if c not in colV[u] and c not in colD[v]:
                colV[u][c] = eid
                colD[v][c] = eid
                ans[eid] = c
                break
        else:
            # need to recolor
            for c in range(1, Delta + 1):
                if c not in colV[u]:
                    # try to free c on dock side via augmenting path
                    seen = set()
                    if dfs(u, c, seen):
                        colV[u][c] = eid
                        colD[v][c] = eid
                        ans[eid] = c
                        break

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
