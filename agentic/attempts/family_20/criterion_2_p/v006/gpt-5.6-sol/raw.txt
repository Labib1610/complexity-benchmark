import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)
nL = next(it)
nR = next(it)
m = next(it)

U = [0] * m
V = [0] * m
deg_l = [0] * nL
deg_r = [0] * nR

for i in range(m):
    u = next(it) - 1
    v = next(it) - 1
    U[i] = u
    V[i] = v
    deg_l[u] += 1
    deg_r[v] += 1

if m == 0:
    sys.stdout.write("YES\n\n")
    raise SystemExit

delta = max(max(deg_l), max(deg_r))
answer = [0] * m


def actual_max_degree(edges):
    dl = {}
    dr = {}
    maximum = 0
    for eid in edges:
        u = U[eid]
        v = V[eid]
        x = dl.get(u, 0) + 1
        dl[u] = x
        if x > maximum:
            maximum = x
        x = dr.get(v, 0) + 1
        dr[v] = x
        if x > maximum:
            maximum = x
    return maximum


def split_balanced(edges):
    vertex_id = {}
    adj = []

    def get_vertex(key):
        idx = vertex_id.get(key)
        if idx is None:
            idx = len(adj)
            vertex_id[key] = idx
            adj.append([])
        return idx

    a = []
    b = []
    original = []

    for eid in edges:
        x = get_vertex(U[eid])
        y = get_vertex(nL + V[eid])
        idx = len(a)
        a.append(x)
        b.append(y)
        original.append(eid)
        adj[x].append(idx)
        adj[y].append(idx)

    odd_l = []
    odd_r = []
    for key, idx in vertex_id.items():
        if len(adj[idx]) & 1:
            if key < nL:
                odd_l.append(idx)
            else:
                odd_r.append(idx)

    def add_dummy(x, y):
        idx = len(a)
        a.append(x)
        b.append(y)
        original.append(-1)
        adj[x].append(idx)
        adj[y].append(idx)

    common = min(len(odd_l), len(odd_r))
    for i in range(common):
        add_dummy(odd_l[i], odd_r[i])

    next_dummy_key = nL + nR

    i = common
    while i < len(odd_l):
        x = odd_l[i]
        y = odd_l[i + 1]
        z = get_vertex(next_dummy_key)
        next_dummy_key += 1
        add_dummy(x, z)
        add_dummy(y, z)
        i += 2

    i = common
    while i < len(odd_r):
        x = odd_r[i]
        y = odd_r[i + 1]
        z = get_vertex(next_dummy_key)
        next_dummy_key += 1
        add_dummy(x, z)
        add_dummy(y, z)
        i += 2

    used = bytearray(len(a))
    first = []
    second = []

    for start in range(len(adj)):
        while adj[start] and used[adj[start][-1]]:
            adj[start].pop()
        if not adj[start]:
            continue

        stack_v = [start]
        stack_e = []
        circuit = []

        while stack_v:
            x = stack_v[-1]
            while adj[x] and used[adj[x][-1]]:
                adj[x].pop()

            if adj[x]:
                eid = adj[x].pop()
                if used[eid]:
                    continue
                used[eid] = 1
                y = b[eid] if a[eid] == x else a[eid]
                stack_v.append(y)
                stack_e.append(eid)
            else:
                stack_v.pop()
                if stack_e:
                    circuit.append(stack_e.pop())

        for parity, local_eid in enumerate(circuit):
            eid = original[local_eid]
            if eid >= 0:
                if parity & 1:
                    second.append(eid)
                else:
                    first.append(eid)

    return first, second


def critical_matching(edges, d):
    l_index = {}
    r_index = {}
    l_vertices = []
    r_vertices = []
    dl = []
    dr = []

    for eid in edges:
        u = U[eid]
        v = V[eid]

        li = l_index.get(u)
        if li is None:
            li = len(l_vertices)
            l_index[u] = li
            l_vertices.append(u)
            dl.append(0)
        dl[li] += 1

        ri = r_index.get(v)
        if ri is None:
            ri = len(r_vertices)
            r_index[v] = ri
            r_vertices.append(v)
            dr.append(0)
        dr[ri] += 1

    nl = len(l_vertices)
    nr = len(r_vertices)

    critical_l = [i for i, x in enumerate(dl) if x == d]
    critical_r = [i for i, x in enumerate(dr) if x == d]

    if not critical_l and not critical_r:
        return [], edges

    S = 0
    L0 = 1
    R0 = L0 + nl
    T = R0 + nr
    SS = T + 1
    TT = SS + 1
    N = TT + 1

    graph = [[] for _ in range(N)]

    def add_edge(x, y, cap):
        forward = [y, len(graph[y]), cap]
        backward = [x, len(graph[x]), 0]
        graph[x].append(forward)
        graph[y].append(backward)
        return len(graph[x]) - 1

    critical_l_set = bytearray(nl)
    critical_r_set = bytearray(nr)
    for x in critical_l:
        critical_l_set[x] = 1
    for x in critical_r:
        critical_r_set[x] = 1

    for i in range(nl):
        add_edge(S, L0 + i, 0 if critical_l_set[i] else 1)

    refs = [None] * len(edges)
    for pos, eid in enumerate(edges):
        li = l_index[U[eid]]
        ri = r_index[V[eid]]
        node = L0 + li
        idx = add_edge(node, R0 + ri, 1)
        refs[pos] = (node, idx)

    for i in range(nr):
        add_edge(R0 + i, T, 0 if critical_r_set[i] else 1)

    total_required = len(critical_l) + len(critical_r)
    add_edge(T, S, total_required + 1)

    for i in critical_l:
        add_edge(SS, L0 + i, 1)
    for i in critical_r:
        add_edge(R0 + i, TT, 1)

    if critical_l:
        add_edge(S, TT, len(critical_l))
    if critical_r:
        add_edge(SS, T, len(critical_r))

    level = [-1] * N
    ptr = [0] * N

    def bfs():
        for i in range(N):
            level[i] = -1
        level[SS] = 0
        q = deque([SS])
        while q:
            x = q.popleft()
            nx = level[x] + 1
            for e in graph[x]:
                if e[2] and level[e[0]] < 0:
                    level[e[0]] = nx
                    q.append(e[0])
        return level[TT] >= 0

    def dfs(x, pushed):
        if x == TT:
            return pushed
        gx = graph[x]
        while ptr[x] < len(gx):
            e = gx[ptr[x]]
            if e[2] and level[e[0]] == level[x] + 1:
                flow = dfs(e[0], min(pushed, e[2]))
                if flow:
                    e[2] -= flow
                    graph[e[0]][e[1]][2] += flow
                    return flow
            ptr[x] += 1
        return 0

    flow = 0
    while flow < total_required and bfs():
        for i in range(N):
            ptr[i] = 0
        while flow < total_required:
            pushed = dfs(SS, total_required - flow)
            if not pushed:
                break
            flow += pushed

    chosen = []
    remaining = []
    for pos, eid in enumerate(edges):
        node, idx = refs[pos]
        if graph[node][idx][2] == 0:
            chosen.append(eid)
        else:
            remaining.append(eid)

    return chosen, remaining


tasks = [(list(range(m)), delta, 0)]

while tasks:
    edges, capacity, offset = tasks.pop()
    if not edges:
        continue

    d = actual_max_degree(edges)
    if d == 1:
        color = offset + 1
        for eid in edges:
            answer[eid] = color
        continue

    if d & 1:
        matching, rest = critical_matching(edges, d)
        color = offset + d
        for eid in matching:
            answer[eid] = color
        if rest:
            tasks.append((rest, d - 1, offset))
    else:
        first, second = split_balanced(edges)
        half = d // 2
        if second:
            tasks.append((second, half, offset + half))
        if first:
            tasks.append((first, half, offset))

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answer)))
sys.stdout.write("\n")
