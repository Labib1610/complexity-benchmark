import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))

    edges = []
    deg = [0] * (nL + nR + 1)
    offset = nL

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        deg[u] += 1
        deg[offset + v] += 1

    delta = max(deg) if m > 0 else 0
    if delta == 0:
        # m == 0
        out = ["YES", ""]
        sys.stdout.write("\n".join(out))
        return

    # Greedy edge coloring: for each edge, pick the smallest color
    # not used by either endpoint. Since the graph is bipartite,
    # this greedy approach uses at most delta colors.
    color = [0] * (nL + nR + 1)
    ans = [0] * m

    # Use a dictionary to track available colors per vertex
    # But for efficiency, we use a "next available" pointer approach
    # combined with a set of used colors.
    # For large delta, we need a smarter approach.
    
    # Actually, for bipartite graphs, the greedy algorithm with
    # "smallest available color" works and uses at most delta colors.
    # We need an efficient implementation.
    
    # For each vertex, track used colors in a set and the next candidate color.
    used = [set() for _ in range(nL + nR + 1)]
    nxt = [1] * (nL + nR + 1)

    for i in range(m):
        u, v = edges[i]
        vv = offset + v
        
        # Find smallest available color for u
        cu = nxt[u]
        while cu in used[u]:
            cu += 1
        
        # Find smallest available color for v
        cv = nxt[vv]
        while cv in used[vv]:
            cv += 1
        
        c = max(cu, cv)
        ans[i] = c
        used[u].add(c)
        used[vv].add(c)
        nxt[u] = c
        nxt[vv] = c

    out = ["YES", " ".join(map(str, ans))]
    sys.stdout.write("\n".join(out))

solve()
