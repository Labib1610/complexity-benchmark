import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n_L, n_R, m = map(int, sys.stdin.readline().split())
    
    # We will model this as a graph where:
    # Each edge represents a fiber strand
    # Each node represents either a transmitter or a receiver
    # We want to color the edges such that no two edges sharing a node have the same color
    
    # Degree of each node
    deg_L = [0] * (n_L + 1)
    deg_R = [0] * (n_R + 1)
    
    edges = []
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        deg_L[u] += 1
        deg_R[v] += 1
    
    D = max(max(deg_L[1:]), max(deg_R[1:])) if m > 0 else 0
    
    # For each node, maintain a set of used colors
    adj_L = [set() for _ in range(n_L + 1)]
    adj_R = [set() for _ in range(n_R + 1)]
    
    result = []
    
    # For each edge, assign the smallest available color not used by its endpoints
    for u, v in edges:
        # Find the smallest color not in adj_L[u] or adj_R[v]
        color = 1
        while color in adj_L[u] or color in adj_R[v]:
            color += 1
        adj_L[u].add(color)
        adj_R[v].add(color)
        result.append(str(color))
    
    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    main()
