import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    
    edges = []
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
    
    # Degree counts
    degL = [0] * (nL + 1)
    degR = [0] * (nR + 1)
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    
    max_deg = 0
    for d in degL:
        if d > max_deg: max_deg = d
    for d in degR:
        if d > max_deg: max_deg = d
    
    if max_deg == 0:
        print("YES")
        print("")
        return
    
    # Build adjacency lists
    adjL = [[] for _ in range(nL + 1)]
    adjR = [[] for _ in range(nR + 1)]
    for idx, (u, v) in enumerate(edges):
        adjL[u].append((v, idx))
        adjR[v].append((u, idx))
    
    # Current pointer for each vertex
    ptrL = [0] * (nL + 1)
    ptrR = [0] * (nR + 1)
    
    # Match arrays: matchL[u] = (v, idx) or None
    matchL = [None] * (nL + 1)
    matchR = [None] * (nR + 1)
    
    # BFS queue
    q = deque()
    
    # Initialize: all vertices with degree < max_deg are free
    for u in range(1, nL + 1):
        if degL[u] < max_deg:
            q.append(('L', u))
    for v in range(1, nR + 1):
        if degR[v] < max_deg:
            q.append(('R', v))
    
    while q:
        side, u = q.popleft()
        
        if side == 'L':
            # Find next unmatched edge from u
            while ptrL[u] < len(adjL[u]):
                v, idx = adjL[u][ptrL[u]]
                ptrL[u] += 1
                if matchR[v] is None:
                    # Match u with v
                    matchL[u] = (v, idx)
                    matchR[v] = (u, idx)
                    break
            else:
                continue  # No unmatched edge found
        else:  # side == 'R'
            # Find next unmatched edge from v
            while ptrR[u] < len(adjR[u]):
                w, idx = adjR[u][ptrR[u]]
                ptrR[u] += 1
                if matchL[w] is None:
                    # Match w with u (u is v in the original notation)
                    matchL[w] = (u, idx)
                    matchR[u] = (w, idx)
                    break
            else:
                continue  # No unmatched edge found
        
        # Successfully matched an edge, now check if any vertex became saturated
        # Check left vertex
        if degL[u] == max_deg:
            # u just became saturated, add to queue
            q.append(('L', u))
        # Check right vertex (the one we matched with)
        v = matchL[u][0] if side == 'L' else u
        if degR[v] == max_deg:
            q.append(('R', v))
    
    # Assign colors based on matches
    color = [0] * m
    for u in range(1, nL + 1):
        if matchL[u] is not None:
            _, idx = matchL[u]
            color[idx] = 1
    
    # For uncolored edges, assign colors greedily
    # We need to ensure no two edges sharing a vertex have the same color
    # Since we have max_deg colors and the matching covers all saturated vertices,
    # we can greedily assign colors
    
    # For each vertex, track which colors are used
    usedL = [set() for _ in range(nL + 1)]
    usedR = [set() for _ in range(nR + 1)]
    
    # First, mark colors from matched edges
    for u in range(1, nL + 1):
        if matchL[u] is not None:
            v, idx = matchL[u]
            usedL[u].add(1)
            usedR[v].add(1)
    
    # Now assign colors to uncolored edges
    for idx, (u, v) in enumerate(edges):
        if color[idx] == 0:
            # Find smallest available color
            for c in range(1, max_deg + 1):
                if c not in usedL[u] and c not in usedR[v]:
                    color[idx] = c
                    usedL[u].add(c)
                    usedR[v].add(c)
                    break
    
    print("YES")
    print(" ".join(map(str, color)))

if __name__ == "__main__":
    solve()
