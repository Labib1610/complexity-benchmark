import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)
nL = next(it)
nR = next(it)
m = next(it)

U = [0] * m
V = [0] * m
for i in range(m):
    U[i] = next(it) - 1
    V[i] = next(it) - 1

answer = [0] * m

if m == 0:
    sys.stdout.write("YES\n\n")
    raise SystemExit

deg_l_global = [0] * nL
deg_r_global = [0] * nR
for u, v in zip(U, V):
    deg_l_global[u] += 1
    deg_r_global[v] += 1

delta = max(max(deg_l_global), max(deg_r_global))


def balanced_split(edges):
    size = len(edges)
    super_node = nL + nR

    a = [0] * size
    b = [0] * size
    original = list(edges)
    adjacency = {}

    for i, eid in enumerate(edges):
        x = U[eid]
        y = nL + V[eid]
        a[i] = x
        b[i] = y
        adjacency.setdefault(x, []).append(i)
        adjacency.setdefault(y, []).append(i)

    odd_vertices = [x for x, lst in adjacency.items() if len(lst) & 1]

    if odd_vertices:
        adjacency[super_node] = []
        for x in odd_vertices:
            eid = len(a)
            a.append(super_node)
            b.append(x)
            original.append(-1)
            adjacency[super_node].append(eid)
            adjacency[x].append(eid)

    used = bytearray(len(a))
    left_part = []
    right_part = []

    def run_tour(start):
        node_stack = [start]
        edge_stack = []
        circuit = []

        while node_stack:
            x = node_stack[-1]
            lst = adjacency[x]
            while lst and used[lst[-1]]:
                lst.pop()

            if not lst:
                node_stack.pop()
                if edge_stack:
                    circuit.append(edge_stack.pop())
                continue

            eid = lst.pop()
            if used[eid]:
                continue
            used[eid] = 1

            y = b[eid] if a[eid] == x else a[eid]
            node_stack.append(y)
            edge_stack.append(eid)

        for pos, local_eid in enumerate(circuit):
            real_eid = original[local_eid]
            if real_eid >= 0:
                if pos & 1:
                    right_part.append(real_eid)
                else:
                    left_part.append(real_eid)

    if odd_vertices:
        run_tour(super_node)

    for x in list(adjacency.keys()):
        lst = adjacency[x]
        while lst and used[lst[-1]]:
            lst.pop()
        if lst:
            run_tour(x)

    return left_part, right_part


def extract_matching(edges, degree_bound, deg_l, deg_r, adj_l, adj_r):
    required_l = {u for u, d in deg_l.items() if d == degree_bound}
    required_r = [v for v, d in deg_r.items() if d == degree_bound]

    match_l = {}
    match_r = {}

    # First find a matching saturating all required right vertices.
    inf = 10 ** 18
    dist = {}

    def bfs():
        q = deque()
        found = False
        for r in required_r:
            if r not in match_r:
                dist[r] = 0
                q.append(r)
            else:
                dist[r] = inf

        while q:
            r = q.popleft()
            nd = dist[r] + 1
            for eid in adj_r[r]:
                l = U[eid]
                old_eid = match_l.get(l)
                if old_eid is None:
                    found = True
                else:
                    old_r = V[old_eid]
                    if dist.get(old_r, inf) == inf:
                        dist[old_r] = nd
                        q.append(old_r)
        return found

    def dfs_r(r):
        for eid in adj_r[r]:
            l = U[eid]
            old_eid = match_l.get(l)
            if old_eid is None:
                match_l[l] = eid
                match_r[r] = eid
                return True

            old_r = V[old_eid]
            if dist.get(old_r, inf) == dist[r] + 1 and dfs_r(old_r):
                match_l[l] = eid
                match_r[r] = eid
                return True

        dist[r] = inf
        return False

    remaining = len(required_r)
    while remaining:
        if not bfs():
            break
        gained = 0
        for r in required_r:
            if r not in match_r and dfs_r(r):
                gained += 1
        if gained == 0:
            break
        remaining -= gained

    # Transfer unmatched status among left vertices until every required
    # left vertex is matched, without unmatching any right vertex.
    seen = {}
    stamp = 0

    def dfs_l(l):
        nonlocal stamp
        if seen.get(l) == stamp:
            return False
        seen[l] = stamp

        for eid in adj_l[l]:
            r = V[eid]
            old_eid = match_r.get(r)

            if old_eid is None:
                match_l[l] = eid
                match_r[r] = eid
                return True

            old_l = U[old_eid]

            if old_l not in required_l:
                match_l.pop(old_l, None)
                match_l[l] = eid
                match_r[r] = eid
                return True

            if dfs_l(old_l):
                match_l[l] = eid
                match_r[r] = eid
                return True

        return False

    for l in required_l:
        if l not in match_l:
            stamp += 1
            if not dfs_l(l):
                raise RuntimeError("Failed to construct required matching")

    return set(match_r.values())


def color_edges(edges, base_color, bound):
    if not edges:
        return

    deg_l = {}
    deg_r = {}
    adj_l = {}
    adj_r = {}

    actual = 0
    for eid in edges:
        u = U[eid]
        v = V[eid]

        nd = deg_l.get(u, 0) + 1
        deg_l[u] = nd
        if nd > actual:
            actual = nd

        nd = deg_r.get(v, 0) + 1
        deg_r[v] = nd
        if nd > actual:
            actual = nd

        adj_l.setdefault(u, []).append(eid)
        adj_r.setdefault(v, []).append(eid)

    if actual < bound:
        bound = actual

    if bound == 1:
        color = base_color
        for eid in edges:
            answer[eid] = color
        return

    if bound & 1:
        matching = extract_matching(
            edges, bound, deg_l, deg_r, adj_l, adj_r
        )
        matching_color = base_color + bound - 1
        remaining = []

        for eid in edges:
            if eid in matching:
                answer[eid] = matching_color
            else:
                remaining.append(eid)

        color_edges(remaining, base_color, bound - 1)
    else:
        first, second = balanced_split(edges)
        half = bound // 2
        color_edges(first, base_color, half)
        color_edges(second, base_color + half, half)


color_edges(list(range(m)), 1, delta)

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answer)))
sys.stdout.write("\n")
