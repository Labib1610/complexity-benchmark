import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    
    try:
        nA = int(next(iterator))
        nB = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj_A = [[] for _ in range(nA + 1)]
    adj_B = [[] for _ in range(nB + 1)]
    edges = []
    
    for i in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        edges.append((u, v))
        adj_A[u].append(i)
        adj_B[v].append(i)

    # Calculate K: maximum degree of any node
    K = 0
    for i in range(1, nA + 1):
        if len(adj_A[i]) > K:
            K = len(adj_A[i])
    for i in range(1, nB + 1):
        if len(adj_B[i]) > K:
            K = len(adj_B[i])

    if K == 0:
        print("YES")
        print("")
        return

    # We use a greedy edge coloring approach.
    # Maintain for each node the set of available colors.
    # Since K is small relative to the number of edges in dense cases,
    # but K can be up to 400000, we use a "first available" strategy
    # with a pointer to the next available color to avoid scanning from 1.
    
    # next_color_A[u] is the smallest color >= 1 that might be free for actuator u.
    next_color_A = [1] * (nA + 1)
    next_color_B = [1] * (nB + 1)
    
    # used_A[u] stores colors currently used by edges incident to actuator u.
    # We use a set for O(1) lookups.
    used_A = [set() for _ in range(nA + 1)]
    used_B = [set() for _ in range(nB + 1)]
    
    # Result array
    ans = [0] * m
    
    # To handle the augmenting path search efficiently, we use a BFS.
    # Since we only need to find one augmenting path and update,
    # and the graph is bipartite, we can use a standard alternating path search.
    
    # For reconstruction of the path:
    # We need to know for each node in the path, which edge led to it.
    # Since edges are indexed, we can store the edge index.
    
    for i in range(m):
        u, v = edges[i]
        
        # Find the first available color for u and v
        # We advance the pointers to skip used colors
        while next_color_A[u] in used_A[u]:
            next_color_A[u] += 1
        while next_color_B[v] in used_B[v]:
            next_color_B[v] += 1
            
        c1 = next_color_A[u]
        c2 = next_color_B[v]
        
        if c1 == c2:
            # Color c1 is free on both u and v, assign it
            ans[i] = c1
            used_A[u].add(c1)
            used_B[v].add(c1)
        else:
            # We need to resolve the conflict.
            # We want to assign color c1 to edge i.
            # c1 is free on u but used on v.
            # We look for an alternating path starting from v:
            # (v, c1) -> (u', c2) -> (v', c1) -> ...
            # where we alternate between colors c1 and c2.
            # The path ends when we find a node where the other color is free.
            
            # We use BFS to find the shortest augmenting path.
            # The path is a sequence of edges.
            # We store the path as a list of edge indices.
            
            path = []
            # We need to avoid revisiting nodes.
            # Since we are only considering edges colored c1 or c2,
            # we only need to mark nodes as visited.
            visited_A = [False] * (nA + 1)
            visited_B = [False] * (nB + 1)
            
            q = deque()
            # Start BFS from v (sensor side), looking for color c1
            # The first step is to find an edge colored c1 incident to v.
            # We iterate through edges incident to v.
            for e_idx in adj_B[v]:
                if ans[e_idx] == c1:
                    u_prime = edges[e_idx][0]
                    if not visited_A[u_prime]:
                        visited_A[u_prime] = True
                        visited_B[v] = True
                        q.append((u_prime, e_idx, path[:])) # (node, edge leading to it, path so far)
            
            found = False
            target_u = -1
            target_path = []
            
            while q and not found:
                curr_u, curr_e_idx, curr_path = q.popleft()
                curr_path.append(curr_e_idx)
                
                # Check if we can stop: is c2 free on curr_u?
                if c2 not in used_A[curr_u]:
                    found = True
                    target_u = curr_u
                    target_path = curr_path
                    break
                
                # Otherwise, continue the alternating path.
                # From curr_u, we must follow an edge colored c2.
                for next_e_idx in adj_A[curr_u]:
                    if ans[next_e_idx] == c2:
                        next_v = edges[next_e_idx][1]
                        if not visited_B[next_v]:
                            visited_B[next_v] = True
                            # From next_v, we must follow an edge colored c1.
                            for next_e2_idx in adj_B[next_v]:
                                if ans[next_e2_idx] == c1:
                                    next_u2 = edges[next_e2_idx][0]
                                    if not visited_A[next_u2]:
                                        visited_A[next_u2] = True
                                        q.append((next_u2, next_e2_idx, curr_path[:]))
            
            if found:
                # We found an augmenting path.
                # The path is: (v, c1) -> (target_u, c2)
                # We swap colors along the path.
                # The edges in the path are stored in target_path.

                        ans[e_idx] = c2
                    else:
                        ans[e_idx] = c1

                used_B[v].add(c1)
                
                # Update used sets for the swapped edges
                # The path starts at v with c1, so v now has c1 used by the new edge.
                # The first edge in path was c1, now it's c2.
                # So v loses c1 (but we just added it back with the new edge),
                # and gains c2.
                # Actually, the logic is simpler:
                # We just need to update the used sets for the endpoints of the swapped edges.
                # But since we don't track which nodes are affected, we can just
                # clear and rebuild the used sets for the affected nodes.
                # However, that would be O(degree) per operation, which is too slow.
                
                # Instead, we can update the used sets incrementally.
                # The path is a sequence of edges.
                # For each edge in the path, we swap its color.
                # So for each edge (u', v') in the path:
                # - If it was c1, it becomes c2. So u' and v' lose c1 and gain c2.
                # - If it was c2, it becomes c1. So u' and v' lose c2 and gain c1.
                
                for e_idx in target_path:
                    u_p, v_p = edges[e_idx]
                    old_c = c2 if ans[e_idx] == c1 else c1 # This is wrong logic, let's re-derive.
                    # Before swap: ans[e_idx] was c1 or c2.
                    # After swap: it's the other one.
                    # So:
                    if ans[e_idx] == c1: # It was c2 before
                        used_A[u_p].discard(c2)
                        used_B[v_p].discard(c2)
                        used_A[u_p].add(c1)
                        used_B[v_p].add(c1)
                    else: # It was c1 before
                        used_A[u_p].discard(c1)
                        used_B[v_p].discard(c1)
                        used_A[u_p].add(c2)
                        used_B[v_p].add(c2)
            else:
                # No augmenting path found for c1.
                # This means c1 is not achievable.
                # We try to assign c2 instead.
                # c2 is free on v but used on u.
                # We look for an alternating path starting from u:
                # (u, c2) -> (v', c1) -> ...
                
                path = []
                visited_A = [False] * (nA + 1)
                visited_B = [False] * (nB + 1)
                q = deque()
                
                for e_idx in adj_A[u]:
                    if ans[e_idx] == c2:
                        v_prime = edges[e_idx][1]
                        if not visited_B[v_prime]:
                            visited_B[v_prime] = True
                            visited_A[u] = True
                            q.append((v_prime, e_idx, path[:]))
                
                found = False
                target_v = -1
                target_path = []
                
                while q and not found:
                    curr_v, curr_e_idx, curr_path = q.popleft()
                    curr_path.append(curr_e_idx)
                    
                    if c1 not in used_B[curr_v]:
                        found = True
                        target_v = curr_v
                        target_path = curr_path
                        break
                    
                    for next_e_idx in adj_B[curr_v]:
                        if ans[next_e_idx] == c1:
                            next_u = edges[next_e_idx][0]
                            if not visited_A[next_u]:
                                visited_A[next_u] = True
                                for next_e2_idx in adj_A[next_u]:
                                    if ans[next_e2_idx] == c2:
                                        next_v2 = edges[next_e2_idx][1]
                                        if not visited_B[next_v2]:
                                            visited_B[next_v2] = True
                                            q.append((next_v2, next_e2_idx, curr_path[:]))
                
                if found:
                    for e_idx in target_path:
                        if ans[e_idx] == c1:
                            ans[e_idx] = c2
                        else:
                            ans[e_idx] = c1
                    
                    ans[i] = c2
                    used_A[u].add(c2)
