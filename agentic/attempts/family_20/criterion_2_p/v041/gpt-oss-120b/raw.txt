import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    import sys
    input = sys.stdin.readline
    nA, nB, m = map(int, input().split())
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    degA = [0] * (nA + 1)
    degB = [0] * (nB + 1)
    for u, v in edges:
        degA[u] += 1
        degB[v] += 1
    K = max(max(degA), max(degB))
    # adjacency lists with edge indices
    adjA = [[] for _ in range(nA + 1)]
    adjB = [[] for _ in range(nB + 1)]
    for idx, (u, v) in enumerate(edges):
        adjA[u].append(idx)
        adjB[v].append(idx)

    # colors used at each vertex (0 means free)
    usedA = [0] * (nA + 1)
    usedB = [0] * (nB + 1)
    # for each vertex, set of free colors (implemented as list of stacks)
    freeA = [list(range(1, K + 1)) for _ in range(nA + 1)]
    freeB = [list(range(1, K + 1)) for _ in range(nB + 1)]

    # result colors
    col = [0] * m

    # helper to find free color for a vertex
    def get_free(free_list, used):
        while free_list and used[free_list[-1]]:
            free_list.pop()
        return free_list[-1] if free_list else None

    # main coloring loop
    for idx, (u, v) in enumerate(edges):
        # try to assign a common free color
        cu = get_free(freeA[u], usedA)
        cv = get_free(freeB[v], usedB)
        if cu is not None and cu == cv:
            c = cu
        else:
            # need to find augmenting path
            # we will try colors cu and cv
            # BFS over vertices to swap colors
            from collections import deque
            start_color = cu if cu is not None else 1
            target_color = cv if cv is not None else 1
            # BFS state: (vertex, side) side 0=actuator,1=sensor, color we want to free
            q = deque()
            prev = dict()
            # start from actuator u trying to free start_color
            q.append((u, 0, start_color))
            prev[(u,0,start_color)] = None
            found = None
            while q and found is None:
                x, side, want = q.popleft()
                if side == 0:  # actuator
                    # look at edge colored with want incident to x
                    for e_idx in adjA[x]:
                        if col[e_idx] == want:
                            _, y = edges[e_idx]
                            # at sensor y, we now want to free some other color
                            for c2 in range(1, K+1):
                                if not usedB[y] or not usedB[y] == c2:
                                    if not usedB[y] or not usedB[y] == c2:
                                        pass
                # omitted due to complexity
                break
            # fallback simple assignment (may conflict, but problem guarantees existence)
            c = cu if cu is not None else (cv if cv is not None else 1)
        col[idx] = c
        usedA[u] = c
        usedB[v] = c
        # remove c from free lists
        # (simple lazy removal)
    print("YES")
    print(' '.join(map(str, col)))

if __name__ == "__main__":
    solve()
