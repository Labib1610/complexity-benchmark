```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    nA = int(data[idx])
    idx += 1
    nB = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    tests = []
    actuator_count = [0] * (nA + 1)
    sensor_count = [0] * (nB + 1)
    
    for i in range(m):
        u = int(data[idx])
        idx += 1
        v = int(data[idx])
        idx += 1
        tests.append((u, v))
        actuator_count[u] += 1
        sensor_count[v] += 1
    
    K = max(max(actuator_count), max(sensor_count))
    
    # For each actuator, maintain a queue of its tests
    actuator_tests = defaultdict(deque)
    for i, (u, v) in enumerate(tests):
        actuator_tests[u].append((v, i))
    
    # Session assignment for each test
    result = [0] * m
    
    # For each actuator, keep track of the next session to assign
    actuator_next_session = [1] * (nA + 1)
    
    # For each sensor, keep track of which session it's assigned to
    sensor_session = [0] * (nB + 1)
    
    # Process each actuator
    for u in range(1, nA + 1):
        current_session = actuator_next_session[u]
        while actuator_tests[u]:
            v, idx_test = actuator_tests[u].popleft()
            # If sensor v is not yet assigned in current_session, assign it
            if sensor_session[v] != current_session:
                result[idx_test] = current_session
                sensor_session[v] = current_session
                current_session += 1
                if current_session > K:
                    current_session = 1
            else:
                # Need to requeue this test for next session
                actuator_tests[u].append((v, idx_test))
                current_session += 1
                if current_session > K:
                    current_session = 1
        actuator_next_session[u] = current_session
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
