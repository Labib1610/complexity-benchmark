```python
import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    nA = int(input_data[0])
    nB = int(input_data[1])
    m = int(input_data[2])
    
    if m == 0:
        print("YES")
        print()
        return

    degA = [0] * (nA + 1)
    degB = [0] * (nB + 1)
    
    edges_u = [0] * m
    edges_v = [0] * m
    
    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        edges_u[i] = u
        edges_v[i] = v
        degA[u] += 1
        degB[v] += 1
        
    K = 0
    for u in range(1, nA + 1):
        if degA[u] > K:
            K = degA[u]
    for v in range(1, nB + 1):
        if degB[v] > K:
            K = degB[v]
            
    # We will use the standard bipartite edge coloring algorithm with alternating paths.
    # To be extremely fast, we maintain:
    # color_of_edge[edge_id]
    # matchA[u][c] = edge_id
    # matchB[v][c] = edge_id
    # Since K <= m, we can use arrays/dicts. But allocating K-sized lists for all vertices is O(N * K).
    # Since max degree is K, each vertex only has at most K incident edges.
    # We can use a flat array or list of dicts. 
    # But list of dicts might have overhead. Since total degree sum is 2m, 
    # actually, dict lookup/insert in Python is very fast.
    
    matchA = [{} for _ in range(nA + 1)]
    matchB = [{} for _ in range(nB + 1)]
    ans = [0] * m
    
    # We also need a fast way to find a free color for u and v.
    # Since degree(u) < K and degree(v) < K at the moment of insertion,
    # free colors exist in range 1..K.
    
    for e in range(m):
        u = edges_u[e]
        v = edges_v[e]
        
        # Find a free color for u
        mA = matchA[u]
        mB = matchB[v]
        
        c1 = 1
        while c1 in mA:
            c1 += 1
            
        c2 = 1
        while c2 in mB:
            c2 += 1
            
        if c1 == c2:
            mA[c1] = e
            mB[c1] = e
            ans[e] = c1
        else:
            # We will start an alternating path from v with color c1 (which is free at u, but used at v if c1 in mB)
            # Actually, standard method:
            # Color e with c1.
            # If c1 is free at v, done.
            # Otherwise, let edge e' at v have color c1. We flip the alternating c1-c2 path starting at v.
            
            # Trace and flip alternating path:
            cur = v
            curr_col = c1
            next_col = c2
            is_A = False # cur is in B
            
            # To flip efficiently:
            # We can follow the path and update the matching along the way.
            # We need to recolor edges.
            # Since it's a path starting at v with color c1,
            # cur = v (in B) has an edge of color c1.
            # We will swap color c1 and c2 along this path.
            
            # Let's collect the path vertices and edges or update in-place
            # In-place update:
            while True:
                if not is_A:
                    # cur is in B
                    if curr_col not in matchB[cur]:
                        break
                    edge_id = matchB[cur][curr_col]
                    # This edge goes to some vertex in A
                    nxt_vert = edges_u[edge_id]
                    
                    # Remove edge_id with curr_col from matchB[cur] and matchA[nxt_vert]
                    del matchB[cur][curr_col]
                    del matchA[nxt_vert][curr_col]
                    
                    # Add edge_id with next_col to matchB[cur] and matchA[nxt_vert]
                    matchB[cur][next_col] = edge_id
                    matchA[nxt_vert][next_col] = edge_id
                    ans[edge_id] = next_col
                    
                    cur = nxt_vert
                    is_A = True
                    curr_col, next_col = next_col, curr_col
                else:
                    # cur is in A
                    if curr_col not in matchA[cur]:
                        break
                    edge_id = matchA[cur][curr_col]
                    nxt_vert = edges_v[edge_id]
                    
                    del matchA[cur][curr_col]
                    del matchB[nxt_vert][curr_col]
                    
                    matchA[cur][next_col] = edge_id
                    matchB[nxt_vert][next_col] = edge_id
                    ans[edge_id] = next_col
                    
                    cur = nxt_vert
                    is_A = False
                    curr_col, next_col = next_col, curr_col
                    
            # Now color c1 is definitely free at v (because it was reassigned to c2, or was already free),
            # and color c1 is free at u (it was originally free, and the path started at v with c1,
            # so the path could not reach u with color c1 because u had no c1 edge!).
            mA[c1] = e
            mB[c1] = e
            ans[e] = c1

    print("YES")
    print(*(ans[i] for i in range(m)))

if __name__ == '__main__':
    solve()
```
