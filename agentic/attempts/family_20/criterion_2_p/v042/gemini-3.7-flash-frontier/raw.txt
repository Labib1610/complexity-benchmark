import sys

# Increase recursion depth just in case
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    nL = int(input_data[0])
    nR = int(input_data[1])
    m = int(input_data[2])

    if m == 0:
        sys.stdout.write("YES\n\n")
        return

    u_list = [0] * m
    v_list = [0] * m
    deg_L = [0] * (nL + 1)
    deg_R = [0] * (nR + 1)

    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        u_list[i] = u
        v_list[i] = v + nL  # Shift receiver indices
        deg_L[u] += 1
        deg_R[v] += 1

    max_deg = 0
    for d in deg_L:
        if d > max_deg:
            max_deg = d
    for d in deg_R:
        if d > max_deg:
            max_deg = d

    ans = [0] * m

    # Iterative stack for divide and conquer to avoid recursion overhead
    # Stack items: (edge_indices, max_deg, color_offset)
    initial_edges = list(range(m))
    stack = [(initial_edges, max_deg, 0)]

    total_nodes = nL + nR + 1
    used = [False] * m
    deg = {}
    adj = {}
    ptr = {}

    while stack:
        edges, d, color_offset = stack.pop()
        if not edges or d == 0:
            continue
        if d == 1:
            c = color_offset + 1
            for e in edges:
                ans[e] = c
            continue

        d1 = (d + 1) // 2
        d2 = d // 2

        # Collect active vertices
        active_nodes = []
        for e in edges:
            u = u_list[e]
            v = v_list[e]
            if u not in adj:
                adj[u] = []
                deg[u] = 0
                ptr[u] = 0
                active_nodes.append(u)
            if v not in adj:
                adj[v] = []
                deg[v] = 0
                ptr[v] = 0
                active_nodes.append(v)
            adj[u].append(e)
            adj[v].append(e)
            deg[u] += 1
            deg[v] += 1
            used[e] = False

        odd_nodes = [node for node in active_nodes if deg[node] % 2 == 1]

        E1 = []
        E2 = []

        # Process walks starting from odd degree vertices
        for start_node in odd_nodes:
            while ptr[start_node] < len(adj[start_node]):
                e_idx = adj[start_node][ptr[start_node]]
                ptr[start_node] += 1
                if used[e_idx]:
                    continue
                used[e_idx] = True
                curr = v_list[e_idx] if u_list[e_idx] == start_node else u_list[e_idx]
                walk = [e_idx]
                while True:
                    found = False
                    adj_curr = adj[curr]
                    while ptr[curr] < len(adj_curr):
                        nxt_e = adj_curr[ptr[curr]]
                        ptr[curr] += 1
                        if not used[nxt_e]:
                            used[nxt_e] = True
                            walk.append(nxt_e)
                            curr = v_list[nxt_e] if u_list[nxt_e] == curr else u_list[nxt_e]
                            found = True
                            break
                    if not found:
                        break

                for i, eid in enumerate(walk):
                    if i % 2 == 0:
                        E1.append(eid)
                    else:
                        E2.append(eid)

        # Process Eulerian cycles for remaining edges
        for start_node in active_nodes:
            while ptr[start_node] < len(adj[start_node]):
                e_idx = adj[start_node][ptr[start_node]]
                ptr[start_node] += 1
                if used[e_idx]:
                    continue
                used[e_idx] = True
                curr = v_list[e_idx] if u_list[e_idx] == start_node else u_list[e_idx]
                walk = [e_idx]
                while True:
                    found = False
                    adj_curr = adj[curr]
                    while ptr[curr] < len(adj_curr):
                        nxt_e = adj_curr[ptr[curr]]
                        ptr[curr] += 1
                        if not used[nxt_e]:
                            used[nxt_e] = True
                            walk.append(nxt_e)
                            curr = v_list[nxt_e] if u_list[nxt_e] == curr else u_list[nxt_e]
                            found = True
                            break
                    if not found:
                        break

                for i, eid in enumerate(walk):
                    if i % 2 == 0:
                        E1.append(eid)
                    else:
                        E2.append(eid)

        # Clear maps for active nodes
        for node in active_nodes:
            del adj[node]
            del deg[node]
            del ptr[node]

        # Push subproblems
        stack.append((E2, d2, color_offset + d1))
        stack.append((E1, d1, color_offset))

    out = ["YES\n", " ".join(map(str, ans)), "\n"]
    sys.stdout.write("".join(out))

if __name__ == '__main__':
    solve()
