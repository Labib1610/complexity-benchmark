import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    # degree calculation
    degL = [0] * nL
    degR = [0] * nR
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = max(max(degL) if degL else 0, max(degR) if degR else 0)

    # adjacency structures for colors
    # for each vertex we keep: used colors set, and mapping color -> edge id
    usedL = [set() for _ in range(nL)]
    usedR = [set() for _ in range(nR)]
    col_edgeL = [{} for _ in range(nL)]   # color -> edge id
    col_edgeR = [{} for _ in range(nR)]

    ans = [0] * m

    # helper to assign a color to an edge
    def assign(eid: int, color: int):
        u, v = edges[eid]
        ans[eid] = color
        usedL[u].add(color)
        usedR[v].add(color)
        col_edgeL[u][color] = eid
        col_edgeR[v][color] = eid

    # helper to remove a color from an edge (used during recoloring)
    def unassign(eid: int):
        u, v = edges[eid]
        c = ans[eid]
        usedL[u].remove(c)
        usedR[v].remove(c)
        del col_edgeL[u][c]
        del col_edgeR[v][c]
        ans[eid] = 0

    # recursive recoloring attempt
    def dfs(u: int, target_color: int, seenL: set, seenR: set) -> bool:
        """try to make color target_color free at left vertex u"""
        if target_color not in usedL[u]:
            return True
        if u in seenL:
            return False
        seenL.add(u)
        eid = col_edgeL[u][target_color]
        _, v = edges[eid]

        # try to find a new color for this edge at the right side
        for new_c in range(1, Delta + 1):
            if new_c not in usedR[v]:
                # reassign directly
                unassign(eid)
                assign(eid, new_c)
                return True

        # otherwise try to move the conflicting edge further
        for new_c in range(1, Delta + 1):
            if new_c in usedR[v]:
                eid2 = col_edgeR[v][new_c]
                if eid2 == eid:
                    continue
                u2, _ = edges[eid2]
                if u2 in seenL:
                    continue
                if dfs(u2, new_c, seenL, seenR):
                    # now color new_c is free at v
                    unassign(eid)
                    assign(eid, new_c)
                    return True
        return False

    for eid, (u, v) in enumerate(edges):
        # try to give the smallest possible color
        for c in range(1, Delta + 1):
            if c not in usedL[u] and c not in usedR[v]:
                assign(eid, c)
                break
        else:
            # need recoloring
            for c in range(1, Delta + 1):
                if c not in usedL[u]:
                    # try to free c at v via augmenting path
                    if dfs(v, c, set(), set()):
                        assign(eid, c)
                        break
            else:
                # should never happen
                print("NO")
                return

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
