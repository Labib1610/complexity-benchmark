We can model this as a bipartite graph where transmitters and receivers are the two sides, and each test is an edge.  
The maximum degree Δ is the minimum number of channels needed.  

We can assign channels using a simple greedy edge-coloring algorithm for bipartite graphs, which always uses exactly Δ colors.  

Here’s the implementation:  

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    nL, nR, m = map(int, input().split())
    edges = []
    adj = [[] for _ in range(nL + nR + 1)]
    for i in range(m):
        u, v = map(int, input().split())
        v += nL
        edges.append((u, v))
        adj[u].append((v, i))
        adj[v].append((u, i))
    
    # Find max degree
    deg = [0] * (nL + nR + 1)
    for u, v in edges:
        deg[u] += 1
        deg[v] += 1
    Delta = max(deg)
    
    color = [0] * m
    # Greedy edge coloring for bipartite graph
    # We'll use a simple approach: for each edge, pick smallest available color
    # for both endpoints.
    used_color_u = [0] * (nL + nR + 1)  # bitmask of used colors per vertex
    # Since Delta <= 100000, we can use sets or arrays.
    used_u = [set() for _ in range(nL + nR + 1)]
    
    for idx, (u, v) in enumerate(edges):
        # find smallest color not used by u or v
        c = 1
        while c in used_u[u] or c in used_u[v]:
            c += 1
        color[idx] = c
        used_u[u].add(c)
        used_u[v].add(c)
    
    print("YES")
    print(" ".join(map(str, color)))

if __name__ == "__main__":
    solve()
```
