import sys
from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n_in = int(next(it))
    n_out = int(next(it))
    m = int(next(it))
    
    edges = []
    adj_in = [[] for _ in range(n_in + 1)]
    adj_out = [[] for _ in range(n_out + 1)]
    
    for i in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        adj_in[u].append(i)
        adj_out[v].append(i)
    
    delta = 0
    for i in range(1, n_in + 1):
        if len(adj_in[i]) > delta:
            delta = len(adj_in[i])
    for i in range(1, n_out + 1):
        if len(adj_out[i]) > delta:
            delta = len(adj_out[i])
    
    if delta == 0:
        sys.stdout.write("YES\n")
        if m > 0:
            sys.stdout.write(" ".join(["1"] * m) + "\n")
        else:
            sys.stdout.write("\n")
        return
    
    N = n_in + n_out + 2
    S = n_in + n_out
    T = S + 1
    head = [-1] * N
    to = []
    cap = []
    nxt = []
    
    def add_edge(u, v, c):
        to.append(v)
        cap.append(c)
        nxt.append(head[u])
        head[u] = len(to) - 1
        to.append(u)
        cap.append(0)
        nxt.append(head[v])
        head[v] = len(to) - 1
    
    for u in range(1, n_in + 1):
        add_edge(S, u - 1, delta - len(adj_in[u]))
    for v in range(1, n_out + 1):
        add_edge(n_in + v - 1, T, delta - len(adj_out[v]))
    for i in range(m):
        u, v = edges[i]
        add_edge(u - 1, n_in + v - 1, 1)
    
    level = [0] * N
    q = deque()
    
    def bfs():
        for i in range(N):
            level[i] = -1
        level[S] = 0
        q.append(S)
        while q:
            u = q.popleft()
            e = head[u]
            while e != -1:
                v = to[e]
                if cap[e] > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    q.append(v)
                e = nxt[e]
        return level[T] >= 0
    
    def dfs(u, f):
        if u == T:
            return f
        e = head[u]
        while e != -1:
            v = to[e]
            if cap[e] > 0 and level[v] == level[u] + 1:
                ret = dfs(v, min(f, cap[e]))
                if ret > 0:
                    cap[e] -= ret
                    cap[e ^ 1] += ret
                    return ret
            e = nxt[e]
        level[u] = -1
        return 0
    
    while bfs():
        while True:
            f = dfs(S, delta * m + 1)
            if f == 0:
                break
    
    color = [0] * m
    for u in range(1, n_in + 1):
        e = head[u - 1]
        while e != -1:
            v = to[e]
            if v != S and cap[e] == 0:
                edge_idx = (e - 2 * m) // 2
                color[edge_idx] = 1
            e = nxt[e]
    
    adj = [[] for _ in range(delta + 1)]
    deg = [0] * (delta + 1)
    for i in range(m):
        c = color[i]
        adj[c].append(i)
        deg[c] += 1
    
    for c in range(1, delta + 1):
        adj[c].extend(adj[0])
    
    adj[0] = []
    
    for c in range(1, delta + 1):
        for i in adj[c]:
            u, v = edges[i]
            if color[i] == 0:
                adj_in[u].append(i)
                adj_out[v].append(i)
    
    for u in range(1, n_in + 1):
        for i in adj_in[u]:
            v = edges[i][1]
            adj_out[v].append(i)
    
    for v in range(1, n_out + 1):
        for i in adj_out[v]:
            u = edges[i][0]
            adj_in[u].append(i)
    
    for u in range(1, n_in + 1):
        for i in adj_in[u]:
            v = edges[i][1]
            adj_out[v].append(i)
    
    for v in range(1, n_out + 1):
        for i in adj_out[v]:
            u = edges[i][0]
            adj_in[u].append(i)
    
    for u in range(1, n_in + 1):
        for i in adj_in[u]:

            adj_out[v].append(i)
    
    for v in range(1, n_out + 1):
        for i in adj_out[v]:

            adj_in[u].append(i)
