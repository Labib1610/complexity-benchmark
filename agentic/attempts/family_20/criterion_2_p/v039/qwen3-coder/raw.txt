```python
import sys
from collections import defaultdict
import heapq

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n_U = int(data[idx])
    idx += 1
    n_D = int(data[idx])
    idx += 1
    m = int(data[idx])
    idx += 1
    
    edges = []
    adj_U = defaultdict(list)  # adj_U[u] = list of (v, edge_index)
    adj_D = defaultdict(list)  # adj_D[v] = list of (u, edge_index)
    
    for i in range(m):
        u = int(data[idx]) - 1  # Convert to 0-based indexing
        idx += 1
        v = int(data[idx]) - 1
        idx += 1
        edges.append((u, v))
        adj_U[u].append((v, i))
        adj_D[v].append((u, i))
    
    # Determine the maximum degree (K)
    K = 0
    for u in adj_U:
        K = max(K, len(adj_U[u]))
    for v in adj_D:
        K = max(K, len(adj_D[v]))
    
    if K == 0:
        print("YES")
        return
    
    # For each vertex (upstream and downstream), we need to assign colors (windows) 1..K
    # This is edge coloring where each edge gets a color such that no two edges incident to 
    # the same vertex have the same color.
    # Since max degree is K, by König's theorem, we can edge-color this bipartite graph with K colors.
    
    # We'll use a greedy approach with priority queues to balance the assignment
    result = [0] * m  # result[i] = window for edge i
    
    # For each node, maintain a min-heap of available colors
    # But actually, it's easier to just assign greedily using a simple strategy:
    # For each node, keep track of which colors are used by its edges
    
    # Let's do a greedy coloring:
    # For each edge, pick the smallest color not used by either endpoint yet
    
    color_used_U = [set() for _ in range(n_U)]
    color_used_D = [set() for _ in range(n_D)]
    
    for i in range(m):
        u, v = edges[i]
        # Find the smallest color >= 1 that is not used by u or v
        color = 1
        while color in color_used_U[u] or color in color_used_D[v]:
            color += 1
        result[i] = color
        color_used_U[u].add(color)
        color_used_D[v].add(color)
    
    print("YES")
    print(' '.join(map(str, result)))

solve()
```
