import sys
sys.setrecursionlimit(1 << 25)

def main():
    import sys
    input = sys.stdin.readline
    nL, nR, m = map(int, input().split())
    edges = [tuple(map(int, input().split())) for _ in range(m)]
    degL = [0] * (nL + 1)
    degR = [0] * (nR + 1)
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = max(max(degL), max(degR))
    # adjacency lists of edge ids
    adjL = [[] for _ in range(nL + 1)]
    adjR = [[] for _ in range(nR + 1)]
    for idx, (u, v) in enumerate(edges):
        adjL[u].append(idx)
        adjR[v].append(idx)

    # colors used at each vertex (set implemented as list of bool)
    usedL = [set() for _ in range(nL + 1)]
    usedR = [set() for _ in range(nR + 1)]
    color = [0] * m

    # helper to find free color at vertex
    def free_color(used, Delta):
        for c in range(1, Delta + 1):
            if c not in used:
                return c
        return None

    # BFS for augmenting recolor path
    from collections import deque
    for idx, (u, v) in enumerate(edges):
        if color[idx]:
            continue
        # try direct free color
        cu = free_color(usedL[u], Delta)
        cv = free_color(usedR[v], Delta)
        if cu == cv:
            col = cu
            color[idx] = col
            usedL[u].add(col)
            usedR[v].add(col)
            continue
        # need to recolor via alternating path
        # we will try to make cu available at v (or cv at u)
        # start BFS from u trying to free color cv
        start = u
        target = v
        want = cv  # color we want to free at start side
        other = cu  # the other color
        # BFS over vertices, storing predecessor edge and vertex
        pred = dict()  # vertex -> (prev_vertex, edge_id)
        q = deque()
        q.append(start)
        pred[start] = None
        found = None
        while q:
            x = q.popleft()
            if x <= nL:
                # left side, look at edge colored want
                for e_id in adjL[x]:
                    if color[e_id] != want:
                        continue
                    _, y = edges[e_id]
                    if y not in pred:
                        pred[y] = (x, e_id)
                        q.append(y)
            else:
                # right side, look at edge colored other
                for e_id in adjR[x - nL]:
                    if color[e_id] != other:
                        continue
                    y, _ = edges[e_id]
                    if y not in pred:
                        pred[y] = (x, e_id)
                        q.append(y)
            # check if we reached a vertex where want is free
            if x <= nL:
                if want not in usedL[x]:
                    found = x
                    break
            else:
                if want not in usedR[x - nL]:
                    found = x
                    break
        if found is None:
            # fallback simple assign (should not happen)
            col = cu
            color[idx] = col
            usedL[u].add(col)
            usedR[v].add(col)
            continue
        # flip colors along the path
        cur = found
        while pred[cur] is not None:
            prev, e_id = pred[cur]
            # swap colors on edge e_id
            if color[e_id] == want:
                color[e_id] = other
                # update used sets
                if prev <= nL:
                    usedL[prev].remove(want)
                    usedL[prev].add(other)
                    usedR[cur - nL].remove(want)
                    usedR[cur - nL].add(other)
                else:
                    usedR[prev - nL].remove(want)
                    usedR[prev - nL].add(other)
                    usedL[cur].remove(want)
                    usedL[cur].add(other)
            else:
                color[e_id] = want
                if prev <= nL:
                    usedL[prev].remove(other)
                    usedL[prev].add(want)
                    usedR[cur - nL].remove(other)
                    usedR[cur - nL].add(want)
                else:
                    usedR[prev - nL].remove(other)
                    usedR[prev - nL].add(want)
                    usedL[cur].remove(other)
                    usedL[cur].add(want)
            cur = prev
            want, other = other, want  # swap roles for next step
        # now assign color to original edge
        col = cu
        color[idx] = col
        usedL[u].add(col)
        usedR[v].add(col)

    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    main()
