import sys
from array import array
from collections import deque

sys.setrecursionlimit(1_000_000)


def read_ints():
    data = sys.stdin.buffer.read()
    n = len(data)
    i = 0
    while i < n:
        while i < n and data[i] <= 32:
            i += 1
        if i >= n:
            break
        x = 0
        while i < n and data[i] > 32:
            x = x * 10 + data[i] - 48
            i += 1
        yield x


it = iter(read_ints())
nL = next(it)
nR = next(it)
m = next(it)

left = array('i')
right = array('i')
degL = [0] * nL
degR = [0] * nR

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    left.append(u)
    right.append(v)
    degL[u] += 1
    degR[v] += 1

if m == 0:
    sys.stdout.write("YES\n\n")
    raise SystemExit

Delta = max(max(degL, default=0), max(degR, default=0))

if Delta == 1:
    sys.stdout.write("YES\n" + " ".join(["1"] * m) + "\n")
    raise SystemExit


def pack_vertices(deg, capacity):
    size = len(deg)
    group = [-1] * size

    if capacity == 2:
        loads = []
        singles = []
        for v, d in enumerate(deg):
            if d == 2:
                group[v] = len(loads)
                loads.append(2)
            elif d == 1:
                singles.append(v)
        i = 0
        while i < len(singles):
            g = len(loads)
            group[singles[i]] = g
            if i + 1 < len(singles):
                group[singles[i + 1]] = g
                loads.append(2)
            else:
                loads.append(1)
            i += 2
        return group, loads

    active = [i for i, d in enumerate(deg) if d]
    active.sort(key=deg.__getitem__, reverse=True)

    base = 1
    while base < len(active):
        base <<= 1
    tree = [0] * (base << 1)
    loads = []

    for v in active:
        d = deg[v]
        if tree[1] < d:
            b = len(loads)
            loads.append(d)
            group[v] = b
            p = base + b
            tree[p] = capacity - d
        else:
            p = 1
            while p < base:
                q = p << 1
                if tree[q] >= d:
                    p = q
                else:
                    p = q | 1
            b = p - base
            group[v] = b
            loads[b] += d
            tree[p] -= d

        p >>= 1
        while p:
            a = tree[p << 1]
            bval = tree[p << 1 | 1]
            tree[p] = a if a >= bval else bval
            p >>= 1

    return group, loads


groupL, loadL = pack_vertices(degL, Delta)
groupR, loadR = pack_vertices(degR, Delta)

for i in range(m):
    left[i] = groupL[left[i]]
    right[i] = groupR[right[i]]

N = max(len(loadL), len(loadR))
if len(loadL) < N:
    loadL.extend([0] * (N - len(loadL)))
if len(loadR) < N:
    loadR.extend([0] * (N - len(loadR)))

del groupL, groupR, degL, degR

i = j = 0
needL = Delta - loadL[0] if N else 0
needR = Delta - loadR[0] if N else 0

while i < N and j < N:
    if needL == 0:
        i += 1
        if i < N:
            needL = Delta - loadL[i]
        continue
    if needR == 0:
        j += 1
        if j < N:
            needR = Delta - loadR[j]
        continue

    take = needL if needL < needR else needR
    left.extend([i] * take)
    right.extend([j] * take)
    needL -= take
    needR -= take

del loadL, loadR

total_edges = len(left)
answer = [0] * m
mark = bytearray(total_edges)


def split_even(edges):
    ecount = len(edges)
    mate_l = array('i', [-1]) * ecount
    mate_r = array('i', [-1]) * ecount
    pending_l = [-1] * N
    pending_r = [-1] * N

    for idx, eid in enumerate(edges):
        u = left[eid]
        old = pending_l[u]
        if old == -1:
            pending_l[u] = idx
        else:
            mate_l[idx] = old
            mate_l[old] = idx
            pending_l[u] = -1

        v = right[eid]
        old = pending_r[v]
        if old == -1:
            pending_r[v] = idx
        else:
            mate_r[idx] = old
            mate_r[old] = idx
            pending_r[v] = -1

    seen = bytearray(ecount)
    first = []
    second = []

    for start in range(ecount):
        if seen[start]:
            continue
        cur = start
        side = 0
        color = 0
        while not seen[cur]:
            seen[cur] = 1
            if color == 0:
                first.append(edges[cur])
            else:
                second.append(edges[cur])
            if side == 0:
                cur = mate_l[cur]
            else:
                cur = mate_r[cur]
            side ^= 1
            color ^= 1

    return first, second


def perfect_matching(edges, degree):
    ecount = len(edges)
    slots = array('i', [0]) * ecount
    pos = [0] * N

    for eid in edges:
        u = left[eid]
        idx = u * degree + pos[u]
        slots[idx] = eid
        pos[u] += 1

    pair_u = [-1] * N
    pair_v = [-1] * N
    pair_edge = [-1] * N

    for u in range(N):
        base = u * degree
        end = base + degree
        for k in range(base, end):
            eid = slots[k]
            v = right[eid]
            if pair_v[v] == -1:
                pair_u[u] = v
                pair_v[v] = u
                pair_edge[u] = eid
                break

    dist = [-1] * N
    ptr = [0] * N

    def bfs():
        q = deque()
        found = False
        for u in range(N):
            if pair_u[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = -1

        while q:
            u = q.popleft()
            nd = dist[u] + 1
            base = u * degree
            end = base + degree
            for k in range(base, end):
                v = right[slots[k]]
                pu = pair_v[v]
                if pu == -1:
                    found = True
                elif dist[pu] == -1:
                    dist[pu] = nd
                    q.append(pu)
        return found

    def dfs(u):
        base = u * degree
        end = base + degree
        k = ptr[u]
        if k < base:
            k = base

        while k < end:
            ptr[u] = k + 1
            eid = slots[k]
            v = right[eid]
            pu = pair_v[v]
            if pu == -1 or (dist[pu] == dist[u] + 1 and dfs(pu)):
                pair_u[u] = v
                pair_v[v] = u
                pair_edge[u] = eid
                return True
            k += 1

        dist[u] = -1
        return False

    while True:
        unmatched = False
        for u in range(N):
            if pair_u[u] == -1:
                unmatched = True
                break
        if not unmatched:
            break
        if not bfs():
            break
        for u in range(N):
            ptr[u] = u * degree
        for u in range(N):
            if pair_u[u] == -1:
                dfs(u)

    return pair_edge


all_edges = list(range(total_edges))
stack = [(all_edges, Delta, 1)]

while stack:
    edges, degree, color_start = stack.pop()

    if degree == 1:
        for eid in edges:
            if eid < m:
                answer[eid] = color_start
        continue

    if degree & 1:
        matching = perfect_matching(edges, degree)
        matching_color = color_start + degree - 1

        for eid in matching:
            mark[eid] = 1
            if eid < m:
                answer[eid] = matching_color

        remainder = []
        append_rem = remainder.append
        for eid in edges:
            if not mark[eid]:
                append_rem(eid)

        for eid in matching:
            mark[eid] = 0

        stack.append((remainder, degree - 1, color_start))
    else:
        first, second = split_even(edges)
        half = degree >> 1
        stack.append((second, half, color_start + half))
        stack.append((first, half, color_start))

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answer)))
sys.stdout.write("\n")
