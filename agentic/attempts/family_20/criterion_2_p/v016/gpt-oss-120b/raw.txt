```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    S = int(next(it))
    T = int(next(it))
    m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    # compute max degree D
    deg_src = [0] * S
    deg_dst = [0] * T
    for u, v in edges:
        deg_src[u] += 1
        deg_dst[v] += 1
    D = max(max(deg_src, default=0), max(deg_dst, default=0))
    if D == 0:
        print("YES")
        print()
        return

    # for each vertex store color -> edge index
    src_color = [{} for _ in range(S)]
    dst_color = [{} for _ in range(T)]

    # next free color pointer for each vertex
    src_next = [1] * S
    dst_next = [1] * T

    ans = [0] * m

    # adjacency list of edges with their colors (to traverse alternating paths)
    edge_color = [0] * m

    # helper to get first free color at a vertex
    def first_free(col_dict, nxt):
        while nxt in col_dict:
            nxt += 1
        return nxt

    # DFS for alternating path between colors a and b starting from source vertex u
    def dfs(u, a, b, visited_src, visited_dst):
        # u is source side vertex
        visited_src.add(u)
        # edge of color a incident to u, if any
        if a in src_color[u]:
            v, eidx = src_color[u][a]
            if v in visited_dst:
                pass
            else:
                visited_dst.add(v)
                # try to continue with color b from v
                if b not in dst_color[v]:
                    # found free at destination side
                    return [(u, a, eidx, v)]
                else:
                    # follow edge of color b from v
                    u2, eidx2 = dst_color[v][b]
                    if u2 not in visited_src:
                        path = dfs(u2, a, b, visited_src, visited_dst)
                        if path is not None:
                            return [(u, a, eidx, v)] + path
        return None

    for idx, (u, v) in enumerate(edges):
        # find first free colors
        a = first_free(src_color[u], src_next[u])
        b = first_free(dst_color[v], dst_next[v])

        if a == b:
            col = a
        else:
            # try to recolor using alternating path between a and b
            path = dfs(u, a, b, set(), set())
            if path is None:
                # no path, we can just use a (it is free at source, may be used at dest)
                # but b is free at dest, so we can use b as well; choose a and recolor later
                col = a
            else:
                # flip colors along the path
                for su, ca, eidx, dv in path:
                    # edge eidx currently has color ca, we will change it to the other color
                    other = b if ca == a else a
                    edge_color[eidx] = other
                    # update mappings
                    src_color[su].pop(ca)
                    src_color[su][other] = (dv, eidx)
                    dst_color[dv].pop(ca)
                    dst_color[dv][other] = (su, eidx)
                col = a  # now a is free at both ends
        # assign color col to current edge
        ans[idx] = col
        edge_color[idx] = col
        src_color[u][col] = (v, idx)
        dst_color[v][col] = (u, idx)

        # update next pointers if needed
        if src_next[u] == col:
            src_next[u] = col + 1
        if dst_next[v] == col:
            dst_next[v] = col + 1

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
```
