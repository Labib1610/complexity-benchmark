import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    nA = int(next(it))
    nB = int(next(it))
    m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    degA = [0] * nA
    degB = [0] * nB
    for u, v in edges:
        degA[u] += 1
        degB[v] += 1
    T = max(max(degA, default=0), max(degB, default=0))

    # adjacency: for each vertex store list of (neighbor, edge_id)
    adjA = [[] for _ in range(nA)]
    adjB = [[] for _ in range(nB)]
    for idx, (u, v) in enumerate(edges):
        adjA[u].append((v, idx))
        adjB[v].append((u, idx))

    # colors used at each vertex, stored as dict color->edge_id
    usedA = [{} for _ in range(nA)]
    usedB = [{} for _ in range(nB)]
    ans = [0] * m

    from collections import deque

    def bfs_recolor(start_u, start_color, target_u):
        """Find alternating path starting from start_u (in side A) using start_color,
        ending at a vertex where target_u has a free color. Returns parent map."""
        q = deque()
        q.append(('A', start_u))
        parent = {('A', start_u): None}
        edge_used = {}
        while q:
            side, v = q.popleft()
            if side == 'A':
                # try colors at A vertex
                for col, e_id in usedA[v].items():
                    if col == start_color:
                        # follow edge to B
                        u2, _ = edges[e_id]
                        nb = ('B', u2)
                        if nb not in parent:
                            parent[nb] = (side, v, e_id)
                            q.append(nb)
                # also try free color at A
                if start_color not in usedA[v]:
                    # we have reached a free spot
                    return parent, ('A', v)
            else:  # side B
                for col, e_id in usedB[v].items():
                    if col == start_color:
                        u2, _ = edges[e_id]
                        nb = ('A', u2)
                        if nb not in parent:
                            parent[nb] = (side, v, e_id)
                            q.append(nb)
                if start_color not in usedB[v]:
                    return parent, ('B', v)
        return None, None

    # Simple greedy with limited recoloring (only one swap)
    for idx, (u, v) in enumerate(edges):
        # find smallest free color at both ends
        for c in range(1, T + 1):
            if c not in usedA[u] and c not in usedB[v]:
                ans[idx] = c
                usedA[u][c] = idx
                usedB[v][c] = idx
                break
        else:
            # need recolor: pick any color free at u
            for c in range(1, T + 1):
                if c not in usedA[u]:
                    free_c = c
                    break
            # find edge at v using free_c (must exist)
            e2 = usedB[v][free_c]
            # swap colors of e2 and current edge
            ans[idx] = free_c
            ans[e2] = 0  # will be recolored later
            usedA[u][free_c] = idx
            usedB[v][free_c] = idx
            # now recolor e2 recursively
            stack = [e2]
            while stack:
                e = stack.pop()
                uu, vv = edges[e]
                # find a free color for this edge
                for c in range(1, T + 1):
                    if c not in usedA[uu] and c not in usedB[vv]:
                        ans[e] = c
                        usedA[uu][c] = e
                        usedB[vv][c] = e
                        break
                else:
                    # pick free at uu and push conflicting edge
                    for c in range(1, T + 1):
                        if c not in usedA[uu]:
                            free_c = c
                            break
                    e2 = usedB[vv][free_c]
                    ans[e] = free_c
                    usedA[uu][free_c] = e
                    usedB[vv][free_c] = e
                    stack.append(e2)

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
