import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

nA, nB, m = data[0], data[1], data[2]
U = [0] * m
V = [0] * m

degA = [0] * nA
degB = [0] * nB

p = 3
for i in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2
    U[i] = u
    V[i] = v
    degA[u] += 1
    degB[v] += 1

T = max(max(degA, default=0), max(degB, default=0))
answer = [0] * m


def balanced_split(edge_ids):
    q = len(edge_ids)
    adj = {}

    for local_id, eid in enumerate(edge_ids):
        a = U[eid]
        b = nA + V[eid]
        adj.setdefault(a, []).append(local_id)
        adj.setdefault(b, []).append(local_id)

    odd_vertices = [x for x, edges in adj.items() if len(edges) & 1]

    ends_a = [0] * (q + len(odd_vertices))
    ends_b = [0] * (q + len(odd_vertices))

    for local_id, eid in enumerate(edge_ids):
        ends_a[local_id] = U[eid]
        ends_b[local_id] = nA + V[eid]

    dummy = -1
    if odd_vertices:
        dummy_edges = []
        for j, x in enumerate(odd_vertices):
            local_id = q + j
            ends_a[local_id] = dummy
            ends_b[local_id] = x
            dummy_edges.append(local_id)
            adj[x].append(local_id)
        adj[dummy] = dummy_edges

    total_edges = len(ends_a)
    used = bytearray(total_edges)
    left = []
    right = []

    def process_component(start):
        vertex_stack = [start]
        edge_stack = []
        circuit = []

        while vertex_stack:
            x = vertex_stack[-1]
            edges = adj[x]
            while edges and used[edges[-1]]:
                edges.pop()

            if edges:
                e = edges.pop()
                if used[e]:
                    continue
                used[e] = 1
                y = ends_b[e] if ends_a[e] == x else ends_a[e]
                vertex_stack.append(y)
                edge_stack.append(e)
            else:
                vertex_stack.pop()
                if edge_stack:
                    circuit.append(edge_stack.pop())

        parity = 0
        for e in circuit:
            if e < q:
                if parity == 0:
                    left.append(edge_ids[e])
                else:
                    right.append(edge_ids[e])
            parity ^= 1

    if odd_vertices:
        process_component(dummy)

    for x, edges in adj.items():
        while edges and used[edges[-1]]:
            edges.pop()
        if edges:
            process_component(x)

    return left, right


def extract_matching(edge_ids, D):
    adjL = {}
    adjR = {}

    for eid in edge_ids:
        u = U[eid]
        v = V[eid]
        adjL.setdefault(u, []).append(eid)
        adjR.setdefault(v, []).append(eid)

    criticalL = [u for u, edges in adjL.items() if len(edges) == D]
    criticalR = [v for v, edges in adjR.items() if len(edges) == D]
    criticalR_set = set(criticalR)

    matchL = {}
    matchR = {}

    def augment_required(roots, adjS, matchS, matchO, opposite_of,
                         source_of_matched, replaceable=None):
        while True:
            active_roots = [x for x in roots if x not in matchS]
            if not active_roots:
                return

            dist = {}
            queue = deque()
            for x in active_roots:
                dist[x] = 0
                queue.append(x)

            best = 10 ** 18

            while queue:
                x = queue.popleft()
                dx = dist[x]
                if dx + 1 > best:
                    continue

                current = matchS.get(x, -1)
                for eid in adjS.get(x, ()):
                    if eid == current:
                        continue
                    y = opposite_of(eid)
                    me = matchO.get(y, -1)

                    if me == -1:
                        if dx + 1 < best:
                            best = dx + 1
                    else:
                        x2 = source_of_matched(me)
                        if replaceable is not None and x2 not in replaceable:
                            if dx + 1 < best:
                                best = dx + 1
                        elif dx + 1 < best and x2 not in dist:
                            dist[x2] = dx + 1
                            queue.append(x2)

            if best == 10 ** 18:
                return

            def dfs(x):
                dx = dist[x]
                current = matchS.get(x, -1)

                for eid in adjS.get(x, ()):
                    if eid == current:
                        continue

                    y = opposite_of(eid)
                    me = matchO.get(y, -1)

                    if me == -1:
                        if dx + 1 == best:
                            matchS[x] = eid
                            matchO[y] = eid
                            return True
                    else:
                        x2 = source_of_matched(me)

                        if replaceable is not None and x2 not in replaceable:
                            if dx + 1 == best:
                                del matchS[x2]
                                matchS[x] = eid
                                matchO[y] = eid
                                return True
                        elif (
                            dx + 1 < best
                            and dist.get(x2, -1) == dx + 1
                            and dfs(x2)
                        ):
                            matchS[x] = eid
                            matchO[y] = eid
                            return True

                dist[x] = -1
                return False

            progress = False
            for x in active_roots:
                if x not in matchS and dist.get(x, -1) == 0 and dfs(x):
                    progress = True

            if not progress:
                return

    augment_required(
        criticalL,
        adjL,
        matchL,
        matchR,
        lambda eid: V[eid],
        lambda eid: U[eid]
    )

    augment_required(
        criticalR,
        adjR,
        matchR,
        matchL,
        lambda eid: U[eid],
        lambda eid: V[eid],
        criticalR_set
    )

    return list(matchL.values())


def color_edges(edge_ids, D, offset):
    if not edge_ids:
        return

    if D == 1:
        color = offset + 1
        for eid in edge_ids:
            answer[eid] = color
        return

    if D & 1:
        matching = extract_matching(edge_ids, D)
        matching_set = set(matching)
        color = offset + D

        for eid in matching:
            answer[eid] = color

        remaining = [eid for eid in edge_ids if eid not in matching_set]
        color_edges(remaining, D - 1, offset)
    else:
        first, second = balanced_split(edge_ids)
        half = D // 2
        color_edges(first, half, offset)
        color_edges(second, half, offset + half)


if m:
    color_edges(list(range(m)), T, 0)

out = ["YES", " ".join(map(str, answer))]
sys.stdout.write("\n".join(out))
