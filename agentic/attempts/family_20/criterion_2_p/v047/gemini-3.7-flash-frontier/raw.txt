import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    A = int(input_data[0])
    B = int(input_data[1])
    L = int(input_data[2])

    if L == 0:
        print("YES")
        print("")
        return

    edges_u = [0] * L
    edges_v = [0] * L
    deg = [0] * (A + B + 1)

    idx = 3
    for i in range(L):
        u = int(input_data[idx])
        v = int(input_data[idx + 1]) + A
        idx += 2
        edges_u[i] = u
        edges_v[i] = v
        deg[u] += 1
        deg[v] += 1

    free_colors = [set(range(1, deg[i] + 1)) for i in range(A + B + 1)]
    edge_by_color = [{} for _ in range(A + B + 1)]
    ans = [0] * L

    for e in range(L):
        u = edges_u[e]
        v = edges_v[e]

        c1 = next(iter(free_colors[u]))
        if c1 in free_colors[v]:
            c2 = c1
        else:
            c2 = next(iter(free_colors[v]))
            if c2 in free_colors[u]:
                c1 = c2

        if c1 == c2:
            edge_by_color[u][c1] = (v, e)
            edge_by_color[v][c1] = (u, e)
            free_colors[u].remove(c1)
            free_colors[v].remove(c1)
            ans[e] = c1
        else:
            path = []
            curr = v
            c_cur, c_alt = c1, c2
            while True:
                nxt, edge_id = edge_by_color[curr][c_cur]
                path.append((curr, nxt, edge_id, c_cur, c_alt))
                if c_alt in edge_by_color[nxt]:
                    curr = nxt
                    c_cur, c_alt = c_alt, c_cur
                else:
                    break

            for p_u, p_v, _, old_c, _ in path:
                del edge_by_color[p_u][old_c]
                del edge_by_color[p_v][old_c]

            for p_u, p_v, edge_id, _, new_c in path:
                edge_by_color[p_u][new_c] = (p_v, edge_id)
                edge_by_color[p_v][new_c] = (p_u, edge_id)
                ans[edge_id] = new_c

            free_colors[v].add(c1)
            free_colors[v].remove(c2)

            last_w = path[-1][1]
            old_c = path[-1][3]
            new_c = path[-1][4]
            free_colors[last_w].add(old_c)
            free_colors[last_w].remove(new_c)

            edge_by_color[u][c1] = (v, e)
            edge_by_color[v][c1] = (u, e)
            free_colors[u].remove(c1)
            free_colors[v].remove(c1)
            ans[e] = c1

    print("YES")
    print(*(ans[i] for i in range(L)))

if __name__ == '__main__':
    solve()
