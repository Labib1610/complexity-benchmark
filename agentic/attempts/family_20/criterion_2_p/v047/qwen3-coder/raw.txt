import sys
from collections import defaultdict
import heapq

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    A = int(data[0])
    B = int(data[1])
    L = int(data[2])
    
    links = []
    alpha_deg = [0] * (A + 1)
    beta_deg = [0] * (B + 1)
    
    idx = 3
    for i in range(L):
        u = int(data[idx])
        v = int(data[idx+1])
        links.append((u, v))
        alpha_deg[u] += 1
        beta_deg[v] += 1
        idx += 2
    
    K = max(max(alpha_deg), max(beta_deg))
    
    # For each terminal, maintain a min-heap of available phases
    alpha_phase_heap = [list(range(1, K+1)) for _ in range(A+1)]
    beta_phase_heap = [list(range(1, K+1)) for _ in range(B+1)]
    
    # To avoid list copying, we'll use a smarter approach:
    # For each terminal, track which phases are used
    alpha_used = [set() for _ in range(A+1)]
    beta_used = [set() for _ in range(B+1)]
    
    result = []
    
    for u, v in links:
        # Find a phase not used by either u or v
        used_phases = alpha_used[u].union(beta_used[v])
        # Pick the smallest available phase
        phase = 1
        while phase in used_phases:
            phase += 1
        result.append(str(phase))
        alpha_used[u].add(phase)
        beta_used[v].add(phase)
    
    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    main()
