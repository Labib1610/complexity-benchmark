We can model this as a bipartite graph edge-coloring problem.  
Since the graph is bipartite, we can color its edges with Δ colors using a standard algorithm (Kőnig’s line coloring theorem).  

Here’s the Python implementation:  

```python
import sys
from collections import deque

input = sys.stdin.readline

nL, nR, m = map(int, input().split())
edges = []
adj = [[] for _ in range(nL + nR + 1)]
deg = [0] * (nL + nR + 1)

for i in range(m):
    u, v = map(int, input().split())
    v += nL
    edges.append((u, v))
    adj[u].append((v, i))
    adj[v].append((u, i))
    deg[u] += 1
    deg[v] += 1

Delta = max(deg)
color = [0] * m

# Greedy edge coloring with Kempe chain recoloring
for u in range(1, nL + nR + 1):
    used = [False] * (Delta + 1)
    for v, idx in adj[u]:
        if color[idx]:
            used[color[idx]] = True
    mex = 1
    while used[mex]:
        mex += 1
    for v, idx in adj[u]:
        if not color[idx]:
            color[idx] = mex
            # Recolor conflicts
            conflict = True
            while conflict:
                conflict = False
                for w, idx2 in adj[v]:
                    if idx2 != idx and color[idx2] == color[idx]:
                        # find free color for w
                        used2 = [False] * (Delta + 1)
                        for x, idx3 in adj[w]:
                            if color[idx3]:
                                used2[color[idx3]] = True
                        c2 = 1
                        while used2[c2]:
                            c2 += 1
                        color[idx2] = c2
                        v = w
                        conflict = True
                        break
            break

print("YES")
print(" ".join(map(str, color)))
```
