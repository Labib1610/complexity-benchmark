```python
import sys
from collections import defaultdict
import threading

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n_P, n_S, m = map(int, sys.stdin.readline().split())
    
    edges = []
    adj_P = defaultdict(list)
    adj_S = defaultdict(list)
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        adj_P[u].append((v, i))
        adj_S[v].append((u, i))
    
    # Determine the maximum degree
    max_degree = 0
    for u in range(1, n_P + 1):
        if len(adj_P[u]) > max_degree:
            max_degree = len(adj_P[u])
    for v in range(1, n_S + 1):
        if len(adj_S[v]) > max_degree:
            max_degree = len(adj_S[v])
    
    # We need to edge-color the bipartite graph with max_degree colors
    # This is always possible for bipartite graphs (König's theorem)
    
    # We'll use a greedy approach with a priority-based assignment
    
    # For each vertex, we'll keep track of which colors are used by its neighbors
    result = [0] * m
    used_P = [set() for _ in range(n_P + 1)]
    used_S = [set() for _ in range(n_S + 1)]
    
    # Process edges in some order - let's sort by the sum of degrees of endpoints
    edge_order = []
    for i in range(m):
        u, v = edges[i]
        deg_u = len(adj_P[u])
        deg_v = len(adj_S[v])
        edge_order.append((deg_u + deg_v, i))
    
    edge_order.sort(reverse=True)
    
    # For each edge, assign the smallest available color
    for _, i in edge_order:
        u, v = edges[i]
        # Find the first color not used by either endpoint
        color = 1
        while color in used_P[u] or color in used_S[v]:
            color += 1
        result[i] = color
        used_P[u].add(color)
        used_S[v].add(color)
    
    print("YES")
    print(*result)

threading.Thread(target=main).start()
```
