import sys
from collections import deque

sys.setrecursionlimit(1_000_000)


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n_left, n_right, m = data[0], data[1], data[2]

    if m == 0:
        sys.stdout.write("YES\n\n")
        return

    left_original = [0] * m
    right_original = [0] * m
    deg_left = [0] * n_left
    deg_right = [0] * n_right

    p = 3
    for i in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        left_original[i] = u
        right_original[i] = v
        deg_left[u] += 1
        deg_right[v] += 1

    del data

    delta = max(max(deg_left), max(deg_right))

    def pack_vertices(degrees, limit):
        mapping = [-1] * len(degrees)
        bin_degrees = []
        current = -1

        for vertex, degree in enumerate(degrees):
            if degree == 0:
                continue
            if current == -1 or bin_degrees[current] + degree > limit:
                bin_degrees.append(0)
                current += 1
            mapping[vertex] = current
            bin_degrees[current] += degree

        return mapping, bin_degrees

    map_left, bin_deg_left = pack_vertices(deg_left, delta)
    map_right, bin_deg_right = pack_vertices(deg_right, delta)

    del deg_left, deg_right

    k = max(len(bin_deg_left), len(bin_deg_right))
    bin_deg_left.extend([0] * (k - len(bin_deg_left)))
    bin_deg_right.extend([0] * (k - len(bin_deg_right)))

    edge_left = [0] * m
    edge_right = [0] * m

    for i in range(m):
        edge_left[i] = map_left[left_original[i]]
        edge_right[i] = map_right[right_original[i]]

    del left_original, right_original, map_left, map_right

    # Add dummy edges until the compressed graph is delta-regular.
    i = 0
    j = 0
    need_left = delta - bin_deg_left[0]
    need_right = delta - bin_deg_right[0]

    while i < k and j < k:
        while i < k and need_left == 0:
            i += 1
            if i < k:
                need_left = delta - bin_deg_left[i]

        while j < k and need_right == 0:
            j += 1
            if j < k:
                need_right = delta - bin_deg_right[j]

        if i == k or j == k:
            break

        take = min(need_left, need_right)
        if take:
            edge_left.extend([i] * take)
            edge_right.extend([j] * take)
            need_left -= take
            need_right -= take

    total_edges = len(edge_left)
    answers = [0] * m

    if k == 1:
        for eid in range(m):
            answers[eid] = eid + 1
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, answers)) + "\n")
        return

    def perfect_matching(edges):
        adjacency = [[] for _ in range(k)]
        for eid in edges:
            adjacency[edge_left[eid]].append(eid)

        pair_left = [-1] * k
        pair_right = [-1] * k
        matched_edge = [-1] * k

        # Greedy initialization.
        for u in range(k):
            for eid in adjacency[u]:
                v = edge_right[eid]
                if pair_right[v] == -1:
                    pair_left[u] = v
                    pair_right[v] = u
                    matched_edge[u] = eid
                    break

        distance = [-1] * k

        while True:
            queue = deque()
            found = False

            for u in range(k):
                if pair_left[u] == -1:
                    distance[u] = 0
                    queue.append(u)
                else:
                    distance[u] = -1

            while queue:
                u = queue.popleft()
                nd = distance[u] + 1
                for eid in adjacency[u]:
                    v = edge_right[eid]
                    next_u = pair_right[v]
                    if next_u == -1:
                        found = True
                    elif distance[next_u] == -1:
                        distance[next_u] = nd
                        queue.append(next_u)

            if not found:
                break

            pointer = [0] * k

            def dfs(u):
                adj = adjacency[u]
                while pointer[u] < len(adj):
                    eid = adj[pointer[u]]
                    pointer[u] += 1
                    v = edge_right[eid]
                    next_u = pair_right[v]

                    if next_u == -1 or (
                        distance[next_u] == distance[u] + 1 and dfs(next_u)
                    ):
                        pair_left[u] = v
                        pair_right[v] = u
                        matched_edge[u] = eid
                        return True

                distance[u] = -1
                return False

            for u in range(k):
                if pair_left[u] == -1:
                    dfs(u)

        return matched_edge

    def split_even(edges):
        q = len(edges)
        mate_left = [-1] * q
        mate_right = [-1] * q
        last_left = [-1] * k
        last_right = [-1] * k

        for local_id, eid in enumerate(edges):
            u = edge_left[eid]
            previous = last_left[u]
            if previous == -1:
                last_left[u] = local_id
            else:
                mate_left[local_id] = previous
                mate_left[previous] = local_id
                last_left[u] = -1

            v = edge_right[eid]
            previous = last_right[v]
            if previous == -1:
                last_right[v] = local_id
            else:
                mate_right[local_id] = previous
                mate_right[previous] = local_id
                last_right[v] = -1

        side = bytearray(q)

        for start in range(q):
            if side[start]:
                continue

            previous = -1
            current = start
            value = 1

            while not side[current]:
                side[current] = value
                next_edge = mate_left[current]
                if next_edge == previous:
                    next_edge = mate_right[current]
                previous, current = current, next_edge
                value = 3 - value

        first = []
        second = []
        first_append = first.append
        second_append = second.append

        for local_id, eid in enumerate(edges):
            if side[local_id] == 1:
                first_append(eid)
            else:
                second_append(eid)

        return first, second

    def color_regular(edges, degree, first_color):
        if degree == 1:
            color = first_color
            for eid in edges:
                if eid < m:
                    answers[eid] = color
            return

        if degree & 1:
            matching = perfect_matching(edges)
            selected = bytearray(len(edge_left))
            color = first_color

            for eid in matching:
                selected[eid] = 1
                if eid < m:
                    answers[eid] = color

            remaining = [eid for eid in edges if not selected[eid]]
            color_regular(remaining, degree - 1, first_color + 1)
        else:
            first, second = split_even(edges)
            half = degree // 2
            color_regular(first, half, first_color)
            color_regular(second, half, first_color + half)

    color_regular(list(range(total_edges)), delta, 1)

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, answers)) + "\n")


if __name__ == "__main__":
    main()
