We can model the ports as vertices of a bipartite graph, where primary and secondary ports are the two partitions and cables are edges.  
The task is to edge-color this bipartite graph using exactly `Δ` colors, where `Δ` is the maximum degree. This is always possible for bipartite graphs (Kőnig’s line coloring theorem).

We can do edge-coloring using a greedy algorithm with alternating paths, often called **Kőnig's line coloring algorithm**. This is efficient here given `m ≤ 400000`.

Here is the Python implementation:

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    nP, nS, m = map(int, input().split())
    edges = []
    adjP = [[] for _ in range(nP)]
    adjS = [[] for _ in range(nS)]
    
    for i in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adjP[u].append((v, i))
        adjS[v].append((u, i))
    
    # Max degree
    delta = 0
    for i in range(nP):
        delta = max(delta, len(adjP[i]))
    for i in range(nS):
        delta = max(delta, len(adjS[i]))
    
    if m == 0:
        print("YES")
        print()
        return
    
    # Instead of using Delta colors initially, we construct color assignment.
    # We'll maintain color of each edge and used colors for each vertex.
    color = [0] * m  # 0 means uncolored
    usedP = [set() for _ in range(nP)]
    usedS = [set() for _ in range(nS)]
    
    # Try to color each edge
    # We apply simple algorithm: find smallest free color at both endpoints, if conflict, do alternating path.
    
    # This version using alternating path for bipartite graph edge coloring.
    for eidx in range(m):
        u, v = edges[eidx]
        # find smallest free color at u
        cu = 1
        while cu in usedP[u] or cu in usedS[v]:
            cu += 1
        if cu <= delta:
            color[eidx] = cu
            usedP[u].add(cu)
            usedS[v].add(cu)
            continue
        
        # If no common free color, we need recoloring via alternating path
        # We'll pick color a free at u, color b free at v, then flip along path.
        a = 1
        while a in usedP[u]:
            a += 1
        b = 1
        while b in usedS[v]:
            b += 1
        # Now a is free at u, b is free at v
        # We must find alternating path starting from v with color a, ending with color b.
        # Path starts at secondary v, moving along edge with color a to some primary,
        # then edge with color b to some secondary, etc.
        
        # Build alternating path
        path_edges = []  # list of edge indices
        curr_vertex = ('S', v)
        curr_color = a
        visited = set()
        
        while True:
            # find edge of color curr_color incident to curr_vertex
            found_edge = -1
            if curr_vertex[0] == 'S':
                idx = curr_vertex[1]
                for eid in range(m):
                    if edges[eid][1] == idx and color[eid] == curr_color:
                        found_edge = eid
                        break
            else:
                idx = curr_vertex[1]
                for eid in range(m):
                    if edges[eid][0] == idx and color[eid] == curr_color:
                        found_edge = eid
                        break
            if found_edge == -1:
                break
            path_edges.append(found_edge)
            # move to other endpoint
            if curr_vertex[0] == 'S':
                nxt = ('P', edges[found_edge][0])
            else:
                nxt = ('S', edges[found_edge][1])
            if (nxt, curr_color) in visited:
                break
            visited.add((nxt, curr_color))
            curr_vertex = nxt
            # switch color
            curr_color = b if curr_color == a else a
        
        # Now we have path_edges, flip colors
        for idx in path_edges:
            old = color[idx]
            new = b if old == a else a
            if old != 0:
                # remove old
                pu, pv = edges[idx]
                usedP[pu].discard(old)
                usedS[pv].discard(old)
            color[idx] = new
            usedP[edges[idx][0]].add(new)
            usedS[edges[idx][1]].add(new)
        
        # Now color the edge eidx
        color[eidx] = a
        usedP[u].add(a)
        usedS[v].add(a)
    
    # Output
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(color[i]) for i in range(m)) + "\n")

if __name__ == "__main__":
    solve()
```
