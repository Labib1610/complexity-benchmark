We need to assign each cable a service window from 1 to k, where k is the maximum degree among all ports, such that no two cables sharing a port get the same window. This is exactly a proper edge-coloring of a bipartite graph, and by König’s theorem, the edge chromatic number equals the maximum degree. So we can use a standard algorithm for edge-coloring a bipartite graph in O(m log Δ) or O(m Δ) time, but with m up to 400,000 we need an efficient approach.

We can use the **Kőnig line-coloring algorithm** that works in O(m log Δ) using a divide-and-conquer on the degree, or the classic **greedy with color interchange** that runs in O(m Δ) which might be too slow if Δ is large. But since it's bipartite, we can use the **O(m log Δ)** method by splitting the graph into two subgraphs with half the max degree, recursively coloring, and then merging.

Here’s the plan:

1. Read input, build adjacency lists for left and right vertices.
2. Compute k = max degree.
3. Use a recursive edge-coloring function:
   - If k == 0: no edges.
   - If k == 1: all edges get color 1.
   - Otherwise, split edges into two groups such that each vertex has at most ceil(deg/2) edges in each group. This is possible by Eulerian trail decomposition: since the graph is bipartite, we can find an Eulerian orientation and alternate edges.
   - Recursively color each subgraph with k1 = ceil(k/2) and k2 = floor(k/2) colors.
   - Then merge: for the second subgraph, add k1 to its colors.

This yields an O(m log Δ) algorithm.

                        stack.append((nb, e, v))
                        found = True
                        break
                if not found:
                    v, e_in, prev = stack.pop()
                    if e_in is not None:
                        trail.append((prev, v, e_in))
            # trail is in reverse order
            trail.reverse()
            # Assign groups alternately
            g = 0
            for u, v, e in trail:
                if e >= 0:
                    edge_group[e] = g
                g = 1 - g
        # Split edge_indices
        group0 = []
        group1 = []
        for e in edge_indices:
            if edge_group[e] == 0:
                group0.append(e)
            else:
                group1.append(e)
        k0 = (k + 1) // 2
        k1 = k // 2
        rec(group0, k0)
        rec(group1, k1)
        # Merge
        for e in group1:
            color[e] += k0

    rec(list(range(m)), k)
    print("YES")
    print(' '.join(str(color[i]) for i in range(m)))

if __name__ == "__main__":
    solve()

import sys
sys.setrecursionlimit(1 << 25)

def solve():
    import sys
    input = sys.stdin.readline

    nL, nR, m = map(int, input().split())
    edges = []
    adjL = [[] for _ in range(nL)]
    adjR = [[] for _ in range(nR)]

    for i in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adjL[u].append((v, i))
        adjR[v].append((u, i))

    if m == 0:
        print("YES")
        print()
        return

    degL = [len(adjL[i]) for i in range(nL)]
    degR = [len(adjR[i]) for i in range(nR)]
    k = max(max(degL) if nL else 0, max(degR) if nR else 0)

    color = [0] * m

    def rec(edge_indices, k):
        if k == 0:
            return
        if k == 1:
            for e in edge_indices:
                color[e] = 1
            return

        # Build adjacency for the current subgraph
        adj = {}
        for e in edge_indices:
            u, v = edges[e]
            adj.setdefault(('L', u), []).append((('R', v), e))
            adj.setdefault(('R', v), []).append((('L', u), e))

        # Add dummy edges to make all degrees even
        odd = [v for v in adj if len(adj[v]) % 2 == 1]
        dummy_idx = -1
        for i in range(0, len(odd), 2):
            u = odd[i]
            v = odd[i + 1]
            adj[u].append((v, dummy_idx))
            adj[v].append((u, dummy_idx))
            dummy_idx -= 1

        used = set()
        edge_group = {}

        # Find Eulerian circuits and assign alternating groups
        for start in list(adj.keys()):
            if all((nb, e) in used for nb, e in adj[start]):
                continue

            stack = [(start, None, None)]  # (vertex, edge_used, prev_vertex)
            trail = []

            while stack:
                v, e_in, prev = stack[-1]
                found = False
                for nb, e in adj[v]:
                    if (v, nb, e) not in used and (nb, v, e) not in used:
                        used.add((v, nb, e))
                        used.add((nb, v, e))
                        stack.append((nb, e, v))
                        found = True
                        break
                if not found:
                    v, e_in, prev = stack.pop()
                    if e_in is not None:
                        trail.append((prev, v, e_in))

            trail.reverse()

            g = 0
            for u, v, e in trail:
                if e >= 0:
                    edge_group[e] = g
                g = 1 - g

        group0 = []
        group1 = []
        for e in edge_indices:
            if edge_group[e] == 0:
                group0.append(e)
            else:
                group1.append(e)

        k0 = (k + 1) // 2
        k1 = k // 2

        rec(group0, k0)
        rec(group1, k1)

        for e in group1:
            color[e] += k0

    rec(list(range(m)), k)

    print("YES")
    print(' '.join(str(color[i]) for i in range(m)))


if __name__ == "__main__":
    solve()
