import sys
from collections import deque

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    
    edges = []
    deg_L = [0] * (nL + 1)
    deg_R = [0] * (nR + 1)
    
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        deg_L[u] += 1
        deg_R[v] += 1
    
    K = max(max(deg_L[1:]), max(deg_R[1:])) if m > 0 else 0
    
    # Build bipartite graph for edge coloring
    adj = [[] for _ in range(nL + 1)]
    for i, (u, v) in enumerate(edges):
        adj[u].append((v, i))
    
    # Current available color for each vertex
    # We use 1-based indexing for colors
    avail_L = [0] * (nL + 1)
    avail_R = [0] * (nR + 1)
    
    # Color assignment for edges
    color = [0] * m
    
    # For each edge, we need to find a color
    for u in range(1, nL + 1):
        for v, edge_idx in adj[u]:
            # Find available colors
            used_L = avail_L[u]
            used_R = avail_R[v]
            
            # Find a color not used by u or v
            # Colors are 1 to K
            # We need to find c such that bit c is not set in used_L or used_R
            
            # Simple case: if there's a color available for both
            combined = used_L | used_R
            found = -1
            for c in range(1, K + 1):
                if not (combined & (1 << c)):
                    found = c
                    break
            
            if found != -1:
                color[edge_idx] = found
                avail_L[u] |= (1 << found)
                avail_R[v] |= (1 << found)
            else:
                # Need to recolor: find c1 used by u but not v, c2 used by v but not u
                c1 = -1
                c2 = -1
                for c in range(1, K + 1):
                    if (used_L & (1 << c)) and not (used_R & (1 << c)):
                        c1 = c
                    if (used_R & (1 << c)) and not (used_L & (1 << c)):
                        c2 = c
                
                # Find alternating path starting with c1-c2
                # BFS to find augmenting path
                # We need to find a path from u using color c2 that ends at a vertex with free color
                
                # Build reverse index: for each (vertex, color) -> edge index
                # For R vertices: which edge has this color
                R_color_to_edge = {}
                for prev_u, prev_v, prev_idx in [(edges[j][0], edges[j][1], j) for j in range(m) if color[j] != 0]:
                    c = color[prev_idx]
                    if prev_v not in R_color_to_edge:
                        R_color_to_edge[prev_v] = {}
                    R_color_to_edge[prev_v][c] = prev_idx
                
                # BFS
                queue = deque()
                queue.append((u, 'L', c1, c2, []))
                visited_L = set()
                visited_R = set()
                visited_L.add(u)
                
                augment_path = None
                
                while queue and augment_path is None:
                    curr_u, side, col1, col2, path = queue.popleft()
                    
                    if side == 'L':
                        # At L vertex, we want to follow edges with col2
                        for next_v, e_idx in adj[curr_u]:
                            if e_idx == edge_idx:
                                continue
                            if color[e_idx] == col2:
                                if next_v not in visited_R:
                                    visited_R.add(next_v)
                                    new_path = path + [(e_idx, col1, col2)]
                                    # Check if we can stop here
                                    if not (avail_R[next_v] & (1 << col1)):
                                        augment_path = new_path
                                        break
                                    queue.append((next_v, 'R', col2, col1, new_path))
                    else:
                        # At R vertex, we want to follow edges with col1
                        curr_v = curr_u
                        if curr_v in R_color_to_edge and col1 in R_color_to_edge[curr_v]:
                            e_idx = R_color_to_edge[curr_v][col1]
                            next_u = edges[e_idx][0]
                            if next_u not in visited_L:
                                visited_L.add(next_u)
                                new_path = path + [(e_idx, col2, col1)]
                                queue.append((next_u, 'L', col1, col2, new_path))
                
                if augment_path is not None:
                    # Swap colors along the path
                    for e_idx, old_c, new_c in augment_path:
                        old_color = color[e_idx]
                        # Remove old color
                        uu, vv = edges[e_idx]
                        avail_L[uu] &= ~(1 << old_color)
                        avail_R[vv] &= ~(1 << old_color)
                        # Add new color
                        color[e_idx] = new_c
                        avail_L[uu] |= (1 << new_c)
                        avail_R[vv] |= (1 << new_c)
                    
                    # Now assign c1 to current edge
                    color[edge_idx] = c1
                    avail_L[u] |= (1 << c1)
                    avail_R[v] |= (1 << c1)
                else:
                    # Fallback: just find any available color (shouldn't happen in theory)
                    for c in range(1, K + 1):
                        if not (used_L & (1 << c)):
                            color[edge_idx] = c
                            avail_L[u] |= (1 << c)
                            avail_R[v] |= (1 << c)
                            break
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
