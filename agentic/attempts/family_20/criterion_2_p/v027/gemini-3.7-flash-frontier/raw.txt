import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    nL = int(input_data[0])
    nR = int(input_data[1])
    m = int(input_data[2])
    
    if m == 0:
        sys.stdout.write("YES\n\n")
        return
    
    deg_L = [0] * (nL + 1)
    deg_R = [0] * (nR + 1)
    edges = []
    
    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        deg_L[u] += 1
        deg_R[v] += 1
        edges.append((u, v, i))
        
    K = max(max(deg_L), max(deg_R))
    if K == 0:
        sys.stdout.write("YES\n\n")
        return

    # Group vertices to condense the bipartite graph
    group_L = [0] * (nL + 1)
    L_groups = []
    curr_deg = 0
    curr_g = 0
    for u in range(1, nL + 1):
        if deg_L[u] == 0:
            continue
        if curr_deg + deg_L[u] > K:
            curr_g += 1
            curr_deg = 0
        group_L[u] = curr_g
        curr_deg += deg_L[u]
    num_L_groups = curr_g + 1 if any(deg_L[u] > 0 for u in range(1, nL + 1)) else 0

    group_R = [0] * (nR + 1)
    curr_deg = 0
    curr_g = 0
    for v in range(1, nR + 1):
        if deg_R[v] == 0:
            continue
        if curr_deg + deg_R[v] > K:
            curr_g += 1
            curr_deg = 0
        group_R[v] = curr_g
        curr_deg += deg_R[v]
    num_R_groups = curr_g + 1 if any(deg_R[v] > 0 for v in range(1, nR + 1)) else 0

    V_prime = max(num_L_groups, num_R_groups)
    if V_prime == 0:
        V_prime = 1

    # Map edges to condensed graph
    condensed_edges = []
    deg_LG = [0] * V_prime
    deg_RG = [0] * V_prime
    for u, v, eid in edges:
        gu = group_L[u]
        gv = group_R[v]
        condensed_edges.append((gu, gv, eid))
        deg_LG[gu] += 1
        deg_RG[gv] += 1

    # Add dummy edges to make condensed graph K-regular
    ptr_L = 0
    ptr_R = 0
    while ptr_L < V_prime and ptr_R < V_prime:
        need_L = K - deg_LG[ptr_L]
        need_R = K - deg_RG[ptr_R]
        if need_L == 0:
            ptr_L += 1
            continue
        if need_R == 0:
            ptr_R += 1
            continue
        add = min(need_L, need_R)
        for _ in range(add):
            condensed_edges.append((ptr_L, ptr_R, -1))
        deg_LG[ptr_L] += add
        deg_RG[ptr_R] += add

    ans = [0] * m

    sys.setrecursionlimit(max(1000000, 2 * V_prime + 100))

    def color_regular(edge_list, d, color_start):
        if d == 1:
            for u, v, eid in edge_list:
                if eid != -1:
                    ans[eid] = color_start
            return

        if d % 2 == 1:
            # Find a perfect matching
            adj = [[] for _ in range(V_prime)]
            for i, (u, v, eid) in enumerate(edge_list):
                adj[u].append((v, i))

            match_L = [-1] * V_prime
            match_R = [-1] * V_prime
            matched_edge_idx = [-1] * V_prime

            # Greedy initial matching
            for u in range(V_prime):
                for v, e_idx in adj[u]:
                    if match_R[v] == -1:
                        match_L[u] = v
                        match_R[v] = u
                        matched_edge_idx[u] = e_idx
                        break

            # Augmenting paths
            for u in range(V_prime):
                if match_L[u] != -1:
                    continue
                vis = [False] * V_prime

                def dfs(curr):
                    vis[curr] = True
                    for nxt, e_idx in adj[curr]:
                        w = match_R[nxt]
                        if w == -1 or (not vis[w] and dfs(w)):
                            match_L[curr] = nxt
                            match_R[nxt] = curr
                            matched_edge_idx[curr] = e_idx
                            return True
                    return False

                dfs(u)

            in_matching = [False] * len(edge_list)
            for u in range(V_prime):
                e_idx = matched_edge_idx[u]
                in_matching[e_idx] = True
                orig_eid = edge_list[e_idx][2]
                if orig_eid != -1:
                    ans[orig_eid] = color_start

            rem_edges = [edge_list[i] for i in range(len(edge_list)) if not in_matching[i]]
            color_regular(rem_edges, d - 1, color_start + 1)
        else:
            # Even degree: cycle decomposition into two d/2-regular graphs
            total_V = 2 * V_prime
            adj = [[] for _ in range(total_V)]
            num_e = len(edge_list)
            for i, (u, v, eid) in enumerate(edge_list):
                adj[u].append((v + V_prime, i))
                adj[v + V_prime].append((u, i))

            used = [False] * num_e
            head = [0] * total_V
            in_path = [-1] * total_V

            E1 = []
            E2 = []

            for root in range(total_V):
                while head[root] < len(adj[root]):
                    # Check if there's an unused edge
                    while head[root] < len(adj[root]) and used[adj[root][head[root]][1]]:
                        head[root] += 1
                    if head[root] >= len(adj[root]):
                        break

                    path = [root]
                    path_edges = []
                    in_path[root] = 0

                    while path:
                        curr = path[-1]
                        found = False
                        while head[curr] < len(adj[curr]):
                            nxt, e_idx = adj[curr][head[curr]]
                            head[curr] += 1
                            if not used[e_idx]:
                                used[e_idx] = True
                                found = True
                                break

                        if found:
                            if in_path[nxt] != -1:
                                idx_p = in_path[nxt]
                                cycle_edges = path_edges[idx_p:] + [e_idx]
                                for v_node in path[idx_p + 1:]:
                                    in_path[v_node] = -1
                                del path[idx_p + 1:]
                                del path_edges[idx_p:]
                                for k, c_e_idx in enumerate(cycle_edges):
                                    if k % 2 == 0:
                                        E1.append(edge_list[c_e_idx])
                                    else:
                                        E2.append(edge_list[c_e_idx])
                            else:
                                in_path[nxt] = len(path)
                                path.append(nxt)
                                path_edges.append(e_idx)
                        else:
                            in_path[curr] = -1
                            path.pop()
                            if path_edges:
                                path_edges.pop()

            half = d // 2
            color_regular(E1, half, color_start)
            color_regular(E2, half, color_start + half)

    color_regular(condensed_edges, K, 1)

    out = ["YES", " ".join(map(str, ans))]
    sys.stdout.write("\n".join(out) + "\n")

if __name__ == '__main__':
    solve()
