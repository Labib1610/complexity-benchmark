```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    degL = [0] * nL
    degR = [0] * nR
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    Delta = 0
    if degL:
        Delta = max(Delta, max(degL))
    if degR:
        Delta = max(Delta, max(degR))

    usedL = [set() for _ in range(nL)]
    usedR = [set() for _ in range(nR)]
    ptrL = [1] * nL
    ptrR = [1] * nR

    ans = [0] * m
    for idx, (u, v) in enumerate(edges):
        pu = ptrL[u]
        while pu in usedL[u]:
            pu += 1
        pv = ptrR[v]
        while pv in usedR[v]:
            pv += 1
        c = pu if pu > pv else pv
        while c in usedL[u] or c in usedR[v]:
            c += 1
        ans[idx] = c
        usedL[u].add(c)
        usedR[v].add(c)
        if ptrL[u] == c:
            ptrL[u] = c + 1
        if ptrR[v] == c:
            ptrR[v] = c + 1

    out = ['YES', ' '.join(map(str, ans))]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
