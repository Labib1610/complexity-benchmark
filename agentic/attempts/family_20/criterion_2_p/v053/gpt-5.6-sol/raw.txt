import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
it = iter(data)

nL = next(it)
nR = next(it)
m = next(it)

U = [0] * m
V = [0] * m
degL = [0] * nL
degR = [0] * nR

for i in range(m):
    u = next(it) - 1
    v = next(it) - 1
    U[i] = u
    V[i] = v
    degL[u] += 1
    degR[v] += 1

delta = max(max(degL, default=0), max(degR, default=0))

if m == 0:
    sys.stdout.write("YES\n\n")
    raise SystemExit

answer = [0] * m

adjL = [None] * nL
adjR = [None] * nR
mateL = [-1] * nL
mateR = [-1] * nR
distL = [0] * nL
distR = [0] * nR

INF = 10**18


def covering_matching(edges, d):
    touchedL = []
    touchedR = []

    for e in edges:
        u = U[e]
        v = V[e]

        if adjL[u] is None:
            adjL[u] = []
            touchedL.append(u)
            mateL[u] = -1
        adjL[u].append(e)

        if adjR[v] is None:
            adjR[v] = []
            touchedR.append(v)
            mateR[v] = -1
        adjR[v].append(e)

    requiredL = [u for u in touchedL if len(adjL[u]) == d]
    requiredR = [v for v in touchedR if len(adjR[v]) == d]
    requiredRSet = set(requiredR)

    # First saturate all required left vertices.
    while True:
        q = deque()
        for u in requiredL:
            if mateL[u] == -1:
                distL[u] = 0
                q.append(u)
            else:
                distL[u] = INF

        shortest = INF

        while q:
            u = q.popleft()
            du = distL[u]
            if du >= shortest:
                continue

            for e in adjL[u]:
                v = V[e]
                me = mateR[v]
                if me == -1:
                    shortest = du + 1
                else:
                    nu = U[me]
                    if distL[nu] == INF:
                        distL[nu] = du + 1
                        q.append(nu)

        if shortest == INF:
            break

        def dfs_left(u):
            du = distL[u]
            for e in adjL[u]:
                v = V[e]
                me = mateR[v]
                if me == -1:
                    if du + 1 != shortest:
                        continue
                    mateL[u] = e
                    mateR[v] = e
                    return True

                nu = U[me]
                if distL[nu] == du + 1 and dfs_left(nu):
                    mateL[u] = e
                    mateR[v] = e
                    return True

            distL[u] = INF
            return False

        changed = False
        for u in requiredL:
            if mateL[u] == -1 and dfs_left(u):
                changed = True
        if not changed:
            break

    # Then saturate required right vertices without uncovering required left ones.
    while True:
        q = deque()
        for v in requiredR:
            if mateR[v] == -1:
                distR[v] = 0
                q.append(v)
            else:
                distR[v] = INF

        shortest = INF

        while q:
            v = q.popleft()
            dv = distR[v]
            if dv >= shortest:
                continue

            matched_edge = mateR[v]
            for e in adjR[v]:
                if e == matched_edge:
                    continue

                u = U[e]
                old = mateL[u]

                if old == -1 or V[old] not in requiredRSet:
                    shortest = dv + 1
                else:
                    nv = V[old]
                    if distR[nv] == INF:
                        distR[nv] = dv + 1
                        q.append(nv)

        if shortest == INF:
            break

        def dfs_right(v):
            dv = distR[v]
            matched_edge = mateR[v]

            for e in adjR[v]:
                if e == matched_edge:
                    continue

                u = U[e]
                old = mateL[u]

                if old == -1 or V[old] not in requiredRSet:
                    if dv + 1 != shortest:
                        continue
                    if old != -1:
                        mateR[V[old]] = -1
                    mateL[u] = e
                    mateR[v] = e
                    return True

                nv = V[old]
                if distR[nv] == dv + 1 and dfs_right(nv):
                    mateL[u] = e
                    mateR[v] = e
                    return True

            distR[v] = INF
            return False

        changed = False
        for v in requiredR:
            if mateR[v] == -1 and dfs_right(v):
                changed = True
        if not changed:
            break

    matched = []
    matched_set = set()

    for u in touchedL:
        e = mateL[u]
        if e != -1:
            matched.append(e)
            matched_set.add(e)

    remaining = [e for e in edges if e not in matched_set]

    for u in touchedL:
        adjL[u] = None
        mateL[u] = -1
    for v in touchedR:
        adjR[v] = None
        mateR[v] = -1

    return matched, remaining


def euler_split(edges):
    q = len(edges)
    endpoints_a = [0] * q
    endpoints_b = [0] * q
    graph = {}
    odd_left = []
    odd_right = []

    for i, e in enumerate(edges):
        a = U[e]
        b = nL + V[e]
        endpoints_a[i] = a
        endpoints_b[i] = b
        graph.setdefault(a, []).append(i)
        graph.setdefault(b, []).append(i)

    for vertex, lst in graph.items():
        if len(lst) & 1:
            if vertex < nL:
                odd_left.append(vertex)
            else:
                odd_right.append(vertex)

    next_vertex = nL + nR

    def add_virtual(a, b):
        token = len(endpoints_a)
        endpoints_a.append(a)
        endpoints_b.append(b)
        graph.setdefault(a, []).append(token)
        graph.setdefault(b, []).append(token)

    if len(odd_left) & 1:
        add_virtual(odd_left.pop(), odd_right.pop())

    while odd_left:
        a = odd_left.pop()
        b = odd_left.pop()
        dummy = next_vertex
        next_vertex += 1
        add_virtual(a, dummy)
        add_virtual(b, dummy)

    while odd_right:
        a = odd_right.pop()
        b = odd_right.pop()
        dummy = next_vertex
        next_vertex += 1
        add_virtual(a, dummy)
        add_virtual(b, dummy)

    total_tokens = len(endpoints_a)
    used = bytearray(total_tokens)
    position = {vertex: len(lst) - 1 for vertex, lst in graph.items()}

    first = []
    second = []

    for start in list(graph.keys()):
        pos = position[start]
        while pos >= 0 and used[graph[start][pos]]:
            pos -= 1
        position[start] = pos
        if pos < 0:
            continue

        stack_vertices = [start]
        stack_edges = []
        tour = []

        while stack_vertices:
            x = stack_vertices[-1]
            lst = graph[x]
            p = position[x]

            while p >= 0 and used[lst[p]]:
                p -= 1
            position[x] = p

            if p < 0:
                stack_vertices.pop()
                if stack_edges:
                    tour.append(stack_edges.pop())
                continue

            token = lst[p]
            position[x] = p - 1
            if used[token]:
                continue

            used[token] = 1
            a = endpoints_a[token]
            b = endpoints_b[token]
            y = b if x == a else a

            stack_vertices.append(y)
            stack_edges.append(token)

        for i, token in enumerate(tour):
            if token < q:
                if i & 1:
                    second.append(edges[token])
                else:
                    first.append(edges[token])

    return first, second


def color_edges(edges, d, offset):
    if not edges:
        return

    if d == 1:
        for e in edges:
            answer[e] = offset
        return

    if d & 1:
        matching, remaining = covering_matching(edges, d)
        special_color = offset + d - 1
        for e in matching:
            answer[e] = special_color
        color_edges(remaining, d - 1, offset)
    else:
        first, second = euler_split(edges)
        half = d // 2
        color_edges(first, half, offset)
        color_edges(second, half, offset + half)


color_edges(list(range(m)), delta, 1)

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answer)))
sys.stdout.write("\n")
