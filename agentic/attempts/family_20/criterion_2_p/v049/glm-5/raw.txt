This is an edge coloring problem on a bipartite graph. By König's theorem, the chromatic index equals the maximum degree Δ. We can find a valid coloring using Δ colors by repeatedly finding a perfect matching in a regularized bipartite graph.

```python
import sys
from collections import defaultdict, deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    nA = int(next(it))
    nB = int(next(it))
    m = int(next(it))
    
    edges = []
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
    
    if m == 0:
        sys.stdout.write("YES\n\n")
        return
    
    degA = [0] * nA
    degB = [0] * nB
    for u, v in edges:
        degA[u] += 1
        degB[v] += 1
    
    T = max(max(degA) if nA > 0 else 0, max(degB) if nB > 0 else 0)
    if T == 0:
        sys.stdout.write("YES\n\n")
        return
    
    # Pad to T-regular bipartite graph
    # Left side: original nA nodes + dummy nodes to reach T*nA' >= T*nB'
    # Right side: original nB nodes + dummy nodes to reach T*nB' >= T*nA'
    # We need T*nA' = T*nB', so nA' = nB' = max(nA, nB) works if both sides have enough capacity
    
    # Actually simpler: add dummy nodes on whichever side has fewer total capacity
    # Left capacity needed: T * nA_padded
    # Right capacity needed: T * nB_padded
    # We need them equal, so pad smaller side
    
    # Current left capacity = T * nA, right capacity = T * nB
    # Deficits: left needs T*nA - sum(degA) = T*nA - m dummy edges
    #           right needs T*nB - m dummy edges
    # Both deficits are non-negative. Pad the side with smaller n to make capacities equal.
    
    if nA < nB:
        nA_pad = nB
        nB_pad = nB
    else:
        nA_pad = nA
        nB_pad = nA
    
    # Now T * nA_pad = T * nB_pad
    # Left deficit = T * nA_pad - m (dummy edges from left)
    # Right deficit = T * nB_pad - m (dummy edges from right)
    # These are equal, good.
    
    left_deficit = T * nA_pad - m
    right_deficit = T * nB_pad - m
    
    # Build adjacency for Hopcroft-Karp
    # Nodes: left 0..nA_pad-1, right 0..nB_pad-1
    # Real edges + dummy edges
    
    adj = [[] for _ in range(nA_pad)]
    edge_list = []  # (left, right, edge_id) where edge_id < m means real, >= m means dummy
    
    for i, (u, v) in enumerate(edges):
        adj[u].append((v, i))
        edge_list.append((u, v, i))
    
    # Add dummy edges
    dummy_edge_id = m
    # We need to connect dummy left nodes and dummy right nodes
    # Left dummy nodes: nA .. nA_pad-1 (if nA_pad > nA)
    # Right dummy nodes: nB .. nB_pad-1 (if nB_pad > nB)
    # Also need to fill remaining degree of real nodes
    
    # Remaining degree for each left node
    rem_left = [T - degA[i] for i in range(nA)] + [T] * (nA_pad - nA)
    rem_right = [T - degB[i] for i in range(nB)] + [T] * (nB_pad - nB)
    
    # Greedily add dummy edges to fill remaining degrees
    # Use two pointers
    li = 0
    ri = 0
    n_left_dummy = 0
    while n_left_dummy < left_deficit:
        while li < nA_pad and rem_left[li] == 0:
            li += 1
        while ri < nB_pad and rem_right[ri] == 0:
            ri += 1
        if li >= nA_pad or ri >= nB_pad:
            break
        cnt = min(rem_left[li], rem_right[ri])
        for _ in range(cnt):
            adj[li].append((ri, dummy_edge_id))
            edge_list.append((li, ri, dummy_edge_id))
            dummy_edge_id += 1
            n_left_dummy += 1
        rem_left[li] -= cnt
        rem_right[ri] -= cnt
    
    # Now we have a T-regular bipartite graph with nA_pad nodes on each side
    # Decompose into T perfect matchings using repeated Hopcroft-Karp
    
    color = [0] * len(edge_list)  # color for each edge (1-indexed)
    
    # We need to find T perfect matchings
    # For each matching, remove matched edges and assign color
    
    current_adj = [list(adj[i]) for i in range(nA_pad)]
    # current_adj[i] is list of (right_node, edge_id)
    
    # Build reverse adjacency for removal
    # We'll use a different approach: for each color, find perfect matching
    
    # Use a simpler approach: maintain adjacency lists, for each color find matching
    # and remove matched edges
    
    # For efficiency, use Hopcroft-Karp
    
    for color_val in range(1, T + 1):
        # Find perfect matching in current graph
        # current_adj[i] = list of (v, edge_id)
        
        # Build adjacency without edge_ids for matching
        match_l = [-1] * nA_pad
        match_r = [-1] * nB_pad
        
        # Simple augmenting path (since graph is regular, matching exists)
        # Use BFS/DFS
        
        # Build adjacency list of just neighbors
        cur_adj_simple = [[] for _ in range(nA_pad)]
        for i in range(nA_pad):
            for (v, eid) in current_adj[i]:
                cur_adj_simple[i].append(v)
        
        # Hopcroft-Karp
        INF = float('inf')
        dist = [0] * nA_pad
        
        def bfs():
            queue = deque()
            for u in range(nA_pad):
                if match_l[u] == -1:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = INF
            d = INF
            while queue:
                u = queue.popleft()
                if dist[u] < d:
                    for v in cur_adj_simple[u]:
                        if match_r[v] != -1:
                            if dist[match_r[v]] == INF:
                                dist[match_r[v]] = dist[u] + 1
                                queue.append(match_r[v])
                        else:
                            d = dist[u] + 1
            return d != INF
        
        def dfs(u):
            for v in cur_adj_simple[u]:
                if match_r[v] == -1 or (dist[match_r[v]] == dist[u] + 1 and dfs(match_r[v])):
                    match_l[u] = v
                    match_r[v] = u
                    return True
            dist[u] = INF
            return False
        
        matching_count = 0
        while bfs():
            for u in range(nA_pad):
                if match_l[u] == -1:
                    if dfs(u):
                        matching_count += 1
        
        # Assign colors to matched edges and remove them
        for u in range(nA_pad):
            v = match_l[u]
            if v != -1:
                # Find the edge (u, v) in current_adj[u] and get its edge_id
                for idx in range(len(current_adj[u])):
                    if current_adj[u][idx][0] == v:
                        eid = current_adj[u][idx][1]
                        color[eid] = color_val
                        current_adj[u].pop(idx)
                        break
    
    # Output
    result = []
    for i in range(m):
        result.append(str(color[i]))
    
    sys.stdout.write("YES\n" + " ".join(result) + "\n")

solve()
```
