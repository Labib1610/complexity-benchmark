```python
import sys
import heapq
import random

# Increase recursion depth for deep recursive calls
sys.setrecursionlimit(2000000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    nA = int(input_data[0])
    nB = int(input_data[1])
    m = int(input_data[2])
    
    if m == 0:
        print("YES")
        print()
        return

    deg_A = [0] * (nA + 1)
    deg_B = [0] * (nB + 1)
    
    edges_raw = []
    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges_raw.append((u, v))
        deg_A[u] += 1
        deg_B[v] += 1

    Delta = max(max(deg_A), max(deg_B))
    if Delta == 0:
        print("YES")
        print()
        return

    # Merge vertices in A
    active_A = [u for u in range(1, nA + 1) if deg_A[u] > 0]
    heap_A = [(deg_A[u], u, [u]) for u in active_A]
    heapq.heapify(heap_A)
    while len(heap_A) >= 2:
        d1, u1, list1 = heapq.heappop(heap_A)
        d2, u2, list2 = heapq.heappop(heap_A)
        if d1 + d2 <= Delta:
            heapq.heappush(heap_A, (d1 + d2, u1, list1 + list2))
        else:
            heapq.heappush(heap_A, (d1, u1, list1))
            heapq.heappush(heap_A, (d2, u2, list2))
            break

    map_A = {}
    for new_id, (_, _, group) in enumerate(heap_A):
        for u in group:
            map_A[u] = new_id
    num_A = len(heap_A)

    # Merge vertices in B
    active_B = [v for v in range(1, nB + 1) if deg_B[v] > 0]
    heap_B = [(deg_B[v], v, [v]) for v in active_B]
    heapq.heapify(heap_B)
    while len(heap_B) >= 2:
        d1, v1, list1 = heapq.heappop(heap_B)
        d2, v2, list2 = heapq.heappop(heap_B)
        if d1 + d2 <= Delta:
            heapq.heappush(heap_B, (d1 + d2, v1, list1 + list2))
        else:
            heapq.heappush(heap_B, (d1, v1, list1))
            heapq.heappush(heap_B, (d2, v2, list2))
            break

    map_B = {}
    for new_id, (_, _, group) in enumerate(heap_B):
        for v in group:
            map_B[v] = new_id
    num_B = len(heap_B)

    # Equalize number of vertices
    N = max(num_A, num_B)
    cur_deg_A = [0] * N
    cur_deg_B = [0] * N

    edges = []
    for i, (u, v) in enumerate(edges_raw):
        nu = map_A[u]
        nv = map_B[v]
        edges.append((nu, nv, i))
        cur_deg_A[nu] += 1
        cur_deg_B[nv] += 1

    # Add dummy edges to make the graph Delta-regular
    dummy_eid = m
    ptr_A = 0
    ptr_B = 0
    while ptr_A < N and ptr_B < N:
        rem_A = Delta - cur_deg_A[ptr_A]
        rem_B = Delta - cur_deg_B[ptr_B]
        if rem_A == 0:
            ptr_A += 1
            continue
        if rem_B == 0:
            ptr_B += 1
            continue
        take = min(rem_A, rem_B)
        for _ in range(take):
            edges.append((ptr_A, ptr_B, dummy_eid))
            dummy_eid += 1
        cur_deg_A[ptr_A] += take
        cur_deg_B[ptr_B] += take

    color_res = [0] * m
    random.seed(42)

    def split_euler(curr_edges, n_L, n_R):
        total_V = n_L + n_R
        adj = [[] for _ in range(total_V)]
        for local_id, (u, v, gid) in enumerate(curr_edges):
            v_idx = n_L + v
            adj[u].append((v_idx, local_id, u, v, gid))
            adj[v_idx].append((u, local_id, u, v, gid))

        used = [False] * len(curr_edges)
        edge_ptr = [0] * total_V
        edges_0 = []
        edges_1 = []

        for start_node in range(total_V):
            if edge_ptr[start_node] < len(adj[start_node]):
                tour_edges = []
                stack_v = [start_node]
                stack_e = []
                while stack_v:
                    curr = stack_v[-1]
                    while edge_ptr[curr] < len(adj[curr]) and used[adj[curr][edge_ptr[curr]][1]]:
                        edge_ptr[curr] += 1
                    if edge_ptr[curr] < len(adj[curr]):
                        nxt, local_id, u, v, gid = adj[curr][edge_ptr[curr]]
                        edge_ptr[curr] += 1
                        used[local_id] = True
                        stack_e.append((u, v, gid))
                        stack_v.append(nxt)
                    else:
                        stack_v.pop()
                        if stack_e:
                            tour_edges.append(stack_e.pop())

                for i, e in enumerate(tour_edges):
                    if i % 2 == 0:
                        edges_0.append(e)
                    else:
                        edges_1.append(e)
        return edges_0, edges_1

    def find_matching(curr_edges, n_L, n_R):
        adj = [[] for _ in range(n_L)]
        for u, v, gid in curr_edges:
            adj[u].append((v, gid))

        match_R = [-1] * n_R
        match_L_edge = [None] * n_L

        for u0 in range(n_L):
            curr = u0
            path = [curr]
            in_path = {curr: 0}
            edge_path = []

            while True:
                v, gid = random.choice(adj[curr])
                if match_R[v] == -1:
                    edge_path.append((v, gid))
                    for k in range(len(edge_path)):
                        uk = path[k]
                        vk, ek_gid = edge_path[k]
                        match_L_edge[uk] = (vk, ek_gid)
                        match_R[vk] = uk
                    break
                else:
                    next_u = match_R[v]
                    if next_u in in_path:
                        idx = in_path[next_u]
                        while len(path) > idx + 1:
                            del in_path[path.pop()]
                            edge_path.pop()
                        curr = next_u
                    else:
                        edge_path.append((v, gid))
                        curr = next_u
                        in_path[curr] = len(path)
                        path.append(curr)

        matched_gids = set(gid for _, gid in match_L_edge)
        rem_edges = []
        for e in curr_edges:
            if e[2] in matched_gids:
                if e[2] < m:
                    color_res[e[2]] = None # Will be set by caller
            else:
                rem_edges.append(e)
        return matched_gids, rem_edges

    def solve(curr_edges, delta, color_offset):
        if delta == 0 or not curr_edges:
            return
        if delta == 1:
            for _, _, gid in curr_edges:
                if gid < m:
                    color_res[gid] = color_offset + 1
            return
        if delta % 2 == 1:
            matched_gids, rem_edges = find_matching(curr_edges, N, N)
            assigned_c = color_offset + delta
            for gid in matched_gids:
                if gid < m:
                    color_res[gid] = assigned_c
            solve(rem_edges, delta - 1, color_offset)
        else:
            e0, e1 = split_euler(curr_edges, N, N)
            half = delta // 2
            solve(e0, half, color_offset)
            solve(e1, half, color_offset + half)

    solve(edges, Delta, 0)

    print("YES")
    print(*(color_res[i] for i in range(m)))

if __name__ == '__main__':
    main()
```
