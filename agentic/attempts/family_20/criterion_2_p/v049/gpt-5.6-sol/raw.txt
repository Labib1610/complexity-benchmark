import sys
from array import array
from collections import deque

sys.setrecursionlimit(1_000_000)


def integers():
    data = sys.stdin.buffer.read()
    num = 0
    in_num = False
    for c in data:
        if 48 <= c <= 57:
            num = num * 10 + c - 48
            in_num = True
        elif in_num:
            yield num
            num = 0
            in_num = False
    if in_num:
        yield num


it = iter(integers())
n_a = next(it)
n_b = next(it)
m = next(it)

orig_u = array('i')
orig_v = array('i')
deg_a = [0] * n_a
deg_b = [0] * n_b

for _ in range(m):
    u = next(it) - 1
    v = next(it) - 1
    orig_u.append(u)
    orig_v.append(v)
    deg_a[u] += 1
    deg_b[v] += 1

if m == 0:
    sys.stdout.write("YES\n\n")
    raise SystemExit

delta = max(max(deg_a), max(deg_b))


def make_groups(degrees, limit):
    mapping = [0] * len(degrees)
    loads = [0]
    group = 0
    for i, d in enumerate(degrees):
        if d and loads[group] and loads[group] + d > limit:
            loads.append(0)
            group += 1
        mapping[i] = group
        loads[group] += d
    return mapping, loads


map_a, load_a = make_groups(deg_a, delta)
map_b, load_b = make_groups(deg_b, delta)

k = max(len(load_a), len(load_b))
load_a.extend([0] * (k - len(load_a)))
load_b.extend([0] * (k - len(load_b)))

edge_u = array('i', (map_a[u] for u in orig_u))
edge_v = array('i', (map_b[v] for v in orig_v))

del orig_u, orig_v, deg_a, deg_b, map_a, map_b

def_a = [delta - x for x in load_a]
def_b = [delta - x for x in load_b]

i = j = 0
while i < k and j < k:
    while i < k and def_a[i] == 0:
        i += 1
    while j < k and def_b[j] == 0:
        j += 1
    if i == k or j == k:
        break

    take = min(def_a[i], def_b[j])
    if take:
        edge_u.extend(array('i', [i]) * take)
        edge_v.extend(array('i', [j]) * take)
        def_a[i] -= take
        def_b[j] -= take

del load_a, load_b, def_a, def_b

total_edges = len(edge_u)
answers = array('i', [0]) * m
marks = array('I', [0]) * total_edges
stamp = 0


def perfect_matching(edges):
    adjacency = [[] for _ in range(k)]
    for eid in edges:
        adjacency[edge_u[eid]].append(eid)

    pair_left = [-1] * k
    pair_right = [-1] * k
    pair_edge = [-1] * k
    dist = [-1] * k

    def dfs(u):
        nd = dist[u] + 1
        for eid in adjacency[u]:
            v = edge_v[eid]
            other = pair_right[v]
            if other == -1 or (dist[other] == nd and dfs(other)):
                pair_left[u] = v
                pair_right[v] = u
                pair_edge[u] = eid
                return True
        dist[u] = -1
        return False

    matched = 0
    while matched < k:
        q = deque()
        for u in range(k):
            if pair_left[u] == -1:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = -1

        while q:
            u = q.popleft()
            nd = dist[u] + 1
            for eid in adjacency[u]:
                other = pair_right[edge_v[eid]]
                if other != -1 and dist[other] == -1:
                    dist[other] = nd
                    q.append(other)

        added = 0
        for u in range(k):
            if pair_left[u] == -1 and dfs(u):
                added += 1
        if added == 0:
            break
        matched += added

    return pair_edge


def split_even(edges):
    global stamp
    stamp += 1
    current_stamp = stamp

    adjacency = [[] for _ in range(2 * k)]
    for eid in edges:
        u = edge_u[eid]
        v = k + edge_v[eid]
        adjacency[u].append(eid)
        adjacency[v].append(eid)

    first = []
    second = []

    for start in range(2 * k):
        while adjacency[start] and marks[adjacency[start][-1]] == current_stamp:
            adjacency[start].pop()
        if not adjacency[start]:
            continue

        stack_vertices = [start]
        stack_edges = []
        parity = 0

        while stack_vertices:
            vtx = stack_vertices[-1]
            adj = adjacency[vtx]
            while adj and marks[adj[-1]] == current_stamp:
                adj.pop()

            if adj:
                eid = adj.pop()
                if marks[eid] == current_stamp:
                    continue
                marks[eid] = current_stamp
                if vtx < k:
                    nxt = k + edge_v[eid]
                else:
                    nxt = edge_u[eid]
                stack_vertices.append(nxt)
                stack_edges.append(eid)
            else:
                stack_vertices.pop()
                if stack_edges:
                    eid = stack_edges.pop()
                    if parity == 0:
                        first.append(eid)
                    else:
                        second.append(eid)
                    parity ^= 1

    return first, second


stack = [(list(range(total_edges)), delta)]
color = 0

while stack:
    edges, degree = stack.pop()

    if degree == 1:
        color += 1
        for eid in edges:
            if eid < m:
                answers[eid] = color
        continue

    if degree & 1:
        matched_edges = perfect_matching(edges)
        stamp += 1
        current_stamp = stamp
        for eid in matched_edges:
            marks[eid] = current_stamp

        remainder = []
        matching = []
        for eid in edges:
            if marks[eid] == current_stamp:
                matching.append(eid)
            else:
                remainder.append(eid)

        stack.append((remainder, degree - 1))
        stack.append((matching, 1))
    else:
        left, right = split_even(edges)
        half = degree // 2
        stack.append((right, half))
        stack.append((left, half))

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answers)))
sys.stdout.write("\n")
