import sys
import random

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    NS = int(input_data[0])
    NG = int(input_data[1])
    M = int(input_data[2])
    
    if M == 0:
        print("YES")
        print()
        return

    deg_u = [0] * (NS + 1)
    deg_v = [0] * (NG + 1)
    
    edges_raw = []
    idx = 3
    for i in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        deg_u[u] += 1
        deg_v[v] += 1
        edges_raw.append((u, v))
        
    D = max(max(deg_u), max(deg_v))
    if D == 0:
        print("YES")
        print()
        return

    # Merge vertices in U into bins of capacity <= D
    map_U = {}
    curr_bin = 0
    curr_deg = 0
    for u in range(1, NS + 1):
        if deg_u[u] == 0:
            continue
        if curr_deg + deg_u[u] > D:
            curr_bin += 1
            curr_deg = 0
        map_U[u] = curr_bin
        curr_deg += deg_u[u]
    K_U = curr_bin + 1 if map_U else 0

    # Merge vertices in V into bins of capacity <= D
    map_V = {}
    curr_bin = 0
    curr_deg = 0
    for v in range(1, NG + 1):
        if deg_v[v] == 0:
            continue
        if curr_deg + deg_v[v] > D:
            curr_bin += 1
            curr_deg = 0
        map_V[v] = curr_bin
        curr_deg += deg_v[v]
    K_V = curr_bin + 1 if map_V else 0

    K = max(K_U, K_V, 1)

    # Build initial edge list with bin indices
    edge_list = []
    bin_deg_U = [0] * K
    bin_deg_V = [0] * K
    for eid, (u, v) in enumerate(edges_raw):
        bu = map_U[u]
        bv = map_V[v]
        edge_list.append((bu, bv, eid))
        bin_deg_U[bu] += 1
        bin_deg_V[bv] += 1

    # Add dummy edges to make the merged graph D-regular
    def_U = [D - bin_deg_U[i] for i in range(K)]
    def_V = [D - bin_deg_V[i] for i in range(K)]
    
    p_U = 0
    p_V = 0
    while p_U < K and p_V < K:
        while p_U < K and def_U[p_U] == 0:
            p_U += 1
        while p_V < K and def_V[p_V] == 0:
            p_V += 1
        if p_U == K or p_V == K:
            break
        take = min(def_U[p_U], def_V[p_V])
        for _ in range(take):
            edge_list.append((p_U, p_V, -1))
        def_U[p_U] -= take
        def_V[p_V] -= take

    ans = [0] * M

    def decompose(edges, cur_D, base_color):
        if cur_D == 0:
            return
        if cur_D == 1:
            for u, v, eid in edges:
                if eid != -1:
                    ans[eid] = base_color
            return

        if cur_D % 2 == 1:
            # Find a 1-factor using Goel-Kapralov-Khanna random walk algorithm
            adj_U = [[] for _ in range(K)]
            for internal_idx, edge in enumerate(edges):
                u, v, eid = edge
                adj_U[u].append((v, internal_idx))

            match_U = [-1] * K
            match_V = [-1] * K
            match_edge_idx = [-1] * K

            for u in range(K):
                if match_U[u] != -1:
                    continue
                curr = u
                while curr != -1:
                    v, e_idx = random.choice(adj_U[curr])
                    next_u = match_V[v]
                    if next_u != -1:
                        match_U[next_u] = -1
                    match_V[v] = curr
                    match_U[curr] = v
                    match_edge_idx[curr] = e_idx
                    curr = next_u

            in_matching = [False] * len(edges)
            for u in range(K):
                e_idx = match_edge_idx[u]
                in_matching[e_idx] = True
                u_node, v_node, eid = edges[e_idx]
                if eid != -1:
                    ans[eid] = base_color

            remaining_edges = [edges[i] for i in range(len(edges)) if not in_matching[i]]
            decompose(remaining_edges, cur_D - 1, base_color + 1)
        else:
            # cur_D is even: decompose into two cur_D // 2 regular graphs
            adj = [[] for _ in range(2 * K)]
            for idx_e, (u, v, eid) in enumerate(edges):
                adj[u].append((K + v, idx_e))
                adj[K + v].append((u, idx_e))

            used = [False] * len(edges)
            ptr = [0] * (2 * K)
            E1 = []
            E2 = []

            for start in range(2 * K):
                while ptr[start] < len(adj[start]):
                    if used[adj[start][ptr[start]][1]]:
                        ptr[start] += 1
                        continue
                    
                    curr = start
                    cycle_edges = []
                    while True:
                        while ptr[curr] < len(adj[curr]) and used[adj[curr][ptr[curr]][1]]:
                            ptr[curr] += 1
                        if ptr[curr] == len(adj[curr]):
                            break
                        nxt, e_idx = adj[curr][ptr[curr]]
                        ptr[curr] += 1
                        used[e_idx] = True
                        cycle_edges.append(e_idx)
                        curr = nxt
                        if curr == start:
                            break
                    
                    for j, e_idx in enumerate(cycle_edges):
                        if j % 2 == 0:
                            E1.append(edges[e_idx])
                        else:
                            E2.append(edges[e_idx])

            decompose(E1, cur_D // 2, base_color)
            decompose(E2, cur_D // 2, base_color + cur_D // 2)

    random.seed(42)
    decompose(edge_list, D, 1)

    print("YES")
    print(*(ans[i] for i in range(M)))

if __name__ == '__main__':
    solve()
