import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    nA = int(input_data[0])
    nB = int(input_data[1])
    m = int(input_data[2])
    
    if m == 0:
        print("YES")
        print()
        return

    V = nA + nB
    edge_at = [{} for _ in range(V + 1)]
    first_free = [1] * (V + 1)
    ans = [0] * m

    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1]) + nA
        idx += 2
        
        c1 = first_free[u]
        eu = edge_at[u]
        while c1 in eu:
            c1 += 1
        first_free[u] = c1
        
        c2 = first_free[v]
        ev = edge_at[v]
        while c2 in ev:
            c2 += 1
        first_free[v] = c2
        
        if c1 == c2:
            col = c1
        else:
            curr = v
            c = c1
            other_c = c2
            path = []
            while c in edge_at[curr]:
                nxt, e_idx = edge_at[curr][c]
                path.append((curr, nxt, e_idx, c, other_c))
                curr = nxt
                c, other_c = other_c, c
            
            for p_u, p_v, e_idx, old_c, new_c in path:
                del edge_at[p_u][old_c]
                del edge_at[p_v][old_c]
            
            for p_u, p_v, e_idx, old_c, new_c in path:
                edge_at[p_u][new_c] = (p_v, e_idx)
                edge_at[p_v][new_c] = (p_u, e_idx)
                ans[e_idx] = new_c
                if old_c < first_free[p_u]:
                    first_free[p_u] = old_c
                if old_c < first_free[p_v]:
                    first_free[p_v] = old_c
            
            col = c1
        
        edge_at[u][col] = (v, i)
        edge_at[v][col] = (u, i)
        ans[i] = col

    print("YES")
    print(*(ans[i] for i in range(m)))

if __name__ == '__main__':
    solve()
