```python
import sys
from collections import defaultdict
import heapq

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    nL, nR, m = map(int, sys.stdin.readline().split())
    
    # We'll store the graph as adjacency lists
    adjL = defaultdict(list)  # adjL[u] = list of (v, idx) connected to L-port u
    adjR = defaultdict(list)  # adjR[v] = list of (u, idx) connected to R-port v
    
    edges = []
    
    for i in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adjL[u].append((v, i))
        adjR[v].append((u, i))
        edges.append((u, v))
    
    # Find maximum degree
    max_deg = 0
    for u in range(1, nL + 1):
        max_deg = max(max_deg, len(adjL[u]))
    for v in range(1, nR + 1):
        max_deg = max(max_deg, len(adjR[v]))
    
    if max_deg == 0:
        print("YES")
        return
    
    # Color the edges: we need at most max_deg colors
    # This is edge coloring in a bipartite graph, which can be done with max_deg colors
    
    # For each node, we keep track of which colors are used on its edges
    color_used_L = defaultdict(set)  # color_used_L[u] = set of colors used by edges of u
    color_used_R = defaultdict(set)  # color_used_R[v] = set of colors used by edges of v
    
    result = [0] * m  # result[i] = color of edge i (1-based)
    
    # For each L-node, process its edges
    for u in range(1, nL + 1):
        for v, idx in adjL[u]:
            # Find a color not used by u and not used by v
            color = 1
            while color in color_used_L[u] or color in color_used_R[v]:
                color += 1
            result[idx] = color
            color_used_L[u].add(color)
            color_used_R[v].add(color)
    
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
```
