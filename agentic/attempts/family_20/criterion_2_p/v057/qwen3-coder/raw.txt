```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    nA = int(data[0])
    nB = int(data[1])
    m = int(data[2])
    
    edges = []
    degree_A = [0] * (nA + 1)
    degree_B = [0] * (nB + 1)
    
    idx = 3
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx+1])
        edges.append((u, v))
        degree_A[u] += 1
        degree_B[v] += 1
        idx += 2
    
    D = max(max(degree_A), max(degree_B))
    
    # Build adjacency list for graph matching
    adj = defaultdict(list)
    for i, (u, v) in enumerate(edges):
        adj[u].append((v, i))
    
    # Assign colors (windows) using greedy coloring with D colors
    edge_color = [0] * m
    color_used_A = [set() for _ in range(nA + 1)]
    color_used_B = [set() for _ in range(nB + 1)]
    
    for i, (u, v) in enumerate(edges):
        # Find a color not used in both endpoints
        used = color_used_A[u] | color_used_B[v]
        color = 1
        while color in used:
            color += 1
        edge_color[i] = color
        color_used_A[u].add(color)
        color_used_B[v].add(color)
    
    print("YES")
    print(' '.join(map(str, edge_color)))

if __name__ == "__main__":
    main()
```
