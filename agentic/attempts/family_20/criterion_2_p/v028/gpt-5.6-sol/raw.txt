import sys
from collections import deque

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
if not data:
    sys.exit()

n_left, n_right, m = data[0], data[1], data[2]
U = [0] * m
V = [0] * m

deg_left = [0] * n_left
deg_right = [0] * n_right

p = 3
for i in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2
    U[i] = u
    V[i] = v
    deg_left[u] += 1
    deg_right[v] += 1

if m == 0:
    sys.stdout.write("YES\n\n")
    sys.exit()

Delta = max(max(deg_left), max(deg_right))
answer = [0] * m


def balanced_split(edges):
    k = len(edges)
    adj = {}
    A = [0] * k
    B = [0] * k
    odd_left = []
    odd_right = []

    for j, eid in enumerate(edges):
        a = U[eid]
        b = n_left + V[eid]
        A[j] = a
        B[j] = b
        adj.setdefault(a, []).append(j)
        adj.setdefault(b, []).append(j)

    for vertex, incident in adj.items():
        if len(incident) & 1:
            if vertex < n_left:
                odd_left.append(vertex)
            else:
                odd_right.append(vertex)

    common = min(len(odd_left), len(odd_right))
    for i in range(common):
        a = odd_left[i]
        b = odd_right[i]
        j = len(A)
        A.append(a)
        B.append(b)
        adj[a].append(j)
        adj[b].append(j)

    next_virtual = n_left + n_right

    i = common
    while i < len(odd_left):
        virtual = next_virtual
        next_virtual += 1
        a1 = odd_left[i]
        a2 = odd_left[i + 1]

        j = len(A)
        A.append(a1)
        B.append(virtual)
        adj[a1].append(j)
        adj.setdefault(virtual, []).append(j)

        j = len(A)
        A.append(a2)
        B.append(virtual)
        adj[a2].append(j)
        adj[virtual].append(j)
        i += 2

    i = common
    while i < len(odd_right):
        virtual = next_virtual
        next_virtual += 1
        b1 = odd_right[i]
        b2 = odd_right[i + 1]

        j = len(A)
        A.append(virtual)
        B.append(b1)
        adj.setdefault(virtual, []).append(j)
        adj[b1].append(j)

        j = len(A)
        A.append(virtual)
        B.append(b2)
        adj[virtual].append(j)
        adj[b2].append(j)
        i += 2

    used = bytearray(len(A))
    first = []
    second = []

    for start, incident in adj.items():
        while incident and used[incident[-1]]:
            incident.pop()
        if not incident:
            continue

        stack_v = [start]
        stack_e = []
        circuit = []

        while stack_v:
            x = stack_v[-1]
            lst = adj[x]
            while lst and used[lst[-1]]:
                lst.pop()

            if lst:
                e = lst.pop()
                if used[e]:
                    continue
                used[e] = 1
                y = B[e] if A[e] == x else A[e]
                stack_v.append(y)
                stack_e.append(e)
            else:
                stack_v.pop()
                if stack_e:
                    circuit.append(stack_e.pop())

        for parity, local_e in enumerate(circuit):
            if local_e < k:
                if parity & 1:
                    second.append(edges[local_e])
                else:
                    first.append(edges[local_e])

    return first, second


def covering_matching(edges, D):
    adj_left = {}
    adj_right = {}

    for eid in edges:
        u = U[eid]
        v = V[eid]
        adj_left.setdefault(u, []).append(eid)
        adj_right.setdefault(v, []).append(eid)

    required_left = [u for u, es in adj_left.items() if len(es) == D]
    required_right = [v for v, es in adj_right.items() if len(es) == D]
    required_right_set = set(required_right)

    match_left = {}
    match_right = {}

    dist = {}

    def bfs():
        q = deque()
        found = False
        for u in required_left:
            if u not in match_left:
                dist[u] = 0
                q.append(u)
            else:
                dist[u] = -1

        while q:
            u = q.popleft()
            nd = dist[u] + 1
            for eid in adj_left[u]:
                v = V[eid]
                old = match_right.get(v)
                if old is None:
                    found = True
                else:
                    u2 = old[0]
                    if dist.get(u2, -1) == -1:
                        dist[u2] = nd
                        q.append(u2)
        return found

    phase_pos = {}

    def dfs_left(u):
        es = adj_left[u]
        i = phase_pos.get(u, 0)
        while i < len(es):
            eid = es[i]
            i += 1
            phase_pos[u] = i
            v = V[eid]
            old = match_right.get(v)

            if old is None:
                match_left[u] = (v, eid)
                match_right[v] = (u, eid)
                return True

            u2 = old[0]
            if dist.get(u2, -1) == dist[u] + 1 and dfs_left(u2):
                match_left[u] = (v, eid)
                match_right[v] = (u, eid)
                return True

        dist[u] = -1
        return False

    while bfs():
        phase_pos.clear()
        for u in required_left:
            if u not in match_left:
                dfs_left(u)

    seen_right = {}
    stamp = 0

    def relocate_right(r):
        seen_right[r] = stamp
        for eid in adj_right[r]:
            u = U[eid]
            old = match_left.get(u)

            if old is None:
                match_left[u] = (r, eid)
                match_right[r] = (u, eid)
                return True

            old_r = old[0]
            if old_r == r:
                return True

            if old_r not in required_right_set:
                match_right.pop(old_r, None)
                match_left[u] = (r, eid)
                match_right[r] = (u, eid)
                return True

            if seen_right.get(old_r) != stamp and relocate_right(old_r):
                match_left[u] = (r, eid)
                match_right[r] = (u, eid)
                return True

        return False

    for r in required_right:
        if r not in match_right:
            stamp += 1
            relocate_right(r)

    return [item[1] for item in match_left.values()]


def solve(edges, D, first_color):
    if not edges:
        return

    if D == 1:
        for eid in edges:
            answer[eid] = first_color
        return

    if D & 1:
        matching = covering_matching(edges, D)
        matched = set(matching)

        for eid in matching:
            answer[eid] = first_color + D - 1

        remaining = [eid for eid in edges if eid not in matched]

        if D == 3:
            a, b = balanced_split(remaining)
            for eid in a:
                answer[eid] = first_color
            for eid in b:
                answer[eid] = first_color + 1
        else:
            solve(remaining, D - 1, first_color)
    else:
        a, b = balanced_split(edges)
        half = D // 2

        if half == 1:
            for eid in a:
                answer[eid] = first_color
            for eid in b:
                answer[eid] = first_color + 1
        else:
            solve(a, half, first_color)
            solve(b, half, first_color + half)


solve(list(range(m)), Delta, 1)

sys.stdout.write("YES\n")
sys.stdout.write(" ".join(map(str, answer)))
sys.stdout.write("\n")
