    # Use dict for sparse representation

        u, v = edges[ei]
        
        # Find a free color at u
        a = next_free_v[u]
        while a in vessel_color[u]:
            a += 1
        next_free_v[u] = a
        
        # Find a free color at v
        b = next_free_d[v]
        while b in dock_color[v]:
            b += 1
        next_free_d[v] = b
        
        if a == b:
            ans[ei] = a
            vessel_color[u][a] = ei
            dock_color[v][a] = ei
        elif a not in dock_color[v]:
            # a is free at both u and v
            ans[ei] = a
            vessel_color[u][a] = ei
            dock_color[v][a] = ei
            if next_free_d[v] == a:
                next_free_d[v] = a + 1

            # Neither a is free at v, nor b is free at u

            # We want to use color a for edge (u,v)
