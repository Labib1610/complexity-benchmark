```python
import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    nV = int(next(it))
    nD = int(next(it))
    m = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(m)]

    degV = [0] * nV
    degD = [0] * nD
    for u, v in edges:
        degV[u] += 1
        degD[v] += 1
    Delta = max(max(degV, default=0), max(degD, default=0))

    # for each vertex store: color -> neighbor vertex, and set of used colors
    usedV = [dict() for _ in range(nV)]
    usedD = [dict() for _ in range(nD)]

    # result colors per edge (1‑based)
    res = [0] * m

    # helper to find a free color at a vertex
    def free_color(used_dict):
        # used_dict size = degree, small on average
        for c in range(1, Delta + 1):
            if c not in used_dict:
                return c
        return None  # should never happen

    # BFS to find alternating path between colors a (missing at u) and b (missing at v)
    from collections import deque
    def recolor(u_start, v_start, ca, cb):
        # BFS over vertices, storing predecessor and edge info
        # vertices are split into two sides, we keep side info by sign
        # positive id for vessels (0..nV-1), negative for docks (-1..-nD)
        q = deque()
        prev = {}          # vertex -> (prev_vertex, edge_id, old_color)
        q.append(u_start)
        prev[u_start] = None

        target = None
        while q:
            x = q.popleft()
            if x >= 0:          # vessel side
                used = usedV[x]
                # we can move via edge colored ca
                if ca in used:
                    y = used[ca]          # dock neighbor
                    if y not in prev:
                        prev[y] = (x, None, ca)   # edge id unknown, just color
                        q.append(y)
                else:
                    # ca is free here -> we reached a free spot
                    target = x
                    break
            else:               # dock side, store as negative index
                d = -x - 1
                used = usedD[d]
                # move via edge colored cb
                if cb in used:
                    y = used[cb]          # vessel neighbor
                    if y not in prev:
                        prev[y] = (x, None, cb)
                        q.append(y)
                else:
                    target = x
                    break

        # reconstruct and flip colors along the path
        cur = target
        while prev[cur] is not None:
            p, _, col = prev[cur]
            # edge between p and cur has color col, we need to swap ca<->cb along path
            if p >= 0:
                # p vessel, cur dock
                dv = p
                dd = -cur - 1
                # remove old mapping
                if col == ca:
                    del usedV[dv][ca]
                    del usedD[dd][ca]
                    # add cb
                    usedV[dv][cb] = dd
                    usedD[dd][cb] = dv
                else:  # col == cb
                    del usedV[dv][cb]
                    del usedD[dd][cb]
                    usedV[dv][ca] = dd
                    usedD[dd][ca] = dv
            else:
                # p dock, cur vessel
                dd = -p - 1
                dv = cur
                if col == ca:
                    del usedD[dd][ca]
                    del usedV[dv][ca]
                    usedD[dd][cb] = dv
                    usedV[dv][cb] = dd
                else:
                    del usedD[dd][cb]
                    del usedV[dv][cb]
                    usedD[dd][ca] = dv
                    usedV[dv][ca] = dd
            cur = p

    for idx, (u, v) in enumerate(edges):
        if v in usedV[u]:
            # already colored (should not happen)
            continue
        # try to assign a free common color
        cu = free_color(usedV[u])
        cv = free_color(usedD[v])
        if cu == cv:
            # both free same color
            col = cu
            usedV[u][col] = v
            usedD[v][col] = u
            res[idx] = col
            continue
        # otherwise we need recoloring
        # we will try to make cu free at v via alternating path
        # ensure cu is missing at v after recolor
        recolor(u, v, cu, cv)
        # now cu is free at both ends
        col = cu
        usedV[u][col] = v
        usedD[v][col] = u
        res[idx] = col

    out = ['YES', ' '.join(str(x) for x in res)]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
```
