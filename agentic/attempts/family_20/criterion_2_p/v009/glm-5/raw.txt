```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    
    if m == 0:
        sys.stdout.write("YES\n\n")
        return
    
    degL = [0] * (nL + 2)
    degR = [0] * (nR + 2)
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
        degL[u] += 1
        degR[v] += 1
    
    delta = 0
    for i in range(1, nL + 1):
        if degL[i] > delta:
            delta = degL[i]
    for i in range(1, nR + 1):
        if degR[i] > delta:
            delta = degR[i]
    
    if delta == 0:
        sys.stdout.write("YES\n\n")
        return
    
    # Build adjacency lists for left and right ports
    # Each port's incident edges stored as list of edge indices
    headL = [-1] * (nL + 2)
    headR = [-1] * (nR + 2)
    nxtL = [-1] * m
    nxtR = [-1] * m
    
    for i in range(m):
        u, v = edges[i]
        nxtL[i] = headL[u]
        headL[u] = i
        nxtR[i] = headR[v]
        headR[v] = i
    
    color = [0] * m
    used = [False] * (delta + 1)
    
    # Process each edge in input order
    for i in range(m):
        u, v = edges[i]
        
        # Mark colors used by adjacent edges on left port u
        e = headL[u]
        while e != -1:
            if e != i:
                c = color[e]
                if c:
                    used[c] = True
            e = nxtL[e]
        
        # Mark colors used by adjacent edges on right port v
        e = headR[v]
        while e != -1:
            if e != i:
                c = color[e]
                if c:
                    used[c] = True
            e = nxtR[e]
        
        # Find smallest available color
        c = 1
        while c <= delta and used[c]:
            c += 1
        
        color[i] = c
        
        # Reset used array
        e = headL[u]
        while e != -1:
            if e != i:
                c2 = color[e]
                if c2:
                    used[c2] = False
            e = nxtL[e]
        
        e = headR[v]
        while e != -1:
            if e != i:
                c2 = color[e]
                if c2:
                    used[c2] = False
            e = nxtR[e]
    
    out = ["YES"]
    out.append(" ".join(map(str, color)))
    sys.stdout.write("\n".join(out) + "\n")

main()
```
