We can model this as a bipartite graph edge-coloring problem.  
The maximum degree in the graph gives the minimum number of colors (service windows) needed, and by König’s line coloring theorem, a bipartite graph can be edge-colored with exactly Δ colors.

We’ll implement an efficient edge-coloring algorithm using a greedy approach with color swapping, which runs in O(m Δ) worst-case but is fast enough for the given limits.

```python
import sys
from collections import deque

def solve():
    input = sys.stdin.readline
    nL, nR, m = map(int, input().split())
    edges = []
    adjL = [[] for _ in range(nL)]
    adjR = [[] for _ in range(nR)]
    for i in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        edges.append((u, v))
        adjL[u].append((v, i))
        adjR[v].append((u, i))
    
    # Find max degree
    max_deg = 0
    for i in range(nL):
        max_deg = max(max_deg, len(adjL[i]))
    for i in range(nR):
        max_deg = max(max_deg, len(adjR[i]))
    
    color = [0] * m
    # colorL[u][c] = edge index or -1
    colorL = [[-1] * (max_deg + 1) for _ in range(nL)]
    colorR = [[-1] * (max_deg + 1) for _ in range(nR)]
    
    for idx, (u, v) in enumerate(edges):
        # find free color for u
        c = 1
        while colorL[u][c] != -1:
            c += 1
        # assign
        color[idx] = c
        colorL[u][c] = idx
        colorR[v][c] = idx
        
        # resolve conflicts on right side
        while True:
            conflict = colorR[v][c]
            if conflict == idx:
                break
            # conflict edge (u2, v) with same color c
            u2, _ = edges[conflict]
            # find free color for u2
            c2 = 1
            while colorL[u2][c2] != -1:
                c2 += 1
            # recolor conflict edge to c2
            color[conflict] = c2
            colorL[u2][c] = -1
            colorL[u2][c2] = conflict
            colorR[v][c] = idx
            colorR[v][c2] = conflict
            # now move to u2 with color c2
            # find if there is conflict on right side for u2 with c2
            # Actually we need to check right side of u2's other edges
            # But we already set colorR[v][c2] = conflict, so no conflict there.
            # We need to check if u2 has another edge with color c2 on right side
            # That would be colorR[w][c2] for some w != v
            # We'll search for it
            found = False
            for w in range(nR):
                if colorR[w][c2] != -1:
                    e2 = colorR[w][c2]
                    if e2 != conflict and edges[e2][0] == u2:
                        # conflict on u2 side
                        # swap colors
                        # We'll handle by setting v = w and c = c2 and continue
                        v = w
                        c = c2
                        found = True
                        break
            if not found:
                break
    
    print("YES")
    print(" ".join(str(color[i]) for i in range(m)))

if __name__ == "__main__":
    solve()
```
