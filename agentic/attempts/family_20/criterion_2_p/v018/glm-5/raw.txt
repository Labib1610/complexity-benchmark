import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    nL = int(next(it))
    nR = int(next(it))
    m = int(next(it))
    
    edges = []
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        edges.append((u, v))
    
    # Calculate degrees
    degL = [0] * (nL + 1)
    degR = [0] * (nR + 1)
    for u, v in edges:
        degL[u] += 1
        degR[v] += 1
    
    max_load = 0
    if nL > 0:
        max_load = max(max_load, max(degL))
    if nR > 0:
        max_load = max(max_load, max(degR))
    
    if m == 0:
        print("YES")
        print("")
        return
    
    # Build adjacency lists
    adjL = [[] for _ in range(nL + 1)]
    adjR = [[] for _ in range(nR + 1)]
    for i, (u, v) in enumerate(edges):
        adjL[u].append((v, i))
        adjR[v].append((u, i))
    
    # Initialize next available color for each port
    next_color_L = [1] * (nL + 1)
    next_color_R = [1] * (nR + 1)
    
    # Initialize color assignment
    colors = [0] * m
    
    # Process edges
    for i, (u, v) in enumerate(edges):
        c1 = next_color_L[u]
        c2 = next_color_R[v]
        
        if c1 == c2:
            colors[i] = c1
            next_color_L[u] += 1
            next_color_R[v] += 1
        else:
            # Choose the smaller color to minimize gaps
            if c1 < c2:
                colors[i] = c1
                next_color_L[u] += 1
            else:
                colors[i] = c2
                next_color_R[v] += 1
    
    print("YES")
    print(' '.join(map(str, colors)))

if __name__ == "__main__":
    solve()
