import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n_L = int(input_data[0])
    n_R = int(input_data[1])
    m = int(input_data[2])
    
    ans = [0] * m
    if m == 0:
        print("YES")
        print()
        return

    color = [{} for _ in range(n_L + n_R + 1)]
    
    idx = 3
    for e in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1]) + n_L
        idx += 2
        
        cu = color[u]
        cv = color[v]
        
        c1 = 1
        while c1 in cu:
            c1 += 1
            
        if c1 not in cv:
            cu[c1] = (v, e)
            cv[c1] = (u, e)
            ans[e] = c1
        else:
            c2 = 1
            while c2 in cv:
                c2 += 1
                
            path = [v]
            curr = v
            c = c1
            while c in color[curr]:
                nxt, _ = color[curr][c]
                path.append(nxt)
                curr = nxt
                c = c2 if c == c1 else c1
            
            edges_in_path = []
            for i in range(len(path) - 1):
                p1 = path[i]
                p2 = path[i + 1]
                c_old = c1 if i % 2 == 0 else c2
                _, eid = color[p1][c_old]
                edges_in_path.append(eid)
                del color[p1][c_old]
                del color[p2][c_old]
                
            for i in range(len(path) - 1):
                p1 = path[i]
                p2 = path[i + 1]
                c_new = c2 if i % 2 == 0 else c1
                eid = edges_in_path[i]
                color[p1][c_new] = (p2, eid)
                color[p2][c_new] = (p1, eid)
                ans[eid] = c_new
                
            cu[c1] = (v, e)
            cv[c1] = (u, e)
            ans[e] = c1

    print("YES")
    print(*(ans[i] for i in range(m)))

if __name__ == '__main__':
    solve()
