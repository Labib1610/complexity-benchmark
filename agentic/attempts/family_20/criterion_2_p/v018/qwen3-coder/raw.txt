    right_ports = [[] for _ in range(n_R + 1)]

    # Assign windows to edges
    result = [0] * m

    # For each edge, assign a window that's available on both ends
    for i in range(m):

        # To avoid complex heap operations, let's track used windows per port separately
        # Actually, let's re-think the approach

    # Reset approach: use greedy assignment with tracking of used windows per port
