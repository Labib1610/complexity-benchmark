import sys
import random

# Increase recursion depth just in case
sys.setrecursionlimit(200000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N_emp = int(input_data[0])
    N_ven = int(input_data[1])
    M = int(input_data[2])
    
    if M == 0:
        sys.stdout.write("YES\n\n")
        return

    edge_u = [0] * M
    edge_v = [0] * M
    deg_L = [0] * (N_emp + 1)
    deg_R = [0] * (N_ven + 1)
    
    idx = 3
    for i in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edge_u[i] = u
        edge_v[i] = v
        deg_L[u] += 1
        deg_R[v] += 1
        
    F = max(max(deg_L), max(deg_R))
    if F == 0:
        sys.stdout.write("YES\n\n")
        return

    # Binning Left vertices
    bin_L = [0] * (N_emp + 1)
    bins_L = []
    cur_bin = []
    cur_sum = 0
    for u in range(1, N_emp + 1):
        if deg_L[u] == 0:
            continue
        if cur_sum + deg_L[u] > F:
            bins_L.append(cur_bin)
            cur_bin = [u]
            cur_sum = deg_L[u]
        else:
            cur_bin.append(u)
            cur_sum += deg_L[u]
    if cur_bin:
        bins_L.append(cur_bin)

    for b_id, b in enumerate(bins_L):
        for u in b:
            bin_L[u] = b_id

    # Binning Right vertices
    bin_R = [0] * (N_ven + 1)
    bins_R = []
    cur_bin = []
    cur_sum = 0
    for v in range(1, N_ven + 1):
        if deg_R[v] == 0:
            continue
        if cur_sum + deg_R[v] > F:
            bins_R.append(cur_bin)
            cur_bin = [v]
            cur_sum = deg_R[v]
        else:
            cur_bin.append(v)
            cur_sum += deg_R[v]
    if cur_bin:
        bins_R.append(cur_bin)

    for b_id, b in enumerate(bins_R):
        for v in b:
            bin_R[v] = b_id

    B = max(len(bins_L), len(bins_R), 1)

    deg_bin_L = [0] * B
    deg_bin_R = [0] * B

    for i in range(M):
        bu = bin_L[edge_u[i]]
        bv = bin_R[edge_v[i]]
        edge_u[i] = bu
        edge_v[i] = bv
        deg_bin_L[bu] += 1
        deg_bin_R[bv] += 1

    def_L = [F - deg_bin_L[i] for i in range(B)]
    def_R = [F - deg_bin_R[i] for i in range(B)]

    ptr_L = 0
    ptr_R = 0
    dummy_u = []
    dummy_v = []
    while ptr_L < B and ptr_R < B:
        while ptr_L < B and def_L[ptr_L] == 0:
            ptr_L += 1
        while ptr_R < B and def_R[ptr_R] == 0:
            ptr_R += 1
        if ptr_L < B and ptr_R < B:
            take = min(def_L[ptr_L], def_R[ptr_R])
            for _ in range(take):
                dummy_u.append(ptr_L)
                dummy_v.append(ptr_R)
            def_L[ptr_L] -= take
            def_R[ptr_R] -= take

    num_dummies = len(dummy_u)
    total_edges = M + num_dummies
    edge_u.extend(dummy_u)
    edge_v.extend(dummy_v)

    color = [0] * total_edges
    used = [False] * total_edges

    def solve(edge_list, d, color_offset):
        if d == 1:
            for e in edge_list:
                color[e] = color_offset
            return

        if d % 2 == 1:
            # Find a perfect matching using random walk
            adj = [[] for _ in range(B)]
            for e in edge_list:
                adj[edge_u[e]].append(e)

            match_L = [-1] * B
            match_R = [-1] * B

            for start_u in range(B):
                curr_u = start_u
                while curr_u is not None:
                    e = random.choice(adj[curr_u])
                    v = edge_v[e]
                    prev_e = match_R[v]
                    if prev_e != -1:
                        prev_u = edge_u[prev_e]
                        match_L[prev_u] = -1
                        next_u = prev_u
                    else:
                        next_u = None
                    match_R[v] = e
                    match_L[curr_u] = e
                    curr_u = next_u

            matching_set = set(match_L)
            remaining_edges = []
            for e in edge_list:
                if e in matching_set:
                    color[e] = color_offset
                else:
                    remaining_edges.append(e)

            solve(remaining_edges, d - 1, color_offset + 1)
        else:
            # Even d: Eulerian circuit decomposition
            adj = [[] for _ in range(2 * B)]
            for e in edge_list:
                u = edge_u[e]
                v = edge_v[e] + B
                adj[u].append((v, e))
                adj[v].append((u, e))

            ptr = [0] * (2 * B)
            list1 = []
            list2 = []

            for start_node in range(2 * B):
                if ptr[start_node] < len(adj[start_node]):
                    stack = [start_node]
                    edge_stack = []
                    while stack:
                        u = stack[-1]
                        while ptr[u] < len(adj[u]) and used[adj[u][ptr[u]][1]]:
                            ptr[u] += 1
                        if ptr[u] < len(adj[u]):
                            v, eid = adj[u][ptr[u]]
                            ptr[u] += 1
                            used[eid] = True
                            stack.append(v)
                            edge_stack.append((u, eid))
                        else:
                            stack.pop()
                            if edge_stack:
                                u_from, eid = edge_stack.pop()
                                if u_from < B:
                                    list1.append(eid)
                                else:
                                    list2.append(eid)

            solve(list1, d // 2, color_offset)
            solve(list2, d // 2, color_offset + d // 2)

    all_edges = list(range(total_edges))
    solve(all_edges, F, 1)

    sys.stdout.write("YES\n" + " ".join(map(str, color[:M])) + "\n")

if __name__ == '__main__':
    main()
