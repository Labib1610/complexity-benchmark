import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    Nemp = int(next(it))
    Nven = int(next(it))
    M = int(next(it))
    
    edges = []
    for _ in range(M):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        edges.append((u, v))
    
    deg_u = [0] * Nemp
    deg_v = [0] * Nven
    for u, v in edges:
        deg_u[u] += 1
        deg_v[v] += 1
    
    F = 0
    if M > 0:
        F = max(max(deg_u), max(deg_v))
    
    if M == 0:
        sys.stdout.write("YES\n\n")
        return
    
    # Build bipartite graph
    head = [-1] * Nemp
    to = [0] * (2 * M)
    nxt = [0] * (2 * M)
    edge_id = [0] * (2 * M)
    cnt = 0
    for i, (u, v) in enumerate(edges):
        to[cnt] = v
        nxt[cnt] = head[u]
        edge_id[cnt] = i
        head[u] = cnt
        cnt += 1
        to[cnt] = u
        nxt[cnt] = head[v]
        edge_id[cnt] = i
        cnt += 1
    
    # Find Eulerian trails
    ans = [0] * M
    used = [False] * M
    stack = []
    cur = [0] * (Nemp + Nven)
    for i in range(Nemp + Nven):
        cur[i] = head[i]
    
    for s in range(Nemp + Nven):
        if cur[s] == -1:
            continue
        stack.append(s)
        while stack:
            x = stack[-1]
            while cur[x] != -1 and used[edge_id[cur[x]]]:
                cur[x] = nxt[cur[x]]
            if cur[x] == -1:
                stack.pop()
                continue
            e = cur[x]
            y = to[e]
            used[edge_id[e]] = True
            cur[x] = nxt[e]
            stack.append(y)
            if len(stack) >= 2:
                prev = stack[-2]
                curr = stack[-1]
                if prev < Nemp and curr >= Nemp:
                    ans[edge_id[e]] = 1
                else:
                    ans[edge_id[e]] = 2
    
    out = ['YES']
    out.append(' '.join(map(str, ans)))
    sys.stdout.write('\n'.join(out) + '\n')

solve()
