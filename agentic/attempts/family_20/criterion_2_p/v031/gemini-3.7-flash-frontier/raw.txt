import sys

# Increase recursion depth for deep recursions if any
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    nL = int(input_data[0])
    nR = int(input_data[1])
    m = int(input_data[2])
    
    if m == 0:
        print("YES")
        print()
        return

    edges = []
    deg_L = [0] * (nL + 1)
    deg_R = [0] * (nR + 1)
    
    idx = 3
    for i in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        edges.append((u, v, i))
        deg_L[u] += 1
        deg_R[v] += 1
        
    delta = max(max(deg_L), max(deg_R))
    if delta == 0:
        print("YES")
        print(" ".join(["1"] * m))
        return

    ans = [0] * m

    def regularize(edge_list, d):
        active_deg_L = {}
        active_deg_R = {}
        for u, v, _ in edge_list:
            active_deg_L[u] = active_deg_L.get(u, 0) + 1
            active_deg_R[v] = active_deg_R.get(v, 0) + 1

        bins_L = []
        curr_bin = []
        curr_deg = 0
        for u, deg in active_deg_L.items():
            if curr_deg + deg > d:
                bins_L.append(curr_bin)
                curr_bin = [u]
                curr_deg = deg
            else:
                curr_bin.append(u)
                curr_deg += deg
        if curr_bin:
            bins_L.append(curr_bin)

        bins_R = []
        curr_bin = []
        curr_deg = 0
        for v, deg in active_deg_R.items():
            if curr_deg + deg > d:
                bins_R.append(curr_bin)
                curr_bin = [v]
                curr_deg = deg
            else:
                curr_bin.append(v)
                curr_deg += deg
        if curr_bin:
            bins_R.append(curr_bin)

        while len(bins_L) < len(bins_R):
            bins_L.append([])
        while len(bins_R) < len(bins_L):
            bins_R.append([])

        K = len(bins_L)
        bin_of_L = {}
        bin_deg_L = [0] * K
        for b_idx, b in enumerate(bins_L):
            for u in b:
                bin_of_L[u] = b_idx
                bin_deg_L[b_idx] += active_deg_L[u]

        bin_of_R = {}
        bin_deg_R = [0] * K
        for b_idx, b in enumerate(bins_R):
            for v in b:
                bin_of_R[v] = b_idx
                bin_deg_R[b_idx] += active_deg_R[v]

        reg_edges = []
        for u, v, orig_id in edge_list:
            reg_edges.append((bin_of_L[u], bin_of_R[v], orig_id))

        def_L = [d - deg for deg in bin_deg_L]
        def_R = [d - deg for deg in bin_deg_R]
        ptr_L = 0
        ptr_R = 0
        while ptr_L < K and ptr_R < K:
            while ptr_L < K and def_L[ptr_L] == 0:
                ptr_L += 1
            while ptr_R < K and def_R[ptr_R] == 0:
                ptr_R += 1
            if ptr_L < K and ptr_R < K:
                add = min(def_L[ptr_L], def_R[ptr_R])
                for _ in range(add):
                    reg_edges.append((ptr_L, ptr_R, -1))
                def_L[ptr_L] -= add
                def_R[ptr_R] -= add

        return K, reg_edges

    def hopcroft_karp(K, adj):
        pair_u = [-1] * K
        pair_v = [-1] * K
        edge_u = [-1] * K
        dist = [-1] * K

        while True:
            queue = []
            for u in range(K):
                if pair_u[u] == -1:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = -1

            found = False
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                d = dist[u]
                for v, e_idx in adj[u]:
                    next_u = pair_v[v]
                    if next_u == -1:
                        found = True
                    elif dist[next_u] == -1:
                        dist[next_u] = d + 1
                        queue.append(next_u)

            if not found:
                break

            ptr = [0] * K

            def dfs(u):
                d = dist[u]
                for i in range(ptr[u], len(adj[u])):
                    ptr[u] = i + 1
                    v, e_idx = adj[u][i]
                    next_u = pair_v[v]
                    if next_u == -1 or (dist[next_u] == d + 1 and dfs(next_u)):
                        pair_u[u] = v
                        pair_v[v] = u
                        edge_u[u] = e_idx
                        return True
                return False

            for u in range(K):
                if pair_u[u] == -1:
                    dfs(u)

        return edge_u

    def rec_color(edge_list, d, color_offset):
        if not edge_list or d == 0:
            return
        if d == 1:
            for _, _, orig_id in edge_list:
                ans[orig_id] = color_offset + 1
            return

        K, reg_edges = regularize(edge_list, d)

        if d % 2 == 1:
            adj = [[] for _ in range(K)]
            for e_idx, (u, v, _) in enumerate(reg_edges):
                adj[u].append((v, e_idx))

            matched_edge_indices = set(hopcroft_karp(K, adj))
            rem_edges = []
            for e_idx, (u, v, orig_id) in enumerate(reg_edges):
                if e_idx in matched_edge_indices:
                    if orig_id != -1:
                        ans[orig_id] = color_offset + d
                else:
                    if orig_id != -1:
                        rem_edges.append((u, v, orig_id))

            rec_color(rem_edges, d - 1, color_offset)
        else:
            num_v = 2 * K
            adj = [[] for _ in range(num_v)]
            for e_idx, (u, v, _) in enumerate(reg_edges):
                adj[u].append((v + K, e_idx))
                adj[v + K].append((u, e_idx))

            head = [0] * num_v
            used_edge = [False] * len(reg_edges)
            set1 = []
            set2 = []

            for start_node in range(num_v):
                if head[start_node] >= len(adj[start_node]):
                    continue
                stack = [start_node]
                while stack:
                    u = stack[-1]
                    while head[u] < len(adj[u]) and used_edge[adj[u][head[u]][1]]:
                        head[u] += 1
                    if head[u] < len(adj[u]):
                        nxt, e_idx = adj[u][head[u]]
                        head[u] += 1
                        used_edge[e_idx] = True
                        stack.append(nxt)
                    else:
                        stack.pop()
                        if stack:
                            prev = stack[-1]
                            # edge is between prev and u
                            # prev is from stack[-1], u is popped
                            # direction traversed: prev -> u
                            if prev < K:  # L -> R
                                orig_id = reg_edges[e_idx][2]
                                if orig_id != -1:
                                    set1.append((prev, u - K, orig_id))
                            else:  # R -> L
                                orig_id = reg_edges[e_idx][2]
                                if orig_id != -1:
                                    set2.append((u, prev - K, orig_id))

            rec_color(set1, d // 2, color_offset)
            rec_color(set2, d // 2, color_offset + d // 2)

    rec_color(edges, delta, 0)

    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == '__main__':
    solve()
