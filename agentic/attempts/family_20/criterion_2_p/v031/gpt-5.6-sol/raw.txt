import sys
from collections import deque

sys.setrecursionlimit(1_500_000)


def make_groups(degrees, limit):
    items = [(d, i) for i, d in enumerate(degrees) if d]
    items.sort()
    group_of = [-1] * len(degrees)
    group_degrees = []

    lo, hi = 0, len(items) - 1
    gid = 0

    while lo <= hi:
        d, v = items[hi]
        hi -= 1
        total = d
        group_of[v] = gid

        while lo <= hi and total + items[lo][0] <= limit:
            d2, v2 = items[lo]
            lo += 1
            total += d2
            group_of[v2] = gid

        group_degrees.append(total)
        gid += 1

    return group_of, group_degrees


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    n_left, n_right, m = data[0], data[1], data[2]

    if m == 0:
        sys.stdout.write("YES\n\n")
        return

    original_left = [0] * m
    original_right = [0] * m
    degree_left = [0] * n_left
    degree_right = [0] * n_right

    p = 3
    for i in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        original_left[i] = u
        original_right[i] = v
        degree_left[u] += 1
        degree_right[v] += 1

    delta = max(max(degree_left), max(degree_right))

    left_group, left_group_degrees = make_groups(degree_left, delta)
    right_group, right_group_degrees = make_groups(degree_right, delta)

    group_count = max(len(left_group_degrees), len(right_group_degrees))
    left_group_degrees.extend([0] * (group_count - len(left_group_degrees)))
    right_group_degrees.extend([0] * (group_count - len(right_group_degrees)))

    edge_left = [0] * m
    edge_right = [0] * m
    for i in range(m):
        edge_left[i] = left_group[original_left[i]]
        edge_right[i] = right_group[original_right[i]]

    left_deficits = []
    right_deficits = []

    for u, d in enumerate(left_group_degrees):
        if d < delta:
            left_deficits.append([u, delta - d])

    for v, d in enumerate(right_group_degrees):
        if d < delta:
            right_deficits.append([v, delta - d])

    i = j = 0
    while i < len(left_deficits):
        u, a = left_deficits[i]
        v, b = right_deficits[j]
        take = a if a < b else b

        edge_left.extend([u] * take)
        edge_right.extend([v] * take)

        a -= take
        b -= take
        left_deficits[i][1] = a
        right_deficits[j][1] = b

        if a == 0:
            i += 1
        if b == 0:
            j += 1

    total_edges = len(edge_left)
    answer = [0] * m
    used_mark = [0] * total_edges
    selected_mark = [0] * total_edges
    used_token = 0
    selected_token = 0
    next_color = 1

    def perfect_matching(edges):
        nonlocal selected_token

        adjacency = [[] for _ in range(group_count)]
        for eid in edges:
            adjacency[edge_left[eid]].append(eid)

        pair_u = [-1] * group_count
        pair_v = [-1] * group_count

        for u in range(group_count):
            for eid in adjacency[u]:
                v = edge_right[eid]
                if pair_v[v] == -1:
                    pair_u[u] = eid
                    pair_v[v] = eid
                    break

        dist = [-1] * group_count

        while True:
            queue = deque()
            found = False

            for u in range(group_count):
                if pair_u[u] == -1:
                    dist[u] = 0
                    queue.append(u)
                else:
                    dist[u] = -1

            while queue:
                u = queue.popleft()
                nd = dist[u] + 1
                for eid in adjacency[u]:
                    v = edge_right[eid]
                    matched = pair_v[v]
                    if matched == -1:
                        found = True
                    else:
                        u2 = edge_left[matched]
                        if dist[u2] == -1:
                            dist[u2] = nd
                            queue.append(u2)

            if not found:
                break

            def dfs(u):
                for eid in adjacency[u]:
                    v = edge_right[eid]
                    matched = pair_v[v]
                    if matched == -1:
                        pair_u[u] = eid
                        pair_v[v] = eid
                        return True
                    u2 = edge_left[matched]
                    if dist[u2] == dist[u] + 1 and dfs(u2):
                        pair_u[u] = eid
                        pair_v[v] = eid
                        return True
                dist[u] = -1
                return False

            for u in range(group_count):
                if pair_u[u] == -1:
                    dfs(u)

        selected_token += 1
        token = selected_token
        matching = pair_u
        for eid in matching:
            selected_mark[eid] = token

        remainder = [eid for eid in edges if selected_mark[eid] != token]
        return matching, remainder

    def split_even(edges):
        nonlocal used_token
        used_token += 1
        token = used_token

        vertex_count = group_count * 2
        adjacency = [[] for _ in range(vertex_count)]

        for eid in edges:
            u = edge_left[eid]
            v = group_count + edge_right[eid]
            adjacency[u].append(eid)
            adjacency[v].append(eid)

        first = []
        second = []

        for start in range(vertex_count):
            adj = adjacency[start]
            while adj and used_mark[adj[-1]] == token:
                adj.pop()
            if not adj:
                continue

            vertex_stack = [start]
            edge_stack = []
            circuit = []

            while vertex_stack:
                vtx = vertex_stack[-1]
                av = adjacency[vtx]

                while av and used_mark[av[-1]] == token:
                    av.pop()

                if av:
                    eid = av.pop()
                    if used_mark[eid] == token:
                        continue
                    used_mark[eid] = token

                    if vtx < group_count:
                        nxt = group_count + edge_right[eid]
                    else:
                        nxt = edge_left[eid]

                    vertex_stack.append(nxt)
                    edge_stack.append(eid)
                else:
                    vertex_stack.pop()
                    if edge_stack:
                        circuit.append(edge_stack.pop())

            first.extend(circuit[0::2])
            second.extend(circuit[1::2])

        return first, second

    stack = [(list(range(total_edges)), delta)]

    while stack:
        edges, degree = stack.pop()

        if degree == 1:
            color = next_color
            next_color += 1
            for eid in edges:
                if eid < m:
                    answer[eid] = color
            continue

        if degree & 1:
            matching, remainder = perfect_matching(edges)
            color = next_color
            next_color += 1

            for eid in matching:
                if eid < m:
                    answer[eid] = color

            stack.append((remainder, degree - 1))
        else:
            first, second = split_even(edges)
            half = degree // 2
            stack.append((second, half))
            stack.append((first, half))

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, answer)) + "\n")


if __name__ == "__main__":
    main()
