import sys
from collections import defaultdict, deque

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    N_in = int(data[idx])
    idx += 1
    N_out = int(data[idx])
    idx += 1
    M = int(data[idx])
    idx += 1
    
    links = []
    inbound_links = defaultdict(list)
    outbound_links = defaultdict(list)
    
    for i in range(M):
        u = int(data[idx]) - 1  # 0-indexed
        idx += 1
        v = int(data[idx]) - 1  # 0-indexed
        idx += 1
        links.append((u, v))
        inbound_links[u].append(i)
        outbound_links[v].append(i)
    
    # Find maximum degree
    max_degree = 0
    for u in inbound_links:
        max_degree = max(max_degree, len(inbound_links[u]))
    for v in outbound_links:
        max_degree = max(max_degree, len(outbound_links[v]))
    
    D = max_degree
    if D == 0:
        print("YES")
        print()
        return
    
    # Initialize assignment
    assignment = [0] * M
    # For each interface, keep track of which cycles are used
    inbound_cycles = defaultdict(set)
    outbound_cycles = defaultdict(set)
    
    # Process links: we'll use a greedy approach
    # For each link, assign the smallest cycle number not used by its interfaces
    for i in range(M):
        u, v = links[i]
        used = inbound_cycles[u] | outbound_cycles[v]
        cycle = 1
        while cycle in used:
            cycle += 1
        assignment[i] = cycle
        inbound_cycles[u].add(cycle)
        outbound_cycles[v].add(cycle)
    
    print("YES")
    print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
