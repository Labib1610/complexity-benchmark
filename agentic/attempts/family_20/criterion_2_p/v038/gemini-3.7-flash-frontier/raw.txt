An optimal and elegant way to solve bipartite edge coloring in $O(M \log D)$ time is through divide-and-conquer Eulerian trail decomposition. 

By decomposing the graph into alternating trails (paths between odd-degree vertices and Eulerian cycles), we can partition the edges into two subgraphs such that the degree of every vertex is split into $\lceil \deg(v)/2 \rceil$ and $\lfloor \deg(v)/2 \rfloor$. This guarantees that both subgraphs have maximum degree at most $\lceil D/2 \rceil$ and $\lfloor D/2 \rfloor$, respectively, allowing us to recursively color each half.

### Python 3 Source Code

```python
import sys

# Increase recursion depth for deep divide-and-conquer trees
sys.setrecursionlimit(2000000)

def main():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    Nin = int(input_data[0])
    Nout = int(input_data[1])
    M = int(input_data[2])

    if M == 0:
        print("YES")
        print()
        return

    # Read edges
    # Represent inbound nodes as 1..Nin, outbound as Nin+1..Nin+Nout
    u_arr = [0] * M
    v_arr = [0] * M
    deg = [0] * (Nin + Nout + 1)

    idx = 3
    for i in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx + 1]) + Nin
        idx += 2
        u_arr[i] = u
        v_arr[i] = v
        deg[u] += 1
        deg[v] += 1

    D = max(deg)
    ans = [0] * M

    # Eulerian trail decomposition
    head = [-1] * (Nin + Nout + 1)
    to = [0] * (2 * M)
    nxt = [0] * (2 * M)
    edge_id = [0] * (2 * M)
    used_edge = [False] * M
    cur_deg = [0] * (Nin + Nout + 1)

    def solve(edge_list, d, color_offset):
        if not edge_list:
            return
        if d == 1:
            c = color_offset + 1
            for e in edge_list:
                ans[e] = c
            return

        # Active vertices in this subproblem
        nodes = []
        for e in edge_list:
            u = u_arr[e]
            v = v_arr[e]
            if cur_deg[u] == 0:
                nodes.append(u)
            cur_deg[u] += 1
            if cur_deg[v] == 0:
                nodes.append(v)
            cur_deg[v] += 1

        # Build adjacency
        edge_ptr = 0
        for e in edge_list:
            used_edge[e] = False
            u = u_arr[e]
            v = v_arr[e]

            to[edge_ptr] = v
            edge_id[edge_ptr] = e
            nxt[edge_ptr] = head[u]
            head[u] = edge_ptr
            edge_ptr += 1

            to[edge_ptr] = u
            edge_id[edge_ptr] = e
            nxt[edge_ptr] = head[v]
            head[v] = edge_ptr
            edge_ptr += 1

        edges1 = []
        edges2 = []

        # Find paths starting at odd-degree vertices
        for u in nodes:
            if cur_deg[u] % 2 != 0:
                curr = u
                color = 0
                while True:
                    e_idx = -1
                    while head[curr] != -1:
                        p = head[curr]
                        head[curr] = nxt[p]
                        e = edge_id[p]
                        if not used_edge[e]:
                            used_edge[e] = True
                            e_idx = e
                            next_node = to[p]
                            break
                    if e_idx == -1:
                        break
                    if color == 0:
                        edges1.append(e_idx)
                    else:
                        edges2.append(e_idx)
                    color ^= 1
                    curr = next_node

        # Find cycles for remaining even-degree vertices
        for u in nodes:
            while head[u] != -1:
                # Find an unused edge
                start_p = -1
                while head[u] != -1:
                    p = head[u]
                    head[u] = nxt[p]
                    e = edge_id[p]
                    if not used_edge[e]:
                        used_edge[e] = True
                        start_p = p
                        break
                if start_p == -1:
                    break

                curr = to[start_p]
                e_idx = edge_id[start_p]
                edges1.append(e_idx)
                color = 1

                while curr != u:
                    e_idx = -1
                    while head[curr] != -1:
                        p = head[curr]
                        head[curr] = nxt[p]
                        e = edge_id[p]
                        if not used_edge[e]:
                            used_edge[e] = True
                            e_idx = e
                            next_node = to[p]
                            break
                    if e_idx == -1:
                        break
                    if color == 0:
                        edges1.append(e_idx)
                    else:
                        edges2.append(e_idx)
                    color ^= 1
                    curr = next_node

        # Clean up
        for u in nodes:
            cur_deg[u] = 0
            head[u] = -1

        d1 = (d + 1) // 2
        d2 = d // 2

        solve(edges1, d1, color_offset)
        solve(edges2, d2, color_offset + d1)

    initial_edges = list(range(M))
    solve(initial_edges, D, 0)

    print("YES")
    print(" ".join(map(str, ans)))

if __name__ == '__main__':
    main()
```
