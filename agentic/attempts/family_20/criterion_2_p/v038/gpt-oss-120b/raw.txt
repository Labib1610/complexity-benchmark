import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    N_in = int(next(it))
    N_out = int(next(it))
    M = int(next(it))
    edges = [(int(next(it)) - 1, int(next(it)) - 1) for _ in range(M)]

    deg_in = [0] * N_in
    deg_out = [0] * N_out
    for u, v in edges:
        deg_in[u] += 1
        deg_out[v] += 1
    D = max(max(deg_in, default=0), max(deg_out, default=0))

    # adjacency lists of edge indices for each vertex
    adj_in = [[] for _ in range(N_in)]
    adj_out = [[] for _ in range(N_out)]
    for idx, (u, v) in enumerate(edges):
        adj_in[u].append(idx)
        adj_out[v].append(idx)

    # colors used at each vertex (set implemented as list of bool)
    used_in = [bytearray(D + 1) for _ in range(N_in)]
    used_out = [bytearray(D + 1) for _ in range(N_out)]
    color = [0] * M

    # helper to find a free color for a vertex
    def first_free(used):
        for c in range(1, D + 1):
            if not used[c]:
                return c
        return 0  # should never happen

    # main greedy with augmenting path (standard bipartite edge coloring)
    for idx, (u, v) in enumerate(edges):
        # try to assign a free color directly
        cu = used_in[u]
        cv = used_out[v]
        c = 0
        for col in range(1, D + 1):
            if not cu[col] and not cv[col]:
                c = col
                break
        if c:
            color[idx] = c
            cu[c] = 1
            cv[c] = 1
            continue

        # need to recolor: find alternating path
        # we will try colors a (missing at u) and b (missing at v)
        # find any a missing at u
        a = first_free(cu)
        b = first_free(cv)

        # BFS on vertices to find alternating path from u to v using colors a and b
        from collections import deque
        q = deque()
        prev = {}  # vertex -> (prev_vertex, edge_idx, used_color_at_prev)
        # start from u (in side)
        q.append(('in', u))
        prev[('in', u)] = None

        target = None
        while q and target is None:
            side, x = q.popleft()
            if side == 'in':
                # edges incident to x with color a
                for e_idx in adj_in[x]:
                    if color[e_idx] != a:
                        continue
                    _, y = edges[e_idx]
                    node = ('out', y)
                    if node not in prev:
                        prev[node] = (('in', x), e_idx, a)
                        if not used_out[y][b]:
                            target = node
                            break
                        q.append(node)
            else:  # out side
                # edges incident to x with color b
                for e_idx in adj_out[x]:
                    if color[e_idx] != b:
                        continue
                    u2, _ = edges[e_idx]
                    node = ('in', u2)
                    if node not in prev:
                        prev[node] = (('out', x), e_idx, b)
                        if not used_in[u2][a]:
                            target = node
                            break
                        q.append(node)

        # reconstruct and flip colors along the path
        cur = target
        while cur is not None:
            p = prev[cur]
            if p is None:
                break
            (prev_node, e_idx, col) = p
            # flip color of edge e_idx
            old = color[e_idx]
            new = a if old == b else b
            color[e_idx] = new
            # update used arrays
            uu, vv = edges[e_idx]
            used_in[uu][old] = 0
            used_out[vv][old] = 0
            used_in[uu][new] = 1
            used_out[vv][new] = 1
            cur = prev_node

        # now assign color a to current edge
        color[idx] = a
        used_in[u][a] = 1
        used_out[v][a] = 1

    out = ['YES', ' '.join(map(str, color))]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
