import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    vaults = [0] * (n + 1)
    
    def can_place(material, vault_idx, current_vaults):
        mask = 1 << (material - 1)
        for i in range(1, material):
            if (adj[material] & (1 << (i - 1))) and current_vaults[i] == vault_idx:
                return False
        return True

    def backtrack(material, current_vaults):
        if material > n:
            return True
        
        for v in range(1, 4):
            if can_place(material, v, current_vaults):
                current_vaults[material] = v
                if backtrack(material + 1, current_vaults):
                    return True
                current_vaults[material] = 0
        return False

    if backtrack(1, vaults):
        print("YES")
        print(*(vaults[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
