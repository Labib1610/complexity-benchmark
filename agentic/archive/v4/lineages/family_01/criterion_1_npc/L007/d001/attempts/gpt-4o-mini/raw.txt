def can_color_graph(n, edges):
    from collections import defaultdict, deque
    
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    color = [0] * (n + 1)  # 0 means uncolored; colors will be 1, 2, or 3
    
    def bfs(start):
        queue = deque([start])
        color[start] = 1  # Start coloring with color 1
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            next_color = 1 if current_color == 3 else current_color + 1
            
            for neighbor in graph[node]:
                if color[neighbor] == 0:  # If uncolored
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] == current_color:  # Conflict in coloring
                    return False
        return True
    
    for i in range(1, n + 1):
        if color[i] == 0:  # Not colored yet
            if not bfs(i):
                print("NO")
                return
    
    # Make sure the colors are lexicographically smallest
    result = [color[i] for i in range(1, n + 1)]
    print("YES")
    print(" ".join(map(str, result)))

import sys
input = sys.stdin.read
data = input().strip().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]
can_color_graph(n, edges)
