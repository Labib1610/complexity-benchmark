def can_assign_vaults(n, m, incompatible_pairs):
    from collections import defaultdict, deque

    # Build the graph
    graph = defaultdict(list)
    for u, v in incompatible_pairs:
        graph[u].append(v)
        graph[v].append(u)

    # Color array to store vault assignments
    color = [-1] * (n + 1)
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0  # Start coloring with 0 (representing vault 1)
        
        while queue:
            node = queue.popleft()
            current_color = color[node]
            next_color = (current_color + 1) % 3
            
            for neighbor in graph[node]:
                if color[neighbor] == -1:  # Not colored yet
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] == current_color:  # Conflict
                    return False
        return True

    for i in range(1, n + 1):
        if color[i] == -1:  # Not colored
            if not bfs(i):
                return "NO", []

    # Prepare the output in lexicographically smallest order
    result = [c + 1 for c in color[1:]]  # Convert to 1-based vault numbers
    return "YES", result

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
incompatible_pairs = [tuple(map(int, line.split())) for line in data[1:m+1]]

result = can_assign_vaults(n, m, incompatible_pairs)
if result[0] == "NO":
    print("NO")
else:
    print("YES")
    print(" ".join(map(str, result[1])))
