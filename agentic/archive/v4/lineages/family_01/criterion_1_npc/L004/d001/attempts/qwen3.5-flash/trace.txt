        adj[v].append(u)

    colors = [0] * n

    # We need to find the lexicographically smallest assignment.
    # We iterate colors 1, 2, 3 for each node in order 1..n.
    
    def dfs(idx):
        if idx == n:
            return True
        
        # If no color is possible for current node, backtrack
        if possible[idx] == 0:
            return False
        
        p = possible[idx]

        # We iterate in order to ensure lexicographically smallest solution
        
        # Try color 1 (bit 0)
        if (p & 1):
            c = 1
            colors[idx] = c
            changes = []
            valid = True
