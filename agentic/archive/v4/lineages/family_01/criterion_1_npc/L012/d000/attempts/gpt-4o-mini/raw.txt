def solve():
    import sys
    from collections import defaultdict
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    conflicts = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        conflicts[u - 1].append(v - 1)
        conflicts[v - 1].append(u - 1)
    
    shifts = [-1] * n
    
    def is_valid(worker, shift):
        for conflict in conflicts[worker]:
            if shifts[conflict] == shift:
                return False
        return True
    
    def assign_shifts(worker):
        if worker == n:
            return True
        
        for shift in range(1, 4):
            if is_valid(worker, shift):
                shifts[worker] = shift
                if assign_shifts(worker + 1):
                    return True
                shifts[worker] = -1
        
        return False
    
    if assign_shifts(0):
        print("YES")
        print(" ".join(map(str, shifts)))
    else:
        print("NO")
