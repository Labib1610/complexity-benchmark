def solve():
    # Read all input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # available[u] bitmask: bit 0 for shift 1, bit 1 for shift 2, bit 2 for shift 3
    # 7 is binary 111
    available = [7] * n
    assignment = [0] * n

    def backtrack(idx):
        if idx == n:
            return True
        
        # Pruning: if no colors available for current vertex
        if available[idx] == 0:
            return False

        # Try shifts 1, 2, 3 (bits 1, 2, 4) in order
        for c in (1, 2, 4):
            # Check if shift c is available for current vertex
            if not (c & available[idx]):
                continue
            
            # Try assigning shift c to vertex idx
            # We need to update available colors for uncolored neighbors (v > idx)
            modified = []
            valid = True
            
            for v in adj[idx]:
                if v > idx:
                    # v is uncolored
                    # Remove shift c from v's available colors
                    available[v] &= ~c
                    modified.append(v)
                    if available[v] == 0:
                        valid = False
                        break
            
            if valid:
                assignment[idx] = c
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
            
            # Backtrack: restore available colors for neighbors
            for v in modified:
                available[v] |= c
        
        return False

    if backtrack(0):
        print("YES")
        # Map bits back to shifts 1, 2, 3
        # 1 -> 1, 2 -> 2, 4 -> 3
        res = []
        for x in assignment:
            if x == 1: res.append(1)
            elif x == 2: res.append(2)
            elif x == 4: res.append(3)
        print(*(res))
    else:
        print("NO")
