def solve():
    import sys
    from collections import defaultdict
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    incompatible_pairs = [tuple(map(int, line.split())) for line in data[1:m + 1]]
    
    incompatibility_graph = defaultdict(set)
    
    for u, v in incompatible_pairs:
        incompatibility_graph[u].add(v)
        incompatibility_graph[v].add(u)
    
    assignment = [0] * (n + 1)  # to hold block assignments (1-indexed)
    
    def assign(unit):
        if unit > n:  # all units assigned
            return True
        
        for block in range(1, 4):  # blocks 1, 2, 3
            assignment[unit] = block
            
            # Check compatibility with already assigned units
            if all(assignment[i] == 0 or (i not in incompatibility_graph[unit] or assignment[i] != block) for i in range(1, n + 1)):
                if assign(unit + 1):
                    return True
            
            assignment[unit] = 0  # reset on backtrack
            
        return False
    
    if assign(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")
