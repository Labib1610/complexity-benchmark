def can_color_graph(n, m, edges):
    from collections import defaultdict
    graph = defaultdict(list)
    
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    colors = [0] * (n + 1)  # Use 1-based indexing for nodes

    def is_valid(node, color):
        for neighbor in graph[node]:
            if colors[neighbor] == color:
                return False
        return True

    def assign_colors(node):
        if node > n:
            return True
        
        for color in range(1, 4):  # Colors 1, 2, 3
            if is_valid(node, color):
                colors[node] = color
                if assign_colors(node + 1):
                    return True
                colors[node] = 0  # Backtrack
        
        return False

    if assign_colors(1):
        print("YES")
        print(' '.join(map(str, colors[1:])))
    else:
        print("NO")


import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m+1]]
can_color_graph(n, m, edges)
