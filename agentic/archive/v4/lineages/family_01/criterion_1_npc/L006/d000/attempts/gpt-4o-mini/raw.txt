def main():
    import sys
    from collections import defaultdict, deque

    input = sys.stdin.read
    data = input().splitlines()

    n, m = map(int, data[0].split())
    
    incompatibilities = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        incompatibilities[u].append(v)
        incompatibilities[v].append(u)

    # Vault assignments
    vaults = [-1] * (n + 1)  # 1-indexed
    
    def bfs(start):
        queue = deque([start])
        vaults[start] = 1  # Start assigning from vault 1
        
        while queue:
            current = queue.popleft()
            current_vault = vaults[current]
            next_vault = 1 if current_vault != 1 else 2
            
            for neighbor in incompatibilities[current]:
                if vaults[neighbor] == -1:  # Not assigned yet
                    vaults[neighbor] = next_vault
                    queue.append(neighbor)
                elif vaults[neighbor] == current_vault:  # Conflict found
                    return False
                next_vault = 3 if next_vault == 2 else 2  # Alternate vaults
        
        return True

    for i in range(1, n + 1):
        if vaults[i] == -1:  # Not assigned
            if not bfs(i):
                print("NO")
                return

    # Adjust vault assignments to be lexicographically smallest
    for i in range(1, n + 1):
        if vaults[i] == 0:
            vaults[i] = 1  # Ensure at least 1 is assigned

    print("YES")
    print(" ".join(map(str, vaults[1:])))
    
main()
