def is_valid_assignment(n, graph, assignment):
    for u in range(n):
        for v in graph[u]:
            if assignment[u] == assignment[v]:
                return False
    return True

def assign_slots(n, graph):
    assignment = [0] * n
    def backtrack(index):
        if index == n:
            if is_valid_assignment(n, graph, assignment):
                return assignment[:]
            return None
        
        for slot in range(1, 4):
            assignment[index] = slot
            result = backtrack(index + 1)
            if result:
                return result
            assignment[index] = 0
        
        return None

    return backtrack(0)

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
graph = [[] for _ in range(n)]

for i in range(1, m + 1):
    u, v = map(int, data[i].split())
    graph[u - 1].append(v - 1)
    graph[v - 1].append(u - 1)

assignment = assign_slots(n, graph)

if assignment is None:
    print("NO")
else:
    print("YES")
    print(" ".join(map(str, assignment)))
