def dfs(idx):
    if idx>n: return True
    for c in (1,2,3):
        conflict=False
        for nb in adj[idx]:
            if color[nb]==c:
                conflict=True
                break
        if conflict: continue
        color[idx]=c
        # forward check
        ok=True
        for u in range(idx+1, n+1):
            used=0
            for nb in adj[u]:
                if color[nb]!=0:
                    used |= 1 << (color[nb]-1)
                if used==0b111:
                    break
            if used==0b111:
                ok=False
                break
        if ok:
            if dfs(idx+1):
                return True
        color[idx]=0
    return False
