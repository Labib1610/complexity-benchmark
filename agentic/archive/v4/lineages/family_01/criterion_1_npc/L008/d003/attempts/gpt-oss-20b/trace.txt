          if neighbor>index:
              if (available[neighbor] >> (color-1)) &1:

                  available[neighbor] = old & ~(1<<(color-1))

      if ok and recursion(index+1):
          return True
      # backtrack
      for v, old in changes:
          available[v]=old
      assignment[index]=0
