def main():
    import sys
    from collections import defaultdict

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    graph = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)
    
    colors = [0] * (n + 1)

    def can_color(node, color):
        colors[node] = color
        for neighbor in graph[node]:
            if colors[neighbor] == color:
                return False
            if colors[neighbor] == 0 and not can_color(neighbor, 3 - color):
                return False
        return True

    for node in range(1, n + 1):
        if colors[node] == 0:
            if not can_color(node, 1):
                print("NO")
                return
    
    # Assign channels 1, 2, 3 based on the coloring
    result = []
    for i in range(1, n + 1):
        if colors[i] == 1:
            result.append(1)
        elif colors[i] == 2:
            result.append(2)
        else:
            result.append(3)
    
    print("YES")
    print(" ".join(map(str, result)))

main()
