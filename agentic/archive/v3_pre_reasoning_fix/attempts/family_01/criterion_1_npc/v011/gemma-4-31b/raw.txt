import sys

def solve():
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Result array to store assigned channels
    res = [0] * n
    
    def is_valid(node, color):
        for neighbor in adj[node]:
            if res[neighbor] == color:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        
        # Try channels 1, 2, 3 in order to ensure lexicographical smallest result
        for color in range(1, 4):
            if is_valid(node, color):
                res[node] = color
                if backtrack(node + 1):
                    return True
                res[node] = 0
        return False

    # The constraints n <= 40 suggest a backtracking approach.
    # Since we need the lexicographically smallest sequence, 
    # we simply try colors 1, 2, 3 in order for each station from 1 to n.
    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
