import sys

def solve():
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # assignment[i] stores the window (1, 2, or 3) for person i
    assignment = [0] * (n + 1)

    def is_valid(person, window):
        for neighbor in adj[person]:
            if assignment[neighbor] == window:
                return False
        return True

    def backtrack(person):
        if person > n:
            return True
        
        # To get the lexicographically smallest sequence, 
        # we try windows 1, 2, and 3 in order.
        for window in range(1, 4):
            if is_valid(person, window):
                assignment[person] = window
                if backtrack(person + 1):
                    return True
                assignment[person] = 0
        
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
