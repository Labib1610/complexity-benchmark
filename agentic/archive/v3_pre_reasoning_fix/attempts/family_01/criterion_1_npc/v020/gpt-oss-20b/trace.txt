    node=idx
    for c in range(1,4):
        if allowed[node] & (1<<(c-1))==0: continue
        # check neighbor conflict: but allowed ensures no conflict? Actually neighbor might already have same color but allowed[neighbor] may still include that color if not considered. But we enforce checking before assignment: For each neighbor already colored same, conflict. So we need neighbor check.
        conflict=False
        for nb in adj[node]:
            if color[nb]==c:
                conflict=True
                break
        if conflict: continue
        # assign
        color[node]=c
        changes=[]
        bad=False
        for nb in adj[node]:
            if color[nb]==0:
                old=allowed[nb]
                new=old & ~(1<<(c-1))
                if new!=old:
                    allowed[nb]=new
                    changes.append((nb,old))
                    if new==0:
                        bad=True
                        break
        if not bad and dfs(idx+1):
            return True
        # revert
        for nb,old in changes:
            allowed[nb]=old
        color[node]=0
    return False
