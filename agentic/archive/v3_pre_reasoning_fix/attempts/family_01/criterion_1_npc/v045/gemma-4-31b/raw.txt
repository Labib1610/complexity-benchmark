import sys

def solve():
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list for incompatibilities
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Shift assignments: 0 means unassigned, 1, 2, 3 are shifts
    assignment = [0] * (n + 1)

    def is_safe(node, shift):
        for neighbor in adj[node]:
            if assignment[neighbor] == shift:
                return False
        return True

    def backtrack(node):
        # Base case: all staff members assigned
        if node > n:
            return True
        
        # Try shifts 1, 2, 3 in order to find the lexicographically smallest sequence
        for shift in (1, 2, 3):
            if is_safe(node, shift):
                assignment[node] = shift
                if backtrack(node + 1):
                    return True
                # Backtrack
                assignment[node] = 0
        
        return False

    # Start backtracking from the first staff member
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
