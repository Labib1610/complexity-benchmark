import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    # We need the lexicographically smallest assignment.
    # We can use backtracking with pruning.
    # Since n is up to 40, we need an efficient way to check 3-colorability.
    # 3-coloring is NP-complete, but n=40 is small enough for optimized backtracking
    # if we use bitmasks and pruning.
    
    res = [0] * n
    
    def is_valid(u, color):
        # Check if any neighbor of u already has the assigned color
        # We only check neighbors v < u because those are already assigned
        # This is handled by the backtracking loop.
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3 in order for lexicographical smallest result
        for color in (1, 2, 3):
            # Check if any neighbor of u already has this color
            possible = True
            for v in range(u):
                if (adj[u] & (1 << v)) and res[v] == color:
                    possible = False
                    break
            
            if possible:
                res[u] = color
                # Heuristic: we could check if the remaining graph is still 3-colorable,
                # but for n=40, simple backtracking with the order 1,2,3 often works
                # if the constraints are not specifically designed to break it.
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    # To optimize, we can use a more robust 3-coloring check or 
    # simply rely on the fact that n=40 is small for some cases.
    # However, standard backtracking might TLE. 
    # Let's use a bitmask approach to check constraints quickly.
    
    color_masks = [0, 0, 0, 0] # masks for colors 1, 2, 3 (index 0 unused)
    
    def backtrack_fast(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for c in (1, 2, 3):
            if not (adj[u] & color_masks[c]):
                color_masks[c] |= (1 << u)
                res[u] = c
                if backtrack_fast(u + 1):
                    return True
                color_masks[c] &= ~(1 << u)
        return False

    if backtrack_fast(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

if __name__ == "__main__":
    # Increase recursion depth for n=40
    sys.setrecursionlimit(2000)
    solve()
