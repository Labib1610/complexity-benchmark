def can_assign_slots(n, m, incompatible_pairs):
    from collections import defaultdict
    
    graph = defaultdict(list)
    for u, v in incompatible_pairs:
        graph[u - 1].append(v - 1)
        graph[v - 1].append(u - 1)

    slots = [-1] * n

    def dfs(person, assigned_slot):
        if slots[person] != -1:
            return slots[person] == assigned_slot
        
        slots[person] = assigned_slot
        for neighbor in graph[person]:
            if not dfs(neighbor, 3 - assigned_slot):  # Try to assign the other slot
                return False
        return True

    for person in range(n):
        if slots[person] == -1:
            if not (dfs(person, 1) or dfs(person, 2) or dfs(person, 3)):
                return "NO", []
    
    return "YES", slots

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
incompatible_pairs = [tuple(map(int, line.split())) for line in data[1:m+1]]

result, assignments = can_assign_slots(n, m, incompatible_pairs)
if result == "NO":
    print(result)
else:
    print(result)
    print(" ".join(map(str, assignments)))
