            if c == 2 and cnt2[u] > 0: continue
            if c == 3 and cnt3[u] > 0: continue
            
            # Assign color c to u
            colors[u] = c
            res[u] = c
            
            # Update constraints for neighbors
            doomed = False
            
            # Update counts for uncolored neighbors
            # We update all neighbors first to maintain consistent state before checking doom
            for v in adj[u]:
                if colors[v] == 0:
                    if c == 1:
                        cnt1[v] += 1
                    elif c == 2:
                        cnt2[v] += 1
                    else:
                        cnt3[v] += 1
            
            # Check if any uncolored neighbor is doomed (has all 3 colors forbidden)
            for v in adj[u]:
                if colors[v] == 0:
                    if cnt1[v] > 0 and cnt2[v] > 0 and cnt3[v] > 0:
                        doomed = True
                        break
            
            if doomed:
                # Rollback updates
                for v in adj[u]:
                    if colors[v] == 0:
                        if c == 1:
                            cnt1[v] -= 1
                        elif c == 2:
                            cnt2[v] -= 1
                        else:
                            cnt3[v] -= 1
                colors[u] = 0
                continue
            
            # Recurse to next node
            if dfs(u + 1):
                return True
            
            # Backtrack: undo updates
            for v in adj[u]:
                if colors[v] == 0:
                    if c == 1:
                        cnt1[v] -= 1
                    elif c == 2:
                        cnt2[v] -= 1
                    else:
                        cnt3[v] -= 1
            colors[u] = 0
            
        return False

    # Start DFS from node 1
    if dfs(1):
        print("YES")
        # Print the sequence s_1, s_2, ..., s_n
        print(*(res[1:]))
    else:
        print("NO")

    solve()
