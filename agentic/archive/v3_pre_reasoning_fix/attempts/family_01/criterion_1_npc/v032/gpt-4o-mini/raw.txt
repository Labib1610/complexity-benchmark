def assign_time_slots(n, m, restrictions):
    from collections import defaultdict
    
    graph = defaultdict(set)
    
    for u, v in restrictions:
        graph[u].add(v)
        graph[v].add(u)

    slots = [0] * (n + 1)  # 1-indexed, 0 means unassigned

    def dfs(personnel):
        if personnel > n:
            return True
        
        for slot in range(1, 4):  # Try slots 1, 2, 3
            if all(slots[neighbor] != slot for neighbor in graph[personnel]):
                slots[personnel] = slot
                if dfs(personnel + 1):
                    return True
                slots[personnel] = 0  # Backtrack
        
        return False

    if dfs(1):
        print("YES")
        print(' '.join(map(str, slots[1:])))
    else:
        print("NO")

import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
restrictions = [tuple(map(int, line.split())) for line in data[1:m + 1]]

assign_time_slots(n, m, restrictions)
