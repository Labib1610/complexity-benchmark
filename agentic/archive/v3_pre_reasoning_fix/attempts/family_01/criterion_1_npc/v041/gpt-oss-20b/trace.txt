colors = [0]*n

def dfs(i):
    if i==n:
        return True
    v = i
    # compute neighbor colors
    for color in (1,2,3):
        conflict = False
        for nb in adj[v]:
            if colors[nb]==color:
                conflict = True
                break
        if conflict: continue
        colors[v] = color
        # early pruning: check all uncolored vertices
        ok = True
        for j in range(i+1, n):
            if colors[j]==0:
                # check if any color available
                avail = False
                for col in (1,2,3):
                    bad = False
                    for nb in adj[j]:
                        if colors[nb]==col:
                            bad = True
                            break
                    if not bad:
                        avail = True
                        break
                if not avail:
                    ok = False
                    break
        if ok:
            if dfs(i+1):
                return True
        colors[v] = 0
    return False
