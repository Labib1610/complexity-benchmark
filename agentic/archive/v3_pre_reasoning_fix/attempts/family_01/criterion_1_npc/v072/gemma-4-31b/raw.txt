import sys

def solve():
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # We want the lexicographically smallest assignment.
    # This means we try slot 1, then 2, then 3 for worker 1, then worker 2, and so on.
    assignment = [0] * (n + 1)
    
    def is_valid(worker, slot):
        for neighbor in adj[worker]:
            if assignment[neighbor] == slot:
                return False
        return True

    def backtrack(worker):
        if worker > n:
            return True
        
        # Try slots 1, 2, 3 in order to ensure lexicographical smallest result
        for slot in (1, 2, 3):
            if is_valid(worker, slot):
                assignment[worker] = slot
                if backtrack(worker + 1):
                    return True
                assignment[worker] = 0
        return False

    # Optimization: For n=40, simple backtracking might be too slow.
    # However, 3-coloring is NP-complete. With n=40, we can try a bit of pruning 
    # or just rely on the constraints if the test cases aren't adversarial.
    # To improve speed, we can use a more efficient check or forward checking.
    
    # Forward checking / Constraint Propagation
    # domains[i] is a bitmask of available slots for worker i (bit 0: slot 1, etc.)
    domains = [(1 << 3) - 1] * (n + 1)
    
    def backtrack_with_domains(worker):
        if worker > n:
            return True
        
        # Current possible slots for this worker
        current_domain = domains[worker]
        
        for slot_idx in range(3): # Try slots 1, 2, 3
            if current_domain & (1 << slot_idx):
                slot = slot_idx + 1
                
                # Save domains to restore after backtracking
                saved_domains = []
                possible = True
                
                # Update neighbors' domains
                for neighbor in adj[worker]:
                    if neighbor > worker:
                        if domains[neighbor] & (1 << slot_idx):
                            saved_domains.append((neighbor, domains[neighbor]))
                            domains[neighbor] &= ~(1 << slot_idx)
                            if domains[neighbor] == 0:
                                possible = False
                                break
                
                if possible:
                    assignment[worker] = slot
                    if backtrack_with_domains(worker + 1):
                        return True
                
                # Restore domains
                for neighbor, old_dom in saved_domains:
                    domains[neighbor] = old_dom
                assignment[worker] = 0
                
        return False

    if backtrack_with_domains(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
