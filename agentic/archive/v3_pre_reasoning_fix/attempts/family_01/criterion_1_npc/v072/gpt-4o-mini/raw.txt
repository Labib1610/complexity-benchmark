def can_allocate(n, m, restrictions):
    from collections import defaultdict
    
    graph = defaultdict(set)
    for u, v in restrictions:
        graph[u].add(v)
        graph[v].add(u)

    # To store the slot assignments for workers
    slots = [0] * (n + 1)

    def backtrack(worker):
        if worker > n:
            return True
        
        for slot in range(1, 4):
            # Check if this slot can be assigned to the worker
            valid = True
            for neighbor in graph[worker]:
                if slots[neighbor] == slot:
                    valid = False
                    break
            
            if valid:
                slots[worker] = slot
                if backtrack(worker + 1):
                    return True
                slots[worker] = 0  # backtrack

        return False

    if backtrack(1):
        return "YES", slots[1:]  # Ignore index 0
    else:
        return "NO",

import sys
input = sys.stdin.read
data = input().splitlines()

first_line = data[0].split()
n = int(first_line[0])
m = int(first_line[1])
restrictions = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = can_allocate(n, m, restrictions)

if result[0] == "NO":
    print(result[0])
else:
    print(result[0])
    print(" ".join(map(str, result[1])))
