import sys

def solve():
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # assignment stores the block for each task (1-indexed)
    assignment = [0] * (n + 1)
    
    def is_safe(u, block):
        for v in adj[u]:
            if assignment[v] == block:
                return False
        return True

    def can_color(u):
        """
        Checks if the remaining tasks [u...n] can be colored given the 
        current assignments [1...u-1].
        Since n=40 is small but 3^40 is too large, we use backtracking.
        To optimize, we use a simple backtracking with a small heuristic
        or just raw backtracking since we only need to know if ANY solution exists.
        """
        if u > n:
            return True
        
        # Try colors 1, 2, 3
        for block in (1, 2, 3):
            if is_safe(u, block):
                assignment[u] = block
                if can_color(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To find the lexicographically smallest sequence, we try to assign 
    # the smallest possible block to task i and check if the rest of the 
    # graph can still be 3-colored.
    
    # However, a simple "try 1, then 2, then 3" for each i and then
    # verifying the rest via backtracking is exactly what the 
    # standard backtracking search does if we explore options in order.
    
    result = [0] * (n + 1)
    
    def find_lex_smallest(u):
        if u > n:
            return True
        
        for block in (1, 2, 3):
            if is_safe(u, block):
                assignment[u] = block
                # We need to check if this choice allows for a completion.
                # Because we want the lexicographically smallest, we greedily
                # pick the smallest block and check if a valid coloring exists.
                if can_color_from(u + 1):
                    return True
                assignment[u] = 0
        return False

    def can_color_from(u):
        if u > n:
            return True
        
        # Basic pruning: check if any unassigned node has no available colors
        # This is a simple version of forward checking.
        for i in range(u, n + 1):
            used = {assignment[v] for v in adj[i] if assignment[v] != 0}
            if len(used) == 3:
                return False
        
        # Try colors for the current node u
        for block in (1, 2, 3):
            if is_safe(u, block):
                assignment[u] = block
                if can_color_from(u + 1):
                    return True
                assignment[u] = 0
        return False

    # We can simply use one backtracking function.
    # The first valid assignment found by trying colors 1, 2, 3 in order
    # for tasks 1, 2, ..., n will be the lexicographically smallest.
    
    final_assignment = [0] * (n + 1)
    def backtrack(u):
        if u > n:
            return True
        
        for block in (1, 2, 3):
            if is_safe(u, block):
                assignment[u] = block
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

if __name__ == "__main__":
    solve()
