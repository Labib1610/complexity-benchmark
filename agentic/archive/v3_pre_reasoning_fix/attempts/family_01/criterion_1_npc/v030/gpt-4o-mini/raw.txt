def main():
    import sys
    from collections import defaultdict

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    restrictions = defaultdict(set)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        restrictions[u].add(v)
        restrictions[v].add(u)
    
    # Attempt to color the graph with 3 colors (1, 2, 3)
    colors = [0] * (n + 1)  # Using 1-based index, 0 means uncolored
    
    def can_color(node, color):
        if colors[node] != 0:
            return colors[node] == color  # Check if already colored correctly
        
        colors[node] = color
        for neighbor in restrictions[node]:
            if not can_color(neighbor, 1 if color != 1 else (2 if color != 2 else 3)):
                return False
        return True
    
    for i in range(1, n + 1):
        if colors[i] == 0:  # Not colored yet
            if not can_color(i, 1):  # Start coloring with color 1
                print("NO")
                return
    
    # Prepare the result in lexicographically smallest order
    result = colors[1:]  # Skip index 0 as it's unused
    print("YES")
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
