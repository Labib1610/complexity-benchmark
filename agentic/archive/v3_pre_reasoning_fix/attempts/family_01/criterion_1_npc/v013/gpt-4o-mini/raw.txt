def allocate_shifts(n, m, conflicts):
    from collections import defaultdict, deque

    graph = defaultdict(list)
    for u, v in conflicts:
        graph[u].append(v)
        graph[v].append(u)

    shifts = [0] * (n + 1)  # Shift assignments, 1-indexed

    for worker in range(1, n + 1):
        if shifts[worker] == 0:  # Not assigned yet
            available_shifts = {1, 2, 3}
            for neighbor in graph[worker]:
                if shifts[neighbor] in available_shifts:
                    available_shifts.remove(shifts[neighbor])
            if available_shifts:
                shifts[worker] = min(available_shifts)

            # BFS to assign shifts to connected components
            queue = deque([worker])
            while queue:
                current = queue.popleft()
                current_shift = shifts[current]
                next_shift = 1 if current_shift == 3 else current_shift + 1
                for neighbor in graph[current]:
                    if shifts[neighbor] == 0:  # Not assigned yet
                        while next_shift in available_shifts:
                            next_shift += 1
                        if next_shift <= 3:
                            shifts[neighbor] = next_shift
                            queue.append(neighbor)

    if any(shifts[i] == 0 for i in range(1, n + 1)):
        print("NO")
    else:
        print("YES")
        print(" ".join(map(str, shifts[1:])))

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
conflicts = [tuple(map(int, line.split())) for line in data[1:m + 1]]

allocate_shifts(n, m, conflicts)
