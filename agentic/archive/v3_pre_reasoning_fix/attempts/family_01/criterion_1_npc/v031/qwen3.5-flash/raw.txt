    input_data = sys.stdin.read().split()

        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))

        return

    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)

    # 1 -> 1 (001)
    # 2 -> 2 (010)

    color_bits = [0, 1, 2, 4]
    
    def backtrack(u):

            return True

        for c in range(1, 4):
            # Check validity against already colored neighbors
            if not (forbidden[u] & color_bits[c]):
                colors[u] = c
                changed = []
                possible = True

                for v in adj[u]:

                    forbidden[v] ^= color_bits[c] # Toggle the bit back to 0
                
                colors[u] = 0
        
        return False

    # Start backtracking from vertex 1
    if backtrack(1):
        print("YES")
        # Print the colors for vertices 1 to n
        print(*(colors[1:]))
    else:
        print("NO")

    solve()
