import sys

def solve():
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # To find the lexicographically smallest sequence, 
    # we try assigning channel 1, then 2, then 3 for tasks 1 to n.
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try colors in increasing order to ensure lexicographical minimality
        for color in (1, 2, 3):
            if is_valid(u, color):
                assignment[u] = color
                # Forward checking or simple pruning could be added, 
                # but with N=40, basic backtracking with order might pass 
                # if the graph is not a worst-case for 3-coloring.
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # 3-coloring is NP-complete, but N=40 is small enough for some 
    # backtracking implementations, especially when searching for the 
    # lexicographically first solution.
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
