def solve():
    import sys
    from collections import defaultdict
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    incompatibilities = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        incompatibilities[u].append(v)
        incompatibilities[v].append(u)
    
    # This will hold the block assigned to each personnel:
    assignment = [0] * (n + 1)
    
    def can_assign(person, block):
        for incompatible in incompatibilities[person]:
            if assignment[incompatible] == block:
                return False
        return True
    
    def backtrack(person=1):
        if person > n:
            return True
        
        for block in range(1, 4):
            if can_assign(person, block):
                assignment[person] = block
                if backtrack(person + 1):
                    return True
                assignment[person] = 0
        
        return False
    
    if backtrack():
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")
