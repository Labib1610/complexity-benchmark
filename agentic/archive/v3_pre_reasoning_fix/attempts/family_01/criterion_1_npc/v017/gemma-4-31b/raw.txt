import sys

def solve():
    # Increase recursion depth for deep DFS if necessary, 
    # though n=40 is small.
    sys.setrecursionlimit(2000)
    
    # Read input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
    
    # We need the lexicographically smallest assignment.
    # This means for employee 1, we try shift 1, then 2, then 3.
    # If a valid completion exists for shift 1, we take it and move to employee 2.
    
    assignment = [0] * (n + 1)
    
    def is_valid(node, color):
        for neighbor in adj[node]:
            if assignment[neighbor] == color:
                return False
        return True

    def backtrack(node):
        if node > n:
            return True
        
        # Try colors 1, 2, 3 in order to ensure lexicographical smallest result
        for color in [1, 2, 3]:
            if is_valid(node, color):
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        
        return False

    # Since n is small (up to 40), a simple backtracking with pruning 
    # might be too slow in the worst case (3^40), but for 3-coloring 
    # on 40 nodes, it often works if the graph isn't specifically 
    # designed to break it. However, to be more robust, we can use 
    # a more optimized approach.
    
    # Actually, for n=40, a standard backtracking search for 3-coloring 
    # can be slow. But the requirement is the lexicographically smallest.
    # Let's use the simple backtracking and see.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
