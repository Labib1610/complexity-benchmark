import sys

def solve():
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # shifts[i] will store the assigned shift for worker i
    shifts = [0] * (n + 1)

    def is_valid(worker, shift):
        for neighbor in adj[worker]:
            if neighbor < worker: # Only check already assigned workers
                if shifts[neighbor] == shift:
                    return False
        return True

    def can_color(worker):
        """
        Checks if the remaining workers can be colored with 3 colors
        given the current assignments.
        Since n is small (40), we can use a backtracking approach with pruning.
        However, for a 3-coloring problem, the standard approach is 
        backtracking. To speed up, we only check if a valid completion exists.
        """
        if worker > n:
            return True
        
        # Try shifts 1, 2, 3 in order
        for s in range(1, 4):
            if is_valid(worker, s):
                shifts[worker] = s
                if can_color(worker + 1):
                    return True
                shifts[worker] = 0
        return False

    # To find the lexicographically smallest sequence, we greedily 
    # pick the smallest possible shift for the current worker and
    # check if a valid coloring for the rest of the workers exists.
    
    # However, a simple greedy approach with a "can_color" check 
    # is effectively backtracking.
    
    # Let's implement a backtracking search that returns the first 
    # (lexicographically smallest) solution found.
    
    final_shifts = [0] * (n + 1)
    
    def backtrack(worker):
        if worker > n:
            return True
        
        for s in range(1, 4):
            if is_valid(worker, s):
                shifts[worker] = s
                if backtrack(worker + 1):
                    return True
                shifts[worker] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(shifts[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
