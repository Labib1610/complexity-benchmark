import sys

def solve():
    # Read n and m
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    # Adjacency list for conflicts
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        try:
            u, v = map(int, sys.stdin.readline().split())
            adj[u].append(v)
            adj[v].append(u)
        except ValueError:
            break

    # Assignment array: 0 means unassigned
    res = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True

    def backtrack(u):
        # Base case: all personnel assigned
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in order to ensure lexicographical smallest result
        for color in (1, 2, 3):
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        
        return False

    # The constraint n <= 40 is small enough for backtracking with 3 colors 
    # if the graph is sparse or highly constrained. 
    # However, 3-coloring is NP-complete. 
    # Given the "lexicographically smallest" requirement, we must try colors in order.
    if backtrack(1):
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    solve()
