import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # We need the lexicographically smallest sequence (s1, s2, ..., sn).
    # This means we should try to assign the smallest possible shift (1, then 2, then 3)
    # to worker 1, then worker 2, and so on.
    # Since n is small (up to 40), we can use backtracking with pruning.
    
    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try shifts 1, 2, 3 in order for lexicographical smallest
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To speed up backtracking for n=40, we can't rely on simple DFS if the graph is dense.
    # However, for n=40, a simple backtracking with the lexicographical constraint
    # usually works unless the graph is specifically designed to thwart it.
    # To be more robust, we can implement a basic check if the remaining graph is 3-colorable,
    # but 3-coloring is NP-complete.
    # Given the constraints and the "lexicographically smallest" requirement, 
    # the search space is explored in the exact order we need.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == "__main__":
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    solve()
