import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need the lexicographically smallest assignment.
    # This means for compound 1, we try vault 1, then 2, then 3.
    # Then for compound 2, etc.
    # Since n is small (up to 40), we can use backtracking with pruning.
    
    assignment = [0] * n
    
    def is_safe(node, color):
        for neighbor in adj[node]:
            if assignment[neighbor] == color:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        
        # Try colors 1, 2, 3 in order for lexicographical smallest result
        for color in (1, 2, 3):
            if is_safe(node, color):
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    # Optimization: The problem is 3-coloring. 
    # For N=40, simple backtracking might be slow in worst case, 
    # but usually works if the graph is sparse or highly constrained.
    # Given the time limit of 2s and N=40, we can try this.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == "__main__":
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    solve()
