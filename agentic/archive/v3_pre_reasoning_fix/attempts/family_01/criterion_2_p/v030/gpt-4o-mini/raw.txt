def factory_roster(n, m, constraints):
    from collections import defaultdict, deque

    graph = defaultdict(list)
    for u, v in constraints:
        graph[u].append(v)
        graph[v].append(u)

    mode = [0] * (n + 1)  # 0 means unassigned, 1 and 2 are the two modes

    def bfs(start):
        queue = deque([start])
        mode[start] = 1  # Start with mode 1
        while queue:
            current = queue.popleft()
            current_mode = mode[current]
            next_mode = 2 if current_mode == 1 else 1
            
            for neighbor in graph[current]:
                if mode[neighbor] == 0:  # If unassigned
                    mode[neighbor] = next_mode
                    queue.append(neighbor)
                elif mode[neighbor] == current_mode:  # Conflict
                    return False
        return True

    for person in range(1, n + 1):
        if mode[person] == 0:  # Not visited yet
            if not bfs(person):
                print("NO")
                return

    # Prepare the result
    result = mode[1:]  # Skip the 0th index
    print("YES")
    print(" ".join(map(str, result)))

import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
constraints = [tuple(map(int, line.split())) for line in data[1:m + 1]]
factory_roster(n, m, constraints)
