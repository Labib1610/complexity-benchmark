def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)

    # However, for BFS, a list with index pointer is also O(N) total.
    # Let's use collections.deque for clarity and efficiency.
    from collections import deque
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for this component
            # We want to minimize s_i, so assign 1
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                next_color = 3 - current_color
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_color
                        queue.append(v)
                    elif color[v] == current_color:
                        print("NO")
                        return

    print("YES")
    # Output s_1 ... s_n
    # Construct the string for output
    # Using sys.stdout.write might be faster
    print(*(color[1:]))

    solve()
