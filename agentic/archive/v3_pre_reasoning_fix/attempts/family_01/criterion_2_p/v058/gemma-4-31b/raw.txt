import sys

# Increase recursion depth for deep DFS trees in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Read n and m from standard input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # A node interfering with itself cannot be assigned any setting
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # settings[i] will store the setting (1 or 2) for unit i. 0 means unassigned.
    settings = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we process nodes from 1 to n.
    # For each connected component, we try to assign the smallest possible value (1) 
    # to the node with the smallest index in that component.
    for i in range(1, n + 1):
        if settings[i] == 0:
            # Start a BFS/DFS to color the component.
            # To ensure lexicographical smallest, we must decide the color of the 
            # first node in the component (the one with the smallest index).
            # Since we want the smallest sequence, we always try setting the 
            # smallest index node in the component to 1.
            
            stack = [i]
            settings[i] = 1
            
            # Use a stack for iterative DFS to avoid recursion limit issues and be efficient
            component_nodes = []
            
            # We use a queue-like approach (BFS) or stack (DFS). BFS is generally safer.
            queue = [i]
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                
                for v in adj[u]:
                    if settings[v] == 0:
                        # Assign opposite color
                        settings[v] = 3 - settings[u]
                        queue.append(v)
                    elif settings[v] == settings[u]:
                        # Conflict found: graph is not bipartite
                        print("NO")
                        return

    # If we reached here, a valid assignment exists.
    print("YES")
    print(*(settings[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
