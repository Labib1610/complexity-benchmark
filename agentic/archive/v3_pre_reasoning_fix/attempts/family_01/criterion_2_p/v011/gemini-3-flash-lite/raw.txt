# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():

    if not input_data:

    idx = 2

        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # We want the lexicographically smallest sequence.
    # This means for each component, we try to set the first node
    # in the component (by index) to 1. If that fails, the component
    # is impossible.
    
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:

            stack = [(i, 1)]
            color[i] = 1

            # Use stack to traverse component
            comp_stack = [i]
            while comp_stack:
                u = comp_stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp_stack.append(v)
                    elif color[v] == color[u]:

                        return
            
            # After coloring the component, we check if this assignment
            # is valid. The greedy strategy works because setting the 
            # smallest index node to 1 forces the rest of the component.
            # Since the graph is bipartite, the coloring is unique up to
            # swapping 1s and 2s. We choose the assignment that makes
