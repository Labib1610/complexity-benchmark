import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will store the assigned channel (1 or 2)
    # 0 means not yet visited
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, for each connected component,
    # we want to assign the smallest possible value (1) to the smallest index in that component
    # if it doesn't force a contradiction. However, since it's a bipartite graph,
    # once the color of one node in a component is fixed, all others are determined.
    # To minimize the sequence lexicographically, for the first uncolored node i (which is the smallest index in its component),
    # we MUST try assigning color 1 first.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component. 
            # Always start the smallest index in the component with color 1.
            stack = [(i, 1)]
            color[i] = 1
            
            # We use a stack for DFS to avoid recursion depth issues and manage the component
            component = []
            visited_in_comp = []
            
            # Standard iterative DFS/BFS to check bipartiteness and color
            queue = [i]
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
    # If we reached here, the graph is bipartite
    print("YES")
    print(*(color[1:]))

if __name__ == "__main__":
    solve()
