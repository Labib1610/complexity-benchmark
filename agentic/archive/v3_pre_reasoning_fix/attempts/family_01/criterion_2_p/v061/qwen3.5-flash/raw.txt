import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = colors[u]
                next_color = 3 - current_color
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_color
                        queue.append(v)
                    elif colors[v] == current_color:
                        print("NO")
                        return

    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()

        return

        m = int(next(iterator))

        return

    # Using 1-based indexing for devices

    for _ in range(m):

        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
    
    # Array to store colors of devices
    # 0: uncolored, 1: segment 1, 2: segment 2

            # If device i is uncolored, it belongs to a new connected component.

            # This choice is optimal because i is the first index in this component

                        print("NO")
                        return

    print("YES")
    # Print the assignment sequence
