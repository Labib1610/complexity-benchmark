import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)   # 0 or 1
    seg = [0] * (n + 1)      # final segment 1 or 2
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS for component
        q = deque([start])
        color[start] = 0
        comp = [start]
        minv = start
        ok = True
        while q:
            u = q.popleft()
            if u < minv:
                minv = u
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == cu:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            print("NO")
            return
        # decide orientation so that minv gets segment 1
        if color[minv] == 0:
            for v in comp:
                seg[v] = 1 if color[v] == 0 else 2
        else:
            for v in comp:
                seg[v] = 2 if color[v] == 0 else 1

    print("YES")
    print(' '.join(str(seg[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
