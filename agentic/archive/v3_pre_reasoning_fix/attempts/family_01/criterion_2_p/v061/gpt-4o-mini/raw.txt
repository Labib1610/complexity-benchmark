def network_partition(n, m, edges):
    from collections import deque, defaultdict

    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    colors = [-1] * (n + 1)

    def bfs(start):
        queue = deque([start])
        colors[start] = 0  # Start coloring with 0
        while queue:
            node = queue.popleft()
            current_color = colors[node]
            for neighbor in graph[node]:
                if colors[neighbor] == -1:  # Not colored yet
                    colors[neighbor] = 1 - current_color  # Alternate color
                    queue.append(neighbor)
                elif colors[neighbor] == current_color:
                    return False  # Found a conflict
        return True

    for i in range(1, n + 1):
        if colors[i] == -1:  # Not visited yet
            if not bfs(i):
                print("NO")
                return

    # Convert colors from {0, 1} to {1, 2} for the output
    result = [c + 1 for c in colors[1:]]
    print("YES")
    print(" ".join(map(str, result)))

import sys
input = sys.stdin.read
data = input().strip().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m+1]]
network_partition(n, m, edges)
