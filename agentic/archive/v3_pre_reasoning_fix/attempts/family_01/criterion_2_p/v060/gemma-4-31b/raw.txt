import sys

# Increase recursion depth for deep components in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    
    adj = [[] for _ in range(N + 1)]
    idx = 2
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will be 0 if not visited, 1 or 2 otherwise
    color = [0] * (N + 1)
    
    # To get the lexicographically smallest sequence, for each connected component:
    # We want to assign 1 to the smallest index in the component if possible.
    # However, the constraints are rigid (bipartite coloring).
    # In a connected component, once the color of one node is fixed, 
    # the colors of all others are determined.
    # Let's say the component has nodes {v1, v2, ... vk} where v1 is the smallest.
    # We can either have (v1=1, others determined) or (v1=2, others determined).
    # To make the sequence lexicographically smallest, we must try to set v1 = 1.
    # But wait, that's not entirely correct. Lexicographical smallest means 
    # we want s_1 to be as small as possible, then s_2, etc.
    # Since s_i can only be 1 or 2, we always prefer 1 over 2.
    # For the first node encountered in a component (which will be the smallest index 
    # because we iterate i from 1 to N), we MUST assign it 1 to minimize the sequence.
    
    for i in range(1, N + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # We assign color[i] = 1 to ensure lexicographical smallest result
            stack = [i]
            color[i] = 1
            
            # Use a simple list as a stack for DFS to avoid recursion limits
            visited_in_comp = []
            
            # We use a manual stack to avoid recursion depth issues
            # but since we need to check bipartiteness, standard DFS/BFS is fine.
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[i] for i in range(1, N + 1)))

if __name__ == "__main__":
    solve()
