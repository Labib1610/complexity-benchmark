from collections import deque

sys.setrecursionlimit(200005)

def main():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store colors (shifts) of each worker
    # 0: unvisited, 1: Shift 1, 2: Shift 2
    colors = [0] * (n + 1)
    
    # Iterate over all workers to handle disconnected components
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS for the new component
            # To ensure lexicographically smallest assignment, 
            # we assign Shift 1 to the first node of the component encountered
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_c = colors[u]
                next_c = 3 - current_c

                        colors[v] = next_c
                        queue.append(v)

                        # Conflict detected: graph is not bipartite
                        print("NO")
                        return

    # If we reach here, a valid assignment exists
    print("YES")
    # Output the shifts for workers 1 to n
    # Using join for efficient string conversion and output
    print(" ".join(map(str, colors[1:])))

import sys
from collections import deque

# Increase recursion depth just in case (though not using recursion)
sys.setrecursionlimit(200005)

def main():
    # Read all input at once for efficiency
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # colors[i] stores the shift of worker i (0: unvisited, 1: Shift 1, 2: Shift 2)
    colors = [0] * (n + 1)
    
    # Iterate through each worker to handle all connected components
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Assign Shift 1 to the first unvisited worker in the component
            # to ensure lexicographically smallest assignment
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = colors[u]
                next_color = 3 - current_color
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_color
                        queue.append(v)
                    elif colors[v] == current_color:
                        # Conflict found: graph is not bipartite
                        print("NO")
                        return

    # If no conflicts, output YES and the assignment
    print("YES")
    print(" ".join(map(str, colors[1:])))

if __name__ == '__main__':

Final check:
The problem asks for "lexicographically smallest sequence of shift numbers".
My logic: For each component, the first node encountered (smallest index) determines the coloring. We force it to 1.
Since components are disjoint, choices are independent.
For component $C$, let $min(C)$ be the smallest index in $C$.
We set $s_{min(C)} = 1$.
Suppose we chose $s_{min(C)} = 2$.
Then the sequence at index $min(C)$ would be 2 instead of 1.
Since $1 < 2$, the sequence starting with 1 is lexicographically smaller than the one starting with 2.
Since $min(C)$ is the first index of this component in the global sequence $1 \dots n$, minimizing $s_{min(C)}$ is the highest priority for this component's contribution to the global sequence.
Thus, the greedy choice is optimal.

import sys
from collections import deque

def solve():
    # Read all input from standard input efficiently
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Build adjacency list for the graph
    # Workers are 1-indexed
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # colors[i] stores the shift of worker i
    # 0: unvisited, 1: Shift 1, 2: Shift 2
    colors = [0] * (n + 1)
    
    # Iterate through all workers to handle disconnected components
    # We iterate from 1 to n to ensure we process components in lexicographical order
    # of their first member.
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS for the new connected component
            # To achieve the lexicographically smallest sequence s_1, ..., s_n,
            # we must assign the smallest possible shift to the first worker
            # encountered in the component (which is 'i' since we iterate 1..n).
            # The smallest shift is 1.
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = colors[u]
                next_color = 3 - current_color  # If current is 1, next is 2; if 2, next is 1
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_color
                        queue.append(v)
                    elif colors[v] == current_color:
                        # Conflict detected: adjacent workers have the same shift
                        # This means the graph is not bipartite
                        print("NO")
                        return

    # If we finish without conflicts, a valid assignment exists
    print("YES")
    # Print the shift assignments for workers 1 to n
    print(" ".join(map(str, colors[1:])))

if __name__ == '__main__':
    solve()
