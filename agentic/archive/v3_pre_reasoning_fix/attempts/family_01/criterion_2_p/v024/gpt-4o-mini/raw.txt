def allocate_channels(n, m, rules):
    from collections import defaultdict, deque
    
    graph = defaultdict(list)
    
    for u, v in rules:
        graph[u].append(v)
        graph[v].append(u)
    
    colors = [0] * (n + 1)  # 0 means uncolored, 1 and 2 will be our channels
    
    def bfs(start):
        queue = deque([start])
        colors[start] = 1  # Start coloring with channel 1
        
        while queue:
            node = queue.popleft()
            current_color = colors[node]
            next_color = 2 if current_color == 1 else 1
            
            for neighbor in graph[node]:
                if colors[neighbor] == 0:  # Not colored
                    colors[neighbor] = next_color
                    queue.append(neighbor)
                elif colors[neighbor] == current_color:  # Conflict
                    return False
        return True
    
    for node in range(1, n + 1):
        if colors[node] == 0:  # Not colored yet
            if not bfs(node):
                print("NO")
                return
    
    print("YES")
    print(" ".join(map(str, colors[1:])))

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
rules = [tuple(map(int, line.split())) for line in data[1:m + 1]]

allocate_channels(n, m, rules)
