The problem asks us to assign $n$ personnel members to one of two modes (Mode 1 or Mode 2) such that certain pairs of personnel are not in the same mode. This is a classic graph coloring problem where we need to check if a graph is bipartite (2-colorable). If it is, we need to output the lexicographically smallest coloring.

Let's represent the personnel as nodes in a graph and the interference constraints as edges. An edge between $u$ and $v$ means $u$ and $v$ cannot have the same mode. If we assign mode 1 to $u$, then $v$ must be assigned mode 2, and vice versa. This is exactly the condition for a graph to be bipartite.

```python
import sys
from collections import deque

# Increase recursion depth just in case, though we use iterative BFS
sys.setrecursionlimit(200005)

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break
            
    # Color array: 0 = uncolored, 1 = Mode 1, 2 = Mode 2
    color = [0] * (n + 1)
    
    # We iterate through each node from 1 to n.
    # If a node is uncolored, it belongs to a new connected component.
    # To ensure the lexicographically smallest assignment, we assign the 
    # first node (smallest index) of the component to Mode 1 (1).
    # This forces the rest of the component to be colored consistently.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the connected component
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                next_color = 3 - current_color  # 1 -> 2, 2 -> 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_color
                        queue.append(v)
                    else:
                        # If already colored, check for consistency
                        if color[v] != next_color:
                            print("NO")
                            return

    # If we successfully colored all components without conflict
    print("YES")
    # Print the colors for nodes 1 to n
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```

```python
import sys
from collections import deque

# Set recursion limit just in case, though we use iterative BFS.
sys.setrecursionlimit(200005)

def solve():
    # Read all input from standard input efficiently
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Build the graph using adjacency list
    # Nodes are 1-indexed, so size is n + 1
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break
            
    # Array to store the mode (color) of each personnel
    # 0: unassigned, 1: Mode 1, 2: Mode 2
    color = [0] * (n + 1)
    
    # Iterate through all personnel to handle connected components
    # We want the lexicographically smallest assignment sequence (s_1, s_2, ..., s_n).
    # This means we want s_1 to be 1 if possible, then s_2 to be 1 if possible, etc.
    # For a connected component, the assignment is determined by the color of one node.
    # Specifically, if we fix the color of the node with the smallest index in the component,
    # all other nodes in the component are determined.
    # Since we iterate i from 1 to n, the first uncolored node i we encounter is the 
    # smallest index in its connected component.
    # To minimize the sequence lexicographically, we must set color[i] = 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the new component
            # Assign color 1 to the smallest index node in this component
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_c = color[u]
                # The neighbor must have the opposite color
                target_c = 3 - current_c  # 1->2, 2->1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target_c
                        queue.append(v)
                    else:
                        # If neighbor is already colored, check for conflict
                        if color[v] != target_c:
                            print("NO")
                            return

    # If we finish without conflicts, a valid assignment exists
    print("YES")
    # Print the assignment sequence for nodes 1 to n
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
