```python
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:

    iterator = iter(input_data)
    try:

        m = int(next(iterator))

        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

        if color[i] == 0:
            # This node is the start of a new component (since we iterate 1..n)
            # To minimize lexicographically, we assign 1 to this node
            color[i] = 1
            queue = deque([i])

                c_v = 3 - c_u

                    if color[v] == 0:
                        color[v] = c_v
                        queue.append(v)
                    elif color[v] == c_u:
                        print("NO")
                        return
    
    print("YES")
```

```python
import sys
from collections import deque

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # 1-based indexing for nodes 1 to n
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # Iterate from 1 to n to handle disconnected components
    # This order ensures we process components based on their smallest node index
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the component containing i
            # To get lexicographically smallest sequence, assign 1 to the first node of the component
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                c_u = color[u]
                c_v = 3 - c_u  # Alternate colors 1 and 2
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = c_v
                        queue.append(v)
                    elif color[v] == c_u:
                        # Conflict found: not bipartite
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```

```python
import sys
from collections import deque

# Increase recursion depth just in case, though we use BFS
sys.setrecursionlimit(300000)

def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    
    # Read m edges
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store the color of each node (0: uncolored, 1: color 1, 2: color 2)
    color = [0] * (n + 1)
    
    # Iterate through all nodes to handle disconnected components
    for i in range(1, n + 1):
        if color[i] == 0:
            # If node i is uncolored, it belongs to a new component.
            # Since we iterate from 1 to n, i is the smallest index in this component.
            # To achieve the lexicographically smallest sequence, we assign color 1 to i.
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                next_color = 3 - current_color  # Toggle between 1 and 2
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_color
                        queue.append(v)
                    elif color[v] == current_color:
                        # Conflict detected: graph is not bipartite
                        print("NO")
                        return

    # If valid assignment found
    print("YES")
    # Print the colors for nodes 1 to n
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
