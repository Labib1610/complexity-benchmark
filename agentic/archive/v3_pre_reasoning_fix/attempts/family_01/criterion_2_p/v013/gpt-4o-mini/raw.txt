def main():
    import sys
    from collections import defaultdict, deque

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    graph = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)

    color = [0] * (n + 1)  # 0 means uncolored, 1 and 2 will be the two colors
    is_bipartite = True

    for start in range(1, n + 1):
        if color[start] == 0:  # not yet colored
            queue = deque([start])
            color[start] = 1  # start coloring with 1

            while queue:
                node = queue.popleft()
                current_color = color[node]

                for neighbor in graph[node]:
                    if color[neighbor] == 0:  # not colored
                        color[neighbor] = 2 if current_color == 1 else 1
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        is_bipartite = False
                        break
            
            if not is_bipartite:
                break

    if not is_bipartite:
        print("NO")
    else:
        print("YES")
        # Since we need to output the lexicographically smallest configuration
        print(' '.join(str(color[i]) if color[i] != 0 else '1' for i in range(1, n + 1)))

main()
