import sys

# Increase recursion depth for deep components
sys.setrecursionlimit(300000)

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            # Self-loop means no valid coloring possible
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # color[i] = 0 (unvisited), 1 or 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, for each connected component,
    # we check if it's bipartite. If it is, we have two choices for the coloring:
    # Choice A: start with 1 for the smallest node in the component.
    # Choice B: start with 2 for the smallest node in the component.
    # However, the lexicographical requirement is global across all nodes.
    # For each component, we must decide whether the smallest node in that 
    # component should be 1 or 2 to minimize the sequence (s1, s2, ..., sn).
    # Since components are independent, for each component, we try to set the 
    # smallest index to 1. If that's possible, we check if it forces any later 
    # indices to be larger than they would be if the smallest index were 2.
    # Actually, for a bipartite component, once the color of one node is fixed, 
    # all others are fixed. Let the smallest node be 'root'. 
    # Option 1: color[root] = 1. This determines all other colors in the component.
    # Option 2: color[root] = 2. This flips all colors (1->2, 2->1).
    # To be lexicographically smallest, we compare the resulting sequences.
    # The first index i in the component where the two options differ will 
    # determine the winner. Since the smallest index is 'root', and Option 1 
    # sets s_{root} = 1 and Option 2 sets s_{root} = 2, Option 1 is always 
    # lexicographically smaller because 1 < 2.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i. i is the smallest index in this component.
            # To minimize lexicographically, we MUST set color[i] = 1.
            stack = [i]
            color[i] = 1
            component_nodes = []
            
            # Use iterative DFS to avoid recursion depth issues
            visited_in_comp = {i}
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component_nodes.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
