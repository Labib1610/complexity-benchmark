import sys

# Increase recursion depth for deep components
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will be 0 (unvisited), 1, or 2
    color = [0] * (n + 1)
    
    # To find the lexicographically smallest sequence s_1, s_2, ..., s_n,
    # we process workers in order from 1 to n.
    # For each connected component, the choice of color for the first node 
    # encountered determines the colors of all others in that component.
    # Since we want lexicographically smallest, for the first node 'u' of a 
    # component, we should try assigning it color 1 first.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color the component
            # We use a stack for iterative DFS to avoid recursion limits
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # We need to track the nodes in the component to potentially flip them
            # However, the greedy choice for the smallest index in the component
            # is always to set it to 1 if possible.
            # Since this is a bipartite graph, if we set the root to 1,
            # the assignment for the rest is fixed.
            # If we set the root to 2, the assignment for the rest is flipped.
            # To get lexicographically smallest, we must check if setting root to 1
            # is valid (it always is if the component is bipartite).
            # Then we check if setting root to 1 results in a smaller sequence 
            # than setting it to 2. 
            # Actually, if we process nodes in order 1...n, the first node 'i' 
            # we encounter in a component is the most significant digit.
            # Setting color[i] = 1 is always better than color[i] = 2.
            
            q = [i]
            visited_idx = 0
            while visited_idx < len(q):
                u = q[visited_idx]
                visited_idx += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
