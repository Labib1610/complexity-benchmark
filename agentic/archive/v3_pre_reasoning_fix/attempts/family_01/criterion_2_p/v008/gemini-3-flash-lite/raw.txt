import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We want lexicographically smallest assignment.
    # This means for each connected component, we try assigning 
    # the smallest node (first encountered) to 1.
    # Since we must be lexicographically smallest overall, we iterate 1 to n.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Component not yet colored.
            # To be lexicographically smallest, we must decide whether the
            # first node in this component should be 1 or 2.
            # If we set assignment[i] = 1, it's always better than 2.
            # Is it always possible? Yes, if a valid 2-coloring exists.
            
            stack = [(i, 1)]
            assignment[i] = 1
            component = [i]
            
            # Use stack for DFS to avoid recursion limit and to collect component
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
