import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will be 0 (unvisited), 1, or 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, 
    # we must iterate through nodes 1 to n.
    # For each connected component, we want the smallest possible 
    # value for the smallest index in that component.
    # Since we want the smallest sequence, for each component,
    # we should try to assign 1 to the smallest index in that component.
    # Then the coloring of the rest of the component is fixed.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color the component
            # To ensure lexicographical smallest, we start by assigning 1 to i
            stack = [i]
            color[i] = 1
            
            # We use a stack for DFS to avoid recursion depth issues, 
            # though we already increased the limit.
            component = []
            
            # Using a queue for BFS is also fine. Let's use a simple list as a stack.
            ptr = 0
            q = [i]
            while ptr < len(q):
                u = q[ptr]
                ptr += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # The logic above assigns 1 to the smallest index of every component.
    # Since the components are disjoint, this greedily minimizes the sequence
    # from left to right.
    
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
