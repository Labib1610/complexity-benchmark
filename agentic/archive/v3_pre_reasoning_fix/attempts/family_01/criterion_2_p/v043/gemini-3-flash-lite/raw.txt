import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We want the lexicographically smallest sequence (s1, s2, ..., sn).
    # This means we prefer s_i = 1 whenever possible.
    # We iterate i from 1 to n. If i is not yet assigned, we assign it 1,
    # then propagate the constraints using BFS/DFS.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning 1 to i
            stack = [(i, 1)]
            assignment[i] = 1
            component = [i]
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # If 1 failed, try 2
                for node in component:
                    assignment[node] = 0
                
                stack = [(i, 2)]
                assignment[i] = 2
                component = [i]
                possible = True
                idx_comp = 0
                while idx_comp < len(component):
                    u = component[idx_comp]
                    idx_comp += 1
                    
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - assignment[u]
                            component.append(v)
                        elif assignment[v] == assignment[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
