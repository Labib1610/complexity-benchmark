import sys

# Increase recursion depth for deep DFS in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Read input using fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            # A device cannot be different from itself
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # color[i] will store the frequency band (1 or 2)
    # 0 means unvisited
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, 
    # for each connected component, we should try to assign 
    # the smallest possible value to the smallest index in that component.
    # However, the bipartition is fixed once the first node is colored.
    # Let's say a component has two possible colorings: (C1, C2) and (C2, C1).
    # We compare the resulting sequences and pick the lexicographically smaller one.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to find the connected component
            component = []
            stack = [i]
            color[i] = 1
            
            # We use a stack for iterative DFS to avoid recursion limits and be faster
            visited_in_comp = []
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                visited_in_comp.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # There are only two choices: the current coloring or the flipped one.
            # To make it lexicographically smallest, we look at the smallest index
            # in the component. If that index is not 1, we flip it to 1.
            # Actually, since we process i from 1 to n, i is always the smallest index.
            # If color[i] is already 1, we keep it. If it were 2, we'd flip.
            # But we initialized color[i] = 1, so it's already optimal.
            # Wait, that's not entirely true. We need to compare the sequences.
            # Let's re-evaluate: for the first node 'i' in the component, 
            # assigning it 1 is always better than assigning it 2, 
            # because it's the most significant position in the lexicographical order.
            # Since we set color[i] = 1, we have already made the best choice for the 
            # first element of the component.
            
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
