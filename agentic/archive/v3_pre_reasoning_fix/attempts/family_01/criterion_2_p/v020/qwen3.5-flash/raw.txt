from collections import deque

# Set recursion limit just in case, though iterative BFS is used.
sys.setrecursionlimit(200005)

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Initialize adjacency list for the graph
    # Nodes are 1-indexed
    adj = [[] for _ in range(n + 1)]
    
    # Read m edges
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store the bus line assignment (color) for each hub
    # 0 means unvisited/uncolored, 1 or 2 means assigned
    colors = [0] * (n + 1)
    
    # Iterate through each hub to handle disconnected components
    # We iterate from 1 to n to ensure we process components starting from the smallest index
    # which is crucial for lexicographically smallest assignment.
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start a BFS for the new component
            # To achieve lexicographically smallest sequence, we assign color 1 to the smallest index node
            # in the component (which is 'i' since we iterate in increasing order and this node is uncolored).
            colors[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = colors[u]
                next_color = 3 - current_color  # Toggle between 1 and 2
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_color
                        queue.append(v)
                    elif colors[v] == current_color:
                        # Conflict detected: adjacent nodes have the same color
                        # This means the graph is not bipartite
                        sys.stdout.write("NO\n")
                        return

    # If we finish the loop without conflicts, a valid assignment exists
    sys.stdout.write("YES\n")
    # Output the assignment sequence for nodes 1 to n
    # Using sys.stdout.write for faster output
    sys.stdout.write(" ".join(map(str, colors[1:])) + "\n")

if __name__ == '__main__':
    solve()
