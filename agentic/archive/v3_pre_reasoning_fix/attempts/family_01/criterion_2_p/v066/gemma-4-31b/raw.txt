import sys

# Increase recursion depth for deep DFS trees in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            # Self-loop makes it impossible to have different bands
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # color[i] = 0: unvisited, 1: band 1, 2: band 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we process nodes 1 to n.
    # For each connected component, the first node we encounter (the smallest index)
    # should be assigned band 1 if possible.
    # Once the first node of a component is fixed to 1, the rest of the component
    # is uniquely determined.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color the component
            # We use a stack for iterative DFS to avoid recursion limits and for speed
            stack = [(i, 1)]
            color[i] = 1
            
            # To ensure we don't use too much memory or time, we process the component
            component = []
            
            # Standard iterative DFS
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
    # If we reached here, a valid 2-coloring exists.
    # The logic above already assigns the smallest possible band (1) to the
    # smallest index of every connected component, which naturally leads to
    # the lexicographically smallest sequence for a bipartite graph.
    
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
