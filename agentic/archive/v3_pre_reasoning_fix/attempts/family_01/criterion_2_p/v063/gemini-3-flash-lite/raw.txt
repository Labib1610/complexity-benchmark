import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component not colored yet. To get lexicographically smallest,
            # we try to assign 1 to the smallest node in this component.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use iterative DFS to find all nodes in the connected component
            # and verify bipartiteness
            q = [i]
            comp_nodes = [i]
            idx_q = 0
            while idx_q < len(q):
                u = q[idx_q]
                idx_q += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Within this component, we have two possible colorings:
            # 1. Start with color[i] = 1 (current)
            # 2. Start with color[i] = 2 (invert all colors)
            # We want the one that makes the smallest index i have 1.
            # Since we iterate i from 1 to n, the first node in the component
            # is the smallest. We already set it to 1.
            # This is locally optimal for the lexicographical requirement.
            
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
