```python
import sys
from collections import deque

# Increase recursion depth just in case, though we plan to use BFS
sys.setrecursionlimit(200005)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # We need to determine if it's bipartite and find the lexicographically smallest coloring.
    # The greedy strategy: for each connected component, the node with the smallest index
    # determines the color of the whole component. To minimize lexicographically,
    # the smallest index node in a component must be assigned color 1.
    
    # Iterate through all nodes. If a node is uncolored, it starts a new component.
    # Assign it color 1 and propagate.
    
    for i in range(1, n + 1):
        if color[i] != 0:
            continue
        
        # Start BFS for the component containing i
        color[i] = 1
        queue = deque([i])
        
        while queue:
            u = queue.popleft()
            c_u = color[u]
            c_next = 3 - c_u # If c_u is 1, c_next is 2. If c_u is 2, c_next is 1.
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = c_next
                    queue.append(v)
                elif color[v] == c_u:
                    print("NO")
                    return

    print("YES")
    # Print colors for 1 to n
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```

```python
import sys
from collections import deque

# Set recursion limit just in case, though we use iterative BFS.
sys.setrecursionlimit(200005)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    # 1-based indexing for convenience
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Color array: 0 = uncolored, 1 = Channel 1, 2 = Channel 2
    color = [0] * (n + 1)
    
    # Iterate through each node to handle connected components
    # We want lexicographically smallest assignment, so for each component,
    # the node with the smallest index should be assigned Channel 1 (value 1).
    # Since we iterate i from 1 to n, the first uncolored node we encounter
    # is the smallest index in its component.
    
    for i in range(1, n + 1):
        if color[i] != 0:
            continue
        
        # Start BFS for the component containing i
        # Assign i to Channel 1
        color[i] = 1
        queue = deque([i])
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            # The other channel is 3 - current_color (since 1->2, 2->1)
            next_color = 3 - current_color
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = next_color
                    queue.append(v)
                elif color[v] == current_color:
                    # Conflict detected: graph is not bipartite
                    print("NO")
                    return

    # If we successfully colored all nodes
    print("YES")
    # Output the colors for nodes 1 to n separated by spaces
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
