def sensor_channel_allocation(n, m, edges):
    from collections import defaultdict, deque
    
    graph = defaultdict(list)
    
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    colors = [0] * (n + 1)  # 0 means uncolored, 1 and 2 are the two channels
    
    def bfs(start):
        queue = deque([start])
        colors[start] = 1  # Start coloring with channel 1
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if colors[neighbor] == 0:  # Uncolored
                    colors[neighbor] = 3 - colors[node]  # Alternate colors
                    queue.append(neighbor)
                elif colors[neighbor] == colors[node]:  # Conflict
                    return False
        return True
    
    for sensor in range(1, n + 1):
        if colors[sensor] == 0:  # Not colored yet
            if not bfs(sensor):
                print("NO")
                return
    
    print("YES")
    print(" ".join(map(str, colors[1:])))
    
import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]
sensor_channel_allocation(n, m, edges)
