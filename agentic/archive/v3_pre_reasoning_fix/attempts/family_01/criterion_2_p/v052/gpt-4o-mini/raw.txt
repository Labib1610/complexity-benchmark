def dual_channel_allocation(n, m, restrictions):
    from collections import deque
    
    # Create an adjacency list for the graph
    graph = [[] for _ in range(n + 1)]
    for u, v in restrictions:
        graph[u].append(v)
        graph[v].append(u)
    
    # To store the channel assignment
    channel = [0] * (n + 1)  # 0 means unassigned, 1 and 2 will be the channels

    def bfs(start):
        queue = deque([start])
        channel[start] = 1  # Start with channel 1
        
        while queue:
            node = queue.popleft()
            current_channel = channel[node]
            next_channel = 2 if current_channel == 1 else 1
            
            for neighbor in graph[node]:
                if channel[neighbor] == 0:  # Not assigned yet
                    channel[neighbor] = next_channel
                    queue.append(neighbor)
                elif channel[neighbor] == current_channel:  # Conflict found
                    return False
        return True

    for i in range(1, n + 1):
        if channel[i] == 0:  # Not visited yet
            if not bfs(i):
                print("NO")
                return

    print("YES")
    print(" ".join(map(str, channel[1:])))

import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
restrictions = [tuple(map(int, line.split())) for line in data[1:m + 1]]

dual_channel_allocation(n, m, restrictions)
