def shift_assignment(n, m, conflicts):
    from collections import defaultdict, deque
    
    graph = defaultdict(list)
    
    for u, v in conflicts:
        graph[u].append(v)
        graph[v].append(u)
    
    shift = [0] * (n + 1)

    def bfs(start):
        queue = deque([start])
        shift[start] = 1
        
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if shift[neighbor] == 0:  # Not assigned yet
                    shift[neighbor] = 3 - shift[node]  # Assign opposite shift
                    queue.append(neighbor)
                elif shift[neighbor] == shift[node]:  # Conflict detected
                    return False
        return True

    for technician in range(1, n + 1):
        if shift[technician] == 0:  # Not visited
            if not bfs(technician):
                print("NO")
                return

    print("YES")
    print(' '.join(map(str, shift[1:])))
    
import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
conflicts = [tuple(map(int, line.split())) for line in data[1:m+1]]
shift_assignment(n, m, conflicts)
