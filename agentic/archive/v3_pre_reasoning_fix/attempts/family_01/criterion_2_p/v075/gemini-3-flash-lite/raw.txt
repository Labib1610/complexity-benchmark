import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # We want the lexicographically smallest assignment.
    # We iterate from 1 to n. If a node is uncolored, we try assigning it '1'.
    # If that leads to a contradiction, we try '2'.
    # Because we want lexicographical smallest, we greedily pick 1 whenever possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack to find all nodes in the connected component
            # and verify bipartiteness.
            comp_nodes = []
            q = [i]
            comp_nodes.append(i)
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The component is bipartite. We have two possible colorings:
            # 1. Keep as is (color[i] = 1)
            # 2. Flip all colors (color[i] = 2)
            # Since we want lexicographically smallest, we check which one
            # gives a smaller value at the first node i of the component.
            # Wait, that's not quite right. We need the smallest sequence globally.
            # Actually, for each component, we can either use the current coloring
            # or the flipped one. We pick the one that makes the first technician
            # in this component (which is i) 1 if possible.
            if color[i] == 2:
                for node in comp_nodes:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
