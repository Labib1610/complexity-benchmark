import sys

# Increase recursion depth for deep components
sys.setrecursionlimit(300000)

def solve():
    # Using fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # A person cannot be incompatible with themselves in a 2-coloring problem
            # unless the problem allows it, but usually it means NO.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    # color[i] = 0: unvisited, 1: color 1, 2: color 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence (s1, s2, ..., sn),
    # we must process staff members in order from 1 to n.
    # For each connected component, the first member we encounter (the smallest index)
    # should be assigned color 1 if possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS to color the component. 
            # To minimize lexicographically, the smallest node in the component
            # should be assigned 1.
            stack = [i]
            color[i] = 1
            
            # We use a stack for DFS or queue for BFS. 
            # Since we just need to check bipartiteness and assign colors, 
            # any traversal works.
            visited_nodes = []
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                visited_nodes.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return

    # The logic above assigns the smallest index in each component to 1.
    # However, this doesn't necessarily guarantee the lexicographically smallest 
    # global sequence because the colors of other nodes in the component are 
    # forced once the first node is colored.
    # Let's re-verify: if we have a component, we have two choices:
    # Choice A: node_min = 1, others follow.
    # Choice B: node_min = 2, others follow.
    # Since we want the lexicographically smallest sequence, we definitely want 
    # s_{node_min} to be 1. This is because s_{node_min} is the most significant 
    # position in the sequence for that component.
    
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
