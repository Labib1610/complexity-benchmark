The problem asks us to assign each of $n$ individuals to one of two operating periods, 1 or 2. We are given $m$ restrictions, where each restriction is a pair of individuals $(u, v)$ that cannot be in the same period. We need to determine if a valid assignment exists. This is a classic graph coloring problem. Specifically, it's about determining if a graph is bipartite. The individuals are nodes, and the restrictions are edges. import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            q = deque([i])
            while q:
                u = q.popleft()
                c = color[u]
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        q.append(v)
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
