import sys
from collections import deque

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # Iterate through all nodes to handle disconnected components
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the component
            # We want lexicographically smallest, so we set color[i] = 1
            # This is the smallest possible value for the first node in this component.
            
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                
                current_color = color[u]
                next_color = 3 - current_color  # 1 -> 2, 2 -> 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_color
                        queue.append(v)
                    elif color[v] == current_color:
                        # Conflict found: graph is not bipartite
                        print("NO")
                        return
                        
    # If we successfully colored all components
    print("YES")
    # Print colors from 1 to n separated by spaces
    print(*(color[1:]))

if __name__ == '__main__':
    solve()

        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store team assignment (0: unvisited, 1: Team 1, 2: Team 2)
    color = [0] * (n + 1)
    
    # Iterate through all nodes 1 to n
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the connected component containing i
            # Since we want the lexicographically smallest sequence,
            # we must assign the smallest index node in the component to Team 1 if possible.
            # Since i is the smallest index in this component (as we iterate 1..n),
            # we set color[i] = 1.
            
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                
                # The required color for neighbors is the opposite of u's color

                        color[v] = next_c
                        queue.append(v)
                    elif color[v] == color[u]:
                        # If a neighbor has the same color, the graph is not bipartite
                        print("NO")
                        return
    
    # If we complete the loop without conflicts, output YES and the assignment
    print("YES")
    print(" ".join(map(str, color[1:])))
