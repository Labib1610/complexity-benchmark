```python
import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list and adjacency matrix for fast lookup
    # Since n <= 2000, a bitset or adjacency matrix is feasible.
    # In Python, we can use a list of sets or a bitset (integer).
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2

    # To find the lexicographically smallest (a, b, c) where a < b < c:
    # We iterate through a from 1 to n, then b from a+1 to n, then c from b+1 to n.
    # To optimize, we only iterate through neighbors of a that are > a.
    
    # Pre-sort the neighbors for each node to ensure we check in increasing order.
    sorted_adj = [sorted([v for v in adj[u] if v > u]) for u in range(n + 1)]

    for a in range(1, n - 1):
        neighbors_a = sorted_adj[a]
        # We need to find two neighbors b, c of a such that b < c and b is connected to c.
        # Since neighbors_a is sorted, we can iterate through them.
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need c to be a neighbor of a AND a neighbor of b, and c > b.
            # We can iterate through the neighbors of a that are greater than b.
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected.
                # Using the set for O(1) average lookup.
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
