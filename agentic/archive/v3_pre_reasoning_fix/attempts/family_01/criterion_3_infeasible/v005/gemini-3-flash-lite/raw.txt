```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Iterate c from b+1 to n
    # Check if a-b, a-c, b-c are all in adj
    
    # Optimization: Sort neighbors to speed up the process
    # But since n is 2000, O(n^3) is too slow.
    # However, we only need to check if b and c are in adj[a]
    # and if c is in adj[b].
    
    # Iterate through all possible 'a'
    for a in range(1, n + 1):
        # Get all neighbors of a that are greater than a
        neighbors_a = [x for x in adj[a] if x > a]
        neighbors_a.sort()
        
        # Iterate through all possible 'b' in neighbors of 'a'
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # Find common neighbors of a and b that are > b
            # These are candidates for c
            # This is effectively finding triangles in the graph
            common = adj[a].intersection(adj[b])
            candidates_c = [x for x in common if x > b]
            
            if candidates_c:
                candidates_c.sort()
                print(f"{a} {b} {candidates_c[0]}")
                return

if __name__ == '__main__':
    solve()
```
