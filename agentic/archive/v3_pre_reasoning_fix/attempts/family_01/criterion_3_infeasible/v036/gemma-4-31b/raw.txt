import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        ptr += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # We need the lexicographically smallest trio (a, b, c) with a < b < c.
    # This means we want to minimize a, then b, then c.
    # Since n is up to 2000, an O(n^3) approach is too slow.
    # However, we can iterate through all edges (u, v) and find common neighbors.
    # To ensure lexicographical smallest, we should iterate candidates carefully.
    
    # Sort nodes to process in increasing order
    # We want smallest a, then smallest b, then smallest c.
    # Let's iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look for pairs (b, c) such that a < b < c and (a,b), (a,c), (b,c) are edges.
    
    # Optimization: Only consider nodes connected to 'a'.
    for a in range(1, n + 1):
        # neighbors of a that are > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # Now we need to find the smallest b, c in neighbors_a such that (b, c) is an edge.
        # Since we want the smallest b, then smallest c, we iterate through neighbors_a.
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
