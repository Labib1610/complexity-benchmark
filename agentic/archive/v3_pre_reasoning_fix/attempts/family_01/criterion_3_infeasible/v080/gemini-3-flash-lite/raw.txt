import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Represent graph as an adjacency list of bitsets (represented as integers)
    # This allows for efficient intersection checks.
    adj = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        
        # Ensure u < v for consistent storage
        if u > v:
            u, v = v, u
        
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        adj_list[u].append(v)
        adj_list[v].append(u)

    # We need the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate a from 1 to n-2.
    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_a = [v for v in adj_list[a] if v > a]
        neighbors_a.sort()
        
        # Iterate b in neighbors of a
        for b in neighbors_a:
            # We need c > b such that c is neighbor of a AND c is neighbor of b
            # Intersection of bits in adj[a] and adj[b]
            common = adj[a] & adj[b]
            
            # We only care about c > b
            # Mask to keep only bits > b
            mask = ~((1 << (b + 1)) - 1)
            candidates = common & mask
            
            if candidates:
                # Find the smallest c
                c = (candidates & -candidates).bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
