```python
import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) edge lookups
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Convert sets to sorted lists for deterministic iteration
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # We want lexicographically smallest (a, b, c) with a < b < c.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Iterate b from a+1 to n-1
        # To optimize, b must be connected to a
        neighbors_a = adj_list[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need c > b such that c is connected to a AND c is connected to b
            # Intersection of neighbors of a and neighbors of b
            # We only care about c > b
            set_b = adj[b]
            for c in neighbors_a:
                if c > b and c in set_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
