import sys

def solve():
    # Read n and m from the first line of input
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list representation
    # Using sets for O(1) average time complexity lookups
    adj = [set() for _ in range(n + 1)]
    
    # Store edges to iterate over them efficiently
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # Ensure u < v for consistency
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate through possible values of 'a' from 1 to n.
    # For a fixed 'a', we look for the smallest 'b' > a such that (a, b) is an edge.
    # Then we look for the smallest 'c' > b such that (a, c) and (b, c) are edges.
    
    # Pre-sort the adjacency lists to iterate in increasing order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b are neighbors of a that are greater than a
        for b in sorted_adj[a]:
            if b <= a:
                continue
            
            # Candidates for c must be neighbors of both a and b, and greater than b
            # To optimize, we iterate through the smaller adjacency list of a and b
            # However, for the lexicographically smallest, we need to check c in increasing order.
            # The neighbors of a that are > b are already sorted in sorted_adj[a].
            
            # Find the first neighbor of 'a' that is > b and also a neighbor of 'b'
            # Since sorted_adj[a] is sorted, the first one we find will be the smallest c.
            for c in sorted_adj[a]:
                if c <= b:
                    continue
                if c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
