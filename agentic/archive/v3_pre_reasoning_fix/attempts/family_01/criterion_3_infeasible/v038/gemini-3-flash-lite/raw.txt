```python
import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triplet
    # We want to iterate through all possible 'a', then 'b', then 'c'
    # Or more efficiently: iterate through all edges (u, v) as (a, b), 
    # then check if there exists a 'c' such that (a, c) and (b, c) exist.
    
    # To ensure lexicographical order:
    # Iterate a from 1 to n
    #   Iterate b in sorted(adj[a]) where b > a
    #     Iterate c in sorted(adj[a] & adj[b]) where c > b
    #       Return the first one found
    
    for a in range(1, n + 1):
        neighbors_a = sorted([v for v in adj[a] if v > a])
        for b in neighbors_a:
            # Find common neighbors c such that c > b
            # Intersection of adj[a] and adj[b]
            # Since n is up to 2000, we can use bitsets or set intersection
            # For efficiency, iterate over the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
                other = adj[b]
            else:
                candidates = adj[b]
                other = adj[a]
            
            common = sorted([c for c in candidates if c > b and c in other])
            if common:
                print(f"{a} {b} {common[0]}")
                return

if __name__ == '__main__':
    solve()
```
