def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [bytearray(n+1) for _ in range(n+1)]
    neigh = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        if adj[u][v]: continue  # ignore duplicate
        adj[u][v] = 1
        adj[v][u] = 1
        neigh[u].append(v)
        neigh[v].append(u)
    for a in range(1, n+1):
        neigh[a].sort()
        # we can also filter >a but will check inside loops
    for a in range(1, n+1):
        na = neigh[a]
        for b in na:
            if b <= a: continue
            # for c in na: iterate >b
            for c in na:
                if c <= b: continue
                if adj[b][c]:
                    sys.stdout.write(f"{a} {b} {c}")
                    return
