```python
import sys

def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # adj[i] stores the bitmask of neighbors for vertex i (0-indexed)
    adj = [0] * n
    
    # adj_list[i] stores the list of neighbors for vertex i (0-indexed)
    adj_list = [[] for _ in range(n)]

    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        
        adj_list[u].append(v)
        adj_list[v].append(u)

    # Sort adjacency lists to ensure we iterate neighbors in increasing order
    for i in range(n):
        adj_list[i].sort()

    # Iterate through each vertex a as the first member of the triple
    for a in range(n):
        # Iterate over neighbors b of a
        for b in adj_list[a]:
            # We need a < b < c
            if b <= a:
                continue
            
            # We need to find c > b such that c is connected to both a and b.
            common = adj[a] & adj[b]
            
            # We only care about vertices c > b.
            # In 0-indexed terms, c_idx > b.
            # Mask out bits <= b.
            mask = ~((1 << (b + 1)) - 1)
            common &= mask
            
            if common:
                # Find the smallest set bit in common
                lowbit = common & -common
                c_idx = lowbit.bit_length() - 1
                
                # Output 1-based indices
                print(f"{a + 1} {b + 1} {c_idx + 1}")
                return
