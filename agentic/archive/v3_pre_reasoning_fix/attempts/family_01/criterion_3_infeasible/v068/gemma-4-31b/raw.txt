import sys

def solve():
    # Read n and m from the first line
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency lists to store the graph
    # Since we need the lexicographically smallest (a, b, c) with a < b < c,
    # we iterate a from 1 to n, then b from a+1 to n, and check for c.
    adj = [[] for _ in range(n + 1)]
    # Use a bitset-like approach or a set for fast lookup of edges
    # Given N=2000, a bitset (represented as an integer in Python) 
    # is very efficient for intersection operations.
    bit_adj = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # Ensure u < v for consistent storage
        if u > v:
            u, v = v, u
        adj[u].append(v)
        bit_adj[u] |= (1 << v)
        bit_adj[v] |= (1 << u)

    # Sort adjacency lists to process b in increasing order
    for i in range(1, n + 1):
        adj[i].sort()

    # We look for the smallest a, then smallest b, then smallest c.
    for a in range(1, n + 1):
        # For a fixed a, we only consider neighbors of a that are > a.
        neighbors_of_a = [v for v in adj[a] if v > a]
        
        # To find the smallest b, we iterate through neighbors of a in increasing order.
        for b in neighbors_of_a:
            # Now we need to find the smallest c > b such that c is connected to both a and b.
            # The set of candidates for c are neighbors of a that are > b.
            # These are already in neighbors_of_a, we just need to filter.
            
            # Instead of iterating all c, we can intersect the bitsets of a and b.
            # common_neighbors = bit_adj[a] & bit_adj[b]
            # However, we only care about c > b.
            
            # Optimization: Only check neighbors of b that are also neighbors of a and > b.
            # Since we want the smallest c, we can iterate through the sorted adj[b]
            # and check if the element is > b and present in bit_adj[a].
            for c in adj[b]:
                if c > b:
                    if (bit_adj[a] >> c) & 1:
                        print(f"{a} {b} {c}")
                        return

if __name__ == "__main__":
    solve()
