```python
import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        ptr += 2
        # Store edges such that u < v
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))

    # To find the lexicographically smallest triple (a, b, c) where a < b < c:
    # We want the smallest a, then smallest b, then smallest c.
    # We can iterate through all possible 'a' from 1 to n.
    # For a fixed 'a', we look at all its neighbors 'b' where b > a.
    # For each such 'b', we check if they share a common neighbor 'c' where c > b.
    
    # To optimize, we only iterate through nodes that have at least 2 neighbors.
    # Since n is up to 2000, an O(n * m / word_size) or O(n^3 / word_size) approach 
    # using bitsets is efficient. Python's large integers act as bitsets.
    
    bit_adj = [0] * (n + 1)
    for u, v in edges:
        bit_adj[u] |= (1 << v)
        bit_adj[v] |= (1 << u)

    for a in range(1, n + 1):
        # We only care about neighbors of a that are greater than a
        neighbors_a = []
        # Extract bits set in bit_adj[a] that are > a
        # This is effectively iterating through sorted neighbors
        temp_bits = bit_adj[a] >> (a + 1)
        curr = a + 1
        while temp_bits:
            if temp_bits & 1:
                neighbors_a.append(curr)
            temp_bits >>= 1
            curr += 1
        
        # Now we have sorted neighbors b > a.
        # We look for the smallest b, then the smallest c > b.
        for b in neighbors_a:
            # Find common neighbors of a and b that are > b
            # common = (adj[a] AND adj[b]) AND (bits > b)
            common = bit_adj[a] & bit_adj[b]
            # Mask to keep only bits > b
            mask = ~((1 << (b + 1)) - 1)
            candidates = common & mask
            
            if candidates:
                # Find the lowest set bit in candidates
                # candidates.bit_length() - 1 is the highest, 
                # (candidates & -candidates).bit_length() - 1 is the lowest.
                c = (candidates & -candidates).bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
