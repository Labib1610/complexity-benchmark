import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
    
    edges.sort()
    
    # We need the lexicographically smallest triple (a, b, c) with a < b < c.
    # This means we want to minimize a, then b, then c.
    # Iterate through each edge (u, v) as the first two elements of the triple.
    # Since we need (a, b, c) with a < b < c, for each edge (u, v),
    # we look for a common neighbor w such that w > v (to maintain order).
    # To ensure lexicographically smallest, we iterate through edges (a, b)
    # in sorted order and check neighbors.
    
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we look for c > b such that c is in adj[a] and adj[b]
            # To get smallest c, we intersect sorted lists or just iterate
            # through the smaller neighbor list.
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            possible_c = []
            for c in candidates:
                if c > b and c in adj[a] and c in adj[b]:
                    possible_c.append(c)
            
            if possible_c:
                print(f"{a} {b} {min(possible_c)}")
                return

solve()
