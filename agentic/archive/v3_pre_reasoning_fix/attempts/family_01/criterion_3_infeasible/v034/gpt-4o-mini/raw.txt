```python
import sys
from collections import defaultdict

input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
reactive_pairs = defaultdict(set)

for i in range(1, m + 1):
    u, v = map(int, data[i].split())
    reactive_pairs[u].add(v)
    reactive_pairs[v].add(u)

for a in range(1, n + 1):
    for b in reactive_pairs[a]:
        if b > a:  # Ensure a < b
            for c in reactive_pairs[b]:
                if c > b and c in reactive_pairs[a]:  # Ensure b < c and a, b, c are all mutually reactive
                    print(a, b, c)
                    sys.exit()
```
