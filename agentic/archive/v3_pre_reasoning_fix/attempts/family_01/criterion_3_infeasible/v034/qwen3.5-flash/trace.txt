We need to find the lexicographically smallest triangle $(a, b, c)$ with $a < b < c$. This means we want to minimize $a$, then $b$, then $c$.

A naive approach to finding a triangle is to iterate over all possible triplets $(i, j, k)$ and check if $(i, j)$, $(j, k)$, and $(i, k)$ are edges. This would take $O(n^3)$ time, which is too slow given $n=2000$. $2000^3 = 8 \cdot 10^9$, way beyond the limit.

```python
import sys

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency mask for fast intersection
    # Vertices are 1-indexed. We will use bit positions 1 to n.
    # bit k corresponds to vertex k.
    adj_mask = [0] * (n + 1)
    
    # Adjacency list to iterate neighbors in order
    adj_list = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        
        # Set bits in masks
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
        
        adj_list[u].append(v)
        adj_list[v].append(u)
    
    # Sort adjacency lists to ensure iteration order
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate to find the lexicographically smallest triangle
    # Order: smallest a, then smallest b, then smallest c
    for a in range(1, n + 1):
        # We only care about neighbors b > a
        # Since adj_list[a] is sorted, we can iterate and skip small ones
        # Or use bisect to find start index. Iterating is O(deg(a)).
        for b in adj_list[a]:
            if b <= a:
                continue
            
            # We need to find smallest c > b such that (a, c) and (b, c) are edges.
            # This corresponds to finding the smallest bit set in (adj_mask[a] & adj_mask[b])
            # that is > b.
            
            common = adj_mask[a] & adj_mask[b]
            
            # We want to ignore bits <= b.
            # Create a mask with bits 0..b set to 1.
            # Then invert it to get bits 0..b as 0, others as 1.
            # Note: 1 << (b + 1) creates a 1 at position b+1.
            # Subtracting 1 gives 1s at positions 0..b.
            mask_le_b = (1 << (b + 1)) - 1
            common_filtered = common & ~mask_le_b
            
            if common_filtered:
                # Find the lowest set bit
                # low_bit = common_filtered & -common_filtered
                # c = low_bit.bit_length() - 1
                # This gives the index of the bit.
                
                # Optimization: bit_length is fast.
                c = (common_filtered & -common_filtered).bit_length() - 1
                
                # We found the smallest c > b in the intersection.
                # Since we iterate a then b, and find smallest c, this is the answer.
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
```
