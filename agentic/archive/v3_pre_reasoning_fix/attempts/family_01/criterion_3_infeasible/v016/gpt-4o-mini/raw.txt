def find_conflicting_triplet(n, m, conflicts):
    from collections import defaultdict
    
    # Create an adjacency list for conflicts
    conflict_map = defaultdict(set)
    
    for u, v in conflicts:
        conflict_map[u].add(v)
        conflict_map[v].add(u)

    # Sort the keys to ensure we check in lexicographical order
    for a in range(1, n + 1):
        if a not in conflict_map:
            continue
        for b in conflict_map[a]:
            if b <= a:
                continue
            for c in conflict_map[b]:
                if c <= b:
                    continue
                if c in conflict_map[a]:  # Check if a, b, c are all mutually conflicting
                    return a, b, c

# Read input
import sys
input = sys.stdin.read
data = input().strip().splitlines()

n, m = map(int, data[0].split())
conflicts = [tuple(map(int, line.split())) for line in data[1:m + 1]]

# Find and print the conflicting triplet
a, b, c = find_conflicting_triplet(n, m, conflicts)
print(a, b, c)
