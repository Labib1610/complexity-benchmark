def solve():
    import sys
    input_data = sys.stdin.read().strip().split()
    if not input_data: return
    it = iter(input_data)
    n = int(next(it)); m = int(next(it))
    adj_mask = [0]*(n+1)
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj_mask[u] |= 1 << (v-1)
        adj_mask[v] |= 1 << (u-1)
    for a in range(1, n+1):
        mask_a = adj_mask[a]
        for b in range(a+1, n+1):
            if (mask_a >> (b-1)) & 1:
                mask = mask_a & adj_mask[b]
                # zero bits <= b
                mask_after = mask & ~((1 << b) - 1)  # bits 0..b-1 zeroed
                if mask_after:
                    low = mask_after & -mask_after
                    c = low.bit_length()  # because lowest bit position index +1
                    if c <= n:
                        print(a, b, c)
                        return

Check: When we compute low.bit_length() gives position+1 because bit_length returns number of bits needed. For mask 1<<k, bit_length = k+1. So c = low.bit_length() is correct.
