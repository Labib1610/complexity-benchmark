```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Using a list of sets for O(1) average lookup of edges
    adj = [set() for _ in range(n + 1)]
    # To iterate through neighbors efficiently, we keep sorted adjacency lists
    adj_list = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        adj_list[u].append(v)
        adj_list[v].append(u)

    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # We iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # However, n=2000 makes O(n^3) too slow.
    # We can optimize by iterating through a, then through all neighbors b of a where b > a,
    # then checking for common neighbors c of a and b where c > b.
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check b in increasing order
        neighbors_a = sorted([v for v in adj_list[a] if v > a])
        
        for b in neighbors_a:
            # We need c such that c > b and c is connected to both a and b.
            # Since we want the smallest c, we can iterate through the smaller 
            # of the adjacency lists of a and b.
            
            # Optimization: iterate through the smaller set of neighbors to find common ones
            if len(adj[a]) < len(adj[b]):
                # Check neighbors of a that are also neighbors of b and > b
                # Since neighbors_a is already sorted, we can just look for elements > b
                for c in neighbors_a:
                    if c > b:
                        if c in adj[b]:
                            print(f"{a} {b} {c}")
                            return
            else:
                # Check neighbors of b that are also neighbors of a and > b
                # To find the smallest c > b, we can iterate through sorted neighbors of b
                neighbors_b = sorted([v for v in adj_list[b] if v > b])
                for c in neighbors_b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == "__main__":
    solve()
```
