def find_conflict_triplet(n, m, conflicts):
    from collections import defaultdict

    # Create a set for conflicts
    conflict_set = defaultdict(set)
    for u, v in conflicts:
        conflict_set[u].add(v)
        conflict_set[v].add(u)

    # Find the triplet
    for a in range(1, n + 1):
        for b in conflict_set[a]:
            if b > a:  # Ensure a < b
                for c in conflict_set[b]:
                    if c > b and c in conflict_set[a]:  # Ensure b < c and a, b, c are all in conflict
                        return a, b, c

# Input reading
import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
conflicts = [tuple(map(int, line.split())) for line in data[1:m + 1]]

# Find and print the result
result = find_conflict_triplet(n, m, conflicts)
print(*result)
