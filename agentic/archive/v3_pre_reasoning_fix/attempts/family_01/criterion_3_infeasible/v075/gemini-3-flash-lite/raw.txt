import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookups
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c
    # iterate through all possible a, b, c.
    # Since n <= 2000, we can iterate over all edges (a, b) and check for common neighbors.
    # To ensure lexicographical order, we iterate a from 1 to n, b from a+1 to n, 
    # and then check if they share a common neighbor c > b.
    
    # Sorting neighbors to speed up finding common neighbors
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in adj[a]:
                # Find common neighbors c such that c > b
                # Intersecting neighbors of a and b
                # Using set intersection is efficient
                common = adj[a].intersection(adj[b])
                
                res = []
                for c in common:
                    if c > b:
                        res.append(c)
                
                if res:
                    print(f"{a} {b} {min(res)}")
                    return

if __name__ == '__main__':
    solve()
