import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        # Ensure u < v for consistent storage
        if u > v:
            u, v = v, u
        
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest triple (a, b, c) with a < b < c
    # Iterate through all pairs (a, b) and check if there exists c > b 
    # such that (a, c) and (b, c) are also in the conflict set.
    
    # Sort edges to process in lexicographical order
    edges.sort()
    
    # Optimization: iterate through sorted edges (a, b)
    # For each edge (a, b), find common neighbors c such that c > b.
    # To find the smallest triple, we look for the smallest a, then b, then c.
    
    # Pre-sorting adjacency lists can help find c faster
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        for b in adj[a]:
            if b > a:
                # Check for common neighbors c > b
                # Use intersection of sorted lists
                idx_a = 0
                idx_b = 0
                neighbors_a = adj[a]
                neighbors_b = adj[b]
                
                # We need c in neighbors_a AND neighbors_b with c > b
                i = 0
                j = 0
                while i < len(neighbors_a) and j < len(neighbors_b):
                    val_a = neighbors_a[i]
                    val_b = neighbors_b[j]
                    
                    if val_a == val_b:
                        if val_a > b:
                            print(f"{a} {b} {val_a}")
                            return
                        i += 1
                        j += 1
                    elif val_a < val_b:
                        i += 1
                    else:
                        j += 1

if __name__ == '__main__':
    solve()
