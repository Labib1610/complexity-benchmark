import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list to store the graph
    # To find the lexicographically smallest (a, b, c), 
    # we iterate a from 1 to n, then b from a+1 to n, etc.
    # To optimize, we only check pairs (a, b) that are edges.
    adj = [[] for _ in range(n + 1)]
    # Use a bitset-like approach or a set for O(1) lookup of edges
    # Since n is up to 2000, a boolean matrix (n+1)x(n+1) fits in memory.
    # 2000 * 2000 = 4 million booleans, which is well within 256MB.
    has_edge = [[False] * (n + 1) for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        has_edge[u][v] = True
        has_edge[v][u] = True
        
    # To ensure lexicographical order, sort adjacency lists
    for i in range(1, n + 1):
        adj[i].sort()
        
    # Iterate through possible 'a' in increasing order
    for a in range(1, n - 1):
        # We only care about neighbors of 'a' that are greater than 'a'
        neighbors_a = [v for v in adj[a] if v > a]
        
        # Iterate through pairs of neighbors (b, c) where a < b < c
        # Since neighbors_a is sorted, we can iterate through them linearly
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                if has_edge[b][c]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
