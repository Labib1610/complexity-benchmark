```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and adj[b]
    
    for a in range(1, n - 1):
        # Sort neighbors of a that are > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for b in neighbors_a:
            # Find common neighbors of a and b that are > b
            # We can intersect the sets or iterate
            # Since we want the smallest c, we iterate through sorted neighbors
            # To optimize, we check the smaller neighbor set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
                other = b
            else:
                candidates = adj[b]
                other = a
            
            for c in sorted(candidates):
                if c > b and c in adj[other]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
