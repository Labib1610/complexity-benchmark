def find_mutually_interlocking_triplet(n, m, collisions):
    from collections import defaultdict

    # Create an adjacency list for the graph of collisions
    graph = defaultdict(set)
    for u, v in collisions:
        graph[u].add(v)
        graph[v].add(u)

    # Look for the first triplet (a, b, c) such that a < b < c
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if b in graph[a]:  # Check if there's a collision between a and b
                for c in range(b + 1, n + 1):
                    if c in graph[a] and c in graph[b]:  # Check if there's a collision between a and c, and b and c
                        return a, b, c

# Read input
import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
collisions = [tuple(map(int, line.split())) for line in data[1:m + 1]]

# Find and print the result
result = find_mutually_interlocking_triplet(n, m, collisions)
print(*result)
