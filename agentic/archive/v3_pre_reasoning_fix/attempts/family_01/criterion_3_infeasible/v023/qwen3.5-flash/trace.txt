```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))

    # Adjacency list and masks
    # Using 0-based indexing for internal logic
    adj_list = [[] for _ in range(n)]
    adj_mask = [0] * n

    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj_list[u].append(v)
        adj_list[v].append(u)
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)

    # Sort adjacency lists to ensure we iterate neighbors in increasing order
    for i in range(n):
        adj_list[i].sort()

    # Iterate a from 0 to n-3
    for a in range(n - 2):
        # Iterate neighbors b of a
        # We need b > a
        for b in adj_list[a]:
            if b <= a:
                continue
            
            # We need c > b such that (a, c) and (b, c) are edges
            # c must be a common neighbor of a and b
            # common = adj_mask[a] & adj_mask[b]
            # We need smallest c > b in common
            
            common = adj_mask[a] & adj_mask[b]
            
            # Shift common to remove bits <= b
            # We want smallest bit > b
            shifted = common >> (b + 1)
            
            if shifted > 0:
                # Find the lowest set bit in shifted
                lsb = shifted & -shifted
                # Calculate the original index
                c = lsb.bit_length() - 1 + (b + 1)
                
                # Found the lexicographically smallest triplet
                print(f"{a + 1} {b + 1} {c + 1}")
                return

if __name__ == '__main__':
```

```python
import sys

# Increase integer string conversion limit just in case, though not needed for bitwise ops
sys.set_int_max_str_digits(100000)

def solve():
    # Read all input at once
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for iteration order
    adj_list = [[] for _ in range(n)]
    # Adjacency masks for fast intersection
    adj_mask = [0] * n

    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj_list[u].append(v)
        adj_list[v].append(u)
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)

    # Sort adjacency lists to ensure neighbors are processed in increasing order
    for i in range(n):
        adj_list[i].sort()

    # Iterate through each vertex as the smallest vertex 'a' in the triplet
    for a in range(n - 2):
        # Iterate through neighbors of 'a' as potential 'b'
        # We need b > a to ensure a < b < c
        for b in adj_list[a]:
            if b <= a:
                continue
            
            # Check for a common neighbor 'c' such that c > b
            # c must be connected to both a and b
            # We use bitwise AND to find common neighbors
            common = adj_mask[a] & adj_mask[b]
            
            # We only care about neighbors c > b
            # Shift right by (b + 1) to remove bits <= b
            shifted = common >> (b + 1)
            
            if shifted > 0:
                # Find the smallest bit set in shifted
                # This corresponds to the smallest c > b
                lsb = shifted & -shifted
                # bit_length() returns the number of bits required to represent the integer
                # For a power of 2 (lsb), bit_length() - 1 gives the 0-based index of the bit
                c = lsb.bit_length() - 1 + (b + 1)
                
                # Output the triplet (converting back to 1-based indexing)
                print(f"{a + 1} {b + 1} {c + 1}")
                return

if __name__ == '__main__':
    solve()
```
