def solve():
    import sys
    input=sys.stdin.readline
    n,m=map(int,input().split())
    adj=[0]*n
    neigh=[[] for _ in range(n)]
    for _ in range(m):
        u,v=map(int,input().split())
        u-=1; v-=1
        adj[u]|=1<<v
        adj[v]|=1<<u
        neigh[u].append(v)
        neigh[v].append(u)
    for a in range(n):
        for b in neigh[a]:
            if b<=a: continue
            common=adj[a]&adj[b]
            mask=common >> (b+1)  # shift so bit 0 corresponds to node b+1
            if mask==0: continue
            # find lowest set bit
            idx=(mask & -mask).bit_length()-1
            c=b+1+idx
            print(a+1,b+1,c+1)
            return
