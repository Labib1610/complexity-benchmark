import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    masks = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        masks[u] |= 1 << (v - 1)
        masks[v] |= 1 << (u - 1)
        neigh[u].append(v)
        neigh[v].append(u)

    for i in range(1, n + 1):
        neigh[i].sort()

    for a in range(1, n + 1):
        for b in neigh[a]:
            if b <= a:
                continue
            common = masks[a] & masks[b]
            shift = common >> b  # remove bits <= b-1
            if shift:
                low = shift & -shift
                c = (low.bit_length() - 1) + b + 1
                print(a, b, c)
                return

if __name__ == "__main__":
    main()
