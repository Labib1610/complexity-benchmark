import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c
    # Iterate a from 1 to n
    # Iterate b from a+1 to n
    # Check if b is in adj[a]
    # If so, check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check potential b and c in order
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # Intersection of adj[a] and adj[b] gives common neighbors
            # We need the smallest c > b
            # To optimize, check neighbors of b that are also in neighbors_a
            
            # Since n is up to 2000, we can use bitsets or direct intersection
            # Given the constraints and the nature of the problem, a simple intersection is fine
            
            # Optimization: only check c > b
            neighbors_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
