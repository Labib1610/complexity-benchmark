        n = int(next(iterator))

    except StopIteration:
        return

    # adj_mask stores adjacency as bitmasks
    adj_list = [[] for _ in range(n + 1)]
    adj_mask = [0] * (n + 1)

    for _ in range(m):

        v = int(next(iterator))

        adj_list[v].append(u)
        
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
        
    # Sort neighbor lists to process in lexicographical order
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # Iterate to find the lexicographically smallest triangle
    for a in range(1, n + 1):
        for b in adj_list[a]:
            # We need a < b < c. 
            # Since we iterate 'a' from 1 upwards, we just need b > a.
            if b <= a:
                continue
            
            # Find common neighbors of a and b
            common = adj_mask[a] & adj_mask[b]

            # Shift the mask to align bit b+1 to position 0.
            # Bits in 'common' correspond to node indices.
            shifted = common >> (b + 1)

                lsb = shifted & -shifted
                # bit_length() - 1 gives the 0-based index of the LSB
                c_idx = lsb.bit_length() - 1

            if b <= a:

            # Intersection of neighbors of a and b gives all common neighbors.
