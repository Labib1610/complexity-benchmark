import sys

def solve():
    # Using fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))

    # To find the lexicographically smallest (a, b, c) where a < b < c:
    # We iterate through potential 'a' from 1 to n.
    # For a fixed 'a', we look for pairs (b, c) such that a < b < c and 
    # (a, b), (a, c), and (b, c) are edges.
    
    # To optimize, we only consider neighbors of 'a' that are greater than 'a'.
    for a in range(1, n - 1):
        # neighbors_a contains all v > a such that (a, v) is an edge
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        # We need to find the smallest pair (b, c) from neighbors_a such that (b, c) is an edge.
        # Since we want the lexicographically smallest (a, b, c), and we are iterating 'a' 
        # in increasing order, the first 'a' that yields any (b, c) will be part of the 
        # lexicographically smallest triplet.
        # However, we must check all (b, c) for that specific 'a' to find the minimum (b, c).
        
        best_bc = None
        
        # Iterate through neighbors_a to find the smallest b, then smallest c.
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if (b, c) is an edge
                if c in adj[b]:
                    # Since neighbors_a is sorted and we iterate i then j,
                    # the first (b, c) we find for this 'a' is the lexicographically smallest.
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
