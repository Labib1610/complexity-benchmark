```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookups
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through a from 1 to n-2
    # Iterate through b from a+1 to n-1
    # Check if a and b are connected
    # If they are, find the smallest c > b such that c is connected to both a and b
    
    # Sort neighbors to make finding c more efficient
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # We only care about neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # Need to find c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can iterate or intersect
            # Given n <= 2000, O(n^2) or slightly more is fine
            
            # Intersection of neighbors of a and b that are > b
            # Use sets for faster intersection
            common = adj[a].intersection(adj[b])
            valid_c = [x for x in common if x > b]
            
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
