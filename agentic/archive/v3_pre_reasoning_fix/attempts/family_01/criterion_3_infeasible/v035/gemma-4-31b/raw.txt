import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list and adjacency matrix/set
    # Given n <= 2000, a bitset or a boolean matrix is feasible.
    # In Python, we can use a list of sets or a list of integers (bitmasks).
    # Since we need lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b from a+1 to n, then c from b+1 to n.
    
    adj = [[] for _ in range(n + 1)]
    # Using a bitset approach for fast intersection in Python
    # Each personnel's neighbors represented as a large integer (bitmask)
    masks = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
        masks[u] |= (1 << v)
        masks[v] |= (1 << u)

    # Sort adjacency lists to iterate in increasing order
    for i in range(1, n + 1):
        adj[i].sort()

    # Lexicographically smallest (a, b, c) means:
    # Minimize a, then minimize b, then minimize c.
    for a in range(1, n - 1):
        # Optimization: only check neighbors of a that are > a
        neighbors_a = adj[a]
        # We only care about neighbors v > a
        valid_neighbors = [v for v in neighbors_a if v > a]
        
        for i in range(len(valid_neighbors)):
            b = valid_neighbors[i]
            # We need to find the smallest c > b such that c is a neighbor of both a and b.
            # This is equivalent to finding the lowest set bit in (masks[a] & masks[b])
            # that is greater than b.
            
            common = masks[a] & masks[b]
            if common == 0:
                continue
            
            # To find the smallest c > b:
            # Clear bits <= b
            # Mask out bits 0 to b: ~( (1 << (b + 1)) - 1 )
            mask_gt_b = ~((1 << (b + 1)) - 1)
            candidates = common & mask_gt_b
            
            if candidates:
                # Find the lowest set bit
                # c = (candidates & -candidates).bit_length() - 1
                # In Python 3.10+, candidates.bit_count() or similar, but bit_length is universal.
                # The lowest set bit is (candidates & -candidates)
                lowbit = candidates & -candidates
                c = lowbit.bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
