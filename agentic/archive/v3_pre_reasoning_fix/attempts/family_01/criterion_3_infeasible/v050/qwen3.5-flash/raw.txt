The problem asks us to find the lexicographically smallest triplet of transmitters $(a, b, c)$ such that $1 \le a < b < c \le n$ and there are edges $(a, b)$, $(a, c)$, and $(b, c)$ in the given graph. We are given $n$ transmitters and $m$ edges representing overlaps. We are guaranteed that at least one such triplet (a triangle) exists. The constraints are $n \le 2000$ and $m \le 2 \cdot 10^5$.

We need to minimize $a$ first, then $b$, then $c$. This means we should iterate through possible values of $a$ from $1$ to $n$. Once we fix $a$, we need to find the smallest $b > a$ such that there is an edge between $a$ and $b$. Then, for this $a$ and $b$, we need to find the smallest $c > b$ such that there are edges between $a$ and $c$, and between $b$ and $c$. If we find such a $c$, then $(a, b, c)$ is the lexicographically smallest triplet starting with $a$ and $b$. Since we iterate $a$ in increasing order and $b$ in increasing order, the first valid triplet we find will be the lexicographically smallest one.

Since we need to iterate over neighbors of $a$ and then check if they are connected to $b$, an adjacency matrix is very convenient for checking existence of $(a, c)$ and $(b, c)$.

import sys

# Increase recursion depth just in case, though not needed here

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency masks for bitwise operations
    # adj_mask[u] will have the v-th bit set if there is an edge (u, v)
    # We use 1-based indexing for bits, so bit v corresponds to transmitter v
    adj_mask = [0] * (n + 1)
    
    # Adjacency lists to iterate neighbors efficiently
    adj_list = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        
        # Update masks
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
        
        # Update lists
        adj_list[u].append(v)
        adj_list[v].append(u)

    # Sort adjacency lists to ensure we process neighbors in increasing order
    # This is crucial for finding the lexicographically smallest triplet
    for i in range(1, n + 1):
        adj_list[i].sort()

    # Iterate through each transmitter 'a'
    for a in range(1, n + 1):
        # Iterate through neighbors of 'a' to find 'b'
        # We want smallest 'b' > 'a'
        neighbors = adj_list[a]
        for b in neighbors:
            if b <= a:
                continue
            
            # We have an edge (a, b) and b > a.
            # Now we need to find the smallest 'c' > 'b' such that (a, c) and (b, c) exist.
            # This is equivalent to finding the smallest bit set in (adj_mask[a] & adj_mask[b])
            # that is greater than b.
            
            # Compute common neighbors mask
            common = adj_mask[a] & adj_mask[b]
            
            # We need bits > b. Shift right by b + 1 to remove bits <= b.
            # If common has bits > b set, val will be non-zero.
            val = common >> (b + 1)
            
            if val == 0:
                continue
            
            # Find the lowest set bit in val.
            # val & -val isolates the lowest set bit.
            lsb = val & -val
            
            # Calculate the index c.
            # If lsb corresponds to bit k (0-indexed) in val (i.e., 1 << k),
            # then the bit index in common is k + b + 1.
            # The bit_length of (1 << k) is k + 1.
            # So k = bit_length - 1.
            # c = (bit_length - 1) + b + 1 = bit_length + b.
            c = lsb.bit_length() + b
            
            # We found the lexicographically smallest triplet (a, b, c)
            print(f"{a} {b} {c}")
            return

import sys

sys.setrecursionlimit(2000)

def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # adj_mask[u] will store a bitmask where the v-th bit is set if (u, v) is an edge
    # We use 1-based indexing for transmitters, so bits 1..n are used.
    adj_mask = [0] * (n + 1)
    
    # adj_list[u] will store the list of neighbors of u
    adj_list = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        
        # Update masks
        adj_mask[u] |= (1 << v)
        adj_mask[v] |= (1 << u)
        
        # Update lists
        adj_list[u].append(v)
        adj_list[v].append(u)

    # Sort adjacency lists to ensure we process neighbors in increasing order.
    # This is important for finding the lexicographically smallest triplet (a, b, c)
    # where we minimize a, then b, then c.
    for i in range(1, n + 1):
        adj_list[i].sort()

    # Iterate through each transmitter 'a' as the first element of the triplet
    for a in range(1, n + 1):
        # Iterate through neighbors 'b' of 'a'
        # We require b > a for the triplet (a, b, c)
        neighbors = adj_list[a]
        for b in neighbors:
            if b <= a:
                continue
            
            # At this point, we have an edge (a, b) with a < b.
            # We need to find the smallest 'c' > 'b' such that (a, c) and (b, c) exist.
            # This means c must be a common neighbor of a and b, and c > b.
            
            # Find common neighbors using bitwise AND of masks
            common = adj_mask[a] & adj_mask[b]
            
            # We want the smallest bit set in 'common' that is greater than b.
            # We can achieve this by shifting 'common' right by (b + 1).
            # If the result is non-zero, the lowest set bit corresponds to the smallest c > b.
            val = common >> (b + 1)
            
            if val == 0:
                continue
            
            # Isolate the lowest set bit of val
            lsb = val & -val
            
            # Calculate the index c.
            # If lsb is 2^k, then the bit at position k in val is set.
            # This corresponds to bit position k + b + 1 in common.
            # The index of the bit k (0-indexed) is bit_length(k) - 1? 
            # No, lsb.bit_length() returns k + 1.
            # So c = (k + 1) + b.
            c = lsb.bit_length() + b
            
            # We found the lexicographically smallest triplet

            return
