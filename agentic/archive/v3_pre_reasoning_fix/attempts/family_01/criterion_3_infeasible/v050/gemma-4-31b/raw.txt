import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list and adjacency matrix (bitsets for efficiency)
    # Since n is up to 2000, a bitset (integer in Python) 
    # allows for fast intersection of neighbors.
    adj_list = [[] for _ in range(n + 1)]
    adj_bitset = [0] * (n + 1)
    
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        ptr += 2
        
        # Ensure u < v for easier processing
        if u > v:
            u, v = v, u
            
        adj_list[u].append(v)
        adj_bitset[u] |= (1 << v)
        adj_bitset[v] |= (1 << u)

    # To find the lexicographically smallest (a, b, c) where a < b < c:
    # We iterate a from 1 to n, then b from (a+1) to n.
    # If (a, b) is an edge, we check for common neighbors c > b.
    
    # Sort adjacency lists to iterate b in increasing order
    for i in range(1, n + 1):
        adj_list[i].sort()

    for a in range(1, n - 1):
        # Optimization: only check neighbors of a that are > a
        neighbors_a = adj_list[a]
        
        # We need b > a and c > b.
        # Only consider neighbors of a that are greater than a.
        filtered_neighbors = [v for v in neighbors_a if v > a]
        
        for i in range(len(filtered_neighbors)):
            b = filtered_neighbors[i]
            
            # We need to find the smallest c > b such that c is a neighbor of both a and b.
            # The common neighbors of a and b are: adj_bitset[a] & adj_bitset[b]
            common = adj_bitset[a] & adj_bitset[b]
            
            if common:
                # We need the smallest c > b.
                # Mask out all bits <= b.
                mask = ~((1 << (b + 1)) - 1)
                valid_c = common & mask
                
                if valid_c:
                    # Find the lowest set bit in valid_c
                    # (valid_c & -valid_c).bit_length() - 1 gives the index of the lowest set bit
                    c = (valid_c & -valid_c).bit_length() - 1
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
