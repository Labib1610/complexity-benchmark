import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list for graph
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # We only care about edges where u < v to facilitate lexicographical search
        if u > v:
            u, v = v, u
        adj[u].append(v)
    
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # We iterate a from 1 to n.
    # Then we iterate b from a+1 to n.
    # Then we check if there is a c > b such that (a,c) and (b,c) are edges.
    
    # Sort the adjacency lists to allow efficient intersection or iteration
    for i in range(1, n + 1):
        adj[i].sort()
        
    # To optimize the search for c, we can use a bitset or a boolean matrix
    # since n <= 2000, a boolean matrix fits in memory (2000*2000 = 4*10^6).
    # However, Python's bitsets (integers) are very efficient.
    
    # Create bitmasks for each node's neighbors
    masks = [0] * (n + 1)
    # Re-process edges to build masks
    # We only need to store neighbors > current node to satisfy a < b < c
    # Actually, to check if (a,c) and (b,c) exist, we need all neighbors.
    
    # Resetting and rebuilding adjacency for the mask
    # We use the original input pairs
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        masks[u] |= (1 << v)
        masks[v] |= (1 << u)

    # Search for the lexicographically smallest triple
    for a in range(1, n - 1):
        # Optimization: only iterate over neighbors of a that are > a
        # The sorted adjacency list adj[a] contains all neighbors v > a
        for b in adj[a]:
            # We need c > b such that (a,c) and (b,c) are edges.
            # This means c must be in (neighbors of a) AND (neighbors of b) AND (c > b).
            
            # Intersection of neighbors of a and b
            common = masks[a] & masks[b]
            
            # We need the smallest c > b. 
            # We can mask out bits <= b.
            # (1 << (b + 1)) - 1 creates a mask of bits 0 to b.
            # Bitwise NOT (~) or XORing with a large mask removes them.
            
            # Filter for bits > b
            candidates = common & ~((1 << (b + 1)) - 1)
            
            if candidates:
                # Find the lowest set bit (the smallest c > b)
                # Using bit_length and trailing zeros logic
                # In Python 3.10+, int.bit_count() exists, but for lowest bit:
                # (candidates & -candidates).bit_length() - 1
                c = (candidates & -candidates).bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
