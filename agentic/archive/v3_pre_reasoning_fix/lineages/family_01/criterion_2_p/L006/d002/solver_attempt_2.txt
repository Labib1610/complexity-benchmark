import sys

# The problem asks us to assign 1 or 2 to each node such that for every edge (u, v),
# padding[u] + padding[v] = 3.
# This means if padding[u] = 1, then padding[v] must be 2, and vice versa.
# This is a bipartite coloring problem.
# For each connected component, we can pick the value of the first node (1 or 2).
# To get the lexicographically smallest sequence, we iterate through nodes 1 to n.
# If a node is not yet colored, we try assigning 1 first (to keep it lexicographically small).
# If that leads to a contradiction, we try 2. If both fail, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning 1 to the first node of the component
            # If it fails, try 2.
            # Since we want lexicographically smallest, we greedily pick 1 if possible.
            # Actually, for a connected component, picking the value of the first node
            # determines all others. We check if 1 works, if not, check if 2 works.
            
            def check(start_node, val):
                comp = []
                stack = [(start_node, val)]
                assignment[start_node] = val
                comp.append((start_node, val))
                
                while stack:
                    u, val_u = stack.pop()
                    target = 3 - val_u
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = target
                            comp.append((v, target))
                            stack.append((v, target))
                        elif assignment[v] != target:
                            return False, comp
                return True, comp

            # Try 1
            temp_assignment = assignment[:]
            success, comp = check(i, 1)
            if not success:
                # Reset and try 2
                for node, val in comp:
                    assignment[node] = 0
                success, comp = check(i, 2)
                if not success:
                    print("NO")
                    return
    
    print("YES")
    print(*(assignment[1:]))

solve()
