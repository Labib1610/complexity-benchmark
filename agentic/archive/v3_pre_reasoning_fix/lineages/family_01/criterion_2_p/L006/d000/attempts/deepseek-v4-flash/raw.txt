import sys
from collections import deque

def main():
    input = sys.stdin.buffer.readline
    n_m = input().split()
    if not n_m: return
    n, m = map(int, n_m)
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    color = [-1]*(n+1)
    for i in range(1, n+1):
        if color[i] != -1:
            continue
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return
    ans = ' '.join(str(color[i]+1) for i in range(1,n+1))
    sys.stdout.write("YES\n" + ans + "\n")

if __name__ == "__main__":
    main()

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))

    del data, it

    color = [-1] * (n + 1)

        if color[i] != -1:
            continue
        color[i] = 0
        q = [i]
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return
    ans = " ".join(str(color[i] + 1) for i in range(1, n + 1))
    sys.stdout.write("YES\n" + ans + "\n")
