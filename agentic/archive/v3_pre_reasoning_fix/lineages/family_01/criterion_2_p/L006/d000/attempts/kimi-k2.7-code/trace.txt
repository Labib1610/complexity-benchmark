def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        if u == v:
            # impossible, but still read all; can set flag
            # We'll just add self-loop and detect later? Detection: when processing edge u-v, if neighbor same, color equal => conflict.
            pass
        adj[u].append(v)
        adj[v].append(u)
    color = [-1]*n
    from collections import deque
    possible = True
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            dq = deque([i])
            while dq and possible:
                v = dq.popleft()
                cv = color[v]
                for w in adj[v]:
                    if color[w] == -1:
                        color[w] = cv ^ 1
                        dq.append(w)
                    elif color[w] == cv:
                        possible = False
                        break
    out = []
    if not possible:
        out.append("NO")
    else:
        out.append("YES")
        out.append(' '.join(str(c+1) for c in color))
    sys.stdout.write('\n'.join(out))

if __name__ == '__main__':
    solve()
