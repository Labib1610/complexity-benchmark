import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest assignment.
    # This means we prefer 1 over 2 for field 1, then field 2, etc.
    # This is a 2-coloring problem.
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color the connected component.
            # To get lexicographically smallest, we try assigning 1 to the current node.
            stack = [i]
            color[i] = 1
            
            while stack:
                u = stack.pop()
                
                # Sort neighbors to ensure deterministic behavior if needed,
                # though for 2-coloring, the choice is forced once the first node is set.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
