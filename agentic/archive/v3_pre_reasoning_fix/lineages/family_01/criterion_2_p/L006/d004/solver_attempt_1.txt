import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest sequence.
    # This means we prefer assigning 1 to field 1, then 1 to field 2, etc.
    # This is a 2-coloring problem.
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to the first uncolored node in the component
            # to keep the sequence lexicographically smallest.
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if color[v] == 0:
                        # To keep lexicographically smallest, we want to assign 1
                        # whenever possible.
                        # If we are at node u with color c, neighbors must be 3-c.
                        new_c = 3 - c
                        color[v] = new_c
                        stack.append((v, new_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
