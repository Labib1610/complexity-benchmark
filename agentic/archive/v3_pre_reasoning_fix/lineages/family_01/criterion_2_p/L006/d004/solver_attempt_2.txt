    if not input_data:

    idx = 2

        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)

    # to the smallest possible index whenever possible.
    # We iterate 1 to n. If a node is uncolored, we try assigning it 1.
    # This is a standard 2-coloring problem (bipartite check).
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assigning 1 to i

            # first node determines the rest. To be lexicographically smallest,
            # we must pick 1 for the smallest node in the component.

            head = 0
