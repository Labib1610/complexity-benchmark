import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest sequence, we want s_1=1, then s_2=1 if possible, etc.
    # This is a 2-coloring problem. For each connected component, we can choose the color
    # of the first node encountered (the one with the smallest index).
    # To make the sequence lexicographically smallest, for each component, we try to 
    # assign 1 to the smallest index node in that component.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from node i
            # Assign color 1 to node i to keep it lexicographically smallest
            stack = [(i, 1)]
            colors[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
