import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data: return
    it = iter(data)
    n = next(it); m = next(it)
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = next(it); v = next(it)
        if u == v:
            # self-loop impossible
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
    color = [0]*(n+1)
    for start in range(1,n+1):
        if color[start]:
            continue
        color[start] = 1
        q = [start]
        head = 0
        while head < len(q):
            v = q[head]; head += 1
            cv = color[v]
            nc = 3 - cv
            for to in adj[v]:
                if color[to] == 0:
                    color[to] = nc
                    q.append(to)
                elif color[to] == cv:
                    print("NO")
                    return
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color[1:])))
    sys.stdout.write("\n")

if __name__ == "__main__": main()
