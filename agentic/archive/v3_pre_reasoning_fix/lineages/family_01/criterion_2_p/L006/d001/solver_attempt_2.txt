import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unvisited, 1 = matte, 2 = glossy
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # For each connected component, we try setting the first node to 1.
    # If that fails, we try 2.
    
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # We want the smallest possible value for the first node in this component
            # which is 1.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                print("NO")
                return
            
            # Store result
            for node in component:
                res[node] = color[node]
                
    print("YES")
    print(*(res[1:]))

solve()
