```python
import sys

# The problem asks for a 2-coloring of a graph such that no two adjacent nodes
# have the same color. This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not yet colored, we assign it color 1 (the smallest possible).
# Then we perform a BFS/DFS to color its connected component.

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we want to assign 1
    # to the smallest possible index whenever we start a new component.
    # Since we iterate i from 1 to n, if colors[i] is 0, we assign 1.
    # This greedy approach works because the choice of color for a component
    # is independent of other components, and 1 is always preferred over 2.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs if using DFS, 
    # but here we use iterative stack-based approach.
    solve()
```
