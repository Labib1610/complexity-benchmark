def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    adj = [[] for _ in range(n)]

        it += 2

                u = stack.pop()

                        stack.append(v)

                        return
    sys.stdout.write("YES\n" + " ".join(map(str, color)))
