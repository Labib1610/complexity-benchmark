import sys

# The condition is that for every edge (u, v), b[u] * b[v] = 2.
# Since b[i] is in {1, 2}, the only way for the product to be 2 is:
# (1 * 2 = 2) or (2 * 1 = 2).
# This means the graph must be bipartite, and for each connected component,
# if we assign a state to one node, all other nodes in that component are fixed.
# Specifically, if node u has state b[u], its neighbor v must have state 3 - b[u].
# To get the lexicographically smallest sequence, for each component, we try
# assigning 1 to the first node (the one with the smallest index) and propagate.
# If that leads to a contradiction, we try assigning 2. If both fail, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning 1 to the first node of the component
            # If it fails, try 2.
            comp = []
            stack = [(i, 1)]
            assignment[i] = 1
            comp.append(i)
            
            possible = True
            idx_stack = 0
            while idx_stack < len(comp):
                u = comp[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        comp.append(v)
                    elif assignment[v] == assignment[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset and try assigning 2
                for node in comp:
                    assignment[node] = 0
                
                assignment[i] = 2
                comp = [i]
                idx_stack = 0
                possible = True
                while idx_stack < len(comp):
                    u = comp[idx_stack]
                    idx_stack += 1
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - assignment[u]
                            comp.append(v)
                        elif assignment[v] == assignment[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
