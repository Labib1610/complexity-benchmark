import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the flow rates f_u and f_v satisfy f_u + f_v = 3.
# Since f_i is either 1 or 2, this means f_u != f_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want f_i = 1 whenever possible.
# For each connected component, we have two choices for coloring:
# 1. Assign the first node in the component (smallest index) to 1.
# 2. Assign the first node in the component to 2.
# We choose the one that makes the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Try setting the smallest index in component to 1
            # If color[i] is 1, this is the current assignment.
            # If color[i] is 2, we can flip all colors in this component.
            
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            
            # We want the color of min_node to be 1 if possible.
            # Current color of min_node is color[min_node].
            if color[min_node] == 2:
                # Flip all
                for node, c in component:
                    ans[node] = 3 - c
            else:
                for node, c in component:
                    ans[node] = c
                    
    print("YES")
    print(*(ans[1:]))

solve()
