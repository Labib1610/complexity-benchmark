import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is not colored, we try assigning 1 first.
    # This is a standard bipartite check.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1

            # Use iterative DFS to avoid recursion depth issues and manage component
            # However, to ensure lexicographical smallest, we must be careful.
            # Actually, for a bipartite graph, once the first node of a component
            # is fixed, the rest are determined. To make it lexicographically smallest,
            # we should always try to assign 1 to the smallest available node.
            
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:

                        q.append(v)
