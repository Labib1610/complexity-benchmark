import sys

# The problem states that for every intersection (u, v), the product of 
# thicknesses b_u * b_v must be 2. Since b_i is either 1 or 2, 
# the only way to get a product of 2 is if one beam is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 0 and 1.
# If a component is bipartite, we have two choices for the assignment:
# 1. Color 0 -> 1, Color 1 -> 2
# 2. Color 0 -> 2, Color 1 -> 1
# To get the lexicographically smallest sequence, for each connected component,
# we check which of the two assignments gives a smaller value for the first 
# node encountered in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two choices: 
            # Choice A: color 0 -> 1, color 1 -> 2
            # Choice B: color 0 -> 2, color 1 -> 1
            # We want the lexicographically smallest sequence.
            # Find the smallest index in the component.
            first_node = min(component)
            
            # If we pick Choice A, ans[first_node] = 1 if color[first_node] == 0 else 2
            # If we pick Choice B, ans[first_node] = 2 if color[first_node] == 0 else 1
            # To minimize, we want ans[first_node] to be 1 if possible.
            
            if color[first_node] == 0:
                # Choice A: 0->1, 1->2. ans[first_node] = 1
                # Choice B: 0->2, 1->1. ans[first_node] = 2
                # Pick A
                for node in component:
                    ans[node] = 1 if color[node] == 0 else 2
            else:
                # Choice A: 0->1, 1->2. ans[first_node] = 2
                # Choice B: 0->2, 1->1. ans[first_node] = 1
                # Pick B
                for node in component:
                    ans[node] = 2 if color[node] == 0 else 1
                    
    print("YES")
    print(*(ans[1:]))

solve()
