```python
import sys

# The condition is: polarity(u) XOR polarity(v) = 3.
# Polarity indices are {1, 2}.
# 1 XOR 1 = 0
# 1 XOR 2 = 3
# 2 XOR 1 = 3
# 2 XOR 2 = 0
# So, polarity(u) XOR polarity(v) = 3 means polarity(u) != polarity(v).
# This is a standard 2-coloring problem on a graph.
# We want the lexicographically smallest sequence (b1, b2, ..., bn).
# To minimize lexicographically, we prefer 1 over 2.
# For each connected component, we try assigning 1 to the first node encountered.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
