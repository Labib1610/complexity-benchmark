import sys

# The constraint b_u * b_v = 2 with b_i in {1, 2} means:
# If b_u = 1, then b_v = 2.
# If b_u = 2, then b_v = 1.
# This is a standard 2-coloring problem on a graph.
# If b_u != b_v, the graph is bipartite.
# We want to minimize the sequence (b_1, b_2, ..., b_n) lexicographically.
# This means we prefer 1 over 2 for each node in order.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To minimize lexicographically, we iterate 1 to n.
    # If a node is not colored, we try assigning 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            component = []
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
