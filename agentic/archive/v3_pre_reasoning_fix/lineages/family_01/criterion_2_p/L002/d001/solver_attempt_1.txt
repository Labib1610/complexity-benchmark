import sys

# The problem states that for every monitor (u, v), the sum of resonance levels must be 3.
# Since resonance levels are 1 or 2, this means if field u is 1, field v must be 2,
# and if field u is 2, field v must be 1.
# This is equivalent to saying s[u] != s[v].
# This is a bipartite coloring problem.
# We need to color the graph with 1s and 2s such that no two adjacent nodes have the same color.
# To get the lexicographically smallest sequence, we iterate from 1 to n.
# If a node is not yet colored, we try to color it 1. If that leads to a contradiction,

def solve():

    if not input_data:

    n = int(input_data[0])
    m = int(input_data[1])

    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if res[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [(i, 1)]

            # We try setting the first node of the component to 1.
            
            comp_nodes = []
            stack = [i]
