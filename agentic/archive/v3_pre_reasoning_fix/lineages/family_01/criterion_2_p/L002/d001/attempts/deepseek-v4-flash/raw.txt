import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data: return
    it = iter(data)
    n = next(it); m = next(it)
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = next(it); v = next(it)
        adj[u].append(v)
        adj[v].append(u)
    color = [-1]*(n+1)
    for i in range(1,n+1):
        if color[i] != -1:
            continue
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            cu = color[u]
            nv = 1 - cu
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = nv
                    q.append(v)
                elif color[v] == cu:
                    print("NO")
                    return
    out = ['YES', ' '.join('1' if color[i] == 0 else '2' for i in range(1,n+1))]
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__": main()

import sys

def main():
    input = sys.stdin.buffer.readline
    n_m = input().split()
    if not n_m:
        return
    n, m = map(int, n_m)
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        color[start] = 0
        stack = [start]
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - cu
                    stack.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return

    ans = ['1' if color[i] == 0 else '2' for i in range(1, n + 1)]
    sys.stdout.write("YES\n" + " ".join(ans) + "\n")

if __name__ == "__main__":
    main()

import sys

def main():
    input = sys.stdin.buffer.readline

    first = input().split()
    if not first:
        return

    n, m = map(int, first)
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)

    for start in range(1, n + 1):
        if color[start] != -1:
            continue

        color[start] = 0
        stack = [start]

        while stack:
            u = stack.pop()
            cu = color[u]

            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - cu
                    stack.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    ans = ['1' if color[i] == 0 else '2' for i in range(1, n + 1)]
    print("YES")
    print(" ".join(ans))

if __name__ == "__main__":
    main()
