import sys

# The problem states that for any two fields linked by a comparator,
# the sum of their frequency tiers must be exactly 3.
# Since tiers must be positive integers, the only possible values are 1 and 2.
# If field u has tier s_u, then field v must have tier 3 - s_u.
# This is a bipartite coloring problem where each connected component
# must be 2-colorable with colors {1, 2}.
# To get the lexicographically smallest assignment, for each connected component,
# we try assigning 1 to the smallest index node and propagate.
# If that fails, we try assigning 2 to the smallest index node and propagate.
# If both fail, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning 1 to node i
            comp = []
            stack = [(i, 1)]
            assignment[i] = 1
            comp.append(i)
            
            possible = True
            head = 0
            while head < len(comp):
                u = comp[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        comp.append(v)
                    elif assignment[v] == assignment[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset and try assigning 2 to node i
                for node in comp:
                    assignment[node] = 0
                
                assignment[i] = 2
                comp = [i]
                possible = True
                head = 0
                while head < len(comp):
                    u = comp[head]
                    head += 1
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - assignment[u]
                            comp.append(v)
                        elif assignment[v] == assignment[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
