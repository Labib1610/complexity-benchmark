```python
import sys

# The interference between states 1 and 2 is |1^2 - 2^2| = |1 - 4| = 3.
# The interference between states 1 and 1 is |1^2 - 1^2| = 0.
# The interference between states 2 and 2 is |2^2 - 2^2| = 0.
# Since every monitor must detect an interference of exactly 3,
# every connected pair (u, v) must have different states (one 1, one 2).
# This means the graph must be bipartite.
# We want the lexicographically smallest sequence (s1, s2, ..., sn).
# This is equivalent to setting s_i = 1 whenever possible.
# For each connected component, we can pick the state of the first node
# (in index order) to be 1, which forces the states of all other nodes
# in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer color 1 for the first node
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
