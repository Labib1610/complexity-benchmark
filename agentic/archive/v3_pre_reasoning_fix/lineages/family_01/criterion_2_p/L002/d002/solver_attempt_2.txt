import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # resonance levels: 1 or 2. 
    # This is a bipartite coloring problem.
    # We want the lexicographically smallest sequence.
    # For each connected component, we can choose the color of the first node
    # to be 1 or 2. To make it lexicographically smallest, we try 1 first.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we process nodes in order 1..n
            # and for each component, we try to set the first node to 1.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            visited_comp = {i}
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if v not in visited_comp:
                        visited_comp.add(v)
                        q.append(v)
            
            # Now color the component. Since we want lexicographically smallest,
            # we check if setting the smallest index in the component to 1 works.
            # Actually, for any bipartite component, there are only 2 ways to color it.
            # We pick the one that results in a smaller value at the first node
            # encountered in the component.
            
            # Standard bipartite check
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
