```python
import sys

# The problem states that for every channel (u, v), if u is in state S_u, 
# it sends nutrients that require v to be in state 3 - S_u.
# Similarly, if v is in state S_v, it sends nutrients that require u to be in state 3 - S_v.
# Thus, for every channel (u, v), we must have S_v = 3 - S_u, which is S_u != S_v.
# This is equivalent to saying the graph must be bipartite.
# To get the lexicographically smallest assignment, we iterate through each 
# connected component and assign the smaller state (1) to the first node 
# encountered in the component, then propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want state[i] = 1
            stack = [(i, 1)]
            state[i] = 1
            while stack:
                u, s = stack.pop()
                target_s = 3 - s
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = target_s
                        stack.append((v, target_s))
                    elif state[v] != target_s:
                        print("NO")
                        return
    
    print("YES")
    print(*(state[1:]))

solve()
```
