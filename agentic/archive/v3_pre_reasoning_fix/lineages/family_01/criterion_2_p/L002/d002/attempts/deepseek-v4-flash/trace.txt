from collections import deque

def solve():
    import sys
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data: return
    it = iter(data)
    n = next(it); m = next(it)
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = next(it); v = next(it)
        adj[u].append(v)
        adj[v].append(u)
    color = [0]*(n+1)
    for i in range(1,n+1):
        if color[i]==0:
            color[i]=1
            q=deque([i])
            while q:
                u=q.popleft()
                cu=color[u]
                nc=2 if cu==1 else 1
                for v in adj[u]:
                    if color[v]==0:
                        color[v]=nc
                        q.append(v)
                    elif color[v]==cu:
                        print("NO")
                        return
    print("YES")
    sys.stdout.write(" ".join(map(str, color[1:])))
