def main():

    if not input_data:

    n = int(input_data[0]); m = int(input_data[1])
    adj = [[] for _ in range(n)]

        if color[start] != -1:

            u = stack.pop()

                elif color[v] != expected:

                    return

    sys.stdout.write(" ".join(str(c + 1) for c in color))
