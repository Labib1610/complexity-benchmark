```python
import sys

# The problem states that for a link (u, v), if u is on band b_u, 
# it sends a signal that arrives as band (3 - b_u).
# Thus, v must be on band (3 - b_u).
# This implies b_v = 3 - b_u, or b_u + b_v = 3.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest assignment (b_1, b_2, ..., b_n).
# To achieve this, we iterate from 1 to n. If a node is not visited,
# we assign it the smallest possible band (1) and propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer 1 over 2
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, assignment[v]))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
