import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum of voltages must be 3" means if u is 1, v must be 2, and vice versa.
    # This is a bipartite coloring problem where 1 and 2 are the two colors.
    # Lexicographically smallest means we prefer 1 over 2 whenever possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try to assign 1 to the smallest index
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # We need to process components. To ensure lexicographical smallest,
            # we should greedily assign 1 to the smallest uncolored node.
            # Actually, the standard bipartite coloring works:
            # If a component is bipartite, there are only 2 ways to color it.
            # We pick the one that makes the smallest index in the component 1.
            
            comp_nodes = []
            q = [i]
            color[i] = 1
            comp_nodes.append(i)
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if the current coloring is the best for this component
            # The smallest index in this component is min(comp_nodes)
            # If color[min(comp_nodes)] == 2, we flip all colors in this component
            min_node = min(comp_nodes)
            if color[min_node] == 2:
                for node in comp_nodes:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
