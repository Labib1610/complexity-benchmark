```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        # No constraints, all 1s is lexicographically smallest
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color nodes with BFS/DFS - each node must be opposite to its neighbors
    # Node values: 0 - unvisited, 1 - color 1, 2 - color 2
    color = [0] * (n + 1)
    
    for start in range(1, n + 1):
        if color[start] == 0:
            # Start BFS coloring
            queue = deque([start])
            color[start] = 1
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 3 - current_color  # 1->2, 2->1
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] != next_color:
                        # Conflict found - impossible to color
                        print("NO")
                        return
    
    # If we reach here, a valid coloring exists
    # For lexicographically smallest, we prefer color 1 over 2 for node 1,
    # but since we started with color 1, this is already handled.
    # However, if node 1 was not in any edge, we'd want it to be 1.
    # Our algorithm handles this correctly because we try to assign 1 first.
    
    # But actually, we need to ensure that among equivalent solutions,
    # we choose the lexicographically smallest one.
    # Since our BFS starting with color 1 already ensures this property
    # (because we always try color 1 first for the first unvisited component),
    # we just need to convert colors to actual values.
    
    # In our case, color 1 corresponds to polarity index 1, 
    # and color 2 corresponds to polarity index 2.
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            # Isolated node, assign 1 (lexicographically smallest)
            result.append("1")
        else:
            result.append(str(color[i]))
    
    print("YES")
    print(" ".join(result))

solve()
```
