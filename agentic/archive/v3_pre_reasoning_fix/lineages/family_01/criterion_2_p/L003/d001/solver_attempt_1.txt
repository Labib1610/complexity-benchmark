```python
import sys

# The problem asks to assign 1 or 2 to each node such that for every edge (u, v),
# |voltage(u) - voltage(v)| = 1.
# This is equivalent to 2-coloring a graph where 1 and 2 are the two colors.
# If the graph is bipartite, we can color it.
# To minimize the sequence (s1, s2, ..., sn) lexicographically, we want s_i = 1
# whenever possible.
# For each connected component, there are two possible colorings (swap 1 and 2).
# We choose the coloring that makes the smallest index node in the component 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to find component and check bipartiteness
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # We have a valid 2-coloring for this component.
            # The two possible colorings are:
            # 1. Current color[v]
            # 2. 3 - color[v]
            # We want to pick the one that makes the smallest index node in the
            # component have color 1.
            
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
```
