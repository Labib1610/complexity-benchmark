import sys
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    color = [-1] * n
    for i in range(n):
        if color[i] != -1:
            continue
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            cu = color[u]
            nv = 1 - cu
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = nv
                    q.append(v)
                elif color[v] == cu:
                    print("NO")
                    return
    out = ' '.join(str(c + 1) for c in color)
    sys.stdout.write("YES\n" + out + "\n")

if __name__ == "__main__": main()

def main():
    import sys
    input = sys.stdin.buffer.readline
    n_m = input().split()
    if not n_m: return
    n, m = map(int, n_m)
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1; v -= 1
        adj[u].append(v)
        adj[v].append(u)
    color = [-1] * n
    for i in range(n):
        if color[i] != -1:

        color[i] = 0
        stack = [i]
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - cu
                    stack.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(c + 1) for c in color))
    sys.stdout.write("\n")
