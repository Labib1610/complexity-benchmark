```python
import sys

# The problem states that for each warning (u, v), the sum of calibration integers must be 3.
# Since the integers must be 1 or 2, this means if machine u is 1, machine v must be 2,
# and if machine u is 2, machine v must be 1.
# This is a classic 2-coloring problem on a graph where edges represent the constraint
# that the two nodes must have different values.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # colors: 0 = unvisited, 1 = color 1, 2 = color 2
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we iterate from 1 to n.
    # If a node is unvisited, we assign it the smallest possible value (1).
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
