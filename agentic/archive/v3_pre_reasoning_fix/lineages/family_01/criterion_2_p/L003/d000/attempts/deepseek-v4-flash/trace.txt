import sys

def main():
    data = sys.stdin.buffer.readline().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.buffer.readline().split())
        u -= 1; v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    for i in range(n):
        if color[i] != -1:
            continue
        color[i] = 0
        stack = [i]
        ok = True
        while stack:
            v = stack.pop()
            for to in adj[v]:
                if color[to] == -1:
                    color[to] = color[v] ^ 1
                    stack.append(to)
                elif color[to] == color[v]:
                    sys.stdout.write("NO\n")
                    return
    ans = ' '.join(str(c + 1) for c in color)
    sys.stdout.write("YES\n" + ans + "\n")

if __name__ == "__main__":
