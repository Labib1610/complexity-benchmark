def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    color = [-1]*(n+1)
    from collections import deque
    ans = [0]*(n+1)
    for i in range(1,n+1):
        if color[i] != -1: continue
        dq = deque([i])
        color[i] = 0
        ans[i] = 1
        while dq:
            u = dq.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    ans[v] = color[v] + 1
                    dq.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO")
                    return
    out = ['YES'] + [' '.join(map(str, ans[1:]))]
    sys.stdout.write('\n'.join(out))
