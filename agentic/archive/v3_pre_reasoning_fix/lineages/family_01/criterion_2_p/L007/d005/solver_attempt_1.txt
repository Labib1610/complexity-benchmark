```python
import sys

# The problem states that for each sensor (u, v), the sum of phase states must be 3.
# Since phase states are {1, 2}, this means if one is 1, the other must be 2.
# This is equivalent to saying f[u] != f[v].
# This is a standard 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest sequence, so we iterate from 1 to n.
# For each connected component, we try assigning 1 to the smallest uncolored node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we want the first node in the 
            # component to be 1.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_c
                        stack.append((v, next_c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
