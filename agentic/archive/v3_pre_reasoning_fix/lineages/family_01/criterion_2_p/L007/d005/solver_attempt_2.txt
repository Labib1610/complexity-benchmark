```python
import sys

# The problem states that for a link between u and v to be active,
# the bitwise XOR of their modes (1 or 2) must be 3.
# 1 is binary 01, 2 is binary 10.
# 1 XOR 2 = 3 (binary 11).
# So, for every link (u, v), we must have mode[u] != mode[v].
# This is exactly the definition of a bipartite graph.
# We need to check if the graph is bipartite.
# If it is, we need the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for each node.
# We can use BFS/DFS to color the graph. For each connected component,
# we pick the smallest node and assign it 1, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    modes = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if modes[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we assign 1 to the first node
            stack = [(i, 1)]
            modes[i] = 1
            while stack:
                u, color = stack.pop()
                next_color = 3 - color
                for v in adj[u]:
                    if modes[v] == 0:
                        modes[v] = next_color
                        stack.append((v, next_color))
                    elif modes[v] == modes[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(modes[1:]))

solve()
```
