import sys

# The problem asks for an assignment of tiers {1, 2} to n shards such that
# for every edge (u, v), tier(u) * tier(v) = 2.
# This means if tier(u) = 1, then tier(v) must be 2.
# If tier(u) = 2, then tier(v) must be 1.
# This is exactly the condition for a bipartite graph.
# If the graph is bipartite, we can color it with 2 colors (1 and 2).
# To get the lexicographically smallest sequence, we want 1s as early as possible.
# For each connected component, we have two choices:
# 1. Color nodes in set A with 1 and set B with 2.
# 2. Color nodes in set A with 2 and set B with 1.
# We choose the one that makes the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            is_bipartite = True
            idx_comp = 0
            while idx_comp < len(component):
                u, c = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # Option 1: nodes with color 1 get tier 1, nodes with color 2 get tier 2
            # Option 2: nodes with color 1 get tier 2, nodes with color 2 get tier 1
            # We want the smallest tier for the smallest index in the component.
            
            comp_nodes = sorted([node for node, c in component])
            first_node = comp_nodes[0]
            
            # If color[first_node] == 1, Option 1 gives tier 1 to first_node
            # If color[first_node] == 2, Option 2 gives tier 1 to first_node
            if color[first_node] == 1:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
