```python
import sys

# The condition is that for every edge (u, v), (state[u] & state[v]) == 0.
# Since states are in {1, 2}, this means:
# 1 & 1 = 1 (invalid)
# 2 & 2 = 2 (invalid)
# 1 & 2 = 0 (valid)
# 2 & 1 = 0 (valid)
# This is equivalent to saying that the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (say 0 and 1).
# If a component is bipartite, we can assign state 1 to one partition and state 2 to the other.
# To get the lexicographically smallest sequence, for each connected component,
# we try assigning state 1 to the smallest index node in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to find component and check bipartiteness
            component = []
            queue = [i]
            color[i] = 1
            component.append(i)
            
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # To get lexicographically smallest, we want the smallest index in the component
            # to have state 1 if possible.
            # The two possible assignments for a component are:
            # 1. color 1 -> state 1, color 2 -> state 2
            # 2. color 1 -> state 2, color 2 -> state 1
            
            # Find the smallest index in this component
            min_node = min(component)
            
            # If we assign color[min_node] to state 1:
            if color[min_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
