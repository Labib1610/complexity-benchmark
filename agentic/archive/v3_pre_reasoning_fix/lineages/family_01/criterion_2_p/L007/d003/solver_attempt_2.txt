```python
import sys

# The problem asks for a configuration of pressure states {1, 2} for n tanks
# such that for every valve (u, v), the product of states is 2.
# If state(u) * state(v) = 2, then one must be 1 and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph.
# If the graph is bipartite, a solution exists.
# We need the lexicographically smallest sequence, so we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unvisited, 1 = state 1, 2 = state 2
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we try assigning 1 to the current node
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
