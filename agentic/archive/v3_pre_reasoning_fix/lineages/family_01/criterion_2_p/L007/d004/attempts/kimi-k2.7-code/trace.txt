def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    color = [0]*(n+1)
    from collections import deque
    q = deque()
    for i in range(1,n+1):
        if color[i]==0:
            color[i]=1
            q.append(i)
            while q:
                u=q.popleft()
                cu=color[u]
                for w in adj[u]:
                    if color[w]==0:
                        color[w] = -cu
                        q.append(w)
                    elif color[w]==cu:
                        print("NO")
                        return
    # map color 1->1, -1->2
    out = ['YES', ' '.join('1' if c==1 else '2' for c in color[1:])]
    sys.stdout.write('\n'.join(out))
