```python
import sys

# The condition is: pressure_u^2 + pressure_v^2 = 5.
# Since pressure states are in {1, 2}, 1^2 + 2^2 = 1 + 4 = 5.
# This means for every valve (u, v), one tank must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# We need to find the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for tank 1, then tank 2, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = state 1, 2 = state 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we greedily assign 1 to the first node
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
