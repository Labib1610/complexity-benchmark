import sys

def main():
    input = sys.stdin.buffer.readline
    first = input().split()
    if not first:
        return
    n = int(first[0]); m = int(first[1])
    adj = [[] for _ in range(n + 1)]

    for _ in range(m):
        line = input().split()
        # if line blank? no
        u = int(line[0]); v = int(line[1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            stack = [start]
            while stack:
                u = stack.pop()
                needed = 3 - color[u]
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = needed
                        stack.append(v)
                    elif color[v] != needed:
                        print("NO")
                        return

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color[1:])))
    sys.stdout.write("\n")

if __name__ == "__main__":
