import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
    color = [-1]*n
    ans = [1]*n
    for i in range(n):
        if color[i] != -1: continue
        # BFS/DFS component
        color[i] = 0
        comp = [i]
        dq = deque([i])
        while dq:
            u = dq.popleft()
            cu = color[u]
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = cu ^ 1
                    dq.append(w)
                    comp.append(w)
                elif color[w] == cu:
                    print("NO")
                    return
        flip = 1 - color[i]  # make first vertex value 1
        for v in comp:
            ans[v] = (color[v] ^ flip) + 1
    out = ["YES", " ".join(map(str, ans))]
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
