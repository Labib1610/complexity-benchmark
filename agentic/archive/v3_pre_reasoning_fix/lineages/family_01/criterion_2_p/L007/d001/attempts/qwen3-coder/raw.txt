```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    graph = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        graph[u].append(v)
        graph[v].append(u)
        edges.append((u, v))
    
    # Try both assignments for the first node
    # We'll use BFS to check bipartiteness and find a valid assignment
    
    # Color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * (n + 1)
    
    # Check each connected component
    for start in range(1, n + 1):
        if color[start] == 0:
            # BFS to assign colors
            queue = deque([start])
            color[start] = 1
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in graph[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = 3 - current_color  # Flip color (1->2, 2->1)
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        # Conflict found - not bipartite
                        print("NO")
                        return
    
    # If we reach here, the graph is bipartite
    # Now we need to verify that all edges have endpoints with different values
    # In a bipartite graph with our coloring, this is automatically satisfied
    
    # Generate the result - we want lexicographically smallest
    # That means we prefer 1 over 2, so we can just use our color assignment
    # But we need to map colors 1,2 to sensitivity levels 1,2
    
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            # Isolated node - assign 1 (lexicographically smallest)
            result.append("1")
        else:
            # Map color to sensitivity level
            # Since we used 1 and 2 for colors, and we want sensitivity levels 1 and 2,
            # we can just use the color directly
            result.append(str(color[i]))
    
    print("YES")
    print(" ".join(result))

solve()
```
