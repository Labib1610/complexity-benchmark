```python
import sys

# The problem asks for a configuration of 1s and 2s such that for every edge (u, v),
# |s_u - s_v| = 1. This is equivalent to saying that s_u != s_v.
# This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want s_1 to be 1.
# If a connected component contains 1, we must fix its coloring based on s_1 = 1.
# If a component does not contain 1, we can choose the coloring such that the 
# smallest index in that component is 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we want s_i = 1 whenever possible.
    # For each component, we can pick the smallest index 'start' and set color[start] = 1.
    # Then perform BFS/DFS to color the rest.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS from i
            color[i] = 1
            queue = [i]
            while queue:
                u = queue.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
