import sys

def main():
    input = sys.stdin.buffer.readline
    n_m = input().split()
    if not n_m:
        return
    n, m = map(int, n_m)
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    color = [0]*(n+1)
    for start in range(1, n+1):
        if color[start]:
            continue
        color[start] = 1
        q = [start]
        head = 0
        while head < len(q):
            u = q[head]
            head += 1
            cu = color[u]
            need = 3 - cu

                if color[v] == 0:
                    color[v] = need
                    q.append(v)
                elif color[v] != need:
                    sys.stdout.write("NO\n")
                    return
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color[1:])))
