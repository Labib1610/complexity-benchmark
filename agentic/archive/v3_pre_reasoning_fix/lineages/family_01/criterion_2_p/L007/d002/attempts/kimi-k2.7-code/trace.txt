import sys
def solve():
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        adj[u].append(v)
        adj[v].append(u)
    color = [0]*n
    for i in range(n):
        if color[i]==0:
            stack=[i]
            color[i]=1
            while stack:
                u=stack.pop()
                cu=color[u]
                for w in adj[u]:
                    if color[w]==0:
                        color[w]=3-cu
                        stack.append(w)
                    elif color[w]==cu:
                        print("NO")
                        return
    print("YES")
    out=' '.join(str(c) for c in color)
    sys.stdout.write(out)
if __name__=='__main__':
    solve()
