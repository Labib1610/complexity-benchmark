import sys

# The condition is: power[u] + power[v] = 3.
# Since power[i] >= 1, the only possible values are 1 and 2.
# If power[u] = 1, then power[v] = 2.
# If power[u] = 2, then power[v] = 1.
# This is a bipartite coloring problem where each edge (u, v) forces
# color[u] != color[v].
# We want the lexicographically smallest sequence, so we prefer 1 over 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unvisited, 1 = power 1, 2 = power 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we try to assign 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                target_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target_c
                        stack.append((v, target_c))
                    elif color[v] != target_c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
