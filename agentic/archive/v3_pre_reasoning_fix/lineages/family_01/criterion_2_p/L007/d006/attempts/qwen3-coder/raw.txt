import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    graph = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        graph[u].append(v)
        graph[v].append(u)
    
    # Color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS
    for start in range(1, n + 1):
        if color[start] == 0:
            queue = deque([start])
            color[start] = 1
            
            while queue:
                node = queue.popleft()
                
                for neighbor in graph[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = 3 - color[node]  # Flip color (1->2, 2->1)
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If we get here, it's bipartite
    print("YES")
    
    # For lexicographically smallest assignment, we want nodes with no edges to be assigned 1
    # But since we're looking for any valid assignment that's lexicographically smallest,
    # and our coloring already ensures that, we just need to make sure we start with color 1
    # for isolated nodes. Our algorithm already does this.
    
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append(1)  # Isolated node, assign to hold 1
        else:
            result.append(color[i])
    
    print(' '.join(map(str, result)))

solve()
