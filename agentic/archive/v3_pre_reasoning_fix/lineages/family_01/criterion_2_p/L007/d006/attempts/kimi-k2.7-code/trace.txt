def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        adj[u].append(v)
        adj[v].append(u)
    color = [0]*n
    from collections import deque
    for i in range(n):
        if color[i]==0:
            color[i]=1
            dq = deque([i])
            while dq:
                v = dq.popleft()
                cv = color[v]
                for w in adj[v]:
                    if color[w]==0:
                        color[w] = 3-cv
                        dq.append(w)
                    elif color[w]==cv:
                        print("NO")
                        return
    out = ["YES", " ".join(str(c) for c in color)]
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
