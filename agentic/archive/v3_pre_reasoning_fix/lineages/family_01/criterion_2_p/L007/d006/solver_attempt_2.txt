```python
import sys

# The problem asks for a 2-coloring of a graph such that every edge connects
# vertices of different colors (1 and 2). This is equivalent to checking if
# the graph is bipartite. If it is, we want the lexicographically smallest
# assignment of 1s and 2s.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = state 1, 2 = state 2
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is unvisited, we start a BFS/DFS. To make it lexicographically
    # smallest, we prefer assigning state 1 whenever possible.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
