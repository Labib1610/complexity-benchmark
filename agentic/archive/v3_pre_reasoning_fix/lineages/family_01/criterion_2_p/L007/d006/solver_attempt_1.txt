```python
import sys

# The condition is that for every edge (u, v), mode(u) XOR mode(v) = 3.
# The modes are {1, 2}.
# 1 XOR 1 = 0
# 1 XOR 2 = 3
# 2 XOR 1 = 3
# 2 XOR 2 = 0
# Thus, the condition mode(u) XOR mode(v) = 3 is equivalent to saying
# mode(u) != mode(v).
# This means the graph must be bipartite.
# We need to assign 1 or 2 to each node such that adjacent nodes have different values.
# To get the lexicographically smallest sequence, we want p_1 to be 1.
# If p_1 = 1, then all neighbors of 1 must be 2, all neighbors of neighbors must be 1, etc.
# This is a standard 2-coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # To get lexicographically smallest, we prefer 1 for the first node of each component
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
