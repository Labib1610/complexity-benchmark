import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # 2-SAT approach:
    # Each employee i can be in shift 1 (x_i=0) or shift 2 (x_i=1).
    # Conflict (u, v) means x_u != x_v.
    # This is a bipartite graph check.
    color = [-1] * (n + 1)
    
    # To handle the sequential requirement:
    # We need to check if a partial assignment is valid.
    # A partial assignment is valid if the remaining graph is bipartite.
    # We can pre-calculate the connected components and their bipartite coloring.
    
    comp = [-1] * (n + 1)
    comp_id = 0
    bipartite = True
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            comp[i] = comp_id
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        comp[v] = comp_id
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            comp_id += 1
            
    # For each component, there are two ways to color it:
    # 1. Original coloring (color[i])
    # 2. Flipped coloring (1 - color[i])
    # We need to decide for each component which way to color it
    # to satisfy the sequential requirement.
    
    # Actually, the problem is simpler:
    # For each component, we can either keep the coloring or flip it.
    # Let c_k be the choice for component k (0 or 1).
    # Employee i is in shift (color[i] ^ c_{comp[i]}) + 1.
    # We want to pick shift 1 if possible.
    
    # The sequential requirement:
    # For i=1 to n:
    # Can we set shift[i] = 1?
    # This is possible if there exists a valid assignment for all components
    # such that shift[i] = 1.
    
    # Since components are independent, we just need to check if
    # setting shift[i] = 1 is consistent with previous choices.
    
    final_shifts = [0] * (n + 1)
    comp_fixed = {} # comp_id -> fixed_color_of_a_node_in_it
    
    print("YES")
    for i in range(1, n + 1):
        c_id = comp[i]
        # Try setting shift[i] = 1
        # This means color[i] ^ c_k = 0 => c_k = color[i]
        target_c = color[i]
        
        if c_id in comp_fixed:
            final_shifts[i] = (color[i] ^ comp_fixed[c_id]) + 1
        else:
            # Check if setting c_k = target_c is valid
            # It is always valid because components are independent
            comp_fixed[c_id] = target_c
            final_shifts[i] = 1
            
    print(*(final_shifts[1:]))

solve()
