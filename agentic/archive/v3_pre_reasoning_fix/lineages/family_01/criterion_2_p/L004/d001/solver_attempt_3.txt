```python
import sys

# The problem asks for a 2-coloring of a graph such that the resulting 
# sequence of colors (1 or 2) is lexicographically smallest.
# This is equivalent to checking if the graph is bipartite and then 
# greedily assigning color 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If an employee is not colored, we try assigning color 1.
    # If that leads to a conflict, we assign color 2.
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
