def solve():
    import sys
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        adj[u].append(v)
        adj[v].append(u)
    color = [0]*n
    from collections import deque
    for i in range(n):
        if color[i]==0:
            color[i]=1
            dq = deque([i])
            while dq:
                u = dq.popleft()
                cu = color[u]
                nv = 3 - cu
                for w in adj[u]:
                    if color[w]==0:
                        color[w]=nv
                        dq.append(w)
                    elif color[w]!=nv:
                        sys.stdout.write("NO")
                        return
    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
