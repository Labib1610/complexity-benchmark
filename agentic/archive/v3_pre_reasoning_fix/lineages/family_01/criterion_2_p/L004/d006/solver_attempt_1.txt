```python
import sys

# The problem asks for a 2-coloring of a graph such that no two adjacent nodes
# have the same color. This is equivalent to checking if the graph is bipartite.
# We need the lexicographically smallest assignment, meaning we prefer 1 over 2
# for each node in order from 1 to n.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] = 0 (unvisited), 1 (tier 1), 2 (tier 2)
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is unvisited, we start a BFS/DFS. To make it lexicographically
    # smallest, we assign 1 to the first node of each connected component.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
