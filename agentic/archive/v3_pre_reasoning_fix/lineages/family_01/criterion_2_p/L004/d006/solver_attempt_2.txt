```python
import sys

# The condition "product of modes is 2" means for every edge (u, v),
# one server must be 1 and the other must be 2.
# This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color it with 1 and 2.
# The set S is the set of servers assigned mode 1.
# We want to maximize the presence of lower-numbered servers in S.
# This means for each connected component, we have two choices of coloring.
# Let the two colorings be C1 and C2.
# In C1, nodes in partition A are 1, nodes in partition B are 2.
# In C2, nodes in partition A are 2, nodes in partition B are 1.
# We want to choose the coloring such that the smallest node index
# that is 1 is as small as possible.
# Actually, the condition "lowest-numbered server present in exactly one of the two sets"
# means we compare the sets S_A and S_B.
# Let the smallest node index be i. If i is in S_A but not S_B, S_A is preferred.
# For each component, let the two partitions be P1 and P2.
# Let min(P1) be the smallest index in P1, min(P2) be the smallest index in P2.
# If min(P1) < min(P2), we should assign 1 to P1 to make min(P1) part of S.
# If min(P2) < min(P1), we should assign 1 to P2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Partition 1: nodes with color 1, Partition 2: nodes with color 2
            p1 = [node for node, c in component if c == 1]
            p2 = [node for node, c in component if c == 2]
            
            m1 = min(p1) if p1 else float('inf')
            m2 = min(p2) if p2 else float('inf')
            
            # We want the smallest index to be in S (mode 1).
            # If m1 < m2, we assign mode 1 to p1.
            # If m2 < m1, we assign mode 1 to p2.
            if m1 < m2:
                for node in p1: ans[node] = 1
                for node in p2: ans[node] = 2
            else:
                for node in p1: ans[node] = 2
                for node in p2: ans[node] = 1
                
    print("YES")
    print(*(ans[1:]))

solve()
```
