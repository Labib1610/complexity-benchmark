```python
import sys

# The problem states: XOR of bands must be 3.
# Bands are 1 (binary 01) and 2 (binary 10).
# 1 XOR 2 = 3.
# This means for every link (u, v), band[u] must be different from band[v].
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1s and 2s).
# For each connected component, we have two choices:
# 1. Assign the first node in the component to 1.
# 2. Assign the first node in the component to 2.
# To get the lexicographically smallest result, for each component,
# we try to assign 1 to the smallest index node in that component.
# If that works, we check if assigning 1 to the smallest index node
# results in a smaller sequence than assigning 2.
# Actually, since we want the lexicographically smallest sequence,
# for each connected component, we find the smallest index node 's'.
# We must assign band[s] = 1 to make the sequence lexicographically smallest.
# Then we propagate the constraints. If we encounter a contradiction, output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want color[i] = 1
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
