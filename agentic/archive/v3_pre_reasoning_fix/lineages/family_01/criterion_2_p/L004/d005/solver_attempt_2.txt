```python
import sys

# The problem requires that for every link (u, v), the product of tiers is 2.
# Since tiers are 1 or 2, the product is 2 if and only if one server is 1 and the other is 2.
# This is equivalent to a bipartite graph coloring problem where 1 and 2 are the two colors.
# We need to find the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for server 1, then server 2, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = tier 1, 2 = tier 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component.
            # To get lexicographically smallest, we want to assign 1 to the first node
            # of each component if possible.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
