import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest sequence, we want to assign 1
    # to the smallest possible index whenever possible.
    # This is a standard 2-coloring problem.
    # For each connected component, we can pick the starting node's color.
    # To make the sequence lexicographically smallest, for each component,
    # we check if assigning 1 to the smallest index in that component
    # leads to a valid coloring.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign color 1 to i
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The current coloring is valid.
            # Check if we should flip the colors in this component.
            # We want the smallest index in this component to have color 1.
            # The smallest index is 'i' because we iterate 1 to n.
            # Since we already set color[i] = 1, this is already the 
            # lexicographically smallest choice for this component.
            
    print("YES")
    print(*(color[1:]))

solve()
