import sys

# The condition is that for every connection (u, v), (polarity[u] & polarity[v]) == 0.
# Polarities are 1 (binary 01) or 2 (binary 10).
# 1 & 1 = 1 (non-zero)
# 2 & 2 = 2 (non-zero)
# 1 & 2 = 0 (zero)
# 2 & 1 = 0 (zero)
# This means for every connection (u, v), polarity[u] must be different from polarity[v].
# This is exactly the definition of a bipartite graph.
# We need to color the graph with 1s and 2s such that no two adjacent nodes have the same color.
# To minimize the sequence (p1, p2, ..., pn) lexicographically, we want p1=1, then p2=1 if possible, etc.
# Since it's a bipartite graph, for each connected component, there are two possible colorings.
# If we fix the color of one node in a component, the rest are determined.
# To minimize the sequence, for each component, we check the smallest index node 'i'.
# We can either set color[i] = 1 or color[i] = 2.
# If we set color[i] = 1, we get a specific coloring for the component.
# If we set color[i] = 2, we get the inverse coloring.
# We compare the two colorings lexicographically and pick the smaller one.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. Current color assignment
            # 2. Flipped color assignment (1->2, 2->1)
            # We want the one that is lexicographically smaller.
            # Since we want to minimize the sequence, we look at the smallest index in the component.
            # Let the smallest index be 'min_idx'.
            # If color[min_idx] is 1, the current assignment is better than the flipped one
            # because the flipped one would have color[min_idx] = 2.
            # If color[min_idx] is 2, the flipped one is better.
            
            min_idx = min(component)
            if color[min_idx] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
