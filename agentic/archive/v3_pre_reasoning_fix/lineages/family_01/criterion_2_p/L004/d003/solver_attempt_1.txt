```python
import sys

# The problem requires that for every link (u, v), power[u] * power[v] = 2.
# Since power settings are 1 or 2, the only way to get a product of 2 is
# (1, 2) or (2, 1). This means the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# To get the lexicographically smallest sequence, we want unit 1 to be 1.
# If unit 1 is 1, then all its neighbors must be 2, their neighbors 1, etc.
# If a component is not connected to unit 1, we want the smallest unit in that
# component to be 1 to keep the sequence lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we process components.
    # For each component, we try setting the smallest index to 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, setting color[i] = 1
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
