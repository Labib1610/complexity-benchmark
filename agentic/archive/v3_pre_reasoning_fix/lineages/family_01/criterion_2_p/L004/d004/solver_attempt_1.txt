import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Calibration factors: 1 or 2.
    # Belt (u, v) requires factor(u) != factor(v) because 
    # one must be twice the other (1 and 2).
    # This is a bipartite coloring problem.
    
    color = [0] * (n + 1) # 0: unassigned, 1: factor 1, 2: factor 2
    
    # To get lexicographically smallest, we prefer 1 over 2.
    # We iterate 1 to n. If uncolored, start BFS/DFS.
    # Since we want lexicographically smallest, we try assigning 1 first.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
