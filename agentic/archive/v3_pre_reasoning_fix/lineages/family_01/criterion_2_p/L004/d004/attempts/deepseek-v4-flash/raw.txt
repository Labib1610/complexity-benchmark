import sys

def main():
    input = sys.stdin.buffer.readline
    n_m = input().split()
    if not n_m:
        return
    n = int(n_m[0]); m = int(n_m[1])
    g = [[] for _ in range(n+1)]
    for _ in range(m):
        u, v = map(int, input().split())
        g[u].append(v)
        g[v].append(u)

    color = [-1]*(n+1)
    for s in range(1, n+1):
        if color[s] != -1:
            continue
        color[s] = 0
        stack = [s]
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in g[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    stack.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return
    out = " ".join(str(color[i] + 1) for i in range(1, n+1))
    sys.stdout.write("YES\n" + out + "\n")

if __name__ == "__main__":
    main()

import sys

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)

    del data

    color = [-1] * (n + 1)
    for s in range(1, n + 1):
        if color[s] != -1:
            continue
        color[s] = 0
        stack = [s]
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    stack.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO\n")
                    return

    ans = [str(color[i] + 1) for i in range(1, n + 1)]
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(ans))
    sys.stdout.write("\n")
