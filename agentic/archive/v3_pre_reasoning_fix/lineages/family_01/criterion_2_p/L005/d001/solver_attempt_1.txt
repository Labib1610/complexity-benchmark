import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    assignment = [0] * (n + 1)
    
    # We need to find the lexicographically smallest assignment (1s preferred)
    # This is equivalent to 2-coloring. Each connected component can be 
    # colored in two ways (swap 1 and 2).
    
    components = []
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            comp = []
            stack = [(i, 1)]
            visited[i] = True
            color[i] = 1
            comp.append(i)
            
            idx_stack = 0
            while idx_stack < len(stack):
                u, c = stack[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                        comp.append(v)
                    elif color[v] == c:
                        print("NO")
                        return
            components.append(comp)
            
    # For each component, we have two choices:
    # Choice A: nodes with color 1 get shift 1, nodes with color 2 get shift 2
    # Choice B: nodes with color 1 get shift 2, nodes with color 2 get shift 1
    # We want to pick the choice that makes the assignment lexicographically smallest.
    
    final_assignment = [0] * (n + 1)
    for comp in components:
        # Find the smallest index in this component
        min_node = min(comp)
        
        # If we set min_node to shift 1:
        # If color[min_node] == 1, we must use Choice A
        # If color[min_node] == 2, we must use Choice B
        
        if color[min_node] == 1:
            # Choice A: color 1 -> 1, color 2 -> 2
            for node in comp:
                final_assignment[node] = color[node]
        else:
            # Choice B: color 1 -> 2, color 2 -> 1
            for node in comp:
                final_assignment[node] = 3 - color[node]
                
    print("YES")
    print(*(final_assignment[1:]))

solve()
