import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition s[u] + s[v] = 3 means s[u] != s[v].
    # This is a bipartite coloring problem.
    # We want the lexicographically smallest sequence.
    # For each connected component, we have two choices:
    # 1. Start with 1 at the smallest node in the component.
    # 2. Start with 2 at the smallest node in the component.
    # To get the lexicographically smallest sequence, we should pick the 
    # setting for the smallest node in the component such that the 
    # resulting sequence is minimized.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component nodes
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find all nodes in component
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # Let the colors be 1 and 2.
            # We can either keep the current coloring or swap 1s and 2s.
            # We want the smallest node in the component to have 1 if possible.
            # Actually, we want the sequence to be lexicographically smallest.
            # Let min_node be the smallest index in this component.
            # If we set color[min_node] = 1, we get one sequence.
            # If we set color[min_node] = 2, we get another.
            # We compare the two sequences at the first index where they differ.
            # Since the component is fixed, we just need to check which 
            # assignment makes the smallest node 1.
            
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
