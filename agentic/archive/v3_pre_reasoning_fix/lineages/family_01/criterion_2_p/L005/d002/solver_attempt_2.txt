import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum of ratings must be 3" means if u has rating 1, v must have 2,
    # and if u has 2, v must have 1. This is a bipartite coloring problem.
    # We want to minimize the sequence (s1, s2, ..., sn) lexicographically.
    # This means we prefer 1 over 2 for each node in order.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start a BFS/DFS for this connected component.
            # To minimize lexicographically, we try assigning 1 to the current node.
            stack = [(i, 1)]
            assignment[i] = 1
            
            component = []
            
            while stack:
                u, val = stack.pop()
                component.append((u, val))
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - val
                        stack.append((v, 3 - val))
                    elif assignment[v] == val:
                        print("NO")
                        return
            
    print("YES")
    print(*(assignment[1:]))

solve()
