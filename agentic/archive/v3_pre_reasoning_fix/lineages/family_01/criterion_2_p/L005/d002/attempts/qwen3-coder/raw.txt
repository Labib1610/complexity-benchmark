```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color array: -1 means unvisited, 0 means color 0, 1 means color 1
    # We'll map colors to pressures later
    color = [-1] * n
    visited = [False] * n
    
    # Check if graph is bipartite and find coloring
    for i in range(n):
        if not visited[i]:
            queue = deque([i])
            color[i] = 0
            visited[i] = True
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 1 - current_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # Verify all edges have sum 3
    for u, v in edges:
        # Color 0 -> pressure 1, Color 1 -> pressure 2
        p_u = 1 if color[u] == 0 else 2
        p_v = 1 if color[v] == 0 else 2
        if p_u + p_v != 3:
            print("NO")
            return
    
    # To get lexicographically smallest sequence, we need to decide which color 
    # maps to which pressure. Try both mappings and check lex order.
    
    # Mapping 0 -> 1, 1 -> 2
    result1 = [1 if color[i] == 0 else 2 for i in range(n)]
    
    # Mapping 0 -> 2, 1 -> 1  
    result2 = [2 if color[i] == 0 else 1 for i in range(n)]
    
    # Choose the lexicographically smaller one
    if result1 <= result2:
        result = result1
    else:
        result = result2
        
    print("YES")
    print(" ".join(map(str, result)))

solve()
```
