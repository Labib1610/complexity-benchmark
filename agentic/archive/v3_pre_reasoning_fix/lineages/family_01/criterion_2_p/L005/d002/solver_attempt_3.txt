import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum of pressure states must be 3" means
    # if valve u has state s_u, then valve v must have state 3 - s_u.
    # This is a standard 2-coloring problem on a graph.
    # We want the lexicographically smallest sequence (s_1, s_2, ..., s_n).
    # To achieve this, we greedily assign 1 to the first unvisited node in each component.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from i, assigning color 1
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                
                # Neighbors must have color 3 - c
                target_c = 3 - c
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = target_c
                        stack.append((v, target_c))
                    elif colors[v] != target_c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
