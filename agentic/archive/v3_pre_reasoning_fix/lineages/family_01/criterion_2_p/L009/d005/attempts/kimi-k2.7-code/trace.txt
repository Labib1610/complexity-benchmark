import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data: 
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n+1)]
    self_loop = False
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        if u == v:
            self_loop = True
        adj[u].append(v)
        adj[v].append(u)
    if self_loop:
        sys.stdout.write("NO")
        return
    color = [-1]*(n+1)
    comp = []
    from collections import deque
    for i in range(1, n+1):
        if color[i] != -1:
            continue
        # BFS
        comp.clear()
        dq = deque([i])
        color[i] = 0
        while dq:
            u = dq.popleft()
            comp.append(u)
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    dq.append(v)
                elif color[v] == cu:
                    sys.stdout.write("NO")
                    return
        if color[i] == 1:
            for v in comp:
                color[v] ^= 1
    out = ['YES']
    out.append(' '.join(str(color[i]+1) for i in range(1, n+1)))
    sys.stdout.write('\n'.join(out))

if __name__ == '__main__':
    solve()
