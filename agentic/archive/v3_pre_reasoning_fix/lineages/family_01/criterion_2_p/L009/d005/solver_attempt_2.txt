```python
import sys

# The problem states:
# Window 1: [0, 2]
# Window 2: [1, 3]
# Intersection of Window 1 and Window 1: [0, 2] (length 2)
# Intersection of Window 2 and Window 2: [1, 3] (length 2)
# Intersection of Window 1 and Window 2: [1, 2] (length 1)
# Intersection of Window 2 and Window 1: [1, 2] (length 1)
#
# For a handoff to be successful, the intersection must be exactly 1 hour.
# This means for every pair (u, v), window(u) must be different from window(v).
# This is equivalent to 2-coloring a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest sequence of 1s and 2s.
# This means for each connected component, we try to assign 1 to the smallest index node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
