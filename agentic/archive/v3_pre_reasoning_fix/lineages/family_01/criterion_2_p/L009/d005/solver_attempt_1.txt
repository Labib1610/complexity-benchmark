import sys

# The problem asks for a 2-coloring of a graph such that every edge connects
# nodes of different colors (1 and 2). This is equivalent to checking if the
# graph is bipartite. To get the lexicographically smallest sequence, we
# iterate through each connected component and try assigning 1 to the smallest
# index node in that component. If that leads to a valid coloring, we keep it.
# Since we want the lexicographically smallest sequence, for each component,
# we try assigning 1 to the smallest uncolored node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # After coloring the component, check if this is the best we can do.
            # Actually, for a bipartite graph, once the first node is fixed,
            # the rest are determined. Since we want the smallest sequence,
            # we always prefer color 1 for the smallest index in the component.
            # This greedy approach works because components are independent.
            
    print("YES")
    print(*(color[1:]))

solve()
