import sys

# The problem states that for each connection (u, v), the stability metric
# (p_u + p_v) / 2 must be 1.5. This means p_u + p_v = 3.
# Since p_u, p_v are in {1, 2}, this implies p_u != p_v.
# This is a bipartite graph coloring problem.
# We need to check if the graph is bipartite. If it is, we want the
# lexicographically smallest assignment.
# For each connected component, we have two possible colorings (1, 2) or (2, 1).
# To make the sequence lexicographically smallest, for each component,
# we look at the smallest index 'i' in that component. We should assign
# p_i = 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a stack to traverse the component
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring the component, check if we should flip
            # We want the smallest index in the component to have color 1
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
