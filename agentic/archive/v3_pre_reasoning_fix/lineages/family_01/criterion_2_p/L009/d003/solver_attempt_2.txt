import sys

def solve():
    # The condition: P_all ^ P_ext = 3
    # P_all = g_1 ^ g_2 ^ ... ^ g_n
    # P_ext = P_all ^ g_u ^ g_v
    # So P_all ^ P_ext = g_u ^ g_v = 3
    # Since g_i is 1 or 2, g_u ^ g_v = 3 means one is 1 and the other is 2.
    # This is a 2-coloring problem on a graph where each edge (u, v)
    # requires g_u != g_v.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try to assign 1 to the first node of each component
            # to get the lexicographically smallest sequence.
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
