def solve():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it); m = next(it)
    adj = [[] for _ in range(n)]

                        stack.append(v)

                        return

import sys

def solve():
    import sys
    input = sys.stdin.buffer.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1; v -= 1
        adj[u].append(v)
        adj[v].append(u)
    color = [-1] * n
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            stack = [i]
            while stack:
                u = stack.pop()
                cu = color[u]
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = cu ^ 1
                        stack.append(v)
                    elif color[v] == cu:
                        sys.stdout.write("NO\n")
                        return
    sys.stdout.write("YES\n" + ' '.join(str(c + 1) for c in color) + "\n")

if __name__ == "__main__":
    solve()
