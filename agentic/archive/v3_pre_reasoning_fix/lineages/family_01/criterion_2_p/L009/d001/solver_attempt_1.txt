import sys

# The condition x_u + x_v = 3 means that if x_u = 1, then x_v = 2, and if x_u = 2, then x_v = 1.
# This is a bipartite graph problem. Each connected component can be colored in two ways.
# To get the lexicographically smallest assignment, for each connected component,
# we should try to assign 1 to the smallest index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # x_u + x_u = 3 => 2*x_u = 3, impossible
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want the smallest index
            # in this component to be 1.
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if this assignment is the best for this component.
            # We have two choices: 
            # 1. assignment[i] = 1
            # 2. assignment[i] = 2 (which flips all values in the component)
            # We want the smallest index in the component to have the smallest possible value.
            # The smallest index in the component is 'i' because we iterate 1..n.
            # So we want assignment[i] = 1.
            # Since we already set assignment[i] = 1, this is already the best.
            
    print("YES")
    print(*(assignment[1:]))

solve()
