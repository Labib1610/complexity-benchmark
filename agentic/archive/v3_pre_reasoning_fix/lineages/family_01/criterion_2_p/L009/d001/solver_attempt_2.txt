import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # To minimize sum_{i=1}^n s_i * 2^{n-i}, we want s_1 to be 1, then s_2 to be 1, etc.
    # This is equivalent to lexicographical minimization of (s_1, s_2, ..., s_n).
    # Since s_i is 1 or 2, we prefer 1 over 2 at each step i=1 to n.
    
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard 2-coloring check
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # For this component, we have two possible assignments:
            # 1. color[i] = 1, then all nodes v have color[v]
            # 2. color[i] = 2, then all nodes v have 3 - color[v]
            # We want to pick the one that makes the sequence (s_1, ..., s_n) lexicographically smallest.
            
            # Find the first node in this component
            first_node = min(component)
            
            # If we set color[first_node] = 1, the assignment is fixed for the component.
            # If we set color[first_node] = 2, the assignment is fixed.
            # We compare the two assignments at the smallest index i in the component.
            
            # Actually, we just need to check which assignment gives a smaller value at the 
            # smallest index i in the component.
            
            # Let's re-evaluate: we want to minimize the sum.
            # For each component, we have two choices. We should pick the choice that 
            # assigns 1 to the smallest index in the component if possible.
            
            # Let's check the color of the smallest index in the component.
            # If color[first_node] == 1, we keep it.
            # If color[first_node] == 2, we flip all colors in the component.
            
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
