import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)

    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        color[i] = 0
        q = deque([i])
        while q:
            v = q.popleft()
            cv = color[v]
            for nb in adj[v]:
                if color[nb] == -1:
                    color[nb] = cv ^ 1
                    q.append(nb)
                elif color[nb] == cv:
                    sys.stdout.write('NO')
                    return

    ans = ' '.join(str(color[i] + 1) for i in range(1, n + 1))
    sys.stdout.write('YES\n' + ans)

if __name__ == '__main__':
    solve()
