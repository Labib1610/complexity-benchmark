```python
import sys

# The problem asks for a gain multiplier of 1 or 2 for each amplifier.
# The product of multipliers for each connection (u, v) must be 2.
# Let g_i be the multiplier for amplifier i.
# g_u * g_v = 2.
# Since g_i is in {1, 2}, the only way to get a product of 2 is if one is 1 and the other is 2.
# This is a bipartite coloring problem.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# To get the lexicographically smallest sequence, we want g_1 = 1.
# If g_1 = 1, then all neighbors of 1 must be 2, all neighbors of neighbors must be 1, etc.
# We perform a BFS/DFS to color each connected component.
# For each component, if we fix the color of one node, the rest are determined.
# To make the sequence lexicographically smallest, for each component, we try to set the
# smallest index node in that component to 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want the first node in the component
            # to be 1 if possible.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
