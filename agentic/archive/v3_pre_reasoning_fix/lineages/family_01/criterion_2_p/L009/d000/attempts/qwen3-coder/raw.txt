```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 = unvisited, 1 = channel 1, 2 = channel 2
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS for each connected component
    for start in range(1, n + 1):
        if color[start] == 0:  # Not yet visited
            queue = deque([start])
            color[start] = 1  # Assign channel 1 to start node
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        # Assign opposite channel
                        color[neighbor] = 3 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # Conflict found - not bipartite
                        print("NO")
                        return
    
    # If we reach here, graph is bipartite
    print("YES")
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:  # Isolated node, assign channel 1
            result.append("1")
        else:
            result.append(str(color[i]))
    
    print(" ".join(result))

solve()
```
