```python
import sys

# The problem states:
# Window 1: maintenance [0, 2]
# Window 2: maintenance [1, 3]
# Sync check active when exactly one of the two servers is in maintenance.
# Let s_u be the start time (0 or 1). Maintenance interval is [s_u, s_u + 2].
# Sync check active during [s_u, s_u + 2] XOR [s_v, s_v + 2].
# The length of the symmetric difference of [s_u, s_u + 2] and [s_v, s_v + 2]
# is 2 * |s_u - s_v|.
# We need 2 * |s_u - s_v| = 2, which means |s_u - s_v| = 1.
# Since s_u, s_v \in {0, 1}, this means s_u != s_v.
# Thus, for every edge (u, v), we must have s_u != s_v.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we assign 1 to the first
# node of each connected component, then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for component
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                next_color = 2 if color == 1 else 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = next_color
                        stack.append((v, next_color))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
