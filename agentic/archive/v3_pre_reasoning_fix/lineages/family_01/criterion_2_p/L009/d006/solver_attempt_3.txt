import sys

# The problem requires that for every edge (u, v), gain[u] * gain[v] = 2.
# Since gain[i] is either 1 or 2, the only way to get a product of 2 is
# if one amplifier has gain 1 and the other has gain 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# To get the lexicographically smallest sequence, we want to assign 1 to
# as many early indices as possible.
# For each connected component, there are two possible colorings (swap 1 and 2).
# We choose the coloring that results in a smaller value at the first node
# of the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to find component and check bipartiteness
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings:
            # 1. Keep as is
            # 2. Swap 1 and 2
            # We want the lexicographically smallest result.
            # Find the first node in this component.
            first_node = min(node for node, c in component)
            
            # Try coloring 1: color[first_node] = 1
            # Try coloring 2: color[first_node] = 2
            # We want the one that makes the assignment at 'first_node' 1.
            # If color[first_node] is 1, we keep it. If it's 2, we swap.
            
            if color[first_node] == 1:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
