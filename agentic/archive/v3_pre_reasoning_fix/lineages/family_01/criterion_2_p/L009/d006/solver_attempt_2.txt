```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the start times are different.
# Time 1: Transmit [0, 1], Receive [1, 2]
# Time 2: Transmit [1, 2], Receive [2, 3]
# Handoff condition: One Transmit, one Receive at the same time.
# If u is Time 1 and v is Time 2:
#   u Transmit [0, 1], Receive [1, 2]
#   v Transmit [1, 2], Receive [2, 3]
#   At [1, 2], u is Receive and v is Transmit. This works.
# If u is Time 2 and v is Time 1:
#   u Transmit [1, 2], Receive [2, 3]
#   v Transmit [0, 1], Receive [1, 2]
#   At [1, 2], u is Transmit and v is Receive. This works.
# If u and v are both Time 1:
#   u Transmit [0, 1], Receive [1, 2]
#   v Transmit [0, 1], Receive [1, 2]
#   No overlap where one is Transmit and one is Receive.
# If u and v are both Time 2:
#   u Transmit [1, 2], Receive [2, 3]
#   v Transmit [1, 2], Receive [2, 3]
#   No overlap where one is Transmit and one is Receive.
# Thus, the condition is simply that u and v must have different start times.
# This is a standard bipartite graph coloring problem.
# We need the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is uncolored, we assign it 1 and perform BFS/DFS.
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
