```python
import sys

# The problem asks for an assignment of 1 or 2 to each node such that for every edge (u, v),
# (tier[u] & tier[v]) == 0.
# Since 1 is binary 01 and 2 is binary 10, the condition (tier[u] & tier[v]) == 0
# means that if tier[u] is 1, tier[v] cannot be 1. If tier[u] is 2, tier[v] cannot be 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (say 0 and 1).
# Let the two sets of the bipartition be A and B.
# If we assign 1 to set A and 2 to set B, all edges (u, v) will have (1 & 2) = 0.
# If we assign 2 to set A and 1 to set B, all edges (u, v) will have (2 & 1) = 0.
# We want the lexicographically smallest sequence.
# We have two possible assignments:
# 1. For each connected component, assign 1 to nodes in set A and 2 to nodes in set B.
# 2. For each connected component, assign 2 to nodes in set A and 1 to nodes in set B.
# To get the lexicographically smallest sequence, for each component, we compare the
# resulting sequences and pick the one that is smaller at the first index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        else:
            print("NO")
            return
        idx += 2

    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two options for this component:
            # Option 1: color 0 -> 1, color 1 -> 2
            # Option 2: color 0 -> 2, color 1 -> 1
            # We want the smallest value at the smallest index in the component.
            comp_nodes = sorted([node for node, c in component])
            first_node = comp_nodes[0]
            
            # Find color of first_node
            c_first = -1
            for node, c in component:
                if node == first_node:
                    c_first = c
                    break
            
            # If c_first is 0:
            # Opt 1: first_node gets 1
            # Opt 2: first_node gets 2
            # We prefer Opt 1.
            if c_first == 0:
                for node, c in component:
                    res[node] = 1 if c == 0 else 2
            else:
                for node, c in component:
                    res[node] = 2 if c == 0 else 1
                    
    print("YES")
    print(*(res[1:]))

solve()
```
