```python
import sys

# The problem states that for each interaction (u, v), the metabolic load
# must be 1^2 + 2^2 = 5. This means for every edge (u, v), one marker
# must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, for each connected component,
# we have two choices:
# 1. Assign 1 to the smallest index node in the component.
# 2. Assign 2 to the smallest index node in the component.
# We choose the assignment that results in a smaller value at the smallest index.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            idx_comp = 0
            while idx_comp < len(component):
                u, c = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For this component, we have two colorings:
            # Option 1: color[v] as is
            # Option 2: 3 - color[v]
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            min_node = min(node for node, c in component)
            
            # If we keep color[min_node], the value is color[min_node].
            # If we flip, the value is 3 - color[min_node].
            # We want the value at min_node to be 1 if possible.
            
            flip = False
            if color[min_node] == 2:
                flip = True
            
            for node, c in component:
                if flip:
                    ans[node] = 3 - c
                else:
                    ans[node] = c
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
