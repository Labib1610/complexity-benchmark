```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try to 2-color the graph (check if bipartite)
    color = [0] * (n + 1)  # 0 means unvisited, 1 and 2 are colors
    queue = deque()
    
    # Process each connected component
    for start in range(1, n + 1):
        if color[start] == 0:
            color[start] = 1
            queue.append(start)
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = 3 - current_color  # Switch between 1 and 2
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # Verify the solution
    for u, v in edges:
        if color[u] & color[v] != 0:  # Bitwise AND should be 0
            print("NO")
            return
    
    # Output the lexicographically smallest solution
    # Since we start with node 1 and assign it color 1, 
    # and we always assign the smallest possible color first,
    # this gives us the lexicographically smallest valid sequence
    print("YES")
    print(" ".join(str(color[i]) for i in range(1, n + 1)))

solve()
```
