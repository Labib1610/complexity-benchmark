```python
import sys

# The problem states that for each connection (u, v), the stability metric
# (power_tier[u] XOR power_tier[v]) must be 3.
# Power tiers are 1 (binary 01) or 2 (binary 10).
# 1 XOR 1 = 0
# 1 XOR 2 = 3
# 2 XOR 1 = 3
# 2 XOR 2 = 0
# Thus, for every connection (u, v), power_tier[u] must be different from power_tier[v].
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest sequence.
# To get the lexicographically smallest sequence, we want p_i = 1 whenever possible.
# For each connected component, we can pick the smallest node and assign it 1,
# then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2.
            # For each component, the first node encountered (smallest index)
            # should be assigned 1.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
