import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [[] for _ in range(n+1)]
    idx = 2
    for _ in range(m):
        u = data[idx]; v = data[idx+1]; idx += 2
        adj[u].append(v)
        adj[v].append(u)
    color = [0]*(n+1)
    possible = True
    for i in range(1, n+1):
        if color[i] == 0:
            color[i] = 1
            q = deque([i])
            while q:
                u = q.popleft()
                cu = color[u]
                nc = 2 if cu == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = nc
                        q.append(v)
                    elif color[v] == cu:
                        print("NO")
                        return
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(str(color[i]) for i in range(1, n+1)))
    # newline optional
    sys.stdout.write("\n")

if __name__ == "__main__": main()
