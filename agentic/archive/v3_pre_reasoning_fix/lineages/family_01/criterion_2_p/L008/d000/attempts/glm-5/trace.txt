from collections import deque

def solve():

    idx = 0
    n, m = int(data[idx]), int(data[idx+1])

    color = [0] * (n+1)  # 0: unvisited, 1: team 1, 2: team 2

        color[start] = 1
        queue = deque([start])

                    color[v] = 3 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
