```python
import sys

# The problem states that for every overlapping pair (u, v), the depth coordinate
# of one must be twice the other. Since depths are {1, 2}, this means one must
# be 1 and the other must be 2. This is a standard 2-coloring problem on a graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want gear 1 to be 1 if possible.
# If gear 1 is 1, its neighbors must be 2, their neighbors must be 1, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a component is not colored, we try assigning 1 to the smallest index.
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
