```python
import sys

# The problem asks for an assignment of 1 or 2 to each node such that for every edge (u, v),
# t_u * t_v = 2.
# Since t_i is in {1, 2}, the only way t_u * t_v = 2 is if one is 1 and the other is 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want t_1 to be 1 if possible.
# If t_1 = 1, then all neighbors of 1 must be 2, all neighbors of those neighbors must be 1, etc.
# We process each connected component. For each component, we try setting the first node
# encountered to 1, and propagate the constraints. If we find a conflict, it's impossible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we prefer 1 over 2.
            # For the first node in a component, we assign 1.
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        new_color = 3 - color
                        assignment[v] = new_color
                        stack.append((v, new_color))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
