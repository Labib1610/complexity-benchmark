```python
import sys

# The problem states that for each joint (u, v), the arithmetic mean of the phases
# s_u and s_v must be 1.5.
# (s_u + s_v) / 2 = 1.5  =>  s_u + s_v = 3
# Since s_i must be a positive integer strictly less than 3, s_i can only be 1 or 2.
# If s_u + s_v = 3 and s_i in {1, 2}, then one must be 1 and the other must be 2.
# This means s_u != s_v.
# This is equivalent to a 2-coloring problem on a graph where each edge (u, v)
# requires s_u != s_v.
# We want the lexicographically smallest assignment (s_1, s_2, ..., s_n).
# To get the lexicographically smallest, we prefer 1 over 2 for each node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 first
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
