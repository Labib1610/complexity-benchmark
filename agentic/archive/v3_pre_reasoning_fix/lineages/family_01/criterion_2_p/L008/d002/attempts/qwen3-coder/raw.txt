```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    graph = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        graph[u].append(v)
        graph[v].append(u)
    
    # Color nodes with BFS/DFS. Each component is bipartite.
    # We want to assign colors such that adjacent nodes multiply to 2.
    # That means one node is 1 and the other is 2.
    # So we need to 2-color each component (bipartite check).
    # Color 0 -> tier 1, Color 1 -> tier 2
    
    color = {}
    result = [None] * (n + 1)  # 1-indexed
    
    for start in range(1, n + 1):
        if start not in color:
            # BFS coloring
            queue = deque()
            color[start] = 0
            queue.append(start)
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                
                for v in graph[u]:
                    if v not in color:
                        color[v] = 1 - current_color
                        queue.append(v)
                    elif color[v] == current_color:
                        print("NO")
                        return
            
    # Now assign tiers based on color
    # To get lexicographically smallest, we prefer tier 1 (color 0) for earlier nodes
    for i in range(1, n + 1):
        if color[i] == 0:
            result[i] = 1
        else:
            result[i] = 2
    
    print("YES")
    print(" ".join(map(str, result[1:])))

if __name__ == "__main__":
    main()
```
