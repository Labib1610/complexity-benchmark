```python
import sys

# The stability equation is s_u^2 + s_v^2 - s_u * s_v = 3
# Given s_i in {1, 2}:
# If s_u = 1, s_v = 1: 1 + 1 - 1 = 1 != 3
# If s_u = 1, s_v = 2: 1 + 4 - 2 = 3 (Valid)
# If s_u = 2, s_v = 1: 4 + 1 - 2 = 3 (Valid)
# If s_u = 2, s_v = 2: 4 + 4 - 4 = 4 != 3
# Thus, for every joint (u, v), s_u must be different from s_v.
# This is a standard bipartite graph coloring problem.
# We want the lexicographically smallest assignment.
# To get the lexicographically smallest, we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we start a BFS/DFS.
    # To make it lexicographically smallest, we assign 1 to the first node
    # of each connected component.
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
