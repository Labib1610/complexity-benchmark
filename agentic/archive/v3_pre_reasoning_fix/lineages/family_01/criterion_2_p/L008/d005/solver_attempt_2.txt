import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # colors: 0 = unassigned, 1 = level 1, 2 = level 2
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is unvisited, we start a BFS/DFS.
    # To ensure lexicographical smallest, we prefer assigning 1 whenever possible.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                
                # Sort neighbors to process in order if needed, 
                # but standard bipartite check is sufficient.
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
