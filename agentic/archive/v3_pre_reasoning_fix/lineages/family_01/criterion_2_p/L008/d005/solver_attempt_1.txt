import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need to 2-color the graph.
    # To get the lexicographically smallest sequence, we prefer depth 1 for smaller indices.
    # We iterate from 1 to n. If a node is not colored, start a BFS/DFS.
    # To ensure lexicographical smallest, when starting a component, assign 1 to the smallest node.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                
                # To maintain lexicographical order, we want to process neighbors
                # such that we try to assign 1 whenever possible.
                # However, 2-coloring is fixed once the first node of a component is set.
                # The lexicographical requirement means we want the smallest index to have 1.
                # Since we iterate 1..n, the first node of any component will always be 1.
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
