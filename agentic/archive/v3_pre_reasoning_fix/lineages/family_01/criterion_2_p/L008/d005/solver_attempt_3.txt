```python
import sys

# The problem states that for each field (u, v), it must be anchored to both vault 1 and vault 2.
# A field is anchored to a vault if at least one of its artifacts is in that vault.
# So, for each field (u, v):
# 1. At least one of {u, v} must be in vault 1.
# 2. At least one of {u, v} must be in vault 2.
# This means {u, v} cannot be both in vault 1, and {u, v} cannot be both in vault 2.
# Therefore, u and v must have different vault assignments.
# This is equivalent to saying the graph must be bipartite.
# To get the lexicographically smallest assignment, we want v_1=1, then v_2=1 if possible, etc.
# We can use BFS/DFS to color the graph with 2 colors (1 and 2).
# For each connected component, we have two choices of coloring (swap 1 and 2).
# To make the assignment lexicographically smallest, for each component, we pick the 
# coloring that assigns 1 to the smallest index in that component if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # We have two colorings for this component:
            # 1. Current color[v]
            # 2. 3 - color[v]
            # We want the one that is lexicographically smaller.
            # Find the smallest index in the component.
            min_node = min(component)
            
            # If we keep current coloring, min_node gets color[min_node].
            # If we swap, min_node gets 3 - color[min_node].
            # We want the assignment where min_node gets 1.
            
            target_color_for_min = 1
            if color[min_node] == target_color_for_min:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
