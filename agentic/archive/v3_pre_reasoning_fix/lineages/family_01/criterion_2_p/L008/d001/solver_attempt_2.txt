import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # 0: unassigned, 1: shift 1, 2: shift 2
    
    # We need to determine the assignment greedily for 1 to n.
    # For each connected component, we have two possible colorings (swap 1 and 2).
    # The priority system says: for the first node in the component, 
    # try to assign it to 1. If that's valid, keep it. If not, assign to 2.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # New component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # The priority system: for the smallest index 'i' in this component,
            # we want color[i] to be 1.
            # If color[i] is already 1, we are good.
            # If color[i] is 2, we must flip all colors in this component.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                assignment[node] = color[node]
                
    print("YES")
    print(*(assignment[1:]))

solve()
