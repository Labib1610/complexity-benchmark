```python
import sys

# The problem states that for each joint (u, v), s_u + s_v = 3.
# Since s_i is either 1 or 2, this means s_u != s_v.
# This is a standard bipartite graph coloring problem.
# We need to color the graph with 1 and 2 such that no two adjacent nodes have the same color.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not yet colored, we assign it 1 (the smallest possible value).
# Then we perform a BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To ensure lexicographical smallest, we always try to assign 1 first.
            # If we encounter a conflict, it's impossible.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
