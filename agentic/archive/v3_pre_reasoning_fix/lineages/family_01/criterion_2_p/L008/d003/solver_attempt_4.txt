```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# level(u) + level(v) = 3. Since levels are {1, 2}, this means level(u) != level(v).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we want gear 1 to be 1 if possible.
# If gear 1 is 1, then all its neighbors must be 2, their neighbors 1, and so on.
# We iterate through all connected components. For each component, we try assigning
# the smallest unassigned node to 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    levels = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if levels[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we always prefer level 1
            stack = [(i, 1)]
            levels[i] = 1
            while stack:
                u, l = stack.pop()
                for v in adj[u]:
                    if levels[v] == 0:
                        levels[v] = 3 - l
                        stack.append((v, levels[v]))
                    elif levels[v] == l:
                        print("NO")
                        return
                        
    print("YES")
    print(*(levels[1:]))

solve()
```
