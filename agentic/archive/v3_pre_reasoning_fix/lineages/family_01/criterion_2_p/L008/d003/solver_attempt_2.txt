```python
import sys

# The problem asks to assign t_i in {1, 2} such that for each pair (u, v),
# t_u * t_v = 2.
# Since t_i can only be 1 or 2, the only way t_u * t_v = 2 is if one is 1 and the other is 2.
# This is a bipartite coloring problem.
# We want to minimize sum_{i=1}^n t_i * 3^{n-i}.
# This is equivalent to lexicographically minimizing the sequence (t_1, t_2, ..., t_n).
# Since we want to minimize the sequence, we prefer t_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1) # 0: unassigned, 1: tier 1, 2: tier 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To minimize the sequence lexicographically, we greedily pick 1 for the smallest index
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
