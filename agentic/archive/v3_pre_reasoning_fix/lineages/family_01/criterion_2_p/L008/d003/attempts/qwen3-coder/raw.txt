```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        # No constraints, assign all to level 1 for lexicographically smallest
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color assignment: 1 or 2, -1 means unvisited
    color = [-1] * (n + 1)
    
    # Check if bipartite using BFS
    for start in range(1, n + 1):
        if color[start] == -1:  # Unvisited component
            queue = deque()
            queue.append(start)
            color[start] = 1  # Assign color 1 to start
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 3 - color[node]  # Assign opposite color
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If we reach here, a valid assignment exists
    print("YES")
    result = []
    for i in range(1, n + 1):
        if color[i] == -1:
            result.append("1")  # Isolated node, assign 1
        else:
            result.append(str(color[i]))
    
    print(" ".join(result))

solve()
```
