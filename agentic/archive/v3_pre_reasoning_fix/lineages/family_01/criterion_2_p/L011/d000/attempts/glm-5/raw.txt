from collections import deque

def solve():
    input_data = sys.stdin.buffer.read().split()
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)  # 0 = unvisited, 1 or 2

        if color[start] != 0:
            continue
        # BFS from start, assign start = 1
        color[start] = 1
        queue = deque([start])
        while queue:

                if color[v] == 0:
                    color[v] = 3 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
    
    print("YES")
