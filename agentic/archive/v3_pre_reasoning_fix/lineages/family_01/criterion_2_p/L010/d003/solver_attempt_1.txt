```python
import sys

# The problem states that for every pair (u, v) in the interference records,
# the sum of their multipliers must be 3. Since multipliers are 1 or 2,
# this means if (u, v) is an interference, then multiplier(u) != multiplier(v).
# This is equivalent to a 2-coloring problem on a graph where edges represent
# interference. We want the lexicographically smallest sequence of 1s and 2s.
# Lexicographically smallest means we prefer 1 over 2 at each index i from 1 to n.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color: 0 = unassigned, 1 = multiplier 1, 2 = multiplier 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the
            # first uncolored node in the component.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
