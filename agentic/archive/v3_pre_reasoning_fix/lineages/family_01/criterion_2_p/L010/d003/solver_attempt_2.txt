import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # state: 0 = unvisited, 1 = State 1, 2 = State 2
    state = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # For each connected component, we try assigning 1 to the first unvisited node.
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            state[i] = 1
            component = []
            
            # Use a stack to perform DFS
            while stack:
                u, s = stack.pop()
                component.append((u, s))
                
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - s
                        stack.append((v, 3 - s))
                    elif state[v] == s:
                        print("NO")
                        return
    
    print("YES")
    print(*(state[1:]))

if __name__ == '__main__':
    solve()
