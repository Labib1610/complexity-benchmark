import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum of priority levels is 3" means if u has priority p,
    # v must have priority 3-p. This is a bipartite coloring problem.
    # We want the lexicographically smallest sequence (p1, p2, ..., pn).
    # This means we prefer 1 over 2 for each vessel if possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the first node
            # of each component.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            visited_in_comp = {i}
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if v not in visited_in_comp:
                        visited_in_comp.add(v)
                        q.append(v)
            
            # Try assigning 1 to the current node i
            # If that fails, try 2.
            # Actually, for a bipartite component, once the first node is fixed,
            # all others are fixed.
            # We check if 1 is valid for node i.
            
            def check(start_node, start_color):
                temp_color = {start_node: start_color}
                stack = [start_node]
                while stack:
                    u = stack.pop()
                    for v in adj[u]:
                        if v not in temp_color:
                            temp_color[v] = 3 - temp_color[u]
                            stack.append(v)
                        elif temp_color[v] == temp_color[u]:
                            return None
                return temp_color

            res = check(i, 1)
            if res is None:
                print("NO")
                return
            
            # Apply the coloring
            for node, c in res.items():
                color[node] = c
                
    print("YES")
    print(*(color[1:]))

solve()
