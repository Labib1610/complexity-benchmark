import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:

    del data

    color = [0] * (n + 1)

            stack = [start]

            while stack:
                v = stack.pop()

                nc = 3 - cv

                    if color[to] == 0:
                        color[to] = nc
                        stack.append(to)
