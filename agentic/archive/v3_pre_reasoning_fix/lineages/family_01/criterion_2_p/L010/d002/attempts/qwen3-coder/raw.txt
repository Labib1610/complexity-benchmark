import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)  # 0 means unvisited, 1 and 2 are the two colors/solvents
    
    # Check if the graph is bipartite using BFS
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        queue = deque()
        queue.append(start)
        color[start] = 1  # Assign color 1 to the starting node
        
        while queue:
            u = queue.popleft()
            current_color = color[u]
            opposite_color = 3 - current_color  # flips 1->2 and 2->1
            
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = opposite_color
                    queue.append(v)
                elif color[v] != opposite_color:
                    print("NO")
                    return
    
    # If we reach here, the graph is bipartite
    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(color[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
