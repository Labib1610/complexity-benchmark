import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = Solvent 1, 2 = Solvent 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a component is not colored, we try assigning 1 first.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            # We want to assign 1 to the current node if possible
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Collect component nodes
            q = [i]
            visited_idx = 0
            while visited_idx < len(q):
                u = q[visited_idx]
                visited_idx += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = -1 # Temporary mark
                        q.append(v)
            
            # Reset component colors to 0 to try assigning
            for node in component:
                color[node] = 0
            
            # Try assigning 1 to the first node of the component
            # If that fails, try 2.
            def check(start_node, start_color):
                stack = [(start_node, start_color)]
                color[start_node] = start_color
                assigned = [start_node]
                while stack:
                    u, c = stack.pop()
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - c
                            stack.append((v, 3 - c))
                            assigned.append(v)
                        elif color[v] == c:
                            for node in assigned:
                                color[node] = 0
                            return False
                return True

            if not check(i, 1):
                if not check(i, 2):
                    print("NO")
                    return
                    
    print("YES")
    print(*(color[1:]))

solve()
