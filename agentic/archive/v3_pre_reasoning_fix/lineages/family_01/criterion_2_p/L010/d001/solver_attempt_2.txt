import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unassigned, 1 = Facility 1, 2 = Facility 2
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a component is not colored, we try assigning 1 to the first node.
    # This is a greedy approach for 2-coloring.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assigning 1 to i
            stack = [i]
            color[i] = 1
            
            # We need to process the component. To ensure lexicographical
            # smallest, we must be careful. However, for a bipartite graph,
            # once the first node of a component is fixed, the rest are 
            # determined.
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
