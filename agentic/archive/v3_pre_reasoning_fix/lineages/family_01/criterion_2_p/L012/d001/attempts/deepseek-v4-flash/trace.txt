import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    color = [0] * n
    for start in range(n):
        if color[start] == 0:
            color[start] = 1
            stack = [start]
            while stack:
                u = stack.pop()
                cu = color[u]
                need = 3 - cu
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = need
                        stack.append(v)
                    elif color[v] != need:
                        sys.stdout.write("NO\n")
                        return
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color)))
    # newline optional but okay
    sys.stdout.write("\n")

if __name__ == "__main__":
