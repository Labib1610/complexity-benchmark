import sys
from collections import defaultdict, deque

def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    adj = defaultdict(list)
    
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    color = [-1] * n  # -1 means unvisited, 0 and 1 are the two colors (modes 1 and 2)
    
    # Check bipartiteness using BFS for each component
    for start in range(n):
        if color[start] == -1:  # Unvisited node
            queue = deque()
            queue.append(start)
            color[start] = 0  # Assign mode 1 (0 in our 0-based coloring)
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - current_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # If we reach here, the graph is bipartite
    # We need the lexicographically smallest assignment
    # Since we started with color 0 (mode 1) for the first node in each component,
    # and followed the constraint, this already gives us the lex smallest valid assignment
    # But we need to make sure that for each component, we start with 0 unless there's a better way.
    # Actually, since we're doing BFS and always starting with 0, it should be fine.
    
    # Let's re-verify by checking if we can improve by flipping the colors in any component
    # But since we want the lexicographically smallest, and we're assigning 0 first (which maps to mode 1),
    # and the constraints force the assignment, our BFS approach already gives the lex smallest.
    
    result = []
    for i in range(n):
        if color[i] == 0:
            result.append('1')
        else:
            result.append('2')
    
    print("YES")
    print(' '.join(result))

solve()
