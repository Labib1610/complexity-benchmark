import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # mode: 0 = unassigned, 1 = mode 1, 2 = mode 2
    mode = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a component is not yet assigned, we try assigning mode 1 to the 
    # smallest node in that component.
    for i in range(1, n + 1):
        if mode[i] == 0:
            # Start BFS/DFS from i, assigning mode 1
            stack = [(i, 1)]
            mode[i] = 1
            component = []
            
            while stack:
                u, m_u = stack.pop()
                component.append((u, m_u))
                
                for v in adj[u]:
                    m_v = 3 - m_u
                    if mode[v] == 0:
                        mode[v] = m_v
                        stack.append((v, m_v))
                    elif mode[v] != m_v:
                        print("NO")
                        return
    
    print("YES")
    print(*(mode[1:]))

solve()
