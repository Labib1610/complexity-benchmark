```python
import sys

# The problem states: for each channel (u, v), the product of filter powers must be 2.
# Filter powers are 1 or 2.
# Product = 1 * 1 = 1 (Invalid)
# Product = 1 * 2 = 2 (Valid)
# Product = 2 * 1 = 2 (Valid)
# Product = 2 * 2 = 4 (Invalid)
# Thus, for every edge (u, v), one must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# We want the lexicographically smallest sequence of 1s and 2s.
# For each connected component, we can assign 1 to the first node encountered
# (to keep it lexicographically small) and then propagate the constraints.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # 0: unvisited, 1: power 1, 2: power 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To make it lexicographically smallest, we prefer 1 for the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 2 if c == 1 else 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
