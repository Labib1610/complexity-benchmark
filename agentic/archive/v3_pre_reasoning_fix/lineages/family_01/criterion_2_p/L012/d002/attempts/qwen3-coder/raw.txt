```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # For each edge, the product of filter powers must be 2
    # This means one filter must be 1 and the other must be 2
    
    # This is a 2-coloring problem where:
    # - Each node gets value 1 or 2
    # - For each edge (u,v): assignment[u] * assignment[v] = 2
    
    # Since 1*1=1, 1*2=2, 2*2=4, we need exactly one 1 and one 2 for each edge
    
    # This means the graph must be bipartite
    # Color nodes such that adjacent nodes have different "requirements"
    # But it's not that simple - let's think differently
    
    # Let's model this as a graph where each connected component must be bipartite
    # For each component, try both colorings and pick lexicographically smaller one
    
    color = [-1] * n  # -1: unvisited, 0: should be 1, 1: should be 2
    result = [0] * n  # Final assignment: 1 or 2
    
    for start in range(n):
        if color[start] != -1:
            continue
            
        # BFS to check if bipartite and find a valid coloring
        queue = deque()
        queue.append(start)
        color[start] = 0
        
        # Collect nodes in component
        component = []
        
        while queue:
            u = queue.popleft()
            component.append(u)
            
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    return
        
        # For this component, we have two possible assignments:
        # color 0 -> 1, color 1 -> 2  (pattern A)
        # color 0 -> 2, color 1 -> 1  (pattern B)
        
        # We want lexicographically smallest, so we check which pattern gives that
        
        # Group nodes by their color
        nodes_color0 = [node for node in component if color[node] == 0]
        nodes_color1 = [node for node in component if color[node] == 1]
        
        # Pattern A: color0->1, color1->2
        # Pattern B: color0->2, color1->1
        
        # Check which is lexicographically smaller
        use_pattern_A = True
        
        # Find the first position where they differ
        all_nodes = sorted(component)
        for node in all_nodes:
            val_A = 1 if color[node] == 0 else 2
            val_B = 2 if color[node] == 0 else 1
            if val_A < val_B:
                use_pattern_A = True
                break
            elif val_A > val_B:
                use_pattern_A = False
                break
        
        # Assign values based on chosen pattern
        if use_pattern_A:
            for node in nodes_color0:
                result[node] = 1
            for node in nodes_color1:
                result[node] = 2
        else:
            for node in nodes_color0:
                result[node] = 2
            for node in nodes_color1:
                result[node] = 1
    
    # Verify solution
    for u, v in edges:
        if result[u] * result[v] != 2:
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, result)))

solve()
```
