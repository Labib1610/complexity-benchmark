```python
import sys

# The condition x_u ^ x_v = 3 means:
# If x_u = 1, then 1 ^ x_v = 3 => x_v = 2
# If x_u = 2, then 2 ^ x_v = 3 => x_v = 1
# This is equivalent to saying x_u != x_v.
# The problem is to 2-color the graph such that adjacent nodes have different colors (1 or 2).
# We want to minimize sum_{i=1}^n x_i * 3^{n-i}.
# Since 3^{n-i} is positive and decreasing with i, we want to make x_i as small as possible
# for smaller i. Specifically, for each connected component, we want to assign the 
# smaller value (1) to the smallest index i in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To minimize the sum, we want the smallest index in this component to be 1.
            # Since we iterate i from 1 to n, the first node we encounter in a 
            # component is always the smallest index in that component.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # All nodes in this component are colored.
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
```
