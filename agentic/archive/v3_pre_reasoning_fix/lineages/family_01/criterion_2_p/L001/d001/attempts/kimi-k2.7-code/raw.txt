import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    color = [-1]*(n+1)
    ans = [0]*(n+1) # 1-based
    for i in range(1, n+1):
        if color[i] != -1:
            continue
        # BFS component
        queue = deque([i])
        color[i] = 0
        comp = [i]
        min_v = i
        while queue:
            u = queue.popleft()
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = color[u] ^ 1
                    queue.append(w)
                    comp.append(w)
                    if w < min_v:
                        min_v = w
                elif color[w] == color[u]:
                    sys.stdout.write("NO")
                    return
        flip = 1 if color[min_v] == 1 else 0
        for v in comp:
            ans[v] = (color[v] ^ flip) + 1
    out = ["YES"]
    out.append(" ".join(str(ans[i]) for i in range(1, n+1)))
    sys.stdout.write("\n".join(out))
