import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [0]*n
    adj_list = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1; v -= 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        adj_list[u].append(v)
        adj_list[v].append(u)
    pcnt = [0,1,1,2,1,2,2,3]  # popcount for 0..7
    all_mask = (1 << n) - 1

    def assign(avail, v, c):
        bit = 1 << c
        if not (avail[v] & bit):
            return False
        avail[v] = bit
        stack = [v]
        while stack:
            u = stack.pop()
            cu = avail[u]
            for w in adj_list[u]:
                if avail[w] & cu:
                    avail[w] &= ~cu
                    if avail[w] == 0:
                        return False
                    if avail[w] & (avail[w] - 1) == 0:
                        stack.append(w)
        return True

    def propagate_all(avail):
        for v in range(n):
            if avail[v] == 0:
                return False
            if avail[v] & (avail[v] - 1) == 0:
                if not assign(avail, v, (avail[v].bit_length() - 1)):
                    return False
        return True

    def get_components(mask):
        comps = []
        rem = mask
        while rem:
            start = (rem & -rem).bit_length() - 1
            visited = 0
            stack = [start]
            while stack:
                v = stack.pop()
                if visited & (1 << v):
                    continue
                visited |= 1 << v
                nb = adj[v] & mask & ~visited
                while nb:
                    lsb = nb & -nb
                    w = lsb.bit_length() - 1
                    stack.append(w)
                    nb -= lsb
            comps.append(visited)
            rem &= ~visited
        return comps

    def unassigned_in(comp, avail):
        mask = 0
        m = comp
        while m:
            lsb = m & -m
            v = lsb.bit_length() - 1
            if avail[v] & (avail[v] - 1):
                mask |= lsb
            m -= lsb
        return mask

    def all_domain7(comp, avail):
        m = comp
        while m:
            lsb = m & -m
            if avail[lsb.bit_length() - 1] != 7:
                return False
            m -= lsb
        return True

        best_pc = 4
        best_deg = -1
        m = comp
        while m:
            lsb = m & -m
            v = lsb.bit_length() - 1
            pc = pcnt[avail[v]]
            deg = adj[v].bit_count()
            if pc < best_pc or (pc == best_pc and deg > best_deg):
                best = v
                best_pc = pc
                best_deg = deg
            m -= lsb
        return best

                return None
            m -= lsb

        if un == 0:

        comps = get_components(un)

                res = solve_comp(sub, avail)
                if res is None:
                    return None
                avail = res
            return avail

            new_avail = avail.copy()
            if not assign(new_avail, v, 0):
                return None
            return solve_comp(sub, new_avail)

            c = lsb.bit_length() - 1
            colors -= lsb
            new_avail = avail.copy()
            if assign(new_avail, v, c):

                if res is not None:

            return False
        un = 0
        for i in range(n):
            if avail[i] & (avail[i] - 1):

        if un == 0:
            return True
        comps = get_components(un)
        for comp in comps:
            res = solve_comp(comp, avail)
            if res is None:
                return False
            avail = res
        return True

    if not can_color([7] * n):
        print("NO")
        return

    ans = [0] * n
    for i in range(n):
        for c in range(3):
            avail = [7] * n
            ok = True
            for j in range(i):
                avail[j] = 1 << ans[j]
            avail[i] = 1 << c
            if can_color(avail):
                ans[i] = c
                break
        else:
            # Should not happen
            print("NO")
            return
