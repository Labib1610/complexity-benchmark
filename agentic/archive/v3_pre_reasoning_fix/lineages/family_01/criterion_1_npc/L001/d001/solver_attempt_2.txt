import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def can_assign(u, hopper):
        for neighbor in adj[u]:
            if assignment[neighbor] == hopper:
                return False
        return True
    
    # We need to find the lexicographically smallest sequence.
    # Since we must assign ingredients 1 to n in order, we use backtracking.
    # However, the problem implies a greedy choice at each step:
    # "The system attempts to place ingredient 1 in the lowest-numbered hopper it can, 
    # then does the same for ingredient 2, and so on..."
    # This suggests a greedy approach with backtracking if a dead end is reached.
    
    def backtrack(u):
        if u > n:
            return True
        
        for h in range(1, 4):
            if can_assign(u, h):
                assignment[u] = h
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
