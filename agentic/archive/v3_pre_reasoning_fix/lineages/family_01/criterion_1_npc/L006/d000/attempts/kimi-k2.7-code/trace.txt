def solve():
    import sys
    sys.setrecursionlimit(1000000)
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    self_loop = False
    for _ in range(m):
        u = int(next(it))-1
        v = int(next(it))-1
        if u == v:
            self_loop = True
        adj[u].append(v)
        adj[v].append(u)
    if self_loop:
        print("NO")
        return

    # Solver functions
    FULL = 7
    def apply(D, active, deg, stack):
        # propagate singletons
        q = [v for v in range(n) if active[v] and D[v].bit_count() == 1]
        while q:
            v = q.pop()
            if not active[v]:
                continue
            c = D[v]
            # deactivate v
            stack.append((v, D[v], active[v], None))
            active[v] = False
            for u in adj[v]:
                if active[u]:
                    stack.append((u, None, None, deg[u]))
                    deg[u] -= 1
                    if D[u] & c:
                        stack.append((u, D[u], None, None))
                        new = D[u] & ~c
                        if new == 0:
                            return False
                        D[u] = new
                        if new.bit_count() == 1:
                            q.append(u)
        # peel
        pq = [v for v in range(n) if active[v] and D[v].bit_count() > deg[v]]
        while pq:
            v = pq.pop()
            if not active[v]:
                continue
            stack.append((v, D[v], active[v], None))
            active[v] = False
            for u in adj[v]:
                if active[u]:
                    stack.append((u, None, None, deg[u]))
                    deg[u] -= 1
                    if D[u].bit_count() > deg[u]:
                        pq.append(u)
        return True

    def undo(stack, target):
        while len(stack) > target:
            v, oldD, oldA, oldDdeg = stack.pop()
            if oldD is not None:
                D[v] = oldD
            if oldA is not None:
                active[v] = oldA
            if oldDdeg is not None:
                deg[v] = oldDdeg

    def search(D, active, deg):
        start = len(stack)
        if not apply(D, active, deg, stack):
            undo(stack, start)
            return False
        # find branch vertex
        best = -1
        best_size = 4
        best_deg = -1
        for v in range(n):
            if active[v]:
                sz = D[v].bit_count()
                if sz < best_size or (sz == best_size and deg[v] > best_deg):
                    best = v
                    best_size = sz
                    best_deg = deg[v]
        if best == -1:
            return True
        v = best
        # order colors by least constrained
        bits = []
        mask = D[v]
        b = 1
        while mask:
            if mask & 1:
                cnt = 0
                for u in adj[v]:
                    if active[u] and D[u] & b:
                        cnt += 1
                bits.append((cnt, b))
            b <<= 1
            mask >>= 1
        bits.sort()
        for _, b in bits:
            start2 = len(stack)
            stack.append((v, D[v], None, None))
            D[v] = b
            if search(D, active, deg):
                return True
            undo(stack, start2)
        undo(stack, start)
        return False

    # initial feasibility
    D0 = [FULL]*n
    active0 = [True]*n
    deg0 = [len(adj[i]) for i in range(n)]
    if not search(D0, active0, deg0):
        print("NO")
        return

    assigned = [0]*n
    for i in range(n):
        chosen = 0
        for tier in (1,2,3):
            bit = 1 << (tier-1)
            # quick conflict with already assigned neighbors
            conflict = False
            for u in adj[i]:
                if assigned[u] == tier:
                    conflict = True
                    break
            if conflict:
                continue
            D = [FULL]*n
            for k in range(i):
                if assigned[k]:
                    D[k] = 1 << (assigned[k]-1)
            D[i] = bit
            active = [True]*n
            deg = [len(adj[i]) for _ in range(n)]  # careful: len(adj[i]) not adj[i] list? Use deg list.
            # Actually deg for each vertex: len(adj[v])
            deg = [len(adj[v]) for v in range(n)]
            if search(D, active, deg):
                chosen = tier
                break
        if chosen == 0:
            print("NO")
            return
        assigned[i] = chosen

    out = sys.stdout
    out.write("YES\n")
    out.write(' '.join(map(str, assigned)))
