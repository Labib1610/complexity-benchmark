from itertools import product

def solve():
    n, m = map(int, input().split())
    
    # Create adjacency list to store conflicts
    conflicts = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, input().split())
        conflicts[u].append(v)
        conflicts[v].append(u)
    
    # Try all possible tier assignments (1, 2, 3)
    # Generate in lexicographically smallest order
    for assignment in product([1, 2, 3], repeat=n):
        # Check if this assignment is valid
        valid = True
        for i in range(1, n + 1):
            tier_i = assignment[i - 1]
            # Check all conflicting microservices
            for j in conflicts[i]:
                if j <= n and assignment[j - 1] == tier_i:
                    valid = False
                    break
            if not valid:
                break
        
        if valid:
            print("YES")
            print(' '.join(map(str, assignment)))
            return
    
    print("NO")

solve()
