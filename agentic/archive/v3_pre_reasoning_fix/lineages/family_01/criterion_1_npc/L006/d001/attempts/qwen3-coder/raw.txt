def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list representation of constraints
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Try all possible assignments in lexicographical order
    # We can use backtracking with early pruning
    assignment = [0] * (n + 1)  # assignment[i] is the drive for shard i
    
    def is_valid(shard, drive):
        # Check if assigning 'drive' to 'shard' conflicts with neighbors
        for neighbor in adj[shard]:
            if assignment[neighbor] == drive:
                return False
        return True
    
    def backtrack(shard):
        if shard > n:
            return True
            
        for drive in [1, 2, 3]:
            if is_valid(shard, drive):
                assignment[shard] = drive
                if backtrack(shard + 1):
                    return True
                assignment[shard] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
