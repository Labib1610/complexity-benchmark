def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all possible assignments in lexicographical order
    # We'll use backtracking with early termination
    heights = [0] * (n + 1)  # 1-indexed
    
    def is_valid(node):
        """Check if current assignment for node is valid with its neighbors"""
        for neighbor in adj[node]:
            if heights[neighbor] != 0 and heights[neighbor] == heights[node]:
                return False
        return True
    
    def backtrack(index):
        """Try to assign heights starting from index"""
        if index > n:
            return True
            
        for h in [1, 2, 3]:
            heights[index] = h
            if is_valid(index):
                if backtrack(index + 1):
                    return True
            heights[index] = 0
            
        return False
    
    # Check if the graph is 3-colorable
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, heights[1:])))
    else:
        print("NO")

solve()
