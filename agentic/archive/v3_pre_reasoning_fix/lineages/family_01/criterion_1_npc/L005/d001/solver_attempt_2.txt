```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    heights = [0] * (n + 1)
    
    def is_valid(u, h):
        for v in adj[u]:
            if heights[v] != 0 and heights[v] == h:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for h in range(1, 4):
            if is_valid(u, h):
                heights[u] = h
                if backtrack(u + 1):
                    return True
                heights[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(heights[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
