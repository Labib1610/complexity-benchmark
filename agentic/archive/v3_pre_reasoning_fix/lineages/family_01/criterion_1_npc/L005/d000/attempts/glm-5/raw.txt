def solve():
    input = sys.stdin.read().split()
    idx = 0
    n = int(input[idx]); idx += 1
    m = int(input[idx]); idx += 1
    
    adj = [[] for _ in range(n)]
    for _ in range(m):

        v = int(input[idx]) - 1; idx += 1
        adj[u].append(v)
        adj[v].append(u)

        for color in [1, 2, 3]:
            if is_valid(node, color):
                colors[node] = color
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False
    
    if backtrack(0):
