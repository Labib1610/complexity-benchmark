```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(dish):
        if dish > n:
            return True
        
        # Try spice blends 1, 2, 3 in order for lexicographical smallest
        for spice in range(1, 4):
            can_use = True
            for neighbor in adj[dish]:
                if assignment[neighbor] == spice:
                    can_use = False
                    break
            
            if can_use:
                assignment[dish] = spice
                if backtrack(dish + 1):
                    return True
                assignment[dish] = 0
        
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
