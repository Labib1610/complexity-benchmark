    forbid = [0,0,0,0]  # index 1..3

        if c:
            bit = 1<<v
            colored |= bit
            forbid[c] |= adj[v]
    # check dead

        return False
    uncolored = n - bit_count(colored)
    def dfs():
        nonlocal colored, uncolored
        if uncolored == 0:
            return True
        # choose MRV vertex
        best_v = -1; best_avail = 4; best_deg = -1
        mask = ~colored & ALL
        # iterate bits
        mm = mask
        while mm:
            v = (mm & -mm).bit_length() - 1  # or use bit_length

            avail = 3 - ((forbid[1]>>v & 1) + (forbid[2]>>v & 1) + (forbid[3]>>v & 1))
            if avail == 0:

            deg = len(adj_list[v])
            if avail < best_avail or (avail == best_avail and deg > best_deg):

                best_avail = avail
                best_deg = deg
                if best_avail == 1:

        v = best_v
        # try colors
        for c in (1,2,3):

            uncolored -= 1

                    return True
            # backtrack

            uncolored += 1
        return False
