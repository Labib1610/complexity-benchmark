    colors = []
    for i in range(n):
        d=dom[i]
        if d & (d-1): # two bits
            # get two bits
            b1 = d & -d
            b2 = d & (d-1) # actually remaining
            colors.append((b1, b2))
        else:
            colors.append((d, d))
    N = n
    g = [[] for _ in range(2*N)]
    def add_imp(a, va, b, vb): # literal (a, va) implies (b, vb)
        g[a*2 + (0 if va==0 else 1)].append(b*2 + (0 if vb==0 else 1))
    for u,v in edges:
        cu0, cu1 = colors[u]
        cv0, cv1 = colors[v]
        for xu, cu in enumerate((cu0, cu1)):
            for xv, cv in enumerate((cv0, cv1)):
                if cu == cv:
                    # forbid xu,xv
                    # clause (x_u != xu) OR (x_v != xv)
                    # implications xu -> x_v != xv, xv -> x_u != xu
                    add_imp(u, xu, v, 1-xv)
                    add_imp(v, xv, u, 1-xu)
    # SCC
    # Kosaraju
    visited=[False]*(2*N)
    order=[]
    def dfs1(v):
        visited[v]=True
        for to in g[v]:
            if not visited[to]: dfs1(to)
        order.append(v)
    for i in range(2*N):
        if not visited[i]: dfs1(i)
    rg=[[] for _ in range(2*N)]
    for v in range(2*N):
        for to in g[v]:
            rg[to].append(v)
    comp=[-1]*(2*N)
    def dfs2(v, c):
        comp[v]=c
        for to in rg[v]:
            if comp[to]==-1: dfs2(to, c)
    cid=0
    for v in reversed(order):
        if comp[v]==-1:
            dfs2(v, cid); cid+=1
    for i in range(N):
        if comp[2*i] == comp[2*i+1]: return False
    return True
