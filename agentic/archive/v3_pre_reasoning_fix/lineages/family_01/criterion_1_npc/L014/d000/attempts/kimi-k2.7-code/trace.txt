def propagate(domains, queue):
    # queue contains vertices whose domain became singleton and need to propagate
    while queue:
        v = queue.pop()
        mask = domains[v]
        if mask == 0: return False
        if mask & (mask-1) != 0: continue # not singleton (shouldn't happen)
        c = (mask.bit_length() - 1)
        # remove c from neighbors
        nb = adj[v]
        while nb:
            lsb = nb & -nb
            u = (lsb.bit_length() - 1)
            nb ^= lsb
            if domains[u] & (1<<c):
                newmask = domains[u] ^ (1<<c)
                if newmask == 0:
                    return False
                domains[u] = newmask
                if newmask & (newmask-1) == 0:
                    queue.append(u)
    return True

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    data = sys.stdin.read().strip().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [0]*n
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        if 0 <= u < n and 0 <= v < n:
            adj[u] |= 1 << v
            adj[v] |= 1 << u
    # popcount table for 0..7
    pc = [0]*8
    for i in range(8): pc[i] = i.bit_count()
    memo = {}
    FULL = (1<<3)-1 # 7

    def propagate(d, stack):
        while stack:
            v = stack.pop()
            mask = d[v]
            if mask == 0 or mask & (mask - 1):
                continue
            cbit = mask
            nb = adj[v]
            nbm = nb
            while nbm:
                lsb = nbm & -nbm
                u = lsb.bit_length() - 1
                nbm ^= lsb
                if d[u] & cbit:
                    d[u] &= ~cbit
                    if d[u] == 0:
                        return False
                    if d[u] & (d[u] - 1) == 0:
                        stack.append(u)
        return True

    def dfs(d):
        # d is mutable list; will be reduced
        stack = [i for i in range(n) if d[i] and (d[i] & (d[i] - 1) == 0)]
        if not propagate(d, stack):
            return False
        # kernelization
        rem_mask = 0
        for i in range(n):
            if d[i]:
                rem_mask |= 1 << i
        changed = True
        while changed:
            changed = False
            m = rem_mask
            while m:
                lsb = m & -m
                v = lsb.bit_length() - 1
                m ^= lsb
                if d[v] & (d[v] - 1):  # non-singleton
                    deg = (adj[v] & rem_mask).bit_count()
                    if pc[d[v]] > deg:
                        d[v] = 0
                        rem_mask ^= lsb
                        changed = True
        key = bytes(d)
        if key in memo:
            return memo[key]
        # choose branch vertex
        v = -1
        best = 4
        maxdeg = -1
        m = rem_mask
        while m:
            lsb = m & -m
            u = lsb.bit_length() - 1
            m ^= lsb
            mask = d[u]
            if mask & (mask - 1):
                p = pc[mask]
                deg = (adj[u] & rem_mask).bit_count()
                if p < best or (p == best and deg > maxdeg):
                    best = p
                    maxdeg = deg
                    v = u
        if v == -1:
            memo[key] = True
            return True
        mask = d[v]
        # try colors in order of maybe least constraining? For speed, try colors that appear in fewest neighbor domains.
        # Precompute for each color in domain a score = number of neighbors (in rem_mask) that have that color in domain.
        # Choose color with smallest score.
        colors = []
        for c in (0,1,2):
            if mask & (1<<c):
                # count neighbors in rem_mask with color c available
                cnt = 0
                nbm = adj[v] & rem_mask
                while nbm:
                    lsb = nbm & -nbm
                    u = lsb.bit_length() - 1
                    nbm ^= lsb
                    if d[u] & (1<<c):
                        cnt += 1
                colors.append((cnt, c))
        colors.sort()
        for _, c in colors:
            d2 = d[:]  # shallow copy of list of ints
            d2[v] = 1 << c
            if dfs(d2):
                memo[key] = True
                return True
        memo[key] = False
        return False

    def feasible(d):
        return dfs(d[:])

    res = [-1]*n
    for i in range(n):
        # compute colors used by already colored neighbors
        forbidden = 0
        nb = adj[i]
        nbm = nb
        while nbm:
            lsb = nbm & -nbm
            u = lsb.bit_length() - 1
            nbm ^= lsb
            if res[u] != -1:
                forbidden |= 1 << res[u]
        chosen = -1
        for c in (0,1,2):
            if forbidden & (1<<c):
                continue
            d = [FULL]*n
            for j in range(i):
                if res[j] != -1:
                    d[j] = 1 << res[j]
            d[i] = 1 << c
            if feasible(d):
                chosen = c
                break
        if chosen == -1:
            sys.stdout.write("NO")
            return
        res[i] = chosen

    out = ["YES", " ".join(str(c+1) for c in res)]
    sys.stdout.write("\n".join(out))
