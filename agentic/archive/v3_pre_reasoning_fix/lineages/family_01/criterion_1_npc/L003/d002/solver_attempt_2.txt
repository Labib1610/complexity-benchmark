import sys

def solve():
    # n is capped at 40 (ten years * 4 quarters)
    # tiers are 1 to 3 (months in a fiscal quarter)
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)

    # We need to assign tiers 1, 2, or 3 to each portfolio 1..n
    # such that if (u, v) is an edge, tier[u] != tier[v].
    # We want the lexicographically smallest sequence (tier[1], tier[2], ..., tier[n]).
    
    assignment = [0] * (n + 1)

    def backtrack(u):
        if u > n:
            return True
        
        # Try tiers 1, 2, 3 in order to ensure lexicographical smallest
        for t in range(1, 4):
            possible = True
            for neighbor in adj[u]:
                if assignment[neighbor] == t:
                    possible = False
                    break
            
            if possible:
                assignment[u] = t
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
