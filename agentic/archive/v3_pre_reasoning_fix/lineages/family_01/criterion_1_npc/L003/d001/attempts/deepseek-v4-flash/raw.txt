from functools import lru_cache

@lru_cache(maxsize=None)
def possible(m1, m2, m3, domain):
    if domain == 0:
        return True
    colored = m1 | m2 | m3
    # reduction
    active = domain
    # We'll maintain deg for active vertices. Since n small, use list length n.
    deg = [0]*n
    # initialize deg for active vertices: popcount(adj[v] & (colored | active))
    # But active will change; we can compute initial using current active.
    # Need iterate over active bits.
    a = active
    while a:
        lsb = a & -a
        v = lsb.bit_length() - 1
        deg[v] = (adj[v] & (colored | active)).bit_count()
        a ^= lsb
    # queue of active vertices with deg <=2
    queue = []
    a = active
    while a:
        lsb = a & -a
        v = lsb.bit_length() - 1
        if deg[v] <= 2:
            queue.append(v)
        a ^= lsb
    while queue:
        v = queue.pop()
        if not (active >> v) & 1:
            continue
        if deg[v] > 2:
            continue
        # remove v
        active ^= (1 << v)
        # update neighbors
        neigh = adj[v] & active
        while neigh:
            lsb = neigh & -neigh
            u = lsb.bit_length() - 1
            deg[u] -= 1
            if deg[u] <= 2:
                queue.append(u)
            neigh ^= lsb
    if active == 0:
        return True
    # optional component decomposition
    comps = []
    unvisited = active
    while unvisited:
        start_lsb = unvisited & -unvisited
        start = start_lsb.bit_length() - 1
        comp = 0
        stack = [start]
        unvisited ^= start_lsb
        while stack:
            v = stack.pop()
            comp |= (1 << v)
            neigh = adj[v] & active & unvisited
            while neigh:
                lsb = neigh & -neigh
                u = lsb.bit_length() - 1
                unvisited ^= lsb
                stack.append(u)
                neigh ^= lsb
        comps.append(comp)
    if len(comps) > 1:
        for comp in comps:
            if not possible(m1, m2, m3, comp):
                return False
        return True
    # single component active
    # choose vertex with min available colors, tie max degree
    best_v = -1
    best_cnt = 4
    best_deg = -1
    best_avail = 0
    a = active
    while a:
        lsb = a & -a
        v = lsb.bit_length() - 1
        avail = 0
        if not (adj[v] & m1):
            avail |= 1
        if not (adj[v] & m2):
            avail |= 2
        if not (adj[v] & m3):
            avail |= 4
        if avail == 0:
            return False
        cnt = avail.bit_count()
        # deg[v] is current degree? We updated deg for active vertices. It should be current.
        d = deg[v]
        if cnt < best_cnt or (cnt == best_cnt and d > best_deg):
            best_cnt = cnt
            best_v = v
            best_deg = d
            best_avail = avail
        a ^= lsb
    if best_cnt == 1:
        # force
        c = (best_avail.bit_length() - 1)  # 0,1,2
        bit = 1 << best_v
        if c == 0:
            return possible(m1 | bit, m2, m3, active & ~bit)
        elif c == 1:
            return possible(m1, m2 | bit, m3, active & ~bit)
        else:
            return possible(m1, m2, m3 | bit, active & ~bit)
    # branch
    # color symmetry if colored == 0: only try color1
    if colored == 0:
        colors = [0]  # only color 1
    else:
        colors = []
        if best_avail & 1: colors.append(0)
        if best_avail & 2: colors.append(1)
