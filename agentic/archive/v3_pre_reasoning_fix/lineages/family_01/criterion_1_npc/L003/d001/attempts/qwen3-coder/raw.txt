def solve():
    import sys
    from itertools import product
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Try all possible assignments of engines to portfolios
    # Since n <= 40 and we have 3 engines, trying all 3^n combinations is too expensive
    # However, we can try all assignments in lexicographical order using backtracking
    
    # We'll use a greedy backtracking approach that tries engine 1, 2, 3 in order
    assignment = [0] * (n + 1)
    
    def is_valid(portfolio, engine):
        for neighbor in adj[portfolio]:
            if assignment[neighbor] == engine:
                return False
        return True
    
    def backtrack(portfolio):
        if portfolio > n:
            return True
            
        for engine in [1, 2, 3]:
            if is_valid(portfolio, engine):
                assignment[portfolio] = engine
                if backtrack(portfolio + 1):
                    return True
                assignment[portfolio] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
