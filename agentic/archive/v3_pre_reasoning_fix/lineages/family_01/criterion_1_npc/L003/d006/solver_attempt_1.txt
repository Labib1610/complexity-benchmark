import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Precompute fingerprints
    # F[k][j] = (k * (j+1)) % 4 for j in 0,1,2
    F = {}
    for k in [1, 2, 3]:
        F[k] = [(k * (j + 1)) % 4 for j in range(3)]

    # Precompute compatibility
    # Condition: sum |F[du][j] - F[dv][j]| == 4
    compat = [[False for _ in range(4)] for _ in range(4)]
    for d1 in [1, 2, 3]:
        for d2 in [1, 2, 3]:
            s = sum(abs(F[d1][j] - F[d2][j]) for j in range(3))
            if s == 4:
                compat[d1][d2] = True

    assignment = [0] * (n + 1)

    def backtrack(u):
        if u > n:
            return True
        
        for d in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] != 0:
                    if not compat[d][assignment[v]]:
                        possible = False
                        break
            
            if possible:
                assignment[u] = d
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
