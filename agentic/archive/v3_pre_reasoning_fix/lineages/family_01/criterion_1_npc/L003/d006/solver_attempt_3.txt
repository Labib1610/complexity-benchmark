```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: d_u * x + d_v * y = 0 and d_v * x + d_u * y = 0
    # This is a linear system with matrix [[d_u, d_v], [d_v, d_u]].
    # It has a unique solution (x=0, y=0) if and only if the determinant is non-zero.
    # det = d_u^2 - d_v^2 != 0 => d_u != d_v and d_u != -d_v.
    # Since d_i in {1, 2, 3}, d_u != -d_v is always true.
    # Thus, the condition simplifies to d_u != d_v.
    # This is a standard graph coloring problem with 3 colors.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
