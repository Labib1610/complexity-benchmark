def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Precompute fingerprints for tiers 1, 2, 3
    fingerprints = {}
    for k in [1, 2, 3]:
        fingerprints[k] = tuple((k * j) % 4 for j in range(1, 4))
    
    # Precompute which tier pairs satisfy the separation rule
    valid_pairs = set()
    tiers = [1, 2, 3]
    for i in tiers:
        for j in tiers:
            diff = sum(abs(fingerprints[i][jj] - fingerprints[j][jj]) for jj in range(3))
            if diff == 4:
                valid_pairs.add((i, j))
                valid_pairs.add((j, i))
    
    # Read constraints
    edges = []
    adj = [[] for _ in range(n + 1)]
    for i in range(m):
        idx = 2 + 2 * i
        u = int(data[idx])
        v = int(data[idx + 1])
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all 3^n assignments with backtracking for early pruning
    # But since n<=40, we need a smarter approach
    
    # This is a graph coloring problem with specific color constraints
    # We can model this as a 2-SAT or use constraint satisfaction
    # However, given small constraints, let's try a smart backtracking
    
    # Since we want lexicographically smallest, we'll assign values from 1 to 3
    # in order for each node, and backtrack when constraints are violated
    
    # Let's build the constraint graph: for each edge (u,v), (d_u,d_v) must be in valid_pairs
    
    assignment = [0] * (n + 1)
    
    def is_valid(node, tier):
        for neighbor in adj[node]:
            if assignment[neighbor] != 0:
                if (tier, assignment[neighbor]) not in valid_pairs:
                    return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for tier in [1, 2, 3]:
            if is_valid(node, tier):
                assignment[node] = tier
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
