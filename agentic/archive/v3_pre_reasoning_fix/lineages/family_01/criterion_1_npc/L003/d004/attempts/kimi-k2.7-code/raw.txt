import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [0]*n
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        adj[u] |= 1<<v
        adj[v] |= 1<<u
    color = [0]*n
    avail = [7]*n
    found = False
    ans = [1]*n

    # map single bit to color
    bit_to_col = {1:1,2:2,4:3}

    out = sys.stdout

    def dfs():
        nonlocal found
        if found:
            return
        # find first uncolored
        i = 0
        while i < n and color[i]:
            i += 1
        if i == n:
            found = True
            ans[:] = color[:]
            return
        # try colors in lex order
        mask = avail[i]
        # colors 1..3 correspond bits 1,2,4
        for col in (1,2,3):
            bit = 1 << (col-1)
            if mask & bit:
                changes = []
                if propagate_assign(i, col, changes):
                    dfs()
                    if found:
                        return
                undo(changes)
        return

    def propagate_assign(v, col, changes):
        # iterative stack of assignments
        stack = [(v, col)]
        while stack:
            x, c = stack.pop()
            if color[x]:
                if color[x] != c:
                    return False
                continue
            # check availability
            if not (avail[x] & (1 << (c-1))):
                return False
            # assign
            changes.append((x, avail[x], 0)) # store old avail; color was 0
            color[x] = c
            avail[x] = 0
            b = 1 << (c-1)
            # iterate neighbors
            nb = adj[x]
            while nb:
                lsb = nb & -nb
                u = (lsb.bit_length() - 1)
                nb ^= lsb
                if color[u] == 0:
                    old = avail[u]
                    new = old & ~b
                    if new == 0:
                        return False
                    if new != old:
                        changes.append((u, old, 0))
                        avail[u] = new
                        # if only one color, force
                        if new & (new - 1) == 0:
                            stack.append((u, bit_to_col[new]))
        return True

    def undo(changes):
        for v, old_avail, _ in reversed(changes):
            avail[v] = old_avail
            color[v] = 0

    dfs()
    if found:
        out.write("YES\n")
        out.write(' '.join(map(str, ans)))
    else:
        out.write("NO")

if __name__ == '__main__':
    solve()

    n = int(next(it))
