```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Condition: s_u * s_v != k^2
    # Possible stages: 1, 2, 3
    # 1*1 = 1 (square) -> s_u, s_v cannot both be 1
    # 2*2 = 4 (square) -> s_u, s_v cannot both be 2
    # 3*3 = 9 (square) -> s_u, s_v cannot both be 3
    # 1*2 = 2 (not square)
    # 1*3 = 3 (not square)
    # 2*3 = 6 (not square)
    # So, for any edge (u, v), s_u != s_v.
    # This is equivalent to 3-coloring the graph.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
```
