import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(task, stage):
        for neighbor in adj[task]:
            if assignment[neighbor] != 0 and abs(assignment[neighbor] - stage) < 1:
                # The condition is |su - sv| >= 1.
                # This means su != sv.
                # Wait, the problem says |su - sv| >= 1.
                # Since stages are {1, 2, 3}, |su - sv| >= 1 is equivalent to su != sv.
                # Let's re-read: "For any two tasks u and v that share memory registers, 
                # their assigned execution stages su and sv must satisfy the condition |su - sv| >= 1."
                # If su = 1 and sv = 1, |1 - 1| = 0, which is < 1.
                # So yes, it is simply su != sv.
                if assignment[neighbor] == stage:
                    return False
        return True

    def backtrack(task):
        if task > n:
            return True
        
        for stage in [1, 2, 3]:
            if is_valid(task, stage):
                assignment[task] = stage
                if backtrack(task + 1):
                    return True
                assignment[task] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
