    assignment = [-1] * n

            return True

def propagate(task, stage, assignment, valid_stages):
    """Propagate constraints after assigning stage to task."""
    queue = [(task, stage)]
    propagated = []
    
    while queue:
        t, s = queue.pop(0)
        for neighbor in adj[t]:
            if assignment[neighbor] == -1:
                if valid_stages[neighbor] & (1 << (s - 1)):
                    valid_stages[neighbor] &= ~(1 << (s - 1))
                    propagated.append((neighbor, s))
                    if valid_stages[neighbor] == 0:
                        # Conflict, restore and return False
                        for n, st in propagated:
                            valid_stages[n] |= (1 << (st - 1))
                        return False, propagated
                    # If only one valid stage left, assign it
                    if valid_stages[neighbor] == 1 << 0:
                        assignment[neighbor] = 1
                        queue.append((neighbor, 1))
                    elif valid_stages[neighbor] == 1 << 1:
                        assignment[neighbor] = 2
                        queue.append((neighbor, 2))
                    elif valid_stages[neighbor] == 1 << 2:
                        assignment[neighbor] = 3
                        queue.append((neighbor, 3))
    
    return True, propagated
