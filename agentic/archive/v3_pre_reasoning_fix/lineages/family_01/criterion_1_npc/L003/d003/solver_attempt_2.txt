import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(task, priority):
        for neighbor in adj[task]:
            if assignment[neighbor] == priority:
                return False
        return True

    def can_complete(task):
        if task > n:
            return True
        
        for p in range(1, 4):
            if is_valid(task, p):
                assignment[task] = p
                if can_complete(task + 1):
                    return True
                assignment[task] = 0
        return False

    # To satisfy the lexicographical requirement, we greedily pick the smallest
    # priority for each task and check if a valid completion exists.
    
    def find_lexicographical(task):
        if task > n:
            return True
        
        for p in range(1, 4):
            if is_valid(task, p):
                assignment[task] = p
                # Check if this choice allows a valid completion for the rest
                if can_complete(task + 1):
                    if find_lexicographical(task + 1):
                        return True
                assignment[task] = 0
        return False

    if find_lexicographical(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
