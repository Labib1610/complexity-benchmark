```
    data = sys.stdin.buffer.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        if u == v:
            print("NO")
            return
        adj[u].append(v); adj[v].append(u)
    FULL = 7
    sys.setrecursionlimit(1000000)

    def propagate(domains):
        changes = []
        stack = []
        for v in range(n):
            d = domains[v]
            if d == 0:
                return False, changes
            if d.bit_count() == 1:
                stack.append(v)
        while stack:
            v = stack.pop()
            cmask = domains[v]
            for u in adj[v]:
                du = domains[u]
                if du == cmask:
                    return False, changes
                if du & cmask:
                    new = du ^ cmask
                    if new == 0:
                        return False, changes
                    changes.append((u, du))
                    domains[u] = new
                    if new.bit_count() == 1:
                        stack.append(u)
        return True, changes

    def is_colorable(domains):
        ok, changes = propagate(domains)
        if not ok:
            return False
        v = -1
        best = 4
        for i in range(n):
            d = domains[i]
            if d.bit_count() > 1:
                pc = d.bit_count()
                if pc < best:
                    best = pc
                    v = i
                    if best == 2:
                        break
        if v == -1:
            return True
        old = domains[v]
        d = old
        while d:
            color = d & -d
            domains[v] = color
            if is_colorable(domains):
                return True
            domains[v] = old
            d -= color
        for u, du in reversed(changes):
            domains[u] = du
        return False

    assign = [0]*n
    bits = [1,2,4]
    for i in range(n):
        chosen = 0
        for c in (1,2,3):
            cm = bits[c-1]
            conflict = False
            for u in adj[i]:
                if u < i and assign[u] == c:
                    conflict = True
                    break
            if conflict:
                continue
            assign[i] = c
            domains = [FULL]*n
            for j in range(i+1):
                if assign[j]:
                    domains[j] = bits[assign[j]-1]
            if is_colorable(domains):
                chosen = c
                break
            assign[i] = 0
        if chosen == 0:
            print("NO")
            return

    out = sys.stdout
    out.write("YES\n")
    out.write(' '.join(map(str, assign)))
```

```
def propagate(domains):
    changes = []
    stack = []
    for v in range(n):
        d = domains[v]
        if d == 0:
            return False, changes
        if d.bit_count() == 1:
            stack.append(v)
    while stack:
        v = stack.pop()
        cmask = domains[v]
        for u in adj[v]:
            du = domains[u]
            if du == cmask:
                return False, changes
            if du & cmask:
                new = du ^ cmask
                if new == 0:
                    return False, changes
                changes.append((u, du))
                domains[u] = new
                if new.bit_count() == 1:
                    stack.append(u)
    return True, changes

def solve(domains):
    ok, changes = propagate(domains)
    if not ok:
        return False
    # choose branch vertex
    v = -1
    best = 4
    best_deg = -1
    for i in range(n):
        d = domains[i]
        pc = d.bit_count()
        if pc > 1:
            deg = degree[i]
            if pc < best or (pc == best and deg > best_deg):
                best = pc
                v = i
                best_deg = deg
                if best == 2 and best_deg >= ...: # can't break because maybe same degree
                    pass
    if v == -1:
        return True
    old = domains[v]
    d = old
    while d:
        color = d & -d
        domains[v] = color
        if solve(domains):
            return True
        domains[v] = old
        d -= color
    for u, du in reversed(changes):
        domains[u] = du
    return False
```
