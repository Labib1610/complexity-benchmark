def is_colorable(fixed):
    n = len(fixed)
    avail = [7]*n
    stack = []
    for v,c in enumerate(fixed):
        if c != -1:
            mask = 1 << c
            if avail[v] & mask == 0:
                return False
            avail[v] = mask
            stack.append(v)
    # propagate precolored
    while stack:
        v = stack.pop()
        mask = avail[v]
        # mask is single bit
        c = (mask.bit_length() - 1)
        for u in adj_list[v]:
            if avail[u] & mask:

                if new == 0:
                    return False

                if new & (new - 1) == 0:
                    stack.append(u)

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    idx = 0
    n = data[idx]; idx+=1
    m = data[idx]; idx+=1
    adj = [[] for _ in range(n)]
    deg = [0]*n
    for _ in range(m):
        u = data[idx]-1; idx+=1
        v = data[idx]-1; idx+=1
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1; deg[v] += 1

    popcnt = [0,1,1,2,1,2,2,3]

    def is_colorable(fixed):
        avail = [7]*n
        stack = []
        for v,c in enumerate(fixed):
            if c != -1:
                mask = 1 << c
                if (avail[v] & mask) == 0:
                    return False
                avail[v] = mask
                stack.append(v)
        while stack:
            v = stack.pop()
            mask = avail[v]
            for u in adj[v]:
                if avail[u] & mask:
                    new = avail[u] & ~mask
                    if new == 0:
                        return False
                    avail[u] = new
                    if new & (new - 1) == 0:
                        stack.append(u)
        return dfs(avail)

    def dfs(avail):
        # choose uncolored vertex with smallest domain
        v = -1
        best = 4
        for i in range(n):
            cnt = popcnt[avail[i]]
            if cnt > 1:
                if cnt < best:
                    best = cnt
                    v = i
                elif cnt == best and v != -1 and deg[i] > deg[v]:
                    v = i
        if v == -1:
            return True
        mask = avail[v]
        # try colors 0,1,2
        for c in range(3):
            if mask & (1 << c):
                new_avail = avail[:]  # copy
                # assign v = c
                new_avail[v] = 1 << c
                stack = [v]
                ok = True
                while stack:
                    x = stack.pop()
                    xm = new_avail[x]
                    for u in adj[x]:
                        if new_avail[u] & xm:
                            new = new_avail[u] & ~xm
                            if new == 0:
                                ok = False
                                break
                            new_avail[u] = new
                            if new & (new - 1) == 0:
                                stack.append(u)
                    if not ok:
                        break
                if ok and dfs(new_avail):
                    return True
        return False

    fixed = [-1]*n
    if not is_colorable(fixed):
        sys.stdout.write("NO")
        return

    for i in range(n):
        for c in range(3):
            fixed[i] = c
            if is_colorable(fixed):
                break
        else:
            sys.stdout.write("NO")
            return

    out = "YES\n" + " ".join(str(c+1) for c in fixed)
    sys.stdout.write(out)

    import sys
    sys.setrecursionlimit(1 << 25)
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    idx = 0
    n = data[idx]; idx += 1
    m = data[idx]; idx += 1
    adj = [[] for _ in range(n)]

        v = data[idx] - 1; idx += 1
        if u == v:
            # self-loop: will make coloring impossible

            pass

        adj[v].append(u)

        while stack:

                    if new == 0:
                        return False
                    avail[u] = new

                        stack.append(u)
        return dfs(avail)

    def dfs(avail):
        v = -1
        best = 4
        for i in range(n):
            cnt = popcnt[avail[i]]
            if cnt > 1:
                if cnt < best:
                    best = cnt
                    v = i
                elif cnt == best and v != -1 and deg[i] > deg[v]:
                    v = i
        if v == -1:
            return True
        mask = avail[v]
        for c in range(3):
            if mask & (1 << c):
                new_avail = avail[:]
                new_avail[v] = 1 << c
                stack = [v]
                ok = True
                while stack:
                    x = stack.pop()
                    xm = new_avail[x]
                    for u in adj[x]:
                        if new_avail[u] & xm:
                            new = new_avail[u] & ~xm
                            if new == 0:
                                ok = False
                                break
                            new_avail[u] = new
                            if new & (new - 1) == 0:
                                stack.append(u)
                    if not ok:
                        break
                if ok and dfs(new_avail):
                    return True
        return False

    fixed = [-1] * n
    if not is_colorable(fixed):
        sys.stdout.write("NO\n")
        return

    for i in range(n):
        chosen = -1
        for c in range(3):
            # skip if conflicts with an already fixed neighbor
            conflict = False
            for u in adj[i]:
                if fixed[u] == c:
                    conflict = True
                    break

            fixed[i] = c
            if is_colorable(fixed):
                chosen = c
                break
        if chosen == -1:
            sys.stdout.write("NO\n")
            return

    sys.stdout.write("YES\n" + " ".join(str(c + 1) for c in fixed))

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    idx = 0
    n = data[idx]; idx += 1
    m = data[idx]; idx += 1
    adj = [[] for _ in range(n)]
    deg = [0] * n
    for _ in range(m):
        u = data[idx] - 1; idx += 1
        v = data[idx] - 1; idx += 1
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1

    if m == 0:
        sys.stdout.write("YES\n" + " ".join(["1"] * n))
        return

    for i in range(n):
        adj[i].sort(key=deg.__getitem__, reverse=True)

    popcnt = [0, 1, 1, 2, 1, 2, 2, 3]

    def dfs(avail):
        v = -1
        best = 4
        for i in range(n):
            cnt = popcnt[avail[i]]
            if cnt > 1:
                if cnt < best:
                    best = cnt
                    v = i
                elif cnt == best and v != -1 and deg[i] > deg[v]:
                    v = i
        if v == -1:
            return True
        mask = avail[v]
        for c in range(3):
            if mask & (1 << c):
                new_avail = avail[:]
                new_avail[v] = 1 << c
                stack = [v]
                ok = True
                while stack:

                    for u in adj[x]:
                        if new_avail[u] & xm:
                            new = new_avail[u] & ~xm
                            if new == 0:
                                ok = False
                                break
                            new_avail[u] = new
                            if new & (new - 1) == 0:
                                stack.append(u)
                    if not ok:
                        break
                if ok and dfs(new_avail):
                    return True
        return False

    def is_colorable(fixed):
        avail = [7] * n
        stack = []
        for v, c in enumerate(fixed):
            if c != -1:
                mask = 1 << c
                if (avail[v] & mask) == 0:
                    return False
                avail[v] = mask
                stack.append(v)

                    if new == 0:
                        return False
                    avail[u] = new

                break

            return
