import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Collision metric: 2^su + 2^sv must not be a power of 2.
    # Possible stages: 1, 2, 3.
    # Metrics:
    # 1,1: 2^1 + 2^1 = 4 (2^2) - INVALID
    # 1,2: 2^1 + 2^2 = 6 - VALID
    # 1,3: 2^1 + 2^3 = 10 - VALID
    # 2,2: 2^2 + 2^2 = 8 (2^3) - INVALID
    # 2,3: 2^2 + 2^3 = 12 - VALID
    # 3,3: 2^3 + 2^3 = 16 (2^4) - INVALID
    #
    # Constraints:
    # If u, v share:
    # su != sv (cannot be 1,1 or 2,2 or 3,3)
    # AND (su, sv) cannot be (1,1), (2,2), (3,3)
    # Wait, the rule is: 2^su + 2^sv != 2^k
    # 1,1 -> 4 (2^2) NO
    # 2,2 -> 8 (2^3) NO
    # 3,3 -> 16 (2^4) NO
    # 1,2 -> 6 OK
    # 1,3 -> 10 OK
    # 2,3 -> 12 OK
    # So the rule is simply: su != sv.
    # This is just a 3-coloring problem.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, stage):
        for v in adj[u]:
            if assignment[v] == stage:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for stage in [1, 2, 3]:
            if is_valid(u, stage):
                assignment[u] = stage
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
