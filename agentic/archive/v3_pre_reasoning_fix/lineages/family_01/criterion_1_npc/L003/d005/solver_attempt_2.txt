import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Resonance condition: (2 * du * dv) % (du + dv) != 0
    # Possible pairs (du, dv) where du, dv in {1, 2, 3}:
    # (1, 1): 2*1*1 / (1+1) = 2/2 = 1 (Divisible -> Invalid)
    # (1, 2): 2*1*2 / (1+2) = 4/3 (Not divisible -> Valid)
    # (1, 3): 2*1*3 / (1+3) = 6/4 (Not divisible -> Valid)
    # (2, 1): Valid
    # (2, 2): 2*2*2 / (2+2) = 8/4 = 2 (Divisible -> Invalid)
    # (2, 3): 2*2*3 / (2+3) = 12/5 (Not divisible -> Valid)
    # (3, 1): Valid
    # (3, 2): Valid
    # (3, 3): 2*3*3 / (3+3) = 18/6 = 3 (Divisible -> Invalid)
    
    # So, du != dv is the only condition.
    # This is equivalent to 3-coloring the graph.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
