class ThreeColorSolver:
    __slots__ = ('n','adj','color','avail')
    def __init__(self, n, adj):
        self.n = n
        self.adj = adj
        self.color = [0]*n
        self.avail = [7]*n
    def assign(self, v, c):
        changes = []
        q = [v]
        self.color[v] = c
        olda = self.avail[v]
        self.avail[v] = 1 << (c-1)
        changes.append((v, 0, olda))
        idx = 0
        while idx < len(q):
            x = q[idx]; idx += 1
            cx = self.color[x]
            mask = 1 << (cx-1)
            for y in self.adj[x]:
                if self.color[y] == cx:
                    self._undo(changes)
                    return None
                if self.color[y] == 0 and (self.avail[y] & mask):
                    olday = self.avail[y]
                    neway = olday & ~mask
                    self.avail[y] = neway
                    changes.append((y, 0, olday))
                    if neway == 0:
                        self._undo(changes)
                        return None
                    if neway & (neway - 1) == 0:
                        cy = neway.bit_length()
                        self.color[y] = cy
                        changes.append((y, 0, neway))
                        q.append(y)
        return changes
    def _undo(self, changes):
        for v, oldc, olda in reversed(changes):
            self.color[v] = oldc
            self.avail[v] = olda
    def dfs(self, i):
        n = self.n
        while i < n and self.color[i] != 0:
            i += 1
        if i == n:
            return True
        mask = self.avail[i]
        # if mask == 0, fail (should not happen)
        while mask:
            lsb = mask & -mask
            c = lsb.bit_length()
            mask ^= lsb
            changes = self.assign(i, c)
            if changes is not None:
                if self.dfs(i+1):
                    return True
                self._undo(changes)
        return False
