    import sys
    sys.setrecursionlimit(10000)
    data = sys.stdin.read().split()
    
    n = int(data[0])
    m = int(data[1])
    
    conflicts = [set() for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        idx += 2
        conflicts[u].add(v)
        conflicts[v].add(u)
    
    assignment = [-1] * n
    forbid_count = [[0, 0, 0] for _ in range(n)]
    forbidden = [0] * n
    
    def propagate(i):
        # If any unassigned worker has only one valid shift, assign it
        changed = True
        while changed:
            changed = False
            for j in range(i, n):
                if assignment[j] != -1:
                    continue
                valid_shifts = [s for s in range(3) if forbid_count[j][s] == 0]
                if len(valid_shifts) == 0:
                    return False  # Conflict
                if len(valid_shifts) == 1:
                    shift = valid_shifts[0]
                    assignment[j] = shift + 1
                    bit = 1 << shift
                    for k in conflicts[j]:
                        if k > j and assignment[k] == -1:
                            forbid_count[k][shift] += 1
                            if forbid_count[k][shift] == 1:
                                forbidden[k] |= bit
                    changed = True
        return True

        if i == n:
            return True
        
        # Skip already assigned workers
        while i < n and assignment[i] != -1:
            i += 1
        if i == n:
            return True
