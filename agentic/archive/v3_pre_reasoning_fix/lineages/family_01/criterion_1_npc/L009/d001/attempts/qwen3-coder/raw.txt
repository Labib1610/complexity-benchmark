def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Try all 3^n combinations in lexicographical order
    # We'll use backtracking with pruning
    assignment = [0] * (n + 1)
    
    def is_valid(task, mode):
        for neighbor in adj[task]:
            if assignment[neighbor] == mode:
                return False
        return True
    
    def backtrack(task):
        if task > n:
            return True
            
        for mode in [1, 2, 3]:
            if is_valid(task, mode):
                assignment[task] = mode
                if backtrack(task + 1):
                    return True
                assignment[task] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
