import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    avail = [7] * n  # 1: color1, 2: color2, 4: color3
    stack = []
    
    def undo(stack_len):
        while len(stack) > stack_len:
            typ, v, old_val = stack.pop()
            if typ == 'color':
                color[v] = old_val
            else:
                avail[v] = old_val
                
    def assign(v, c):
        # assign color c to vertex v, propagate
        # returns True if no contradiction, False otherwise
        old_color = color[v]
        color[v] = c
        stack.append(('color', v, old_color))
        bit = 1 << (c-1)
        for nb in adj[v]:
            if color[nb] == 0:
                old_avail = avail[nb]
                if old_avail & bit:
                    new_avail = old_avail & ~bit
                    avail[nb] = new_avail
                    stack.append(('avail', nb, old_avail))
                    if new_avail == 0:
                        return False
                    if new_avail in (1,2,4):
                        forced_c = 1 if new_avail == 1 else (2 if new_avail == 2 else 3)
                        if not assign(nb, forced_c):
                            return False
        return True
        
    def propagate_units():
        while True:
            unit_v = -1
            for i in range(n):
                if color[i] == 0 and avail[i] in (1,2,4):
                    unit_v = i
                    break
            if unit_v == -1:
                return True
            forced_c = 1 if avail[unit_v] == 1 else (2 if avail[unit_v] == 2 else 3)
            stack_len = len(stack)
            if not assign(unit_v, forced_c):
                undo(stack_len)
                return False
                
    def dfs():
        # unit propagation
        if not propagate_units():
            return False
        # check if all assigned
        if all(color[i] != 0 for i in range(n)):
            return True
        # find smallest unassigned
        v = -1
        for i in range(n):
            if color[i] == 0:
                v = i
                break
        # v should have at least 2 available colors (since units propagated)
        # try colors in order 1,2,3
        for c in (1,2,3):
            if avail[v] & (1 << (c-1)):
                stack_len = len(stack)
                if assign(v, c):
                    if dfs():
                        return True
                undo(stack_len)
        return False
        
    if dfs():
        print("YES")
        print(' '.join(str(c) for c in color))
    else:
        print("NO")

if __name__ == "__main__":
