def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [0]*n
    neigh = [[] for _ in range(n)]
    deg = [0]*n
    for _ in range(m):

        adj[u] |= 1<<v
        adj[v] |= 1<<u
        neigh[u].append(v)
        neigh[v].append(u)
        deg[u]+=1; deg[v]+=1

        domain = [7]*n
        assigned = [-1]*n
        assigned_count = 0
        changes = []

        def undo_to(start):
            nonlocal assigned_count
            while len(changes) > start:
                v, old_domain, old_assigned = changes.pop()
                if old_assigned == -1 and assigned[v] != -1:
                    assigned_count -= 1
                assigned[v] = old_assigned
                domain[v] = old_domain

        def try_assign(v, c):
            nonlocal assigned_count
            start = len(changes)
            # record assignment
            changes.append((v, domain[v], assigned[v]))
            assigned[v] = c
            domain[v] = 1 << c
            assigned_count += 1

            while queue:
                u = queue.pop()

                for w in neigh[u]:
                    if assigned[w] == -1 and (domain[w] & bit):
                        new_domain = domain[w] & ~bit
                        if new_domain == 0:
                            undo_to(start)
                            return False
                        changes.append((w, domain[w], assigned[w]))
                        domain[w] = new_domain
                        if new_domain & (new_domain - 1) == 0:

                            changes.append((w, new_domain, assigned[w]))
                            assigned[w] = col
                            domain[w] = new_domain
                            assigned_count += 1
                            queue.append(w)

        def dfs():
            nonlocal assigned_count
            if assigned_count == n:
                return True
            best = -1
            best_pc = 4
            best_deg = -1
            for i in range(n):
                if assigned[i] == -1:
                    d = domain[i]
                    if d == 0:
                        return False
