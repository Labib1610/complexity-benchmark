def solve():

    if not data: return
    it = iter(data)

    adj = [0]*n

        u = int(next(it))-1; v=int(next(it))-1
        adj[u] |= 1<<v
        adj[v] |= 1<<u

    def can_color(prefix):
        color = [0]*n

        uncolored = n

            if c == 0: continue

                    elif color[j] == c:
                        return False
        undo = []
        def undo_last():
            nonlocal uncolored
            v, old_color, changes = undo.pop()
            color[v] = old_color
            uncolored += 1
            for u, old in changes:
                avail[u] = old
        def assign(v, c):
            nonlocal uncolored
            if color[v] != 0:
                return False
            bit = 1 << (c-1)
            if not (avail[v] & bit):
                return False
            old_color = color[v]  # 0

            uncolored -= 1

            nb = adj[v]

                u = lsb.bit_length() - 1

                if color[u] == 0:

                    new = old & ~bit
                    if new != old:

                        avail[u] = new
                        if new == 0:
                            ok = False

                uncolored += 1
                for u, old in changes:
                    avail[u] = old
                return False

            return True

            start_len = len(undo)
            # propagate forced
            while True:
                forced = -1
                forced_color = 0
                for i in range(n):
