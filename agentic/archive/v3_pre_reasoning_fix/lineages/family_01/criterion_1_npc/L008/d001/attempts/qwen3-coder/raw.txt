import sys
from collections import defaultdict

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list for the graph
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try all possible assignments of zones 1, 2, 3 to each node
    # But we need to be smart about it - this is a graph coloring problem
    # We want to 3-color the graph such that no adjacent nodes have the same color
    # And among all valid colorings, we want the lexicographically smallest one
    
    # For each node, we'll try colors 1, 2, 3 in order
    # This will naturally give us the lexicographically smallest solution
    
    color = [0] * (n + 1)  # color[i] represents the zone for shard i
    
    # We'll use backtracking with constraint checking
    def is_valid(node, c):
        # Check if assigning color c to node is valid
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for c in [1, 2, 3]:
            if is_valid(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0  # backtrack
                
        return False
    
    # Start backtracking from node 1
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, color[1:])))
    else:
        print("NO")

solve()
