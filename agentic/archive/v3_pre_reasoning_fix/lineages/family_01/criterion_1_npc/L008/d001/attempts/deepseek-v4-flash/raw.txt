def can_extend(ans):
    U = 0
    for i,c in enumerate(ans):
        if c == 0:
            U |= 1 << i
    if U == 0:
        return True
    f0 = f1 = f2 = 0
    for v,c in enumerate(ans):
        if c == 1:
            f0 |= adj[v] & U
        elif c == 2:
            f1 |= adj[v] & U
        elif c == 3:
            f2 |= adj[v] & U
    # optional: if any uncolored vertex has no avail, return False
    memo = {}
    def dfs(U, f0, f1, f2):
        if U == 0:
            return True
        key = (U, f0, f1, f2)
        if key in memo:
            return memo[key]
        temp = U
        best_v = -1
        best_cnt = 4
        best_deg = -1
        best_av = 0
        while temp:
            lsb = temp & -temp
            v = lsb.bit_length() - 1
            temp ^= lsb
            av = 7
            if (f0 >> v) & 1:
                av &= ~1
            if (f1 >> v) & 1:
                av &= ~2
            if (f2 >> v) & 1:
                av &= ~4
            cnt = av.bit_count()
            if cnt == 0:
                memo[key] = False
                return False
            deg = (adj[v] & U).bit_count()
            if cnt < best_cnt or (cnt == best_cnt and deg > best_deg):
                best_cnt = cnt
                best_v = v
                best_deg = deg
                best_av = av
                if cnt == 1:
                    # can't get lower; but maybe deg? break? If cnt==1, no lower, but there might be another cnt1 with deg0? We want min cnt; if cnt==1, break okay because no lower. But if current deg maybe not max among cnt1; but any cnt1 is forced, so okay.

            res = dfs(U2, f0, f1, f2)

            return res

            c = lsb.bit_length() - 1

        return False
