import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data: return
    it = iter(data)
    n = int(next(it)); m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it))-1; v = int(next(it))-1
        adj[u].append(v); adj[v].append(u)
    ans = [0]*n
    visited = [False]*n
    for s in range(n):
        if not visited[s]:
            # BFS component
            comp = []
            stack = [s]
            visited[s] = True
            while stack:
                v = stack.pop()
                comp.append(v)
                for w in adj[v]:
                    if not visited[w]:
                        visited[w] = True
                        stack.append(w)
            comp.sort()
            # check bipartite
            color_bip = {}
            # BFS from comp[0]
            from collections import deque
            q = deque([comp[0]])
            color_bip[comp[0]] = 0
            is_bip = True
            while q and is_bip:
                v = q.popleft()
                for w in adj[v]:
                    if w not in color_bip:
                        color_bip[w] = 1 - color_bip[v]
                        q.append(w)
                    elif color_bip[w] == color_bip[v]:
                        is_bip = False
                        break
            if is_bip:
                for v in comp:
                    ans[v] = 1 if color_bip[v] == 0 else 2
            else:
                # exact DFS
                k = len(comp)
                comp_order = comp  # sorted
                color = [0]*n
                allowed = [7]*n
                # only vertices in comp matter
                assigned = []
                changes = []
                sys.setrecursionlimit(10000)
                # define propagate and dfs as nested functions capturing comp_order, adj, color, allowed, assigned, changes
                def propagate(v, c):
                    to_assign = [(v,c)]
                    while to_assign:
                        u, col = to_assign.pop()
                        if color[u] != 0:
                            if color[u] != col:
                                return False
                            continue
                        color[u] = col
                        assigned.append(u)
                        bit = 1 << (col-1)
                        for w in adj[u]:
                            if color[w] == 0:
                                if allowed[w] & bit:
                                    changes.append((w, allowed[w]))
                                    new = allowed[w] & ~bit
                                    allowed[w] = new
                                    if new == 0:
                                        return False
                                    if new & (new - 1) == 0:
                                        cw = (new & -new).bit_length()
                                        to_assign.append((w, cw))
                    return True

                def dfs():
                    # find first uncolored
                    v = -1
                    for idx in comp_order:
                        if color[idx] == 0:
                            v = idx
                            break
                    if v == -1:
                        return True
                    mask = allowed[v]
                    for c in (1,2,3):
                        if mask & (1 << (c-1)):
                            start_a = len(assigned)
                            start_c = len(changes)

                                allowed[w] = old

                                u = assigned.pop()
                                color[u] = 0
                    return False

                if not dfs():
                    print("NO")
                    return
                for v in comp:
                    ans[v] = color[v]
