class ThreeColorLex:
    def __init__(self, n, adj):
        self.n = n
        self.adj = adj
        self.neighbors = [ [j for j in range(n) if adj[i]>>j & 1] for i in range(n) ]
        self.color = [0]*n
        self.avail = [7]*n
        self.changes = []
        self.ans = [0]*n

    def set_color(self, v, cbit):
        self.changes.append((v, self.color[v], self.avail[v]))
        self.color[v] = cbit
        self.avail[v] = 0

    def propagate(self, v):
        stack = [v]
        color = self.color; avail = self.avail; neigh = self.neighbors; changes = self.changes
        while stack:
            u = stack.pop()
            cu = color[u]
            for w in neigh[u]:
                if color[w]:
                    continue
                if avail[w] & cu:
                    changes.append((w, color[w], avail[w]))
                    avail[w] &= ~cu
                    if avail[w] == 0:
                        return False
                    if (avail[w] & (avail[w] - 1)) == 0:
                        changes.append((w, color[w], avail[w]))
                        color[w] = avail[w]
                        avail[w] = 0
                        stack.append(w)
        return True

    def dfs(self, pos):
        n = self.n
        color = self.color
        while pos < n and color[pos]:
            pos += 1
        if pos == n:
            # record answer
            self.ans = [ (c.bit_length() if c else 0) for c in color ]  # but color bits 1,2,4 -> bit_length gives 1,2,3. For c=0, 0.
            return True
        bits = self.avail[pos]
        # try colors 1,2,3
        for cbit in (1,2,4):
            if not (bits & cbit):
                continue
            saved_len = len(self.changes)
            self.set_color(pos, cbit)
            if self.propagate(pos):
                if self.dfs(pos+1):
                    return True
            # undo
            while len(self.changes) > saved_len:
                v, oc, oa = self.changes.pop()
                color[v] = oc
                self.avail[v] = oa
        return False

    def solve(self):
        if self.dfs(0):
            return self.ans
        else:
            return None

class LexThreeColor:
    __slots__ = ('n','adj','neighbors','color','avail','changes')
    def __init__(self, n, adj):
        self.n = n
        self.adj = adj
        self.neighbors = [ [j for j in range(n) if (adj[i]>>j)&1] for i in range(n) ]
        self.color = [0]*n
        self.avail = [7]*n
        self.changes = []

    def set_color(self, v, cbit):
        self.changes.append((v, self.color[v], self.avail[v]))
        self.color[v] = cbit
        self.avail[v] = 0

    def propagate(self, v):
        stack = [v]
        color = self.color; avail = self.avail; neighbors = self.neighbors; changes = self.changes
        while stack:
            u = stack.pop()
            cu = color[u]
            for w in neighbors[u]:
                if color[w]:
                    continue
                if avail[w] & cu:
                    changes.append((w, color[w], avail[w]))
                    avail[w] &= ~cu
                    if avail[w] == 0:
                        return False
                    if (avail[w] & (avail[w] - 1)) == 0:
                        changes.append((w, color[w], avail[w]))
                        color[w] = avail[w]
                        avail[w] = 0
                        stack.append(w)
        return True

    def dfs(self, pos):
        n = self.n
        color = self.color
        neighbors = self.neighbors
        while pos < n and color[pos]:
            pos += 1
        if pos == n:
            return True
        bits = self.avail[pos]
        # check if pos has any uncolored neighbor
        has_future = False
        for w in neighbors[pos]:
            if color[w] == 0:
                has_future = True
                break
        if not has_future:
            # choose smallest available color
