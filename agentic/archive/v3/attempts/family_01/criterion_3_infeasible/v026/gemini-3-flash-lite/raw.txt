import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        if u > v:
            u, v = v, u
        edges.append((u, v))
    
    # To find lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all possible a, then b, then c.
    # To optimize: iterate through a, then through all neighbors b of a, 
    # then check if there is a common neighbor c such that c > b.
    
    # Pre-sort neighbors for each node to ensure we can iterate in order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        neighbors_a = sorted_adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c such that b < c and c is in adj[a] and adj[b]
            # Since we want the lexicographically smallest, we check neighbors of b
            # that are also in neighbors of a and are > b.
            neighbors_b = adj[b]
            
            # Intersecting neighbors_a (which is sorted) and neighbors_b
            # We only care about c > b
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
