```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list using sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    # Adjacency list using lists for iteration
    adj_list = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        adj_list[u].append(v)
        adj_list[v].append(u)
    
    # Sort the adjacency lists to facilitate lexicographical search
    for i in range(1, n + 1):
        adj_list[i].sort()

    # We need the lexicographically smallest triple (a, b, c) where a < b < c.
    # Lexicographical order means we prioritize smallest a, then smallest b, then smallest c.
    # We iterate a from 1 to n, then b from a+1 to n, then check if there's a c > b.
    
    for a in range(1, n + 1):
        # Only consider neighbors of a that are greater than a
        neighbors_a = [v for v in adj_list[a] if v > a]
        
        # To find the smallest b, we iterate through neighbors of a in increasing order
        for b in neighbors_a:
            # We need a c such that c > b and c is connected to both a and b.
            # To find the smallest c, we check the intersection of adj[a] and adj[b].
            # Since we want the smallest c > b, we can iterate through the 
            # smaller of the two adjacency lists or use set intersection.
            
            # Optimization: We only care about c > b.
            # We can iterate through the neighbors of b that are > b and check if they are in adj[a].
            for c in adj_list[b]:
                if c > b:
                    if c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == "__main__":
    solve()
```
