def find_mutual_interference_triple(n, m, edges):
    from collections import defaultdict

    graph = defaultdict(set)

    # Building the graph
    for u, v in edges:
        graph[u].add(v)
        graph[v].add(u)

    # Check for triples
    for a in range(1, n + 1):
        for b in graph[a]:
            if b > a:  # ensuring a < b
                for c in graph[b]:
                    if c > b and c in graph[a]:  # ensuring b < c and a, b, c are mutually connected
                        return a, b, c

# Input reading
import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]

# Finding the result
result = find_mutual_interference_triple(n, m, edges)
print(*result)
