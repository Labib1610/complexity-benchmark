import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list to store the graph
    adj = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # Store edges such that we only look at neighbors with higher indices
        if u < v:
            adj[u].append(v)
        else:
            adj[v].append(u)
            
    # Sort neighbors for each node to ensure we find the lexicographically smallest triplet
    for i in range(1, n + 1):
        adj[i].sort()
        
    # To check for edges efficiently, we can use a bitset or a set.
    # Since N <= 2000, a bitset (represented by a large integer in Python) 
    # or a boolean matrix is feasible.
    # Using a list of sets for O(1) average lookup.
    has_edge = [set() for _ in range(n + 1)]
    # Re-populate the edges for fast lookup
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        has_edge[u].add(v)
        has_edge[v].add(u)

    # We want the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate through potential 'a' from 1 to n.
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate through potential 'b' from neighbors of 'a' (where b > a).
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now we look for 'c' such that c > b and (a, c) and (b, c) are edges.
            # Since we sorted neighbors_a, we only check elements after index i.
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # We already know (a, b) and (a, c) are edges. Check if (b, c) is an edge.
                if c in has_edge[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
