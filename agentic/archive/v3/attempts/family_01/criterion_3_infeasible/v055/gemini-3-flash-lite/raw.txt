import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use bitsets for fast adjacency checks
    # adj[i] will store a bitmask of neighbors of i
    # Since n <= 2000, bitsets are efficient in Python
    adj = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Iterate c from b+1 to n
    # Check if (a, b), (a, c), (b, c) are edges
    
    for a in range(1, n - 1):
        # Pre-filter nodes connected to a
        adj_a = adj[a]
        for b in range(a + 1, n):
            if (adj_a >> b) & 1:
                # b is connected to a, now check for c
                # c must be connected to both a and b
                adj_b = adj[b]
                common = adj_a & adj_b
                
                # Check for any bit set at position > b
                # We need the smallest such c
                # Mask out bits <= b
                mask = common >> (b + 1)
                if mask:
                    # Find the position of the lowest set bit
                    # (mask & -mask) gives the lowest bit
                    # bit_length() gives the position
                    c = (mask & -mask).bit_length() + b
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
