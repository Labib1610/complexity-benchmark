```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check potential b's in order
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # We need a c such that b < c and c in adj[a] and c in adj[b]
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            valid_c = [x for x in common if x > b]
            if valid_c:
                print(f"{a} {b} {min(valid_c)}")
                return

if __name__ == '__main__':
    solve()
```
