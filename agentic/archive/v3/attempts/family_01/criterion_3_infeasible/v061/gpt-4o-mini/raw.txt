def find_mutually_incompatible_triplet(n, m, incompatibilities):
    from collections import defaultdict
    
    # Create an adjacency list for incompatibilities
    incompat_graph = defaultdict(set)
    for u, v in incompatibilities:
        incompat_graph[u].add(v)
        incompat_graph[v].add(u)
    
    # Check for each pair (a, b) if there's a third member c that is incompatible with both
    for a in range(1, n + 1):
        for b in incompat_graph[a]:
            if a < b:
                for c in incompat_graph[a].intersection(incompat_graph[b]):
                    if b < c:
                        return a, b, c

# Read input
import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
incompatibilities = [tuple(map(int, line.split())) for line in data[1:m + 1]]

# Find and print the result
result = find_mutually_incompatible_triplet(n, m, incompatibilities)
print(*result)
