import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
    
    # To find lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then iterate over all b in adj[a] such that b > a,
    # then check if there is any c in adj[a] AND adj[b] such that c > b.
    
    # Sort adjacency lists to ensure we pick smallest b and c
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # We need a common neighbor c of a and b such that c > b
            # Use sets for O(1) lookup
            set_a = set(neighbors_a)
            set_b = set(adj[b])
            
            # Intersection of neighbors of a and b
            common = sorted(list(set_a.intersection(set_b)))
            for c in common:
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
