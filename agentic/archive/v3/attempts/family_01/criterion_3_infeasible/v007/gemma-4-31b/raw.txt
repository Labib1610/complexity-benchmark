```python
import sys

def solve():
    # Increase recursion depth for deep graphs, though not needed for this approach
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency lists to store the graph
    # Use sets for O(1) lookup to check if an edge exists
    adj = [[] for _ in range(n + 1)]
    adj_set = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # Store edges such that the adjacency list is sorted for lexicographical search
        adj[u].append(v)
        adj[v].append(u)
        adj_set[u].add(v)
        adj_set[v].add(u)
    
    # Sort adjacency lists to ensure we iterate in increasing order of node IDs
    for i in range(1, n + 1):
        adj[i].sort()
        
    # We want the lexicographically smallest triple (a, b, c) with a < b < c.
    # This means we minimize a, then b, then c.
    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_of_a = [v for v in adj[a] if v > a]
        
        # If a has fewer than 2 neighbors > a, it cannot be the smallest element of a triangle
        if len(neighbors_of_a) < 2:
            continue
            
        # Now we look for b and c among neighbors_of_a such that b < c and b is connected to c.
        # Since neighbors_of_a is already sorted, we iterate b and c in increasing order.
        for i in range(len(neighbors_of_a)):
            b = neighbors_of_a[i]
            for j in range(i + 1, len(neighbors_of_a)):
                c = neighbors_of_a[j]
                # Check if b and c are connected
                if c in adj_set[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
```
