def find_mutually_interfering_triple(n, m, edges):
    from collections import defaultdict
    
    graph = defaultdict(set)
    
    for u, v in edges:
        graph[u].add(v)
        graph[v].add(u)
    
    for a in range(1, n + 1):
        for b in graph[a]:
            if b > a:  # Ensuring a < b
                for c in graph[b]:
                    if c > b and c in graph[a]:  # Ensuring b < c and a, b, c interfere
                        return a, b, c

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = find_mutually_interfering_triple(n, m, edges)
print(*result)
