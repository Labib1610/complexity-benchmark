import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Store adjacency as a list of sets for fast lookup
    # Using a list of bits (integers) can be faster for dense graphs,
    # but with n=2000, bitsets or sets are fine.
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # We iterate a from 1 to n-2, b from a+1 to n-1, and check if there exists c > b
    # such that c is in adj[a] and c is in adj[b].
    
    # Pre-sort adjacency lists to make intersection faster
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b are neighbors of a
        neighbors_a = sorted_adj[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need c > b such that c is in adj[a] and c in adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            # but since we only need the smallest c, we can just iterate.
            
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
