import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list using sets for O(1) lookup
    # We only store edges where u < v to simplify and save space
    adj = [set() for _ in range(n + 1)]
    # We also need a sorted list of neighbors to find the lexicographically smallest triple
    adj_list = [[] for _ in range(n + 1)]
    
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        ptr += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj_list[u].append(v)
    
    # Sort adjacency lists to ensure we check candidates in lexicographical order
    for i in range(1, n + 1):
        adj_list[i].sort()

    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # 1. Iterate a from 1 to n
    # 2. Iterate b from the neighbors of a that are > a
    # 3. Iterate c from the neighbors of a that are > b
    # 4. Check if b and c are connected.
    
    # Optimization: To avoid O(n^3), we can use bitsets if n is small, 
    # but n=2000 and Python doesn't have native bitsets like C++.
    # However, the requirement is the lexicographically smallest triple.
    # We can iterate a and b, then intersect their adjacency sets.
    
    # Pre-calculate full adjacency sets for fast lookup of (b, c)
    # Since we only stored u < v, we need to be able to check any pair.
    full_adj = [set() for _ in range(n + 1)]
    # We reuse the input data to fill this
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        ptr += 2
        full_adj[u].add(v)
        full_adj[v].add(u)

    for a in range(1, n + 1):
        # Neighbors of a that are greater than a
        neighbors_a = adj_list[a]
        
        # Iterate through pairs (b, c) from neighbors_a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # We need b > a, which is guaranteed by adj_list[a] and how we filled it
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if b and c are connected
                # b < c is guaranteed by the sorted nature of neighbors_a
                if c in full_adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    solve()
