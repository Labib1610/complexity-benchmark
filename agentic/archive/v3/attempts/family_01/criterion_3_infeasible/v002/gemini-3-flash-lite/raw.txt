import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find the lexicographically smallest triple
    edges.sort()
    
    # We need to find a, b, c such that 1 <= a < b < c <= n
    # and (a,b), (a,c), (b,c) are all edges.
    # To get the lexicographically smallest (a, b, c):
    # Iterate through each edge (u, v) as the potential (a, b)
    # Then find if there exists a c > b such that c is in adj[u] and adj[v].
    
    for a in range(1, n + 1):
        # Sort neighbors to check for smallest c
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we have edge (a, b). Look for c > b
            # c must be in both adj[a] and adj[b]
            # To optimize, intersect the two sets
            common = adj[a].intersection(adj[b])
            
            # Filter for c > b
            candidates = [c for c in common if c > b]
            if candidates:
                print(f"{a} {b} {min(candidates)}")
                return

if __name__ == '__main__':
    solve()
