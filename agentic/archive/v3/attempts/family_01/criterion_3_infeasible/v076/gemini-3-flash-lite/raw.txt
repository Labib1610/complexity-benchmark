```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency list representation with sets for fast lookups
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through a from 1 to n-2
    # Iterate through b from a+1 to n-1
    # Iterate through c from b+1 to n
    # Since we want the lexicographically smallest, this order works perfectly.
    # To optimize, we only check b if a is connected to b,
    # and we only check c if a is connected to c AND b is connected to c.
    
    # Sort adjacency lists to potentially speed up finding the smallest c
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Candidates for b must be greater than a and connected to a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # Candidates for c must be greater than b, connected to a, and connected to b
            # We intersect neighbors of a and b to find potential c's
            # Since neighbors_a is sorted, we can iterate efficiently
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
