import sys

def solve():
    # Fast I/O
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency matrix for O(1) edge check
    # Using 1-based indexing for convenience
    adj = [[0] * (n + 1) for _ in range(n + 1)]
    
    # Adjacency lists for iteration
    adj_list = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u][v] = 1
        adj[v][u] = 1
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # Sort adjacency lists to allow efficient iteration and filtering
    for i in range(1, n + 1):
        adj_list[i].sort()
        
    # We iterate a from 1 to n.
    # For each a, we look for b, c such that a < b < c and edges (a,b), (a,c), (b,c) exist.
    # To ensure lexicographically smallest (a, b, c), we iterate a increasing,
    # then b increasing, then c increasing.
    
    for a in range(1, n - 1):
        # We are interested in neighbors of a that are > a.
        # Since adj_list[a] is sorted, we can iterate starting from the first element > a.
        # However, simply iterating and checking x > a is O(d_a).
        # Given constraints, O(d_a) is acceptable.
        
        # We can filter neighbors > a.
        # neighbors_a = [x for x in adj_list[a] if x > a]
        # This creates a new list. To save memory/time, we can iterate indices.
        # But creating the list is cleaner and likely fast enough.
        # Let's do it.
        
        neighbors_a = [x for x in adj_list[a] if x > a]
        
        # Now we need to find b, c in neighbors_a such that b < c and (b, c) is an edge.
        # Since neighbors_a is sorted, we can iterate b as neighbors_a[i]
        # and c as neighbors_a[j] with j > i.
        # The first pair (b, c) that satisfies (b, c) is an edge will be the answer
        # because b is minimized, then c is minimized.
        
        len_na = len(neighbors_a)
        for i in range(len_na):
            b = neighbors_a[i]
            # Optimization: if we find a c, we return immediately.
            # Since we iterate b in increasing order, the first valid (b, c) found
            # for this 'a' will have the smallest 'b'.
            # And since we iterate c in increasing order (j > i), it will have smallest 'c'.
            
            # Iterate c candidates
            for j in range(i + 1, len_na):
                c = neighbors_a[j]
                # Check if edge (b, c) exists
                if adj[b][c]:
                    # Found the lexicographically smallest triangle
                    print(f"{a} {b} {c}")
                    return
