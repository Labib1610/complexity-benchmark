import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        edges.append((min(u, v), max(u, v)))
        idx += 2
        
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c,
    # we iterate through possible values of a, then b, then c.
    # Since n <= 2000, we can iterate through all edges (a, b) and check
    # if there is a node c > b such that c is connected to both a and b.
    
    # Sort edges to ensure we check potential 'a' and 'b' in increasing order
    edges.sort()
    
    # Pre-sort adjacency lists for faster intersection
    adj_list = [sorted(list(s)) for s in adj]
    
    # Iterate through each edge (a, b) where a < b
    for a, b in edges:
        # We need a node c such that b < c and c is in adj[a] and adj[b]
        # To minimize lexicographical order, we want smallest a, then b, then c.
        # Since we iterate edges in sorted order, we already have the smallest (a, b).
        # Now find the smallest c > b.
        
        # Use set intersection for O(min(deg(a), deg(b)))
        # To find smallest c > b efficiently:
        set_a = adj[a]
        set_b = adj[b]
        
        # We only care about c > b
        # Iterate over the smaller adjacency list
        if len(set_a) < len(set_b):
            candidates = set_a
            other = set_b
        else:
            candidates = set_b
            other = set_a
            
        best_c = float('inf')
        found = False
        for c in candidates:
            if c > b:
                if c in other:
                    if c < best_c:
                        best_c = c
                        found = True
        
        if found:
            print(f"{a} {b} {best_c}")
            return

if __name__ == '__main__':
    solve()
