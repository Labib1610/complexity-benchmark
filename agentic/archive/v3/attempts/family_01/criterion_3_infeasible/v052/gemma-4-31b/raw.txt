```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list for graph representation
    adj = [[] for _ in range(n + 1)]
    # Adjacency matrix for O(1) lookup (using bitsets for speed in Python)
    # Since n <= 2000, a list of integers acting as bitsets is efficient
    bit_adj = [0] * (n + 1)
    
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr + 1])
        ptr += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
        bit_adj[u] |= (1 << v)
        bit_adj[v] |= (1 << u)

    # To find the lexicographically smallest (a, b, c) where a < b < c:
    # Iterate a from 1 to n, then b from a+1 to n, then check for c.
    # To optimize, we only iterate through existing edges.
    
    # Sort adjacency lists to ensure we process b in increasing order
    for i in range(1, n + 1):
        adj[i].sort()

    for a in range(1, n + 1):
        # Only consider b > a
        for b in adj[a]:
            if b <= a:
                continue
            
            # We need c > b such that c is connected to both a and b.
            # This is the intersection of adj[a] and adj[b].
            # Using bitwise AND for fast intersection.
            common = bit_adj[a] & bit_adj[b]
            
            # We need the smallest c > b.
            # We can find the lowest set bit in 'common' that is > b.
            # Mask out bits <= b
            mask = ~((1 << (b + 1)) - 1)
            candidates = common & mask
            
            if candidates:
                # Find the index of the lowest set bit
                # candidates.bit_length() is not helpful for lowest bit.
                # (candidates & -candidates).bit_length() - 1 gives the index.
                c = (candidates & -candidates).bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
