import sys

def solve():
    # Read n and m from the first line
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list to store the graph
    adj = [[] for _ in range(n + 1)]
    # Adjacency matrix (using bitsets/bitmasks for efficiency in Python)
    # Since n <= 2000, we can use an array of integers as bitsets
    # However, for n=2000, a list of sets or a bitset might be faster.
    # In Python, the most efficient way to check existence of an edge 
    # for this size is often a list of sets or a bitmask (using large ints).
    
    # We'll use a list of sets for O(1) average lookup
    adj_set = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].append(v)
        adj_set[u].add(v)
        adj_set[v].add(u)

    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # We iterate a from 1 to n, then b from a+1 to n, then check for c.
    # However, iterating all pairs (a, b) is O(n^2). 
    # For each pair (a, b) that is an edge, we need to find the smallest c > b 
    # such that c is connected to both a and b.
    
    # Sort the adjacency lists to ensure we process neighbors in increasing order
    for i in range(1, n + 1):
        adj[i].sort()

    # Optimization: only iterate through existing edges (a, b) where a < b
    for a in range(1, n + 1):
        # Only consider neighbors of a that are greater than a
        neighbors_a = [v for v in adj[a] if v > a]
        
        for b in neighbors_a:
            # Now we need the smallest c > b such that (a, c) and (b, c) are edges.
            # We can iterate through the neighbors of b that are > b and check if they are in adj_set[a].
            # Or iterate through neighbors of a that are > b and check if they are in adj_set[b].
            # To find the smallest c, we should iterate the smaller neighbor list.
            
            # We look for c in the neighbors of b that are > b
            # Since adj[b] is sorted, the first c we find that is also in adj_set[a] is the smallest.
            for c in adj[b]:
                if c > b:
                    if c in adj_set[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == "__main__":
    solve()
