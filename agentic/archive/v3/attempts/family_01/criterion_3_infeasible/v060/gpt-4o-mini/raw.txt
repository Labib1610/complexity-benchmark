def find_smallest_triad(n, m, edges):
    from collections import defaultdict

    graph = defaultdict(set)

    for u, v in edges:
        graph[u].add(v)
        graph[v].add(u)

    smallest_triad = None

    for a in range(1, n + 1):
        for b in graph[a]:
            if b > a:  # Ensure a < b
                for c in graph[b]:
                    if c > b and c in graph[a]:  # Ensure b < c and a < c
                        triad = (a, b, c)
                        if smallest_triad is None or triad < smallest_triad:
                            smallest_triad = triad

    return smallest_triad

import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = find_smallest_triad(n, m, edges)
print(result[0], result[1], result[2])
