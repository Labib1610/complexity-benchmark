import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list and bitsets (represented as integers in Python)
    # Using bitsets for O(N/64) intersection check
    adj_bitset = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        
        # To find the lexicographically smallest (a, b, c) with a < b < c:
        # We iterate a from 1 to n, then b from a+1 to n, then check for c > b.
        adj_list[u].append(v)
        adj_bitset[u] |= (1 << v)
        adj_bitset[v] |= (1 << u)

    # Sort adjacency lists to ensure we check b in increasing order
    for i in range(1, n + 1):
        adj_list[i].sort()

    # Lexicographical search for (a, b, c)
    for a in range(1, n + 1):
        # For a fixed a, we only care about neighbors b > a
        neighbors_a = [v for v in adj_list[a] if v > a]
        
        # To optimize, we can use the bitset of a's neighbors
        # mask_a contains 1s for all v such that (a, v) is an edge
        mask_a = adj_bitset[a]
        
        for b in neighbors_a:
            # We need c such that c > b and (a, c) is an edge and (b, c) is an edge.
            # This is equivalent to finding the smallest c in the intersection
            # of neighbors of a and neighbors of b, where c > b.
            
            # mask_b contains 1s for all v such that (b, v) is an edge
            mask_b = adj_bitset[b]
            
            # Common neighbors of a and b
            common = mask_a & mask_b
            
            # We need the smallest c > b. 
            # Clear bits <= b in the common bitset.
            # (1 << (b + 1)) - 1 creates a mask of bits 0 to b.
            # XORing or ANDing with the complement removes them.
            remaining = common & ~((1 << (b + 1)) - 1)
            
            if remaining:
                # Find the lowest set bit (the smallest c)
                # bit_length() and trailing zeros can be used, 
                # but (remaining & -remaining).bit_length() - 1 is a fast way to get index.
                # Since we need the absolute smallest c, we can use .bit_length() 
                # or simply iterate if the number of set bits is small.
                # In Python, (remaining & -remaining).bit_length() - 1 is efficient.
                low_bit = remaining & -remaining
                c = low_bit.bit_length() - 1
                print(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
