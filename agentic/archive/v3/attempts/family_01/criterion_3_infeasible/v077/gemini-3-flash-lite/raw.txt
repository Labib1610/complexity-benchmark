```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if a and b are connected
    # If so, find intersection of neighbors of a and b that are > b
    
    for a in range(1, n + 1):
        # Sort neighbors to maintain lexicographical order search
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we look for c such that b < c and c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can intersect efficiently
            # but for n=2000, bitsets or simple set intersection is fast enough
            
            # Get intersection of neighbors of a and b
            common = adj[a].intersection(adj[b])
            
            # Find smallest c > b
            best_c = n + 1
            for c in common:
                if c > b and c < best_c:
                    best_c = c
            
            if best_c <= n:
                print(f"{a} {b} {best_c}")
                return

solve()
```
