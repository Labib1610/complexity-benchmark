import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c),
    # we iterate through possible values of a, then b, then c.
    # Since we need a < b < c, we iterate a from 1 to n-2,
    # b from a+1 to n-1, and c from b+1 to n.
    
    # Optimization: Pre-sort neighbor lists to check in order
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b such that b > a and b is connected to a
        neighbors_a = [b for b in sorted_adj[a] if b > a]
        for b in neighbors_a:
            # Check for common neighbors c such that c > b
            # Intersection of neighbors of a and neighbors of b
            # We want the smallest c > b
            set_a = adj[a]
            set_b = adj[b]
            
            # Iterate through sorted neighbors of b that are > b
            # to find the first one that is also in set_a
            for c in sorted_adj[b]:
                if c > b and c in set_a:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
