def frequency_channel_assignment():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    graph = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)
    
    color = [-1] * (n + 1)  # -1 means uncolored
    valid = True
    
    def bfs(start):
        queue = deque([start])
        color[start] = 0  # Start coloring with 0
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == -1:  # Not colored yet
                    color[neighbor] = 1 - color[node]  # Alternate color
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:  # Conflict found
                    return False
        return True
    
    for i in range(1, n + 1):
        if color[i] == -1:  # Not colored, start a new component
            if not bfs(i):
                valid = False
                break
    
    if not valid:
        print("NO")
    else:
        # Convert 0, 1 to 1, 2 for output
        result = [1 + color[i] for i in range(1, n + 1)]
        print("YES")
        print(" ".join(map(str, result)))

frequency_channel_assignment()
