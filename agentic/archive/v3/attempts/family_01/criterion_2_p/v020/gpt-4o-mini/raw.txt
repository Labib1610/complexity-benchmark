def grid_bus_assignment(n, m, edges):
    from collections import defaultdict, deque
    
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    color = [0] * (n + 1)
    
    def bfs(start):
        queue = deque([start])
        color[start] = 1  # Start coloring with 1
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == 0:  # If not colored
                    color[neighbor] = 3 - color[node]  # Alternate colors (1 <-> 2)
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:  # Conflict detected
                    return False
        return True

    for i in range(1, n + 1):
        if color[i] == 0:  # Not colored yet
            if not bfs(i):
                print("NO")
                return

    # Prepare the result
    result = [color[i] for i in range(1, n + 1)]
    print("YES")
    print(" ".join(map(str, result)))

import sys
input = sys.stdin.read
data = input().splitlines()

n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]

grid_bus_assignment(n, m, edges)
