import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is unvisited, we try assigning it shift 1, then propagate.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for the connected component
            # To ensure lexicographically smallest, we prefer 1 over 2.
            # However, for a bipartite component, there are only two colorings.
            # We try assigning 1 to the first node of the component.
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            while stack:
                u, color = stack.pop()
                component.append((u, color))
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, 3 - color))
                    elif assignment[v] == color:
                        print("NO")
                        return
            
            # Check if this choice (assignment[i]=1) is the lexicographically smallest
            # for this component. If we swap 1s and 2s, we get another valid assignment.
            # We want the one where the smallest index in the component has the smallest value.
            # The current component nodes were colored based on assignment[i]=1.
            # Let's see if swapping makes the assignment lexicographically smaller.
            # We find the smallest index 'min_idx' in this component.
            # We want assignment[min_idx] to be 1.
            min_idx = min(c[0] for c in component)
            if assignment[min_idx] == 2:
                for node, color in component:
                    assignment[node] = 3 - color
                    
    print("YES")
    print(*(assignment[1:]))

solve()
