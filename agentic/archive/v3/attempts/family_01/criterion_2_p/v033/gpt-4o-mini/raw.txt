def main():
    import sys
    from collections import defaultdict, deque

    input = sys.stdin.read
    data = input().splitlines()

    n, m = map(int, data[0].split())
    graph = defaultdict(list)

    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)

    color = [0] * (n + 1)  # 0 means uncolored, 1 and 2 are the two groups
    is_bipartite = True

    def bfs(start):
        nonlocal is_bipartite
        queue = deque([start])
        color[start] = 1  # Start coloring with group 1

        while queue and is_bipartite:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == 0:  # If uncolored
                    color[neighbor] = 2 if color[node] == 1 else 1
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:  # Conflict found
                    is_bipartite = False
                    return

    for i in range(1, n + 1):
        if color[i] == 0:
            bfs(i)
            if not is_bipartite:
                break

    if not is_bipartite:
        print("NO")
    else:
        print("YES")
        print(' '.join(str(c) for c in color[1:]))

main()
