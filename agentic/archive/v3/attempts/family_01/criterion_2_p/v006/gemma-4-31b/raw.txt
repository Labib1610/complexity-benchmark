import sys

# Increase recursion depth for deep components
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will be 0 if unvisited, 1 or 2 otherwise
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, for each connected 
    # component, we want to assign the smallest possible values to the 
    # smallest indices. 
    # However, for a bipartite component, once the color of one node is fixed, 
    # the colors of all other nodes in that component are determined.
    # Let the two possible bipartite colorings be C1 and C2.
    # C1: root is 1, others are derived.
    # C2: root is 2, others are derived (C2 is just C1 flipped).
    # We compare C1 and C2 lexicographically and pick the smaller one.
    # The smallest index in the component determines the priority.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to find all nodes in this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard BFS to find component and check bipartiteness
            # Using a list as a queue for simplicity since we need the component
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # We have a valid bipartite coloring for this component.
            # The current coloring is 'color'. The alternative is '3 - color'.
            # To get the lexicographically smallest sequence, we look at the
            # smallest index in the component. If we can set it to 1, we should.
            # Since the component was started from 'i' (the smallest index 
            # in this component), and we set color[i] = 1, the current 
            # coloring is already the lexicographically smallest for this component.
            # Because any other valid coloring would have color[i] = 2, 
            # which is lexicographically larger.
            
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
