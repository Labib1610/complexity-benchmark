import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1) # 0: unvisited, 1: color 1, 2: color 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS from i, assign color 1
            color[i] = 1
            queue = deque([i])
            while queue:
                u = queue.popleft()
                c = color[u]
                next_c = 3 - c
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        queue.append(v)
                    elif color[v] != next_c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Using 1-based indexing for convenience, size n + 1
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break

    # Color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * (n + 1)
    
    # Iterate through all nodes to handle disconnected components
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS for the component containing i
            # We assign color 1 to the first node encountered (lexicographically smallest for this component)
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_color = color[u]
                next_color = 3 - current_color  # 1 -> 2, 2 -> 1

                    if color[v] == 0:
                        color[v] = next_color
                        queue.append(v)
                    elif color[v] != next_color:
                        # Conflict detected: graph is not bipartite
                        print("NO")
                        return

    # If no conflicts found
    print("YES")
    # Output the colors for nodes 1 to n
    # Using sys.stdout.write for faster output is possible but print is usually fine
    # Joining strings is efficient
