    # Using list of lists
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))

        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)

    for i in range(1, n + 1):
        if color[i] == 0:
            # Start traversal for this component
            # Assign channel 1 to the first node to ensure lexicographically smallest
            color[i] = 1
            stack = [i]
            
            while stack:
                u = stack.pop()
                c_u = color[u]
                # The neighbor must have color 3 - c_u (since 1+2=3)
                # 3 - 1 = 2, 3 - 2 = 1
                
                c_v = 3 - c_u
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = c_v
                        stack.append(v)
                    elif color[v] != c_v:
                        # Conflict detected
                        print("NO")
                        return

    # If we reach here, a valid assignment exists
    print("YES")
    # Construct output string
    # Using sys.stdout.write for efficiency
    # map(str, color[1:]) converts integers to strings
    sys.stdout.write(" ".join(map(str, color[1:])) + "\n")
