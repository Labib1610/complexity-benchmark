input n,m
adj=[[] for _ in range(n+1)]
for _ in range(m):
    u,v = map(int,input().split())
    adj[u].append(v)
    adj[v].append(u)
color=[-1]*(n+1)
ans=[0]*(n+1)
from collections import deque
for i in range(1,n+1):
    if color[i]==-1:
        # bfs
        q=deque([i])
        color[i]=0
        comp_nodes=[i]
        while q:
            u=q.popleft()
            for v in adj[u]:
                if color[v]==-1:
                    color[v]=color[u]^1
                    q.append(v)
                    comp_nodes.append(v)
                elif color[v]==color[u]:
                    print('NO'); return
        min_idx=min(comp_nodes)
        if color[min_idx]==0:
            flip=0
        else:
            flip=1
        for node in comp_nodes:
            ans[node]=(color[node]^flip)+1
print('YES')
print(' '.join(str(ans[i]) for i in range(1,n+1)))
