# Set recursion limit just in case, though iterative BFS is used.
sys.setrecursionlimit(200000)

def solve():
    # Efficiently read all input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    # 1-based indexing for sensors
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
    
    # Array to store the phase setting (1 or 2)
    # 0 indicates the sensor has not been assigned a phase yet
    phase = [0] * (n + 1)
    
    # Iterate through each sensor to find connected components
    for i in range(1, n + 1):
        if phase[i] == 0:
            # Start BFS for the connected component containing sensor i
            # To achieve the lexicographically smallest configuration,
            # we must assign phase 1 to the smallest index node in the component.
            # Since we iterate from 1 to n, i is the smallest index in this component.
            phase[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_phase = phase[u]
                # The required phase for neighbors is the opposite (1 <-> 2)
                neighbor_phase = 3 - current_phase
                
                for v in adj[u]:
                    if phase[v] == 0:
                        phase[v] = neighbor_phase
                        queue.append(v)
                    elif phase[v] == current_phase:
                        # Conflict found: adjacent nodes have the same phase
                        # This implies the graph is not bipartite
                        print("NO")
                        return
    
    # If we successfully assigned phases to all sensors without conflicts
    print("YES")
    # Output the phases for sensors 1 to n
    sys.stdout.write(' '.join(map(str, phase[1:])) + '\n')
