from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))

        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            queue = deque([i])
            while queue:
                u = queue.popleft()
                c_u = color[u]

                    if color[v] == 0:
                        color[v] = c_v_target
                        queue.append(v)
                    elif color[v] != c_v_target:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

import sys
from collections import deque

# Increase recursion depth just in case, though we use iterative BFS.
sys.setrecursionlimit(200005)

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * (n + 1)
    
    # Iterate through all stations to handle disconnected components
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS from the smallest index in this component
            # To ensure lexicographically smallest sequence, set s[i] = 1
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                
                # Determine the required color for neighbors
                # If u is 1, neighbor must be 2. If u is 2, neighbor must be 1.
                # This can be calculated as 3 - color[u]
                neighbor_color = 3 - color[u]
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = neighbor_color
                        queue.append(v)
                    elif color[v] != neighbor_color:
                        # Conflict detected: neighbors have same color
                        print("NO")
                        return

    # If no conflicts found
    print("YES")
    # Output colors for 1 to n
    # Using join is generally safer/faster for large outputs than * unpacking
    print(" ".join(map(str, color[1:])))

if __name__ == '__main__':
    solve()

import sys
from collections import deque

# Increase recursion depth just in case, though not used here.
sys.setrecursionlimit(200005)

def solve():
    # Read all input from standard input at once
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Build adjacency list
    # Using 1-based indexing for convenience
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store colors (1 or 2), 0 indicates unvisited
    color = [0] * (n + 1)
    
    # Iterate through all nodes to handle all connected components
    for i in range(1, n + 1):
        if color[i] == 0:
            # This is a new component.
            # To achieve the lexicographically smallest configuration,
            # we set the first node of the component (smallest index i) to 1.
            # This forces the rest of the component's values relative to this.
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                
                # The neighbor must have the opposite color
                # If u is 1, neighbor is 2. If u is 2, neighbor is 1.
                # 3 - 1 = 2, 3 - 2 = 1
                target_color = 3 - color[u]
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target_color
                        queue.append(v)
                    elif color[v] != target_color:
                        # Conflict: adjacent nodes have the same color
                        print("NO")
                        return

    # If we successfully colored the graph
    print("YES")
    # Print the colors for stations 1 to n
    print(" ".join(map(str, color[1:])))

if __name__ == '__main__':
    solve()

def solve():
    # Read all input from standard input efficiently
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    try:

        m = int(next(iterator))
    except StopIteration:
        return

    # Build adjacency list for the graph
    # Using 1-based indexing for stations 1 to n
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store the frequency band for each station
    # 0 indicates unvisited, 1 and 2 are the available bands
    color = [0] * (n + 1)
    
    # Iterate through all stations to handle all connected components
    # This ensures we visit every station and handle disconnected parts of the graph
    for i in range(1, n + 1):
        if color[i] == 0:
            # We found a new connected component.
            # To achieve the lexicographically smallest configuration, we should assign
            # the smallest possible value to the earliest station in the component.
            # Since we iterate i from 1 to n, 'i' is the smallest index in this component.
            # We assign band 1 to station i. This forces the assignment of all other
            # stations in this component due to the constraints.
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                
                # If station u is in band 1, its neighbors must be in band 2.
                # If station u is in band 2, its neighbors must be in band 1.
                # This can be computed as 3 - color[u].
                target_color = 3 - color[u]
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target_color
                        queue.append(v)
                    elif color[v] != target_color:
                        # Conflict detected: adjacent stations have the same frequency band.
                        # This means the graph is not bipartite.
                        print("NO")
                        return

    print("YES")
    # Output the configuration for stations 1 to n
    print(" ".join(map(str, color[1:])))
