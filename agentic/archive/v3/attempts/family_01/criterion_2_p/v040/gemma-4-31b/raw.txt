import sys

# Increase recursion depth for deep DFS trees in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # colors: 0 = unvisited, 1 = channel 1, 2 = channel 2
    colors = [0] * (n + 1)

    # we iterate through units 1 to n. If a unit is not yet assigned,
    # we must try to assign it to 1 first.
    # Because the graph might be disconnected, we process each component.
    # In a bipartite component, once the color of one node is fixed,
    # the colors of all other reachable nodes are determined.
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Try assigning the smallest available color (1) to the first node of the component
            # to aim for lexicographical minimality.
            # However, the choice of color for the root of a component doesn't 
            # necessarily guarantee the lexicographically smallest sequence for the whole array
            # unless we carefully consider the constraints.
            # Actually, for a bipartite component, there are only two valid colorings:
            # Coloring A: root=1, neighbors=2, ...
            # Coloring B: root=2, neighbors=1, ...
            # To get the lexicographically smallest sequence, we compare the two.
            # But since we process nodes in order 1...n, assigning root i = 1
            # is always better than assigning root i = 2 if both are valid.
            
            stack = [(i, 1)]
            component = []
            colors[i] = 1
            
            # BFS/DFS to find all nodes in the component and check bipartiteness
            # Using a stack for iterative DFS to avoid recursion limit issues
            visit_stack = [i]
            component.append(i)
            
            # We use a simple iterative approach to color the component
            # and check for contradictions.
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        q.append(v)
                        component.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # Let's check if flipping the colors (1->2, 2->1) results in a
            # lexicographically smaller sequence for the nodes in this component.
            # The nodes are processed in order 1...n. The first node in the component
            # that we encountered is 'i'. If we set colors[i] = 1, the sequence
            # starts with 1 at index i, which is always smaller than starting with 2.
            # Thus, assigning colors[i] = 1 is always optimal for the lexicographical requirement.
            # (The logic: for the first index j in the component, s_j = 1 is better than s_j = 2).
            # Since we iterate i from 1 to n, i is the smallest index in its component.
            
    print("YES")
    print(*(colors[1:]))

if __name__ == "__main__":
    solve()
