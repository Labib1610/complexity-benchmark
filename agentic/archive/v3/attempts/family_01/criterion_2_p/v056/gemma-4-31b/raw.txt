import sys

# Increase recursion depth for deep DFS in large graphs
sys.setrecursionlimit(300000)

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    # modes[i] will be 0 (unassigned), 1, or 2
    modes = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we want to assign 1 
    # to the smallest possible indices. Since the graph can have multiple 
    # connected components, we iterate from 1 to n.
    # For each component, we determine if it's bipartite.
    # If it is, there are exactly two possible colorings (swapping 1 and 2).
    # To make the sequence lexicographically smallest, for each component, 
    # we compare the two valid colorings and pick the one that results in 
    # a smaller sequence.
    # However, a simpler approach: for the first node 'r' encountered in 
    # a component, we try assigning it mode 1. Then we propagate. 
    # But wait, assigning the smallest index in a component to 1 
    # doesn't necessarily guarantee the lexicographically smallest 
    # sequence for the whole component. 
    # Actually, it does: if we have two valid colorings A and B for a component,
    # and the smallest index in that component is i, then one coloring has 
    # A[i]=1 and the other has B[i]=2. Since A[i] < B[i], coloring A is 
    # lexicographically smaller regardless of the values at indices > i.
    
    for i in range(1, n + 1):
        if modes[i] == 0:
            # Start a BFS/DFS to color this component
            # We force the smallest index in the component to be 1
            stack = [i]
            modes[i] = 1
            component = []
            
            # Use a manual stack to avoid recursion depth issues and 
            # collect all nodes in the current component
            visited_in_comp = []
            q = [i]
            idx = 0
            while idx < len(q):
                u = q[idx]
                idx += 1
                visited_in_comp.append(u)
                for v in adj[u]:
                    if modes[v] == 0:
                        modes[v] = 3 - modes[u]
                        q.append(v)
                    elif modes[v] == modes[u]:
                        print("NO")
                        return
            
    print("YES")
    print(*(modes[1:]))

if __name__ == "__main__":
    solve()
