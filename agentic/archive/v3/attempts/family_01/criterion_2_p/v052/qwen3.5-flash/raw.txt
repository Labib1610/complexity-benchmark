```python
from collections import deque

# Increase recursion depth just in case (though not using recursion)
sys.setrecursionlimit(200005)

def solve():
    # Read all input from standard input
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break
    
    # Array to store the channel assignment for each node
    # 0: unassigned, 1: channel 1, 2: channel 2
    color = [0] * (n + 1)
    
    # Iterate through all nodes to handle disconnected components
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a BFS traversal for the connected component containing i
            # To achieve the lexicographically smallest assignment,
            # we assign the smallest possible value (1) to the node with the smallest index
            # in the component. Since we iterate i from 1 to n, i is the smallest index
            # for the current component.
            
            color[i] = 1
            queue = deque([i])

                u = queue.popleft()
                current_color = color[u]
                # The neighbor must have the opposite color
                neighbor_color = 3 - current_color
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = neighbor_color
                        queue.append(v)
                    elif color[v] == current_color:
                        # Conflict detected: adjacent nodes have the same color
                        # This means the graph is not bipartite
                        print("NO")
                        return
    
    # If we successfully colored all nodes without conflict
    print("YES")
    # Print the assignments for nodes 1 to n
    print(*(color[i] for i in range(1, n + 1)))
```

```python
import sys
from collections import deque

def solve():
    # Read all input from standard input efficiently
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n + 1)]
    
    # Parse m edges
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break
    
    # Color array: 0 = unassigned, 1 = channel 1, 2 = channel 2
    color = [0] * (n + 1)
    
    # Iterate through all nodes to ensure all components are processed
    # This order (1 to n) ensures we pick the lexicographically smallest assignment
    # because for any component, the node with the smallest index will be processed first
    # and assigned the smallest valid color (1).
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS from node i
            # Assign color 1 to the starting node (lexicographically smallest choice)
            color[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_c = color[u]
                next_c = 3 - current_c  # Toggle between 1 and 2
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        queue.append(v)
                    elif color[v] == current_c:
                        # Conflict: two adjacent nodes have the same color
                        print("NO")
                        return
    
    # If valid assignment found
    print("YES")
    # Output the result
    # Using sys.stdout.write for speed with large output
    sys.stdout.write(" ".join(map(str, color[1:])) + "\n")
```
