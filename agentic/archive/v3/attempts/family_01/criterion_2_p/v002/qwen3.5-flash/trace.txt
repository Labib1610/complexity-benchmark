import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))

    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            queue = deque([i])
            while queue:
                u = queue.popleft()
                current_color = colors[u]
                next_color = 3 - current_color
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = next_color
                        queue.append(v)
                    elif colors[v] != next_color:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':

import sys
from collections import deque

# Increase recursion depth just in case, though we use iterative BFS.
sys.setrecursionlimit(200005)

def main():
    # Read all input from stdin at once
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    
    # Parse edges
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store the chamber assignment (1 or 2)
    # 0 indicates unvisited
    chamber = [0] * (n + 1)
    
    # Iterate through all nodes to handle disconnected components
    for i in range(1, n + 1):
        if chamber[i] == 0:
            # Start BFS for the component containing i
            # To ensure lexicographically smallest sequence, assign chamber 1 to the 
            # first node encountered in the component (which is the smallest index in that component)
            chamber[i] = 1
            queue = deque([i])
            
            while queue:
                u = queue.popleft()
                current_chamber = chamber[u]
                next_chamber = 3 - current_chamber  # Toggle between 1 and 2
                
                for v in adj[u]:
                    if chamber[v] == 0:
                        chamber[v] = next_chamber
                        queue.append(v)
                    elif chamber[v] != next_chamber:
                        # Conflict found: graph is not bipartite
                        print("NO")
                        return

    # If no conflicts found, print YES and the assignment
    print("YES")
    # Output the assignments for nodes 1 to n
    print(*(chamber[1:]))

if __name__ == '__main__':
    main()
