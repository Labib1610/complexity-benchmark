```python
import sys

def solve():
    # Read all input from standard input efficiently
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list for the graph
    # Using 1-based indexing for convenience
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    # Array to store the color of each worker
    # 0: uncolored, 1: block 1, 2: block 2
    color = [0] * (n + 1)
    
    # Iterate through each worker from 1 to n
    # This ensures we process components in order of their smallest index
    # which helps in achieving the lexicographically smallest sequence
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start a traversal (DFS) for this component
            # To ensure lexicographically smallest sequence, we assign color 1
            # to the first uncolored node encountered (which is the smallest index
            # in this component).
            color[i] = 1
            stack = [i]
            
            while stack:
                u = stack.pop()
                current_c = color[u]
                next_c = 3 - current_c  # Switch between 1 and 2
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append(v)
                    elif color[v] != next_c:
                        # Conflict detected: graph is not bipartite
                        print("NO")
                        return

    # If we successfully colored all workers
    print("YES")
    # Output the colors for workers 1 to n
    sys.stdout.write(" ".join(map(str, color[1:])) + "\n")

if __name__ == '__main__':
    solve()
