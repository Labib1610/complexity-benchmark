import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(500000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    def dfs(u, color):
        assignment[u] = color
        for v in adj[u]:
            if assignment[v] == 0:
                if not dfs(v, 3 - color):
                    return False
            elif assignment[v] == color:
                return False
        return True

    # To get the lexicographically smallest result, we want 1s as much as possible.
    # We iterate from 1 to n. If a node is unassigned, we try assigning it 1.
    # If that forces a contradiction in its component, we must assign 2.
    # However, this is a bipartite check. Each connected component has two possible
    # colorings (swap 1s and 2s). We pick the one that makes the first node 
    # in the component 1, unless that leads to a lexicographically smaller 
    # arrangement (which is always 1).
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning 1 to the current node i
            # We need to check if this component is valid
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            possible = True
            head = 0
            while head < len(component):
                u = component[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        possible = False
                        break
                if not possible: break
            
            if not possible:
                # If 1 failed, try 2
                for node in component:
                    assignment[node] = 0
                
                assignment[i] = 2
                possible = True
                head = 0
                component = [i]
                while head < len(component):
                    u = component[head]
                    head += 1
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - assignment[u]
                            component.append(v)
                        elif assignment[v] == assignment[u]:
                            possible = False
                            break
                    if not possible: break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
