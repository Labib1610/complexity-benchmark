```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(next(iterator))
        v = int(next(iterator))
        adj[u].append(v)
        adj[v].append(u)
        
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if ans[i] == 0:
            ans[i] = 1
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if ans[v] == 0:
                        ans[v] = 3 - ans[u]
                        q.append(v)
                    elif ans[v] == ans[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(ans[1:]))

if __name__ == '__main__':
    solve()
```

```python
import sys
from collections import deque

# Increase recursion depth just in case, though we use iterative BFS.
sys.setrecursionlimit(200005)

def solve():
    # Read all input from stdin
    input_data = sys.stdin.read().split()
    
    if not input_data:
        return

    iterator = iter(input_data)
    
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    # Adjacency list
    adj = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        try:
            u = int(next(iterator))
            v = int(next(iterator))
            adj[u].append(v)
            adj[v].append(u)
        except StopIteration:
            break
            
    # Array to store the mode for each component
    # 0: unvisited, 1: Mode 1, 2: Mode 2
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if ans[i] == 0:
            # Start BFS for this component
            # We greedily assign Mode 1 to the smallest index node in the component
            ans[i] = 1
            q = deque([i])
            
            while q:
                u = q.popleft()
                
                for v in adj[u]:
                    if ans[v] == 0:
                        # Assign opposite mode
                        ans[v] = 3 - ans[u]
                        q.append(v)
                    elif ans[v] == ans[u]:
                        # Conflict detected
                        print("NO")
                        return
    
    # If we reached here, a valid assignment exists
    print("YES")
    # Print the modes for components 1 to n
    print(*(ans[1:]))

if __name__ == '__main__':
```
