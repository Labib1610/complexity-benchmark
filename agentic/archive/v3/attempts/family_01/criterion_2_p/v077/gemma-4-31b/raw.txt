```python
import sys

# The problem asks for the lexicographically smallest sequence of frequencies (1 or 2)
# for a bipartite graph. Lexicographically smallest means we want to assign '1' 
# to the smallest indices possible.
# For each connected component, we have two choices for coloring (1, 2) or (2, 1).
# To minimize the sequence lexicographically, for each component, we should 
# compare the two possible assignments and pick the one that is lexicographically smaller.
# This is equivalent to checking the smallest index in the component: 
# that index MUST be assigned '1' to minimize the sequence.

def solve():
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2
        
    color = [0] * (n + 1) # 0: unvisited, 1: freq 1, 2: freq 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # New component found. To make it lexicographically smallest,
            # the smallest index in this component should be assigned '1'.
            # Since we iterate i from 1 to n, 'i' is the smallest index in this component.
            stack = [i]
            color[i] = 1
            component = [i]
            
            # Use iterative DFS to avoid recursion depth issues and collect component nodes
            idx = 0
            while idx < len(stack):
                u = stack[idx]
                idx += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring the component starting with color[i] = 1,
            # we check if swapping colors (1 <-> 2) would result in a lexicographically smaller sequence.
            # Since i is the smallest index in the component and we set color[i] = 1,
            # any swap would set color[i] = 2, which is lexicographically larger.
            # Thus, the current assignment for this component is already the optimal one.

    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
