import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We want the lexicographically smallest sequence of modes (1, 2, ..., n).
    # This means for each connected component, we try assigning node i to 1 first.
    # Since it's a bipartite check, once a node is assigned, all others are fixed.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS component
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            # Use a list as a stack for DFS to avoid recursion depth issues
            dfs_stack = [i]
            component.append(i)
            
            while dfs_stack:
                u = dfs_stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        dfs_stack.append(v)
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if this component's lexicographically smallest assignment is fixed
            # Actually, the problem asks for the lexicographically smallest sequence
            # (s1, s2, ..., sn). This means we want s1 to be 1 if possible, then s2, etc.
            # In a bipartite component, there are only two ways to color it.
            # Let the component be fixed by assigning node 'start' = 1 or 2.
            # We compare the two resulting sequences for this component and pick the smaller.
            
            # Re-run for the two possible colorings of the component
            def get_coloring(start_node, start_val):
                res = {}
                res[start_node] = start_val
                q = [start_node]
                while q:
                    u = q.pop()
                    for v in adj[u]:
                        if v not in res:
                            res[v] = 3 - res[u]
                            q.append(v)
                        elif res[v] == res[u]:
                            return None
                return res

            c1 = get_coloring(i, 1)
            c2 = get_coloring(i, 2)
            
            # Compare c1 and c2 lexicographically
            # We only care about the indices in this component
            comp_nodes = sorted(component)
            
            use_c1 = True
            if c1 is not None and c2 is not None:
                for node in comp_nodes:
                    if c1[node] < c2[node]:
                        use_c1 = True
                        break
                    elif c1[node] > c2[node]:
                        use_c1 = False
                        break
            elif c1 is not None:
                use_c1 = True
            else:
                use_c1 = False
                
            final_coloring = c1 if use_c1 else c2
            for node in component:
                assignment[node] = final_coloring[node]
                
    print("YES")
    print(*(assignment[1:]))

solve()
