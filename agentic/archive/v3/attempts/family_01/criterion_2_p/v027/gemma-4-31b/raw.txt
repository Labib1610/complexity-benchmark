import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # color[i] will store the unit (1 or 2). 0 means unvisited.
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment:
    # We iterate through compounds from 1 to n.
    # If a compound isn't assigned yet, it's the start of a new connected component.
    # We want the sequence s_1, s_2, ..., s_n to be lexicographically smallest.
    # This means we want s_i to be 1 if possible.
    # For the first unvisited node i in a component, assigning s_i = 1 is always 
    # better than s_i = 2. Since the relative colors in a bipartite component 
    # are fixed, assigning the root to 1 determines the rest.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i and assign it color 1
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                curr_color = color[u]
                next_color = 2 if curr_color == 1 else 1
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_color
                        stack.append(v)
                    elif color[v] == curr_color:
                        print("NO")
                        return

    # If we reached here, the graph is bipartite.
    print("YES")
    print(*(color[i] for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
