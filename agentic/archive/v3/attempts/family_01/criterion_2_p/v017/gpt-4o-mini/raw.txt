def shift_partitioning(n, m, constraints):
    from collections import defaultdict, deque

    graph = defaultdict(list)
    for u, v in constraints:
        graph[u].append(v)
        graph[v].append(u)

    shifts = [-1] * (n + 1)  # -1 means unassigned

    def bfs(start):
        queue = deque([start])
        shifts[start] = 1  # Start with shift 1
        
        while queue:
            current = queue.popleft()
            current_shift = shifts[current]
            next_shift = 2 if current_shift == 1 else 1
            
            for neighbor in graph[current]:
                if shifts[neighbor] == -1:  # If not assigned
                    shifts[neighbor] = next_shift
                    queue.append(neighbor)
                elif shifts[neighbor] == current_shift:  # Conflict found
                    return False
        return True

    for worker in range(1, n + 1):
        if shifts[worker] == -1:  # Not yet assigned
            if not bfs(worker):
                print("NO")
                return

    print("YES")
    print(" ".join(map(str, shifts[1:])))
    
import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
constraints = [tuple(map(int, line.split())) for line in data[1:m + 1]]
shift_partitioning(n, m, constraints)
