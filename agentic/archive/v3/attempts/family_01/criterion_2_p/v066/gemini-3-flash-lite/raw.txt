import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we process nodes 1 to n.
    # For each connected component, we try assigning color 1 to the first
    # encountered node (which is the smallest node index in that component).
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component not visited, start BFS/DFS
            # We use a stack for iterative DFS to avoid recursion limits
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                # To maintain lexicographical order as much as possible,
                # we sort neighbors, though the constraint is about the
                # overall sequence s_1, s_2, ..., s_n.
                # Actually, simply processing in order 1..n is sufficient.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
