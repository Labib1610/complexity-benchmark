def solve():
    import sys
    from collections import deque
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    graph = [[] for _ in range(n + 1)]
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        graph[u].append(v)
        graph[v].append(u)

    color = [0] * (n + 1)  # 0 means uncolored, 1 and 2 will be the two colors
    possible = True

    def bfs(start):
        nonlocal possible
        queue = deque([start])
        color[start] = 1  # Start coloring with 1

        while queue and possible:
            node = queue.popleft()
            current_color = color[node]
            for neighbor in graph[node]:
                if color[neighbor] == 0:  # Not colored
                    color[neighbor] = 2 if current_color == 1 else 1
                    queue.append(neighbor)
                elif color[neighbor] == current_color:  # Conflict
                    possible = False
                    return

    for i in range(1, n + 1):
        if color[i] == 0:  # Not colored yet
            bfs(i)
            if not possible:
                break

    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, color[1:])))
        
solve()
