def solve():
    import sys
    from collections import defaultdict, deque
    
    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    incompatibilities = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        incompatibilities[u - 1].append(v - 1)
        incompatibilities[v - 1].append(u - 1)
    
    assignment = [-1] * n  # -1 means unassigned
    
    def bfs(start):
        queue = deque([start])
        assignment[start] = 0  # Start assigning shift 1 (0 in 0-index)
        
        while queue:
            current = queue.popleft()
            current_shift = assignment[current]
            for neighbor in incompatibilities[current]:
                if assignment[neighbor] == -1:
                    # Assign the next shift, wrapping around 3
                    assignment[neighbor] = (current_shift + 1) % 3
                    queue.append(neighbor)
                elif assignment[neighbor] == current_shift:
                    return False  # Conflict found
        return True

    for worker in range(n):
        if assignment[worker] == -1:  # Not assigned yet
            if not bfs(worker):
                print("NO")
                return
    
    # Convert to 1-indexed shifts
    shifts = [x + 1 for x in assignment]
    print("YES")
    print(" ".join(map(str, shifts)))
