import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr]) - 1
        v = int(input_data[ptr+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2

    # The problem is 3-coloring a graph with n <= 40.
    # We need the lexicographically smallest sequence.
    # This means for each node i from 0 to n-1, we try colors 1, 2, 3 in order.
    # Since n is small (40), we can use backtracking with pruning.
    
    colors = [0] * n

    def is_safe(v, c):
        for neighbor in adj[v]:
            if colors[neighbor] == c:
                return False
        return True

    # To optimize, we can check if the remaining graph is 3-colorable.
    # However, for n=40, a simple backtracking with a few heuristics 
    # might be too slow. But since we need the LEXICOGRAPHICALLY smallest,
    # we must try colors 1, 2, 3 for node 0, then node 1, etc.
    
    # Forward checking or constraint propagation can help.
    # domain[i] = set of available colors for node i
    domains = [set([1, 2, 3]) for _ in range(n)]

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors in increasing order to get lexicographical smallest
        # The domain for u is already restricted by previous assignments
        available_colors = sorted(list(domains[u]))
        
        for c in available_colors:
            # Check if assigning c to u is consistent with neighbors
            # (Already handled by domain updates, but for safety:)
            if not is_safe(u, c):
                continue
            
            colors[u] = c
            
            # Save domains to restore later
            changed_nodes = []
            possible = True
            
            for v in adj[u]:
                if v > u and c in domains[v]:
                    domains[v].remove(c)
                    changed_nodes.append(v)
                    if not domains[v]:
                        possible = False
                        break
            
            if possible:
                if backtrack(u + 1):
                    return True
            
            # Restore domains
            for v in changed_nodes:
                domains[v].add(c)
            colors[u] = 0
            
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == "__main__":
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    solve()
