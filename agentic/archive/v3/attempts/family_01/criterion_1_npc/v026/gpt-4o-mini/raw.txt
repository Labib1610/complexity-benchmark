def spectrum_division(n, m, edges):
    from collections import defaultdict
    import sys
    input = sys.stdin.read
    data = input().splitlines()
    
    # Build the graph
    graph = defaultdict(list)
    for i in range(m):
        u, v = map(int, data[i + 2].split())
        graph[u].append(v)
        graph[v].append(u)

    # To store the frequencies of each unit
    freq = [0] * (n + 1)

    def can_color(node):
        # Try to color the current node with the smallest possible color
        used = [False] * 4  # We have 3 colors, index 0 is unused
        for neighbor in graph[node]:
            if freq[neighbor] > 0:
                used[freq[neighbor]] = True
        
        # Assign the smallest available color
        for color in range(1, 4):
            if not used[color]:
                freq[node] = color
                return True
        
        return False

    # Try to color each node
    for i in range(1, n + 1):
        if freq[i] == 0:  # Not colored yet
            if not can_color(i):
                print("NO")
                return

    print("YES")
    print(" ".join(map(str, freq[1:])))

import sys

data = input().splitlines()
n, m = map(int, data[0].split())
edges = [tuple(map(int, line.split())) for line in data[1:m + 1]]

spectrum_division(n, m, edges)
