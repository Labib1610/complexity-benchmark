import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(person_idx):
        if person_idx == n:
            return True
        
        # Try assigning window 1, 2, 3 in order for lexicographical smallest
        for window in [1, 2, 3]:
            can_assign = True
            # Check constraints with already assigned people
            for prev_idx in range(person_idx):
                if (adj[person_idx + 1] & (1 << prev_idx)):
                    if assignment[prev_idx] == window:
                        can_assign = False
                        break
            
            if can_assign:
                assignment[person_idx] = window
                if backtrack(person_idx + 1):
                    return True
                assignment[person_idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
