```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    shifts = [0] * (n + 1)
    
    # We need to find the lexicographically smallest assignment.
    # Because n is small (40), we can use backtracking with pruning.
    # The requirement "each worker assigned to the lowest numbered shift 
    # that produces a globally valid schedule" implies we check greedily.
    
    def can_complete(u):
        if u > n:
            return True
        
        # Try shifts 1, 2, 3
        for s in range(1, 4):
            # Check if shift s is valid for worker u
            valid = True
            for neighbor in adj[u]:
                if neighbor < u and shifts[neighbor] == s:
                    valid = False
                    break
            
            if valid:
                shifts[u] = s
                if can_complete(u + 1):
                    return True
                shifts[u] = 0
        return False

    if can_complete(1):
        print("YES")
        print(*(shifts[1:]))
    else:
        print("NO")

solve()
```
