```python
import sys

def solve():
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    ptr = 2
    for _ in range(m):
        u = int(input_data[ptr])
        v = int(input_data[ptr+1])
        adj[u].append(v)
        adj[v].append(u)
        ptr += 2

    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in increasing order to ensure lexicographical smallest result
        for color in (1, 2, 3):
            if is_valid(u, color):
                assignment[u] = color
                # Heuristic check: can the remaining graph be 3-colored?
                # Since n is small (40), simple backtracking with pruning or 
                # just straightforward backtracking might be too slow for some cases,
                # but for n=40, the search space is reduced significantly by constraints.
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # For n=40, simple backtracking can be O(3^n), which is too slow.
    # However, 3-coloring is NP-complete. With n=40, we need a bit more efficiency.
    # We use a simple backtracking with the order of colors 1, 2, 3.
    # To speed up, we can check if the remaining graph is still 3-colorable, 
    # but that's as hard as the original problem.
    # Let's use a basic backtracking first; for n=40, many graphs are pruned quickly.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
