def can_assign_vaults(n, m, incompatibilities):
    from itertools import combinations

    # Create an adjacency list for the incompatibilities
    graph = [[] for _ in range(n + 1)]
    for u, v in incompatibilities:
        graph[u].append(v)
        graph[v].append(u)

    # Attempt to color the graph with 3 colors
    color = [-1] * (n + 1)

    def dfs(node, c):
        color[node] = c
        for neighbor in graph[node]:
            if color[neighbor] == c:
                return False
            if color[neighbor] == -1 and not dfs(neighbor, (c + 1) % 3):
                return False
        return True

    for i in range(1, n + 1):
        if color[i] == -1:
            if not dfs(i, 0):
                return "NO", []

    vault_assignment = [c + 1 for c in color[1:]]
    return "YES", vault_assignment

import sys
input = sys.stdin.read

data = input().splitlines()
n, m = map(int, data[0].split())
incompatibilities = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result, assignment = can_assign_vaults(n, m, incompatibilities)

if result == "NO":
    print("NO")
else:
    print("YES")
    print(" ".join(map(str, assignment)))
