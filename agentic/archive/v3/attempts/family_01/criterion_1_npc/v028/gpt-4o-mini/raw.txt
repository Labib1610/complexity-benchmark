def main():
    import sys
    from collections import defaultdict, deque

    input = sys.stdin.read
    data = input().splitlines()
    
    n, m = map(int, data[0].split())
    incompatibilities = defaultdict(list)
    
    for i in range(1, m + 1):
        u, v = map(int, data[i].split())
        incompatibilities[u].append(v)
        incompatibilities[v].append(u)

    # Colors for vaults: 1, 2, 3
    color = [0] * (n + 1)

    def bfs(start):
        queue = deque([start])
        color[start] = 1 # Start coloring with 1
        
        while queue:
            node = queue.popleft()
            next_color = 1 if color[node] != 1 else 2
            
            for neighbor in incompatibilities[node]:
                if color[neighbor] == 0:  # Not colored yet
                    color[neighbor] = next_color
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:  # Conflict
                    return False
            
            # Adjust next_color to cycle through 1, 2, 3
            next_color = 3 if next_color == 2 else 1
        
        return True

    for i in range(1, n + 1):
        if color[i] == 0:  # Not colored
            if not bfs(i):
                print("NO")
                return

    # Adjusting colors to make sure we have at most 3 vaults and lexicographically smallest
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1  # Assign to vault 1 by default

    # Output the result
    print("YES")
    print(' '.join(map(str, color[1:])))
    
if __name__ == "__main__":
    main()
