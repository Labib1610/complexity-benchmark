import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    assignment = [0] * (n + 1)

    def is_safe(node, color):
        for neighbor in adj[node]:
            if assignment[neighbor] == color:
                return False
        return True

    def can_color(node):
        """
        Check if the remaining graph can be 3-colored given the current assignments.
        Since n is small (up to 40), we can use backtracking with pruning.
        """
        if node > n:
            return True
        
        # Try colors 1, 2, 3 for the current node
        for color in range(1, 4):
            if is_safe(node, color):
                assignment[node] = color
                if can_color(node + 1):
                    return True
                assignment[node] = 0
        return False

    # To find the lexicographically smallest sequence, we greedily try 
    # assigning the smallest possible color (1, 2, 3) to core i, 
    # and then check if a valid 3-coloring exists for the remaining cores.
    
    # We'll use a more efficient check for the remaining graph.
    # Because n is small, we can use a simple backtracking with a bit of optimization.
    
    def check_validity(current_node):
        # Temporary copy to avoid mutating the main assignment during the check
        temp_assignment = list(assignment)
        
        def backtrack(node):
            if node > n:
                return True
            
            # If already assigned (by the greedy process), move to next
            if temp_assignment[node] != 0:
                return backtrack(node + 1)
            
            # Try colors 1, 2, 3
            for color in range(1, 4):
                safe = True
                for neighbor in adj[node]:
                    if temp_assignment[neighbor] == color:
                        safe = False
                        break
                if safe:
                    temp_assignment[node] = color
                    if backtrack(node + 1):
                        return True
                    temp_assignment[node] = 0
            return False
        
        return backtrack(current_node + 1)

    # Greedy construction for lexicographical smallest
    for i in range(1, n + 1):
        found = False
        for color in range(1, 4):
            if is_safe(i, color):
                assignment[i] = color
                # Check if this choice allows for a valid completion
                if check_validity(i):
                    found = True
                    break
                assignment[i] = 0
        
        if not found:
            print("NO")
            return

    print("YES")
    print(*(assignment[1:]))

if __name__ == "__main__":
    # Increase recursion depth for backtracking
    sys.setrecursionlimit(2000)
    solve()
