import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # colors[i] = 0 (unassigned), 1, 2, or 3
    colors = [0] * (n + 1)

    def is_safe(v, c):
        for neighbor in adj[v]:
            if colors[neighbor] == c:
                return False
        return True

    def can_color(node_idx):
        """
        Checks if the remaining nodes can be colored with 3 colors.
        Since n is small (40), we can use a backtracking approach with 
        some basic pruning or just a simple DFS.
        """
        if node_idx > n:
            return True
        
        # Try colors 1, 2, 3 for the current node
        for c in (1, 2, 3):
            if is_safe(node_idx, c):
                colors[node_idx] = c
                if can_color(node_idx + 1):
                    return True
                colors[node_idx] = 0
        return False

    # To find the lexicographically smallest sequence, we try to assign 
    # the smallest possible color to transmitter 1, then transmitter 2, etc.
    # For each assignment, we must check if a valid 3-coloring still exists 
    # for the remaining nodes.
    
    # Because n=40 is small but potentially too large for raw backtracking
    # in every step, we use a slightly more optimized check.
    
    def check_feasibility(current_node):
        # Simple backtracking to check if a valid coloring exists for nodes current_node...n
        # given the fixed colors of nodes 1...current_node-1.
        
        # To speed up, we can use a simple heuristic or just rely on the small n.
        # However, a raw DFS might be too slow if called n*3 times.
        # Let's implement a basic DFS.
        
        temp_colors = list(colors)
        
        def dfs(u):
            if u > n:
                return True
            
            # Optimization: if u is already colored (shouldn't happen for u > current_node), skip
            # But here we only call dfs for u from current_node + 1 to n.
            
            # Try colors in any order, we just need to know if ANY exists.
            for c in (1, 2, 3):
                safe = True
                for v in adj[u]:
                    if temp_colors[v] == c:
                        safe = False
                        break
                if safe:
                    temp_colors[u] = c
                    if dfs(u + 1):
                        return True
                    temp_colors[u] = 0
            return False
        
        return dfs(current_node + 1)

    # Greedy assignment with feasibility check
    for i in range(1, n + 1):
        found = False
        for c in (1, 2, 3):
            if is_safe(i, c):
                colors[i] = c
                # Check if this choice allows a valid completion
                # We need to check if nodes (i+1 ... n) can be colored.
                if check_feasibility(i):
                    found = True
                    break
                colors[i] = 0
        if not found:
            print("NO")
            return

    print("YES")
    print(*(colors[1:]))

if __name__ == "__main__":
    # Increase recursion depth for DFS
    sys.setrecursionlimit(2000)
    solve()
