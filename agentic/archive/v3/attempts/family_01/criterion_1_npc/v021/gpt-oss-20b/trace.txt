def dfs(pos):
    if pos>n: return True
    # compute available colors for pos
    for c in 1..3:
        conflict = False
        for nb in adj[pos]:
            if nb < pos and color[nb] == c:
                conflict = True; break
        if conflict: continue
        # assign
        color[pos] = c
        # update usedMask for neighbors > pos
        changes = []
        for nb in adj[pos]:
            if nb > pos:
                prev = usedMask[nb]
                usedMask[nb] |= (1 << (c-1))
                changes.append((nb, prev))
        # check forward feasibility
        feasible = True
        for j in range(pos+1, n+1):
            # if usedMask[j] == 0b111: no colors left
            if usedMask[j] == 0b111:
                feasible = False
                break
        if feasible:
            if dfs(pos+1): return True
        # revert
        for nb, prev in changes:
            usedMask[nb] = prev
        color[pos] = 0
    return False
