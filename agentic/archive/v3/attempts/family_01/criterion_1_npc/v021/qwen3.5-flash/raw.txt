    colors = [0] * n 
    # forbidden: bitmask of forbidden colors. 1<<0, 1<<1, 1<<2 for colors 1, 2, 3
    forbidden = [0] * n

    # We assign colors to 0, 1, ..., n-1.

        if forbidden[u] == 7:

                continue

            possible = True
