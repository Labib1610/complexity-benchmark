import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # We need the lexicographically smallest sequence (s1, s2, ..., sn).
    # We can try to assign values 1, 2, 3 to assets 0, 1, ..., n-1 greedily.
    # Since n is small (up to 40), we can use backtracking with pruning.
    # To optimize, we can check if the remaining graph is 3-colorable, 
    # but 3-coloring is NP-complete. However, with n=40, a well-pruned 
    # backtracking search should be sufficient.
    
    res = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if v < u and res[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3 in order for lexicographical smallest result
        for color in (1, 2, 3):
            if is_valid(u, color):
                res[u] = color
                # Forward checking or simple backtracking
                # Given n=40, simple backtracking might be slow, 
                # but since we only need the first valid sequence, 
                # and we try colors in increasing order, it's the best we can do.
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    # To speed up backtracking for n=40, we can use a slightly more 
    # sophisticated approach if needed, but let's start with this.
    # Actually, 3-coloring for n=40 can be very slow. 
    # Let's add a simple pruning: check if any unassigned node has no available colors.
    
    def backtrack_with_pruning(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3
        for color in (1, 2, 3):
            if is_valid(u, color):
                res[u] = color
                
                # Pruning: check if any future node i > u has no available colors
                possible = True
                for i in range(u + 1, n):
                    # Check colors available for node i
                    # A color is available if no neighbor j < i has that color
                    # (Wait, this only checks neighbors already assigned)
                    # A better check: if node i has neighbors assigned to 1, 2, and 3, it's impossible.
                    used_colors = set()
                    for v in adj[i]:
                        if v <= u:
                            used_colors.add(res[v])
                    if len(used_colors) == 3:
                        possible = False
                        break
                
                if possible and backtrack_with_pruning(u + 1):
                    return True
                res[u] = 0
        return False

    if backtrack_with_pruning(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

if __name__ == "__main__":
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(2000)
    solve()
