def can_schedule(n, m, restrictions):
    from itertools import combinations
    from collections import defaultdict

    # Create an adjacency list for restrictions
    graph = defaultdict(set)
    for u, v in restrictions:
        graph[u].add(v)
        graph[v].add(u)

    # Initialize time slots assignment
    slots = [0] * (n + 1)  # slots[i] = 0 means unassigned

    def backtrack(person):
        if person > n:
            return True  # All personnel assigned

        for slot in range(1, 4):  # Try slots 1, 2, 3
            valid = True
            for neighbor in graph[person]:
                if slots[neighbor] == slot:  # Check for conflicts
                    valid = False
                    break
            if valid:
                slots[person] = slot  # Assign the slot
                if backtrack(person + 1):  # Recur for next person
                    return True
                slots[person] = 0  # Backtrack

        return False  # No valid assignment found

    if backtrack(1):
        print("YES")
        print(" ".join(map(str, slots[1:])))
    else:
        print("NO")


import sys
input = sys.stdin.read
data = input().splitlines()
n, m = map(int, data[0].split())
restrictions = [tuple(map(int, line.split())) for line in data[1:m+1]]
can_schedule(n, m, restrictions)
