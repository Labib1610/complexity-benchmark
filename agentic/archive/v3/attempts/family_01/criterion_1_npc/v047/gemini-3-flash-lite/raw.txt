import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition s_u * s_v in {2, 3, 6} means s_u * s_v != 1, 4, 9.
    # Since s_i \in {1, 2, 3}:
    # If s_u = 1, then s_v cannot be 1.
    # If s_u = 2, then s_v cannot be 2.
    # If s_u = 3, then s_v cannot be 3.
    # Effectively, s_u != s_v.
    # This is simply a graph coloring problem with 3 colors.
    
    assignment = [0] * (n + 1)
    
    def is_valid(worker, color):
        for neighbor in adj[worker]:
            if assignment[neighbor] == color:
                return False
        return True
    
    def backtrack(worker):
        if worker > n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(worker, color):
                assignment[worker] = color
                if backtrack(worker + 1):
                    return True
                assignment[worker] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
