def can_assign_technicians(n, m, pairs):
    from collections import defaultdict
    
    # Create a graph of incompatible technicians
    graph = defaultdict(set)
    for u, v in pairs:
        graph[u].add(v)
        graph[v].add(u)
    
    # To store the assigned windows
    assignment = [0] * (n + 1)
    
    # Function to perform DFS and try to assign windows
    def dfs(tech):
        if tech > n:
            return True
        for window in range(1, 4):
            # Check if the current technician can be assigned to this window
            if all(assignment[neighbor] != window for neighbor in graph[tech]):
                assignment[tech] = window
                if dfs(tech + 1):
                    return True
                assignment[tech] = 0  # Backtrack
        return False
    
    if dfs(1):
        return "YES", assignment[1:]  # Omit the zero index
    else:
        return "NO", []

# Read input
import sys
input = sys.stdin.read
data = input().splitlines()

first_line = data[0].split()
n = int(first_line[0])
m = int(first_line[1])

pairs = [tuple(map(int, line.split())) for line in data[1:m + 1]]

result = can_assign_technicians(n, m, pairs)

if result[0] == "NO":
    print(result[0])
else:
    print(result[0])
    print(" ".join(map(str, result[1])))
