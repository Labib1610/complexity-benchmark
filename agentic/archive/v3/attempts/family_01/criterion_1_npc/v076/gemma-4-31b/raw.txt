import sys

def solve():
    # Increase recursion depth for deep backtracking if necessary
    sys.setrecursionlimit(2000)
    
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest sequence, we try assigning 
    # channel 1, then 2, then 3 for unit 1, then unit 2, and so on.
    # Since n is small (up to 40), backtracking with pruning should work.
    
    res = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try channels 1, 2, 3 in order for lexicographical smallest
        for color in (1, 2, 3):
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    # Optimization: 3-coloring is NP-complete, but for n=40, simple backtracking
    # might be slow. However, the lexicographical requirement forces a specific
    # order. Let's use basic backtracking first.
    if backtrack(1):
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
