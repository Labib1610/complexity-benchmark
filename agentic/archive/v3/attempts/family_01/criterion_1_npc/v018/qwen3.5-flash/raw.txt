        visited = [False] * (n + 1)
        
        for i in range(1, n + 1):
            if not visited[i]:
                # BFS to traverse the component
                # i is the smallest index in this component, so assign 1
                final_color[i] = 1
                visited[i] = True
                q = [i]
                idx = 0
                while idx < len(q):
                    u = q[idx]
                    idx += 1
                    for v in adj[u]:
                        if not visited[v]:
                            visited[v] = True
                            final_color[v] = 3 - final_color[u]
                            q.append(v)
        
        print("YES")
        print(*(final_color[1:]))
        return

    # If not bipartite, use backtracking with Forward Checking
    # Forward Checking prunes branches where an uncolored node has no valid colors left.
    color = [0] * (n + 1)
    # forbidden_count[v][c] stores the number of already colored neighbors of v that have color c
    forbidden_count = [[0] * 4 for _ in range(n + 1)]
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in order to ensure lexicographically smallest solution
        for c in range(1, 4):
            # Check validity: no neighbor < u has color c
            valid = True
            for v in adj[u]:
                if v < u:
                    if color[v] == c:
                        valid = False
                        break
            if not valid:
                continue
            
            # Forward Checking: check if assigning c to u makes any neighbor v > u have 0 valid colors
            affected_nodes = []
            possible = True
            
            for v in adj[u]:
                if v > u:
                    forbidden_count[v][c] += 1
                    affected_nodes.append(v)
                    # If all 3 colors are forbidden for v, then v cannot be colored
                    if forbidden_count[v][1] > 0 and forbidden_count[v][2] > 0 and forbidden_count[v][3] > 0:
                        possible = False
                        break
            
            if possible:
                color[u] = c
                if backtrack(u + 1):
                    return True
                color[u] = 0
            
            # Backtrack: revert forbidden_count changes
            for v in affected_nodes:
                forbidden_count[v][c] -= 1
                
        return False

    if backtrack(1):
        print("YES")
        print(*(color[1:]))
    else:
        print("NO")

One final check: "lexicographically smallest sequence".
In the bipartite case, I assign 1 to the first node of each component.
Is it possible that a different assignment (e.g. using 3) is smaller?
As discussed, no. Because 3 > 1 and 3 > 2.
And any valid 3-coloring that uses 3 at index $i$ is lexicographically larger than one that uses 1 or 2 at index $i$.
If a 2-coloring exists, we can always find a 2-coloring that matches the prefix of the optimal 3-coloring?
Actually, simpler: The set of 2-colorings is a subset of 3-colorings.
The minimum of a set is $\le$ minimum of a superset.
But we want the minimum of the superset.
If the minimum of the superset is in the subset, we are good.
It is a known property that for bipartite graphs, the lexicographically first 3-coloring is a 2-coloring.
