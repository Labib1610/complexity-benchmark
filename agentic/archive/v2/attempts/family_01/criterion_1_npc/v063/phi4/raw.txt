def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    edges = []
    index = 2
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u - 1, v - 1))
        index += 2

    from itertools import permutations
    
    def is_valid(assignment):
        for (u, v) in edges:
            if assignment[u] == assignment[v]:
                return False
        return True
    
    min_assignment = None
    
    # Generate all possible assignments for n units to 3 channels
    for perm in permutations(range(1, 4), n):
        if is_valid(perm):
            if min_assignment is None or perm < min_assignment:
                min_assignment = perm
    
    if min_assignment:
        print("YES")
        print(' '.join(map(str, min_assignment)))
    else:
        print("NO")

# The code block below will be executed when the script is run.
if __name__ == "__main__":
    solve()
