import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)  # 0 unassigned, 1..3 colors
    solution = []

    # Precompute neighbor list for quick check
    neigh = adj

    def is_feasible_uncolored() -> bool:
        # For each uncolored node, ensure at least one color available
        for v in range(1, n + 1):
            if color[v] == 0:
                # Check availability of colors
                for c in (1, 2, 3):
                    ok = True
                    for u in neigh[v]:
                        if color[u] == c:
                            ok = False
                            break
                    if ok:
                        break
                else:  # no color possible
                    return False
        return True

    def dfs(pos: int) -> bool:
        if pos > n:
            return True
        v = pos
        for c in (1, 2, 3):
            conflict = False
            for u in neigh[v]:
                if color[u] == c:
                    conflict = True
                    break
            if conflict:
                continue
            color[v] = c
            if is_feasible_uncolored():
                if dfs(pos + 1):
                    return True
            color[v] = 0
        return False

    if dfs(1):
        out = ["YES"]
        out.append(" ".join(str(color[i]) for i in range(1, n + 1)))
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
