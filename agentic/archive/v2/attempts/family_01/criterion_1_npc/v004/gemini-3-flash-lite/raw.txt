```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(worker):
        if worker > n:
            return True
        
        # Try shifts 1, 2, 3 in order to find the lexicographically smallest
        for shift in range(1, 4):
            can_assign = True
            for neighbor in adj[worker]:
                if neighbor < worker and assignment[neighbor] == shift:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[worker] = shift
                if backtrack(worker + 1):
                    return True
                assignment[worker] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
