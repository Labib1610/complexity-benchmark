import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    allowed = [0b111 for _ in range(n)]  # bits for colors 1,2,3
    assigned = [0] * n  # 0 unassigned, else 1..3
    found = False

    def dfs(pos: int) -> bool:
        nonlocal found
        if pos == n:
            found = True
            return True
        for color in (1, 2, 3):
            bit = 1 << (color - 1)
            # check if any assigned neighbor has same color
            conflict = False
            for nb in adj[pos]:
                if assigned[nb] == color:
                    conflict = True
                    break
            if conflict:
                continue
            # apply forward checking
            changed = []
            ok = True
            for nb in adj[pos]:
                if assigned[nb] == 0 and (allowed[nb] & bit):
                    allowed[nb] ^= bit
                    changed.append(nb)
                    if allowed[nb] == 0:
                        ok = False
                        break
            if not ok:
                # revert
                for nb in changed:
                    allowed[nb] |= bit
                continue
            assigned[pos] = color
            if dfs(pos + 1):
                return True
            assigned[pos] = 0
            # revert allowed
            for nb in changed:
                allowed[nb] |= bit
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, assigned)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
