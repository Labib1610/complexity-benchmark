```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    colors = [0] * n
    color_mask = [0, 0, 0]  # bitmask of vertices colored with vault 1,2,3

    sys.setrecursionlimit(1000000)
    found = False

    def dfs(pos: int) -> bool:
        nonlocal found
        if pos == n:
            return True
        for c in range(3):
            if adj[pos] & color_mask[c]:
                continue  # conflict
            # assign
            colors[pos] = c + 1
            color_mask[c] |= 1 << pos
            if dfs(pos + 1):
                return True
            # backtrack
            color_mask[c] ^= 1 << pos
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
