```python
import sys
sys.setrecursionlimit(1000000)

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n  # 0 = unassigned, 1/2/3 = color

    def dfs(i):
        if i == n:
            return True
        for c in (1, 2, 3):
            conflict = False
            for nb in adj[i]:
                if colors[nb] == c:
                    conflict = True
                    break
            if not conflict:
                colors[i] = c
                if dfs(i + 1):
                    return True
                colors[i] = 0
        return False

    if dfs(0):
        out_lines = ["YES", " ".join(str(colors[i]) for i in range(n))]
        sys.stdout.write("\n".join(out_lines))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    main()
```
