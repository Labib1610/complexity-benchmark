import sys

def can_partition(n, m, incompatible_pairs):
    # Create an adjacency list for the graph of incompatibilities
    adj = [[] for _ in range(n)]
    for u, v in incompatible_pairs:
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # Try to assign vaults 1, 2, or 3 to each substance
    def is_valid(assignment):
        for i in range(n):
            if assignment[i] == 0:
                continue
            for neighbor in adj[i]:
                if assignment[neighbor] == assignment[i]:
                    return False
        return True

    # Recursive function to try all assignments and find the lexicographically smallest one
    def backtrack(index, assignment):
        nonlocal found_assignment
        if index == n:
            if is_valid(assignment):
                found_assignment = assignment[:]
            return
        
        for vault in range(1, 4):
            assignment[index] = vault
            if is_valid(assignment[:index+1]):
                backtrack(index + 1, assignment)
            assignment[index] = 0

    found_assignment = None
    backtrack(0, [0] * n)

    if found_assignment:
        print("YES")
        print(' '.join(map(str, found_assignment)))
    else:
        print("NO")

# Read input from standard input
input_data = sys.stdin.read().strip().split()
n, m = int(input_data[0]), int(input_data[1])
incompatible_pairs = []
index = 2
for _ in range(m):
    u, v = int(input_data[index]), int(input_data[index+1])
    incompatible_pairs.append((u, v))
    index += 2

can_partition(n, m, incompatible_pairs)
