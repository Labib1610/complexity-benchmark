import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(emp_idx):
        if emp_idx == n:
            return True
        
        # Try shifts 1, 2, 3 in order to ensure lexicographically smallest
        for shift in [1, 2, 3]:
            can_assign = True
            for prev in range(emp_idx):
                if (adj[emp_idx + 1] & (1 << prev)):
                    if assignment[prev] == shift:
                        can_assign = False
                        break
            
            if can_assign:
                assignment[emp_idx] = shift
                if backtrack(emp_idx + 1):
                    return True
                assignment[emp_idx] = 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
