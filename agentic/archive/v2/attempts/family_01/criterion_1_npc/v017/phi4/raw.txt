from sys import stdin

def main():
    def dfs(node):
        if node == n + 1:
            return True
        for shift in range(1, 4):
            valid = True
            for neighbor in graph[node]:
                if shifts[neighbor] == shift:
                    valid = False
                    break
            if valid:
                shifts[node] = shift
                if dfs(node + 1):
                    return True
        if shifts[node] != -1:
            shifts[node] = -1
        return False

    n, m = map(int, stdin.readline().split())
    graph = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, stdin.readline().split())
        graph[u].append(v)
        graph[v].append(u)

    shifts = [-1] * (n + 1)
    
    if dfs(1):
        print("YES")
        print(' '.join(map(str, shifts[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
