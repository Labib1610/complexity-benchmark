import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neigh = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        neigh[u].append(v)
        neigh[v].append(u)

    col = [0] * n           # 0 = unassigned, else 1..3
    allowed = [7] * n       # bitmask of allowed colors (bits 0..2)

    found = False
    result = [0] * n

    def dfs(idx: int) -> bool:
        nonlocal found
        if idx == n:
            result[:] = col[:]
            found = True
            return True
        i = idx
        for c in (1, 2, 3):
            # conflict with already colored neighbors
            conflict = False
            for j in neigh[i]:
                if col[j] == c:
                    conflict = True
                    break
            if conflict:
                continue
            col[i] = c
            changed = []
            ok = True
            bit = 1 << (c - 1)
            for j in neigh[i]:
                if col[j] == 0:
                    old = allowed[j]
                    new = old & ~bit
                    if new == 0:
                        ok = False
                    allowed[j] = new
                    changed.append((j, old))
            if ok:
                if dfs(idx + 1):
                    return True
            # revert changes
            for j, old in changed:
                allowed[j] = old
            col[i] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, result)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
