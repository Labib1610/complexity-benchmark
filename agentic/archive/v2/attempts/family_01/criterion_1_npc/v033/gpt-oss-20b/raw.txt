import sys

def main():
    sys.setrecursionlimit(10000)
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # availability bitmask: bit 0 -> color 1, bit 1 -> color 2, bit 2 -> color 3
    avail = [0b111] * n
    color = [0] * n  # 0 means uncolored, otherwise 1/2/3

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        # try colors in increasing order to respect lexicographic order
        for col in (1, 2, 3):
            bit = 1 << (col - 1)
            if not (avail[idx] & bit):
                continue  # this color not allowed
            color[idx] = col
            changed = []
            ok = True
            for nb in adj[idx]:
                if color[nb] == 0:  # uncolored neighbor
                    prev = avail[nb]
                    new = prev & ~bit
                    if new == 0:
                        ok = False
                    avail[nb] = new
                    changed.append((nb, prev))
            if ok and dfs(idx + 1):
                return True
            # backtrack
            for nb, prev in changed:
                avail[nb] = prev
            color[idx] = 0
        return False

    if dfs(0):
        out = ['YES'] + [' '.join(map(str, color))]
        sys.stdout.write('\n'.join(out))
    else:
        sys.stdout.write('NO')

if __name__ == "__main__":
    main()
