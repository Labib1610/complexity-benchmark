```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_safe(u, channel):
        for v in adj[u]:
            if assignment[v] == channel:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for channel in range(1, 4):
            if is_safe(u, channel):
                assignment[u] = channel
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem asks for the lexicographically smallest sequence (s_1, s_2, ..., s_n).
    # Since n is small (up to 40), and we need lexicographical order,
    # the greedy approach with backtracking works perfectly.
    # At each step u from 1 to n, we try channel 1, then 2, then 3.
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
