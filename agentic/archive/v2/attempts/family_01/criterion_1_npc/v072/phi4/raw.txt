```python
from sys import stdin

# Reading input
input = stdin.read
data = input().split()
n = int(data[0])
m = int(data[1])

# Initialize adjacency list for restricted pairs
restricted_pairs = [[] for _ in range(n + 1)]
index = 2
for i in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    restricted_pairs[u].append(v)
    restricted_pairs[v].append(u)
    index += 2

# Initialize time slot assignments and their counts
time_slots = [0] * (n + 1)
slot_count = [0, 0, 0]

def is_valid_assignment(worker, slot):
    for neighbor in restricted_pairs[worker]:
        if time_slots[neighbor] == slot:
            return False
    return True

# Depth First Search to assign time slots
def dfs(worker):
    if worker > n:
        print("YES")
        print(' '.join(map(str, time_slots[1:])))
        exit(0)
    
    for slot in range(1, 4):
        if is_valid_assignment(worker, slot) and (slot_count[slot] < len(restricted_pairs[worker]) + 1):
            time_slots[worker] = slot
            slot_count[slot] += 1
            
            dfs(worker + 1)
            
            # Backtrack
            time_slots[worker] = 0
            slot_count[slot] -= 1

    return False

# Start DFS from the first worker
if not dfs(1):
    print("NO")
```
