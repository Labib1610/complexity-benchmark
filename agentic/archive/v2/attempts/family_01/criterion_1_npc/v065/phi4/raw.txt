from itertools import combinations

def hazardous_reagent_distribution():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    edges = []
    index = 2
    
    for _ in range(m):
        u, v = int(data[index]), int(data[index + 1])
        edges.append((u - 1, v - 1))
        index += 2

    adj_list = [[] for _ in range(n)]
    
    for u, v in edges:
        adj_list[u].append(v)
        adj_list[v].append(u)

    def is_valid(assignment):
        for (a, b) in combinations(range(n), 2):
            if assignment[a] == assignment[b] and b in adj_list[a]:
                return False
        return True

    def backtrack(index, assignment):
        if index == n:
            return assignment
        
        for unit in range(1, 4):
            assignment[index] = unit
            if is_valid(assignment):
                result = backtrack(index + 1, assignment)
                if result is not None:
                    return result
            assignment[index] = 0

        return None

    assignment = [0] * n
    result = backtrack(0, assignment)

    if result is None:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, result)))

if __name__ == "__main__":
    hazardous_reagent_distribution()
