import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We need to find the lexicographically smallest assignment.
    # We can use backtracking to fill assignment[1] to assignment[n].
    # Since n <= 40, we need to be careful with efficiency.
    # However, 3^40 is too large, but the constraints on the graph
    # (m edges) prune the search space significantly.
    
    def backtrack(k):
        if k > n:
            return True
        
        # Try vault 1, then 2, then 3 to ensure lexicographical order
        for vault in range(1, 4):
            possible = True
            for neighbor in adj[k]:
                if assignment[neighbor] == vault:
                    possible = False
                    break
            
            if possible:
                assignment[k] = vault
                if backtrack(k + 1):
                    return True
                assignment[k] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
