import sys
sys.setrecursionlimit(10000)

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neighbors = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        neighbors[u].append(v)
        neighbors[v].append(u)

    colors = [0] * n           # 0 = unassigned, 1/2/3 assigned
    allowed = [7] * n          # bitmask of allowed colors (bits 0-2)

    solution = [0] * n
    found = False

    def dfs(pos):
        nonlocal found
        if found:
            return
        if pos == n:
            for i in range(n):
                solution[i] = colors[i]
            found = True
            return

        i = pos
        if allowed[i] == 0:
            return

        for c in (1, 2, 3):
            if not (allowed[i] & (1 << (c - 1))):
                continue
            conflict = False
            for j in neighbors[i]:
                if colors[j] == c:
                    conflict = True
                    break
            if conflict:
                continue

            colors[i] = c
            changed = []
            ok = True
            for j in neighbors[i]:
                if colors[j] == 0:
                    old = allowed[j]
                    new = old & ~(1 << (c - 1))
                    if new == 0:
                        ok = False
                    else:
                        allowed[j] = new
                    changed.append((j, old))
            if ok:
                # Quick check for any uncolored node with no colors left
                for k in range(pos + 1, n):
                    if colors[k] == 0 and allowed[k] == 0:
                        ok = False
                        break
            if ok:
                dfs(pos + 1)
                if found:
                    return
            # revert
            for j, old in changed:
                allowed[j] = old
            colors[i] = 0

    dfs(0)

    out = sys.stdout
    if not found:
        out.write("NO\n")
    else:
        out.write("YES\n")
        out.write(" ".join(str(solution[i]) for i in range(n)))
        out.write("\n")

if __name__ == "__main__":
    main()
