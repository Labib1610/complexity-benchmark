```python
import sys

def main():
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = data[idx], data[idx + 1]
        idx += 2
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * (n + 1)

    sys.setrecursionlimit(10000)

    def dfs(i):
        if i > n:
            return True
        for c in (1, 2, 3):
            conflict = False
            for nb in adj[i]:
                if nb < i and colors[nb] == c:
                    conflict = True
                    break
            if conflict:
                continue
            colors[i] = c
            if dfs(i + 1):
                return True
            colors[i] = 0
        return False

    if dfs(1):
        out = ["YES", " ".join(str(colors[i]) for i in range(1, n + 1))]
    else:
        out = ["NO"]
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    main()
```
