def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    incompatible = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(data[idx]), int(data[idx + 1])
        incompatible[u].append(v)
        incompatible[v].append(u)
        idx += 2

    def is_valid(assignment):
        for i in range(1, n + 1):
            block_i = assignment[i - 1]
            for j in incompatible[i]:
                if assignment[j - 1] == block_i:
                    return False
        return True

    from itertools import permutations
    
    blocks = [1, 2, 3]
    best_assignment = None
    
    for perm in permutations(blocks * (n // 3 + 1), n):
        if is_valid(perm):
            best_assignment = perm[:n]
            break
    
    if best_assignment:
        print("YES")
        print(*best_assignment)
    else:
        print("NO")

if __name__ == "__main__":
    main()
