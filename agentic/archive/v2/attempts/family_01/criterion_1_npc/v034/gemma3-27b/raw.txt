def solve():
    n, m = map(int, input().split())
    incompatible = []
    for _ in range(m):
        u, v = map(int, input().split())
        incompatible.append((u, v))
    
    def is_valid(assignment):
        for u, v in incompatible:
            if assignment[u-1] == assignment[v-1]:
                return False
        return True
    
    def find_schedule():
        import itertools
        
        best_schedule = None
        
        for schedule in itertools.product(range(1, 4), repeat=n):
            if is_valid(schedule):
                if best_schedule is None:
                    best_schedule = list(schedule)
                else:
                    if list(schedule) < best_schedule:
                        best_schedule = list(schedule)
        
        return best_schedule
    
    schedule = find_schedule()
    
    if schedule is None:
        print("NO")
    else:
        print("YES")
        print(*schedule)

solve()
