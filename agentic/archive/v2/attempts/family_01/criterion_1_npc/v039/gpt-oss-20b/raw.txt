```python
import sys
import threading

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    assigned = [0] * n  # 0 means uncolored, else 1..3
    color_masks = [0, 0, 0]  # bitmask of vertices colored with color 1,2,3

    # helper to get available colors for vertex v
    def avail_mask(v: int) -> int:
        mask = 0
        av = adj[v]
        for c in range(3):
            if (av & color_masks[c]) == 0:
                mask |= 1 << c
        return mask

    sys.setrecursionlimit(10000)

    def dfs(colored_cnt: int) -> bool:
        # forced propagation
        while True:
            changed = False
            for v in range(n):
                if assigned[v] != 0:
                    continue
                mask = avail_mask(v)
                if mask == 0:
                    return False
                if mask & (mask - 1) == 0:  # only one bit set
                    c = (mask.bit_length() - 1)
                    assigned[v] = c + 1
                    color_masks[c] ^= 1 << v
                    colored_cnt += 1
                    changed = True
            if not changed:
                break

        if colored_cnt == n:
            return True

        # pick first uncolored vertex
        for v in range(n):
            if assigned[v] == 0:
                break
        mask = avail_mask(v)
        for c in range(3):
            if mask & (1 << c):
                assigned[v] = c + 1
                color_masks[c] ^= 1 << v
                if dfs(colored_cnt + 1):
                    return True
                color_masks[c] ^= 1 << v
                assigned[v] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, assigned)))
    else:
        print("NO")

def main():
    solve()

if __name__ == "__main__":
    threading.Thread(target=main).start()
```
