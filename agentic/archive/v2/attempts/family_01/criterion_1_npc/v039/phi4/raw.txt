```python
import sys

def solve():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    edges = []
    index = 2
    
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index + 1]) - 1
        edges.append((u, v))
        index += 2

    def is_valid(assignment):
        for (u, v) in edges:
            if assignment[u] == assignment[v]:
                return False
        return True
    
    def find_assignment():
        from itertools import permutations
        
        channels = [1, 2, 3]
        
        # Generate all possible assignments for n transmitters using channels {1, 2, 3}
        for perm in permutations(channels * (n // len(channels) + 1), n):
            if is_valid(perm[:n]):
                return "YES\n" + ' '.join(map(str, perm[:n]))
        
        return "NO"
    
    print(find_assignment())
```
