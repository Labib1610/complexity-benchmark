```python
import sys
from itertools import permutations

def main():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    incompatible_pairs = []
    index = 2
    for _ in range(m):
        u, v = int(data[index]), int(data[index + 1])
        incompatible_pairs.append((u - 1, v - 1))
        index += 2
    
    # Generate all possible assignments of domains to modules
    domain_permutations = permutations([1, 2, 3], n)
    
    valid_assignment_found = False
    best_assignment = None
    
    for perm in domain_permutations:
        valid = True
        
        # Check each incompatible pair
        for u, v in incompatible_pairs:
            if perm[u] == perm[v]:
                valid = False
                break
        
        if valid:
            if not valid_assignment_found or perm < best_assignment:
                valid_assignment_found = True
                best_assignment = perm
    
    if valid_assignment_found:
        print("YES")
        print(" ".join(map(str, best_assignment)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
