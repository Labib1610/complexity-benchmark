import sys

def main():
    input = sys.stdin.read
    data = input().split()
    
    index = 0
    
    n = int(data[index])
    m = int(data[index + 1])
    index += 2
    
    adj_list = [[] for _ in range(n)]
    
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index + 1]) - 1
        adj_list[u].append(v)
        adj_list[v].append(u)
        index += 2
    
    def is_valid(assignment, person):
        time_block = assignment[person]
        
        for neighbor in adj_list[person]:
            if assignment[neighbor] == time_block:
                return False
        
        return True

    def backtrack(assignment, person):
        if person == n:
            return True
        
        for time_block in range(1, 4):
            assignment[person] = time_block
            
            if is_valid(assignment, person) and backtrack(assignment, person + 1):
                return True
            
            assignment[person] = 0
        
        return False
    
    assignment = [0] * n
    
    if backtrack(assignment, 0):
        print("YES")
        print(' '.join(map(str, assignment)))
    else:
        print("NO")

if __name__ == "__main__":
    main()
