import sys

def main():
    sys.setrecursionlimit(10000)
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * n                     # 0 = uncolored, 1..3 colored
    possible = [0b111] * n               # bitmask of allowed colors
    solution = None

    def dfs(idx):
        nonlocal solution
        if idx == n:
            solution = colors.copy()
            return True
        v = idx
        mask = possible[v]
        for c in (1, 2, 3):
            if mask & (1 << (c - 1)):
                colors[v] = c
                changes = []
                fail = False
                for u in adj[v]:
                    if colors[u] == 0:
                        old = possible[u]
                        new = old & ~(1 << (c - 1))
                        if new == 0:
                            fail = True
                            break
                        if new != old:
                            changes.append((u, old))
                            possible[u] = new
                if not fail:
                    if dfs(idx + 1):
                        return True
                # revert
                for u, oldmask in changes:
                    possible[u] = oldmask
                colors[v] = 0
        return False

    if dfs(0):
        out_lines = ["YES"]
        out_lines.append(" ".join(str(x) for x in solution))
        sys.stdout.write("\n".join(out_lines))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    main()
