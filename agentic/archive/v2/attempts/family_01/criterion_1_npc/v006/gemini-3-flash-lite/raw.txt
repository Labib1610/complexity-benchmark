```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(curr):
        if curr == n:
            return True
        
        # Try assigning 1, 2, 3 in order for lexicographical smallest
        for color in [1, 2, 3]:
            possible = True
            for prev in range(curr):
                if (adj[curr] & (1 << prev)):
                    if assignment[prev] == color:
                        possible = False
                        break
            
            if possible:
                assignment[curr] = color
                if backtrack(curr + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
