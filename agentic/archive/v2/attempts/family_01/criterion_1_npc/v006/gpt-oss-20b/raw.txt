```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[False]*n for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u][v] = adj[v][u] = True

    colors = [0]*n  # 0 means unassigned
    found = False

    def dfs(pos: int) -> bool:
        nonlocal found
        if pos == n:
            return True
        for c in (1,2,3):
            conflict = False
            for j in range(pos):
                if adj[pos][j] and colors[j] == c:
                    conflict = True
                    break
            if conflict:
                continue
            colors[pos] = c
            if dfs(pos+1):
                return True
            colors[pos] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
