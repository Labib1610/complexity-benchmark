import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adjacency = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adjacency[u] |= 1 << v
        adjacency[v] |= 1 << u

    colors = [0] * n          # 0 = unassigned, 1/2/3 assigned
    colorMasks = [0, 0, 0, 0] # index by color, bitmask of nodes colored with that color

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        node = idx
        # Try colors in increasing order for lexicographic minimality
        for col in (1, 2, 3):
            if adjacency[node] & colorMasks[col]:
                continue  # conflict with same color neighbor
            colors[node] = col
            colorMasks[col] |= 1 << node
            if dfs(idx + 1):
                return True
            colorMasks[col] &= ~(1 << node)
            colors[node] = 0
        return False

    if dfs(0):
        out = ["YES", " ".join(str(c) for c in colors)]
        sys.stdout.write("\n".join(out))
    else:
        sys.stdout.write("NO")

if __name__ == "__main__":
    solve()
