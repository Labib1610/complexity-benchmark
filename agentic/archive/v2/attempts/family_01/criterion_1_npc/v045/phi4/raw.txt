import sys
from itertools import combinations

def read_input():
    input = sys.stdin.read
    data = input().split()
    idx = 0
    
    n = int(data[idx])
    m = int(data[idx + 1])
    idx += 2
    
    incompatibilities = []
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        incompatibilities.append((u - 1, v - 1))
        idx += 2
        
    return n, m, incompatibilities

def is_valid_assignment(n, edges, assignment):
    for (a, b) in edges:
        if assignment[a] == assignment[b]:
            return False
    return True

def solve(n, edges):
    min_shifts = [1] * n  # Start with the smallest lexicographical shifts
    
    def backtrack(index):
        if index == n:
            if is_valid_assignment(n, edges, min_shifts):
                for shift in min_shifts:
                    print(shift, end=" ")
                sys.exit()
            return
        
        for shift in range(1, 4):  # Try assigning shifts 1, 2, and 3
            min_shifts[index] = shift
            if is_valid_assignment(n, edges, min_shifts):
                backtrack(index + 1)
    
    backtrack(0)
    print("NO")

def main():
    n, m, incompatibilities = read_input()
    solve(n, incompatibilities)

if __name__ == "__main__":
    main()
