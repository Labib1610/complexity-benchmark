def solve():
    n, m = map(int, input().split())
    incompatible = []
    for _ in range(m):
        u, v = map(int, input().split())
        incompatible.append((u, v))

    assignment = [0] * n

    def is_valid(person, slot):
        for u, v in incompatible:
            if (u == person + 1 and assignment[v - 1] == slot) or \
               (v == person + 1 and assignment[u - 1] == slot):
                return False
        return True

    def backtrack(person):
        if person == n:
            return True
        
        for slot in [1, 2, 3]:
            if is_valid(person, slot):
                assignment[person] = slot
                if backtrack(person + 1):
                    return True
                assignment[person] = 0  # Backtrack
        
        return False

    if backtrack(0):
        print("YES")
        print(*assignment)
    else:
        print("NO")

solve()
