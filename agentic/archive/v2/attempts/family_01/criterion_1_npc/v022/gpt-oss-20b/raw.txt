```python
import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u != v:
            adj[u].add(v)
            adj[v].add(u)

    colors = [0] * n  # 0 unassigned, 1-3 assigned

    def dfs(pos: int) -> bool:
        if pos == n:
            return True
        for c in (1, 2, 3):
            conflict = False
            for nb in adj[pos]:
                if colors[nb] == c:
                    conflict = True
                    break
            if conflict:
                continue
            colors[pos] = c
            if dfs(pos + 1):
                return True
            colors[pos] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
