import sys
from itertools import combinations

def hardware_bus_allocation():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    edges = []
    index = 2
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u - 1, v - 1))
        index += 2

    # Try all possible bus assignments (3^n possibilities)
    def is_valid(assignment):
        for u, v in edges:
            if assignment[u] == assignment[v]:
                return False
        return True
    
    buses = [1, 2, 3]
    
    min_assignment = None
    
    def generate_assignments(current_assignment, pos):
        nonlocal min_assignment
        
        if len(current_assignment) == n:
            if is_valid(current_assignment):
                if min_assignment is None or current_assignment < min_assignment:
                    min_assignment = current_assignment[:]
            return
        
        for bus in buses:
            current_assignment.append(bus)
            generate_assignments(current_assignment, pos + 1)
            current_assignment.pop()
    
    generate_assignments([], 0)
    
    if min_assignment is None:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, min_assignment)))

hardware_bus_allocation()
