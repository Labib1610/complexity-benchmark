import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * (n + 1)          # assigned color, 0 means uncolored
    used_colors = [0] * (n + 1)     # bitmask of colors already used by neighbors

    answer = []

    def backtrack(idx: int) -> bool:
        if idx > n:
            # all vertices colored
            answer.extend(colors[1:])
            return True
        for col in (1, 2, 3):
            mask = 1 << (col - 1)
            if used_colors[idx] & mask:
                continue  # color already used by a neighbor
            # assign
            colors[idx] = col
            changes = []
            for u in adj[idx]:
                if colors[u] == 0:
                    changes.append((u, used_colors[u]))
                    used_colors[u] |= mask
            # feasibility check
            feasible = True
            for u in range(idx + 1, n + 1):
                if colors[u] == 0 and used_colors[u] == 0b111:
                    feasible = False
                    break
            if feasible:
                if backtrack(idx + 1):
                    return True
            # revert
            colors[idx] = 0
            for u, old in changes:
                used_colors[u] = old
        return False

    if backtrack(1):
        print("YES")
        print(' '.join(map(str, answer)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
