from sys import stdin

def solve():
    def assign_slots(n, m, edges):
        graph = [[] for _ in range(n + 1)]
        for u, v in edges:
            graph[u].append(v)
            graph[v].append(u)

        slots = [0] * (n + 1)  # 0 means unassigned

        def is_valid(person, slot):
            for neighbor in graph[person]:
                if slots[neighbor] == slot:
                    return False
            return True

        def backtrack(person):
            if person > n:
                return True
            
            for slot in [1, 2, 3]:
                if is_valid(person, slot):
                    slots[person] = slot
                    if backtrack(person + 1):
                        return True
                    slots[person] = 0

            return False

        if not backtrack(1):
            print("NO")
        else:
            print("YES")
            print(*slots[1:])

    n, m = map(int, stdin.readline().split())
    edges = [tuple(map(int, stdin.readline().split())) for _ in range(m)]
    assign_slots(n, m, edges)

solve()
