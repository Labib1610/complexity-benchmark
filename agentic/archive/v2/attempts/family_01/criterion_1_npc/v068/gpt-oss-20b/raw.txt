import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neighbors = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if v not in neighbors[u]:
            neighbors[u].append(v)
        if u not in neighbors[v]:
            neighbors[v].append(u)

    allowed = [0b111] * n          # bits for colors 1,2,3
    color = [0] * n                # 0 means uncolored

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        for c in (1, 2, 3):
            bit = 1 << (c - 1)
            if not (allowed[idx] & bit):
                continue
            color[idx] = c
            changes = []
            fail = False
            for nb in neighbors[idx]:
                if color[nb] == 0:
                    prev = allowed[nb]
                    allowed[nb] &= ~bit
                    changes.append((nb, prev))
                    if allowed[nb] == 0:
                        fail = True
                        break
            if not fail and dfs(idx + 1):
                return True
            # revert changes
            for nb, prev in changes:
                allowed[nb] = prev
            color[idx] = 0
        return False

    if dfs(0):
        sys.stdout.write(' '.join(str(c) for c in color))
    else:
        sys.stdout.write('NO')

if __name__ == "__main__":
    solve()
