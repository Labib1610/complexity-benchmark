from itertools import combinations

def main():
    n, m = map(int, input().split())
    interference = [set() for _ in range(n)]
    
    for _ in range(m):
        u, v = map(lambda x: int(x) - 1, input().split())
        interference[u].add(v)
        interference[v].add(u)
        
    # Try all combinations of channel assignments
    def is_valid(assignment):
        for i in range(n):
            for j in interference[i]:
                if assignment[i] == assignment[j]:
                    return False
        return True

    best_assignment = [3] * n  # Start with the worst lexicographical order (all 3's)
    found_solution = False
    
    for channels in combinations(range(1, 4), n):
        if is_valid(channels):
            if not found_solution or channels < tuple(best_assignment):
                best_assignment = list(channels)
                found_solution = True

    if found_solution:
        print("YES")
        print(*best_assignment)
    else:
        print("NO")

if __name__ == "__main__":
    main()
