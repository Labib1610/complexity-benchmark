def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))

    def is_valid(allocation):
        for u, v in edges:
            if allocation[u-1] == allocation[v-1]:
                return False
        return True

    def find_lexicographically_smallest_allocation():
        allocation = [0] * n
        
        def backtrack(index):
            if index == n:
                if is_valid(allocation):
                    return allocation
                else:
                    return None
            
            for channel in range(1, 4):
                allocation[index] = channel
                result = backtrack(index + 1)
                if result:
                    return result
            
            return None

        return backtrack(0)
    
    allocation = find_lexicographically_smallest_allocation()
    
    if allocation is None:
        print("NO")
    else:
        print("YES")
        print(*allocation)

solve()
