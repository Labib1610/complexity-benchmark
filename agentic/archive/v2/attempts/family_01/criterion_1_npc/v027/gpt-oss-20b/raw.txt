import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n
    found = False

    # Simple 3-coloring backtracking
    def dfs(idx):
        nonlocal found
        if found:
            return
        if idx == n:
            found = True
            return
        for c in (1, 2, 3):
            conflict = False
            for nb in adj[idx]:
                if colors[nb] == c:
                    conflict = True
                    break
            if conflict:
                continue
            colors[idx] = c
            dfs(idx + 1)
            if found:
                return
            colors[idx] = 0

    dfs(0)

    out = []
    if not found:
        out.append("NO")
    else:
        out.append("YES")
        out.append(" ".join(map(str, colors)))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
