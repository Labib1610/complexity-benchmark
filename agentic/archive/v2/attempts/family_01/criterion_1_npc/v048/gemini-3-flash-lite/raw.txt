import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        
        # Try assigning window 1, then 2, then 3 to get lexicographical smallest
        for window in [1, 2, 3]:
            # Check for conflicts
            conflict = False
            for prev in range(idx):
                if assignment[prev] == window:
                    if (adj[idx + 1] & (1 << prev)):
                        conflict = True
                        break
            
            if not conflict:
                assignment[idx] = window
                if backtrack(idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
