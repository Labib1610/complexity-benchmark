import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    avail = [7] * n          # bitmask 111 -> all colors available
    color = [0] * n          # 0 means uncolored

    sys.setrecursionlimit(1000)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        mask = avail[idx]
        if mask == 0:
            return False
        # iterate colors 1..3
        for c in (1, 2, 3):
            bit = 1 << (c - 1)
            if not (mask & bit):
                continue
            color[idx] = c
            changes = []
            ok = True
            for nb in adj[idx]:
                if nb > idx:
                    old = avail[nb]
                    new = old & ~bit
                    if new != old:
                        changes.append((nb, old))
                        avail[nb] = new
                        if new == 0:
                            ok = False
                            break
            if ok and dfs(idx + 1):
                return True
            # revert changes
            for nb, old in changes:
                avail[nb] = old
            color[idx] = 0
        return False

    possible = dfs(0)

    out_lines = []
    if possible:
        out_lines.append("YES")
        out_lines.append(' '.join(str(color[i]) for i in range(n)))
    else:
        out_lines.append("NO")
    sys.stdout.write('\n'.join(out_lines))

if __name__ == "__main__":
    solve()
