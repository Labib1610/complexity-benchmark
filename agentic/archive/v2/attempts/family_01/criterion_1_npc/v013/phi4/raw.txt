import sys
from itertools import combinations

def solve(n, m, conflicts):
    def is_valid(assignment):
        for u, v in conflicts:
            if assignment[u-1] == assignment[v-1]:
                return False
        return True

    def backtrack(i, assignment):
        if i == n:
            return True
        
        for shift in range(1, 4):
            assignment[i] = shift
            if is_valid(assignment) and backtrack(i + 1, assignment):
                return True
            assignment[i] = 0
        
        return False

    assignment = [0] * n
    if backtrack(0, assignment):
        print("YES")
        print(*assignment)
    else:
        print("NO")

if __name__ == "__main__":
    input_data = sys.stdin.read().strip().split()
    idx = 0
    
    n = int(input_data[idx])
    m = int(input_data[idx + 1])
    idx += 2

    conflicts = []
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        conflicts.append((u, v))
        idx += 2
    
    solve(n, m, conflicts)
