import sys

def solve():
    n, m = map(int, input().split())
    graph = [[] for _ in range(n)]
    
    for _ in range(m):
        u, v = map(lambda x: int(x) - 1, input().split())
        graph[u].append(v)
        graph[v].append(u)

    # Try all possible distributions of n operators into three slots
    from itertools import combinations
    
    def is_valid_assignment(assignment):
        for i in range(n):
            for j in graph[i]:
                if assignment[i] == assignment[j]:
                    return False
        return True

    min_assignment = None

    for sizes in combinations(range(n + 1), 2):
        if sum(sizes) > n:
            continue
        a, b = sizes[0], sizes[1]
        c = n - a - b
        
        if c < 0 or c > n:
            continue
        
        slots = [1] * a + [2] * b + [3] * c
        for perm in itertools.permutations(slots):
            assignment = list(perm)
            if is_valid_assignment(assignment):
                if min_assignment is None or assignment < min_assignment:
                    min_assignment = assignment

    if min_assignment is not None:
        print("YES")
        print(" ".join(map(str, min_assignment)))
    else:
        print("NO")

import itertools
solve()
