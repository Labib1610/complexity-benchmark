def solve():
    n, m = map(int, input().split())
    restrictions = []
    for _ in range(m):
        u, v = map(int, input().split())
        restrictions.append((u, v))
    
    def is_valid(assignment):
        for u, v in restrictions:
            if assignment[u-1] == assignment[v-1]:
                return False
        return True
    
    def find_assignment():
        assignment = [0] * n
        
        def backtrack(index):
            if index == n:
                if is_valid(assignment):
                    return True
                else:
                    return False
            
            for channel in range(1, 4):
                assignment[index] = channel
                if backtrack(index + 1):
                    return True
                assignment[index] = 0  # Backtrack
            
            return False
        
        if backtrack(0):
            return assignment
        else:
            return None
    
    def find_lexicographically_smallest_assignment():
        assignment = [0] * n

        def backtrack(index):
            if index == n:
                if is_valid(assignment):
                    return assignment
                else:
                    return None

            for channel in range(1, 4):
                assignment[index] = channel
                result = backtrack(index + 1)
                if result:
                    return result
                assignment[index] = 0
            return None

        return backtrack(0)
    
    assignment = find_lexicographically_smallest_assignment()
    
    if assignment:
        print("YES")
        print(*assignment)
    else:
        print("NO")

solve()
