```python
import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * n  # 0 unassigned, 1-3 assigned

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        for c in (1, 2, 3):
            conflict = False
            for nb in adj[idx]:
                if nb < idx and colors[nb] == c:
                    conflict = True
                    break
            if conflict:
                continue
            colors[idx] = c
            if dfs(idx + 1):
                return True
            colors[idx] = 0
        return False

    if dfs(0):
        out = ['YES', '\n', ' '.join(str(colors[i]) for i in range(n))]
        sys.stdout.write(''.join(out))
    else:
        sys.stdout.write('NO')

if __name__ == "__main__":
    solve()
```
