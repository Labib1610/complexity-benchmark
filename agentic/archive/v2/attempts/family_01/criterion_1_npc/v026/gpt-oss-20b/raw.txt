import sys
sys.setrecursionlimit(1000000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        node = idx + 1
        for c in (1, 2, 3):
            conflict = False
            for nb in adj[node]:
                if color[nb] == c:
                    conflict = True
                    break
            if not conflict:
                color[node] = c
                if dfs(idx + 1):
                    return True
                color[node] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(str(color[i]) for i in range(1, n + 1)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
