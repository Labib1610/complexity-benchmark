import sys

def find_primary_mutual_alliance_triad():
    n, m = map(int, input().split())
    adjacency_matrix = [[0 for _ in range(n)] for _ in range(n)]
    
    for _ in range(m):
        u, v = map(int, input().split())
        adjacency_matrix[u-1][v-1] = 1
        adjacency_matrix[v-1][u-1] = 1
    
    for a in range(n):
        for b in range(a+1, n):
            if adjacency_matrix[a][b]:
                for c in range(b+1, n):
                    if adjacency_matrix[a][c] and adjacency_matrix[b][c]:
                        print(f"{a + 1} {b + 1} {c + 1}")
                        return

find_primary_mutual_alliance_triad()
