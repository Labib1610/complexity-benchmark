import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    neighbors = [0] * (n + 1)           # bitset representation
    adj = [[] for _ in range(n + 1)]    # adjacency lists

    for _ in range(m):
        u = next(it)
        v = next(it)
        bit_u = 1 << (v - 1)
        bit_v = 1 << (u - 1)
        neighbors[u] |= bit_u
        neighbors[v] |= bit_v
        adj[u].append(v)
        adj[v].append(u)

    for a in range(1, n - 1):
        for b in adj[a]:
            if b <= a:
                continue
            inter = neighbors[a] & neighbors[b]
            shift = inter >> b
            if shift:
                low = shift & -shift
                p = low.bit_length() - 1
                c = b + p + 1
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
