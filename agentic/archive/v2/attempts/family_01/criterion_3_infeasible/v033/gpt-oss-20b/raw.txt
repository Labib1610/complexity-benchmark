```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [bytearray(n + 1) for _ in range(n + 1)]
    adj_list = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = 1
        adj[v][u] = 1
        adj_list[u].append(v)
        adj_list[v].append(u)

    for i in range(1, n + 1):
        adj_list[i].sort()

    for a in range(1, n + 1):
        neigh = [v for v in adj_list[a] if v > a]
        L = len(neigh)
        for i in range(L):
            b = neigh[i]
            for j in range(i + 1, L):
                c = neigh[j]
                if adj[b][c]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
```
