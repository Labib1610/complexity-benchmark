import sys

def find_minimal_bottleneck():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        index += 2
    
    # Create adjacency list
    adj_list = {i: [] for i in range(1, n+1)}
    for u, v in edges:
        if u < v:
            adj_list[u].append(v)
        else:
            adj_list[v].append(u)

    # Sort the adjacency lists to ensure lexicographical order
    for key in adj_list:
        adj_list[key].sort()
    
    min_triangle = (float('inf'), float('inf'), float('inf'))
    
    # Check each possible triangle (a, b, c) with a < b < c
    for a in range(1, n+1):
        for b in adj_list[a]:
            if b > a:
                for c in adj_list[b]:
                    if c > b and c in adj_list[a]:  # Ensure all three pairs are present
                        triangle = (a, b, c)
                        if triangle < min_triangle:
                            min_triangle = triangle
    
    print(*min_triangle)

find_minimal_bottleneck()
