import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate through all a from 1 to n,
    # then iterate through sorted neighbors b of a,
    # then check if there exists c such that (a, c) and (b, c) exist.
    
    # Sort adjacency lists to ensure we find lexicographical order efficiently
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
        
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = adj[a]
        # Iterate b in neighbors of a such that b > a
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need the smallest c > b such that c is in adj[a] and adj[b]
            # We can intersect the sorted lists adj[a] and adj[b]
            # Since we need the smallest c > b, we look after b in the lists
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                for c in adj[a]:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj[b]:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
