```python
import sys

# Read inputs
n, m = map(int, input().split())
incompatibilities = [tuple(map(int, input().split())) for _ in range(m)]

# Sort pairs to ensure a < b in each pair
incompatibilities.sort()

# Convert the list of pairs into a set for O(1) lookups
incompat_set = set(incompatibilities)

# Iterate over all possible triplets (a, b, c)
for a in range(1, n + 1):
    # Create a dictionary to track the second element `b` for each first element `a`
    next_b_for_a = {}
    
    for u, v in incompatibilities:
        if u == a and v > a:  # Ensure we only consider pairs where a < b
            next_b_for_a[v] = True
    
    # Extract the sorted list of possible values for `b` when paired with `a`
    possible_bs = sorted(next_b_for_a.keys())
    
    for i in range(len(possible_bs)):
        b = possible_bs[i]
        
        # Create a dictionary to track the third element `c` for each pair (a, b)
        next_c_for_ab = {}
        
        for u, v in incompatibilities:
            if (u == a and v > b) or (v == a and u > b):  # Ensure we only consider pairs where a < c and b < c
                if u > b:  # `c` should be greater than both `a` and `b`
                    next_c_for_ab[v] = True
        
        # Extract the sorted list of possible values for `c` when paired with (a, b)
        possible_cs = sorted(next_c_for_ab.keys())
        
        for j in range(len(possible_cs)):
            c = possible_cs[j]
            
            if (a, c) in incompat_set and (b, c) in incompat_set:
                # Found the lexicographically smallest triplet (a, b, c)
                print(a, b, c)
                sys.exit()
```
