import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    bits = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        bits[u] |= 1 << v
        bits[v] |= 1 << u

    full_mask = (1 << n) - 1
    mask_after = [full_mask ^ ((1 << (i + 1)) - 1) for i in range(n)]

    for a in range(n):
        ba = bits[a]
        for b in range(a + 1, n):
            if not (ba >> b) & 1:
                continue
            inter = ba & bits[b] & mask_after[b]
            if inter:
                low = inter & -inter
                c = low.bit_length() - 1
                # c > b automatically due to mask_after
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    solve()
