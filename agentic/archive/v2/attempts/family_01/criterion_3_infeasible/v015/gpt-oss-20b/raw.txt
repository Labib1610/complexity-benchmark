```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    for i in range(n - 2):
        ai = adj[i]
        for j in range(i + 1, n - 1):
            if not (ai >> j) & 1:
                continue
            common = ai & adj[j]
            mask = common >> (j + 1)
            if mask == 0:
                continue
            low = mask & -mask
            pos = low.bit_length() - 1
            k = pos + j + 1
            print(i + 1, j + 1, k + 1)
            return

if __name__ == "__main__":
    solve()
```
