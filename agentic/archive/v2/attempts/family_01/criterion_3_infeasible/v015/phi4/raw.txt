```python
import sys

# Read input data
input = sys.stdin.read
data = input().split()

n = int(data[0])
m = int(data[1])

# Initialize an adjacency list for storing incompatibilities
incompatibility = [[False] * (n + 1) for _ in range(n + 1)]

# Read the pairs of incompatible workers
index = 2
for _ in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    incompatibility[u][v] = True
    incompatibility[v][u] = True
    index += 2

# Find the lexicographically smallest triplet with all pairwise incompatibilities
for a in range(1, n + 1):
    for b in range(a + 1, n + 1):
        if incompatibility[a][b]:
            for c in range(b + 1, n + 1):
                if incompatibility[a][c] and incompatibility[b][c]:
                    print(a, b, c)
                    sys.exit()
```
