```python
import sys

input = sys.stdin.read
data = input().split()

n = int(data[0])
m = int(data[1])

incompatibility = [[False] * n for _ in range(n)]

index = 2
for _ in range(m):
    u = int(data[index]) - 1
    v = int(data[index + 1]) - 1
    incompatibility[u][v] = True
    incompatibility[v][u] = True
    index += 2

found = False
for a in range(n):
    for b in range(a + 1, n):
        if not incompatibility[a][b]:
            continue
        for c in range(b + 1, n):
            if incompatibility[a][c] and incompatibility[b][c]:
                print(f"{a+1} {b+1} {c+1}")
                found = True
                break
        if found:
            break
    if found:
        break
```
