import sys

def main():
    data = list(map(int, sys.stdin.read().strip().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)
    adj = [0] * (n + 1)  # 1-indexed
    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    for a in range(1, n - 1):
        for b in range(a + 1, n):
            if (adj[a] >> (b - 1)) & 1:
                intersection = adj[a] & adj[b]
                shift = intersection >> b
                if shift:
                    tz = (shift & -shift).bit_length() - 1
                    c = b + tz + 1
                    print(a, b, c)
                    return

if __name__ == "__main__":
    main()
