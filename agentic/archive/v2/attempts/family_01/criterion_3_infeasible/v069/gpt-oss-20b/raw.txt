```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [0] * (n + 1)  # 1-indexed

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    for a in range(1, n - 1):
        a_bits = adj[a]
        for b in range(a + 1, n):
            if ((a_bits >> (b - 1)) & 1) == 0:
                continue
            inter = a_bits & adj[b]
            inter_shifted = inter >> b  # shift to ignore indices <= b
            if inter_shifted:
                lowest = inter_shifted & -inter_shifted
                c = (lowest.bit_length() - 1) + b + 1
                sys.stdout.write(f"{a} {b} {c}")
                return

if __name__ == "__main__":
    solve()
```
