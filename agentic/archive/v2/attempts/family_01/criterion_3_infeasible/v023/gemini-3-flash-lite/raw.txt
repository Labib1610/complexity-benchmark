```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        if u > v:
            u, v = v, u
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest triplet
    # We want smallest a, then smallest b, then smallest c.
    # Triplet (a, b, c) with a < b < c.
    # This means (a, b) is an edge, (a, c) is an edge, (b, c) is an edge.
    
    edges.sort()
    
    # Iterate through all edges (a, b) as the first two nodes of the triplet
    for a, b in edges:
        # We need to find a c > b such that c is connected to both a and b.
        # To ensure lexicographical order, we check c in increasing order.
        # Since we need the overall smallest (a, b, c), checking edges in sorted order
        # and then c in sorted order is efficient.
        
        # Intersection of neighbors of a and b
        common = adj[a].intersection(adj[b])
        
        # We need c > b
        candidates = [c for c in common if c > b]
        if candidates:
            print(f"{a} {b} {min(candidates)}")
            return

solve()
```
