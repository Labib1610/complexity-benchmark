import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    # adjacency masks (0‑based)
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # precompute mask of indices greater than i
    mask_gt = [0] * n
    for i in range(n - 1):
        mask_gt[i] = ((1 << (n - i - 1)) - 1) << (i + 1)
    mask_gt[n - 1] = 0

    # iterate a, then b> a, then find smallest c > b
    for a in range(n):
        neigh_a_gt = adj[a] & mask_gt[a]
        # iterate over bits set in neigh_a_gt
        nb_mask = neigh_a_gt
        while nb_mask:
            b_bit = nb_mask & -nb_mask
            b = b_bit.bit_length() - 1
            nb_mask ^= b_bit

            # intersection of neighbors of a and b, only > b
            candidate = adj[a] & adj[b] & mask_gt[b]
            if candidate:
                c_bit = candidate & -candidate
                c = c_bit.bit_length() - 1
                # output 1‑based
                print(f"{a+1} {b+1} {c+1}")
                return

if __name__ == "__main__":
    solve()
