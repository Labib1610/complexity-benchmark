import sys

def find_hazard_set(n, m, interference_entries):
    graph = [[0] * n for _ in range(n)]
    
    # Read the interference entries and construct the adjacency matrix
    for u, v in interference_entries:
        u -= 1
        v -= 1
        if u > v:
            u, v = v, u
        graph[u][v] = 1

    # Try to find the smallest set (a, b, c)
    for a in range(n):
        for b in range(a + 1, n):
            if graph[a][b]:
                for c in range(b + 1, n):
                    if graph[a][c] and graph[b][c]:
                        return a + 1, b + 1, c + 1

# Read input
input_data = sys.stdin.read().strip().split()
n = int(input_data[0])
m = int(input_data[1])

interference_entries = []
index = 2
for _ in range(m):
    u = int(input_data[index])
    v = int(input_data[index + 1])
    interference_entries.append((u, v))
    index += 2

# Find the lexicographically smallest hazard set
a, b, c = find_hazard_set(n, m, interference_entries)

# Print the result
print(a, b, c)
