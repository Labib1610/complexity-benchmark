```python
import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We want the lexicographically smallest triple (a, b, c)
    # Iterate a from 1 to n-2
    for a in range(1, n + 1):
        # Candidates for b must be neighbors of a and b > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # Candidates for c must be neighbors of both a and b, and c > b
            # We can find this by intersecting neighbors of a and neighbors of b
            # and filtering for > b. Since we want the lexicographically 
            # smallest, we look for the smallest c.
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
            else:
                candidates = adj[b]
            
            found_c = -1
            for c in candidates:
                if c > b and c in adj[a] and c in adj[b]:
                    if found_c == -1 or c < found_c:
                        found_c = c
            
            if found_c != -1:
                print(f"{a} {b} {found_c}")
                return

if __name__ == '__main__':
    solve()
```
