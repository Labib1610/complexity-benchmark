```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    
    index = 0
    n = int(data[index])
    m = int(data[index + 1])
    index += 2
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index + 1]) - 1
        index += 2
        adj[u].add(v)
        adj[v].add(u)

    a, b, c = None, None, None

    for i in range(n):
        for j in adj[i]:
            if j > i:
                for k in adj[j]:
                    if k > j and k in adj[i]:
                        if a is None or (i < a) or (i == a and j < b) or (i == a and j == b and k < c):
                            a, b, c = i, j, k

    print(a + 1, b + 1, c + 1)

if __name__ == "__main__":
    main()
```
