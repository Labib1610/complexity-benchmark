import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    for i in range(1, n + 1):
        adj[i].sort()
    adj_set = [set(lst) for lst in adj]
    for a in range(1, n + 1):
        neigh_a = adj[a]
        for b in neigh_a:
            if b <= a:
                continue
            neigh_b_set = adj_set[b]
            for c in neigh_a:
                if c <= b:
                    continue
                if c in neigh_b_set:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
