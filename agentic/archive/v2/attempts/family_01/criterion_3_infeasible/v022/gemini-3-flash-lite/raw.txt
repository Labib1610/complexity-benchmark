```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) edge lookups
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        if u > v:
            u, v = v, u
        
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triple
    # A triple (a, b, c) with a < b < c exists if (a, b) is an edge
    # and there exists c such that (a, c) and (b, c) are edges.
    # To find the lexicographically smallest, we iterate through all edges (a, b)
    # where a < b, and check for common neighbors c > b.
    
    edges.sort()
    
    best = (n + 1, n + 1, n + 1)
    
    for a, b in edges:
        # We need a < b < c.
        # Since we iterate edges (a, b) in sorted order,
        # we check for common neighbors c > b.
        
        # Optimization: if a is already greater than best[0], we can't do better.
        if a > best[0]:
            break
            
        # Find intersection of adj[a] and adj[b]
        # Only care about c > b to satisfy a < b < c
        for c in adj[a]:
            if c > b:
                if c in adj[b]:
                    # Found a triple (a, b, c)
                    if (a, b, c) < best:
                        best = (a, b, c)
    
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
