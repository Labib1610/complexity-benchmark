```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to ensure we process in lexicographical order (a, b)
    edges.sort()
    
    # We want the smallest (a, b, c).
    # Iterate through all pairs (a, b) that are connected.
    # For each (a, b), find the smallest c > b such that c is connected to both a and b.
    
    for a, b in edges:
        # Find common neighbors of a and b that are > b
        # Since we want smallest (a, b, c), checking the common neighbors
        # of a and b is sufficient.
        common = adj[a].intersection(adj[b])
        
        valid_c = [c for c in common if c > b]
        if valid_c:
            c = min(valid_c)
            print(f"{a} {b} {c}")
            return

if __name__ == '__main__':
    solve()
```
