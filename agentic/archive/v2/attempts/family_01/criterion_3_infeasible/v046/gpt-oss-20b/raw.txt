import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [[] for _ in range(n)]
    bits = [0] * n

    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        adj[u].append(v)
        adj[v].append(u)
        bits[u] |= 1 << v
        bits[v] |= 1 << u

    for a in range(n):
        neigh = adj[a]
        # iterate neighbors greater than a to ensure a < b
        for b in neigh:
            if b <= a:
                continue
            inter = bits[a] & bits[b]
            # mask bits greater than b
            mask = inter >> b
            if mask:
                lowbit = mask & -mask
                c0 = lowbit.bit_length() - 1
                c = b + c0 + 1  # convert to 1-based
                print(a + 1, b + 1, c)
                return

if __name__ == "__main__":
    solve()
