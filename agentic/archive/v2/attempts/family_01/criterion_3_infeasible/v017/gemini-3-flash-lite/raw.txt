import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     Check if b is in adj[a]
    #       If yes, check for a common neighbor c > b
    
    # Sort neighbors to speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Only consider b > a
        neighbors_a = [x for x in adj_list[a] if x > a]
        for b in neighbors_a:
            # We need a c such that c > b and c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can iterate through them
            # and check if they are in adj[b]
            
            # Use sets for O(1) lookup
            for c in neighbors_a:
                if c > b and c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
