import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We want the lexicographically smallest (a, b, c) with a < b < c.
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Candidates for b are neighbors of a such that b > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # Candidates for c are neighbors of both a and b such that c > b
            # Intersection of adj[a] and adj[b]
            # Since n is up to 2000, we can use set intersection or bitsets
            # For efficiency, iterate over the smaller neighbor set
            if len(adj[a]) < len(adj[b]):
                common = adj[a].intersection(adj[b])
            else:
                common = adj[b].intersection(adj[a])
            
            # Find the smallest c > b in the common set
            res_c = -1
            for val in common:
                if val > b:
                    if res_c == -1 or val < res_c:
                        res_c = val
            
            if res_c != -1:
                print(f"{a} {b} {res_c}")
                return

if __name__ == '__main__':
    solve()
