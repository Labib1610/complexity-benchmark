import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adjacency = [0] * n
    neigh = [[] for _ in range(n)]

    idx = 2
    for _ in range(m):
        u, v = data[idx], data[idx + 1]
        idx += 2
        if u == v:
            continue
        u -= 1
        v -= 1
        adjacency[u] |= (1 << v)
        adjacency[v] |= (1 << u)
        neigh[u].append(v + 1)  # store 1‑based for easy comparison
        neigh[v].append(u + 1)

    # Make neighbor lists unique and sorted
    for i in range(n):
        neigh[i] = sorted(set(neigh[i]))

    for a in range(1, n + 1):
        for b in neigh[a - 1]:
            if b <= a:
                continue
            common = adjacency[a - 1] & adjacency[b - 1]
            if common == 0:
                continue
            shifted = common >> b
            if shifted == 0:
                continue
            low = shifted & -shifted
            k = low.bit_length() - 1
            c = b + k + 1
            print(a, b, c)
            return

if __name__ == "__main__":
    solve()
