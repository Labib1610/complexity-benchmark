```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_bits = [0] * (n + 1)
    neigh = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bit_u = 1 << (u - 1)
        bit_v = 1 << (v - 1)
        adj_bits[u] |= bit_v
        adj_bits[v] |= bit_u
        neigh[u].append(v)
        neigh[v].append(u)

    for a in range(1, n + 1):
        # sort neighbors to ensure increasing order
        neigh[a].sort()
        for b in neigh[a]:
            if b <= a:
                continue
            common = adj_bits[a] & adj_bits[b]
            mask = common >> b          # discard bits <= b
            if mask:
                c_pos = (mask & -mask).bit_length() - 1
                c = b + 1 + c_pos
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
```
