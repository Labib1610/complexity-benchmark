def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
    
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    triads = []
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            for c in range(b + 1, n + 1):
                if b in adj[a] and c in adj[a] and c in adj[b]:
                    triads.append((a, b, c))
    
    lexicographically_smallest = triads[0]
    for triad in triads:
        if triad < lexicographically_smallest:
            lexicographically_smallest = triad
    
    print(lexicographically_smallest[0], lexicographically_smallest[1], lexicographically_smallest[2])

solve()
