```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency list represented as sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Iterate c from b+1 to n
    # Check if (a,b), (a,c), (b,c) are in adj
    
    # Optimization: Sort neighbors to speed up intersection
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n - 1):
        # Candidates for b must be neighbors of a
        neighbors_a = adj_list[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Candidates for c must be neighbors of both a and b
            # We look for c > b such that c is in adj[a] and adj[b]
            # Since adj_list[a] and adj_list[b] are sorted, we can intersect
            # or just iterate through the smaller neighbor list
            neighbors_b = adj[b]
            
            # To ensure smallest c, check all neighbors of a that are > b
            for c in neighbors_a:
                if c > b:
                    if c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return

solve()
```
