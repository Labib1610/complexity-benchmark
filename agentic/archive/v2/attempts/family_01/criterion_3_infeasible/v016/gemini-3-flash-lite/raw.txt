import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        # Candidates for b must be neighbors of a and b > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        # Iterate b from neighbors_a
        for b in neighbors_a:
            # Candidates for c must be neighbors of a AND neighbors of b, and c > b
            # We intersect the neighbors of a and b that are > b
            # Optimization: iterate through the smaller neighbor set
            if len(adj[a]) < len(adj[b]):
                common = [x for x in adj[a] if x > b and x in adj[b]]
            else:
                common = [x for x in adj[b] if x > b and x in adj[a]]
            
            if common:
                c = min(common)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
