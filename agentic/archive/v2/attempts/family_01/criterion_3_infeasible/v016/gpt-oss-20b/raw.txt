```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    # adjacency matrix (1-indexed)
    adj = [bytearray(n + 1) for _ in range(n + 1)]
    # adjacency list for neighbors
    neigh = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u][v] = 1
        adj[v][u] = 1
        neigh[u].append(v)
        neigh[v].append(u)

    for a in range(1, n - 1):
        # neighbors of a sorted
        for b in sorted(neigh[a]):
            if b <= a:
                continue
            for c in sorted(neigh[b]):
                if c <= b:
                    continue
                if adj[a][c]:
                    sys.stdout.write(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    main()
```
