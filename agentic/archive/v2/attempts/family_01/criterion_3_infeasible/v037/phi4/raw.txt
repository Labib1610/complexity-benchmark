def find_smallest_triplet():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    graph = {i: set() for i in range(1, n + 1)}
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        graph[u].add(v)
        graph[v].add(u)
        index += 2
    
    def is_triplet(a, b, c):
        return (b in graph[a] and c in graph[a]) and (c in graph[b])

    for a in range(1, n + 1):
        neighbors_a = sorted(graph[a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c <= b:
                    continue
                if is_triplet(a, b, c):
                    print(f"{a} {b} {c}")
                    return
