```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neighbors_mask = [0] * (n + 1)
    adjacency = [[] for _ in range(n + 1)]

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bit_u = 1 << (u - 1)
        bit_v = 1 << (v - 1)
        neighbors_mask[u] |= bit_v
        neighbors_mask[v] |= bit_u
        adjacency[u].append(v)
        adjacency[v].append(u)

    for a in range(1, n + 1):
        neigh_a = neighbors_mask[a]
        for b in sorted(adjacency[a]):
            if b <= a:
                continue
            inter = neigh_a & neighbors_mask[b]
            # clear bits <= b (positions 0..b-1)
            mask = inter & (~((1 << b) - 1))
            if mask:
                low = mask & -mask
                c = (low.bit_length() - 1) + 1
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
```
