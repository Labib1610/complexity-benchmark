def solve():
    n, m = map(int, input().split())
    incompatible_pairs = set()
    for _ in range(m):
        u, v = map(int, input().split())
        incompatible_pairs.add(tuple(sorted((u, v))))
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            for c in range(b + 1, n + 1):
                if (a, b) in incompatible_pairs and (a, c) in incompatible_pairs and (b, c) in incompatible_pairs:
                    print(a, b, c)
                    return

solve()
