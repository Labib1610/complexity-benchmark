import sys

def find_smallest_triplet(n, m, links):
    adj = [set() for _ in range(n + 1)]
    
    for u, v in links:
        adj[u].add(v)
        adj[v].add(u)

    for a in range(1, n - 1):
        for b in adj[a]:
            if b > a:
                for c in adj[b]:
                    if c > b and c in adj[a]:
                        return (a, b, c)

def main():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    links = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        links.append((u, v))
        index += 2
    
    result = find_smallest_triplet(n, m, links)
    print(result[0], result[1], result[2])

if __name__ == "__main__":
    main()
