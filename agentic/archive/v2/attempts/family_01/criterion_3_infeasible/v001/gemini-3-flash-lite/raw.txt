```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and c is in adj[b]
    
    # Optimization: Sort adjacency lists to potentially speed up checks
    # But since we need lexicographical order, we iterate a, b, c in order.
    # The set lookup is O(1).
    
    for a in range(1, n + 1):
        # Candidates for b must be neighbors of a and > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        for b in neighbors_a:
            # Candidates for c must be neighbors of both a and b and > b
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
```
