import sys

def find_conflict_triplet(n, m, pairs):
    # Create a set for quick lookup of incompatible pairs
    incompatible = set()
    
    # Read all incompatible pairs into a set for O(1) average time complexity lookups
    for u, v in pairs:
        if u > v:
            u, v = v, u
        incompatible.add((u, v))
        
    # Iterate over all possible triplets (a, b, c) with 1 <= a < b < c <= n
    for a in range(1, n - 1):
        for b in range(a + 1, n):
            for c in range(b + 1, n + 1):
                # Check if all pairs (a,b), (a,c), (b,c) are in the incompatible set
                if ((a, b) in incompatible and (a, c) in incompatible and (b, c) in incompatible):
                    return a, b, c

# Read input from standard input
input = sys.stdin.read
data = input().split()

n = int(data[0])
m = int(data[1])

pairs = []
index = 2
for _ in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    pairs.append((u, v))
    index += 2

# Find and print the conflict triplet
a, b, c = find_conflict_triplet(n, m, pairs)
print(a, b, c)
