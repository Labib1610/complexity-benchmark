def find_smallest_incompatibility_triplet(n, m, conflicts):
    # Create an adjacency list for each node
    graph = {i: set() for i in range(1, n + 1)}
    
    # Fill the adjacency list with conflicts
    for u, v in conflicts:
        if u < v:
            graph[u].add(v)
        else:
            graph[v].add(u)

    # Iterate over all possible triplets (a, b, c) with a < b < c
    for a in range(1, n + 1):
        neighbors_a = sorted(graph[a])
        for b in neighbors_a:
            if b <= a:
                continue
            neighbors_b = graph[b]
            common_neighbors = neighbors_a + [x for x in neighbors_b if x > b]

            # Sort the combined list of neighbors to find smallest triplet
            common_neighbors.sort()
            
            for c in common_neighbors:
                if c <= b or c not in graph[a]:
                    continue
                return a, b, c

# Read input from standard input
import sys
input = sys.stdin.read
data = input().split()

n = int(data[0])
m = int(data[1])

conflicts = []
index = 2
for _ in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    conflicts.append((u, v))
    index += 2

# Find and print the smallest incompatibility triplet
a, b, c = find_smallest_incompatibility_triplet(n, m, conflicts)
print(a, b, c)
