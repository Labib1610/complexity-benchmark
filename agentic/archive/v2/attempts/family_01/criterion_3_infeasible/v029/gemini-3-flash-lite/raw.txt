import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Store adjacency as a list of sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find the lexicographically smallest triplet
    # We want smallest a, then smallest b, then smallest c.
    # Iterating through sorted edges (u, v) where u < v
    edges.sort()
    
    # To find the lexicographically smallest triplet (a, b, c)
    # Iterate through all possible 'a' from 1 to n
    for a in range(1, n + 1):
        # Candidates for b must be connected to a and b > a
        neighbors_a = sorted([b for b in adj[a] if b > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            # Now we need a c such that c > b, c is in adj[a], and c is in adj[b]
            # Since we want the smallest (a, b, c), we check sorted neighbors
            # Intersection of adj[a] and adj[b]
            # To optimize, check the smaller set
            if len(adj[a]) < len(adj[b]):
                candidates = adj[a]
                other = adj[b]
            else:
                candidates = adj[b]
                other = adj[a]
            
            for c in sorted(candidates):
                if c > b and c in other:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
