import sys

def find_mutual_interference():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    interference = [set() for _ in range(n + 1)]
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        interference[u].add(v)
        interference[v].add(u)
        index += 2
    
    for i in range(1, n + 1):
        neighbors = sorted(interference[i])
        for j in range(len(neighbors)):
            x = neighbors[j]
            for k in range(j + 1, len(neighbors)):
                y = neighbors[k]
                if y in interference[x]:
                    print(i, x, y)
                    sys.exit()
