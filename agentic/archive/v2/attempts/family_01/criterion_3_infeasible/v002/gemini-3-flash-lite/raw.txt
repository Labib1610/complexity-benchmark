import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # Sort edges to facilitate finding lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (a, b) and check if there exists c > b 
    # such that (a, c) and (b, c) are edges.
    # To ensure lexicographical smallest, we can sort nodes and check.
    
    # Optimization: iterate through all edges (u, v) and find common neighbors
    # Since we want the smallest a, then smallest b, then smallest c:
    for a in range(1, n + 1):
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we have a and b, look for smallest c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            c_candidates = sorted([x for x in common if x > b])
            
            if c_candidates:
                print(f"{a} {b} {c_candidates[0]}")
                return

if __name__ == '__main__':
    solve()
