import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c),
    # we iterate a from 1 to n, b from a+1 to n, and c from b+1 to n.
    # To optimize, we only check b if a is incompatible with b,
    # and only check c if a is incompatible with c AND b is incompatible with c.
    
    # Pre-sort neighbors for consistent iteration
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Candidates for b must be neighbors of a
        neighbors_a = adj_list[a]
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Candidates for c must be neighbors of both a and b
            # Intersection of neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # We want the smallest c > b. 
            # Iterate through neighbors of a that are > b
            for c in neighbors_a:
                if c <= b:
                    continue
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
