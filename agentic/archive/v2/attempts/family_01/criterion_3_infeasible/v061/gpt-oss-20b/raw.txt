import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    n = int(data[0])
    adjacency = [0] * (n + 1)
    adjacency_lists = [[] for _ in range(n + 1)]

    # Read edges two by two
    for i in range(1, len(data), 2):
        u = int(data[i])
        v = int(data[i + 1])
        if u == v:
            continue
        adjacency[u] |= 1 << (v - 1)
        adjacency[v] |= 1 << (u - 1)
        adjacency_lists[u].append(v)
        adjacency_lists[v].append(u)

    # Ensure ascending order of neighbours for deterministic search
    for lst in adjacency_lists[1:]:
        lst.sort()

    for a in range(1, n - 1):
        for b in adjacency_lists[a]:
            if b <= a:
                continue
            common = adjacency[a] & adjacency[b]
            # Shift so that only vertices > b remain
            shifted = common >> b
            if shifted == 0:
                continue
            # Find the smallest vertex in the shifted set
            lowest_bit = shifted & -shifted
            pos = lowest_bit.bit_length() - 1
            c = b + pos + 1
            print(a, b, c)
            return

if __name__ == "__main__":
    solve()
