import sys

# Read inputs
input = sys.stdin.read
data = input().split()
n, m = int(data[0]), int(data[1])
incompatibilities = [tuple(map(int, data[i:i+2])) for i in range(2, len(data), 2)]

# Create a set of tuples to ensure quick lookup and handle unordered pairs
incompatible_set = set((min(u, v), max(u, v)) for u, v in incompatibilities)

# Find the lexicographically smallest triplet
for a in range(1, n+1):
    for b in range(a+1, n+1):
        if (a, b) not in incompatible_set:
            continue
        for c in range(b+1, n+1):
            if (a, c) in incompatible_set and (b, c) in incompatible_set:
                print(a, b, c)
                sys.exit()
