```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
        idx += 2
        
    # Sort edges to help find lexicographically smallest (a, b, c)
    # We want smallest a, then smallest b, then smallest c.
    # An edge (u, v) with u < v is a candidate for (a, b).
    # We iterate through all edges (u, v) as potential (a, b) 
    # and check if there exists a c > v such that (u, c) and (v, c) exist.
    
    edges.sort()
    
    best = None
    
    # Iterate through each edge (a, b) where a < b
    for a, b in edges:
        # We need a c such that b < c and c in adj[a] and c in adj[b]
        # To minimize the triple (a, b, c), for a fixed (a, b), 
        # we want the smallest c > b.
        
        # Intersection of adj[a] and adj[b]
        # Only check elements larger than b
        common = adj[a].intersection(adj[b])
        
        candidates = [c for c in common if c > b]
        if candidates:
            c = min(candidates)
            if best is None or (a, b, c) < best:
                best = (a, b, c)
                
        # Optimization: if we already have a best and current a > best[0],
        # we can stop because any subsequent (a, b, c) will have a larger a.
        if best is not None and a > best[0]:
            break
            
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
```
