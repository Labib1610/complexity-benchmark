import sys

def find_conflict_group(n, m, conflicts):
    # Create an adjacency list for workers with conflicts
    adj = [[] for _ in range(n + 1)]
    
    for u, v in conflicts:
        if u > v:
            u, v = v, u
        adj[u].append(v)
        
    # Sort the adjacency lists to help find lexicographically smallest triplet
    for i in range(1, n + 1):
        adj[i].sort()
    
    # Iterate through each worker and check possible triplets
    for a in range(1, n + 1):
        for b in adj[a]:
            if b > a:
                # Check for mutual conflicts between a, b and any other worker c
                for c in adj[b]:
                    if c > b and a in adj[c]:  # Ensure the triplet (a < b < c) with all conflicts
                        return a, b, c

def main():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    conflicts = [(int(data[i*2 + 2]), int(data[i*2 + 3])) for i in range(m)]
    
    result = find_conflict_group(n, m, conflicts)
    print(*result)

if __name__ == "__main__":
    main()
