import sys

def find_smallest_incompatible_triplet():
    # Read number of workers and incompatible pairs
    n, m = map(int, input().split())
    
    # Create a set for each worker to store incompatible workers
    incompatibilities = [set() for _ in range(n + 1)]
    
    # Read the incompatible pairs
    for _ in range(m):
        u, v = map(int, input().split())
        if len(incompatibilities[u]) < 2:
            incompatibilities[u].add(v)
        if len(incompatibilities[v]) < 2:
            incompatibilities[v].add(u)

    # Find the lexicographically smallest triplet (a, b, c) such that
    # a < b < c and all pairs among them are incompatible.
    for a in range(1, n + 1):
        for b in incompatibilities[a]:
            if b > a:
                for c in incompatibilities[a]:
                    if c > b and c in incompatibilities[b]:
                        print(a, b, c)
                        return

if __name__ == "__main__":
    find_smallest_incompatible_triplet()
