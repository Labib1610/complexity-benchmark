import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets for fast O(1) lookup
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a, then check for a common neighbor c > b
    
    # Convert adjacency sets to sorted lists for deterministic iteration if needed,
    # though simply checking existence in sets is sufficient.
    
    for a in range(1, n + 1):
        # Candidates for b must be connected to a and b > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for b in neighbors_a:
            # We need a c such that c > b, c is connected to a, and c is connected to b
            # This is the intersection of adj[a] and adj[b]
            # Since we need the smallest triplet, we look for the smallest c > b
            
            # Optimization: iterate through the smaller adjacency list
            if len(adj[a]) < len(adj[b]):
                common = [x for x in adj[a] if x > b and x in adj[b]]
            else:
                common = [x for x in adj[b] if x > b and x in adj[a]]
            
            if common:
                c = min(common)
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
