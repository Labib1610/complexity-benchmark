import sys

def find_highest_priority_vulnerability():
    # Reading input
    n, m = map(int, input().split())
    
    prohibited_pairs = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, input().split())
        prohibited_pairs[u].add(v)
        prohibited_pairs[v].add(u)
    
    # Checking for the highest-priority triple
    for a in range(1, n + 1):
        for b in sorted(prohibited_pairs[a]):
            if b > a:
                for c in sorted(prohibited_pairs[a]):
                    if c > b and c in prohibited_pairs[b]:
                        print(a, b, c)
                        return

find_highest_priority_vulnerability()
