import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency list using sets for O(1) lookups
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and adj[b]
    
    # Optimization: Sort adjacency lists to potentially break early
    # But since we need the smallest triple, simple iteration is correct
    # and efficient enough for n=2000 given the constraints and bitset/set speed.
    
    # Pre-sorting adjacency lists for faster intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # Use bitsets or sets. For n=2000, bitsets are very fast.
    # Convert adjacency lists to bitsets (integers in Python)
    adj_bits = [0] * (n + 1)
    for i in range(1, n + 1):
        for neighbor in adj[i]:
            adj_bits[i] |= (1 << neighbor)
            
    for a in range(1, n + 1):
        # Candidates for b are neighbors of a that are > a
        neighbors_a = [x for x in adj[a] if x > a]
        for b in neighbors_a:
            # Candidates for c are neighbors of both a and b that are > b
            common = adj_bits[a] & adj_bits[b]
            # We want smallest c > b
            # Shift right by b + 1 and find lowest bit
            target = common >> (b + 1)
            if target > 0:
                # Get the index of the lowest set bit
                c = (target & -target).bit_length() + b
                print(f"{a} {b} {c}")
                return

if __name__ == '__main__':
    solve()
