import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_bits = [0] * n
    adj_lists = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj_bits[u] |= 1 << v
        adj_bits[v] |= 1 << u
        adj_lists[u].append(v)
        adj_lists[v].append(u)

    for a in range(n):
        for b in adj_lists[a]:
            if b <= a:
                continue
            mask = adj_bits[a] & adj_bits[b]
            shifted = mask >> (b + 1)
            if shifted:
                lsb = shifted & -shifted
                c = (b + 1) + (lsb.bit_length() - 1)
                print(a + 1, b + 1, c + 1)
                return

if __name__ == "__main__":
    solve()
