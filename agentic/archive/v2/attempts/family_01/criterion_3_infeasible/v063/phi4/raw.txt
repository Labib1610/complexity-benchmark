```python
import sys
from collections import defaultdict

def main():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        if u > v:
            u, v = v, u
        edges.append((u, v))
        index += 2
    
    adj_list = defaultdict(list)
    for u, v in edges:
        adj_list[u].append(v)
        adj_list[v].append(u)

    edges.sort()

    smallest_triplet = (float('inf'), float('inf'), float('inf'))

    for i in range(m):
        a, b = edges[i]
        
        if b < a:
            continue
        
        common_neighbors = set(adj_list[a]) & set(adj_list[b])
        
        for c in common_neighbors:
            if c > b:
                smallest_triplet = min(smallest_triplet, (a, b, c))
    
    print(f"{smallest_triplet[0]} {smallest_triplet[1]} {smallest_triplet[2]}")

if __name__ == "__main__":
    main()
```
