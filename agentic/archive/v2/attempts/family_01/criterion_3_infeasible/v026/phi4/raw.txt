import sys

def find_smallest_clique(n, edges):
    adj = [set() for _ in range(n + 1)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)

    # Try to find the smallest clique (a < b < c)
    for a in range(1, n + 1):
        for b in sorted(adj[a]):
            if b > a:
                common_neighbors = adj[a] & adj[b]
                for c in sorted(common_neighbors):
                    if c > b:
                        return a, b, c

# Read input
input_data = sys.stdin.read().splitlines()
n, m = map(int, input_data[0].split())
edges = [tuple(map(int, line.split())) for line in input_data[1:]]

# Find and print the smallest clique
result = find_smallest_clique(n, edges)
print(' '.join(map(str, result)))
