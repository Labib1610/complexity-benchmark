import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_bits = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj_bits[u] |= 1 << v
        adj_bits[v] |= 1 << u

    for a in range(n - 2):
        bits_a = adj_bits[a]
        for b in range(a + 1, n - 1):
            if not (bits_a >> b) & 1:
                continue
            bits_common = bits_a & adj_bits[b]
            mask = bits_common >> (b + 1)
            if mask == 0:
                continue
            offset = (mask & -mask).bit_length() - 1
            c = b + 1 + offset
            print(a + 1, b + 1, c + 1)
            return

if __name__ == "__main__":
    solve()
