import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    for a in range(1, n - 1):
        mask_a = adj[a]
        for b in range(a + 1, n + 1):
            if not (mask_a >> (b - 1)) & 1:
                continue
            mask_common = mask_a & adj[b]
            # Remove bits <= b
            mask_shifted = mask_common >> b
            if mask_shifted:
                # lowest set bit
                p = (mask_shifted & -mask_shifted).bit_length() - 1
                c = p + b + 1
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
