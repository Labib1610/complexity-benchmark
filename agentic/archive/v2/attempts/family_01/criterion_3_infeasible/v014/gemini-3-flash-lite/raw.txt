import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        idx += 2
    
    # Sort edges to ensure we process in lexicographical order
    # An edge (a, b) with a < b
    edges.sort()
    
    # To find the lexicographically smallest triplet (a, b, c) with a < b < c:
    # We iterate through all edges (a, b). For each edge, we look for a node c
    # such that c > b and both (a, c) and (b, c) are in the adjacency set.
    # To get the lexicographically smallest (a, b, c), we iterate a from 1 to n,
    # then b in increasing order, then check common neighbors c.
    
    # Pre-sort adjacency lists to make finding c easier
    sorted_adj = [sorted(list(s)) for s in adj]
    
    # We want smallest a, then b, then c.
    # Iterate through all possible a
    for a in range(1, n + 1):
        # Iterate through all possible b > a
        # Only consider b that are connected to a
        for b in sorted_adj[a]:
            # Now we need a c such that b < c and c is connected to both a and b
            # Iterate through c in the intersection of adj[a] and adj[b]
            # Since sorted_adj[a] and sorted_adj[b] are sorted, we can use two pointers
            # or simply iterate through the smaller set.
            
            # Using set intersection for efficiency:
            # We need c > b
            set_a = adj[a]
            set_b = adj[b]
            
            # Optimization: iterate through the smaller adjacency list
            if len(set_a) < len(set_b):
                candidates = set_a
            else:
                candidates = set_b
                
            for c in candidates:
                if c > b:
                    if c in set_a and c in set_b:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
