import sys

def find_first_conflict_triplet():
    input = sys.stdin.read
    data = input().split()
    
    index = 0
    n = int(data[index])
    m = int(data[index + 1])
    index += 2
    
    overlap_graph = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        overlap_graph[u].append(v)
        overlap_graph[v].append(u)
    
    # Sort adjacency lists to ensure we pick the smallest lexicographically
    for i in range(1, n + 1):
        overlap_graph[i] = sorted(overlap_graph[i])
    
    # Find the lexicographically smallest triplet (a, b, c) such that
    # a < b < c and there is an edge between every pair of these three nodes.
    for a in range(1, n + 1):
        neighbors_a = overlap_graph[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c <= b:
                    continue
                if c in overlap_graph[b]:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    find_first_conflict_triplet()
