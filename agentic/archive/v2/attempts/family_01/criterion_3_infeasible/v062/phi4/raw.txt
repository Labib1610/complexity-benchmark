def find_smallest_coupling_loop(n, m, couplings):
    # Create an adjacency list for the graph representation of modules
    adj = [[] for _ in range(n + 1)]
    
    # Fill the adjacency list with given couplings
    for u, v in couplings:
        if u > v:  # Ensure that we always add (min(u,v), max(u,v))
            u, v = v, u
        adj[u].append(v)
        
    # Sort each module's coupling list to facilitate lexicographical comparison
    for i in range(1, n + 1):
        adj[i].sort()
    
    # Iterate over all possible triples (a, b, c) where a < b < c
    for a in range(1, n - 1):
        for b in adj[a]:
            if b > a:
                for c in adj[b]:
                    if c > b and c in adj[a]:  # Check mutual coupling among a, b, c
                        return (a, b, c)
    return None

# Read input from standard input
import sys
input = sys.stdin.read
data = input().split()

n = int(data[0])
m = int(data[1])

couplings = []
index = 2
for _ in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    couplings.append((u, v))
    index += 2

# Find and print the lexicographically smallest triple (a, b, c)
result = find_smallest_coupling_loop(n, m, couplings)
print(" ".join(map(str, result)))
