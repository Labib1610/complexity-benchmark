import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if there exists c > b such that c is in adj[a] and c is in adj[b]
    
    # Pre-sorting adjacency lists to make intersection faster or checking easier
    sorted_adj = [sorted(list(adj[i])) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # We only care about neighbors b > a
        neighbors_a = [x for x in sorted_adj[a] if x > a]
        for b in neighbors_a:
            # We need a node c such that c > b, c in adj[a], and c in adj[b]
            # Since adj[a] and adj[b] are sorted, we can find common elements
            # efficiently.
            
            # Intersection of neighbors_a and neighbors_b
            # Since we need c > b, we filter neighbors_b for elements > b
            neighbors_b = sorted_adj[b]
            
            # Using two pointers to find intersection
            i = 0
            j = 0
            len_a = len(neighbors_a)
            len_b = len(neighbors_b)
            
            while i < len_a and j < len_b:
                val_a = neighbors_a[i]
                val_b = neighbors_b[j]
                
                if val_a == val_b:
                    if val_a > b:
                        print(f"{a} {b} {val_a}")
                        return
                    i += 1
                    j += 1
                elif val_a < val_b:
                    i += 1
                else:
                    j += 1

solve()
