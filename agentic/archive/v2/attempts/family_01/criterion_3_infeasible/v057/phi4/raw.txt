import sys

def find_conflict_triplet(n, restricted_pairs):
    restrictions = {i: set() for i in range(1, n + 1)}
    
    for u, v in restricted_pairs:
        restrictions[u].add(v)
        restrictions[v].add(u)

    # Sort the keys of restriction dictionary to ensure lexicographical order
    personnel_sorted = sorted(restrictions.keys())

    for a in personnel_sorted:
        neighbors_a = sorted(restrictions[a])
        for i, b in enumerate(neighbors_a):
            if a >= b:
                continue
            for c in neighbors_a[i + 1:]:
                if b >= c or c not in restrictions[b]:
                    continue
                return a, b, c

    # This point should never be reached as per problem guarantees.
    return None

# Read input from standard input
input = sys.stdin.read
data = input().split()
n = int(data[0])
m = int(data[1])

restricted_pairs = []
index = 2
for _ in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    restricted_pairs.append((u, v))
    index += 2

# Find the conflict triplet
triplet = find_conflict_triplet(n, restricted_pairs)

# Output the result to standard output
print(" ".join(map(str, triplet)))
