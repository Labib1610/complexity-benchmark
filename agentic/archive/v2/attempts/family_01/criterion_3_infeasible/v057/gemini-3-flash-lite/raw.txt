import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list of sets or boolean arrays for fast lookup
    # Since n <= 2000, an adjacency list is efficient enough
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Convert sets to sorted lists for deterministic iteration
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # We need to find the lexicographically smallest triplet (a, b, c)
    # where a < b < c and (a,b), (b,c), (a,c) are all edges.
    # We iterate a from 1 to n-2, b from a+1 to n-1, c from b+1 to n.
    
    # Optimization: iterate a, then b, then check if c exists in both adj[a] and adj[b]
    # To make it fast, convert adj[a] to a set for O(1) lookup
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we check b and c in lexicographical order
        neighbors_a = sorted(list(adj_sets[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now find c such that c is in adj[a] and adj[b] and c > b
            # We can intersect the neighbors of a and b
            # Since we need the smallest c for the current (a, b), we look at sorted neighbors
            common = adj_sets[a].intersection(adj_sets[b])
            for c in sorted(list(common)):
                if c > b:
                    print(f"{a} {b} {c}")
                    return

solve()
