```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    bitsets = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    edges = []

    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u > v:
            u, v = v, u
        bitsets[u] |= 1 << (v - 1)
        bitsets[v] |= 1 << (u - 1)
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))

    for lst in adj:
        lst.sort()

    for a in range(1, n + 1):
        for b in adj[a]:
            if b <= a:
                continue
            inter = bitsets[a] & bitsets[b]
            shifted = inter >> b
            if shifted:
                low = shifted & -shifted
                offset = low.bit_length() - 1
                c = b + offset + 1
                print(a, b, c)
                return

if __name__ == "__main__":
    solve()
```
