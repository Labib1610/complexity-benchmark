import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Convert sets to sorted lists to iterate in order
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # We want lexicographically smallest (a, b, c) with a < b < c.
    # Iterate a from 1 to n-2, b from a+1 to n-1, c from b+1 to n.
    # To optimize: for a fixed a and b, check if any common neighbor c exists.
    # Since we need the lexicographically smallest, we iterate in order.
    
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        # Only consider b > a
        for b in neighbors_a:
            if b <= a:
                continue
            
            # Now we need c > b such that c is in adj[a] and c is in adj[b]
            # Intersecting neighbors_a and neighbors_b
            # Since both are sorted, we can use two pointers or set intersection
            # Given the constraints and the guarantee, this will find the result.
            
            # Optimization: iterate through the smaller neighbor list
            if len(neighbors_a) < len(adj_list[b]):
                for c in neighbors_a:
                    if c > b and c in adj[b]:
                        print(f"{a} {b} {c}")
                        return
            else:
                for c in adj_list[b]:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
