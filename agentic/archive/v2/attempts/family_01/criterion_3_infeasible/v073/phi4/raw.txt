import sys

def find_protocol_violation():
    input = sys.stdin.read
    data = input().split()
    
    index = 0
    
    n = int(data[index])
    m = int(data[index + 1])
    index += 2
    
    links = [[False] * (n + 1) for _ in range(n + 1)]
    
    for i in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        links[u][v] = True
        links[v][u] = True
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if not links[a][b]:
                continue
            for c in range(b + 1, n + 1):
                if links[a][c] and links[b][c]:
                    print(f"{a} {b} {c}")
                    return
