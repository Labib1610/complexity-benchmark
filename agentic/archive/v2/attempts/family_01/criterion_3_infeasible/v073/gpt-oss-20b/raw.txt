import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)

    for a in range(1, n + 1):
        neigh_mask = adj[a] >> a  # bits for nodes > a
        temp = neigh_mask
        while temp:
            low_b = temp & -temp
            b = a + (low_b.bit_length() - 1) + 1  # node number
            common = adj[a] & adj[b]
            tmp = common >> b  # shift to discard nodes <= b
            if tmp:
                low_c = tmp & -tmp
                c = b + (low_c.bit_length() - 1) + 1
                print(a, b, c)
                return
            temp -= low_b

if __name__ == "__main__":
    solve()
