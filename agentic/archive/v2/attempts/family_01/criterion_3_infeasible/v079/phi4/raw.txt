```python
import sys

# Reading input data
input = sys.stdin.read
data = input().split()

n = int(data[0])
m = int(data[1])

edges = []
index = 2
for _ in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    edges.append((u, v))
    index += 2

# Initialize adjacency matrix and interference flag
inf = [[False] * (n + 1) for _ in range(n + 1)]

for u, v in edges:
    inf[u][v] = True
    inf[v][u] = True

# Find the lexicographically smallest triplet
answer = None
for a in range(1, n + 1):
    for b in range(a + 1, n + 1):
        if inf[a][b]:
            for c in range(b + 1, n + 1):
                if inf[b][c] and inf[c][a]:
                    if answer is None or (a, b, c) < answer:
                        answer = (a, b, c)

# Output the result
print(*answer)
```
