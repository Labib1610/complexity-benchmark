import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for fast lookup
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # Iterate c from b+1 to n
    # Check if c is connected to a and b
    
    # Sorting adjacency sets to potentially improve cache/speed
    adj_list = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        # Only consider b > a
        neighbors_a = [x for x in adj_list[a] if x > a]
        for b in neighbors_a:
            # Check for c > b such that c is in adj[a] AND adj[b]
            # Intersection of neighbors of a and b that are > b
            # Use sets for O(1) lookup
            set_a = adj[a]
            set_b = adj[b]
            
            # Optimization: iterate over the smaller adjacency list
            if len(set_a) < len(set_b):
                candidates = [x for x in set_a if x > b and x in set_b]
            else:
                candidates = [x for x in set_b if x > b and x in set_a]
            
            if candidates:
                print(f"{a} {b} {min(candidates)}")
                return

if __name__ == '__main__':
    solve()
