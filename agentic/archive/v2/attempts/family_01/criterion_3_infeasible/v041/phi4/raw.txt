import sys
from collections import defaultdict

def find_critical_vulnerability():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    index = 2
    conflicts = defaultdict(set)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        conflicts[u].add(v)
        conflicts[v].add(u)
        index += 2
    
    for a in range(1, n + 1):
        if len(conflicts[a]) < 2:
            continue
        
        possible_b = sorted(list(conflicts[a]))
        
        for b_index in range(len(possible_b)):
            b = possible_b[b_index]
            
            if b <= a:
                continue
            
            common_conflict = conflicts[a] & conflicts[b]
            
            for c in possible_b[b_index + 1:]:
                if c > b and c in common_conflict:
                    print(a, b, c)
                    return
