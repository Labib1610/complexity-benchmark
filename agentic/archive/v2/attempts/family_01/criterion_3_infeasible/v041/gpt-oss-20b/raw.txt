import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_list = [[] for _ in range(n + 1)]
    adj_bit = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj_list[u].append(v)
        adj_list[v].append(u)
        adj_bit[u] |= 1 << (v - 1)
        adj_bit[v] |= 1 << (u - 1)

    for a in range(1, n - 1):
        nbrs = [v for v in adj_list[a] if v > a]
        if len(nbrs) < 2:
            continue
        nbrs.sort()
        L = len(nbrs)
        for i in range(L - 1):
            b = nbrs[i]
            bit_b = adj_bit[b]
            for j in range(i + 1, L):
                c = nbrs[j]
                if bit_b >> (c - 1) & 1:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
