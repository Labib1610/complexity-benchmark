```python
from sys import stdin

# Read number of units and channels
n, m = map(int, stdin.readline().split())

# Initialize adjacency list for each unit with self-loop to simplify checks
adjacency_list = [[] for _ in range(n)]
for i in range(m):
    u, v = map(lambda x: int(x) - 1, stdin.readline().split())
    adjacency_list[u].append(v)
    adjacency_list[v].append(u)

# Set the units in each list to be unique and sorted
for unit_adjacency in adjacency_list:
    unit_adjacency.sort()
    unit_adjacency[:] = set(unit_adjacency)

# Iterate over all possible triples (a, b, c) where a < b < c
for a in range(n):
    for b in adjacency_list[a]:
        if b > a:  # Ensure b > a to maintain lexicographical order
            common_units = sorted(set(adjacency_list[b]).intersection(adjacency_list[a]))
            for c in common_units:
                if c > b:  # Ensure c > b to maintain lexicographical order
                    print(a + 1, b + 1, c + 1)
                    exit(0)
```
