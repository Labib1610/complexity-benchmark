```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    mask = [0] * (n + 1)
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        mask[u] |= 1 << (v - 1)
        mask[v] |= 1 << (u - 1)
        adj[u].append(v)
        adj[v].append(u)
    for lst in adj:
        lst.sort()
    for a in range(1, n + 1):
        for b in adj[a]:
            if b > a:
                inter = mask[a] & mask[b]
                shifted = inter >> b
                if shifted:
                    low = shifted & -shifted
                    pos = low.bit_length() - 1
                    c = pos + b + 1
                    sys.stdout.write(f"{a} {b} {c}")
                    return

if __name__ == "__main__":
    main()
```
