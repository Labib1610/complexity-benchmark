```python
import sys
input = sys.stdin.read

def find_lexicographically_smallest_triad():
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx + 1])
    idx += 2
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        adj[u].add(v)
        adj[v].add(u)
        idx += 2

    for a in range(n):
        neighbors_a = adj[a]
        for b in neighbors_a:
            if b <= a: 
                continue
            neighbors_b = adj[b]
            common_neighbors = neighbors_a & neighbors_b
            for c in sorted(common_neighbors):
                if c > b:
                    print(a + 1, b + 1, c + 1)
                    return

if __name__ == "__main__":
    find_lexicographically_smallest_triad()
```
