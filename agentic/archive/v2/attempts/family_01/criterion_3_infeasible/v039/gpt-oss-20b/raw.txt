```python
import sys

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)
    adj = [set() for _ in range(n)]
    bit = [0] * n
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        adj[u].add(v)
        adj[v].add(u)
        bit[u] |= 1 << v
        bit[v] |= 1 << u

    for a in range(n - 2):
        # neighbors of a that are greater than a
        neigh = [v for v in adj[a] if v > a]
        if len(neigh) < 2:
            continue
        neigh.sort()
        ln = len(neigh)
        for i in range(ln - 1):
            b = neigh[i]
            for j in range(i + 1, ln):
                c = neigh[j]
                if (bit[b] >> c) & 1:
                    # lexicographically smallest found
                    sys.stdout.write(f"{a+1} {b+1} {c+1}")
                    return

if __name__ == "__main__":
    main()
```
