def solve():
    n, m = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)
    
    ans = None
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            for c in range(b + 1, n + 1):
                if b in adj[a] and c in adj[a] and c in adj[b]:
                    if ans is None:
                        ans = (a, b, c)
                    else:
                        if (a, b, c) < ans:
                            ans = (a, b, c)
    
    print(ans[0], ans[1], ans[2])

solve()
