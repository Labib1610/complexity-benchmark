```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use bitsets (Python large integers) for efficient adjacency checking
    # adj[i] will store a bitmask of all nodes j such that (i, j) is an interference pair
    adj = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if a and b are connected
    # If connected, check if there exists c > b such that c is connected to both a and b
    # c must be in the intersection of adj[a] and adj[b]
    
    for a in range(1, n + 1):
        # Candidates for b must be connected to a
        adj_a = adj[a]
        # Iterate b in increasing order
        for b in range(a + 1, n + 1):
            if (adj_a >> b) & 1:
                # b is connected to a, now find smallest c > b connected to both
                common = adj_a & adj[b]
                # We need the smallest bit set in 'common' that is > b
                # Mask out bits <= b
                mask = common >> (b + 1)
                if mask > 0:
                    # Find index of lowest set bit
                    # bit_length() gives position of highest bit, 
                    # but we can use (mask & -mask).bit_length() - 1
                    # Actually, (mask & -mask) is the lowest bit.
                    c = (mask & -mask).bit_length() + b
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
```
