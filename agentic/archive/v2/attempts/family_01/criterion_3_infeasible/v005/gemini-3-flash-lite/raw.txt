import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Adjacency matrix for O(1) lookup
    # Given n <= 2000, 2000*2000 bits/bools is fine
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # We want the lexicographically smallest (a, b, c) with a < b < c
    # Iterate a from 1 to n-2
    for a in range(1, n - 1):
        # Get neighbors of a as potential b, c
        neighbors_a = sorted(list(adj[a]))
        
        # We need b, c in neighbors_a such that b < c and (b, c) is an edge
        # To ensure lexicographically smallest, iterate b then c
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now find c in neighbors_a such that c > b and c is in adj[b]
            # To optimize, we can intersect neighbors_a and neighbors_b
            # Since we need the smallest c, we can iterate through neighbors_b
            # or check neighbors_a
            
            # Optimization: check the smaller neighbor list
            neighbors_b = adj[b]
            
            # Find common neighbors of a and b that are > b
            # Use the smaller list to iterate
            if len(neighbors_a) < len(neighbors_b):
                for c in neighbors_a:
                    if c > b and c in neighbors_b:
                        print(f"{a} {b} {c}")
                        return
            else:
                # Iterate through sorted neighbors of b to find smallest c
                sorted_neighbors_b = sorted(list(neighbors_b))
                for c in sorted_neighbors_b:
                    if c > b and c in adj[a]:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
