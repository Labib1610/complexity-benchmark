```python
import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use an adjacency list to store the graph
    # Use a bitset-like approach or adjacency sets for fast intersection
    adj = [set() for _ in range(n + 1)]
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is connected to a
    # If so, find common neighbors of a and b that are > b
    # The smallest such neighbor will be the smallest c
    
    for a in range(1, n + 1):
        # We only care about neighbors b > a
        neighbors_a = sorted([x for x in adj[a] if x > a])
        
        for b in neighbors_a:
            # Check for potential c
            # c must be a neighbor of a AND a neighbor of b
            # and c > b
            
            # Intersection of adj[a] and adj[b]
            # Since we need the smallest c > b, we can iterate
            # through the smaller set and check membership
            
            set_a = adj[a]
            set_b = adj[b]
            
            # Optimization: iterate through the smaller set to find c
            if len(set_a) < len(set_b):
                candidates = set_a
                check_set = set_b
            else:
                candidates = set_b
                check_set = set_a
            
            best_c = float('inf')
            found = False
            for c in candidates:
                if c > b:
                    if c in check_set:
                        if c < best_c:
                            best_c = c
                            found = True
            
            if found:
                print(f"{a} {b} {best_c}")
                return

if __name__ == '__main__':
    solve()
```
