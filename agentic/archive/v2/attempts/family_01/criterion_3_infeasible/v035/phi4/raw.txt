def find_safety_audit_trio():
    import sys
    input = sys.stdin.read
    
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    restrictions = [set() for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        restrictions[u].add(v)
        restrictions[v].add(u)
        index += 2
    
    for a in range(1, n + 1):
        for b in restrictions[a]:
            if a < b:
                for c in restrictions[b]:
                    if a < c and b < c and a in restrictions[c]:
                        print(a, b, c)
                        return
