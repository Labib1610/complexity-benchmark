```python
import sys

def main():
    data = sys.stdin.buffer.read().split()
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    neighbor = [0] * n
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        neighbor[u] |= 1 << v
        neighbor[v] |= 1 << u
        adj[u].append(v)
        adj[v].append(u)

    for a in range(n):
        for b in sorted(adj[a]):
            if b <= a:
                continue
            # intersection of neighbors of a and b
            common = neighbor[a] & neighbor[b]
            shifted = common >> b
            if shifted:
                low = shifted & -shifted
                pos = low.bit_length() - 1
                c = b + pos + 1
                # output 1-indexed
                sys.stdout.write(f"{a+1} {b+1} {c}\n")
                return

if __name__ == "__main__":
    main()
```
