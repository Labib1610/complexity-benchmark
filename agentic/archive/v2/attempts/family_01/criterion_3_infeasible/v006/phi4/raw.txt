import sys

def find_conflict_triple(n, m, conflicts):
    # Create an adjacency list for graph representation of conflicts
    adj = {i: set() for i in range(1, n+1)}
    for u, v in conflicts:
        adj[u].add(v)
        adj[v].add(u)

    # Iterate over each node and its neighbors to find a triple (a, b, c)
    for a in sorted(adj.keys()):
        neighbors_a = sorted(adj[a])
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                # Check if all pairs (a, b), (a, c), and (b, c) are conflicts
                if c in adj[b]:
                    return a, b, c

# Read input
input_data = sys.stdin.read().strip().split()
n, m = int(input_data[0]), int(input_data[1])
conflicts = [tuple(map(int, input_data[i:i+2])) for i in range(2, len(input_data), 2)]

# Find and print the lexicographically smallest conflict triple
result = find_conflict_triple(n, m, conflicts)
print(" ".join(map(str, result)))
