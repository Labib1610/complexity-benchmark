import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    bitmask = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        bitmask[u] |= 1 << (v - 1)
        bitmask[v] |= 1 << (u - 1)

    for a in range(1, n - 1):
        neigh_a = bitmask[a]
        for b in range(a + 1, n + 1):
            if not ((neigh_a >> (b - 1)) & 1):
                continue  # no edge a-b
            inter = neigh_a & bitmask[b]
            shift = inter >> b  # bits for nodes > b
            if shift:
                pos = (shift & -shift).bit_length() - 1
                c = b + 1 + pos
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
