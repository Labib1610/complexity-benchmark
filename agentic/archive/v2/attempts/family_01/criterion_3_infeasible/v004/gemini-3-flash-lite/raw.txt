import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find lexicographically smallest (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    #   Iterate b from a+1 to n-1
    #     Check if (a, b) is an edge
    #       Iterate c from b+1 to n
    #         Check if (a, c) and (b, c) are edges
    
    # Optimization: Sort adjacency lists to speed up intersection
    for i in range(1, n + 1):
        adj[i] = sorted(list(adj[i]))
    
    # To make the intersection faster, use adjacency sets for lookups
    adj_sets = [set(adj[i]) for i in range(n + 1)]
    
    for a in range(1, n + 1):
        # Candidates for b are neighbors of a
        neighbors_a = adj[a]
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c > b such that c is in adj[a] and adj[b]
            # Since neighbors_a is sorted, we can iterate through it
            # starting from index i+1
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c > b and c in adj_sets[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
