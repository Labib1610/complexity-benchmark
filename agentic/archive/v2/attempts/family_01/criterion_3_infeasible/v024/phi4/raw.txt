```python
import sys

input = sys.stdin.read
data = input().split()
n, m = int(data[0]), int(data[1])
edges = [(int(data[i*2 + 2]), int(data[i*2 + 3])) for i in range(m)]

adjacency_list = [[] for _ in range(n + 1)]
for u, v in edges:
    adjacency_list[u].append(v)
    adjacency_list[v].append(u)

# Ensure each unit's connections are sorted
for i in range(1, n + 1):
    adjacency_list[i] = sorted(adjacency_list[i])

def find_primary_feedback_cluster():
    for a in range(1, n - 1):
        if len(adjacency_list[a]) > (n - 2):  # Only consider units with enough connections
            for b in adjacency_list[a]:
                if b < a:
                    continue
                for c in adjacency_list[a][adjacency_list[a].index(b):]:  # Start from b to ensure a < b < c
                    if c > b and c in adjacency_list[b]:
                        return (a, b, c)

a, b, c = find_primary_feedback_cluster()
print(a, b, c)
```
