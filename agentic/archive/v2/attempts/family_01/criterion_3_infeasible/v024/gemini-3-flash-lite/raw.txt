import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) edge lookups
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate through a from 1 to n-2
    # Iterate through b from a+1 to n-1
    # Iterate through c from b+1 to n
    # Check if (a, b), (b, c), and (a, c) exist.
    # Since we want the smallest, we can break as soon as we find one.
    
    # Optimization: Sort adjacency lists to potentially prune or iterate faster
    # but with set lookups, the nested loops are O(n^2) or slightly better 
    # depending on density. Given N=2000, O(N^2) is acceptable.
    
    for a in range(1, n + 1):
        # Sort neighbors to try smaller b and c first
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Now we need a c > b such that c is in adj[a] and adj[b]
            # Intersecting neighbors_a and neighbors_b
            # Since neighbors_a is sorted, we can iterate through it
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c > b and c in adj[b]:
                    print(f"{a} {b} {c}")
                    return

if __name__ == '__main__':
    solve()
