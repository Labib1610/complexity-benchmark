import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_bits = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj_bits[u] |= 1 << (v - 1)
        adj_bits[v] |= 1 << (u - 1)

    for a in range(1, n + 1):
        bits_a = adj_bits[a]
        for b in range(a + 1, n + 1):
            # check edge a-b
            if not (bits_a >> (b - 1)) & 1:
                continue
            # intersection of neighbors of a and b, shifted to ignore indices <= b
            mask = (bits_a & adj_bits[b]) >> b
            if mask:
                # lowest set bit in mask
                low = mask & -mask
                pos = low.bit_length() - 1  # position relative to shift
                c = b + pos + 1
                sys.stdout.write(f"{a} {b} {c}\n")
                return

if __name__ == "__main__":
    solve()
