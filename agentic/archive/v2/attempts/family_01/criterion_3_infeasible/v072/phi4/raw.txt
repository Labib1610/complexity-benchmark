import sys

def find_lexicographically_smallest_triad():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = {}
    index = 2
    for _ in range(m):
        u, v = int(data[index]), int(data[index + 1])
        if u not in edges:
            edges[u] = set()
        if v not in edges:
            edges[v] = set()
        edges[u].add(v)
        edges[v].add(u)
        index += 2
    
    for a in range(1, n + 1):
        for b in sorted(edges[a]):
            if b > a and a in edges[b]:
                common_neighbors = edges[a].intersection(edges[b])
                for c in sorted(common_neighbors):
                    if c > b:
                        print(a, b, c)
                        return

find_lexicographically_smallest_triad()
