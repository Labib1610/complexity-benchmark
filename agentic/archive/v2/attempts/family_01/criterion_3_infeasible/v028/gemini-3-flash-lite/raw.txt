import sys

def solve():
    # Read n and m
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    # Use adjacency sets for O(1) edge lookup
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        # Ensure u < v for the edge representation
        if u > v:
            u, v = v, u
        adj[u].add(v)
        adj[v].add(u)
        edges.append((u, v))
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all pairs (a, b) that have an edge and a < b.
    # Then check if there exists c > b such that c is connected to both a and b.
    
    # Sort edges to help process in a way that finds lexicographical smallest
    edges.sort()
    
    # Store candidates (a, b, c)
    # We want to minimize a, then b, then c.
    # We can iterate a from 1 to n, b from a+1 to n.
    # If edge (a, b) exists, find all common neighbors c > b.
    
    for a in range(1, n + 1):
        # Neighbors of a
        neighbors_a = adj[a]
        # Only consider b > a
        sorted_neighbors_a = sorted([x for x in neighbors_a if x > a])
        
        for b in sorted_neighbors_a:
            # We look for c > b such that c is in neighbors_a and neighbors_b
            neighbors_b = adj[b]
            
            # Intersection of neighbors_a and neighbors_b
            # We want the smallest c > b
            common = neighbors_a.intersection(neighbors_b)
            
            best_c = -1
            for c in common:
                if c > b:
                    if best_c == -1 or c < best_c:
                        best_c = c
            
            if best_c != -1:
                print(f"{a} {b} {best_c}")
                return

if __name__ == '__main__':
    solve()
