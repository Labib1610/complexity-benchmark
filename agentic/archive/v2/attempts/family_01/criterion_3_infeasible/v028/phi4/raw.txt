import sys

def find_triad(n, m, edges):
    # Create an adjacency list for the graph
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    # Sort connections to ensure lexicographical order
    for i in range(1, n + 1):
        adj[i].sort()

    # Find the smallest triad (a < b < c) such that a, b, and c are all connected
    for a in range(1, n + 1):
        for b in adj[a]:
            if b > a:
                for c in adj[b]:
                    if c > b and c in adj[a]:
                        return (a, b, c)

# Read input from standard input
input = sys.stdin.read
data = input().split()
n = int(data[0])
m = int(data[1])

edges = []
index = 2
for _ in range(m):
    u = int(data[index])
    v = int(data[index + 1])
    edges.append((u, v))
    index += 2

# Find and print the smallest triad
result = find_triad(n, m, edges)
print(" ".join(map(str, result)))
