import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        
        if u > v:
            u, v = v, u
        
        adj[u].add(v)
        edges.append((u, v))
    
    # We need the lexicographically smallest triple (a, b, c) such that a < b < c
    # and all (a, b), (a, c), (b, c) are in the graph.
    
    # Sort edges to help find the smallest a, then b, then c
    edges.sort()
    
    # To find the lexicographically smallest triple (a, b, c), iterate through all edges (a, b)
    # and check if there exists a c > b such that (a, c) and (b, c) are edges.
    # Since we want the smallest (a, b, c), we iterate through possible a, then b.
    
    # Optimization: iterate through all edges (a, b) and intersect adj[a] and adj[b]
    # to find candidates for c.
    
    best_triple = None
    
    for a in range(1, n + 1):
        # We only care about edges (a, b) where a < b
        # Find all neighbors b of a such that b > a
        neighbors_a = sorted([v for v in adj[a] if v > a])
        
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            
            # If we already found a triple with a smaller 'a' or (if same 'a') smaller 'b',
            # we can potentially prune.
            if best_triple and (a > best_triple[0] or (a == best_triple[0] and b >= best_triple[1])):
                continue
            
            # Find common neighbors c of a and b such that c > b
            # Intersection of adj[a] and adj[b]
            common = adj[a].intersection(adj[b])
            c_candidates = [c for c in common if c > b]
            
            if c_candidates:
                c = min(c_candidates)
                if best_triple is None or (a, b, c) < best_triple:
                    best_triple = (a, b, c)
        
        # If we have a best_triple, and we are moving to a > best_triple[0], we can stop
        if best_triple and a > best_triple[0]:
            break
            
    print(f"{best_triple[0]} {best_triple[1]} {best_triple[2]}")

if __name__ == '__main__':
    solve()
