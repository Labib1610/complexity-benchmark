import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    # adjacency matrix
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    # adjacency bitmask per node
    masks = [0] * (n + 1)

    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u][v] = adj[v][u] = True
        masks[u] |= 1 << (v - 1)
        masks[v] |= 1 << (u - 1)

    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            if not adj[a][b]:
                continue
            # common neighbors of a and b
            common = masks[a] & masks[b]
            # remove nodes <= b
            mask = common >> b
            if mask:
                lsb = mask & -mask
                idx = lsb.bit_length() - 1
                c = idx + b + 1
                if c <= n:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
