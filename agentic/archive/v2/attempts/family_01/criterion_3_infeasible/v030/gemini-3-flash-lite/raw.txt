import sys

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].add(v)
        adj[v].add(u)
        if u > v:
            u, v = v, u
        edges.append((u, v))
    
    # Sort edges to help find lexicographically smallest triple
    edges.sort()
    
    # To find the lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through each edge (a, b) and check if there exists a node c
    # such that c > b and c is in both adj[a] and adj[b].
    # To ensure lexicographical order, we check candidates in increasing order of a, then b.
    
    # Pre-sort adjacency lists to make intersection faster
    sorted_adj = [sorted(list(s)) for s in adj]
    
    for a in range(1, n + 1):
        for b in sorted_adj[a]:
            if a >= b:
                continue
            
            # Now we have an edge (a, b). Look for c > b.
            # Intersection of sorted_adj[a] and sorted_adj[b]
            # Since we only need the smallest c > b, we can iterate
            # through the smaller list.
            
            set_a = adj[a]
            set_b = adj[b]
            
            # Optimize intersection: iterate through the smaller adjacency list
            if len(set_a) < len(set_b):
                candidates = sorted_adj[a]
            else:
                candidates = sorted_adj[b]
            
            for c in candidates:
                if c > b:
                    if c in set_a and c in set_b:
                        print(f"{a} {b} {c}")
                        return

if __name__ == '__main__':
    solve()
