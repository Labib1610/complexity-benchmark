from sys import stdin

def find_faulty_group():
    n, m = map(int, stdin.readline().split())
    interference = [set() for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, stdin.readline().split())
        interference[u].add(v)
        interference[v].add(u)

    # Sort the channels to ensure we find the lexicographically smallest group
    sorted_channels = sorted(interference.keys())

    # Iterate over all possible triplets (a, b, c) with a < b < c
    for i in range(n):
        a = sorted_channels[i]
        for j in range(i + 1, n):
            b = sorted_channels[j]
            if b not in interference[a]:
                continue
            for k in range(j + 1, n):
                c = sorted_channels[k]
                if c in interference[a] and c in interference[b]:
                    print(a, b, c)
                    return

find_faulty_group()
