import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # To find the lexicographically smallest triple (a, b, c) with a < b < c:
    # Iterate a from 1 to n-2
    # Iterate b from a+1 to n-1
    # Check if b is in adj[a]
    # If so, iterate c from b+1 to n
    # Check if c is in adj[a] and c is in adj[b]
    
    # Optimization: Sort adjacency lists for faster intersection or just use sets
    # Given n=2000, O(n^3) is too slow. However, we only need to check common neighbors.
    # The number of edges is up to 2e5.
    
    for a in range(1, n + 1):
        # Sort neighbors to ensure we pick the smallest b and c
        neighbors_a = sorted(list(adj[a]))
        for i in range(len(neighbors_a)):
            b = neighbors_a[i]
            if b <= a:
                continue
            
            # Find common neighbors of a and b that are > b
            # Intersection of neighbors_a and neighbors_b
            # Since we need the smallest b and smallest c, this is efficient
            neighbors_b = adj[b]
            for j in range(i + 1, len(neighbors_a)):
                c = neighbors_a[j]
                if c in neighbors_b:
                    print(f"{a} {b} {c}")
                    return

solve()
