```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].add(v)
        adj[v].add(u)
        idx += 2
        
    # Convert sets to sorted lists to optimize intersection checking
    adj_list = [sorted(list(adj[i])) for i in range(n + 1)]
    
    # We want lexicographically smallest (a, b, c) with a < b < c
    # Iterate a from 1 to n
    for a in range(1, n + 1):
        neighbors_a = adj_list[a]
        # We only care about neighbors > a
        nodes_a = [x for x in neighbors_a if x > a]
        if len(nodes_a) < 2:
            continue
            
        # Iterate b from nodes_a
        for i in range(len(nodes_a)):
            b = nodes_a[i]
            # Now we need a c > b such that c is in both neighbors_a and neighbors_b
            # Since both adj_list[a] and adj_list[b] are sorted, we can use intersection
            # However, for n=2000, bitsets or simple set intersection is fast enough.
            # Using set intersection:
            common = adj[a].intersection(adj[b])
            # Filter for c > b
            candidates = [x for x in common if x > b]
            if candidates:
                print(f"{a} {b} {min(candidates)}")
                return

if __name__ == '__main__':
    solve()
```
