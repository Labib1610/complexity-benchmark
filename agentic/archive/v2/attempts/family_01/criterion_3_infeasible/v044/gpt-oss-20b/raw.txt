import sys

def solve() -> None:
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj_set = [set() for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj_set[u].add(v)
        adj_set[v].add(u)

    # sorted adjacency lists for deterministic order
    adj_list = [sorted(adj_set[i]) for i in range(n + 1)]

    for a in range(1, n + 1):
        neigh_a = adj_list[a]
        for b in neigh_a:
            if b <= a:
                continue
            # choose the smaller neighbor list to iterate over
            if len(neigh_a) < len(adj_list[b]):
                small = neigh_a
                large_set = adj_set[b]
            else:
                small = adj_list[b]
                large_set = adj_set[a]
            for c in small:
                if c <= b:
                    continue
                if c in large_set:
                    print(a, b, c)
                    return

if __name__ == "__main__":
    solve()
