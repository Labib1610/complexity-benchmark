def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append(tuple(sorted((u, v))))
    
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    best_committee = None
    
    for a in range(1, n + 1):
        for b in range(a + 1, n + 1):
            for c in range(b + 1, n + 1):
                if b in adj[a] and c in adj[a] and c in adj[b]:
                    if best_committee is None:
                        best_committee = (a, b, c)
                    else:
                        if a < best_committee[0]:
                            best_committee = (a, b, c)
                        elif a == best_committee[0] and b < best_committee[1]:
                            best_committee = (a, b, c)
                        elif a == best_committee[0] and b == best_committee[1] and c < best_committee[2]:
                            best_committee = (a, b, c)
    
    print(best_committee[0], best_committee[1], best_committee[2])

solve()
