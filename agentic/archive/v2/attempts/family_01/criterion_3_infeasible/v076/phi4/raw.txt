```python
n, m = map(int, input().split())
incompatibility = [set() for _ in range(n + 1)]

for _ in range(m):
    u, v = map(int, input().split())
    incompatibility[u].add(v)
    incompatibility[v].add(u)

result = None

for a in range(1, n + 1):
    for b in sorted(incompatibility[a]):
        if b > a:
            common_compatible_units = incompatibility[a] & incompatibility[b]
            for c in sorted(common_compatible_units):
                if c > b:
                    result = (a, b, c)
                    break
            if result:
                break
    if result:
        break

print(*result)
```
