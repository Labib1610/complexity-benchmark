import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    masks = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        masks[u] |= 1 << v
        masks[v] |= 1 << u

    for a in range(n):
        mask_a = masks[a]
        for b in range(a + 1, n):
            if not (mask_a >> b) & 1:
                continue
            common = mask_a & masks[b]
            shifted = common >> (b + 1)
            if shifted:
                lowbit = shifted & -shifted
                tz = lowbit.bit_length() - 1
                c = tz + b + 2  # 1-indexed vertex
                print(a + 1, b + 1, c)
                return

if __name__ == "__main__":
    solve()
