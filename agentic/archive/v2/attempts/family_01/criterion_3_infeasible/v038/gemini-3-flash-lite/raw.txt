import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [set() for _ in range(n + 1)]
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u > v:
            u, v = v, u
        adj[u].add(v)
        edges.append((u, v))
        
    # To find lexicographically smallest (a, b, c) with a < b < c:
    # Iterate through all edges (u, v) as candidates for (a, b),
    # then check if there exists a w > v such that (u, w) and (v, w) exist.
    # To ensure lexicographical order, sort the edges first.
    
    edges.sort()
    
    # Pre-calculate adjacency as bitsets for faster intersection
    # Given n <= 2000, bitsets are efficient.
    adj_bits = [0] * (n + 1)
    for u, v in edges:
        adj_bits[u] |= (1 << v)
        adj_bits[v] |= (1 << u)
        
    best = (n + 1, n + 1, n + 1)
    
    for i in range(1, n + 1):
        # Optimization: if i is already larger than best[0], stop
        if i >= best[0]:
            break
            
        neighbors = sorted([v for v in adj[i] if v > i])
        for idx1 in range(len(neighbors)):
            u = neighbors[idx1]
            if i >= best[0] and u >= best[1]:
                continue
            
            # Intersection of adj[i] and adj[u]
            common = adj_bits[i] & adj_bits[u]
            
            # We need a w > u such that w is in common
            # Find the smallest w > u
            w = -1
            # Check bits > u
            for bit in range(u + 1, n + 1):
                if (common >> bit) & 1:
                    w = bit
                    break
            
            if w != -1:
                if (i, u, w) < best:
                    best = (i, u, w)
                    
    print(f"{best[0]} {best[1]} {best[2]}")

if __name__ == '__main__':
    solve()
