import sys

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        adj[u].append(v)
        adj[v].append(u)

    for lst in adj:
        lst.sort()

    masks = [0] * n
    for i in range(n):
        mask = 0
        for v in adj[i]:
            mask |= 1 << v
        masks[i] = mask

    for a in range(n):
        mask_a = masks[a]
        for b in adj[a]:
            if b <= a:
                continue
            common = mask_a & masks[b]
            if common == 0:
                continue
            shift = b + 1
            x = common >> shift
            if x == 0:
                continue
            lsb = x & -x
            pos = lsb.bit_length() - 1
            c = shift + pos
            print(a + 1, b + 1, c + 1)
            return

if __name__ == "__main__":
    solve()
