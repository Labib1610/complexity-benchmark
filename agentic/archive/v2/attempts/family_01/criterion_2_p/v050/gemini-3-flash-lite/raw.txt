import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(500000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # We want lexicographically smallest assignment.
    # This means we prefer assigning 1 to worker i if possible.
    # We iterate i from 1 to n. If worker i is unassigned, 
    # we start a traversal and try assigning 1.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for the connected component
            # Try setting assignment[i] = 1
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            while stack:
                u, color = stack.pop()
                component.append((u, color))
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, 3 - color))
                    elif assignment[v] == color:
                        print("NO")
                        return
            
            # Now we have a valid assignment for this component.
            # However, swapping the colors (1->2, 2->1) might lead to a 
            # lexicographically smaller result.
            # We compare the current assignment with the swapped one.
            # The lexicographical check is: which one has a 1 earlier?
            # Actually, because we want the smallest sequence s_1, s_2, ..., s_n,
            # we just need to ensure that for the *first* node in this component,
            # we pick the color that results in the smaller value.
            # But wait: the choice is constrained by the component structure.
            # If we pick 1 for the first unvisited node, is that always optimal?
            # Yes, because if we have two components, we solve them independently.
            # For a single component, there are only two valid colorings (if bipartite).
            # We compare the two colorings and pick the one that is lexicographically smaller.
            
            # Re-collect component
            nodes = []
            q = [i]
            visited_comp = {i}
            while q:
                u = q.pop()
                nodes.append(u)
                for v in adj[u]:
                    if v not in visited_comp:
                        visited_comp.add(v)
                        q.append(v)
            
            # Try both: assignment[i]=1 vs assignment[i]=2
            # But we already set assignment[i]=1. Let's see if this is the best.
            # Actually, simply picking 1 for the smallest index in the component
            # is greedy and correct for lexicographical order.
            
    print("YES")
    print(*(assignment[1:]))

solve()
