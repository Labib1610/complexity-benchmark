```python
import sys
from collections import defaultdict

def main():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    conflict_rules = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        conflict_rules.append((u, v))
        index += 2
    
    graph = defaultdict(list)
    for u, v in conflict_rules:
        graph[u].append(v)
        graph[v].append(u)
    
    shifts = [0] * (n + 1)  # Shifts array to store shift of each technician

    def dfs(node, current_shift):
        if shifts[node] != 0:
            return shifts[node] == current_shift
        
        shifts[node] = current_shift
        for neighbor in graph[node]:
            if not dfs(neighbor, 3 - current_shift):  # Toggle between 1 and 2
                return False
        return True
    
    for technician in range(1, n + 1):
        if shifts[technician] == 0:  # Not yet assigned a shift
            if not dfs(technician, 1):  # Try to assign shift 1 first
                print("NO")
                return

    print("YES")
    print(' '.join(map(str, shifts[1:])))
```
