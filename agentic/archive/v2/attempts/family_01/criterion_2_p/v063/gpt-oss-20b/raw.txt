import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS to color component
        q = deque([start])
        color[start] = 0
        comp_nodes = [start]
        is_bipartite = True
        while q:
            u = q.popleft()
            cu = color[u]
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = cu ^ 1
                    q.append(w)
                    comp_nodes.append(w)
                elif color[w] == cu:
                    is_bipartite = False
                    break
            if not is_bipartite:
                break
        if not is_bipartite:
            print("NO")
            return
        # Determine orientation: smallest node gets 1
        min_node = min(comp_nodes)
        if color[min_node] == 0:
            mapping = {0: 1, 1: 2}
        else:
            mapping = {0: 2, 1: 1}
        for v in comp_nodes:
            ans[v] = mapping[color[v]]

    print("YES")
    print(' '.join(map(str, ans[1:])))

if __name__ == "__main__":
    solve()
