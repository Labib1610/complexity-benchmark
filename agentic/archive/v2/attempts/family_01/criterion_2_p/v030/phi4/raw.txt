from collections import defaultdict

def find_largest_component(adj):
    visited = [False] * len(adj)
    max_comp_size = -1
    max_comp_root = -1

    def dfs(u, root):
        nonlocal comp_size
        visited[u] = True
        comp_size += 1
        for v in adj[u]:
            if not visited[v]:
                dfs(v, root)

    for u in range(len(adj)):
        if not visited[u]:
            comp_size = 0
            dfs(u, u)
            if comp_size > max_comp_size:
                max_comp_size = comp_size
                max_comp_root = u

    return max_comp_root, max_comp_size

def is_bipartite(graph):
    n = len(graph)
    color = [-1] * n
    
    def bfs(src):
        queue = [src]
        color[src] = 0
        
        while queue:
            u = queue.pop(0)
            
            for v in graph[u]:
                if color[v] == -1:
                    color[v] = 1 - color[u]
                    queue.append(v)
                elif color[v] == color[u]:
                    return False
        return True

    for i in range(n):
        if color[i] == -1 and not bfs(i):
            return False, []
    
    # Find the lexicographically smallest assignment
    mode_assignment = [2 if c else 1 for c in color]
    
    return True, mode_assignment

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    
    n = int(data[idx])
    m = int(data[idx + 1])
    idx += 2
    
    adj = defaultdict(list)
    
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    largest_comp_root, largest_comp_size = find_largest_component(adj)
    
    if largest_comp_size != n:
        print("YES")
        mode_assignment = [1] * n
        for i in range(n):
            if not adj[i]:
                mode_assignment[i] = 2
        print(' '.join(map(str, mode_assignment)))
    else:
        is_bipartite_possible, mode_assignment = is_bipartite(adj)
        
        if is_bipartite_possible:
            print("YES")
            print(' '.join(map(str, mode_assignment)))
        else:
            print("NO")

if __name__ == "__main__":
    main()
