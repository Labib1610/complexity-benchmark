```python
def solve():
    n, m = map(int, input().split())
    conflicts = []
    for _ in range(m):
        u, v = map(int, input().split())
        conflicts.append((u, v))
    
    def is_valid(assignment):
        for u, v in conflicts:
            if assignment[u-1] == assignment[v-1]:
                return False
        return True
    
    def find_assignment():
        for i in range(2**n):
            assignment = []
            temp = i
            for _ in range(n):
                assignment.append((temp % 2) + 1)
                temp //= 2
            
            if is_valid(assignment):
                return assignment
        return None
    
    assignment = find_assignment()
    
    if assignment:
        print("YES")
        print(*assignment)
    else:
        print("NO")

solve()
```
