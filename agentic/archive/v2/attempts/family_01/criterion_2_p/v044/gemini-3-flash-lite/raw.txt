import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we try to assign 1 
    # to station i if possible. If not, we assign 2.
    # This is a 2-coloring problem.
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for the connected component
            # To ensure lexicographical smallest, we prioritize color 1
            stack = [i]
            color[i] = 1
            
            while stack:
                u = stack.pop()
                
                # Check neighbors
                for v in adj[u]:
                    if color[v] == 0:
                        # If neighbor is uncolored, color it with the opposite
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return

    # Check if we can improve lexicographical order:
    # Actually, for a connected component, there are only two colorings (swap 1 and 2).
    # To get the lexicographical smallest, for each component, we check if 
    # starting with 1 at the smallest index gives a better sequence.
    # However, the graph might be disconnected.
    # The logic above greedily assigns 1 to the first encountered node in each component.
    # This is not necessarily the lexicographical smallest.
    # We must decide for each component whether to use (1, 2) or (2, 1) mapping.
    
    final_colors = [0] * (n + 1)
    visited = [False] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Now color component with 0/1
            comp_color = {node: -1 for node in component}
            start_node = min(component)
            comp_color[start_node] = 0
            q = [start_node]
            while q:
                u = q.pop()
                for v in adj[u]:
                    if comp_color[v] == -1:
                        comp_color[v] = 1 - comp_color[u]
                        q.append(v)
                    elif comp_color[v] == comp_color[u]:
                        print("NO")
                        return
            
            # Two options:
            # Option A: start_node gets 1 (comp_color[x] + 1)
            # Option B: start_node gets 2 (2 - comp_color[x])
            # We compare which one is lexicographically smaller
            opt1 = [0] * len(component)
            opt2 = [0] * len(component)
            for node in component:
                opt1[node-1] = comp_color[node] + 1
                opt2[node-1] = 2 - comp_color[node]
            
            # Compare opt1 and opt2
            # Find the first index where they differ
            for node in sorted(component):
                if opt1[node-1] != opt2[node-1]:
                    if opt1[node-1] < opt2[node-1]:
                        for n_ in component: final_colors[n_] = opt1[n_-1]
                    else:
                        for n_ in component: final_colors[n_] = opt2[n_-1]
                    break
            else:
                for n_ in component: final_colors[n_] = opt1[n_-1]

    print("YES")
    print(*(final_colors[1:]))

solve()
