import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0 or 1
    result = [0] * n          # 1 or 2
    for start in range(n):
        if color[start] != -1:
            continue
        # BFS to check bipartite and collect component nodes
        q = deque([start])
        color[start] = 0
        comp_nodes = [start]
        while q:
            u = q.popleft()
            cu = color[u]
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = cu ^ 1
                    q.append(w)
                    comp_nodes.append(w)
                elif color[w] == cu:
                    print("NO")
                    return
        # Determine orientation: give color 0 -> 1 if the smallest node has color 0
        min_node = min(comp_nodes)
        if color[min_node] == 0:
            map0, map1 = 1, 2
        else:
            map0, map1 = 2, 1
        for v in comp_nodes:
            result[v] = map0 if color[v] == 0 else map1

    print("YES")
    sys.stdout.write(" ".join(map(str, result)))

if __name__ == "__main__":
    solve()
