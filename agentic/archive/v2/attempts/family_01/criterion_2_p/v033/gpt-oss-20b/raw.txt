```python
import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # provisional colors 0/1

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        comp = []
        q = deque([start])
        color[start] = 0
        min_idx = start

        while q:
            u = q.popleft()
            comp.append(u)
            if u < min_idx:
                min_idx = u
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = 1 - color[u]
                    q.append(w)
                elif color[w] == color[u]:
                    sys.stdout.write("NO")
                    return

        # ensure the smallest vertex in this component gets group 1
        if color[min_idx] == 1:
            for v in comp:
                color[v] = 1 - color[v]

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(str(color[i] + 1) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
