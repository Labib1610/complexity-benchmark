def solve():
    n, m = map(int, input().split())
    restrictions = []
    for _ in range(m):
        u, v = map(int, input().split())
        restrictions.append((u, v))

    assignment = [0] * n
    
    def is_valid(assignment):
        for u, v in restrictions:
            if assignment[u-1] == assignment[v-1]:
                return False
        return True
    
    def find_assignment(index):
        if index == n:
            if is_valid(assignment):
                return True
            else:
                return False
        
        assignment[index] = 1
        if find_assignment(index + 1):
            return True
        
        assignment[index] = 2
        if find_assignment(index + 1):
            return True
        
        return False

    if find_assignment(0):
        print("YES")
        print(*assignment)
    else:
        print("NO")

solve()
