```python
import sys
from collections import defaultdict

def main():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        index += 2
    
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    colors = [0] * (n + 1)
    is_possible = True
    
    def dfs(node, color):
        nonlocal is_possible
        if colors[node] != 0:
            if colors[node] != color:
                is_possible = False
            return
        
        colors[node] = color
        for neighbor in graph[node]:
            dfs(neighbor, 3 - color)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            dfs(i, 1)
    
    if not is_possible:
        print("NO")
    else:
        print("YES")
        result = [str(colors[i]) for i in range(1, n + 1)]
        print(' '.join(result))

if __name__ == "__main__":
    main()
```
