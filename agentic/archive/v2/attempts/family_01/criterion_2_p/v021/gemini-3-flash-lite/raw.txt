import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # channel assignment: 0 = unassigned, 1 = channel 1, 2 = channel 2
    assignment = [0] * (n + 1)
    
    # We want to pick 1 whenever possible.
    # This is a 2-coloring problem. Each connected component can be colored in at most 2 ways.
    # To get lexicographically smallest, for each component, we try setting the 
    # first node (smallest index) to 1. If that fails, we set it to 2.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # New component
            # Try assigning 1 to the first node encountered in this component
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Try the other color for the root
                for node in component:
                    assignment[node] = 0
                
                assignment[i] = 2
                component = [i]
                possible = True
                idx_comp = 0
                while idx_comp < len(component):
                    u = component[idx_comp]
                    idx_comp += 1
                    
                    for v in adj[u]:
                        if assignment[v] == 0:
                            assignment[v] = 3 - assignment[u]
                            component.append(v)
                        elif assignment[v] == assignment[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
