from sys import stdin

def read():
    return stdin.readline().strip()

def solve():
    n, m = map(int, read().split())
    adj_list = [[] for _ in range(n + 1)]
    
    for _ in range(m):
        u, v = map(int, read().split())
        adj_list[u].append(v)
        adj_list[v].append(u)

    color = [0] * (n + 1)
    valid = True

    def dfs(node, c):
        nonlocal valid
        if color[node] != 0:
            return color[node] == c
        color[node] = c
        for neighbor in adj_list[node]:
            if not dfs(neighbor, 3 - c):
                valid = False
        return True

    for i in range(1, n + 1):
        if color[i] == 0:
            if not dfs(i, 1):
                valid = False
    
    if not valid:
        print("NO")
    else:
        print("YES")
        result = [min(color[i], 2) for i in range(1, n + 1)]
        print(' '.join(map(str, result)))

if __name__ == "__main__":
    solve()
