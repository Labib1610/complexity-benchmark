import sys

def research_facility_scheduling():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    edges = []
    index = 2
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        index += 2

    # Graph adjacency list
    from collections import defaultdict
    graph = defaultdict(list)
    
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    # If no restrictions, output YES and all researchers in lab 1
    if m == 0:
        print("YES")
        print(' '.join(['1'] * n))
        return

    # Try to color the graph using two colors (1 and 2)
    color = [-1] * (n + 1)  # -1 means uncolored, 1 and 2 are the colors
    
    def is_bipartite(node, c):
        if color[node] != -1:
            return color[node] == c
        color[node] = c
        for neighbor in graph[node]:
            if not is_bipartite(neighbor, 3 - c):  # Toggle between 1 and 2
                return False
        return True
    
    # Try to color all components of the graph
    for i in range(1, n + 1):
        if color[i] == -1:
            if not is_bipartite(i, 1):
                print("NO")
                return

    print("YES")
    result = []
    for i in range(1, n + 1):
        result.append(str(color[i]))
    
    print(' '.join(result))

research_facility_scheduling()
