import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = next(it) - 1
        v = next(it) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n  # 0 or 1
    for start in range(n):
        if color[start] != -1:
            continue
        # BFS for component, start with color 0
        dq = deque([start])
        color[start] = 0
        while dq:
            u = dq.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    dq.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    # Build lexicographically smallest assignment
    res = [0] * n
    visited = [False] * n
    for i in range(n):
        if not visited[i]:
            # component's smallest node is i
            # assign it color 1
            stack = [i]
            visited[i] = True
            comp_nodes = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
                        comp_nodes.append(v)
            # determine mapping: root gets 1
            # root color is color[i] (0 or 1)
            # map: if root color is 0 -> 0->1,1->2 ; else if root color is 1 -> 1->1,0->2
            mapping = [0, 0]
            if color[i] == 0:
                mapping[0] = 1
                mapping[1] = 2
            else:
                mapping[1] = 1
                mapping[0] = 2
            for v in comp_nodes:
                res[v] = mapping[color[v]]

    print("YES")
    print(' '.join(map(str, res)))

if __name__ == "__main__":
    solve()
