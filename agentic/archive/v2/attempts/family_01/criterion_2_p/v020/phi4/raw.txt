from sys import stdin

def solve():
    n, m = map(int, input().split())
    adj = {i: set() for i in range(1, n+1)}
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)
        
    s = [1] * (n + 1)  # Default assignment to line 1
    valid = True
    
    def dfs(node, color):
        nonlocal valid
        if node in colors:
            return colors[node] == color
        
        colors[node] = color
        for neighbor in adj[node]:
            if not dfs(neighbor, 3 - color):  # Alternate between 1 and 2
                valid = False
                return False
        return True
    
    colors = {}
    
    for i in range(1, n+1):
        if i not in colors:
            if not dfs(i, 1):
                valid = False
                break
    
    if valid:
        print("YES")
        print(' '.join(map(str, s[1:])))
    else:
        print("NO")

solve()
