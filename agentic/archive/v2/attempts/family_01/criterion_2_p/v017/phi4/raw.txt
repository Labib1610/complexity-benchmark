from sys import stdin

def shift_partitioning():
    # Read n and m
    n, m = map(int, stdin.readline().split())
    
    # Initialize adjacency list for workers
    adj_list = [[] for _ in range(n + 1)]
    
    # Read constraints
    for _ in range(m):
        u, v = map(int, stdin.readline().split())
        adj_list[u].append(v)
        adj_list[v].append(u)
    
    # Array to store shift assignments: 0 means unassigned
    shifts = [0] * (n + 1)
    
    # Try assigning Shift 1 to the first worker and see if it works
    def try_assign(shift_1):
        shifts[shift_1] = 1  # Assign Shift 1 to worker shift_1
        
        # BFS/DFS to assign shifts ensuring no two incompatible workers share the same shift
        stack = [shift_1]
        
        while stack:
            current = stack.pop()
            
            for neighbor in adj_list[current]:
                if shifts[neighbor] == 0:  # If not yet assigned
                    # Assign opposite shift to neighbor
                    shifts[neighbor] = 3 - shifts[current]
                    stack.append(neighbor)
                elif shifts[neighbor] == shifts[current]:  # Conflict detected
                    return False
        
        return True
    
    if try_assign(1):
        print("YES")
        print(' '.join(map(str, shifts[1:])))
    else:
        shifts = [0] * (n + 1)  # Reset shifts
        if try_assign(2):  # Try assigning Shift 2 to the first worker and see if it works
            print("YES")
            print(' '.join(map(str, shifts[1:])))
        else:
            print("NO")

# Run the function for shift partitioning
shift_partitioning()
