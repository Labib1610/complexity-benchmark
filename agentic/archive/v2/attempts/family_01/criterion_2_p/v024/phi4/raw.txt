def frequency_channel_allocation():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    edges = []
    index = 2
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        index += 2
        
    from collections import defaultdict
    graph = defaultdict(list)
    
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    # Try to assign channels lexicographically smallest way starting with channel 1
    assignment = {}
    channels = [0] * (n + 1)  # 0 means unassigned
    
    def is_valid(unit, channel):
        for neighbor in graph[unit]:
            if channels[neighbor] == channel:
                return False
        return True
    
    def assign_channels(start_unit):
        queue = [start_unit]
        channels[start_unit] = 1
        
        while queue:
            current = queue.pop(0)
            
            # Try assigning the smallest possible valid channel to neighbors
            for neighbor in graph[current]:
                if channels[neighbor] == 0:  # Unassigned
                    if is_valid(neighbor, 1):
                        channels[neighbor] = 1
                    elif is_valid(neighbor, 2):
                        channels[neighbor] = 2
                    else:
                        return False
                    
                    queue.append(neighbor)
        
        return True
    
    for unit in range(1, n + 1):
        if channels[unit] == 0:  # Unassigned
            if not assign_channels(unit):
                print("NO")
                return

    print("YES")
    print(' '.join(map(str, channels[1:])))
