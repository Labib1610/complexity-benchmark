import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not yet assigned, we greedily assign it to 1.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS component processing
            stack = [i]
            assignment[i] = 1
            component = [i]
            
            # Using a list as a stack for iterative DFS to avoid recursion depth issues
            # and to maintain order
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    # The construction above is greedy but might not be globally lexicographically smallest.
    # However, for 2-coloring, the choices are symmetric per connected component.
    # For each component, we have two valid colorings: (c1, c2, ...) or (3-c1, 3-c2, ...).
    # We want the coloring such that the first node in the component (by index) is 1.
    # Since we process 1..n, the first uncolored node is always assigned 1.
    
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
