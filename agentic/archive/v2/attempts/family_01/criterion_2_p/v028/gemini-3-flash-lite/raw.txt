import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
    
    # We want the lexicographically smallest assignment.
    # This means we try to set s[i] = 1 whenever possible.
    # This is a 2-coloring problem.
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try to color this connected component.
            # To get lexicographical smallest, we greedily assign 1 to the 
            # smallest available node in the component.
            stack = [i]
            color[i] = 1
            component = []
            
            # Use iterative DFS to avoid recursion depth issues
            dfs_stack = [i]
            while dfs_stack:
                u = dfs_stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        dfs_stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # The component is colored. Now we check if we can swap 1s and 2s
            # to make the assignment lexicographically smaller.
            # For each component, we have two valid colorings: (c) and (3-c).
            # We compare them and pick the one that is lexicographically smaller.
            # Wait, the lexicographical requirement is for the whole sequence s_1...s_n.
            # Since components are independent, we just pick the smaller assignment for each.
            
            c1 = 0
            c2 = 0
            # Count nodes in the component
            nodes = sorted(component)
            
            # Check which coloring is better for this component
            # coloring 1: color[u]
            # coloring 2: 3 - color[u]
            # We want to minimize the sequence.
            # Find the first node in this component.
            first_node = nodes[0]
            if color[first_node] == 2:
                # Flip all colors in this component
                for node in component:
                    color[node] = 3 - color[node]
