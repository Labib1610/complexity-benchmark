```python
from sys import stdin

# Reading input
input = lambda: stdin.readline().rstrip()

n, m = map(int, input().split())
g = [[] for _ in range(n + 1)]
for _ in range(m):
    u, v = map(int, input().split())
    g[u].append(v)
    g[v].append(u)

def possible(a, b):
    return a[0] == 1 and b[0] != 2 or a[0] == 2 and b[0] != 1

# DFS to assign channels
stack = [None]
visited = [False] * (n + 1)
ans = [None] * (n + 1)

while stack:
    u = stack.pop()
    if not visited[u]:
        # Assign the smallest possible channel that doesn't interfere with neighbors
        v = []
        for i in g[u]:
            if ans[i]: v.append(ans[i])
        
        # Determine which channels are available based on neighbors' assignments
        s = {1, 2}
        for x in v:
            s -= set(x)
            
        if len(s) == 0:
            print('NO')
            exit()
        elif len(s) == 1:
            ans[u] = list(s)
        else:
            # Assign the lexicographically smallest possible channel
            if min(v, default=3) == 2:
                ans[u] = [1]
            else:
                ans[u] = [2]
        
        visited[u] = True
        for i in g[u]:
            if not visited[i]: stack.append(i)

print('YES')
for i in range(1, n + 1): print(ans[i][0], end=' ')
```
