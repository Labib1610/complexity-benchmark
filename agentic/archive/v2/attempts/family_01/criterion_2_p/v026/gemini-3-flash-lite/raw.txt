sys.setrecursionlimit(300000)

    if not input_data:
        return

    idx = 2

        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # try setting color 1 first, then propagate constraints.
    
    for i in range(1, n + 1):

                u = comp_list[comp_idx]
                comp_idx += 1
