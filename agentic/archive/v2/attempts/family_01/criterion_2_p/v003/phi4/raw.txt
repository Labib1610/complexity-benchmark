```python
import sys
from collections import defaultdict

def main():
    input = sys.stdin.read
    data = input().split()
    
    index = 0
    
    n = int(data[index])
    m = int(data[index + 1])
    index += 2
    
    adj = defaultdict(list)
    
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    valid = True
    
    def dfs(node, color):
        nonlocal valid
        if colors[node] != 0:
            return colors[node] == color
        colors[node] = color
        for neighbor in adj[node]:
            if not dfs(neighbor, 3 - color):
                valid = False
        return True
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            if not dfs(i, 1):
                valid = False
    
    if not valid:
        print("NO")
    else:
        print("YES")
        result = []
        for i in range(1, n + 1):
            result.append(str(colors[i]))
        print(' '.join(result))

if __name__ == "__main__":
    main()
```
