import sys
from collections import deque

def solve() -> None:
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # 0/1 colors for bipartite test
    comp = [0] * (n + 1)            # component id per node
    flip = [0] * (n + 1)            # flip flag per component
    comp_cnt = 0
    ok = True

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        comp_cnt += 1
        dq = deque([start])
        color[start] = 0
        comp[start] = comp_cnt
        min_idx = start

        while dq:
            u = dq.popleft()
            if u < min_idx:
                min_idx = u
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    comp[v] = comp_cnt
                    dq.append(v)
                elif color[v] == color[u]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            break

        # Decide flip so that the smallest indexed node gets mode 1
        flip[comp_cnt] = 0 if color[min_idx] == 0 else 1

    if not ok:
        print("NO")
        return

    result = []
    for i in range(1, n + 1):
        mode = (color[i] ^ flip[comp[i]]) + 1
        result.append(str(mode))

    print("YES")
    print(' '.join(result))

if __name__ == "__main__":
    solve()
