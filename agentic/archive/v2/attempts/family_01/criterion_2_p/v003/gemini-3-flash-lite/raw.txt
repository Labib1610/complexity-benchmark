import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a component is unassigned, we assign it '1' (the smaller mode).
    # Then we propagate the constraints using BFS/DFS.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start a component (connected component)
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, mode = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        # Must be the opposite mode
                        new_mode = 3 - mode
                        assignment[v] = new_mode
                        stack.append((v, new_mode))
                    elif assignment[v] == mode:
                        # Conflict found
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
