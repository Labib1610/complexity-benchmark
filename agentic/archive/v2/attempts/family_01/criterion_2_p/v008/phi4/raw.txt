```python
import sys
from collections import defaultdict

def read_input():
    input = sys.stdin.read
    data = input().split()
    n = int(data[0])
    m = int(data[1])
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        index += 2
    return n, m, edges

def can_assign_shifts(n, edges):
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    # Initialize shift assignments with None (no assignment yet)
    shifts = [None] * (n + 1)  # Shifts are 1-indexed

    def assign_shift(worker, shift):
        if shifts[worker] is not None:
            return shifts[worker] == shift
        shifts[worker] = shift
        for neighbor in graph[worker]:
            if not assign_shift(neighbor, 3 - shift):  # Toggle between 1 and 2
                return False
        return True

    for worker in range(1, n + 1):
        if shifts[worker] is None:  # Not yet assigned a shift
            if not assign_shift(worker, 1):  # Try assigning shift 1 first
                return "NO", []
    
    return "YES", [shifts[i] for i in range(1, n + 1)]

def main():
    n, m, edges = read_input()
    result, assignment = can_assign_shifts(n, edges)
    print(result)
    if result == "YES":
        print(" ".join(map(str, assignment)))

if __name__ == "__main__":
    main()
```
