import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we iterate from 1 to n.
    # If a node is unvisited, we start a BFS/DFS. To prioritize '1' over '2',
    # we always try assigning '1' first.
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for the connected component
            # Using a stack for iterative DFS to avoid recursion limits
            stack = [i]
            assignment[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        # We want the smallest assignment, so try 1, then 2.
                        # Since this is a bipartite graph, the neighbor must be
                        # the opposite of the current node.
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
    
    # After filling, check if it's the lexicographically smallest.
    # Actually, the logic above (greedy assign 1 if possible) works because 
    # for each component, there are only two possible colorings (swap 1 and 2).
    # We choose the one where the smallest index in the component gets 1.
    print("YES")
    print(*(assignment[1:]))

solve()
