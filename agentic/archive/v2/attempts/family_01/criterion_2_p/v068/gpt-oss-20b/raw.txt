import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        adj[u].append(v)
        adj[v].append(u)

    visited = [-1] * (n + 1)  # provisional colors 0/1
    result = [0] * (n + 1)    # final colors 1/2

    for start in range(1, n + 1):
        if visited[start] != -1:
            continue
        # start new component
        visited[start] = 0
        result[start] = 1
        q = deque([start])

        while q:
            u = q.popleft()
            cu = visited[u]
            for v in adj[u]:
                if visited[v] == -1:
                    visited[v] = cu ^ 1
                    result[v] = visited[v] + 1
                    q.append(v)
                elif visited[v] == cu:
                    # not bipartite
                    sys.stdout.write("NO\n")
                    return

    # all components bipartite
    out = ["YES\n"] + [" ".join(str(result[i]) for i in range(1, n + 1)) + "\n"]
    sys.stdout.write("".join(out))

if __name__ == "__main__":
    solve()
