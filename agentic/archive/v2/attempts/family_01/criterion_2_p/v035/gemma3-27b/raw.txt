def solve():
    n, m = map(int, input().split())
    incompatibilities = []
    for _ in range(m):
        u, v = map(int, input().split())
        incompatibilities.append((u, v))
    
    def is_valid(assignment):
        for u, v in incompatibilities:
            if assignment[u-1] == assignment[v-1]:
                return False
        return True
    
    def find_lexicographically_smallest():
        for i in range(2**n):
            assignment = []
            for j in range(n):
                if (i >> j) & 1:
                    assignment.append(1)
                else:
                    assignment.append(2)
            
            if is_valid(assignment):
                return assignment
        return None
    
    result = find_lexicographically_smallest()
    
    if result is None:
        print("NO")
    else:
        print("YES")
        print(*result)

solve()
