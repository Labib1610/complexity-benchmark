import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    stack = []
    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        # start new component from smallest vertex i
        stack.append(i)
        color[i] = 0
        while stack:
            u = stack.pop()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    stack.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    out = ['1' if color[i] == 0 else '2' for i in range(1, n + 1)]
    sys.stdout.write("YES\n" + " ".join(out))

if __name__ == "__main__":
    solve()
