import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] = 0 (unvisited), 1, 2
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    # We want lexicographically smallest configuration.
    # For each connected component, we have two choices:
    # 1. Start DFS from the smallest index node in the component
    # 2. Assign 1 to the smallest index node, and propagate constraints.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Component nodes
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to store nodes in the order they are visited
            # to propagate colors correctly.
            q = [i]
            idx_q = 0
            while idx_q < len(q):
                u = q[idx_q]
                idx_q += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have the component colored with 1s and 2s.
            # Check if swapping colors gives a smaller lexicographical result.
            # We want the smallest index in this component to be 1 if possible.
            # Actually, simply checking the first node is not enough.
            # We need to check if the current assignment is better than the flipped one.
            
            # Find the first node in this component
            first_node = min(component)
            if color[first_node] == 2:
                # Flip colors
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
