# Increase recursion depth for deep DFS

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, 
    # we want node 1 to be 1, then node 2 to be 1 if possible, etc.
    # For each connected component, we try setting the first node to 1.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            # Try setting current node to 1
            stack = [(i, 1)]

            comp_nodes = []

            assignment[i] = 1
            comp_nodes.append(i)
            
            idx_q = 0
            while idx_q < len(q):
                u = q[idx_q]
                idx_q += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                        comp_nodes.append(v)
                    elif assignment[v] == assignment[u]:

            # The assignment is fixed by the choice of the first node (1)
