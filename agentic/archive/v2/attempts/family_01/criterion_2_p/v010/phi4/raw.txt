import sys
from collections import deque

def shift_partitioning():
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    
    n = int(data[idx])
    m = int(data[idx + 1])
    idx += 2
    
    # Create adjacency list for graph representation
    adj_list = [[] for _ in range(n)]
    
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx + 1]) - 1
        idx += 2
        adj_list[u].append(v)
        adj_list[v].append(u)
        
    # To store the team assignment, initially all are unassigned (-1)
    teams = [-1] * n
    
    def bfs(start):
        queue = deque([start])
        teams[start] = 1
        
        while queue:
            node = queue.popleft()
            current_team = teams[node]
            next_team = 3 - current_team
            
            for neighbor in adj_list[node]:
                if teams[neighbor] == -1:
                    # Assign the opposite team to the neighbor
                    teams[neighbor] = next_team
                    queue.append(neighbor)
                elif teams[neighbor] == current_team:
                    return False
        return True
    
    possible = True
    for i in range(n):
        if teams[i] == -1:  # If not yet assigned
            if not bfs(i):
                possible = False
                break
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, teams)))

shift_partitioning()
