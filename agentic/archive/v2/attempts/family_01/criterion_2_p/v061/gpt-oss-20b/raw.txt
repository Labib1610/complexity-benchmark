```python
import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = data[idx], data[idx + 1]
        idx += 2
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)
    res = [0] * (n + 1)

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        q = deque([start])
        color[start] = 0
        comp = [start]
        bip = True
        while q and bip:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == cu:
                    bip = False
                    break
        if not bip:
            sys.stdout.write("NO")
            return
        # assign segments: color 0 -> 1, color 1 -> 2
        for node in comp:
            res[node] = 1 if color[node] == 0 else 2

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, res[1:])))

if __name__ == "__main__":
    solve()
```
