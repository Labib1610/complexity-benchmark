import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    # Use fast I/O
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To find the lexicographically smallest assignment, we iterate from 1 to n.
    # If an item is unassigned, we try assigning it to line 1.
    # This greedy approach works because assigning an item to line 1 is always
    # preferred if it doesn't lead to a contradiction.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for the connected component
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            while stack:
                u, color = stack.pop()
                component.append((u, color))
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, 3 - color))
                    elif assignment[v] == color:
                        print("NO")
                        return
            
            # Since we want the lexicographically smallest assignment, 
            # for each connected component, we have two choices:
            # (assignment[v]) or (3 - assignment[v]).
            # We compare the two possibilities for the smallest index in the component.
            
            # Actually, the greedy choice of assigning the first unvisited node 
            # to 1 is correct for lexicographical minimality because 
            # if node i is the smallest, assigning it to 1 is mandatory 
            # for the smallest sequence.
            
    print("YES")
    print(*(assignment[1:]))

solve()
