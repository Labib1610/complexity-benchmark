import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(500000)

def solve():
    input = sys.stdin.read().split()
    if not input:
        return
    
    N = int(input[0])
    M = int(input[1])
    
    adj = [[] for _ in range(N + 1)]
    idx = 2
    for _ in range(M):
        u = int(input[idx])
        v = int(input[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To find the lexicographically smallest assignment, we iterate i from 1 to N.
    # If node i is not colored, we try to set color[i] = 1 (Vault 1).
    # If that leads to a contradiction, we try color[i] = 2.
    # Since this is a bipartite graph coloring, once we pick a color for a 
    # component, all other nodes in that component are forced.
    
    color = [0] * (N + 1)
    
    for i in range(1, N + 1):
        if color[i] == 0:
            # Start a BFS/DFS to color this component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack to traverse the connected component
            idx_stack = 0
            comp_stack = [i]
            color[i] = 1
            
            # Collect all nodes in this connected component first
            q = [i]
            visited_comp = {i}
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if v not in visited_comp:
                        visited_comp.add(v)
                        q.append(v)
            
            # For the lexicographically smallest, we must ensure the component 
            # root is assigned 1 if possible.
            # Actually, the greedy choice is: for the first uncolored node i,
            # assign color 1. This propagates.
            
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
