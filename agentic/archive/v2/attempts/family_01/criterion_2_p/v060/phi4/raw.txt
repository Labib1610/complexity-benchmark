from sys import stdin

def main():
    n, m = map(int, stdin.readline().split())
    
    edges = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(lambda x: int(x) - 1, stdin.readline().split())
        edges[u].append(v)
        edges[v].append(u)

    def dfs(node, color):
        if colors[node] != -1:
            return colors[node] == color
        colors[node] = color
        for neighbor in edges[node]:
            if not dfs(neighbor, 1 - color):
                return False
        return True

    colors = [-1] * n
    is_possible = True
    for i in range(n):
        if colors[i] == -1 and not dfs(i, 0):
            is_possible = False
            break
    
    if not is_possible:
        print("NO")
    else:
        assignment = [colors[i] + 1 for i in range(n)]
        print("YES")
        print(' '.join(map(str, assignment)))

if __name__ == "__main__":
    main()
