import sys
from collections import deque

def main():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    visited = [False] * n
    parity = [0] * n
    period = [0] * n  # final period (1 or 2)

    for start in range(n):
        if visited[start]:
            continue
        # BFS for this component
        q = deque([start])
        visited[start] = True
        parity[start] = 0
        comp_nodes = [start]
        is_bipartite = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    parity[v] = parity[u] ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                else:
                    if parity[v] == parity[u]:
                        is_bipartite = False
                        break
            if not is_bipartite:
                break
        if not is_bipartite:
            sys.stdout.write("NO")
            return

        # determine orientation based on smallest index node
        min_idx = min(comp_nodes)
        if parity[min_idx] == 0:
            flip = 0
        else:
            flip = 1
        for node in comp_nodes:
            period[node] = (parity[node] ^ flip) + 1

    out = ['YES', '\n', ' '.join(map(str, period))]
    sys.stdout.write(''.join(out))

if __name__ == "__main__":
    main()
