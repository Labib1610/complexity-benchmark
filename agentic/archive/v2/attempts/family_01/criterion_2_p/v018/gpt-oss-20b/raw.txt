import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)  # 0 or 1
    stack = []
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # start new component
        stack.append(start)
        color[start] = 0
        component = []
        ok = True
        while stack:
            u = stack.pop()
            component.append(u)
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - color[u]
                    stack.append(v)
                elif color[v] == color[u]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            sys.stdout.write("NO")
            return
        # ensure smallest vertex gets color 0 (maps to block 1)
        min_vertex = min(component)
        if color[min_vertex] == 1:
            for v in component:
                color[v] ^= 1

    # produce answer
    res = ['1' if color[i] == 0 else '2' for i in range(1, n + 1)]
    sys.stdout.write("YES\n" + " ".join(res))

if __name__ == "__main__":
    solve()
