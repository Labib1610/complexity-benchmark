```python
from sys import stdin
input = lambda: stdin.readline().rstrip()
from collections import defaultdict

def main():
    n, m = map(int, input().split())
    adj = defaultdict(set)
    
    for _ in range(m):
        u, v = map(int, input().split())
        adj[u].add(v)
        adj[v].add(u)

    # Initialize frequencies to None
    freq = [None] * (n + 1)
    
    def dfs(node, current_freq):
        if freq[node] is not None:
            return freq[node] == current_freq
        
        freq[node] = current_freq
        next_freq = 3 - current_freq
        
        for neighbor in adj[node]:
            if not dfs(neighbor, next_freq):
                return False
        
        return True
    
    possible = True
    for i in range(1, n + 1):
        if freq[i] is None:
            if not (dfs(i, 1) or dfs(i, 2)):
                possible = False
                break
    
    if possible:
        print("YES")
        print(' '.join(map(str, freq[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
```
