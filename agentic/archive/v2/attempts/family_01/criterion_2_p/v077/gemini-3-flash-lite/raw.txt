import sys

# Increase recursion depth for deep graphs
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
    
    # We want the lexicographically smallest sequence (s1, s2, ..., sn).
    # This means we prefer assigning 1 to smaller indices whenever possible.
    # We use a greedy approach with DFS/BFS on connected components.
    # For each component, we have two possible colorings (swap 1 and 2).
    # We pick the one that results in a smaller value at the smallest index in that component.
    
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # New component
            component = []
            stack = [i]
            colors[i] = 1
            component.append(i)
            
            # Use stack to find all nodes in component and color them
            # We don't care about the specific color yet, just consistency
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        component.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
            
            # Now we have a consistent coloring for this component.
            # Let min_idx be the smallest index in this component.
            # We want colors[min_idx] to be 1 if possible.
            # If colors[min_idx] is 2, we flip all colors in this component.
            min_idx = min(component)
            if colors[min_idx] == 2:
                for node in component:
                    colors[node] = 3 - colors[node]
                    
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
