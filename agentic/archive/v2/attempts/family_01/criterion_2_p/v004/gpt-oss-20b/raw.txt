import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)   # -1 unvisited, 0 or 1
    res = [0] * (n + 1)      # final 1/2 assignments

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS for component
        q = deque([start])
        color[start] = 0
        comp = []
        while q:
            u = q.popleft()
            comp.append(u)
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = 1 - cu
                    q.append(v)
                elif color[v] == cu:
                    # conflict
                    sys.stdout.write("NO")
                    return
        # decide orientation
        min_idx = min(comp)
        if color[min_idx] == 1:  # currently gives 2 at earliest index
            for u in comp:
                color[u] = 1 - color[u]
        # assign results
        for u in comp:
            res[u] = color[u] + 1

    out_lines = ["YES", " ".join(map(str, res[1:]))]
    sys.stdout.write("\n".join(out_lines))

if __name__ == "__main__":
    solve()
