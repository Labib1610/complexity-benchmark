from sys import stdin

def main():
    input = stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    index = 2
    
    adj = {i: [] for i in range(1, n + 1)}
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
        
    colors = [None] * (n + 1)
    
    def dfs(node, color):
        if colors[node] is not None:
            return colors[node] == color
        colors[node] = color
        for neighbor in adj[node]:
            if not dfs(neighbor, 3 - color):
                return False
        return True
    
    for i in range(1, n + 1):
        if colors[i] is None:
            if not dfs(i, 1):
                print("NO")
                return
    
    print("YES")
    result = [colors[i] for i in range(1, n + 1)]
    print(' '.join(map(str, result)))

if __name__ == "__main__":
    main()
