import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # We want lexicographically smallest assignment (1s and 2s).
    # For each connected component, we try setting the first node to 1.
    # If that fails, try setting it to 2.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning color 1 to the first node in this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        new_c = 3 - c
                        color[v] = new_c
                        stack.append((v, new_c))
                        component.append((v, new_c))
                    elif color[v] == c:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Reset component and try assigning 2 to the starting node
                for node, c in component:
                    color[node] = 0
                
                stack = [(i, 2)]
                color[i] = 2
                possible = True
                while stack:
                    u, c = stack.pop()
                    for v in adj[u]:
                        if color[v] == 0:
                            new_c = 3 - c
                            color[v] = new_c
                            stack.append((v, new_c))
                        elif color[v] == c:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return
                    
    print("YES")
    print(*(color[1:]))

solve()
