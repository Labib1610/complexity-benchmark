import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # We need lexicographically smallest assignment.
    # This means for each connected component, we try setting the first
    # encountered node to 1. If it leads to a contradiction, we try 2.
    # Within a component, once the first node is fixed, all others are determined.
    
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try to color this component starting with i as 1
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            comp_nodes = [i]
            idx_comp = 0
            
            # Using BFS/DFS to traverse component
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Try setting i to 2
                for node in q:
                    color[node] = 0
                
                q = [i]
                color[i] = 2
                head = 0
                possible = True
                while head < len(q):
                    u = q[head]
                    head += 1
                    for v in adj[u]:
                        if color[v] == 0:
                            color[v] = 3 - color[u]
                            q.append(v)
                        elif color[v] == color[u]:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return
            
            for node in q:
                res[node] = color[node]
                
    print("YES")
    print(*(res[1:]))

solve()
