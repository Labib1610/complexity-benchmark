```python
import sys
from collections import deque

def solve() -> None:
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    it = iter(data)
    n = next(it)
    m = next(it)

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = next(it)
        v = next(it)
        if u == v:
            # self-loop forces non-bipartite
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)  # 0 or 1
    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # start new component with color 0 (mode 1)
        color[start] = 0
        q = deque([start])
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                elif color[v] == cu:
                    print("NO")
                    return

    # Output result
    modes = [str(c + 1) for c in color[1:]]
    print("YES")
    print(" ".join(modes))

if __name__ == "__main__":
    solve()
```
