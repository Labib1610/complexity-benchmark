import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is not yet colored, we greedily pick 1, then propagate.
    # This is a standard 2-coloring problem (bipartite check).
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS component
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            # Using a stack for iterative DFS to avoid recursion limit issues
            q = [i]
            assignment[i] = 1
            idx_q = 0
            while idx_q < len(q):
                u = q[idx_q]
                idx_q += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
