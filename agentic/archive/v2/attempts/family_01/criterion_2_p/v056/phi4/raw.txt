import sys

def main():
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    conditions = []
    index = 2
    
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index + 1]) - 1
        conditions.append((u, v))
        index += 2

    adj_list = [[] for _ in range(n)]
    for u, v in conditions:
        adj_list[u].append(v)
        adj_list[v].append(u)

    mode_assignment = [0] * n
    
    def is_bipartite(node, color):
        stack = [(node, color)]
        
        while stack:
            current, current_color = stack.pop()
            
            if mode_assignment[current] == 0:
                mode_assignment[current] = current_color
            elif mode_assignment[current] != current_color:
                return False
            
            for neighbor in adj_list[current]:
                if mode_assignment[neighbor] == 0:
                    stack.append((neighbor, 3 - current_color))
        
        return True

    valid = True
    for i in range(n):
        if mode_assignment[i] == 0 and not is_bipartite(i, 1):
            valid = False
            break
    
    if not valid:
        print("NO")
    else:
        result = [str(2 if x == 0 else x) for x in mode_assignment]
        print("YES")
        print(' '.join(result))

if __name__ == "__main__":
    main()
