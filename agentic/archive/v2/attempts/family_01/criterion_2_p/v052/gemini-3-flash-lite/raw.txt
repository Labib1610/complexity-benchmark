import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest, we iterate 1..n
    # and try to assign 1 if possible, else 2.
    # Because this is a bipartite check, once a component's 
    # start node is assigned, the rest are fixed.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # New component. Try assigning 1 to this node.
            stack = [(i, 1)]
            assignment[i] = 1
            component = []
            
            # BFS/DFS to assign the whole connected component
            idx_comp = 0
            comp_nodes = [i]
            assignment[i] = 1
            
            # Use stack for DFS
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

solve()
