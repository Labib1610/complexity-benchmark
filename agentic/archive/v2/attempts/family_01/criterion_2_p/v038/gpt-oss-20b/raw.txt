import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0 or 1 relative to component root
    ans = [0] * n             # final 1/2 values
    for start in range(n):
        if color[start] != -1:
            continue
        # BFS component
        dq = deque([start])
        color[start] = 0
        comp = [start]
        ok = True
        while dq:
            u = dq.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    dq.append(v)
                    comp.append(v)
                elif color[v] == cu:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            print("NO")
            return
        # Determine orientation: set side of min vertex to 1
        min_v = min(comp)
        min_color = color[min_v]
        for v in comp:
            ans[v] = 1 if color[v] == min_color else 2

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
