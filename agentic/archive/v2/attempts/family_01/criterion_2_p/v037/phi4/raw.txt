```python
import sys
from collections import defaultdict

def main():
    input = sys.stdin.read
    data = input().split()
    
    idx = 0
    n = int(data[idx])
    m = int(data[idx + 1])
    idx += 2
    
    graph = defaultdict(set)
    
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        graph[u].add(v)
        graph[v].add(u)
        idx += 2
    
    shift_assignment = [1] * (n + 1)  # Assign all to shift 1 initially
    possible = True
    
    for person in range(1, n + 1):
        if len(graph[person]) == n - 1:
            # If the person is incompatible with everyone else, no solution exists
            possible = False
            break
        
        if shift_assignment[person] == 1:
            continue
        
        conflict_shifts = set(shift_assignment[i] for i in graph[person])
        
        if len(conflict_shifts) == 2:
            # If both shifts are already represented among incompatible persons, no solution exists
            possible = False
            break
        elif len(conflict_shifts) == 1 and 2 not in conflict_shifts:
            # If only one shift is represented and it's not shift 2, assign shift 2 to this person
            shift_assignment[person] = 2
    
    if not possible or any(len(graph[i]) == n - 1 for i in range(1, n + 1)):
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, shift_assignment[1:])))

if __name__ == "__main__":
    main()
```
