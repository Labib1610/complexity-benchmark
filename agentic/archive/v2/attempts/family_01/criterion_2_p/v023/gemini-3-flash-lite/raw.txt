import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    N = int(input_data[0])
    M = int(input_data[1])
    
    adj = [[] for _ in range(N + 1)]
    idx = 2
    for _ in range(M):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (N + 1)
    
    # To get the lexicographically smallest sequence,
    # we iterate from 1 to N and try assigning color 1 first.
    # If a component is not colored, we start BFS/DFS from it.
    # Since we want the smallest sequence, for each connected component,
    # we check if assigning 1 to the smallest node in the component
    # leads to a valid coloring.
    
    for i in range(1, N + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # Assign color 1 to the current node i
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                # Sort neighbors to ensure deterministic behavior if needed,
                # though for 2-coloring, the choice is forced once the first node is set.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
