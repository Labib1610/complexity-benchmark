def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u, v))
    
    def is_valid(assignment):
        for u, v in edges:
            if assignment[u-1] == assignment[v-1]:
                return False
        return True
    
    def find_assignment():
        assignment = [0] * n
        
        def backtrack(index):
            if index == n:
                if is_valid(assignment):
                    return assignment
                else:
                    return None
            
            assignment[index] = 1
            result = backtrack(index + 1)
            if result:
                return result
            
            assignment[index] = 2
            result = backtrack(index + 1)
            if result:
                return result
            
            return None
        
        return backtrack(0)
    
    assignment = find_assignment()
    
    if assignment is None:
        print("NO")
    else:
        print("YES")
        print(*assignment)

solve()
