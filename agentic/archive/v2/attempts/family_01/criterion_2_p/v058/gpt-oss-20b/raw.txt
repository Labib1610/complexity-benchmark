```python
import sys
from collections import deque

def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return
    n, m = data[0], data[1]
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = data[idx]
        v = data[idx + 1]
        idx += 2
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)  # 0 or 1
    for i in range(1, n + 1):
        if color[i] != -1:
            continue
        # start new component
        color[i] = 0
        q = deque([i])
        while q:
            v = q.popleft()
            cv = color[v]
            for nb in adj[v]:
                if color[nb] == -1:
                    color[nb] = 1 - cv
                    q.append(nb)
                elif color[nb] == cv:
                    sys.stdout.write("NO")
                    return

    # All components bipartite, construct answer
    out = ['1' if color[i] == 0 else '2' for i in range(1, n + 1)]
    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(out))

if __name__ == "__main__":
    main()
```
