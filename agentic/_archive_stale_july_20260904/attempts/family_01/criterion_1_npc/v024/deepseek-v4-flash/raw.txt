def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[False] * n for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u][v] = adj[v][u] = True

    # Check if graph is 3-colorable and find lexicographically smallest coloring
    # Since n <= 40, we can use backtracking with pruning
    color = [0] * n  # 0 = uncolored, 1,2,3 = colors

    # Precompute adjacency list
    adj_list = [[] for _ in range(n)]
    for i in range(n):
        for j in range(n):
            if adj[i][j]:
                adj_list[i].append(j)

    # Check if graph is bipartite (2-colorable) for early exit
    # But we need 3-coloring, so just do backtracking

    # Greedy with backtracking for lexicographically smallest
    # We'll try to assign colors one by one
    # To get lexicographically smallest, we try colors in order for each vertex

    # But we need to ensure we can complete the coloring
    # Use backtracking with forward checking

    # Precompute degree for ordering (optional)

    # Simple backtracking
    def backtrack(idx):
        if idx == n:
            return True
        for c in range(1, 4):
            # Check if we can assign color c to vertex idx
            ok = True
            for nb in adj_list[idx]:
                if nb < idx and color[nb] == c:
                    ok = False
                    break
            if ok:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
