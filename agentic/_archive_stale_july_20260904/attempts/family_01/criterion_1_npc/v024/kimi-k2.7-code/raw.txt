import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    comp_of = [-1] * n
    local_index = [0] * n
    components = []

    for s in range(n):
        if comp_of[s] != -1:
            continue
        stack = [s]
        comp_of[s] = len(components)
        verts = []
        while stack:
            v = stack.pop()
            verts.append(v)
            a = adj[v]
            while a:
                lsb = a & -a
                u = lsb.bit_length() - 1
                a ^= lsb
                if comp_of[u] == -1:
                    comp_of[u] = len(components)
                    stack.append(u)
        k = len(verts)
        for i, v in enumerate(verts):
            local_index[v] = i
        local_adj = [0] * k
        neigh = [[] for _ in range(k)]
        for i, v in enumerate(verts):
            a = adj[v]
            mask = 0
            while a:
                lsb = a & -a
                u = lsb.bit_length() - 1
                a ^= lsb
                if comp_of[u] == len(components):
                    j = local_index[u]
                    mask |= 1 << j
                    neigh[i].append(j)
            local_adj[i] = mask
        components.append([verts, local_adj, neigh, k, {}])

    sys.setrecursionlimit(10000)

    def can_color(comp_idx, f1, f2, f3):
        comp = components[comp_idx]
        local_adj, neigh, k, memo = comp[1], comp[2], comp[3], comp[4]
        color = [0] * k
        avail = [7] * k

        m = f1
        while m:
            lsb = m & -m
            v = lsb.bit_length() - 1
            m ^= lsb
            if color[v]:
                return False
            if local_adj[v] & f1:
                return False
            color[v] = 1
            avail[v] = 1

        m = f2
        while m:
            lsb = m & -m
            v = lsb.bit_length() - 1
            m ^= lsb
            if color[v]:
                return False
            if local_adj[v] & f2:
                return False
            color[v] = 2
            avail[v] = 2

        m = f3
        while m:
            lsb = m & -m
            v = lsb.bit_length() - 1
            m ^= lsb
            if color[v]:
                return False
            if local_adj[v] & f3:
                return False
            color[v] = 3
            avail[v] = 4

        fixed_mask = f1 | f2 | f3
        stack = [v for v in range(k) if color[v]]

        while stack:
            v = stack.pop()
            cbit = avail[v]
            for w in neigh[v]:
                if color[w] == 0 and (avail[w] & cbit):
                    avail[w] ^= cbit
                    if avail[w] == 0:
                        return False
                    if avail[w] & (avail[w] - 1) == 0:
                        if avail[w] == 1:
                            color[w] = 1
                        elif avail[w] == 2:
                            color[w] = 2
                        else:
                            color[w] = 3
                        fixed_mask |= 1 << w
                        stack.append(w)

        g1 = g2 = g3 = 0
        assigned = 0
        for v in range(k):
            c = color[v]
            if c:
                assigned |= 1 << v
                if c == 1:
                    g1 |= 1 << v
                elif c == 2:
                    g2 |= 1 << v
                else:
                    g3 |= 1 << v

        key = (g1, g2, g3)
        if key in memo:
            return memo[key]

        if assigned == (1 << k) - 1:
            memo[key] = True
            return True

        unassigned_mask = ((1 << k) - 1) ^ assigned
        best = -1
        best_sz = 4
        best_deg = -1
        for v in range(k):
            if color[v] == 0:
                sz = avail[v].bit_count()
                deg = (local_adj[v] & unassigned_mask).bit_count()
                if sz < best_sz or (sz == best_sz and deg > best_deg):
                    best = v
                    best_sz = sz
                    best_deg = deg

        bits = avail[best]
        for cbit in (1, 2, 4):
            if bits & cbit:
                if cbit == 1:
                    res = can_color(comp_idx, g1 | (1 << best), g2, g3)
                elif cbit == 2:
                    res = can_color(comp_idx, g1, g2 | (1 << best), g3)
                else:
                    res = can_color(comp_idx, g1, g2, g3 | (1 << best))
                if res:
                    memo[key] = True
                    return True

        memo[key] = False
        return False

    ans = [0] * n

    for i in range(n):
        ci = comp_of[i]
        li = local_index[i]
        verts = components[ci][0]
        base1 = base2 = base3 = 0
        for v in verts:
            c = ans[v]
            if c == 1:
                base1 |= 1 << local_index[v]
            elif c == 2:
                base2 |= 1 << local_index[v]
            elif c == 3:
                base3 |= 1 << local_index[v]

        ok = False
        for c in (1, 2, 3):
            conflict = False
            nb = adj[i]
            while nb:
                lsb = nb & -nb
                j = lsb.bit_length() - 1
                nb ^= lsb
                if ans[j] == c:
                    conflict = True
                    break
            if conflict:
                continue
            if c == 1:
                f1, f2, f3 = base1 | (1 << li), base2, base3
            elif c == 2:
                f1, f2, f3 = base1, base2 | (1 << li), base3
            else:
                f1, f2, f3 = base1, base2, base3 | (1 << li)
            if can_color(ci, f1, f2, f3):
                ans[i] = c
                ok = True
                break

        if not ok:
            print("NO")
            return

    out = ["YES", " ".join(str(x) for x in ans)]
    sys.stdout.write("\n".join(out))

solve()
