We can model this as a graph coloring problem with 3 colors.  
Since \( n \leq 40 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        edges.append((u, v))
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    colors = [0] * n
    
    def is_safe(node, color):
        for nei in adj[node]:
            if colors[nei] == color:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
        for c in range(1, 4):
            if can_safe(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False
    
    def can_safe(node, c):
        for nei in adj[node]:
            if colors[nei] == c:
                return False
        return True
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
