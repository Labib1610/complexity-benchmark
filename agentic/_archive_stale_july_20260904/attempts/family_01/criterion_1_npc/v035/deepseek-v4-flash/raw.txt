We can solve this using backtracking with pruning, since \( n \le 40 \) is small enough for a well-optimized search.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    conflicts = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        conflicts[u].append(v)
        conflicts[v].append(u)

    assignment = [0] * n

    def backtrack(i):
        if i == n:
            return True
        for shift in (1, 2, 3):
            ok = True
            for j in conflicts[i]:
                if assignment[j] == shift:
                    ok = False
                    break
            if ok:
                assignment[i] = shift
                if backtrack(i + 1):
                    return True
                assignment[i] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
