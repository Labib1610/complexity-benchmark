def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Create adjacency list
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Try all possible colorings in lexicographical order
    # Use backtracking with early termination
    coloring = [0] * (n + 1)  # 0 means uncolored, 1,2,3 are the colors
    
    def is_valid(node, color):
        # Check if assigning 'color' to 'node' conflicts with neighbors
        for neighbor in adj[node]:
            if coloring[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for color in range(1, 4):
            if is_valid(node, color):
                coloring[node] = color
                if backtrack(node + 1):
                    return True
                coloring[node] = 0  # backtrack
                
        return False
    
    # Start backtracking from node 1
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, coloring[1:])))
    else:
        print("NO")

solve()
