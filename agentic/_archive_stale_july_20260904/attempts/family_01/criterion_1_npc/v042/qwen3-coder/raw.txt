def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Try all possible colorings
    # We want the lexicographically smallest one
    coloring = [0] * (n + 1)  # coloring[i] is the color of node i
    
    def is_valid(node, color):
        for neighbor in adj[node]:
            if coloring[neighbor] == color:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for color in [1, 2, 3]:
            if is_valid(node, color):
                coloring[node] = color
                if backtrack(node + 1):
                    return True
                coloring[node] = 0
        return False
    
    # Start backtracking from node 1
    if backtrack(1):
        print("YES")
        print(' '.join(map(str, coloring[1:])))
    else:
        print("NO")

solve()
