We can solve this by trying to assign channels greedily while checking feasibility with backtracking.  
Since \( n \leq 40 \), we can use a recursive search with pruning.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    colors = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        for c in range(1, 4):
            ok = True
            # check neighbors
            mask = adj[idx]
            for v in range(n):
                if (mask >> v) & 1:
                    if colors[v] == c:
                        ok = False
                        break
            if ok:
                colors[idx] = c
                if backtrack(idx + 1):
                    return True
                colors[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
