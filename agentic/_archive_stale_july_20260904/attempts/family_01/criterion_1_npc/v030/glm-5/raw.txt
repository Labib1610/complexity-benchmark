import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # We need to find the lexicographically smallest valid assignment.
    # Colors are 1, 2, 3.
    # We use backtracking with pruning.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        for c in range(1, 4):
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, colors)) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
