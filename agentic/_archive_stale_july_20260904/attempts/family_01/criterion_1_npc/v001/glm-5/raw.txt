import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # forbidden[u] = bitmask of shifts that worker u cannot take
    forbidden = [0] * (n + 1)
    color = [0] * (n + 1)
    possible = True

    for u in range(1, n + 1):
        if color[u] != 0:
            continue
        # start BFS/DFS for this component
        stack = [u]
        color[u] = 1
        comp = []
        while stack:
            x = stack.pop()
            comp.append(x)
            y = 1
            while y <= 3:
                if color[x] != y and (adj[x] & (1 << y)) == 0:
                    break
                y += 1
            if y > 3:
                possible = False
                break
            color[x] = y
            # propagate: neighbors cannot take shift y
            nb = adj[x]
            while nb:
                b = nb & -nb
                v = b.bit_length() - 1
                nb ^= b
                if color[v] == 0:
                    if (forbidden[v] & (1 << y)) == 0:
                        forbidden[v] |= 1 << y
                        stack.append(v)
                else:
                    if color[v] == y:
                        possible = False
                        break
            if not possible:
                break
        if not possible:
            break

    if not possible:
        print("NO")
        return

    # Greedy lexicographically smallest assignment
    ans = [0] * (n + 1)
    for u in range(1, n + 1):
        for c in range(1, 4):
            if (forbidden[u] & (1 << c)) == 0:
                ok = True
                nb = adj[u]
                while nb:
                    b = nb & -nb
                    v = b.bit_length() - 1
                    nb ^= b
                    if v < u and ans[v] == c:
                        ok = False
                        break
                if ok:
                    ans[u] = c
                    break
        else:
            print("NO")
            return

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

solve()
