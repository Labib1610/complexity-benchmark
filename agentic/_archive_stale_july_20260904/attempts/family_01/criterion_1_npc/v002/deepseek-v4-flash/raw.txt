We can model this as a graph coloring problem with 3 colors, where each color is a storage unit.  
Since \( n \leq 40 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
    
    color = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        # Try colors 1,2,3 in order for lexicographic smallest
        for c in (1, 2, 3):
            ok = True
            for j in range(n):
                if (adj[idx] >> j) & 1 and color[j] == c:
                    ok = False
                    break
            if ok:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
