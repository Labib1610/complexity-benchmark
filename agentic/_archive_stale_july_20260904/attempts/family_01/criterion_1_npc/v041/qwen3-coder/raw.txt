def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list for the graph
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all possible colorings with 3 colors
    # We want the lexicographically smallest assignment
    # So we'll use backtracking with early termination
    
    # Color assignment: 0 means unassigned, 1,2,3 are the blocks
    color = [0] * (n + 1)
    
    def is_safe(node, c):
        # Check if assigning color c to node is safe
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node > n:
            return True
            
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0
        return False
    
    # Check if graph is 3-colorable
    if backtrack(1):
        print("YES")
        print(*color[1:])
    else:
        print("NO")

solve()
