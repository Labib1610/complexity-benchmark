def solve():
    import sys
    from collections import defaultdict
    
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    conflicts = defaultdict(set)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-indexed
        v = int(data[index+1]) - 1
        conflicts[u].add(v)
        conflicts[v].add(u)
        index += 2
    
    # Try all 3^n assignments, but prune early for efficiency
    # We'll use backtracking with constraint propagation
    
    assignment = [-1] * n
    
    def is_valid(worker, shift):
        for neighbor in conflicts[worker]:
            if assignment[neighbor] == shift:
                return False
        return True
    
    def backtrack(pos):
        if pos == n:
            return True
            
        for shift in [1, 2, 3]:
            if is_valid(pos, shift):
                assignment[pos] = shift
                if backtrack(pos + 1):
                    return True
                assignment[pos] = -1
        return False
    
    # For better performance on structured inputs, we can do a greedy coloring first
    # But since n <= 40 and we're looking for lex smallest, let's just try direct backtracking
    # with early termination based on lexicographic order
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

solve()
