import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * (n + 1) for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u][v] = True
        adj[v][u] = True
    
    # We need to find the lexicographically smallest valid assignment
    # Use backtracking with pruning
    
    assignment = [0] * (n + 1)
    found = False
    
    def backtrack(idx):
        nonlocal found
        if found:
            return
        if idx > n:
            found = True
            return
        
        for env in [1, 2, 3]:
            valid = True
            for prev in range(1, idx):
                if adj[idx][prev] and assignment[prev] == env:
                    valid = False
                    break
            if valid:
                assignment[idx] = env
                backtrack(idx + 1)
                if found:
                    return
                assignment[idx] = 0
    
    backtrack(1)
    
    if found:
        print("YES")
        print(' '.join(map(str, assignment[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
