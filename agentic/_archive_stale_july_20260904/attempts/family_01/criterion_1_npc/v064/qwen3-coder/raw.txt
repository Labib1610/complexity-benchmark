def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Colors: 1, 2, 3 (0 means uncolored)
    colors = [0] * n
    
    # Try to color each node with the smallest possible color
    for i in range(n):
        # If already colored, skip
        if colors[i] != 0:
            continue
            
        # Find forbidden colors for neighbors
        forbidden = [False] * 4  # index 0 unused, 1-3 for colors
        for neighbor in adj[i]:
            if colors[neighbor] != 0:
                forbidden[colors[neighbor]] = True
        
        # Assign the smallest available color
        found_color = False
        for c in range(1, 4):
            if not forbidden[c]:
                colors[i] = c
                found_color = True
                break
                
        # If no color is available, it's impossible
        if not found_color:
            print("NO")
            return
    
    # Verify the solution
    for i in range(n):
        for neighbor in adj[i]:
            if colors[i] == colors[neighbor]:
                print("NO")
                return
                
    print("YES")
    print(' '.join(map(str, colors)))

solve()
