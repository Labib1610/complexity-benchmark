import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # Tier assignment array, 0 means unassigned
    tier = [0] * n
    
    # Check if assigning t to node u is valid
    def is_valid(u, t):
        for v in adj[u]:
            if tier[v] == t:
                return False
        return True
    
    # Backtracking with pruning
    # We need lexicographically smallest, so try tiers 1, 2, 3 in order
    found = False
    
    def backtrack(u):
        nonlocal found
        if found:
            return
        if u == n:
            found = True
            return
        
        for t in [1, 2, 3]:
            if is_valid(u, t):
                tier[u] = t
                backtrack(u + 1)
                if found:
                    return
                tier[u] = 0
    
    backtrack(0)
    
    if not found:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, tier)))

if __name__ == "__main__":
    solve()
