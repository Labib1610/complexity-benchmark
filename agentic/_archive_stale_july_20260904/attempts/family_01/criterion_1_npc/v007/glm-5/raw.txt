import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0]); m = int(data[1])
    adj = [0]*n
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1]); idx += 2
        u -= 1; v -= 1
        adj[u] |= (1<<v)
        adj[v] |= (1<<u)
    
    # forbidden[u] = bitmask of colors that worker u cannot take
    # color 1 -> bit 0, color 2 -> bit 1, color 3 -> bit 2
    forbidden = [0]*n
    for u in range(n):
        f = 0
        # if neighbor has color 1, u cannot have color 1 (product 1*1=1 < 2)
        f |= 0b001
        # if neighbor has color 2, u cannot have color 1 (product 1*2=2 >= 2 ok) or 2 (2*2=4 ok)
        # so no restriction from color 2 neighbor
        # if neighbor has color 3, u cannot have color 1 (product 1*3=3 >= 2 ok)
        # so no restriction from color 3 neighbor
        forbidden[u] = f
    
    # We need to assign colors 1,2,3 to each worker.
    # Constraint: for each edge (u,v), colors must differ (since same color gives product c*c; for c=1 product=1<2, for c=2 product=4>=2, for c=3 product=9>=2).
    # Wait, same color 2 gives product 4 >= 2, same color 3 gives product 9 >= 2.
    # So the only forbidden same-color is color 1.
    # But the problem says "must not be placed on the same shift" for conflicting pairs.
    # So conflicting pairs must be on different shifts entirely.
    # That means it's a proper 3-coloring.
    
    # So we need a proper 3-coloring of the conflict graph.
    # Lexicographically smallest assignment.
    
    # Since n <= 40, we can use backtracking with pruning.
    # But 3^40 is too large. We need smarter approach.
    # Graph 3-coloring is NP-hard, but n=40 is small enough for backtracking with good pruning.
    
    # Let's use backtracking with forward checking / constraint propagation.
    
    color = [0]*n  # 0 = unassigned, 1,2,3 = colors
    
    # available[u] = bitmask of available colors (bit 0 = color 1, bit 1 = color 2, bit 2 = color 3)
    available = [0b111]*n
    
    # For lexicographically smallest, we try colors in order 1,2,3 for each worker from 0 to n-1.
    
    # Precompute: if we assign color c to worker u, which workers' available sets shrink.
    
    # Use recursive backtracking with forward checking.
    
    # To speed up, we can use bit manipulation and memoization on remaining subproblem.
    # But the state depends on which colors are assigned to which, not just the set of unassigned.
    
    # Let's try simple backtracking with forward checking first.
    
    def backtrack(pos):
        if pos == n:
            return True
        # Try colors 1, 2, 3 in order
        for c in range(1, 4):
            if available[pos] & (1 << (c-1)):
                # Check if this color is valid with already assigned neighbors
                valid = True
                for v in range(n):
                    if (adj[pos] >> v) & 1 and color[v] == c:
                        valid = False
                        break
                if not valid:
                    continue
                # Forward checking: update available for unassigned neighbors
                old_available = {}
                for v in range(n):
                    if (adj[pos] >> v) & 1 and color[v] == 0:
                        if available[v] & (1 << (c-1)):
                            old_available[v] = available[v]
                            available[v] &= ~(1 << (c-1))
                            if available[v] == 0:
                                # No colors available for v, backtrack
                                # Restore and fail
                                for vv, old in old_available.items():
                                    available[vv] = old
                                valid = False
                                break
                if valid:
                    color[pos] = c
                    if backtrack(pos + 1):
                        return True
                    color[pos] = 0
                # Restore available
                for v, old in old_available.items():
                    available[v] = old
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(str(c) for c in color))
    else:
        print("NO")

solve()
