We can model this as a graph coloring problem with 3 colors (zones), where incompatible pairs cannot share a color.  
Since \( n \leq 40 \) and \( m \leq 780 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

Here’s the Python solution:  

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * n

    def backtrack(idx):
        if idx == n:
            return True
        for c in range(1, 4):
            # Check if any neighbor already has this color
            ok = True
            for nb in adj[idx]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
