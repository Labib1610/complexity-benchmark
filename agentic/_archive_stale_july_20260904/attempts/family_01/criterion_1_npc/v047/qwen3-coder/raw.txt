import sys
from itertools import combinations

def read_input():
    n, m = map(int, sys.stdin.readline().split())
    incompatibles = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        incompatibles.append((u, v))
    return n, m, incompatibles

def is_valid_deployment(assignment, incompatibles):
    # Create a set of incompatible pairs for faster lookup
    incompatible_set = set(incompatibles)
    
    # Check that no two incompatible personnel are in the same zone
    for u, v in incompatibles:
        if assignment[u-1] == assignment[v-1]:
            return False
    
    return True

def generate_assignments(n):
    # Generate all possible assignments in lexicographical order
    # by treating it as base-3 numbers
    for i in range(3**n):
        assignment = []
        temp = i
        for _ in range(n):
            assignment.append((temp % 3) + 1)
            temp //= 3
        yield assignment[::-1]  # Reverse to get the right order

def solve():
    n, m, incompatibles = read_input()
    
    # Special case: if there are no incompatibilities, assign everyone to zone 1
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # For small n, we can try all possible assignments
    # But since n can be up to 40, we need a smarter approach
    # Let's use backtracking to find the lexicographically smallest valid assignment
    
    # We'll model this as a graph coloring problem
    # Personnel are nodes, incompatible pairs are edges
    # We want to color with 3 colors (zones 1,2,3) such that adjacent nodes have different colors
    # And among all valid colorings, we want the lexicographically smallest one
    
    # Build adjacency list
    adj = [[] for _ in range(n+1)]
    for u, v in incompatibles:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try to color the graph using backtracking
    assignment = [0] * (n+1)  # assignment[i] is the zone for personnel i
    
    def backtrack(personnel):
        if personnel > n:
            return True
            
        # Try assigning zones 1, 2, 3 in order to get lexicographically smallest
        for zone in [1, 2, 3]:
            # Check if this zone is valid for this personnel
            valid = True
            for neighbor in adj[personnel]:
                if assignment[neighbor] == zone:
                    valid = False
                    break
                    
            if valid:
                assignment[personnel] = zone
                if backtrack(personnel + 1):
                    return True
                assignment[personnel] = 0  # Backtrack
                
        return False
    
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, assignment[1:])))
    else:
        print("NO")

solve()
