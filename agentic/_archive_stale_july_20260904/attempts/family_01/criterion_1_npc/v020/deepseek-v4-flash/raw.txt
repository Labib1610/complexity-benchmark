We can model this as a graph coloring problem with 3 colors.  
Since \( n \le 40 \), we can use backtracking with pruning to find the lexicographically smallest valid assignment.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[False] * n for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u][v] = True
        adj[v][u] = True
    
    color = [0] * n  # 0 = unassigned, 1,2,3 = bus numbers
    
    # Precompute forbidden colors for each node
    forbidden = [0] * n  # bitmask of colors already used by neighbors
    
    def backtrack(idx):
        if idx == n:
            return True
        # Try colors 1,2,3 in order for lexicographic smallest
        for c in range(1, 4):
            if not (forbidden[idx] & (1 << c)):
                color[idx] = c
                # update forbidden for neighbors
                added = []
                for j in range(n):
                    if adj[idx][j] and not (forbidden[j] & (1 << c)):
                        forbidden[j] |= (1 << c)
                        added.append(j)
                if backtrack(idx + 1):
                    return True
                # revert
                for j in added:
                    forbidden[j] &= ~(1 << c)
                color[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(str(color[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
