import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Precompute for each subset of vertices whether it is an independent set.
    size = 1 << n
    is_indep = [False] * size
    is_indep[0] = True
    for s in range(1, size):
        low = s & -s
        v = low.bit_length() - 1
        prev = s ^ low
        if is_indep[prev] and (adj[v + 1] & prev) == 0:
            is_indep[s] = True

    # Find all maximal independent sets (MIS) using Bron-Kerbosch with pivoting.
    mis_list = []
    def bron_kerbosch(R, P, X):
        if P == 0 and X == 0:
            mis_list.append(R)
            return
        # Choose pivot u from P | X that maximizes |P & ~adj[u]|
        union = P | X
        # iterate over bits in union
        pivot = -1
        best = -1
        temp = union
        while temp:
            low = temp & -temp
            u = low.bit_length() - 1
            cand = P & (~adj[u + 1])
            cnt = bin(cand).count('1')
            if cnt > best:
                best = cnt
                pivot = u
            temp ^= low
        # iterate over v in P & ~adj[pivot]
        if pivot == -1:
            candidates = P
        else:
            candidates = P & (~adj[pivot + 1])
        temp = candidates
        while temp:
            low = temp & -temp
            v = low.bit_length() - 1
            vbit = 1 << v
            bron_kerbosch(R | vbit, P & (~adj[v + 1]), X & (~adj[v + 1]))
            P ^= vbit
            X |= vbit
            temp ^= low

    full = (1 << n) - 1
    bron_kerbosch(0, full, 0)

    # We need to cover all vertices with at most 3 independent sets.
    # Since each color class is an independent set, and we can always restrict
    # each color class to a maximal independent set (extending it doesn't violate independence),
    # we only need to check combinations of at most 3 MIS.
    # To get lexicographically smallest coloring, we try to assign color 1 to the
    # lexicographically smallest maximal independent set, then color 2 to the
    # smallest MIS of the remaining vertices, etc.

    # Sort MIS by their integer value (which corresponds to lexicographic order of
    # the characteristic vector from low bit to high bit, i.e., vertex 1 first).
    mis_list.sort()

    # Try to find 3 MIS that cover all vertices.
    # We want lexicographically smallest coloring, so we greedily pick the smallest
    # possible MIS for color 1, then smallest for color 2, etc.
    # But we must ensure the remaining can be covered by 2 independent sets.
    # Since we restrict to MIS, the remaining must be coverable by 2 MIS.

    # Precompute: for each MIS, which vertices it covers
    # We'll do a greedy with backtracking limited to 3 levels.

    # Actually, let's just try all combinations of up to 3 MIS, but in lexicographic order
    # of the resulting coloring. The coloring is determined by the first MIS that contains
    # each vertex. So to get lexicographically smallest, we want the first vertex (vertex 1)
    # to be in the first chosen MIS if possible, etc.

    # Greedy approach: pick the smallest MIS (in integer value) that contains vertex 1,
    # then among remaining vertices, pick smallest MIS containing the smallest remaining
    # vertex, etc. But this might not always work if a different first choice is needed.

    # Given n <= 40 and m <= 780, but the constraint "product of IDs <= 1600" means
    # edges only exist between vertices i, j with i*j <= 1600. For i >= 1, j <= 1600/i.
    # So high-numbered vertices (e.g., > 40) have very few edges. But n <= 40 anyway.
    # The number of MIS could be large in general, but with this edge constraint,
    # the graph is very sparse for high-numbered vertices.

    # Let's just do a proper backtracking search for 3-coloring with lexicographic order.
    # We assign colors to vertices 1..n in order, trying color 1, 2, 3.
    # This naturally gives lexicographically smallest if we try colors in increasing order.

    color = [0] * (n + 1)
    # For efficiency, use bitmask of available colors for each vertex.
    # But simple backtracking with pruning should work for n <= 40 given the constraints.

    # Actually, let's think about the structure. The constraint i*j <= 1600 means:
    # Vertex 1 can connect to anyone (1*j <= 1600, j <= 40, all).
    # Vertex 2 can connect to j <= 800, so all.
    # Vertex 3: j <= 533, all.
    # Vertex 4: j <= 400, all.
    # ...
    # Vertex 40: j <= 40, all.
    # So actually for n <= 40, the product constraint doesn't restrict anything since
    # 40*40 = 1600. So the graph can be any graph on 40 vertices with up to 780 edges.
    # 780 = 40*39/2, so it can be a complete graph.

    # For general 3-coloring with n=40, backtracking might be too slow.
    # But we can use the independent set cover approach.

    # Let's use the approach: find if 3 MIS cover all vertices, and find lexicographically
    # smallest such cover where "lexicographically smallest" is determined by the coloring.

    # The coloring from MIS cover: vertex gets color of the first MIS (in our chosen order)
    # that contains it. To minimize lexicographically, we want vertex 1 to have color 1 if
    # possible, meaning vertex 1 should be in the first MIS. Then vertex 2 should be in
    # the first MIS if possible (to get color 1), else in the second MIS (color 2), etc.

    # So the greedy: first MIS should be the lexicographically smallest maximal independent
    # set. But we need the remaining to be 2-colorable. If not, we need to try a different
    # first MIS.

    # Let's implement: iterate over MIS in sorted order for color 1. For each, check if
    # the remaining graph is bipartite. If so, find the lexicographically smallest
    # 2-coloring of the remaining, and combine. Among all valid options, pick the one
    # that gives the lexicographically smallest overall coloring.

    # But comparing all options could be expensive. Instead, we can be smarter:
    # For color 1, we want the lexicographically smallest MIS such that the remaining
    # is bipartite. But "lexicographically smallest MIS" doesn't directly translate to
    # "lexicographically smallest coloring" because a vertex not in the first MIS gets
    # color 2 or 3, and we want to minimize those too.

    # Actually, the lexicographically smallest coloring is achieved by:
    # - Vertex 1 gets color 1 if possible (i.e., there's a valid coloring with vertex 1 = color 1)
    # - Given that, vertex 2 gets color 1 if possible, else color 2 if possible, else color 3
    # - etc.

    # So we can do a standard backtracking that tries colors 1, 2, 3 for each vertex
    # in order, with pruning. The question is whether this is fast enough.

    # Given the special structure (product <= 1600), let's think about which vertices
    # have edges. For n <= 40, all pairs have product <= 1600, so no restriction.
    # But m <= 780, so the graph has at most 780 edges.

    # For 3-coloring, a simple backtracking with forward checking can handle n=40
    # if the graph isn't too dense. For dense graphs (like complete graph K4), it's fast
    # because it quickly finds no solution. For sparse graphs, it's also fast.

    # Let's implement backtracking with these optimizations:
    # 1. Try colors in order 1, 2, 3.
    # 2. Use forward checking: maintain available colors for each unassigned vertex.
    # 3. Most constrained variable heuristic: assign the vertex with fewest available colors.

    # But for lexicographically smallest, we must assign vertices in order 1, 2, ..., n.
    # We can't use MCV if we want lexicographically smallest. However, we can still use
    # forward checking.

    # Let's try simple backtracking in vertex order with forward checking.

    # available[i] = bitmask of available colors for vertex i (bits 0,1,2 for colors 1,2,3)
    available = [7] * (n + 1)  # 7 = 0b111

    # Order vertices by degree (descending) for better pruning? No, we need lex smallest.
    # But we can still prune using forward checking.

    result = None

    def backtrack(idx):
        nonlocal result
        if result is not None:
            return
        if idx > n:
            result = color[1:n+1][:]
            return
        # Try colors in order 1, 2, 3
        for c in range(1, 4):
            if (available[idx] >> (c - 1)) & 1:
                # Check if this color is valid (no conflict with already colored neighbors)
                valid = True
                # We don't need to check since available is maintained
                # Assign
                color[idx] = c
                # Save state
                changed = []
                # Update neighbors
                for j in range(1, n + 1):
                    if (adj[idx] >> j) & 1 and color[j] == 0:
                        if (available[j] >> (c - 1)) & 1:
                            available[j] &= ~(1 << (c - 1))
                            changed.append(j)
                            if available[j] == 0:
                                valid = False
                                break
                        # if color is already not available, no change
                if valid:
                    backtrack(idx + 1)
                    if result is not None:
                        return
                # Restore
                for j in changed:
                    available[j] |= (1 << (c - 1))
                color[idx] = 0

    backtrack(1)

    if result is None:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, result)))

solve()
