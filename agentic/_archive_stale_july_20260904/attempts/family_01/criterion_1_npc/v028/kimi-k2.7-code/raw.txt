```python
import sys

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1

    adj = [0] * n
    neighbors = [[] for _ in range(n)]

    for _ in range(m):
        u = int(data[idx]) - 1; idx += 1
        v = int(data[idx]) - 1; idx += 1
        if u == v:
            print("NO")
            return
        if not ((adj[u] >> v) & 1):
            adj[u] |= 1 << v
            adj[v] |= 1 << u
            neighbors[u].append(v)
            neighbors[v].append(u)

    color = [0] * n
    masks = [0, 0, 0, 0]
    block = [[0, 0, 0, 0] for _ in range(n)]
    ans = None

    def dfs(pos):
        nonlocal ans
        if pos == n:
            ans = color[:]
            return True
        bit = 1 << pos
        ap = adj[pos]
        for c in (1, 2, 3):
            if ap & masks[c]:
                continue
            color[pos] = c
            masks[c] |= bit
            dead = False
            for u in neighbors[pos]:
                if color[u] == 0:
                    block[u][c] += 1
                    if block[u][1] and block[u][2] and block[u][3]:
                        dead = True
                        break
            if not dead and dfs(pos + 1):
                return True
            for u in neighbors[pos]:
                if color[u] == 0:
                    block[u][c] -= 1
            masks[c] ^= bit
            color[pos] = 0
        return False

    if dfs(0):
        sys.stdout.write("YES\n" + " ".join(map(str, ans)))
    else:
        sys.stdout.write("NO")

solve()
```
