```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    # We use backtracking with pruning.
    # assignment[i] stores the block (1, 2, or 3) for worker i.
    assignment = [0] * n
    
    # Precompute neighbors for faster access
    # adj is already a list of sets
    
    def backtrack(index):
        if index == n:
            return True
        
        for block in [1, 2, 3]:
            valid = True
            # Check constraints with already assigned neighbors
            # We only need to check neighbors j < index since others are unassigned
            # Actually, we check all neighbors j such that assignment[j] != 0
            # But since we fill sequentially, j < index implies assignment[j] != 0
            # and j >= index implies assignment[j] == 0.
            # So checking neighbors in range(index) is sufficient and correct.
            for neighbor in adj[index]:
                if neighbor < index:
                    if assignment[neighbor] == block:
                        valid = False
                        break
            
            if valid:
                assignment[index] = block
                if backtrack(index + 1):
                    return True
                # Backtrack: assignment[index] will be overwritten or remains 0 if we return False
                # but explicit reset is cleaner for logic, though not strictly necessary if we overwrite.
                assignment[index] = 0
        
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
