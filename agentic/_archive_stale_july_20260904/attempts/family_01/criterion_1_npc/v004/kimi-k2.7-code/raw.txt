            return False, removed_mask, deg

import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data: return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0]*n
    for _ in range(m):
        u = int(next(it))-1
        v = int(next(it))-1
        if u == v:
            # self-loop impossible
            adj[u] |= 1<<u
        else:
            adj[u] |= 1<<v
            adj[v] |= 1<<u
    FULL = 7
    ALL = (1<<n)-1
    # precompute bit_to_color
    bit_to_color = {1:1,2:2,4:3}

    sys.setrecursionlimit(10000)

    def reduce(dom, removed):
        q = deque()
        active = ALL & ~removed
        am = active
        while am:
            lsb = am & -am
            v = lsb.bit_length() - 1
            q.append(v)
            am ^= lsb
        while q:
            v = q.popleft()
            if (removed >> v) & 1:
                continue
            d = dom[v]
            pc = d.bit_count()
            if pc == 0:
                return False, removed
            active_neighbors = adj[v] & ~removed
            deg = active_neighbors.bit_count()
            if pc == 1:
                c = d
                an = active_neighbors
                while an:
                    lsb = an & -an
                    u = lsb.bit_length() - 1
                    if dom[u] & c:
                        if dom[u] == c:
                            return False, removed
                        dom[u] &= ~c
                        q.append(u)
                    an ^= lsb
                removed |= 1 << v
            elif pc > deg:
                an = active_neighbors
                while an:
                    lsb = an & -an
                    u = lsb.bit_length() - 1
                    q.append(u)
                    an ^= lsb
                removed |= 1 << v
            else:
                # pure color reduction
                dd = d
                pure_bit = 0
                while dd:
                    b = dd & -dd
                    # check if any active neighbor has bit b
                    an = active_neighbors
                    found = False
                    while an:
                        lsb = an & -an
                        u = lsb.bit_length() - 1
                        if dom[u] & b:
                            found = True
                            break
                        an ^= lsb
                    if not found:
                        pure_bit = b
                        break
                    dd ^= b
                if pure_bit:
                    dom[v] = pure_bit
                    q.append(v)
        return True, removed

    def colorable(dom, removed=0):
        ok, removed = reduce(dom, removed)
        if not ok:
            return False
        active = ALL & ~removed
        if active == 0:
            return True
        # choose vertex

        best_pc = 10
        best_deg = -1
        am = active

            v = lsb.bit_length() - 1
            pc = dom[v].bit_count()

                if dom[u] & b:

                an ^= lsb
            cands.append((cnt, b))

            dom2[v] = b
            if colorable(dom2, removed):

        return False
