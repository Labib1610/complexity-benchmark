import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_sets = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj_sets[u].add(v)
        adj_sets[v].add(u)
    adj = [list(s) for s in adj_sets]

    color = [0] * n          # 0 = unassigned, otherwise 1..3
    avail = [7] * n          # bitmask of possible colors, bits 0,1,2 -> colors 1,2,3
    ans = None

    def assign(v: int, c: int, mods: list) -> bool:
        """Assign v=c and propagate forced assignments. mods records changes for undo."""
        if color[v] != 0:
            return color[v] == c
        mods.append((v, color[v], avail[v]))
        color[v] = c
        avail[v] = 1 << (c - 1)
        stack = [v]
        while stack:
            x = stack.pop()
            cx = color[x]
            remove = (~(1 << (cx - 1))) & 7
            for y in adj[x]:
                if color[y] != 0:
                    if color[y] == cx:
                        return False
                    continue
                old = avail[y]
                new_mask = old & remove
                if new_mask == 0:
                    return False
                if new_mask != old:
                    mods.append((y, color[y], old))
                    avail[y] = new_mask
                    # if only one color remains, force it
                    if new_mask & (new_mask - 1) == 0:
                        cy = new_mask.bit_length()
                        color[y] = cy
                        stack.append(y)
        return True

    def dfs() -> bool:
        nonlocal ans
        i = 0
        while i < n and color[i] != 0:
            i += 1
        if i == n:
            ans = color[:]
            return True
        mask = avail[i]
        bit = 1
        for c in (1, 2, 3):
            if mask & bit:
                mods = []
                if assign(i, c, mods):
                    if dfs():
                        return True
                for v, oldc, olda in reversed(mods):
                    color[v] = oldc
                    avail[v] = olda
            bit <<= 1
        return False

    if dfs():
        out = ['YES', ' '.join(str(x) for x in ans)]
    else:
        out = ['NO']
    sys.stdout.write('\n'.join(out))

if __name__ == '__main__':
    solve()
