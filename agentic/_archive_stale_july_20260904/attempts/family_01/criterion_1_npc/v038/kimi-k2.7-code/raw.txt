import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj_mask = [0] * n
    adj_list = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        if u == v:
            continue
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u
        adj_list[u].append(v)
        adj_list[v].append(u)

    colors = [0] * n
    masks = [0, 0, 0, 0]  # colors 1..3
    changes = []

    def set_color(v, c):
        colors[v] = c
        masks[c] |= 1 << v
        changes.append(v)
        for u in adj_list[v]:
            col = colors[u]
            if col:
                if col == c:
                    return False
                continue
            f = 0
            au = adj_mask[u]
            if masks[1] & au:
                f |= 1
            if masks[2] & au:
                f |= 2
            if masks[3] & au:
                f |= 4
            if f == 7:
                return False
            if f == 6:
                if not set_color(u, 1):
                    return False
            elif f == 5:
                if not set_color(u, 2):
                    return False
            elif f == 3:
                if not set_color(u, 3):
                    return False
        return True

    def dfs(idx):
        while idx < n and colors[idx]:
            idx += 1
        if idx == n:
            return True
        for c in (1, 2, 3):
            if adj_mask[idx] & masks[c]:
                continue
            base = len(changes)
            if set_color(idx, c):
                if dfs(idx + 1):
                    return True
            while len(changes) > base:
                v = changes.pop()
                col = colors[v]
                masks[col] &= ~(1 << v)
                colors[v] = 0
        return False

    if dfs(0):
        sys.stdout.write("YES\n" + " ".join(map(str, colors)) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == "__main__":
    solve()
