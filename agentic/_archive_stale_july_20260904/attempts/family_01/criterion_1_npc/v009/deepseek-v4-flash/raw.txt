We can solve this using backtracking with pruning.  
Since \( n \leq 40 \), a simple brute force over all \( 3^n \) assignments is too large, but backtracking with conflict checks will work because the graph is small and we stop early when possible.

Here’s the code:

```python
n, m = map(int, input().split())
conflicts = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    conflicts[u].append(v)
    conflicts[v].append(u)

assignment = [0] * n
found = False

def backtrack(idx):
    global found
    if idx == n:
        found = True
        return True
    for shift in range(1, 4):
        ok = True
        for neighbor in conflicts[idx]:
            if assignment[neighbor] == shift:
                ok = False
                break
        if ok:
            assignment[idx] = shift
            if backtrack(idx + 1):
                return True
            assignment[idx] = 0
    return False

if backtrack(0):
    print("YES")
    print(" ".join(map(str, assignment)))
else:
    print("NO")
```
