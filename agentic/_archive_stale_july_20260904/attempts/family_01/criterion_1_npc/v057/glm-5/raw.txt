import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n
    
    # Check if graph is 3-colorable and find lexicographically smallest coloring
    # Using backtracking with constraint propagation
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    # Get saturation degree (number of different colors used by neighbors)
    def get_saturation(node):
        used = set()
        for neighbor in adj[node]:
            if color[neighbor] != 0:
                used.add(color[neighbor])
        return len(used)
    
    # Get degree of uncolored neighbors
    def get_degree(node):
        return sum(1 for neighbor in adj[node] if color[neighbor] == 0)
    
    def solve_recursive(idx):
        if idx == n:
            return True
        
        # Find next uncolored node with highest saturation, then highest degree
        best_node = -1
        best_sat = -1
        best_deg = -1
        
        for i in range(n):
            if color[i] == 0:
                sat = get_saturation(i)
                deg = get_degree(i)
                if sat > best_sat or (sat == best_sat and deg > best_deg):
                    best_sat = sat
                    best_deg = deg
                    best_node = i
        
        if best_node == -1:
            return True
        
        # Try colors in order 1, 2, 3 for lexicographically smallest
        for c in range(1, 4):
            if is_safe(best_node, c):
                color[best_node] = c
                if solve_recursive(idx + 1):
                    return True
                color[best_node] = 0
        
        return False
    
    if solve_recursive(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

solve()
