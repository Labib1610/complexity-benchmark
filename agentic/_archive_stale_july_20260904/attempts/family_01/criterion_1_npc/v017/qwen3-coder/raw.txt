import sys
from collections import defaultdict

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build adjacency list for the graph
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 means unvisited, 1, 2, 3 are the colors
    color = [0] * (n + 1)
    
    # Check if the graph is 3-colorable using DFS
    def dfs(node, c):
        color[node] = c
        for nei in adj[node]:
            if color[nei] == 0:
                # Try to assign a color different from current node's color
                # We try colors 1, 2, 3 in order
                ok = False
                for next_c in range(1, 4):
                    if next_c != c:
                        if dfs(nei, next_c):
                            ok = True
                            break
                if not ok:
                    return False
            elif color[nei] == c:
                return False
        return True
    
    # Try to color each component
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start coloring with color 1
            if not dfs(i, 1):
                print("NO")
                return
    
    # If we reach here, it's 3-colorable
    print("YES")
    print(' '.join(map(str, color[1:])))

if __name__ == "__main__":
    main()
