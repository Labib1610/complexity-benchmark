```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list for the graph
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 means unvisited, 1 and 2 are the two colors
    color = [0] * (n + 1)
    
    # Check if bipartite using BFS
    for start in range(1, n + 1):
        if color[start] == 0:
            queue = deque()
            queue.append(start)
            color[start] = 1  # Start with color 1
            
            while queue:
                node = queue.popleft()
                current_color = color[node]
                next_color = 3 - current_color  # flips 1->2 and 2->1
                
                for neighbor in adj[node]:
                    if color[neighbor] == 0:
                        color[neighbor] = next_color
                        queue.append(neighbor)
                    elif color[neighbor] == current_color:
                        print("NO")
                        return
    
    # If we reach here, the graph is bipartite
    # The coloring gives us a valid assignment
    # We need the lexicographically smallest one
    # That means we prefer worker 1 to be in shift 1 if possible
    # But our coloring might have assigned worker 1 to shift 2
    # So we need to see if we should flip all colors in a component
    
    visited = [False] * (n + 1)
    result = color[:]  # copy of color
    
    for start in range(1, n + 1):
        if not visited[start]:
            # Find all nodes in this component
            component = []
            queue = deque()
            queue.append(start)
            visited[start] = True
            component.append(start)
            
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
                        component.append(neighbor)
            
            # In the component, check the first worker (smallest index)
            # If it's colored 2, we should flip all colors in this component
            min_node = min(component)
            if result[min_node] == 2:
                for node in component:
                    result[node] = 3 - result[node]
    
    print("YES")
    print(" ".join(str(result[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    main()
```
