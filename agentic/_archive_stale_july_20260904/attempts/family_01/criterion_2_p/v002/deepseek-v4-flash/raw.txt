We can model this as a bipartite graph coloring problem.  
If the graph is bipartite, we can assign zones 1 and 2 alternately.  
To get the lexicographically smallest assignment, we try to assign zone 1 to as many early vertices as possible, using BFS/DFS with a greedy coloring approach.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [0] * n  # 0 = uncolored, 1 or 2

for start in range(n):
    if color[start] == 0:
        color[start] = 1
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = 3 - color[u]
                    q.append(v)
                elif color[v] == color[u]:
                    print("NO")
                    sys.exit()

print("YES")
print(" ".join(str(color[i]) for i in range(n)))
```
