We can model this as a bipartite graph check. If the graph is bipartite, we can assign shifts 1 and 2 to the two partitions.  
To get the lexicographically smallest assignment, we should try to assign shift 1 to as many early workers as possible, which means we should start BFS/DFS from each unvisited node in order, assigning it shift 1 if possible.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 0
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = color[u] ^ 1
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            print("NO")
            sys.exit()

print("YES")
print(" ".join(str(c + 1) for c in color))
```
