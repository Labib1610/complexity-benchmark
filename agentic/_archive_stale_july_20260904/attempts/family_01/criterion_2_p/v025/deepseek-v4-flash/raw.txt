We can solve this by checking if the graph is bipartite.  
If it is, we assign colors (cores) using BFS/DFS, and to get the lexicographically smallest assignment, we try to assign core 1 whenever possible.

Here’s the Python solution:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
graph = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    graph[u].append(v)
    graph[v].append(u)

color = [-1] * n
possible = True

for i in range(n):
    if color[i] == -1:
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            for v in graph[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    print("YES")
    # To get lexicographically smallest, we can flip colors if color[0] == 1
    if color[0] == 1:
        color = [1 - c for c in color]
    print(' '.join(str(c + 1) for c in color))
```
