```python
import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        # No constraints, choose lex smallest: all 1
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build graph
    adj = defaultdict(list)
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
    
    # Color array: 0 = unvisited, 1 = color 1, 2 = color 2
    color = [0] * (n + 1)
    
    # Check if bipartite (2-colorable) component by component
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS to color component
        queue = deque()
        queue.append(start)
        color[start] = 1  # Start with color 1 for lex smallest
        while queue:
            u = queue.popleft()
            current_color = color[u]
            next_color = 3 - current_color  # flips 1->2, 2->1
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = next_color
                    queue.append(v)
                elif color[v] == current_color:
                    print("NO")
                    return
    
    # If we reach here, it's bipartite. Output YES and the coloring
    # But we need the lexicographically smallest assignment.
    # Our coloring already ensures this because we always started with color 1 for each component.
    print("YES")
    result = []
    for i in range(1, n + 1):
        if color[i] == 0:
            result.append("1")  # Isolated vertex, choose 1 for lex smallest
        else:
            result.append(str(color[i]))
    print(" ".join(result))

if __name__ == "__main__":
    main()
```
