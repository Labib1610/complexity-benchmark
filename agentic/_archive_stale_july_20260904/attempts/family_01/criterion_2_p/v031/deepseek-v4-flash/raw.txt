We can model this as a bipartite graph problem.  
If the graph is bipartite, we can assign frequencies 1 and 2 to the two partitions.  
To get the lexicographically smallest assignment, we try to assign frequency 1 to as many early nodes as possible.

Here’s the implementation:

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n
    possible = True

    # BFS to check bipartiteness and color
    from collections import deque
    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            q = deque([i])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        q.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break

    if not possible:
        print("NO")
        return

    # Now we have a valid bipartition.
    # To get lexicographically smallest, we try to assign 1 to as many early nodes as possible.
    # We can do a BFS/DFS that tries to assign 1 if possible without conflict.
    # Since the graph is bipartite, we can flip partitions if needed.
    # We'll do a greedy BFS: assign 1 to node 0, then propagate.

    assign = [0] * n
    visited = [False] * n
    for start in range(n):
        if visited[start]:
            continue
        # Try assigning 1 to start
        assign[start] = 1
        visited[start] = True
        q = deque([start])
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    assign[v] = 3 - assign[u]  # opposite
                    visited[v] = True
                    q.append(v)
                elif assign[v] == assign[u]:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            # Need to flip this component
            # Reset visited for this component and assign opposite
            comp = []
            stack = [start]
            visited[start] = False
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if visited[v]:
                        visited[v] = False
                        stack.append(v)
            # Now assign start = 2
            assign[start] = 2
            visited[start] = True
            q = deque([start])
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if not visited[v]:
                        assign[v] = 3 - assign[u]
                        visited[v] = True
                        q.append(v)

    print("YES")
    print(" ".join(str(assign[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
