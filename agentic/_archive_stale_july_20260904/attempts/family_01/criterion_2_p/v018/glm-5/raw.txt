import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # To find the lexicographically smallest valid configuration,
    # we greedily assign the smallest available value (1) to each node
    # if it doesn't conflict with already assigned neighbors.
    # Since we process nodes 0 to n-1, and neighbors with smaller indices
    # have already been fixed, this greedy approach yields the optimal result.
    
    res = [0] * n
    for i in range(n):
        used = 0
        for neighbor in adj[i]:
            if neighbor < i:
                if res[neighbor] == 1:
                    used |= 1
                elif res[neighbor] == 2:
                    used |= 2
        
        if (used & 1) == 0:
            res[i] = 1
        elif (used & 2) == 0:
            res[i] = 2
        else:
            # Conflict: both 1 and 2 are used by neighbors with smaller indices.
            # This implies a contradiction in the constraints relative to the
            # greedy assignment order, meaning no valid configuration exists.
            sys.stdout.write("NO\n")
            return

    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, res)) + "\n")

if __name__ == '__main__':
    solve()
