```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # color array: 0 = uncolored, 1 = cycle 1, 2 = cycle 2
    color = [0] * n
    
    for i in range(n):
        if color[i] == 0:
            # Start BFS for this component
            # To get lexicographically smallest configuration,
            # we start coloring the smallest-indexed node in each component with 1.
            queue = deque([i])
            color[i] = 1
            
            while queue:
                u = queue.popleft()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        # Conflict found
                        sys.stdout.write("NO\n")
                        return
    
    sys.stdout.write("YES\n")
    sys.stdout.write(" ".join(map(str, color)) + "\n")

if __name__ == "__main__":
    solve()
```
