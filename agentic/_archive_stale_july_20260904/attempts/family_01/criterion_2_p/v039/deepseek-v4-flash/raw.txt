Here’s the Python 3 solution for the problem:  

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    g = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        g[u].append(v)
        g[v].append(u)

    color = [-1] * n
    possible = True

    for i in range(n):
        if color[i] == -1:
            color[i] = 0
            stack = [i]
            while stack:
                u = stack.pop()
                for v in g[u]:
                    if color[v] == -1:
                        color[v] = color[u] ^ 1
                        stack.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break

    if not possible:
        print("NO")
        return

    # Now we have a valid 2-coloring, but we need lexicographically smallest.
    # We can try to flip components to get lexicographically smallest.
    # Since colors are 0/1, we can choose which component gets 0 or 1.
    # For lexicographically smallest, we want earlier nodes to be 1 if possible.
    # But we must keep bipartite property.

    comp_id = [-1] * n
    comps = []
    for i in range(n):
        if comp_id[i] == -1:
            cid = len(comps)
            comps.append([])
            stack = [i]
            comp_id[i] = cid
            while stack:
                u = stack.pop()
                comps[cid].append(u)
                for v in g[u]:
                    if comp_id[v] == -1:
                        comp_id[v] = cid
                        stack.append(v)

    # For each component, we can flip all colors (0->1, 1->0) independently.
    # To get lexicographically smallest, we process nodes in order and decide
    # whether to flip the component of that node if it hasn't been decided yet.
    decided = [False] * len(comps)
    for i in range(n):
        cid = comp_id[i]
        if not decided[cid]:
            # If current color is 1, keep; if 0, flip to make it 1
            if color[i] == 0:
                for u in comps[cid]:
                    color[u] ^= 1
            decided[cid] = True

    print("YES")
    print(' '.join(str(c + 1) for c in color))

if __name__ == "__main__":
    solve()
```
