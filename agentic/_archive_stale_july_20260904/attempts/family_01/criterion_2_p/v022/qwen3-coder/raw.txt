import sys
from collections import defaultdict, deque

def solve():

    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # If no constraints, any assignment works; lex smallest is all 1s
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return

    adj = defaultdict(list)
    edges = []
    
    index = 2

        u = int(data[index]) - 1  # Convert to 0-indexed
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
        index += 2

    color = [-1] * n  # -1 means unvisited, 0 and 1 are colors

    # In a bipartite graph, one side gets value 1, other gets 2

    for i in range(n):
        if not visited[i] and adj[i]:  # Only process nodes with edges
            # BFS to find component and check bipartiteness
            component_nodes = []
            queue = deque()
            queue.append(i)
            visited[i] = True
            component_nodes.append(i)
            
            # Try coloring starting with color 0 for node i
            color[i] = 0
            
            is_bipartite = True
            
            while queue and is_bipartite:
                node = queue.popleft()
                current_color = color[node]
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        component_nodes.append(neighbor)
                        color[neighbor] = 1 - current_color
                        queue.append(neighbor)
                    else:
                        # Check if consistent

                            is_bipartite = False
                            break
            
            if not is_bipartite:
                print("NO")
                return

    # To get lexicographically smallest, we prefer color 0 -> value 1

        #                if color[u] == 1, then xu=2, xv=1 (since color[v]=0)
        # But we need to respect the coloring we derived
        
        # Actually, let's re-color properly:
        # We can choose for each component which color maps to which value

        # we prefer to give it value 1, so color->value mapping:
        # color 0 -> value 1
        # color 1 -> value 2
        
        # But we haven't stored components explicitly. Let's do a proper traversal.
        
        pass # We'll handle this below

    assigned = [False] * n
    result = [0] * n

        if adj[i] and not assigned[i]:  # Node involved in constraints
            # BFS from i
            queue = deque([i])

            # Try to set node i to 1 if possible
            # But we must respect bipartite structure
            
            # Let's just do BFS again and assign values directly
            # Root gets 1, neighbors get 2, their neighbors get 1, etc.

            # So for component containing node i, if not yet assigned:
            if color[i] == -1:
                # Start BFS coloring
                queue = deque([i])
                color[i] = 0
                visited_component = set()

        queue = deque([start])
        color[start] = 0

            u = queue.popleft()
            c = color[u]

            # Isolated node, not in any constraint, assign 1 (lex smallest)
            result.append(1)
        elif color[i] == 0:
            result.append(1)

            result.append(2)
            
    print("YES")
