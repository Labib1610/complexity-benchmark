```python
import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        if u != v:
            adj[v].append(u)

    color = [0] * n
    possible = True

    for i in range(n):
        if color[i] != 0:
            continue
        color[i] = 1
        dq = deque([i])
        while dq and possible:
            u = dq.popleft()
            opp = 3 - color[u]
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = opp
                    dq.append(v)
                elif color[v] != opp:
                    possible = False
                    break

    out = sys.stdout
    if not possible:
        out.write("NO\n")
    else:
        out.write("YES\n")
        out.write(" ".join(map(str, color)))
        out.write("\n")

if __name__ == "__main__":
    solve()
```
