import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n      # 0/1 parity within component
    comp = [-1] * n       # component id
    comp_assigned = []    # None or [final_for_0, final_for_1]
    cid = 0

    for s in range(n):
        if color[s] != -1:
            continue
        dq = deque([s])
        color[s] = 0
        comp[s] = cid
        while dq:
            u = dq.popleft()
            cu = color[u]
            for w in adj[u]:
                if color[w] == -1:
                    color[w] = cu ^ 1
                    comp[w] = cid
                    dq.append(w)
                elif color[w] == cu:
                    sys.stdout.write("NO")
                    return
        comp_assigned.append(None)
        cid += 1

    out = [0] * n
    for i in range(n):
        c = comp[i]
        mapping = comp_assigned[c]
        if mapping is None:
            p = color[i]
            if p == 0:
                mapping = [1, 2]
            else:
                mapping = [2, 1]
            comp_assigned[c] = mapping
        out[i] = mapping[color[i]]

    sys.stdout.write("YES\n" + " ".join(map(str, out)))

solve()
