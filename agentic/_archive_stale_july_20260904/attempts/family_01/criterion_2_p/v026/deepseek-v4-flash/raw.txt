We need to check if the graph is bipartite and, if so, assign colors (1 or 2) with the lexicographically smallest assignment.  
The standard approach is BFS/DFS with two colors, but to get the lexicographically smallest assignment, we should try to assign 1 whenever possible, which means we process nodes in order and if a node is uncolored, we start BFS from it with color 1.

Here’s the implementation:

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 1
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = 3 - color[u]  # 1 -> 2, 2 -> 1
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            print("NO")
            sys.exit()

print("YES")
print(" ".join(str(c) for c in color))
```
