 ```python
import sys

def solve() -> None:
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)  # 0 unvisited, 1 or 2
    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        stack = [start]
        color[start] = 1
        while stack:
            v = stack.pop()
            cv = color[v]
            for nb in adj[v]:
                if color[nb] == 0:
                    color[nb] = 3 - cv
                    stack.append(nb)
                elif color[nb] == cv:
                    sys.stdout.write("NO")
                    return

    sys.stdout.write("YES\n")
    sys.stdout.write(' '.join(map(str, color[1:])))

solve()
```
