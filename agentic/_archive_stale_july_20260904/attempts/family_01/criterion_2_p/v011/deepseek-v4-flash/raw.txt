Here’s the Python 3 solution for the problem:  

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
possible = True

for i in range(n):
    if color[i] == -1:
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    # We have a valid bipartition, but we need lexicographically smallest.
    # Try to assign 1 to as many early nodes as possible.
    # We can do a BFS/DFS with greedy assignment: assign 1 if possible,
    # otherwise 2, and propagate constraints.
    assignment = [0] * n
    visited = [False] * n

    for start in range(n):
        if visited[start]:
            continue
        assignment[start] = 1
        visited[start] = True
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    assignment[v] = 3 - assignment[u]  # opposite channel
                    q.append(v)
                else:
                    if assignment[v] == assignment[u]:
                        # conflict, but we already know graph is bipartite,
                        # so this shouldn't happen if we follow the same component.
                        pass

    # The above greedy may not produce lexicographically smallest globally,
    # so we need to try flipping components if it helps lexicographically.
    # Standard approach: for each component, try both starting colors and pick smaller lexicographically.
    assignment = [0] * n
    visited = [False] * n

    for start in range(n):
        if visited[start]:
            continue
        # Try starting with 1
        temp1 = [0] * n
        temp1[start] = 1
        visited1 = [False] * n
        visited1[start] = True
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited1[v]:
                    visited1[v] = True
                    temp1[v] = 3 - temp1[u]
                    q.append(v)
        # Try starting with 2
        temp2 = [0] * n
        temp2[start] = 2
        visited2 = [False] * n
        visited2[start] = True
        q = deque([start])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited2[v]:
                    visited2[v] = True
                    temp2[v] = 3 - temp2[u]
                    q.append(v)

        # Compare lexicographically for this component
        # We need to merge with already fixed assignments
        # Simpler: collect component nodes
        comp_nodes = []
        stack = [start]
        seen = [False] * n
        seen[start] = True
        while stack:
            u = stack.pop()
            comp_nodes.append(u)
            for v in adj[u]:
                if not seen[v]:
                    seen[v] = True
                    stack.append(v)

        # Choose better option
        # Build candidate assignments for this component
        cand1 = [0] * n
        cand2 = [0] * n
        for u in comp_nodes:
            cand1[u] = temp1[u]
            cand2[u] = temp2[u]

        # Compare with current assignment (which is 0 for unassigned)
        # We'll pick the one that gives smaller overall lexicographic order
        # Since we process components in order of smallest index, we can just
        # pick the smaller candidate for this component.
        # But we must ensure consistency with already assigned nodes (none yet for this comp)
        # So just pick lexicographically smaller candidate for this component.
        # Compare cand1 vs cand2 on comp_nodes sorted
        comp_nodes_sorted = sorted(comp_nodes)
        better = None
        for u in comp_nodes_sorted:
            if cand1[u] != cand2[u]:
                better = 1 if cand1[u] < cand2[u] else 2
                break
        if better is None:
            better = 1
        for u in comp_nodes:
            assignment[u] = cand1[u] if better == 1 else cand2[u]
            visited[u] = True

    print("YES")
    print(" ".join(map(str, assignment)))
```
