import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    idx = 0
    n = int(data[idx]); idx += 1
    m = int(data[idx]); idx += 1

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(data[idx]); idx += 1
        v = int(data[idx]); idx += 1
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)

    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        # BFS and try to assign color 1 (shift 1) to the lexicographically smallest vertex
        color[start] = 1
        queue = deque([start])
        while queue:
            u = queue.popleft()
            cu = color[u]
            cv = 3 - cu
            for v in adj[u]:
                if color[v] == 0:
                    color[v] = cv
                    queue.append(v)
                elif color[v] != cv:
                    print("NO")
                    return

    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
