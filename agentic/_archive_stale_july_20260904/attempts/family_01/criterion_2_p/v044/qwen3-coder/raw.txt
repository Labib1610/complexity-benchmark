import sys
from collections import defaultdict, deque

def solve():
    import sys
    sys.setrecursionlimit(1 << 25)
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for the conflict graph
    adj = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Color array: -1 unvisited, 0 color, 1 color
    color = [-1] * n
    result = [0] * n
    
    # Check bipartiteness using BFS for each component
    for start in range(n):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0  # Assign color 0 to the starting node
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if color[neighbor] == -1:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
            
            # Assign the lexicographically smallest color to the component
            # Since we want technician 1 (index 0) to be in shift 1 when possible,
            # and we started with color 0, we need to ensure consistency.
            # Actually, let's reframe: we want the first node in each component 
            # to be colored 0 (shift 1). But since we're doing BFS from the 
            # first unvisited node, and we start with 0, this should work.
            pass
    
    # Construct the result based on colors
    # color 0 -> shift 1, color 1 -> shift 2
    for i in range(n):
        if color[i] == 0:
            result[i] = 1
        else:
            result[i] = 2
    
    print("YES")
    print(' '.join(map(str, result)))

solve()
