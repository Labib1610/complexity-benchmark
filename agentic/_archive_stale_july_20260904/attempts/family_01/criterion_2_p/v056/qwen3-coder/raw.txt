import sys
from collections import defaultdict, deque

def main():
    import sys
    sys.setrecursionlimit(1 << 25)
    
    n, m = map(int, sys.stdin.readline().split())
    
    # Build graph: for each edge (u,v), we add constraints:
    # u and v must have different colors (polarities)
    # This is a 2-SAT / bipartition problem
    
    adj = defaultdict(list)
    adj_rev = defaultdict(list)
    
    # For each monitor (u,v), we need u != v
    # Which means if u is 1, then v is 2, and vice versa
    # In implication graph:
    # u1 -> v2, v1 -> u2, u2 -> v1, v2 -> u1
    # But we can model this more simply as: u and v must be in different groups
    # So it's a graph coloring problem with 2 colors
    
    edges = []
    graph = [[] for _ in range(n+1)]
    
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u, v))
        graph[u].append(v)
        graph[v].append(u)
    
    # Color the graph with 2 colors (BFS/DFS for each component)
    color = [-1] * (n+1)  # -1 unvisited, 0 and 1 are the two colors
    
    for start in range(1, n+1):
        if color[start] == -1:
            queue = deque()
            queue.append(start)
            color[start] = 0
            
            while queue:
                u = queue.popleft()
                c = color[u]
                
                for v in graph[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        queue.append(v)
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If we reach here, a valid coloring exists
    # We want the lexicographically smallest one
    # That means we prefer color[1] = 0, color[2] = 0, etc.
    # Our algorithm already ensures that by always assigning 0 first
    
    print("YES")
    result = []
    for i in range(1, n+1):
        result.append(str(color[i] + 1))  # Convert 0,1 to 1,2
        
    print(' '.join(result))

if __name__ == "__main__":
    main()
