We can model this as a bipartite graph problem: each particle is a node, and each link is an edge requiring opposite colors (1 and 2).  
If the graph is not bipartite, it’s impossible.  
For lexicographically smallest assignment, we assign 1 to the first unvisited node in each component, then propagate alternating values.

```python
import sys
sys.setrecursionlimit(300000)

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [-1] * n
    
    for i in range(n):
        if color[i] == -1:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(" ".join(str(color[i]) for i in range(n)))

if __name__ == "__main__":
    solve()
```
