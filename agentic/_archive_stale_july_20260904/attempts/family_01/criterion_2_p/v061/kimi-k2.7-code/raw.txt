import sys
from collections import deque

def solve():
    data = sys.stdin.buffer.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    comp = [0] * (n + 1)
    comp_mins = []
    cid = 0

    for i in range(1, n + 1):
        if color[i] != 0:
            continue
        q = deque([i])
        color[i] = 1
        comp[i] = cid
        min_v = i
        while q:
            u = q.popleft()
            other = 3 - color[u]
            for w in adj[u]:
                if color[w] == 0:
                    color[w] = other
                    comp[w] = cid
                    if w < min_v:
                        min_v = w
                    q.append(w)
                elif color[w] == color[u]:
                    sys.stdout.write("NO")
                    return
        comp_mins.append(min_v)
        cid += 1

    flip = [False] * cid
    for c in range(cid):
        if color[comp_mins[c]] == 2:
            flip[c] = True

    out = ["YES"]
    line = []
    for i in range(1, n + 1):
        c = color[i]
        if flip[comp[i]]:
            c = 3 - c
        line.append(str(c))
    out.append(' '.join(line))
    sys.stdout.write('\n'.join(out))

if __name__ == "__main__":
    solve()
