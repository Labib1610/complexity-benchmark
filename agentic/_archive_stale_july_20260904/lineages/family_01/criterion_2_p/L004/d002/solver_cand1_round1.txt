```python
import sys

# The problem states that for any two servers connected by a link, 
# the product of their bands must be 2.
# Since bands are {1, 2}, the only way for the product to be 2 is 
# if one server is 1 and the other is 2.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for lower indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is uncolored, we assign it 1 (the smaller value).
    # Then we perform BFS/DFS to color the connected component.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
