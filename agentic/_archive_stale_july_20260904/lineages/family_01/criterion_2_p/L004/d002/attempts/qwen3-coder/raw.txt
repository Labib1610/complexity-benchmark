```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        # No constraints, all 1s is valid and lex smallest
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color nodes with BFS to check bipartiteness
    color = {}
    assignment = [0] * (n + 1)  # 1-indexed
    
    # Process each connected component
    for start in range(1, n + 1):
        if start not in color:
            # BFS coloring
            queue = deque([start])
            color[start] = 0
            
            while queue:
                node = queue.popleft()
                for neighbor in adj[node]:
                    if neighbor not in color:
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        print("NO")
                        return
    
    # If bipartite, construct assignment
    # Assign 1 to color 0, 2 to color 1 for lex smallest
    for i in range(1, n + 1):
        if i in color and color[i] == 0:
            assignment[i] = 1
        elif i in color and color[i] == 1:
            assignment[i] = 2
        else:
            # Isolated node, assign 1 (lex smallest)
            assignment[i] = 1
    
    # Verify the solution
    for u, v in edges:
        if assignment[u] * assignment[v] != 2:
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, assignment[1:])))

solve()
```
