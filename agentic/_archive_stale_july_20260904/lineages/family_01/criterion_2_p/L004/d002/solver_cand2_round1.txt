```python
import sys

# The problem asks to assign 1 or 2 to each node such that for every edge (u, v),
# channel(u) + channel(v) = 3.
# This means channel(u) != channel(v).
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment.
# For each connected component, we have two choices:
# 1. Assign 1 to the smallest index node in the component.
# 2. Assign 2 to the smallest index node in the component.
# To get the lexicographically smallest result, we want the smallest index
# to have 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            assignment[i] = 1
            component.append((i, 1))
            
            idx_stack = 0
            while idx_stack < len(component):
                u, color = component[idx_stack]
                idx_stack += 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        component.append((v, 3 - color))
                    elif assignment[v] == color:
                        print("NO")
                        return
            
            # Check if we should flip this component to make it lexicographically smaller
            # We want the smallest index in this component to be 1.
            # Find the smallest index in this component.
            min_node = min(node for node, color in component)
            if assignment[min_node] == 2:
                for node, color in component:
                    assignment[node] = 3 - assignment[node]
                    
    print("YES")
    print(*(assignment[1:]))

solve()
```
