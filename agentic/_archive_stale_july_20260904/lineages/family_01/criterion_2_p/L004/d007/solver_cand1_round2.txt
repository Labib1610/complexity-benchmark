```python
import sys

# The resonance formula for a workflow (u, v) is:
# R(z_u, z_v) = (z_u^3 + z_v^3) / (z_u * z_v * (z_u + z_v)) + 1
# If z_u = 1, z_v = 1: R = (1+1) / (1*1*2) + 1 = 1 + 1 = 2
# If z_u = 2, z_v = 2: R = (8+8) / (4*4) + 1 = 1 + 1 = 2
# If z_u = 1, z_v = 2: R = (1+8) / (2*3) + 1 = 9/6 + 1 = 1.5 + 1 = 2.5
#
# Total resonance = sum of R_i = 2.5 * m
# Let x be the number of edges with (1, 2) or (2, 1).
# Let y be the number of edges with (1, 1) or (2, 2).
# x + y = m
# 2.5 * x + 2 * y = 2.5 * m
# 2.5 * x + 2 * (m - x) = 2.5 * m
# 0.5 * x = 0.5 * m => x = m
#
# This means every edge must connect a 1 and a 2.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want 1s at the start.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            # To make it lexicographically smallest, we prefer color 1 for smaller indices
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
