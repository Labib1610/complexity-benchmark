```python
import sys

# The resource metric for a job (u, v) is 5 % (t_u + t_v).
# Possible values for t_u + t_v:
# If t_u=1, t_v=1: sum=2, 5 % 2 = 1
# If t_u=1, t_v=2: sum=3, 5 % 3 = 2
# If t_u=2, t_v=2: sum=4, 5 % 4 = 1
#
# Let x be the number of edges with (1, 1), y be the number of edges with (1, 2),
# and z be the number of edges with (2, 2).
# Total metric = 1*x + 2*y + 1*z = 2*m.
# Since x + y + z = m, we have x + y + z + y = 2*m => y = m - (x + z).
# Substituting y: x + 2(m - x - z) + z = 2m => x + 2m - 2x - 2z + z = 2m => -x - z = 0.
# This implies x = 0 and z = 0.
# Thus, y must be m.
# This means every edge must connect a server with tier 1 and a server with tier 2.
# This is exactly the condition for a graph to be bipartite.
# We need to check if the graph is bipartite. If it is, we assign 1s and 2s
# based on the bipartition. To get the lexicographically smallest sequence,
# for each connected component, we can choose which partition gets 1 and which gets 2.
# We want t_i = 1 as much as possible for smaller i.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Count how many 1s we get if we assign color 1 to tier 1
            # vs color 2 to tier 1.
            c1_count = sum(1 for u, c in component if c == 1)
            c2_count = len(component) - c1_count
            
            # To make lexicographically smallest, we want the smallest index
            # in the component to be 1 if possible.
            first_node = min(u for u, c in component)
            first_node_color = next(c for u, c in component if u == first_node)
            
            # If first_node_color is 1, we can set color 1 -> tier 1, color 2 -> tier 2
            # If first_node_color is 2, we can set color 2 -> tier 1, color 1 -> tier 2
            if first_node_color == 1:
                for u, c in component:
                    res[u] = c
            else:
                for u, c in component:
                    res[u] = 3 - c
                    
    print("YES")
    print(*(res[1:]))

solve()
```
