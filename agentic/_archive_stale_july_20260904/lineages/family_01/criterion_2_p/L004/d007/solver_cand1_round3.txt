```python
import sys

# The expression (z_u/z_v + z_v/z_u) for z_i in {1, 2}
# If z_u == z_v, the value is (1/1 + 1/1) = 2 or (2/2 + 2/2) = 2.
# If z_u != z_v, the value is (1/2 + 2/1) = 0.5 + 2 = 2.5.
# Let k be the number of pairs where z_u != z_v.
# The sum is (m - k) * 2 + k * 2.5 = 2m - 2k + 2.5k = 2m + 0.5k.
# We want 2m + 0.5k = 2.5m, which simplifies to 0.5k = 0.5m, so k = m.
# This means for every pair (u, v), we must have z_u != z_v.
# This is equivalent to checking if the graph is bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    # To get lexicographically smallest, we prefer 1 over 2.
    # We iterate 1 to n. If not colored, start BFS/DFS.
    # For lexicographical smallest, we want z_i = 1 if possible.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

solve()
```
