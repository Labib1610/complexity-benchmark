```python
import sys

# The resonance of a workflow (u, v) is z_u/z_v + z_v/z_u.
# If z_u = z_v, resonance = 1/1 + 1/1 = 2 or 2/2 + 2/2 = 2.
# If z_u != z_v, resonance = 1/2 + 2/1 = 2.5.
# Let k be the number of workflows where z_u != z_v.
# Let m - k be the number of workflows where z_u = z_v.
# Total resonance = k * 2.5 + (m - k) * 2 = 2.5 * m.
# 2.5k + 2m - 2k = 2.5m
# 0.5k = 0.5m => k = m.
# This means every workflow must connect two microservices with different scaling factors.
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we can color the graph with 2 colors (1 and 2).
# To get the lexicographically smallest sequence, we greedily assign 1 to the first
# node of each connected component, then propagate the coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
