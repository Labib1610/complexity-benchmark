```python
import sys

# The problem requires the trace of the product of two matrices to be 6.
# M1 = [[1, 2], [0, 1]], M2 = [[1, 0], [2, 1]]
# M1 * M1 = [[1, 4], [0, 1]], trace = 2
# M2 * M2 = [[1, 0], [4, 1]], trace = 2
# M1 * M2 = [[5, 2], [2, 1]], trace = 6
# M2 * M1 = [[1, 2], [2, 5]], trace = 6
# Thus, for any connected pair (u, v), they must have different bands.
# This is a bipartite graph coloring problem.
# We need to check if the graph is bipartite. If so, assign 1 to one set and 2 to the other.
# To get the lexicographically smallest assignment, we prioritize assigning 1 to smaller indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To ensure lexicographical smallest, we try assigning 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
