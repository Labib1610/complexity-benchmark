```python
import sys

# The condition is: integral from 0 to 3 of (x - z_u)(x - z_v) dx = 1.5
# Let f(z_u, z_v) = integral_0^3 (x^2 - x(z_u + z_v) + z_u * z_v) dx
# = [x^3/3 - x^2/2 * (z_u + z_v) + x * z_u * z_v] from 0 to 3
# = (27/3 - 9/2 * (z_u + z_v) + 3 * z_u * z_v)
# = 9 - 4.5 * (z_u + z_v) + 3 * z_u * z_v
# We want this to equal 1.5:
# 9 - 4.5 * (z_u + z_v) + 3 * z_u * z_v = 1.5
# 7.5 - 4.5 * (z_u + z_v) + 3 * z_u * z_v = 0
# Multiply by 2:
# 15 - 9 * (z_u + z_v) + 6 * z_u * z_v = 0
# Divide by 3:
# 5 - 3 * (z_u + z_v) + 2 * z_u * z_v = 0
#
# Case 1: z_u = 1, z_v = 1: 5 - 3(2) + 2(1) = 5 - 6 + 2 = 1 != 0
# Case 2: z_u = 2, z_v = 2: 5 - 3(4) + 2(4) = 5 - 12 + 8 = 1 != 0
# Case 3: z_u = 1, z_v = 2: 5 - 3(3) + 2(2) = 5 - 9 + 4 = 0 (Valid!)
#
# So the condition is equivalent to z_u != z_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
