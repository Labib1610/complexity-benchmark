import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We need to assign 1 or 2. 
    # Different parity means if u is 1, v must be 2, and vice versa.
    # This is a standard bipartite graph coloring problem.
    # To get the lexicographically smallest assignment, we want 1s as much as possible.
    # For each connected component, we can pick the color of the first node.
    # If we pick 1 for the first node, the rest are determined.
    # If we pick 2 for the first node, the rest are also determined.
    # Since we want lexicographically smallest, we should try to assign 1 to the 
    # smallest index node in each component.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # Assign 1 to the first node i
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

solve()
