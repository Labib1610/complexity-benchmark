```python
import sys

# The harmonic factor for a pair (u, v) is (z_u * z_v + 1).
# z_i is in {1, 2}.
# If z_u = 1, z_v = 1: factor = 1*1 + 1 = 2.
# If z_u = 1, z_v = 2: factor = 1*2 + 1 = 3.
# If z_u = 2, z_v = 1: factor = 2*1 + 1 = 3.
# If z_u = 2, z_v = 2: factor = 2*2 + 1 = 5.
# The product of m factors must be 3^m.
# Since each factor is 2, 3, or 5, the only way the product is 3^m is if
# every single factor is 3.
# A factor is 3 if and only if (z_u, z_v) is (1, 2) or (2, 1).
# This means z_u != z_v for every edge (u, v).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want z_i = 1 whenever possible.
# For each connected component, we can pick the color of the first node (smallest index)
# to be 1, then propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want z_i = 1 for the smallest index
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
