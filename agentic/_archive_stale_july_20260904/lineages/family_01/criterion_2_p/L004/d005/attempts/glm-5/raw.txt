import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # We want lexicographically smallest assignment.
    # Assignments are 1 or 2. Lexicographically smallest means preferring 1s.
    # We use BFS to propagate constraints.
    # State: -1 (unvisited), 0 (assigned 1), 1 (assigned 2).
    state = [-1] * n
    possible = True
    
    for i in range(n):
        if state[i] != -1:
            continue
        
        # Start BFS from node i, assigning it 1 (lexicographically smallest choice)
        queue = deque([i])
        state[i] = 0
        
        while queue:
            u = queue.popleft()
            # If u is assigned 1, neighbors must be 2.
            # If u is assigned 2, neighbors must be 1.
            required_neighbor_state = 1 - state[u]
            
            for v in adj[u]:
                if state[v] == -1:
                    state[v] = required_neighbor_state
                    queue.append(v)
                elif state[v] != required_neighbor_state:
                    possible = False
                    break
            if not possible:
                break
        if not possible:
            break
    
    if not possible:
        print("NO")
    else:
        print("YES")
        # Convert state 0 -> 1, state 1 -> 2
        result = [str(s + 1) for s in state]
        print(" ".join(result))

if __name__ == '__main__':
    solve()
