import sys

# The problem asks to assign each microservice a replication factor of 1 or 2.
# A workflow between u and v has capacity = factor(u) * factor(v).
# We need factor(u) * factor(v) = 2.
# This means one must be 1 and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph where edges represent
# the constraint that the two nodes must have different values.
# If the graph is bipartite, we can color it with 1 and 2.
# To get the lexicographically smallest sequence, we want 1s as early as possible.
# For each connected component, we have two choices for coloring (if bipartite).
# Let the two sets of the bipartition be A and B.
# We can assign 1 to A and 2 to B, or 2 to A and 1 to B.
# We choose the one that makes the sequence lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings:
            # 1: color[v] = color[v]
            # 2: color[v] = 3 - color[v]
            # We want the one that is lexicographically smaller.
            # Find the first node in the component.
            first_node = min(node for node, c in component)
            
            # Option 1: color[first_node] remains as is
            # Option 2: color[first_node] becomes 3 - color[first_node]
            
            # We want the smallest value at the smallest index.
            # Let's check which assignment gives a smaller value at the first node.
            # If color[first_node] is 1, we prefer it. If it's 2, we prefer 3-2=1.
            
            c1 = color[first_node]
            # If we keep current colors, first_node gets c1.
            # If we flip, first_node gets 3-c1.
            # We want the assignment where first_node gets 1.
            
            flip = (c1 == 2)
            for node, c in component:
                ans[node] = (3 - c) if flip else c
                
    print("YES")
    print(*(ans[1:]))

solve()
