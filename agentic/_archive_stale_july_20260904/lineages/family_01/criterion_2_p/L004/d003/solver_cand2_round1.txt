```python
import sys

# The condition is that for any two servers connected by a link,
# the sum of the cubes of their bands must be 9.
# Let the bands be x and y, where x, y in {1, 2}.
# x^3 + y^3 = 9
# If x=1, 1^3 + y^3 = 9 => y^3 = 8 => y = 2.
# If x=2, 2^3 + y^3 = 9 => 8 + y^3 = 9 => y^3 = 1 => y = 1.
# This means for every edge (u, v), band(u) must be different from band(v).
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we want to assign 1
# to as many nodes as possible, prioritizing smaller indices.
# We can use BFS/DFS to color the graph. For each connected component,
# we try assigning 1 to the smallest index node and propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
