import sys

# The problem requires that for every sensor (u, v), pressure(u) + pressure(v) = 3.
# Since pressure is either 1 or 2, this means pressure(u) != pressure(v).
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1s before 2s).
# For each connected component, we have two possible colorings (swap 1 and 2).
# To get the lexicographically smallest, for each component, we check which
# coloring results in a smaller value at the first node encountered in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u, c = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings:
            # 1: color[v] = color[v]
            # 2: color[v] = 3 - color[v]
            # We want the one that makes the smallest index in the component 1.
            
            # Find the smallest index in this component
            min_node = min(node for node, c in component)
            
            # Determine which coloring makes color[min_node] == 1
            # If color[min_node] is already 1, use current. Else flip.
            flip = (color[min_node] == 2)
            
            for node, c in component:
                if flip:
                    ans[node] = 3 - c
                else:
                    ans[node] = c
                    
    print("YES")
    print(*(ans[1:]))

solve()
