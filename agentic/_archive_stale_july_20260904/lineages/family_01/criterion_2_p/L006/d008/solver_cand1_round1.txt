```python
import sys

# The basis is {1, x, x^2}.
# D(1) = 0, D(x) = 1, D(x^2) = 2x. Matrix D:
# [[0, 1, 0],
#  [0, 0, 2],
#  [0, 0, 0]]
# T1 = I + 2D:
# [[1, 2, 0],
#  [0, 1, 4],
#  [0, 0, 1]]
# J(1) = x, J(x) = 0.5x^2, J(x^2) = 0. Matrix J:
# [[0, 0, 0],
#  [1, 0, 0],
#  [0, 0.5, 0]]
# T2 = I + 2J:
# [[1, 0, 0],
#  [2, 1, 0],
#  [0, 1, 1]]
#
# Tr(T1*T1) = Tr([[1, 2, 8], [0, 1, 4], [0, 0, 1]]) = 3
# Tr(T2*T2) = Tr([[1, 0, 0], [4, 1, 0], [1, 1, 1]]) = 3
# Tr(T1*T2) = Tr([[1, 2, 0], [0, 1, 4], [0, 0, 1]] * [[1, 0, 0], [2, 1, 0], [0, 1, 1]])
#           = Tr([[5, 2, 0], [2, 5, 4], [0, 1, 1]]) = 5 + 5 + 1 = 11
# Tr(T2*T1) = 11
#
# So, for Tr(Ti*Tj) = 11, we must have i != j.
# This is a 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
