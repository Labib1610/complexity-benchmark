```python
import sys

# The compatibility score between two grids A and B is sum(A[r][c] * B[c][r]).
# Let C1 be:
# 1 2 0
# 0 1 2
# 0 0 1
# Let C2 be:
# 1 0 0
# 2 1 0
# 0 2 1
# Note that C2 is the transpose of C1.
# Let S(A, B) = sum_{r,c} A[r][c] * B[c][r].
# S(C1, C1) = (1*1 + 2*0 + 0*0) + (0*2 + 1*1 + 2*0) + (0*0 + 0*2 + 1*1) = 1 + 1 + 1 = 3.
# S(C2, C2) = (1*1 + 0*2 + 0*0) + (2*0 + 1*1 + 0*2) + (0*0 + 2*0 + 1*1) = 1 + 1 + 1 = 3.
# S(C1, C2) = sum_{r,c} C1[r][c] * C2[c][r].
# Since C2 = C1^T, S(C1, C2) = sum_{r,c} C1[r][c] * C1[r][c] = sum_{r,c} (C1[r][c])^2.
# C1^2 entries: 1^2 + 2^2 + 0^2 + 0^2 + 1^2 + 2^2 + 0^2 + 0^2 + 1^2 = 1 + 4 + 0 + 0 + 1 + 4 + 0 + 0 + 1 = 11.
# So S(C1, C2) = 11.
# The condition is that for every channel (u, v), the modes must be different (one 1, one 2).
# This is a bipartite coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
