```python
import sys

# The stability equation is 2p_u^2 - 5p_u p_v + 2p_v^2 = 0.
# Let x = p_u, y = p_v. 2x^2 - 5xy + 2y^2 = 0.
# This factors as (2x - y)(x - 2y) = 0.
# So, either y = 2x or x = 2y.
# Since p_u, p_v are in {1, 2}:
# If p_u = 1, then p_v must be 2 (since 2*1 - p_v = 0 => p_v=2 or 1 - 2*p_v = 0 => p_v=0.5).
# If p_u = 2, then p_v must be 1 (since 2*2 - p_v = 0 => p_v=4 or 2 - 2*p_v = 0 => p_v=1).
# In all cases, p_u != p_v.
# This is a bipartite graph coloring problem where each edge requires p_u != p_v.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try color 1 first
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
