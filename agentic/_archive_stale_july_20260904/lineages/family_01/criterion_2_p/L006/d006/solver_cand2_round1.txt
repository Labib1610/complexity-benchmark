import sys

# The transformations are:
# f1(x) = (13x + 7) mod 31
# f2(x) = (12x + 9) mod 31
#
# We need f_a(f_b(x)) = x for all x.
# Let f_a(x) = a1*x + a0, f_b(x) = b1*x + b0
# f_a(f_b(x)) = a1*(b1*x + b0) + a0 = (a1*b1)x + (a1*b0 + a0)
# For this to be x, we need:
# a1*b1 = 1 (mod 31)
# a1*b0 + a0 = 0 (mod 31)
#
# Mode 1: a1=13, a0=7
# Mode 2: a1=12, a0=9
#
# Check pairs:
# (1, 1): 13*13 = 169 = 14 mod 31 != 1. (Invalid)
# (2, 2): 12*12 = 144 = 20 mod 31 != 1. (Invalid)
# (1, 2): 13*12 = 156 = 1 mod 31. (Valid)
#         13*9 + 7 = 117 + 7 = 124 = 4*31 = 0 mod 31. (Valid)
# (2, 1): 12*13 = 156 = 1 mod 31. (Valid)
#         12*7 + 9 = 84 + 9 = 93 = 3*31 = 0 mod 31. (Valid)
#
# So, for any channel (u, v), mode(u) must be different from mode(v).
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence, so we try to assign 1 to
# the first node of each connected component, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS, assign 1 to the smallest index node in component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, color[v]))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
