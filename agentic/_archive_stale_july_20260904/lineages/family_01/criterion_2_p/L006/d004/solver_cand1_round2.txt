```python
import sys

# The condition is: spin(u) XOR spin(v) = 3.
# Possible values for spin are {1, 2}.
# 1 XOR 1 = 0
# 1 XOR 2 = 3
# 2 XOR 1 = 3
# 2 XOR 2 = 0
# Thus, for every link (u, v), spin(u) must be different from spin(v).
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is unvisited, we assign it 1 (the smaller value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer color 1 for the root
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    sys.setrecursionlimit(300000)
    solve()
```
