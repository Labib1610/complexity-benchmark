```python
import sys

# The matrices are:
# M1 = [[1, 2, 0], [0, 1, 2], [0, 0, 1]]
# M2 = [[1, 0, 0], [2, 1, 0], [0, 2, 1]]
#
# Tr(M1 * M1) = Tr([[1, 4, 4], [0, 1, 4], [0, 0, 1]]) = 1 + 1 + 1 = 3
# Tr(M2 * M2) = Tr([[1, 0, 0], [4, 1, 0], [4, 4, 1]]) = 1 + 1 + 1 = 3
# Tr(M1 * M2) = Tr([[5, 2, 0], [4, 5, 2], [0, 2, 1]]) = 5 + 5 + 1 = 11
# Tr(M2 * M1) = Tr([[1, 2, 0], [2, 5, 2], [0, 4, 5]]) = 1 + 5 + 5 = 11
#
# Condition: Tr(A * B) = 11.
# This means if A=B, Tr = 3 != 11. If A!=B, Tr = 11.
# So, for every edge (u, v), mode(u) must be different from mode(v).
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer mode 1 (color 1)
            # for the first node in each connected component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
