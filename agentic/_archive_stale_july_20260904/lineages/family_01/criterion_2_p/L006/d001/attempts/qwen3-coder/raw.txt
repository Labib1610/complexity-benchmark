```python
import sys
from collections import defaultdict, deque

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        # No constraints, assign all to 1 for lex smallest
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1  # Convert to 0-indexed
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Color array: -1 unvisited, 0 means color 0, 1 means color 1
    # We'll map 0 -> 1, 1 -> 2 for the final answer
    color = [-1] * n
    visited = [False] * n
    
    # Check if bipartite (2-colorable) using BFS
    for i in range(n):
        if not visited[i]:
            queue = deque([i])
            color[i] = 0
            visited[i] = True
            
            while queue:
                node = queue.popleft()
                
                for neighbor in adj[node]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        color[neighbor] = 1 - color[node]
                        queue.append(neighbor)
                    elif color[neighbor] == color[node]:
                        # Not bipartite - impossible to satisfy constraints
                        print("NO")
                        return
    
    # Verify all edges have sum 3
    for u, v in edges:
        if (color[u] + 1) + (color[v] + 1) != 3:
            print("NO")
            return
    
    # Convert colors to pressures (0->1, 1->2)
    pressures = [str(c + 1) for c in color]
    
    print("YES")
    print(" ".join(pressures))

solve()
```
