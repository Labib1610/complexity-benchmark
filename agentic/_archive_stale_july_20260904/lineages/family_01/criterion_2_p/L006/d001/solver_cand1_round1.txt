import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # shift[i] = 0 (unassigned), 1, or 2
    shift = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we want shift[i] = 1
    # whenever possible. We iterate from 1 to n. If unassigned, try assigning 1.
    # If that leads to a conflict, then it must be 2.
    
    for i in range(1, n + 1):
        if shift[i] == 0:
            # Start a BFS/DFS component
            component = []
            stack = [i]
            shift[i] = 1
            component.append(i)
            
            # Use a stack to process the component
            # We need to check if this component is bipartite
            # and if assigning 1 to the smallest ID is valid.
            # Actually, for each connected component, there are only 2 ways to color it.
            # We pick the one that makes the smallest ID in the component 1.
            
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if shift[v] == 0:
                        shift[v] = 3 - shift[u]
                        q.append(v)
                    elif shift[v] == shift[u]:
                        print("NO")
                        return
            
            # After coloring, check if the smallest ID in this component is 1.
            # If not, flip the colors (1->2, 2->1).
            min_id = min(q)
            if shift[min_id] == 2:
                for node in q:
                    shift[node] = 3 - shift[node]
                    
    print("YES")
    print(*(shift[1:]))

solve()
