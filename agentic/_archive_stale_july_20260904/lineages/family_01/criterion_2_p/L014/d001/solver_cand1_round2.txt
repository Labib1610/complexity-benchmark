```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# |x_u - x_v| = 1. This is exactly the definition of a bipartite graph.
# We need the lexicographically smallest assignment (x_1, x_2, ..., x_n).
# To achieve this, we iterate from 1 to n. If a node is not yet colored,
# we assign it the smallest possible value (1) and perform BFS/DFS to color
# its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer 1 over 2.
            # Since we process nodes 1..n, we always try to set 
            # the first uncolored node in the component to 1.
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs if using DFS, 
    # but here we use iterative stack-based approach.
    solve()
```
