import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition "sum must be 3" means if u is 1, v must be 2, and vice versa.
    # This is a 2-coloring problem on a graph.
    # We want the lexicographically smallest sequence, so we prefer 1 over 2.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try assigning 1 to the first node
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this component could have been smaller by starting with 2
            # Actually, for each component, we have two choices:
            # Choice A: color[i] = 1, others follow
            # Choice B: color[i] = 2, others follow
            # We pick the one that makes the sequence lexicographically smaller.
            # Since we process nodes 1 to n, we just need to ensure the first node
            # of each component gets the smallest possible value (1).
            # The current logic already assigns 1 to the first uncolored node encountered.
            
    print("YES")
    print(*(color[1:]))

solve()
