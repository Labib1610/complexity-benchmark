```python
import sys

# The problem states that for any two interacting workers, the product of their 
# schedule codes must be 2. Since codes are 1 or 2, the only way to get a product 
# of 2 is if one worker has code 1 and the other has code 2.
# This means the graph must be bipartite. If it is bipartite, we can color it 
# with 1s and 2s. To get the lexicographically smallest assignment, we want 
# worker 1 to have code 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # We want the lexicographically smallest assignment.
    # This means for each connected component, we want the smallest ID in that 
    # component to have code 1 if possible.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, assign color 1 to i
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
