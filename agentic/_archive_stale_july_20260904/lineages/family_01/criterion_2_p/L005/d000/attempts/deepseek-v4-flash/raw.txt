We can solve this by checking if the graph is bipartite using BFS/DFS, and assigning shifts in a way that gives the lexicographically smallest assignment.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
possible = True

for i in range(n):
    if color[i] == -1:
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    print("YES")
    # To get lexicographically smallest, we want as many 1's as possible at the start.
    # Since color 0 -> shift 1, color 1 -> shift 2, we can try flipping all colors
    # if that gives more 1's at the beginning.
    # But lexicographically smallest means: at first position where they differ,
    # the one with 1 is smaller. So we want 1's as early as possible.
    # We can try both colorings (original and flipped) and pick the lexicographically smaller.
    assign1 = [1 if c == 0 else 2 for c in color]
    assign2 = [2 if c == 0 else 1 for c in color]
    if assign1 < assign2:
        print(" ".join(map(str, assign1)))
    else:
        print(" ".join(map(str, assign2)))
```
