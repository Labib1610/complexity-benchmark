import sys

# The problem states that for every edge (u, v), the sum of tiers must be 3.
# Since tiers are 1 or 2, this means if node u has tier s_u, node v must have tier 3 - s_u.
# This is equivalent to saying that if s_u = 1, s_v = 2, and if s_u = 2, s_v = 1.
# This is a bipartite coloring problem.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# For each connected component, there are two possible assignments (swap 1s and 2s).
# To get the lexicographically smallest assignment, for each component, we check
# both valid assignments and pick the one that is lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible assignments for this component:
            # 1. color[v] = 1 -> tier 1, color[v] = 2 -> tier 2
            # 2. color[v] = 1 -> tier 2, color[v] = 2 -> tier 1
            
            # We want the lexicographically smallest result.
            # Compare the two assignments for the nodes in this component.
            # Since we want the smallest overall, we compare the first node in the component.
            
            # Assignment A: color 1 -> 1, color 2 -> 2
            # Assignment B: color 1 -> 2, color 2 -> 1
            
            # To compare, we look at the smallest index node in the component.
            # Let min_node be the node with the smallest index in the component.
            # If color[min_node] == 1:
            #   Assignment A gives tier 1, Assignment B gives tier 2.
            #   A is smaller.
            # If color[min_node] == 2:
            #   Assignment A gives tier 2, Assignment B gives tier 1.
            #   B is smaller.
            
            min_node = min(component)
            if color[min_node] == 1:
                for node in component:
                    ans[node] = color[node]
            else:
                for node in component:
                    ans[node] = 3 - color[node]
                    
    print("YES")
    print(*(ans[1:]))

solve()
