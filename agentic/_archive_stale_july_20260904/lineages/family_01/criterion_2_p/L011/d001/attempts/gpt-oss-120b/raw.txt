import sys
sys.setrecursionlimit(1 << 25)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * n          # 0 or 1
    comp_id = [-1] * n
    comps = []                # list of (nodes list, min_node)

    from collections import deque

    for i in range(n):
        if color[i] != -1:
            continue
        q = deque([i])
        color[i] = 0
        nodes = [i]
        min_node = i
        while q:
            v = q.popleft()
            for nb in adj[v]:
                if color[nb] == -1:
                    color[nb] = color[v] ^ 1
                    q.append(nb)
                    nodes.append(nb)
                    if nb < min_node:
                        min_node = nb
                elif color[nb] == color[v]:
                    print("NO")
                    return
        cid = len(comps)
        for node in nodes:
            comp_id[node] = cid
        comps.append((nodes, min_node))

    flips = [False] * len(comps)
    for idx, (_, min_node) in enumerate(comps):
        flips[idx] = (color[min_node] == 1)   # flip if min node currently color 1

    ans = [0] * n
    for v in range(n):
        cid = comp_id[v]
        if flips[cid]:
            ans[v] = 2 - color[v]   # map 0->2, 1->1
        else:
            ans[v] = color[v] + 1    # map 0->1, 1->2

    print("YES")
    print(' '.join(map(str, ans)))

if __name__ == "__main__":
    solve()
