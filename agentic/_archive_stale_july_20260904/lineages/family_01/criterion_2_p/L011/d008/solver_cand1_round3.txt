```python
import sys

# The resonance of a channel (u, v) is det(M_{u,v}) = x_u * x_v - (1 * -1) = x_u * x_v + 1.
# We need the product of (x_u * x_v + 1) for all m edges to be 3^m.
# Since x_i is in {1, 2}:
# If x_u = 1, x_v = 1: det = 1*1 + 1 = 2
# If x_u = 1, x_v = 2: det = 1*2 + 1 = 3
# If x_u = 2, x_v = 1: det = 2*1 + 1 = 3
# If x_u = 2, x_v = 2: det = 2*2 + 1 = 5
#
# We need the product to be 3^m. This means every edge must have a resonance of 3.
# Resonance is 3 if and only if one node is 1 and the other is 2.
# This is equivalent to saying the graph must be bipartite, and we must assign
# tiers such that every edge connects a 1 and a 2.
#
# For each connected component, if it is not bipartite, output NO.
# If it is bipartite, there are two ways to assign tiers (swap 1s and 2s).
# To get the lexicographically smallest sequence, for each component,
# we check both assignments and pick the one that is smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        component.append(v)
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible assignments:
            # Option 1: nodes with color 0 get 1, nodes with color 1 get 2
            # Option 2: nodes with color 0 get 2, nodes with color 1 get 1
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            comp_nodes = sorted(component)
            first = comp_nodes[0]
            
            # If color[first] == 0:
            # Option 1: first gets 1
            # Option 2: first gets 2
            # We prefer Option 1.
            if color[first] == 0:
                for node in component:
                    ans[node] = 1 if color[node] == 0 else 2
            else:
                for node in component:
                    ans[node] = 2 if color[node] == 0 else 1
                    
    print("YES")
    print(*(ans[1:]))

solve()
```
