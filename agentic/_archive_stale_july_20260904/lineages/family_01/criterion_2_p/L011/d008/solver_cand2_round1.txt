import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        idx += 2
        
    # Let x_i be 1 or 2.
    # Binding deficit for (u, v) is 5 - x_u - x_v.
    # If x_u=1, x_v=1: deficit = 3
    # If x_u=1, x_v=2: deficit = 2
    # If x_u=2, x_v=2: deficit = 1
    # Total lattice deficit = product of (5 - x_u - x_v) for all edges.
    # Baseline energy = product of x_i^(d_i) for all atoms.
    # Since x_i is 1 or 2, x_i^(d_i) is 1 if x_i=1, and 2^(d_i) if x_i=2.
    # Let S2 be the set of atoms where x_i = 2.
    # Baseline energy = 2^(sum_{i in S2} d_i).
    # Total lattice deficit = 3^(number of edges with both endpoints in S1) * 
    #                         2^(number of edges with one endpoint in S1 and one in S2) *
    #                         1^(number of edges with both endpoints in S2).
    # For the product to be a power of 2, the number of edges with both endpoints in S1 must be 0.
    # This means S1 must be an independent set.
    # If S1 is an independent set, then every edge has at least one endpoint in S2.
    # Let E_12 be the set of edges with one endpoint in S1 and one in S2.
    # Let E_22 be the set of edges with both endpoints in S2.
    # Total lattice deficit = 2^|E_12|.
    # Baseline energy = 2^(sum_{i in S2} d_i).
    # We need |E_12| = sum_{i in S2} d_i.
    # Note that sum_{i in S2} d_i = |E_12| + 2 * |E_22|.
    # So we need |E_12| = |E_12| + 2 * |E_22|, which implies |E_22| = 0.
    # This means S2 must also be an independent set.
    # Thus, the graph must be bipartite, and S1 and S2 are the two partitions.
    # If the graph is bipartite, we can color it with 1s and 2s.
    # Let the two partitions be P1 and P2.
    # Case 1: S1 = P1, S2 = P2. Then |E_22| = 0, |E_12| = m.
    # Condition: m = sum_{i in P2} d_i. Since sum_{i in P2} d_i = m, this is always true.
    # Case 2: S1 = P2, S2 = P1. Condition: m = sum_{i in P1} d_i = m, always true.
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # We have two valid assignments: color[i] or 3 - color[i].
    # We want the lexicographically smallest sequence.
    # Compare the two sequences and pick the smaller one.
    res1 = [color[i] for i in range(1, n + 1)]
    res2 = [3 - color[i] for i in range(1, n + 1)]
    
    print("YES")
    if res1 <= res2:
        print(*(res1))
    else:
        print(*(res2))

solve()
