import sys

# The condition is 3 * x_u * x_v = 2 * (x_u + x_v)
# 3 * x_u * x_v - 2 * x_u - 2 * x_v = 0
# Multiply by 3: 9 * x_u * x_v - 6 * x_u - 6 * x_v = 0
# (3 * x_u - 2) * (3 * x_v - 2) = 4
# Let A = 3 * x_u - 2, B = 3 * x_v - 2. A * B = 4.
# Since x_u, x_v >= 1, A, B >= 1.
# Possible pairs (A, B) such that A*B = 4:
# 1) A=1, B=4 => 3*x_u-2=1 => x_u=1; 3*x_v-2=4 => x_v=2
# 2) A=4, B=1 => 3*x_u-2=4 => x_u=2; 3*x_v-2=1 => x_v=1
# 3) A=2, B=2 => 3*x_u-2=2 => 3*x_u=4 (no integer solution)
# So for any edge (u, v), one node must be 1 and the other must be 2.
# This means the graph must be bipartite.
# To get the lexicographically smallest assignment, we want x_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    colors = [0] * (n + 1)
    for i in range(1, n + 1):
        if colors[i] == 0:
            # BFS to color component
            # To make it lexicographically smallest, we want 1s at smaller indices.
            # If we have a choice for the first node of a component, pick 1.
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    # colors are 1 or 2. If color is 1, x=1. If color is 2, x=2.
    # Wait, check if this is lexicographically smallest.
    # We want x_i=1 for as many small i as possible.
    # The current coloring is fixed by the graph structure.
    # For each component, we can either have (color 1 -> 1, color 2 -> 2)
    # or (color 1 -> 2, color 2 -> 1).
    # We choose the one that makes the sequence lexicographically smaller.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            q = [i]
            visited[i] = True
            while q:
                u = q.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        q.append(v)
            
            # Try both assignments for this component
            # Assignment 1: color 1 -> 1, color 2 -> 2
            # Assignment 2: color 1 -> 2, color 2 -> 1
            # We pick the one that is lexicographically smaller.
            # Since we process components independently, we just need to compare
            # the values at the smallest index in the component.
            
            first_node = min(component)
            # If colors[first_node] == 1:
            #   Option 1: x[first_node] = 1
            #   Option 2: x[first_node] = 2
            #   Pick Option 1.
            # If colors[first_node] == 2:
            #   Option 1: x[first_node] = 2
            #   Option 2: x[first_node] = 1
            #   Pick Option 2.
            
            target_color = 1 if colors[first_node] == 1 else 2
            for node in component:
                if colors[node] == target_color:
                    res[node] = 1
                else:
                    res[node] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
