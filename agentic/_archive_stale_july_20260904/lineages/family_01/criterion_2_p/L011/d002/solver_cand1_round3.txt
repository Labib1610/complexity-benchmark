```python
import sys

# The problem states: x_i * x_j = 2, where x_i, x_j in {1, 2}.
# This implies that for every edge (u, v), one must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want x_1 to be 1 if possible.
# If x_1 = 1, then all neighbors of 1 must be 2, all neighbors of those neighbors must be 1, etc.
# We can use BFS/DFS to assign values to each connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is unassigned, we try assigning 1 to it.
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, val = stack.pop()
                target = 3 - val
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = target
                        stack.append((v, target))
                    elif assignment[v] != target:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

solve()
```
