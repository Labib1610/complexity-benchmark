```python
import sys

# The condition is integral from 1 to 2 of S_u(t) * S_v(t) <= 0.
# If S_u(t) = t - x_u and S_v(t) = t - x_v,
# Integral_{1}^{2} (t - x_u)(t - x_v) dt
# = Integral_{1}^{2} (t^2 - (x_u + x_v)t + x_u*x_v) dt
# = [t^3/3 - (x_u + x_v)t^2/2 + x_u*x_v*t] from 1 to 2
# = (8/3 - 2(x_u + x_v) + 2*x_u*x_v) - (1/3 - 1/2(x_u + x_v) + x_u*x_v)
# = 7/3 - 3/2(x_u + x_v) + x_u*x_v
#
# Case 1: x_u = 1, x_v = 1: 7/3 - 3/2(2) + 1 = 7/3 - 3 + 1 = 7/3 - 2 = 1/3 > 0 (Invalid)
# Case 2: x_u = 2, x_v = 2: 7/3 - 3/2(4) + 4 = 7/3 - 6 + 4 = 7/3 - 2 = 1/3 > 0 (Invalid)
# Case 3: x_u = 1, x_v = 2: 7/3 - 3/2(3) + 2 = 7/3 - 4.5 + 2 = 7/3 - 2.5 = 2.333 - 2.5 = -0.166 < 0 (Valid)
#
# Thus, for every edge (u, v), x_u must not equal x_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence x_1, ..., x_n.
# This means we prefer x_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we greedily assign 1 if possible
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
