import sys

# The condition is: sum_{j in neighbors(i)} x_j = d_i * (3 - x_i)
# Let y_i = x_i - 1. Since x_i in {1, 2}, y_i in {0, 1}.
# x_i = y_i + 1.
# sum_{j in neighbors(i)} (y_j + 1) = d_i * (3 - (y_i + 1))
# sum_{j in neighbors(i)} y_j + d_i = d_i * (2 - y_i)
# sum_{j in neighbors(i)} y_j + d_i = 2 * d_i - d_i * y_i
# sum_{j in neighbors(i)} y_j + d_i * y_i = d_i
# This is equivalent to: sum_{j in neighbors(i) union {i}} y_j = d_i
# This is a system of linear equations over GF(2) if we consider the parity,
# but actually, it's a system of linear equations over integers.
# However, since y_i is 0 or 1, this is a standard 2-SAT or bipartite coloring problem.
# Specifically, for each connected component, if it's bipartite with parts A and B:
# If we set all nodes in A to y_A and all nodes in B to y_B:
# For i in A: sum_{j in neighbors(i)} y_j + d_i * y_i = d_i * y_B + d_i * y_A = d_i * (y_A + y_B) = d_i
# This implies y_A + y_B = 1.
# So for each component, we must have y_A != y_B.
# This means one set is 0 and the other is 1.
# If a component is not bipartite, it's impossible unless d_i = 0 (isolated node).
# For isolated nodes, d_i = 0, so 0 = 0, which is always true. x_i can be 1 or 2.
# To get lexicographically smallest x, we want y_i = 0 (x_i = 1) whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            if not adj[i]:
                ans[i] = 1
                continue
            
            component = []
            stack = [(i, 0)]
            color[i] = 0
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        is_bipartite = False
            
            if not is_bipartite:
                print("NO")
                return
            
            # Check if valid: sum_{j in neighbors(i)} y_j + d_i * y_i = d_i
            # For bipartite, y_i = c or 1-c.
            # If y_i = c for all i in color 0, and y_i = 1-c for all i in color 1:
            # For i in color 0: sum_{j in neighbors(i)} (1-c) + d_i * c = d_i * (1-c) + d_i * c = d_i. Correct.
            # So we have two choices for each component: (c=0, 1-c=1) or (c=1, 1-c=0).
            # We want lexicographically smallest x_i, so smallest y_i.
            # For each component, compare the two assignments and pick the one that is smaller.
            
            comp_nodes = [[] for _ in range(2)]
            for node, c in component:
                comp_nodes[c].append(node)
            
            # Option 1: color 0 -> y=0, color 1 -> y=1
            # Option 2: color 0 -> y=1, color 1 -> y=0
            # Compare lexicographically
            opt1 = {}
            for node in comp_nodes[0]: opt1[node] = 0
            for node in comp_nodes[1]: opt1[node] = 1
            
            opt2 = {}
            for node in comp_nodes[0]: opt2[node] = 1
            for node in comp_nodes[1]: opt2[node] = 0
            
            nodes = sorted(comp_nodes[0] + comp_nodes[1])
            for node in nodes:
                if opt1[node] < opt2[node]:
                    for n_ in nodes: ans[n_] = opt1[n_] + 1
                    break
                elif opt2[node] < opt1[node]:
                    for n_ in nodes: ans[n_] = opt2[n_] + 1
                    break
                    
    print("YES")
    print(*(ans[1:]))

solve()
