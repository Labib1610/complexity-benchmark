import sys

# The number of paths from (0,0) to (x_u, x_v) is given by the binomial coefficient
# C(x_u + x_v, x_u).
# Possible states are x_i in {1, 2}.
# Possible pairs (x_u, x_v):
# (1, 1): C(1+1, 1) = C(2, 1) = 2
# (1, 2): C(1+2, 1) = C(3, 1) = 3
# (2, 1): C(2+1, 2) = C(3, 2) = 3
# (2, 2): C(2+2, 2) = C(4, 2) = 6
# The condition "exactly 3 paths" means (x_u, x_v) must be (1, 2) or (2, 1).
# This is equivalent to x_u != x_v.
# The problem is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment.
# For each connected component, we have two choices for the 2-coloring.
# To make the sequence lexicographically smallest, for each component,
# we pick the coloring that assigns 1 to the smallest node index in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Try assigning 1 to the smallest node in the component
            # color[i] is 1. If we keep it, node i gets 1.
            # If we flip, node i gets 2.
            # Since we want lexicographically smallest, we want the smallest node
            # in the component to have 1.
            min_node = min(component)
            if color[min_node] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
