import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # s_i is 1 or 2.
    # Impedance ratio for link (u, v) is s_u/s_v + s_v/s_u.
    # If s_u == s_v, ratio = 1/1 + 1/1 = 2 or 2/2 + 2/2 = 2.
    # If s_u != s_v, ratio = 1/2 + 2/1 = 2.5.
    # Total impedance = 2 * (number of edges with same state) + 2.5 * (number of edges with different states).
    # We want total impedance = 2.5 * m.
    # Let k be the number of edges with different states.
    # 2 * (m - k) + 2.5 * k = 2.5 * m
    # 2m - 2k + 2.5k = 2.5m
    # 0.5k = 0.5m => k = m.
    # This means every edge must connect units with different power states.
    # This is equivalent to checking if the graph is bipartite.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    # To get lexicographically smallest, we want s_i = 1 as much as possible.
    # The graph might have multiple connected components.
    # For each component, we have two choices:
    # Choice 1: nodes with color 1 get s=1, nodes with color 2 get s=2.
    # Choice 2: nodes with color 1 get s=2, nodes with color 2 get s=1.
    # We want to pick the choice for each component that makes the sequence lexicographically smallest.
    
    res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Find the smallest index in this component
            min_node = min(component)
            # If color[min_node] == 1, we can choose s[min_node]=1 or s[min_node]=2.
            # To be lexicographically smallest, we prefer s[min_node]=1.
            # If color[min_node] == 1, setting color 1 -> 1 and color 2 -> 2 makes s[min_node]=1.
            # If color[min_node] == 2, setting color 2 -> 1 and color 1 -> 2 makes s[min_node]=1.
            
            c_min = color[min_node]
            for node in component:
                if color[node] == c_min:
                    res[node] = 1
                else:
                    res[node] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
