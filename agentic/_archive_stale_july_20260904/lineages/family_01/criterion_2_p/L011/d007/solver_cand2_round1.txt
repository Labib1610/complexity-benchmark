```python
import sys

# The problem states:
# S_init = (u+v) + (v-u)z
# P1 = u - vz
# P2 = u + vz
# Modulo z^2 = -1
#
# Let S = A + Bz.
# S * P1 = (A + Bz)(u - vz) = Au - Avz + Bzu - Bvz^2 = (Au + Bv) + (Bu - Av)z
# S * P2 = (A + Bz)(u + vz) = Au + Avz + Bzu + Bvz^2 = (Au - Bv) + (Bu + Av)z
#
# Let x_u, x_v be the states of nodes u and v.
# If x_u = 1, x_v = 1: S_final = S_init * P1 * P1
# S_init * P1 = (u(u+v) + v(v-u)) + (u(v-u) - v(u+v))z
#             = (u^2 + uv + v^2 - uv) + (uv - u^2 - uv - v^2)z
#             = (u^2 + v^2) - (u^2 + v^2)z
# S_final = ((u^2+v^2)u + (u^2+v^2)v) + ((u^2+v^2)u - (u^2+v^2)v)z
#         = (u^2+v^2)(u+v) + (u^2+v^2)(u-v)z
# This does not match the target (u^2+v^2)(u+v) + (u^2+v^2)(v-u)z.
#
# If x_u = 1, x_v = 2: S_final = S_init * P1 * P2
# S_init * P1 = (u^2 + v^2) - (u^2 + v^2)z
# S_final = ((u^2+v^2)u - (-(u^2+v^2)v)) + ((u^2+v^2)u + (u^2+v^2)v)z
#         = (u^2+v^2)(u+v) + (u^2+v^2)(u+v)z ... wait.
#
# Re-evaluating:
# S_init = (u+v) + (v-u)z
# P1 = u - vz, P2 = u + vz
# If x_u=1, x_v=2: S_final = S_init * P1 * P2 = S_init * (u^2 + v^2) = (u^2+v^2)(u+v) + (u^2+v^2)(v-u)z
# This matches the target!
# So for every edge (u, v), we must have x_u != x_v.
# This is a bipartite graph coloring problem.
# Lexicographically smallest: for each connected component, try x_i=1 for the smallest node.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
