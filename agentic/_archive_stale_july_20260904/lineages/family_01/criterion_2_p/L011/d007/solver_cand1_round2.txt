```python
import sys

# The problem asks for x_i in {1, 2} such that the product of (x_u + x_v) for all edges (u, v) is 3^m.
# Since x_i is 1 or 2, x_u + x_v can be 2 (1+1), 3 (1+2), or 4 (2+2).
# The product is 2^a * 3^b * 4^c = 2^(a+2c) * 3^b.
# For this to be 3^m, we must have a=0 and c=0.
# This means x_u + x_v must be 3 for every edge (u, v).
# x_u + x_v = 3 implies that if x_u = 1, then x_v = 2, and if x_u = 2, then x_v = 1.
# This is exactly the condition for a graph to be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want x_1 = 1.
# For each connected component, we set the first node encountered to 1, 
# then propagate the coloring. If a conflict arises, it's not bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if ans[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want x_i = 1 if possible.
            # Since we iterate 1 to n, we always set the first node of a component to 1.
            stack = [(i, 1)]
            ans[i] = 1
            while stack:
                u, color = stack.pop()
                next_color = 3 - color
                for v in adj[u]:
                    if ans[v] == 0:
                        ans[v] = next_color
                        stack.append((v, next_color))
                    elif ans[v] != next_color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(ans[1:]))

solve()
```
