```python
import sys
from collections import defaultdict, deque

def transform_vector(state, x, y):
    if state == 1:
        return (x + 3*y, y)
    else:  # state == 2
        return (x, 3*x + y)

def check_balance(state1, state2):
    # Start with vector (1, 1)
    x, y = 1, 1
    # Apply transformations in order
    x, y = transform_vector(state1, x, y)
    x, y = transform_vector(state2, x, y)
    return x + y == 17

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # For each edge, determine which state combinations are valid
    valid_combinations = {}
    for u, v in edges:
        valid = []
        for state_u in [1, 2]:
            for state_v in [1, 2]:
                if check_balance(state_u, state_v) or check_balance(state_v, state_u):
                    valid.append((state_u, state_v))
        valid_combinations[(u, v)] = valid
    
    # Check if any edge has no valid combinations
    for u, v in edges:
        if not valid_combinations[(u, v)]:
            print("NO")
            return
    
    # Try to find a valid assignment using BFS/DFS
    # We'll use a 2-SAT approach conceptually but implement it directly
    # For each connected component, we'll try to 2-color it with constraints
    
    # Color assignment: 0 means not assigned, 1 and 2 are the states
    assignment = [0] * (n + 1)
    
    # Process each connected component
    visited = [False] * (n + 1)
    
    for start_node in range(1, n + 1):
        if visited[start_node] or not adj[start_node]:
            continue
            
        # BFS to process component
        queue = deque([start_node])
        visited[start_node] = True
        
        # Try state 1 first for lexicographically smallest
        if assignment[start_node] == 0:
            assignment[start_node] = 1
        
        while queue:
            node = queue.popleft()
            node_state = assignment[node]
            
            # Check all neighbors
            for neighbor in adj[node]:
                # Find valid combinations for this edge
                edge_key = (node, neighbor) if (node, neighbor) in valid_combinations else (neighbor, node)
                if edge_key not in valid_combinations:
                    edge_key = (neighbor, node)
                
                valid_pairs = valid_combinations[edge_key]
                
                # Find what state neighbor should have
                valid_neighbor_states = []
                for state1, state2 in valid_pairs:
                    if edge_key == (node, neighbor) and state1 == node_state:
                        valid_neighbor_states.append(state2)
                    elif edge_key == (neighbor, node) and state2 == node_state:
                        valid_neighbor_states.append(state1)
                
                if not valid_neighbor_states:
                    print("NO")
                    return
                
                # If neighbor is not assigned, assign the smallest valid state
                if assignment[neighbor] == 0:
                    assignment[neighbor] = min(valid_neighbor_states)
                    visited[neighbor] = True
                    queue.append(neighbor)
                else:
                    # Check if current assignment is valid
                    if assignment[neighbor] not in valid_neighbor_states:
                        print("NO")
                        return
    
    # Handle isolated nodes (nodes with no edges)
    for i in range(1, n + 1):
        if assignment[i] == 0:
            assignment[i] = 1
    
    # Double-check all edges
    for u, v in edges:
        valid = False
        for state_u, state_v in valid_combinations[(u, v)]:
            if assignment[u] == state_u and assignment[v] == state_v:
                valid = True
                break
        if not valid:
            print("NO")
            return
    
    print("YES")
    print(" ".join(map(str, assignment[1:])))

solve()
```
