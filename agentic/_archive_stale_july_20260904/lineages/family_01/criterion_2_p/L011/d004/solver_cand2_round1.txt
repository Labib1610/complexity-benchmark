import sys

# The resonance equation:
# x_u^6 + x_v^6 + x_u^4 x_v^2 + x_u^2 x_v^4 + 4 x_u^3 x_v^3 - 18 (x_u^3 + x_v^3) - 12 x_u x_v (x_u + x_v) + 117 = 0
# Let f(x, y) = x^6 + y^6 + x^4 y^2 + x^2 y^4 + 4 x^3 y^3 - 18 (x^3 + y^3) - 12 xy (x + y) + 117
# This factors as (x^3 + y^3 - 9)^2 + 3(x^3 - y^3)^2 + (x^2 - y^2)^2 + ... actually, let's test small values.
# For x, y in {1, 2, 3, ...}:
# f(1, 2) = 1 + 64 + 4 + 16 + 32 - 18(9) - 12(2)(3) + 117 = 117 + 117 - 162 - 72 = 0.
# f(1, 1) = 1+1+1+1+4 - 18(2) - 12(2) + 117 = 8 - 36 - 24 + 117 = 65 != 0.
# f(2, 2) = 64+64+64+64+256 - 18(16) - 12(8) + 117 = 512 - 288 - 96 + 117 = 245 != 0.
# f(1, 3) = 1+729+9+81+108 - 18(28) - 12(3)(4) + 117 = 928 - 504 - 144 + 117 = 397 != 0.
# It turns out the equation is satisfied if and only if {x, y} = {1, 2}.
# This means the graph must be bipartite, and for every edge (u, v), one must be 1 and the other 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    # To get lexicographically smallest, we want x_i = 1 whenever possible.
    # For each connected component, we have two choices:
    # Choice A: nodes with color 1 get 1, nodes with color 2 get 2.
    # Choice B: nodes with color 1 get 2, nodes with color 2 get 1.
    # We pick the one that makes the first node in the component 1.
    
    final_assignment = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Determine if we should assign 1 to color 1 or color 2
            # We want the smallest value at the smallest index in the component.
            # Let min_node be the smallest index in this component.
            # If color[min_node] == 1, we want color 1 to be 1.
            # If color[min_node] == 2, we want color 2 to be 1.
            
            min_node = min(component)
            target_color = color[min_node]
            
            for node in component:
                if color[node] == target_color:
                    final_assignment[node] = 1
                else:
                    final_assignment[node] = 2
                    
    print("YES")
    print(*(final_assignment[1:]))

solve()
