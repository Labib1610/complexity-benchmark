import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            
    # The quadratic form E(x) = x^T H x simplifies to:
    # E(x) = sum_{(u,v) in edges} (x_u - x_v)^2
    # We want E(x) = m.
    # Since x_i in {1, 2}, (x_u - x_v)^2 is 1 if x_u != x_v, and 0 if x_u == x_v.
    # So E(x) = number of edges (u, v) such that x_u != x_v.
    # We want this to be exactly m.
    # This means for every edge (u, v), we must have x_u != x_v.
    # This is equivalent to saying the graph must be bipartite.
    
    color = [0] * (n + 1)
    possible = True
    
    # To get the lexicographically smallest sequence, we want x_i = 1 whenever possible.
    # For each connected component, we can pick the color of the first node.
    # If we pick color 1 for the first node, we get a specific assignment.
    # If we pick color 2, we get the other.
    # We want to pick the color such that the resulting sequence is lexicographically smallest.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # Collect all nodes in component
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append((v, color[v]))
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a bipartite component.
            # We can either keep the current coloring or flip it (1->2, 2->1).
            # We want the smallest value at the smallest index i in this component.
            # Since i is the smallest index in this component, we want color[i] to be 1.
            # Our BFS already set color[i] = 1.
            # So we just need to check if this is the best we can do.
            # Actually, for each component, we just need to ensure the first node
            # encountered (the smallest index) is 1.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
