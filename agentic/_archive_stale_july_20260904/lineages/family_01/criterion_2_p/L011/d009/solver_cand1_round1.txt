```python
import sys

# The equation (C + D)X = 3D * 1 can be rewritten for each node i:
# sum_{j in neighbors(i)} x_j + d_i * x_i = 3 * d_i
# Let y_i = x_i - 2. Since x_i in {1, 2}, y_i in {-1, 0}.
# The equation becomes:
# sum_{j in neighbors(i)} (y_j + 2) + d_i * (y_i + 2) = 3 * d_i
# sum_{j in neighbors(i)} y_j + 2 * d_i + d_i * y_i + 2 * d_i = 3 * d_i
# sum_{j in neighbors(i)} y_j + d_i * y_i = -d_i
# Let z_i = -y_i. Since y_i in {-1, 0}, z_i in {0, 1}.
# -sum_{j in neighbors(i)} z_j - d_i * z_i = -d_i
# sum_{j in neighbors(i)} z_j + d_i * z_i = d_i
# This is equivalent to: sum_{j in neighbors(i)} (z_j - z_i) = 0
# This means for every node i, the number of neighbors j with z_j = 1 
# must equal d_i * (1 - z_i).
# If z_i = 1, then sum_{j in neighbors(i)} z_j = 0, so all neighbors must have z_j = 0.
# If z_i = 0, then sum_{j in neighbors(i)} z_j = d_i, so all neighbors must have z_j = 1.
# This is exactly the condition for a bipartite graph coloring where z_i is the color.
# z_i = 1 corresponds to x_i = 1, z_i = 0 corresponds to x_i = 2.
# The graph must be bipartite. To get lexicographically smallest x, we want 
# x_i = 1 as much as possible, which means z_i = 1 as much as possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    z = [-1] * (n + 1)
    for i in range(1, n + 1):
        if z[i] == -1:
            # To make x lexicographically smallest, we prefer x_i=1 (z_i=1)
            # If we set z_i = 1, all neighbors must be 0.
            stack = [(i, 1)]
            z[i] = 1
            while stack:
                u, color = stack.pop()
                for v in adj[u]:
                    if z[v] == -1:
                        z[v] = 1 - color
                        stack.append((v, z[v]))
                    elif z[v] == color:
                        print("NO")
                        return
    
    res = []
    for i in range(1, n + 1):
        # x_i = 2 - z_i
        res.append(str(2 - z[i]))
    
    print("YES")
    print(" ".join(res))

solve()
```
