import sys

# The thermal footprint of a link between u and v is 2^{s_u} + 2^{s_v} - 1.
# s_i \in {1, 2}.
# Possible values for 2^{s_i}: 2^1 = 2, 2^2 = 4.
# Possible values for 2^{s_u} + 2^{s_v} - 1:
# If s_u=1, s_v=1: 2+2-1 = 3
# If s_u=1, s_v=2: 2+4-1 = 5
# If s_u=2, s_v=1: 4+2-1 = 5
# If s_u=2, s_v=2: 4+4-1 = 7
# We want the product of these values to be 5^m.
# This means every link must have a footprint of 5.
# A footprint of 5 occurs if and only if one unit is 1 and the other is 2.
# This is equivalent to saying the graph must be bipartite, and we must
# assign 1 to one partition and 2 to the other.
# To get the lexicographically smallest sequence, we want s_i = 1 as much as possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS to check bipartiteness and assign colors
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For each component, we have two choices:
            # 1. Nodes with color 1 get s=1, nodes with color 2 get s=2
            # 2. Nodes with color 1 get s=2, nodes with color 2 get s=1
            # We want the lexicographically smallest sequence.
            # Compare the two options for the smallest index in the component.
            first_node = min(u for u, c in component)
            # Find the color of the first_node
            c_first = 0
            for u, c in component:
                if u == first_node:
                    c_first = c
                    break
            
            # Option A: c_first -> 1, (3-c_first) -> 2
            # Option B: c_first -> 2, (3-c_first) -> 1
            # We prefer s=1. If c_first is 1, Option A gives s=1 to first_node.
            # If c_first is 2, Option B gives s=1 to first_node.
            # Actually, we just need to ensure the smallest index gets 1 if possible.
            if c_first == 1:
                # Keep as is
                pass
            else:
                # Flip colors
                for j in range(len(component)):
                    u, c = component[j]
                    component[j] = (u, 3 - c)
                    color[u] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
