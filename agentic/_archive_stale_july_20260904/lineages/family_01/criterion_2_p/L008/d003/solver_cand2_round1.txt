```python
import sys

# The problem asks for an assignment s_i in {1, 2} such that the product of (s_u + s_v) 
# for all m edges is 3^m.
# Since s_i is 1 or 2, s_u + s_v can be 1+1=2, 1+2=3, or 2+2=4.
# The product is 3^m if and only if every edge (u, v) satisfies s_u + s_v = 3.
# This means for every edge (u, v), one worker must be 1 and the other must be 2.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want s_1 to be 1.
# If s_1 = 1, then all neighbors of 1 must be 2, their neighbors must be 1, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for each connected component
            # To get lexicographically smallest, we prefer 1 over 2.
            # For the first node in a component, assign 1.
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                next_color = 3 - color
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = next_color
                        stack.append((v, next_color))
                    elif assignment[v] == color:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

solve()
```
