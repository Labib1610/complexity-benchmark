```python
import sys

# The problem states:
# Shift 1: Primary=5, Secondary=10
# Shift 2: Primary=10, Secondary=5
# Link (u, v) is stable if Primary(u) == Secondary(v).
# If u=1, Primary(u)=5. For stability, Secondary(v) must be 5, so v must be 2.
# If u=2, Primary(u)=10. For stability, Secondary(v) must be 10, so v must be 1.
# Thus, for every link (u, v), u and v must have different shift codes.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we iterate 1 to n.
# If a node is unvisited, assign it shift 1 (the smaller value).
# Then perform BFS/DFS to color the connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS for component
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
