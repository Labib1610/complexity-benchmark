import sys

# The local balance condition: sum_{j in N(i)} s_j = d_i * (3 - s_i)
# If s_i = 1, sum_{j in N(i)} s_j = 2 * d_i
# If s_i = 2, sum_{j in N(i)} s_j = 1 * d_i
# Let x_i = s_i - 1. Then x_i in {0, 1}.
# s_i = x_i + 1.
# If x_i = 0 (s_i = 1), sum (x_j + 1) = 2 * d_i => sum x_j + d_i = 2 * d_i => sum x_j = d_i
# If x_i = 1 (s_i = 2), sum (x_j + 1) = 1 * d_i => sum x_j + d_i = d_i => sum x_j = 0
# This is equivalent to: if x_i = 1, then for all neighbors j, x_j = 0.
# If x_i = 0, then sum_{j in N(i)} x_j = d_i.
# This implies that if x_i = 0, all neighbors must have x_j = 1.
# Thus, the graph must be bipartite. If it's bipartite, we can color it with 0s and 1s.
# Let the two partitions be A and B.
# If we set all nodes in A to 0 and all nodes in B to 1:
# For u in A: sum_{v in N(u)} x_v = d_u * 1 = d_u. (Satisfied)
# For v in B: sum_{u in N(v)} x_u = 0. (Satisfied)
# The harmonic condition: product of s_i in cycle = 2^(k/2).
# Since s_i in {1, 2}, product is 2^(number of 2s in cycle).
# We need number of 2s = k/2.
# In a bipartite graph, any cycle has even length k.
# If we alternate 1 and 2, every cycle has exactly k/2 ones and k/2 twos.
# So the product is 2^(k/2). This is always satisfied for bipartite graphs.
# We need to minimize sum s_i * 3^(n-i). This is equivalent to lexicographical minimization.
# We have two choices for bipartite coloring: (A=1, B=2) or (A=2, B=1).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Two possible assignments:
    # 1: color 0 -> 1, color 1 -> 2
    # 2: color 0 -> 2, color 1 -> 1
    res1 = [0] * n
    res2 = [0] * n
    for i in range(1, n + 1):
        res1[i-1] = 1 if color[i] == 0 else 2
        res2[i-1] = 2 if color[i] == 0 else 1
        
    print("YES")
    if res1 < res2:
        print(*(res1))
    else:
        print(*(res2))

solve()
