import sys

# The stability condition: trace(A_{s_u} A_{s_v}) = 6
# A1 = [[1, 2], [0, 1]], A2 = [[1, 0], [2, 1]]
# A1*A1 = [[1, 4], [0, 1]], trace = 2 != 6
# A2*A2 = [[1, 0], [4, 1]], trace = 2 != 6
# A1*A2 = [[5, 2], [2, 1]], trace = 6
# A2*A1 = [[1, 2], [2, 5]], trace = 6
# Thus, s_u != s_v for all edges (u, v). This is a bipartite graph check.
# Harmonic condition: h_u * h_v = 1 (mod 11) and h_i = s_i (mod 3).
# h_i in {1..10}.
# If s_i = 1, h_i in {1, 4, 7, 10}.
# If s_i = 2, h_i in {2, 5, 8}.
# If s_u = 1, s_v = 2: h_u * h_v = 1 (mod 11).
# Possible pairs (h_u, h_v):
# h_u=1: h_v=1 (no, 1%3=1), h_v=12 (no), h_v=1/1=1 (no)
# h_u=1, h_v=1 (1*1=1, 1=1 mod 3, 1=1 mod 3) -> s_u=1, s_v=1 (Violates s_u != s_v)
# Let's check:
# If s_u=1, h_u in {1, 4, 7, 10}. If s_v=2, h_v in {2, 5, 8}.
# 1*h_v = 1 mod 11 => h_v = 1 (not in {2,5,8})
# 4*h_v = 1 mod 11 => h_v = 3 (not in {2,5,8})
# 7*h_v = 1 mod 11 => h_v = 8 (in {2,5,8}!)
# 10*h_v = 1 mod 11 => h_v = 10 (not in {2,5,8})
# So if s_u=1, s_v=2, we must have h_u=7, h_v=8.
# This is consistent with s_u=1 (7%3=1) and s_v=2 (8%3=2).
# So the condition is simply that the graph must be bipartite.
# To minimize sum s_i * 3^(n-i), we want s_i=1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For each component, we have two choices:
            # 1. color 1 -> s=1, color 2 -> s=2
            # 2. color 1 -> s=2, color 2 -> s=1
            # We want to minimize sum s_i * 3^(n-i).
            # Find the first node in the component.
            comp_nodes = sorted([u for u, c in component])
            first_node = comp_nodes[0]
            
            # If we set color[first_node] = 1, then s[first_node] = 1
            # If we set color[first_node] = 2, then s[first_node] = 2
            # Since 1 < 2, we prefer s[first_node] = 1.
            c_first = color[first_node]
            for u, c in component:
                if c == c_first:
                    res[u] = 1
                else:
                    res[u] = 2
                    
    print("YES")
    print(*(res[1:]))

solve()
