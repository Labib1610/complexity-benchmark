import sys

# The condition: perimeter of rectangle with sides s_u and s_v is 2*(s_u + s_v) = 6.
# This implies s_u + s_v = 3.
# Since s_i \in {1, 2}, the only way s_u + s_v = 3 is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1s and 2s.
# To minimize sum(s_i * 3^(n-i)), we want s_i to be 1 whenever possible.
# For each connected component, there are two possible colorings (swap 1s and 2s).
# Let a component have nodes with indices i_1, i_2, ...
# Let the two sets of the bipartition be A and B.
# Option 1: A -> 1, B -> 2. Cost: sum_{i in A} 1 * 3^(n-i) + sum_{j in B} 2 * 3^(n-j)
# Option 2: A -> 2, B -> 1. Cost: sum_{i in A} 2 * 3^(n-i) + sum_{j in B} 1 * 3^(n-j)
# We choose the option that gives the smaller value for each component.
# Since 3^(n-i) is much larger for smaller i, we compare the first node index in the component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = map(int, input_data[:2])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = map(int, input_data[idx:idx+2])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Find the smallest index in the component
            first_node = min(node for node, c in component)
            # Find the color of the smallest index
            for node, c in component:
                if node == first_node:
                    first_color = c
                    break
            
            # If first_color is 1, Option 1 (1s and 2s) is better because 1 * 3^(n-first_node) < 2 * 3^(n-first_node)
            # If first_color is 2, Option 2 (swapping colors) is better
            if first_color == 1:
                for node, c in component:
                    ans[node] = c
            else:
                for node, c in component:
                    ans[node] = 3 - c
                    
    print("YES")
    print(*(ans[1:]))

solve()
