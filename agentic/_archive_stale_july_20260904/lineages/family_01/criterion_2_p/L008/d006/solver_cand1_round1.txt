```python
import sys

# The stability condition: trace(A_{s_u} * A_{s_v}) = 6
# A1 = [[1, 2], [0, 1]], A2 = [[1, 0], [2, 1]]
# A1*A1 = [[1, 4], [0, 1]], trace = 2 != 6
# A2*A2 = [[1, 0], [4, 1]], trace = 2 != 6
# A1*A2 = [[5, 2], [2, 1]], trace = 6
# A2*A1 = [[1, 2], [2, 5]], trace = 6
# Thus, s_u != s_v for every edge (u, v).
# This means the graph must be bipartite.
# The harmonic buildup condition: in every simple cycle, count(1) == count(2).
# In a bipartite graph, every cycle has an even length.
# If we color the graph with 1 and 2, any cycle will have equal numbers of 1s and 2s.
# So the condition is satisfied if and only if the graph is bipartite.
# To minimize sum(s_i * 3^{n-i}), we want s_i = 1 whenever possible.
# For each connected component, we have two choices for the bipartition (swap 1s and 2s).
# We choose the one that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # To minimize sum(s_i * 3^{n-i}), we want s_i=1 for the smallest i.
            # Find the smallest index in this component.
            comp_nodes = sorted([node for node, c in component])
            first_node = comp_nodes[0]
            # If color[first_node] is 2, we can flip all colors in this component.
            flip = (color[first_node] == 2)
            for node, c in component:
                res[node] = (3 - c) if flip else c
                
    print("YES")
    print(*(res[1:]))

solve()
```
