```python
import sys

# The phase vector is p_i = (s_i, s_i^2).
# s_i can be 1 or 2.
# p_i = (1, 1) if s_i = 1
# p_i = (2, 4) if s_i = 2
# Two vectors (x1, y1) and (x2, y2) are linearly independent if x1*y2 - x2*y1 != 0.
# If s_u = 1 and s_v = 1: (1, 1) and (1, 1) -> 1*1 - 1*1 = 0 (Dependent)
# If s_u = 2 and s_v = 2: (2, 4) and (2, 4) -> 2*4 - 2*4 = 0 (Dependent)
# If s_u = 1 and s_v = 2: (1, 1) and (2, 4) -> 1*4 - 2*1 = 2 != 0 (Independent)
# Thus, a link (u, v) is stable if and only if s_u != s_v.
# This is equivalent to checking if the graph is bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    # To minimize sum(s_i * 3^(n-i)), we want s_i to be 1 whenever possible.
    # We iterate 1 to n. If not colored, assign 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
