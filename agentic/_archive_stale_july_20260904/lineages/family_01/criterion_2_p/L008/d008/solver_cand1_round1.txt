import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            
    # (s_u - s_v)^2 is 1 if s_u != s_v, and 0 if s_u == s_v.
    # The equation is sum_{j=1}^m (s_{u_j} - s_{v_j})^2 = m.
    # This means for every pair (u_j, v_j), we must have s_{u_j} != s_{v_j}.
    # This is equivalent to checking if the graph is bipartite.
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have a valid assignment.
    # We want to minimize P(2) = sum s_i * 2^{n-i}.
    # This is equivalent to making s_1 as small as possible, then s_2, etc.
    # For each connected component, we can either use (color) or (3-color).
    # To minimize P(2), for each component, we pick the coloring that makes
    # the first node in the component (the one with the smallest index) 1.
    
    final_assignment = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            # Find all nodes in this component
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # The first node in the component (smallest index) should be 1
            # to minimize the polynomial value.
            root = min(component)
            target_color = 1
            current_color = color[root]
            
            for node in component:
                if color[node] == current_color:
                    final_assignment[node] = target_color
                else:
                    final_assignment[node] = 3 - target_color
                    
    print("YES")
    print(*(final_assignment[1:]))

solve()
