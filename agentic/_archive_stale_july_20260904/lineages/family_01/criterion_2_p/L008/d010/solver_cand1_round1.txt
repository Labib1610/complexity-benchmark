```python
import sys

# The complex phase signatures are c1 = 1 + i and c2 = 1 - i.
# For a link (u, v) with states s_u, s_v:
# If s_u = 1, s_v = 1: (1+i)(1+i) = 1 + 2i - 1 = 2i. Imaginary part = 2.
# If s_u = 2, s_v = 2: (1-i)(1-i) = 1 - 2i - 1 = -2i. Imaginary part = -2.
# If s_u = 1, s_v = 2: (1+i)(1-i) = 1 - i^2 = 2. Imaginary part = 0.
# If s_u = 2, s_v = 1: (1-i)(1+i) = 1 - i^2 = 2. Imaginary part = 0.
# The footprint is the sum of squares of imaginary parts.
# For each link, the square is 4 if s_u == s_v, and 0 if s_u != s_v.
# To have a total footprint of 0, we need s_u != s_v for all links.
# This is equivalent to checking if the graph is bipartite.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want to minimize s1, then s2, etc.
    # This means we prefer state 1 whenever possible.
    # For each connected component, we can pick the state of the first node.
    # To minimize lexicographically, we try s_i = 1 for the first node of each component.
    
    state = [0] * (n + 1)
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            stack = [(i, 1)]
            state[i] = 1
            while stack:
                u, s = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - s
                        stack.append((v, 3 - s))
                    elif state[v] == s:
                        print("NO")
                        return
    
    print("YES")
    print(*(state[1:]))

solve()
```
