import sys

# The condition is that for every edge (u, v), [s_u, s_v] must be a subarray of [1, 2, 1].
# The possible subarrays of [1, 2, 1] are [1, 2] and [2, 1].
# This means s_u != s_v for every edge (u, v).
# This is exactly the definition of a bipartite graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest assignment, we want s_1 = 1.
# If the graph has multiple connected components, for each component,
# we want the smallest possible values.
# For a component, if we fix one node to 1, the rest are determined.
# If we fix it to 2, the rest are also determined (the inverse).
# We want to choose the assignment that is lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try to color this component starting with color[i] = 1
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if starting with 1 is better than starting with 2
            # We want the smallest value at the smallest index in the component.
            # Let min_node be the smallest index in this component.
            # If color[min_node] is 2, we can flip all colors in this component.
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
