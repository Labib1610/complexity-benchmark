```python
import sys

# The condition for a link (u, v) to be stable is gcd(s_u + 2*s_v, 2*s_u + s_v) = 1.
# Possible values for s_i are {1, 2}.
# Case 1: s_u = 1, s_v = 1: gcd(1+2, 2+1) = gcd(3, 3) = 3 != 1. (Invalid)
# Case 2: s_u = 2, s_v = 2: gcd(2+4, 4+2) = gcd(6, 6) = 6 != 1. (Invalid)
# Case 3: s_u = 1, s_v = 2: gcd(1+4, 2+2) = gcd(5, 4) = 1. (Valid)
# Case 4: s_u = 2, s_v = 1: gcd(2+2, 4+1) = gcd(4, 5) = 1. (Valid)
# Thus, for every link (u, v), s_u must not equal s_v.
# This means the graph must be bipartite.
# To get the lexicographically smallest assignment, we greedily assign 1 to the 
# first node of each connected component, then propagate the bipartite coloring.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
