```python
import sys

# The condition for a link (u, v) to be balanced:
# A photon in a rectangle [0, s_u] x [0, s_v] starting at (0,0) at 45 degrees
# hits a corner after exactly one reflection.
# The path length in x is k * s_u, in y is l * s_v.
# The photon hits a corner when k * s_u = l * s_v.
# The number of reflections is (k-1) + (l-1) = k + l - 2.
# We need k + l - 2 = 1, so k + l = 3.
# Possible (k, l) pairs: (1, 2) or (2, 1).
# If (k, l) = (1, 2), then 1 * s_u = 2 * s_v.
# If (k, l) = (2, 1), then 2 * s_u = 1 * s_v.
# Since s_u, s_v in {1, 2}:
# If s_u = 1, s_v = 2: 1*1 = 2*2 is false, 2*1 = 1*2 is true.
# If s_u = 2, s_v = 1: 1*2 = 2*1 is true, 2*2 = 1*1 is false.
# If s_u = 1, s_v = 1: 1*1 = 2*1 (false), 2*1 = 1*1 (false).
# If s_u = 2, s_v = 2: 1*2 = 2*2 (false), 2*2 = 1*2 (false).
# Thus, s_u must be different from s_v.
# This is a standard 2-coloring problem (bipartite graph).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
