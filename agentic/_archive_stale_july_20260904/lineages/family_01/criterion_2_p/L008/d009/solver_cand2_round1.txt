import sys

# The equation is W + 2P = 9m
# W = sum_{i=1}^n d_i * (s_i - 3)^2
# P = sum_{(u,v) in E} s_u * s_v
# Let s_i = 1 or 2.
# (s_i - 3)^2 is 4 if s_i = 1, and 1 if s_i = 2.
# Let x_i = 1 if s_i = 1, and x_i = 0 if s_i = 2.
# Then s_i = 2 - x_i.
# (s_i - 3)^2 = (2 - x_i - 3)^2 = (-1 - x_i)^2 = (1 + x_i)^2 = 1 + 2x_i + x_i^2 = 1 + 3x_i.
# W = sum d_i (1 + 3x_i) = sum d_i + 3 * sum d_i * x_i = 2m + 3 * sum d_i * x_i.
# P = sum_{(u,v) in E} (2 - x_u)(2 - x_v) = sum (4 - 2x_u - 2x_v + x_u * x_v)
# P = 4m - 2 * sum d_i * x_i + sum_{(u,v) in E} x_u * x_v.
# W + 2P = (2m + 3 * sum d_i * x_i) + 2 * (4m - 2 * sum d_i * x_i + sum x_u * x_v)
# W + 2P = 2m + 3 * sum d_i * x_i + 8m - 4 * sum d_i * x_i + 2 * sum x_u * x_v
# W + 2P = 10m - sum d_i * x_i + 2 * sum x_u * x_v
# We want W + 2P = 9m, so:
# 10m - sum d_i * x_i + 2 * sum x_u * x_v = 9m
# sum d_i * x_i - 2 * sum x_u * x_v = m
# This is equivalent to: sum_{(u,v) in E} (x_u + x_v - 2 * x_u * x_v) = m
# Note that x_u + x_v - 2 * x_u * x_v is 1 if x_u != x_v, and 0 if x_u == x_v.
# This is exactly the definition of a bipartite graph.
# The equation holds if and only if the graph is bipartite.
# If bipartite, we can color the graph with 2 colors (x_i = 0 or 1).
# To get lexicographically smallest s_i, we want x_i to be 1 as much as possible.
# s_i = 2 - x_i. To make s_i small, we want x_i = 1.
# So we want to assign x_i = 1 to as many nodes as possible.
# In a bipartite graph, we can set all nodes in one partition to x_i = 1 and the other to x_i = 0.
# To make s_i lexicographically smallest, we check both partitions.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Two possible assignments:
    # 1. x_i = color[i] => s_i = 2 - color[i]
    # 2. x_i = 1 - color[i] => s_i = 2 - (1 - color[i]) = 1 + color[i]
    res1 = [2 - color[i] for i in range(1, n + 1)]
    res2 = [1 + color[i] for i in range(1, n + 1)]
    
    print("YES")
    if res1 < res2:
        print(*(res1))
    else:
        print(*(res2))

solve()
