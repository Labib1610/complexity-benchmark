import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically first assignment, we iterate 1 to n.
    # For each unassigned node, we try assigning 1 first.
    # If that fails, we try 2.
    
    def is_consistent(u, color, current_assignment):
        for v in adj[u]:
            if current_assignment[v] == color:
                return False
        return True

    def dfs(u, color, current_assignment):
        current_assignment[u] = color
        for v in adj[u]:
            if current_assignment[v] == 0:
                if not dfs(v, 3 - color, current_assignment):
                    return False
            elif current_assignment[v] == color:
                return False
        return True

    # We need to process components. Since we want lexicographical order,
    # we must be careful. The lexicographical requirement means we want
    # assignment[1] to be 1 if possible, then assignment[2] to be 1 if possible, etc.
    
    # This is a 2-coloring problem.
    # For each connected component, there are at most 2 ways to color it.
    # We pick the one that makes the smallest node in the component 1 if possible.
    
    visited = [False] * (n + 1)
    final_assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Try assigning 1 to the first node in the component
            temp_assignment = {node: 0 for node in component}
            
            def try_color(u, c, comp_assign):
                comp_assign[u] = c
                for v in adj[u]:
                    if comp_assign[v] == 0:
                        if not try_color(v, 3 - c, comp_assign):
                            return False
                    elif comp_assign[v] == c:
                        return False
                return True
            
            # Try setting the smallest node in component to 1
            start_node = min(component)
            comp_assign1 = {node: 0 for node in component}
            possible1 = try_color(start_node, 1, comp_assign1)
            
            comp_assign2 = {node: 0 for node in component}
            possible2 = try_color(start_node, 2, comp_assign2)
            
            if not possible1 and not possible2:
                print("NO")
                return
            
            if possible1 and possible2:
                # Compare lexicographically
                if comp_assign1[start_node] < comp_assign2[start_node]:
                    for node in component: final_assignment[node] = comp_assign1[node]
                else:
                    for node in component: final_assignment[node] = comp_assign2[node]
            elif possible1:
                for node in component: final_assignment[node] = comp_assign1[node]
            else:
                for node in component: final_assignment[node] = comp_assign2[node]

    print("YES")
    print(*(final_assignment[1:]))

solve()
