import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition "sum to 3" means if u is 1, v must be 2, and vice versa.
    # This is a standard 2-coloring problem on a graph.
    # To get the lexicographically smallest assignment, we want s_i = 1 whenever possible.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from i, try to assign 1 to i
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check if this component could have been smaller lexicographically
            # We already assigned 1 to the smallest index in the component.
            # Since we iterate 1 to n, the first node of any component is always
            # assigned 1, which is the smallest possible value.
            
    print("YES")
    print(*(color[1:]))

solve()
