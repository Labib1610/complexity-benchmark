```python
import sys

# The stability condition:
# f_u(x) + f_v(x) has exactly one local extremum.
# If both are f_1(x) = e^x, sum is 2e^x, no extremum.
# If both are f_2(x) = e^{-x}, sum is 2e^{-x}, no extremum.
# If one is f_1(x) = e^x and one is f_2(x) = e^{-x}, sum is e^x + e^{-x} = 2cosh(x).
# 2cosh(x) has a derivative 2sinh(x), which is 0 at x=0.
# The second derivative is 2cosh(x) > 0, so x=0 is a local minimum.
# Thus, a link is stable if and only if the two nodes have different basis functions.
# This is a bipartite graph coloring problem.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
        else:
            # A self-loop means a node must be different from itself, impossible.
            print("NO")
            return
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, we prefer 1 over 2.
            # Start BFS/DFS with color 1 for the smallest node in the component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
