import sys

# The condition for a link (u, v) to be stable is that T_u * T_v is self-adjoint.
# Let M_u be the matrix for operator u. T_u * T_v is self-adjoint if (M_u * M_v)^T = M_u * M_v.
# M_1 = [[1, 4], [2, 5]], M_2 = [[5, -4], [-2, 1]]
# M_1^T = [[1, 2], [4, 5]], M_2^T = [[5, -2], [-4, 1]]
#
# Check all combinations:
# 1. T_1 * T_1: (M_1 * M_1)^T = M_1^T * M_1^T = [[1, 2], [4, 5]] * [[1, 2], [4, 5]] = [[9, 12], [24, 33]]
#    M_1 * M_1 = [[1, 4], [2, 5]] * [[1, 4], [2, 5]] = [[9, 24], [12, 33]]
#    Not self-adjoint.
# 2. T_2 * T_2: (M_2 * M_2)^T = M_2^T * M_2^T = [[5, -2], [-4, 1]] * [[5, -2], [-4, 1]] = [[33, -12], [-24, 9]]
#    M_2 * M_2 = [[5, -4], [-2, 1]] * [[5, -4], [-2, 1]] = [[33, -24], [-12, 9]]
#    Not self-adjoint.
# 3. T_1 * T_2: (M_1 * M_2)^T = M_2^T * M_1^T = [[5, -2], [-4, 1]] * [[1, 2], [4, 5]] = [[-3, 0], [0, -3]]
#    M_1 * M_2 = [[1, 4], [2, 5]] * [[5, -4], [-2, 1]] = [[-3, 0], [0, -3]]
#    Self-adjoint!
# 4. T_2 * T_1: (M_2 * M_1)^T = M_1^T * M_2^T = [[1, 2], [4, 5]] * [[5, -2], [-4, 1]] = [[-3, 0], [0, -3]]
#    M_2 * M_1 = [[5, -4], [-2, 1]] * [[1, 4], [2, 5]] = [[-3, 0], [0, -3]]
#    Self-adjoint!
#
# Conclusion: A link (u, v) is stable if and only if u != v.
# This is a 2-coloring problem. We want the lexicographically smallest assignment.
# For each connected component, we can assign 1 to the first node and color accordingly.
# If a component is bipartite, there are two ways to color it. We pick the one that
# makes the first node in the component 1 if possible, or 2 if not.
# Actually, for each component, we can choose the coloring such that the smallest index
# node in the component gets 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # To minimize lexicographically, we want the smallest index in the component to be 1.
            # If color[min_node] is 2, we flip all colors in this component.
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
