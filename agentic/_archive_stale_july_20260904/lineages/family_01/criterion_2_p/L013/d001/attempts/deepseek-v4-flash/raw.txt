 We need to assign each node a value of 1 or 2 such that no edge connects two nodes with the same value.  
That’s exactly a bipartite graph check, and if it’s bipartite, we output the lexicographically smallest coloring (1 for the first partition, 2 for the second).  

Here’s the implementation:  

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n

def bfs(start):
    q = deque([start])
    color[start] = 0
    while q:
        u = q.popleft()
        for v in adj[u]:
            if color[v] == -1:
                color[v] = color[u] ^ 1
                q.append(v)
            elif color[v] == color[u]:
                return False
    return True

for i in range(n):
    if color[i] == -1:
        if not bfs(i):
            print("NO")
            sys.exit()

print("YES")
print(' '.join('1' if c == 0 else '2' for c in color))
```
