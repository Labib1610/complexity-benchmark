```python
import sys
from collections import deque

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    iterator = iter(input_data)
    n = int(next(iterator))
    m = int(next(iterator))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # We want to find the lexicographically smallest valid configuration.
    # A configuration is valid if no edge connects two 1s or two 2s.
    # This means for every edge (u, v), s[u] != s[v].
    # This is a graph 2-coloring problem.
    # To find the lexicographically smallest valid coloring:
    # We iterate through vertices 0 to n-1.
    # If a vertex is not yet colored, we try to color it with 1.
    # Then we must propagate constraints: neighbors must be 2, their neighbors 1, etc.
    # This is done via BFS/DFS.
    
    color = [0] * n
    possible = True
    
    for i in range(n):
        if color[i] == 0:
            # Start a BFS for this connected component
            queue = deque([i])
            color[i] = 1
            
            while queue:
                u = queue.popleft()
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        possible = False
                        break
                if not possible:
                    break
        if not possible:
            break
            
    if not possible:
        print("NO")
    else:
        print("YES")
        print(*(c for c in color))

if __name__ == '__main__':
    solve()
```
