import sys

# The problem states that for each comparator (u, v), the product of switch positions s_u * s_v must be 2.
# Since s_i is either 1 or 2, the only way s_u * s_v = 2 is if one is 1 and the other is 2.
# This is a bipartite coloring problem.
# If s_u = 1, then s_v must be 2. If s_u = 2, then s_v must be 1.
# This is equivalent to 2-coloring a graph where edges represent constraints.
# If the graph is bipartite, we can color each connected component with two colors (1 and 2).
# To get the lexicographically smallest configuration, for each component, we decide
# the color of the smallest index node. If we set the smallest index node to 1,
# the rest of the component is determined.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make it lexicographically smallest, we want the smallest index
            # in the component to be 1 if possible.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a list to keep track of nodes in this component
            q = [i]
            color[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if this is the best assignment for this component.
            # The current assignment has color[i] = 1.
            # If we swapped colors (1->2, 2->1), the configuration would be different.
            # We want the smallest index in the component to have the smallest possible value.
            # Since we started with the smallest index i in the component, 
            # and we set color[i] = 1, this is already the smallest possible value for i.
            
    print("YES")

solve()
