```python
import sys

# Let M1 be the transformation:
# If (x+y) is even: (x, y) -> (x, y+1)
# If (x+y) is odd: (x, y) -> (x+1, y)
# Let M2 be the transformation:
# If (x+y) is even: (x, y) -> (x-1, y)
# If (x+y) is odd: (x, y) -> (x, y-1)

# Let's trace M1 then M1:
# Start (x, y).
# If x+y even: (x, y) -> (x, y+1). New sum x+y+1 is odd.
# Then (x, y+1) -> (x+1, y+1).
# If x+y odd: (x, y) -> (x+1, y). New sum x+y+1 is even.
# Then (x+1, y) -> (x+1, y+1).
# M1 followed by M1 is (x, y) -> (x+1, y+1). Not identity.

# Let's trace M2 then M2:
# Start (x, y).
# If x+y even: (x, y) -> (x-1, y). New sum x+y-1 is odd.
# Then (x-1, y) -> (x-1, y-1).
# If x+y odd: (x, y) -> (x, y-1). New sum x+y-1 is even.
# Then (x, y-1) -> (x-1, y-1).
# M2 followed by M2 is (x, y) -> (x-1, y-1). Not identity.

# Let's trace M1 then M2:
# Start (x, y).
# If x+y even: (x, y) -> (x, y+1). New sum x+y+1 is odd.
# Then (x, y+1) -> (x, y+1-1) = (x, y).
# If x+y odd: (x, y) -> (x+1, y). New sum x+y+1 is even.
# Then (x+1, y) -> (x+1-1, y) = (x, y).
# M1 followed by M2 is identity.

# Let's trace M2 then M1:
# Start (x, y).
# If x+y even: (x, y) -> (x-1, y). New sum x+y-1 is odd.
# Then (x-1, y) -> (x-1+1, y) = (x, y).
# If x+y odd: (x, y) -> (x, y-1). New sum x+y-1 is even.
# Then (x, y-1) -> (x, y-1+1) = (x, y).
# M2 followed by M1 is identity.

# Conclusion: A link (u, v) is stable if and only if module(u) != module(v).
# This is a 2-coloring problem on a graph.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)

    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
