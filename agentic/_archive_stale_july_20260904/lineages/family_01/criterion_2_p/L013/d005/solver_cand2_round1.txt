```python
import sys

# The condition T_u(T_v(x)) = 5 - x:
# Protocol 1: T(x) = x
# Protocol 2: T(x) = 5 - x
# T1(T1(x)) = x != 5 - x
# T1(T2(x)) = 5 - x (Valid)
# T2(T1(x)) = 5 - x (Valid)
# T2(T2(x)) = 5 - (5 - x) = x != 5 - x
# Thus, for every tether (u, v), T_u must be different from T_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence of 1s and 2s.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we try assigning 1 first.
    for i in range(1, n + 1):
        if color[i] == 0:
            # BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
