```python
import sys

# The condition: (s_u * s_v) + (s_u + s_v) = 5
# Possible values for s_u, s_v are {1, 2}.
# If s_u = 1, s_v = 1: (1*1) + (1+1) = 3 != 5
# If s_u = 1, s_v = 2: (1*2) + (1+2) = 5 == 5 (Balanced)
# If s_u = 2, s_v = 1: (2*1) + (2+1) = 5 == 5 (Balanced)
# If s_u = 2, s_v = 2: (2*2) + (2+2) = 8 != 5
# Thus, for every tether (u, v), s_u must be different from s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest assignment (1s before 2s).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we prefer 1 over 2.
    # For each connected component, we try assigning 1 to the first node.
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
