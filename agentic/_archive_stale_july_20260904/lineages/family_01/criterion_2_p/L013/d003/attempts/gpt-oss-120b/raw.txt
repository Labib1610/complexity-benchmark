import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # 0 / 1 bipartite color
    res = [0] * (n + 1)             # final values 1 or 2
    visited = [False] * (n + 1)

    for start in range(1, n + 1):
        if visited[start]:
            continue
        # BFS this component
        q = deque([start])
        visited[start] = True
        color[start] = 0
        comp_nodes = [start]
        min_node = start
        ok = True
        while q:
            u = q.popleft()
            for v in adj[u]:
                if not visited[v]:
                    visited[v] = True
                    color[v] = color[u] ^ 1
                    q.append(v)
                    comp_nodes.append(v)
                    if v < min_node:
                        min_node = v
                else:
                    if color[v] == color[u]:
                        print("NO")
                        return
        # decide orientation so that min_node gets value 1
        if color[min_node] == 0:
            # map 0 -> 1, 1 -> 2
            for node in comp_nodes:
                res[node] = 1 if color[node] == 0 else 2
        else:
            # map 0 -> 2, 1 -> 1
            for node in comp_nodes:
                res[node] = 2 if color[node] == 0 else 1

    print("YES")
    print(' '.join(str(res[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
