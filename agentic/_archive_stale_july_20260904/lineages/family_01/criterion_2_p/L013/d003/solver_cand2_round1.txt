```python
import sys

# The problem states that for each bridge (u, v), the sum of energy states squared must be 5.
# Energy states are 1 or 2. 1^2 = 1, 2^2 = 4.
# 1 + 4 = 5. So, for every bridge (u, v), one qubit must be 1 and the other must be 2.
# This is equivalent to a 2-coloring problem on a graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest configuration, for each connected component,
# we try assigning 1 to the smallest index node and propagate the coloring.
# If a conflict arises, it's impossible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we want the first node in the component
            # to be 1 if possible.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

solve()
```
