```python
import sys

# The condition: (s_u XOR s_v) > (s_u AND s_v)
# Possible values for s_u, s_v are {1, 2}.
# Case 1: s_u = 1, s_v = 1 => XOR = 0, AND = 1. 0 > 1 is False.
# Case 2: s_u = 1, s_v = 2 => XOR = 3, AND = 0. 3 > 0 is True.
# Case 3: s_u = 2, s_v = 1 => XOR = 3, AND = 0. 3 > 0 is True.
# Case 4: s_u = 2, s_v = 2 => XOR = 0, AND = 2. 0 > 2 is False.
#
# Conclusion: s_u must be different from s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To minimize lexicographically, we want s_i = 1 whenever possible.
# For each connected component, we can pick the color of the first node
# (the one with the smallest index) to be 1, then propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To be lexicographically smallest, we prefer color 1 for the smallest index
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
