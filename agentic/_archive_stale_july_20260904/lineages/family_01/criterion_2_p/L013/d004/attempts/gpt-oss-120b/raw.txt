```python
import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)

    color = [-1] * (n + 1)          # temporary 0/1 colors
    answer = [0] * (n + 1)          # final 1/2 values

    for start in range(1, n + 1):
        if color[start] != -1:
            continue
        # BFS this component
        q = deque([start])
        color[start] = 0
        comp = [start]
        bip = True
        while q:
            u = q.popleft()
            cu = color[u]
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = cu ^ 1
                    q.append(v)
                    comp.append(v)
                elif color[v] == cu:
                    bip = False
                    break
            if not bip:
                break
        if not bip:
            print("NO")
            return

        # decide orientation: make smallest index node have final value 1
        min_node = min(comp)
        if color[min_node] == 1:          # need to flip
            for v in comp:
                color[v] ^= 1

        # set final answer
        for v in comp:
            answer[v] = 1 if color[v] == 0 else 2

    print("YES")
    print(' '.join(str(answer[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
```
