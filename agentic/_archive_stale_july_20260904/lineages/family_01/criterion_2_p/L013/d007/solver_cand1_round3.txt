import sys

# The condition f_u(f_v(V)) = V for all V is equivalent to M_u * M_v = I.
# M1 = [[2, 1], [1, 1]], M2 = [[1, -1], [-1, 2]]
# M1 * M1 = [[5, 3], [3, 2]] != I
# M2 * M2 = [[2, -3], [-3, 5]] != I
# M1 * M2 = [[2*1 + 1*-1, 2*-1 + 1*2], [1*1 + 1*-1, 1*-1 + 1*2]] = [[1, 0], [0, 1]] = I
# M2 * M1 = [[1*2 + -1*1, 1*1 + -1*1], [-1*2 + 2*1, -1*1 + 2*1]] = [[1, 0], [0, 1]] = I
# So, M_u * M_v = I if and only if (u, v) are different transformations.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 1 and 2.
# To minimize lexicographically, we want s_1 = 1.
# If node 1 is in a connected component, its color is fixed by s_1 = 1.
# For each component, we check if it's bipartite. If it is, we assign 1 to the
# partition containing the smaller index node if possible, but the problem
# asks for the lexicographically smallest sequence s_1, s_2, ..., s_n.
# This means we want s_1=1, then s_2=1 if possible, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # To make s lexicographically smallest, we greedily pick 1 for the
            # first uncolored node in the component.
            stack = [(i, 1)]
            color[i] = 1
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # After coloring, check if we can flip the colors to make it lexicographically smaller
            # The current assignment is one valid coloring. The other is flipping 1<->2.
            # We want the smallest value at the smallest index in the component.
            # Let min_node be the smallest index in this component.
            # If color[min_node] == 2, we flip all colors in this component.
            min_node = min(node for node, c in component)
            if color[min_node] == 2:
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
