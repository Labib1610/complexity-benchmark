```python
import sys

# The interference polynomial is H(x) = (x - k_u)(x - k_v) = x^2 - (k_u + k_v)x + k_u * k_v.
# The derivative is H'(x) = 2x - (k_u + k_v).
# H'(0) = -(k_u + k_v).
# We require H'(0) = -3, so -(k_u + k_v) = -3, which means k_u + k_v = 3.
# Since k_i \in {1, 2}, the only way k_u + k_v = 3 is if one is 1 and the other is 2.
# This means for every edge (u, v), k_u != k_v.
# This is equivalent to checking if the graph is bipartite.
# To find the lexicographically smallest assignment, we iterate through nodes 1 to n.
# If a node is not visited, we assign it 1 (the smallest possible value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                next_color = 3 - color
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = next_color
                        stack.append((v, next_color))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
