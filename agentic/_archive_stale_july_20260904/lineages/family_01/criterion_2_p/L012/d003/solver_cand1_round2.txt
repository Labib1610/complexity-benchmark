```python
import sys

# The condition det(M_{u,v}) = 0 means (s_u + s_v)^2 - 9 = 0.
# So (s_u + s_v)^2 = 9, which implies s_u + s_v = 3 (since s_u, s_v in {1, 2}, s_u + s_v > 0).
# This means for every bridge (u, v), s_u + s_v = 3.
# If s_u = 1, then s_v = 2. If s_u = 2, then s_v = 1.
# This is a bipartite coloring problem.
# We want the lexicographically smallest sequence (s_1, s_2, ..., s_n).
# This means we prefer s_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unvisited, 1 = state 1, 2 = state 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try setting s_i = 1
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        new_c = 3 - c
                        color[v] = new_c
                        stack.append((v, new_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
