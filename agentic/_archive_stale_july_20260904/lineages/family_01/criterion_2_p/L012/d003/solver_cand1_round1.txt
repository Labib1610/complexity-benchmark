```python
import sys

# The constraint is det([[s_u, 1], [2, s_v]]) = s_u * s_v - 2 = 0.
# Since s_i in {1, 2}, s_u * s_v = 2.
# The only way to get 2 is (s_u=1, s_v=2) or (s_u=2, s_v=1).
# This means for every edge (u, v), s_u != s_v.
# This is a bipartite graph coloring problem.
# We want the lexicographically smallest sequence (s_1, s_2, ..., s_n).
# To achieve this, we greedily set s_i = 1 if possible, else s_i = 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we prefer color 1 for the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
