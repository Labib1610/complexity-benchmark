import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The constraint is s_u = 2*s_v or s_v = 2*s_u.
    # Since s_i in {1, 2}, this means one must be 1 and the other 2.
    # This is a bipartite coloring problem.
    
    color = [0] * (n + 1) # 0: unassigned, 1: amplitude 1, 2: amplitude 2
    
    # To get lexicographically smallest, we want s_i = 1 whenever possible.
    # We iterate 1 to n. If uncolored, try assigning 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                target_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target_c
                        stack.append((v, target_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
