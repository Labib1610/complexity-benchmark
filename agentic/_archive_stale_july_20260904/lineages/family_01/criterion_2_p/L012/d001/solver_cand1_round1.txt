```python
import sys

# The problem asks to determine if a graph can be 2-colored such that 
# for every vertex, the number of incident edges assigned to channel 1 
# and channel 2 differ by at most 1.
# This is equivalent to finding an Eulerian orientation or path decomposition.
# For a connected component with degrees d_v:
# 1. If all d_v are even, we can always satisfy the condition (balance 0).
# 2. If there are vertices with odd degrees, we must have an even number of them.
#    We can pair them up and add virtual edges to make all degrees even.
#    The condition "differ by at most 1" is satisfied if and only if 
#    for every vertex v, |deg1(v) - deg2(v)| <= 1.
#    This is possible if and only if the graph is connected and 
#    the number of odd-degree vertices is at most 2 per component.
#    Actually, for any graph, such an assignment exists if and only if 
#    every connected component has at most 2 vertices of odd degree.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    idx = 0
    n = int(input_data[idx])
    m = int(input_data[idx+1])
    idx += 2
    
    adj = [[] for _ in range(n + 1)]
    degree = [0] * (n + 1)
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        degree[u] += 1
        degree[v] += 1
        
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i] and degree[i] > 0:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            odd_count = sum(1 for v in component if degree[v] % 2 != 0)
            if odd_count > 2:
                print("NO")
                return

    print("YES")

solve()
```
