import sys

# The constraint is s_u & s_v = 0 where s_i in {1, 2}.
# 1 is binary 01, 2 is binary 10.
# 1 & 1 = 1 (not 0)
# 2 & 2 = 2 (not 0)
# 1 & 2 = 0 (valid)
# 2 & 1 = 0 (valid)
# This means if there is a link between u and v, s_u must be different from s_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
# For each connected component, we have two possible colorings (swap 1s and 2s).
# We pick the one that makes the first node in the component 1 if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Use a list to keep track of nodes in this component
            # to check if we can make the first node 1
            idx_comp = 0
            q = [i]
            while idx_comp < len(q):
                u = q[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # After coloring, check if color[i] is 1. If not, flip.
            if color[i] == 2:
                for node in q:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
