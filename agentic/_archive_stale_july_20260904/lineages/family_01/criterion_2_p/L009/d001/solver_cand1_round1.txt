```python
import sys

# The problem asks us to assign each staff member to shift 1 or 2 such that
# for every tool (u, v), there is at least one staff member in shift 1
# AND at least one staff member in shift 2.
# Let s_i be the shift of staff i (1 or 2).
# Condition: For each tool (u, v):
# 1. At least one of {u, v} is in shift 1 => NOT (s_u == 2 AND s_v == 2)
# 2. At least one of {u, v} is in shift 2 => NOT (s_u == 1 AND s_v == 1)
# This means s_u != s_v for all tools (u, v).
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 2 colors.
# To get the lexicographically smallest assignment, we iterate 1 to n.
# If a node is unvisited, we assign it to shift 1 (the smaller value)
# and perform BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    shift = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if shift[i] == 0:
            # Start BFS/DFS to color component
            # To get lexicographically smallest, we prefer shift 1
            stack = [(i, 1)]
            shift[i] = 1
            while stack:
                u, s = stack.pop()
                next_s = 3 - s
                for v in adj[u]:
                    if shift[v] == 0:
                        shift[v] = next_s
                        stack.append((v, next_s))
                    elif shift[v] == s:
                        print("NO")
                        return
    
    print("YES")
    print(*(shift[1:]))

solve()
```
