```python
import sys

# The problem asks us to assign each staff member to schedule 1 or 2.
# A tool is owned by (u, v). During any schedule, exactly one owner must be present.
# If u and v are in the same schedule, then during that schedule, both are present (2 owners).
# If u and v are in different schedules, then during schedule 1, only the one in schedule 1 is present,
# and during schedule 2, only the one in schedule 2 is present.
# Thus, for every tool (u, v), u and v must be in different schedules.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not visited, we assign it to schedule 1 (the smaller value) and perform BFS/DFS.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer 1 over 2.
            stack = [(i, 1)]
            assignment[i] = 1
            while stack:
                u, color = stack.pop()
                next_color = 2 if color == 1 else 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = next_color
                        stack.append((v, next_color))
                    elif assignment[v] == color:
                        print("NO")
                        return
    
    print("YES")
    print(*(assignment[1:]))

solve()
```
