import sys

# The problem asks for an assignment of n staff members to routine 1 or 2.
# For each machine (u, v), we need:
# 1. Calibration (Routine 1): At least one of {u, v} is in Routine 1.
# 2. Production (Routine 2): At least one of {u, v} is in Routine 2.
# Let x_i = 0 if staff i is in Routine 1, x_i = 1 if staff i is in Routine 2.
# Calibration: (x_u == 0) OR (x_v == 0) => NOT (x_u == 1 AND x_v == 1)
# Production: (x_u == 1) OR (x_v == 1) => NOT (x_u == 0 AND x_v == 0)
# This means for every edge (u, v), x_u != x_v.
# This is equivalent to saying the graph must be bipartite.
# If the graph is bipartite, we can color it with 2 colors.
# Let the colors be 0 and 1.
# If we assign color 0 to Routine 1 and color 1 to Routine 2,
# then for every edge (u, v), one is 0 and one is 1.
# This satisfies both conditions.
# To get the lexicographically smallest assignment, we want to assign 1 to as many
# early indices as possible, but we must respect the bipartite coloring.
# For each connected component, there are two possible colorings (swap 0 and 1).
# We choose the one that makes the assignment lexicographically smallest.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [-1] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            component = []
            stack = [(i, 0)]
            color[i] = 0
            component.append(i)
            
            idx_stack = 0
            while idx_stack < len(component):
                u = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two choices: color 0 -> 1, color 1 -> 2 OR color 0 -> 2, color 1 -> 1
            # We want the lexicographically smallest sequence.
            # Find the first node in the component.
            first_node = min(component)
            
            # Option 1: color[first_node] -> 1, 1-color[first_node] -> 2
            # Option 2: color[first_node] -> 2, 1-color[first_node] -> 1
            
            # Compare the assignments for the first node
            # We want ans[first_node] to be 1 if possible.
            if color[first_node] == 0:
                # Option 1: 0->1, 1->2
                # Option 2: 0->2, 1->1
                # To make ans[first_node] = 1, choose Option 1
                for node in component:
                    ans[node] = 1 if color[node] == 0 else 2
            else:
                # Option 1: 1->1, 0->2
                # Option 2: 1->2, 0->1
                # To make ans[first_node] = 1, choose Option 1
                for node in component:
                    ans[node] = 1 if color[node] == 1 else 2
                    
    print("YES")
    print(*(ans[1:]))

solve()
