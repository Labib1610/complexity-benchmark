```python
import sys

# The game described is Wythoff's Game.
# A position (c_u, c_v) is a losing position (P-position) if and only if
# the state is (floor(n * phi), floor(n * phi^2)) for some non-negative integer n,
# where phi = (1 + sqrt(5)) / 2.
# The possible channel values are {1, 2}.
# The pairs (c_u, c_v) can be (1, 1), (1, 2), (2, 1), or (2, 2).
# Wythoff's P-positions (n, m) with n <= m:
# (0, 0), (1, 2), (3, 5), (4, 7), ...
# For channels {1, 2}, the only P-position is (1, 2) or (2, 1).
# Thus, for every conflict pair (u, v), we must have c_u != c_v.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with 2 colors (1 and 2).
# To get the lexicographically smallest assignment, we use BFS/DFS
# to color each connected component. For each component, we try assigning
# color 1 to the first node encountered.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for component
            # To get lexicographically smallest, we want color[i] = 1
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
