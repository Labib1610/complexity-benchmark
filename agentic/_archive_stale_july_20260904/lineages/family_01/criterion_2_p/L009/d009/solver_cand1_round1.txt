```python
import sys

# The condition: two circles of radii c_u and c_v with distance 3 between centers
# are externally tangent if and only if c_u + c_v = 3.
# Since c_i is in {1, 2}, this means for every edge (u, v),
# c_u + c_v = 3.
# This implies that if c_u = 1, then c_v = 2, and if c_u = 2, then c_v = 1.
# This is a bipartite coloring problem.
# We need to color the graph with 1s and 2s such that every edge connects
# different values.
# To get the lexicographically smallest assignment, we want to assign 1
# to nodes whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is not colored, we try assigning 1.
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
