import sys
sys.setrecursionlimit(1 << 25)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # 0: unvisited, 1: color 1, 2: color 2
    color = [0] * n
    
    # Process components
    for i in range(n):
        if color[i] == 0:
            # BFS/DFS to color the component
            # We want lexicographically smallest, so we try to assign 1 first
            # But we must check if assigning 1 is valid
            # Actually, we can just assign 1 to the start node and propagate
            # If conflict, try assigning 2 to start node
            # If still conflict, NO
            
            # Let's try assigning 1 to node i
            stack = [i]
            color[i] = 1
            valid = True
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        # Assign opposite color
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        valid = False
                        break
                if not valid:
                    break
            
            if valid:
                continue
            
            # Try assigning 2 to node i
            # Reset colors in this component
            # We need to track which nodes were colored
            # Let's use a different approach: reset and retry
            stack = [i]
            color[i] = 2
            valid = True
            visited_in_component = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                        visited_in_component.append(v)
                    elif color[v] == color[u]:
                        valid = False
                        break
                if not valid:
                    break
            
            if not valid:
                print("NO")
                return
    
    print("YES")
    print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
