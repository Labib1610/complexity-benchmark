```python
import sys

# The condition for a right-angled triangle with sides a, b, c:
# a^2 + b^2 = c^2 or a^2 + c^2 = b^2 or b^2 + c^2 = a^2
# Let f(p) = 23*p + 10.
# f(1) = 33, f(2) = 56.
# Possible side lengths are 33, 56, 65.
# Check: 33^2 + 56^2 = 1089 + 3136 = 4225 = 65^2.
# So a triangle with sides 33, 56, 65 is a right-angled triangle.
# For a bridge (u, v) to be balanced, the set {f(p_u), f(p_v), 65} must be {33, 56, 65}.
# This means one of {f(p_u), f(p_v)} must be 33 and the other must be 56.
# Thus, p_u != p_v.
# The problem reduces to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment (p_1, ..., p_n).
# To get the lexicographically smallest, we prefer p_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == 0:
            # To get lexicographically smallest, try assigning 1 to the first node of each component
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
