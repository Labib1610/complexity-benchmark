```python
import sys

# The problem asks to assign 1 or 2 to each node such that for every edge (u, v),
# weight(u) + weight(v) = 3.
# This means if weight(u) = 1, then weight(v) must be 2, and vice versa.
# This is equivalent to checking if the graph is bipartite.
# If it is bipartite, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for node 1, then node 2, etc.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this connected component
            # To get lexicographically smallest, we try assigning 1 to the first node
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - color
                        stack.append((v, assignment[v]))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
```
