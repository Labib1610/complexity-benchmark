```python
import sys

# The condition for non-degenerate triangle with vertices (c_u, c_v), (c_v, c_u), (1, 1):
# Area = 0.5 * |x1(y2 - y3) + x2(y3 - y1) + x3(y1 - y2)|
# Area = 0.5 * |c_u(c_u - 1) + c_v(1 - c_v) + 1(c_v - c_u)|
# Area = 0.5 * |c_u^2 - c_u + c_v - c_v^2 + c_v - c_u|
# Area = 0.5 * |c_u^2 - c_v^2 - 2c_u + 2c_v|
# Area = 0.5 * |(c_u - c_v)(c_u + c_v) - 2(c_u - c_v)|
# Area = 0.5 * |(c_u - c_v)(c_u + c_v - 2)|
# For non-degenerate, Area != 0, so (c_u - c_v)(c_u + c_v - 2) != 0.
# This means c_u != c_v AND c_u + c_v != 2.
# Since c_i in {1, 2}:
# If c_u = 1, c_v = 1: c_u + c_v = 2 (Degenerate)
# If c_u = 2, c_v = 2: c_u - c_v = 0 (Degenerate)
# If c_u = 1, c_v = 2: c_u - c_v = -1, c_u + c_v - 2 = 1. Product = -1 != 0 (Valid)
# If c_u = 2, c_v = 1: c_u - c_v = 1, c_u + c_v - 2 = 1. Product = 1 != 0 (Valid)
# Thus, the condition is simply c_u != c_v.
# This is a standard 2-coloring problem (bipartite graph check).

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    colors = [0] * (n + 1)
    for i in range(1, n + 1):
        if colors[i] == 0:
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, colors[v]))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
