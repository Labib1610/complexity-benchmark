import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    d = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        d[u] += 1
        d[v] += 1
        
    # The energy formula simplifies to:
    # E = sum_{i=1}^n (W_i - D_i * (3 - r_i))^2 - sum_{i=1}^n D_i^2
    # E = 0 iff sum_{i=1}^n (W_i - D_i * (3 - r_i))^2 = sum_{i=1}^n D_i^2
    # Let x_i = 3 - r_i. Since r_i in {1, 2}, x_i in {1, 2}.
    # W_i = sum_{j in neighbors(i)} r_j = sum_{j in neighbors(i)} (3 - x_j)
    # W_i = 3 * D_i - sum_{j in neighbors(i)} x_j
    # W_i - D_i * x_i = 3 * D_i - sum_{j in neighbors(i)} x_j - D_i * x_i
    # This looks like a constraint satisfaction problem.
    # Actually, the condition simplifies to:
    # sum_{i=1}^n (sum_{j in neighbors(i)} x_j + D_i * x_i - 3 * D_i)^2 = sum D_i^2
    # This is satisfied if for all i: sum_{j in neighbors(i)} x_j + D_i * x_i = 3 * D_i
    # Let y_i = x_i - 1. Then y_i in {0, 1}.
    # sum_{j in neighbors(i)} (y_j + 1) + D_i * (y_i + 1) = 3 * D_i
    # sum_{j in neighbors(i)} y_j + D_i + D_i * y_i + D_i = 3 * D_i
    # sum_{j in neighbors(i)} y_j + D_i * y_i = D_i
    
    # We need to find y_i in {0, 1} such that for all i:
    # sum_{j in neighbors(i)} y_j + D_i * y_i = D_i
    # This is a system of linear equations over GF(2) if we consider the structure,
    # but it's actually simpler: if D_i > 0, y_i is determined by neighbors.
    # If D_i = 0, the equation is 0 = 0, so y_i can be 0 (r_i = 2).
    
    y = [-1] * (n + 1)
    
    def dfs(u, val):
        y[u] = val
        for v in adj[u]:
            if y[v] == -1:
                # sum_{j in neighbors(v)} y_j + D_v * y_v = D_v
                # y_u + sum_{j in neighbors(v), j!=u} y_j + D_v * y_v = D_v
                # This is not a simple propagation.
                pass
    
    # The system sum_{j in neighbors(i)} y_j + D_i * y_i = D_i
    # is equivalent to: for every edge (u, v), y_u + y_v is not necessarily constant.
    # Actually, if we set y_i = 1 for all i, sum y_j + D_i * y_i = D_i + D_i = 2*D_i != D_i.
    # If we set y_i = 0 for all i, 0 = D_i. Only works if D_i = 0.
    # The only solution is y_i = 1 if the graph is bipartite and we color it.
    # Let's check if y_i = 1 for one set and 0 for another works.
    
    import collections
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1 and d[i] > 0:
            q = collections.deque([i])
            color[i] = 0
            while q:
                u = q.popleft()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        q.append(v)
    
    # Check if color works: sum_{j in neighbors(i)} color[j] + D_i * color[i] == D_i
    # If bipartite, sum_{j in neighbors(i)} color[j] = D_i * (1 - color[i])
    # So D_i * (1 - color[i]) + D_i * color[i] = D_i * 1 = D_i. Yes!
    
    res = []
    for i in range(1, n + 1):
        # y_i = color[i] if color[i] != -1 else 0
        # r_i = 3 - (y_i + 1) = 2 - y_i
        val = color[i] if color[i] != -1 else 0
        res.append(2 - val)
        
    # Verify
    for i in range(1, n + 1):
        w_i = sum(res[neighbor-1] for neighbor in adj[i])
        if w_i + d[i] * res[i-1] - 3 * d[i] != 0:
            print("NO")
            return
            
    print("YES")
    print(*(res))

solve()
