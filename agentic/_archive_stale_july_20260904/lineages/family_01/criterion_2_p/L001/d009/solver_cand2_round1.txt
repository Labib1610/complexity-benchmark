import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    deg = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        deg[u] += 1
        deg[v] += 1
        
    # Let x_i = r_i - 1, so x_i in {0, 1}.
    # r_i = x_i + 1.
    # P = sum D_i (x_i + 1) = sum D_i x_i + sum D_i = sum D_i x_i + 2m.
    # P = 3m => sum D_i x_i = m.
    # R = sum (x_u + 1)(x_v + 1) = sum (x_u x_v + x_u + x_v + 1)
    # R = sum x_u x_v + sum (x_u + x_v) + m = sum x_u x_v + sum D_i x_i + m
    # R = 2m => sum x_u x_v + m + m = 2m => sum x_u x_v = 0.
    # sum x_u x_v = 0 means for every edge (u, v), x_u * x_v = 0.
    # This means we cannot have both x_u = 1 and x_v = 1.
    # This is equivalent to saying the set of nodes with x_i = 1 must be an independent set.
    # Let S be the set of nodes where x_i = 1.
    # Condition 1: sum_{i in S} D_i = m.
    # Condition 2: S is an independent set.
    
    # We want lexicographically smallest r, which means smallest x.
    # This is equivalent to finding an independent set S such that sum_{i in S} D_i = m.
    # Since we want lexicographically smallest, we want to pick 0 for as many early nodes as possible.
    # This is equivalent to picking the smallest possible set S.
    
    # This is a variation of the subset sum problem on an independent set.
    # However, the graph might be disconnected.
    # For each connected component, we can find all possible sums of degrees of independent sets.
    # Since we want the lexicographically smallest sequence, we should decide x_1, x_2, ... greedily.
    
    # Actually, the condition "sum_{i in S} D_i = m" with S being an independent set
    # is equivalent to partitioning the graph into independent sets.
    # Wait, the problem is simpler: we need to find if there exists an independent set S with sum of degrees = m.
    
    # Let's use DP. For each component, find all possible sums of degrees of independent sets.
    # Since n, m <= 2e5, this is only feasible if the number of independent sets is small or the structure is simple.
    # But wait, the constraints are 2e5. This suggests a linear or O(n log n) approach.
    
    # Is it possible that the only way to satisfy sum D_i x_i = m is to have a specific structure?
    # If the graph is bipartite, we can pick one partition.
    # Actually, for any graph, if we pick S as one of the color classes of a bipartite graph,
    # sum_{i in S} D_i = m.
    # If the graph is bipartite, we can always pick one of the two color classes.
    # If the graph is not bipartite, we might not be able to satisfy it.
    
    # Let's check if the graph is bipartite.
    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # If bipartite, we have two sets S0 = {i | color[i] == 0} and S1 = {i | color[i] == 1}.
    # sum_{i in S0} D_i = m and sum_{i in S1} D_i = m.
    # We want the lexicographically smallest r.
    # r_i = 1 if x_i = 0, r_i = 2 if x_i = 1.
    # We want to minimize the number of 2s, and prioritize 1s at the beginning.
    # We can choose S0 or S1 to be the set of nodes with r_i = 2.
    # Compare the sequences generated by S0 and S1.
    
    res0 = [1] * n
    for i in range(1, n + 1):
        if color[i] == 0: res0[i-1] = 2
    
    res1 = [1] * n
    for i in range(1, n + 1):
        if color[i] == 1: res1[i-1] = 2
        
    if res0 < res1:
        print("YES")
        print(*(res0))
    else:
        print("YES")
        print(*(res1))

solve()
