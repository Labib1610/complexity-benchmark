```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the colors are different (1 and 2). This is equivalent to checking if the
# graph is bipartite. To get the lexicographically smallest sequence, we
# iterate from 1 to n and assign 1 if possible, otherwise 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we want c_i = 1 whenever possible.
    # For each connected component, we try assigning 1 to the smallest node.
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS from node i, assigning 1
            stack = [i]
            colors[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    solve()
```
