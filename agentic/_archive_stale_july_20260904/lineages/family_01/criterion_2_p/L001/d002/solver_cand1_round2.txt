```python
import sys

# The condition: for a pair (u, v), if they are incompatible,
# their assigned values s_u, s_v in {1, 2} must satisfy:
# s_u^3 + s_v^3 = 9
# Possible values for s^3 are 1^3 = 1 and 2^3 = 8.
# 1 + 8 = 9.
# This means for every incompatible pair (u, v), one must be 1 and the other must be 2.
# This is equivalent to a bipartite coloring problem where each edge (u, v)
# forces s_u != s_v.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    # To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
    # We iterate 1 to n. If not colored, start BFS/DFS.
    # For lexicographical smallest, we prefer color 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
```
