import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            # If a worker is paired with themselves, 1^3 + 1^3 = 2 != 9, 2^3 + 2^3 = 16 != 9.
            # Impossible.
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The rule: x^3 + y^3 = 9 where x, y in {1, 2}.
    # 1^3 + 1^3 = 2
    # 1^3 + 2^3 = 1 + 8 = 9
    # 2^3 + 1^3 = 8 + 1 = 9
    # 2^3 + 2^3 = 16
    # This means for every pair (u, v), one must be 1 and the other must be 2.
    # This is a bipartite coloring problem.
    
    color = [0] * (n + 1)
    # To get the lexicographically smallest sequence, we prefer 1 over 2.
    # We iterate 1 to n, if not colored, try coloring with 1.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                target = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = target
                        stack.append((v, target))
                    elif color[v] != target:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
