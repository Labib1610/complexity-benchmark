import sys
from collections import defaultdict

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    degree = [0] * (n + 1)
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        edges.append((u, v))
        degree[u] += 1
        degree[v] += 1
        index += 2
    
    # We want E = 2m
    # E = sum_i (6 * D_i * r_i^2) - sum_j (14 * r_{u_j} * r_{v_j})
    # Let S1 = sum_i (6 * D_i * r_i^2)
    # Let S2 = sum_j (14 * r_{u_j} * r_{v_j})
    # E = S1 - S2 = 2m
    
    # Since r_i is either 1 or 2, r_i^2 is either 1 or 4.
    # So 6*D_i*r_i^2 is either 6*D_i or 24*D_i
    
    # Brute force checking all 2^n combinations is not feasible.
    # Let's think of this as a quadratic optimization over binary variables.
    # Let x_i = 1 if r_i = 2, and 0 if r_i = 1.
    # Then r_i = 1 + x_i
    # r_i^2 = (1 + x_i)^2 = 1 + 2x_i + x_i^2 = 1 + 2x_i + x_i (since x_i in {0,1}) = 1 + 3x_i
    # So S1 = sum_i 6*D_i*(1 + 3x_i) = 6*sum_i D_i + 18*sum_i D_i*x_i
    
    # For S2: r_u * r_v = (1+x_u)(1+x_v) = 1 + x_u + x_v + x_u*x_v
    # So S2 = 14 * sum_j (1 + x_{u_j} + x_{v_j} + x_{u_j}*x_{v_j})
    #     = 14*m + 14*sum_j x_{u_j} + 14*sum_j x_{v_j} + 14*sum_j x_{u_j}*x_{v_j}
    #     = 14*m + 14*sum_i x_i*degree_i + 14*sum_j x_{u_j}*x_{v_j}
    
    # So E = S1 - S2 = [6*sum D_i + 18*sum D_i*x_i] - [14*m + 14*sum degree_i*x_i + 14*sum x_u*x_v]
    #   = 6*sum D_i - 14*m + (18*sum D_i*x_i - 14*sum degree_i*x_i) - 14*sum x_u*x_v
    #   = 6*sum D_i - 14*m + sum x_i*(18*D_i - 14*degree_i) - 14*sum x_u*x_v
    
    # We want E = 2m
    # So: 6*sum D_i - 14*m + sum x_i*(18*D_i - 14*degree_i) - 14*sum x_u*x_v = 2m
    # Rearranging:
    # sum x_i*(18*D_i - 14*degree_i) - 14*sum x_u*x_v = 2m - 6*sum D_i + 14*m
    #                                                  = 16*m - 6*sum D_i

    # A = 16*m - 6*sum D_i
    # C_i = 18*D_i - 14*degree_i
    # So we want: sum C_i*x_i - 14*sum x_u*x_v = A
    
    total_degree = sum(degree)
    A = 16 * m - 6 * total_degree
    
    # Build adjacency for quadratic terms
    adj = [[] for _ in range(n + 1)]
    edge_count = defaultdict(int)
    for u, v in edges:
        if u > v:
            u, v = v, u
        key = (u, v)
        edge_count[key] += 1
        if edge_count[key] == 1:
            adj[u].append(v)
            adj[v].append(u)
    
    # Now we have a quadratic pseudo-boolean function to optimize.
    # Since we want the lexicographically smallest solution, we can try assignments greedily,
    # but due to quadratic interactions, this is complex.
    # However, let's observe that the problem might reduce to a 2SAT-like structure or graph coloring.
    
    # Let's re-examine the original equation with r_i in {1,2}.
    # E = sum_i 6*D_i*r_i^2 - sum_j 14*r_{u_j}*r_{v_j} = 2m
    
    # Let's consider all r_i = 1 first:
    # r_i^2 = 1, so S1 = sum 6*D_i
    # r_u*r_v = 1, so S2 = 14*m
    # E_initial = 6*sum D_i - 14*m
    
    # If we change r_k from 1 to 2:
    # Change in S1: 6*D_k*(4 - 1) = 18*D_k
    # Change in S2: -14 * (sum_{j: k in edge j} (2*other - 1*other)) 
    #                = -14 * (sum_{j=(k,l)} (2*r_l - r_l)) = -14 * (sum_{j=(k,l)} r_l)
    # So net change = 18*D_k - 14*(sum_{l: (k,l) in edges} r_l)
    
    # This suggests a local search approach won't be straightforward.
    # But since we're looking for exact value, maybe there's a pattern.
    
    # Let's try modeling this as a constraint satisfaction problem.
    # We'll use backtracking with pruning, prioritizing early variables.
    
    # But given constraints up to 2e5, even O(n) per check is too slow.
    # Let's look at the math again.
    
    # Rewrite E:
    # E = sum_i 6*D_i*r_i^2 - sum_{(u,v)} 14*r_u*r_v = 2m
    
    # Multiply both sides by 1/2:

        adj_list[u].append(v)
        adj_list[v].append(u)

        for start in range(1, n + 1):

                visited[start] = True
                while queue:
                    u = queue.pop(0)

                        if not visited[v]:
                            visited[v] = True
                            color[v] = 1 - color[u]
                            queue.append(v)
                        elif color[v] == color[u]:
                            return False
        return True
    
    # Reset visited for actual use

    color = [-1] * (n + 1)

        if not visited[start]:
            queue = [start]

                        queue.append(v)

                        break
                if not bipartite:
                    break
        if not bipartite:
            break
    
    if not bipartite:
        # Not bipartite doesn't necessarily mean impossible, but often does for such constraints
        # Let's just try a greedy assignment and check energy

    result = [1] * (n + 1)

        if color[i] == 1:
            result[i] = 2

        S1 += 6 * degree[i] * result[i] * result[i]

    for u, v in edges:
        S2 += 14 * result[u] * result[v]
    
    E = S1 - S2
    
    if E == 2 * m:

        print("YES")
