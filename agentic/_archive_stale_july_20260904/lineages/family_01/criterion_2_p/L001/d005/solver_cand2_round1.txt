import sys

# The condition r_u^3 + r_v^3 = 9 with r_i in {1, 2}
# 1^3 = 1, 2^3 = 8.
# 1 + 8 = 9.
# So for every edge (u, v), one must be 1 and the other must be 2.
# This is equivalent to checking if the graph is bipartite.
# To minimize the decimal 0.r_1r_2...r_n, we want r_i = 1 whenever possible.
# This is a greedy approach on connected components.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if res[i] == 0:
            # Start BFS/DFS for this component
            # To minimize the decimal, we want the first node in the component
            # to be 1 if possible.
            stack = [(i, 1)]
            res[i] = 1
            component = []
            
            # Use a list to keep track of nodes in this component
            q = [i]
            res[i] = 1
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if res[v] == 0:
                        res[v] = 3 - res[u]
                        q.append(v)
                    elif res[v] == res[u]:
                        print("NO")
                        return
            
    print("YES")

solve()
