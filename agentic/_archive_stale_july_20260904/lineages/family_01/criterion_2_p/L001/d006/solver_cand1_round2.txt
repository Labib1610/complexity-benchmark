import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Stability score for (r_u, r_v):
    # (1, 1) -> 3(2) - 2(1) = 4
    # (1, 2) -> 3(3) - 2(2) = 5
    # (2, 1) -> 3(3) - 2(2) = 5
    # (2, 2) -> 3(4) - 2(4) = 4
    # Let x_i = 0 if r_i = 1, x_i = 1 if r_i = 2.
    # If r_u=1, r_v=1: score 4
    # If r_u=1, r_v=2: score 5
    # If r_u=2, r_v=1: score 5
    # If r_u=2, r_v=2: score 4
    # Score is 5 if r_u != r_v, 4 if r_u == r_v.
    # Total score = 5 * (number of edges with different levels) + 4 * (number of edges with same levels)
    # Total score = 5 * (m - same) + 4 * same = 5m - same.
    # We need 5m - same = 5m => same = 0.
    # This means no two connected monomers can have the same resonance level.
    # This is equivalent to checking if the graph is bipartite.
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                res[u] = c + 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # The graph is bipartite. The two sets are color 0 and color 1.
    # To get the lexicographically smallest sequence, we want r_i = 1 as much as possible.
    # For each connected component, we have two choices:
    # Choice A: nodes with color 0 get r=1, nodes with color 1 get r=2.
    # Choice B: nodes with color 0 get r=2, nodes with color 1 get r=1.
    # For each component, pick the one that makes the first node in the component 1.
    
    final_res = [0] * (n + 1)
    visited = [False] * (n + 1)
    for i in range(1, n + 1):
        if not visited[i]:
            component = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                component.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            
            # Find the smallest index in this component
            start_node = min(component)
            # If we set res[start_node] = 1, then all nodes with color[u] == color[start_node] get 1
            # else they get 2.
            c_start = color[start_node]
            for node in component:
                if color[node] == c_start:
                    final_res[node] = 1
                else:
                    final_res[node] = 2
                    
    print("YES")
    print(*(final_res[1:]))

solve()
