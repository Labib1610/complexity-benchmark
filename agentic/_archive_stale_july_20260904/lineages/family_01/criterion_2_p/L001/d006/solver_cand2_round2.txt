import sys

# The problem asks for a configuration of modes {1, 2} such that for every link (u, v),
# the weight (mode[u] + mode[v]) = 3.
# Since modes are 1 or 2, the only way for (mode[u] + mode[v]) = 3 is if one is 1 and the other is 2.
# This means the graph must be bipartite.
# If the graph is bipartite, we can color it with two colors (1 and 2).
# For each connected component, there are two possible assignments (swapping 1s and 2s).
# To get the lexicographically smallest sequence, for each component, we check both
# assignments and pick the one that is lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            possible = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. color[v] as is
            # 2. 3 - color[v]
            # We want the lexicographically smallest result.
            # Find the first node in the component and try to make it 1.
            comp_nodes = sorted(component)
            first_node = comp_nodes[0]
            
            # Option 1: color[first_node] becomes 1
            # If color[first_node] is 1, use current colors. If 2, flip.
            if color[first_node] == 1:
                config1 = {node: color[node] for node in component}
                config2 = {node: 3 - color[node] for node in component}
            else:
                config1 = {node: 3 - color[node] for node in component}
                config2 = {node: color[node] for node in component}
            
            # Compare config1 and config2 lexicographically
            is_config1_smaller = False
            for node in comp_nodes:
                if config1[node] < config2[node]:
                    is_config1_smaller = True
                    break
                elif config1[node] > config2[node]:
                    is_config1_smaller = False
                    break
            
            best_config = config1 if is_config1_smaller else config2
            for node in component:
                ans[node] = best_config[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
