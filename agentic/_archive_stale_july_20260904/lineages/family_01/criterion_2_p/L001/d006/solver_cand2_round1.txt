```python
import sys

# The equation is: product(2 * (s_u^2 + s_v^2)) = product(5 * s_u * s_v)
# Let s_i be in {1, 2}.
# If s_u = 1, s_v = 1: 2 * (1+1) = 4, 5 * (1*1) = 5. Ratio = 4/5
# If s_u = 1, s_v = 2: 2 * (1+4) = 10, 5 * (1*2) = 10. Ratio = 1
# If s_u = 2, s_v = 1: 2 * (4+1) = 10, 5 * (2*1) = 10. Ratio = 1
# If s_u = 2, s_v = 2: 2 * (4+4) = 16, 5 * (2*2) = 20. Ratio = 16/20 = 4/5
#
# Let x_i = 1 if s_i = 1, x_i = 0 if s_i = 2.
# Ratio for (u, v) is 4/5 if (s_u, s_v) is (1, 1) or (2, 2), and 1 if (1, 2) or (2, 1).
# (s_u, s_v) is (1, 1) or (2, 2) iff s_u == s_v.
# So the equation is: product_{links (u,v) where s_u == s_v} (4/5) = 1.
# This is equivalent to: (4/5)^k = 1, where k is the number of links with s_u == s_v.
# Since 4/5 != 1, this requires k = 0.
# Thus, we need s_u != s_v for all links (u, v).
# This is exactly the condition for a graph to be bipartite.
# We need to find the lexicographically smallest assignment (s_1, ..., s_n)
# where s_i in {1, 2} and s_u != s_v for all edges.
# This means for each connected component, we pick a color for the first node
# (1 is smaller than 2) and propagate.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    colors = [0] * (n + 1)
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS, assign 1 to the first node to be lexicographically smallest
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
