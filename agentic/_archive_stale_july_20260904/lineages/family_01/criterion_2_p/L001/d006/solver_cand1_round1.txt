import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: sum_{i=1}^m (5(r_u + r_v) - 3(r_u * r_v)) = 9m
    # Let r_i = 1 + x_i, where x_i in {0, 1}.
    # 5(2 + x_u + x_v) - 3(1 + x_u + x_v + x_u*x_v) = 9
    # 10 + 5x_u + 5x_v - 3 - 3x_u - 3x_v - 3x_u*x_v = 9
    # 7 + 2x_u + 2x_v - 3x_u*x_v = 9
    # 2x_u + 2x_v - 3x_u*x_v = 2
    # If x_u=0, x_v=1: 2(1) - 0 = 2 (Valid)
    # If x_u=1, x_v=0: 2(1) - 0 = 2 (Valid)
    # If x_u=0, x_v=0: 0 = 2 (Invalid)
    # If x_u=1, x_v=1: 4 - 3 = 1 (Invalid)
    # Thus, for every edge (u, v), x_u != x_v.
    # This means the graph must be bipartite.
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            component = []
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # For each component, we have two choices:
            # Choice 0: nodes with color 0 get r=1, nodes with color 1 get r=2
            # Choice 1: nodes with color 0 get r=2, nodes with color 1 get r=1
            # We want lexicographically smallest sequence.
            # For the first node i in the component, we prefer r_i = 1.
            # If color[i] == 0, choose Choice 0. If color[i] == 1, choose Choice 1.
            
            if color[i] == 0:
                for u, c in component:
                    res[u] = 1 if c == 0 else 2
            else:
                for u, c in component:
                    res[u] = 2 if c == 0 else 1
                    
    print("YES")
    print(*(res[1:]))

solve()
