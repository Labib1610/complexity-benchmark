import sys

# The condition: spin_u XOR spin_v = 3
# Possible spin states are {1, 2}.
# 1 XOR 1 = 0
# 1 XOR 2 = 3
# 2 XOR 1 = 3
# 2 XOR 2 = 0
# So, spin_u XOR spin_v = 3 means spin_u != spin_v.
# This is a standard 2-coloring problem (bipartite graph check).
# To get the lexicographically smallest sequence, we want s_i = 1 whenever possible.
# For each connected component, we have two choices for the coloring (swap 1s and 2s).
# We choose the one that makes the first node in the component 1 if possible,
# but actually, we want the overall sequence to be lexicographically smallest.
# This means for each component, we check the two possible colorings and pick the one
# that results in a smaller sequence. Since we want the smallest sequence,
# we should try to assign 1 to the smallest index in the component if possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # 1. Keep as is (color[v])
            # 2. Flip (3 - color[v])
            # We want the lexicographically smallest result.
            # Find the smallest index in this component.
            # To make the sequence smallest, we want the smallest index to be 1.
            # If we set the smallest index to 1, the rest are determined.
            
            min_node = min(component)
            # If color[min_node] is 1, we keep it. If 2, we flip.
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
