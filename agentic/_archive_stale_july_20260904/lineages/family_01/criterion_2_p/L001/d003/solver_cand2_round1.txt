```python
import sys

# The resonance rule: spin_u XOR spin_v = 3
# Possible states are {1, 2}.
# 1 XOR 1 = 0 (not 3)
# 1 XOR 2 = 3 (valid)
# 2 XOR 1 = 3 (valid)
# 2 XOR 2 = 0 (not 3)
# This means for every bond (u, v), spin_u must be different from spin_v.
# This is equivalent to 2-coloring a graph.
# Lexicographically smallest means we prefer 1 over 2 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    # To get lexicographically smallest, we iterate 1 to n.
    # If a node is uncolored, we assign it 1 (the smallest possible state).
    # Then we propagate the coloring using BFS/DFS.
    for i in range(1, n + 1):
        if colors[i] == 0:
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

solve()
```
