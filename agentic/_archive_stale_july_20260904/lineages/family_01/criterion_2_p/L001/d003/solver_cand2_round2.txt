```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition r_u + r_v = 3 means r_u != r_v.
    # This is a bipartite graph coloring problem.
    # We want the lexicographically smallest sequence (r_1, r_2, ..., r_n).
    # To make it lexicographically smallest, we prefer 1 over 2.
    # For each connected component, we can pick the color of the first node
    # (the one with the smallest index) to be 1.
    
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if res[i] == 0:
            # Start BFS/DFS from i, setting res[i] = 1
            stack = [i]
            res[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if res[v] == 0:
                        res[v] = 3 - res[u]
                        stack.append(v)
                    elif res[v] == res[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(res[1:]))

if __name__ == '__main__':
    solve()
```
