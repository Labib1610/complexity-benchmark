```python
import sys

# The problem is a 2-SAT problem.
# Let state 1 be represented by boolean False (0) and state 2 by True (1).
# Variables: x_i is 0 if state is 1, 1 if state is 2.
# k = (u + v) % 4
# k=0: (u=1 or v=1) AND (u=2 or v=2)
#      (u=0 or v=0) AND (u=1 or v=1)
#      Equivalent to: u != v
# k=1: NOT (u=1 and v=1) AND NOT (u=2 and v=2)
#      NOT (u=0 and v=0) AND NOT (u=1 and v=1)
#      Equivalent to: u == v
# k=2: (u=1 => v=2) AND (u=2 => v=1)
#      (u=0 => v=1) AND (u=1 => v=0)
#      Equivalent to: u != v
# k=3: (u=1 <=> v=2)
#      (u=0 <=> v=1)
#      Equivalent to: u == v

def solve():
    input_data = sys.stdin.read().split()
    if not input_data: return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(2 * n)]
    
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]) - 1, int(input_data[idx+1]) - 1
        idx += 2
        k = (u + v + 2) % 4
        if k == 0 or k == 2:
            # u != v: (u or v) and (not u or not v)
            # u=0, v=1: u_0 -> v_1, v_0 -> u_1, u_1 -> v_0, v_1 -> u_0
            adj[2*u].append(2*v + 1)
            adj[2*v].append(2*u + 1)
            adj[2*u + 1].append(2*v)
            adj[2*v + 1].append(2*u)
        else:
            # u == v: (u or not v) and (not u or v)
            adj[2*u].append(2*v)
            adj[2*v].append(2*u)
            adj[2*u + 1].append(2*v + 1)
            adj[2*v + 1].append(2*u + 1)

    dfn = [-1] * (2 * n)
    low = [-1] * (2 * n)
    scc = [-1] * (2 * n)
    stk = []
    in_stk = [False] * (2 * n)
    timer = 0
    scc_cnt = 0

    def tarjan(u):
        nonlocal timer, scc_cnt
        dfn[u] = low[u] = timer
        timer += 1
        stk.append(u)
        in_stk[u] = True
        for v in adj[u]:
            if dfn[v] == -1:
                tarjan(v)
                low[u] = min(low[u], low[v])
            elif in_stk[v]:
                low[u] = min(low[u], dfn[v])
        if low[u] == dfn[u]:
            while True:
                node = stk.pop()
                in_stk[node] = False
                scc[node] = scc_cnt
                if node == u: break
            scc_cnt += 1

    for i in range(2 * n):
        if dfn[i] == -1: tarjan(i)

    res = [0] * n
    for i in range(n):
        if scc[2*i] == scc[2*i+1]:
            print("NO")
            return
        # Lexicographically smallest: state 1 (0) is preferred over state 2 (1)
        # In 2-SAT, if scc[x] < scc[not x], x is False.
        res[i] = 1 if scc[2*i] < scc[2*i+1] else 2
    
    print("YES")
    print(*(res))

solve()
```
