import sys
from collections import deque

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    g = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(next(it)); v = int(next(it))
        g[u].append(v)
        g[v].append(u)

    color = [-1] * (n + 1)          # bipartite side 0/1
    comp_id = [-1] * (n + 1)
    comps = []                      # list of (vertices list, min_vertex)
    q = deque()
    cid = 0
    for v in range(1, n + 1):
        if color[v] != -1:
            continue
        # start new component
        vertices = []
        min_v = v
        color[v] = 0
        comp_id[v] = cid
        q.append(v)
        ok = True
        while q:
            x = q.popleft()
            vertices.append(x)
            if x < min_v:
                min_v = x
            for nb in g[x]:
                if color[nb] == -1:
                    color[nb] = color[x] ^ 1
                    comp_id[nb] = cid
                    q.append(nb)
                elif color[nb] == color[x]:
                    print("NO")
                    return
        comps.append((vertices, min_v))
        cid += 1

    # decide final resonance levels (1 or 2)
    ans = [0] * (n + 1)
    for vertices, min_v in comps:
        # we want min_v to have resonance 1
        if color[min_v] == 0:
            # side 0 -> 1, side 1 -> 2
            for v in vertices:
                ans[v] = 1 if color[v] == 0 else 2
        else:
            # side 1 -> 1, side 0 -> 2
            for v in vertices:
                ans[v] = 1 if color[v] == 1 else 2

    print("YES")
    print(' '.join(str(ans[i]) for i in range(1, n + 1)))

if __name__ == "__main__":
    solve()
