import sys

# The condition is: sum_{j in neighbors(i)} r_j = d_i * (3 - r_i)
# Let x_i = r_i - 1. Then r_i = x_i + 1.
# The condition becomes: sum_{j in neighbors(i)} (x_j + 1) = d_i * (3 - (x_i + 1))
# sum_{j in neighbors(i)} x_j + d_i = d_i * (2 - x_i)
# sum_{j in neighbors(i)} x_j + d_i = 2 * d_i - d_i * x_i
# sum_{j in neighbors(i)} x_j + d_i * x_i = d_i
# This is equivalent to: sum_{j in neighbors(i)} x_j = d_i * (1 - x_i)
# If x_i = 0 (r_i = 1), then sum_{j in neighbors(i)} x_j = d_i.
# This means all neighbors must have x_j = 1 (r_j = 2).
# If x_i = 1 (r_i = 2), then sum_{j in neighbors(i)} x_j = 0.
# This means all neighbors must have x_j = 0 (r_j = 1).
# This is exactly the condition for a bipartite graph:
# r_i = 1 implies all neighbors are 2, r_i = 2 implies all neighbors are 1.
# This is possible if and only if the graph is bipartite.
# For a bipartite graph, there are two possible colorings (swapping 1s and 2s).
# We want the lexicographically smallest sequence r_1, r_2, ..., r_n.
# Let the two colorings be C1 and C2.
# C1 assigns r_i = 1 if i is in partition A, r_i = 2 if i is in partition B.
# C2 assigns r_i = 2 if i is in partition A, r_i = 1 if i is in partition B.
# We compare C1 and C2 lexicographically.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    color = [-1] * (n + 1)
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return

    # Two possible colorings:
    # C1: r_i = color[i] + 1
    # C2: r_i = (1 - color[i]) + 1
    # We want the lexicographically smallest.
    # Compare r_1 in C1 and C2.
    c1_1 = color[1] + 1
    c2_1 = (1 - color[1]) + 1
    
    if c1_1 < c2_1:
        res = [color[i] + 1 for i in range(1, n + 1)]
    elif c2_1 < c1_1:
        res = [(1 - color[i]) + 1 for i in range(1, n + 1)]
    else:
        # If r_1 is same, compare r_2, etc.
        res1 = [color[i] + 1 for i in range(1, n + 1)]
        res2 = [(1 - color[i]) + 1 for i in range(1, n + 1)]
        res = res1 if res1 < res2 else res2

    print("YES")
    print(*(res))

solve()
