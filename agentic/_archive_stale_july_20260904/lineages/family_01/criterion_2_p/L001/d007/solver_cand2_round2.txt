import sys

# The problem states that for any bond (u, v), the intervals must intersect at exactly one point.
# The available intervals are [0, 1] and [1, 2].
# [0, 1] and [1, 2] intersect at exactly {1}.
# [0, 1] and [0, 1] intersect at [0, 1] (length 1).
# [1, 2] and [1, 2] intersect at [1, 2] (length 1).
# Thus, for every bond (u, v), the interval assigned to u must be different from the interval assigned to v.
# This is equivalent to 2-coloring a graph where each bond is an edge.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest sequence (1s and 2s).
# We iterate from 1 to n, and if a node is not colored, we try assigning 1.
# If assigning 1 leads to a contradiction, we assign 2.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            color[i] = 1
            component = []
            
            # We need to check if this component is bipartite
            # To get lexicographically smallest, we greedily pick 1 if possible
            # But for a connected component, once the first node is fixed, 
            # the rest are determined.
            # Actually, for each component, we can choose the color of the 
            # first node (the one with the smallest index) to be 1.
            
            q = [i]
            color[i] = 1
            while q:
                u = q.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
