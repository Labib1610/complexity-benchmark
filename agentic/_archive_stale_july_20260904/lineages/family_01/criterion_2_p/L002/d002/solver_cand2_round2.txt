import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(10**6)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest assignment.
    # This means we prefer 1 over 2 for sensor 1, then sensor 2, etc.
    # This is a 2-coloring problem.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Try assigning 1 to the current component's root to keep it lexicographically small
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # The greedy choice for the component is to ensure the smallest index
            # in the component gets 1 if possible. Since we process 1..n,
            # the first uncolored node i is always the smallest in its component.
            # Assigning color[i] = 1 is always the best local choice.
            
    print("YES")
    print(*(color[1:]))

solve()
