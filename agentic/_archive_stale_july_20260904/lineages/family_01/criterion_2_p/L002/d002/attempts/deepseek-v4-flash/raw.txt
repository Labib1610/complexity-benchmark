We can model this as a bipartite graph problem.  
If the graph is not bipartite, it’s impossible.  
If it is bipartite, we can assign colors (1 or 2) and choose the lexicographically smallest assignment by trying to assign 1 whenever possible during BFS/DFS.

```python
import sys
from collections import deque

input = sys.stdin.readline

n, m = map(int, input().split())
adj = [[] for _ in range(n)]
for _ in range(m):
    u, v = map(int, input().split())
    u -= 1
    v -= 1
    adj[u].append(v)
    adj[v].append(u)

color = [-1] * n
possible = True

for i in range(n):
    if color[i] == -1:
        color[i] = 0
        q = deque([i])
        while q:
            u = q.popleft()
            for v in adj[u]:
                if color[v] == -1:
                    color[v] = color[u] ^ 1
                    q.append(v)
                elif color[v] == color[u]:
                    possible = False
                    break
            if not possible:
                break
    if not possible:
        break

if not possible:
    print("NO")
else:
    # color 0 -> band 1, color 1 -> band 2
    # But we want lexicographically smallest: try to assign 1 as much as possible
    # We can flip colors in each connected component if it makes earlier nodes 1.
    # Since we already have a valid 2-coloring, we can choose orientation per component.
    visited = [False] * n
    for i in range(n):
        if not visited[i]:
            # BFS to collect component
            comp = []
            q = deque([i])
            visited[i] = True
            while q:
                u = q.popleft()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        q.append(v)
            # Check if flipping helps lexicographically
            # If color[i] == 1, flipping makes it 0 -> smaller
            if color[i] == 1:
                for u in comp:
                    color[u] ^= 1

    ans = [1 if c == 0 else 2 for c in color]
    print("YES")
    print(" ".join(map(str, ans)))
```
