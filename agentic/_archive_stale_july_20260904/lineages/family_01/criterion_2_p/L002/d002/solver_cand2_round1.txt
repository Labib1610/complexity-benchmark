```python
import sys

# The problem states that for each interfering pair (u, v), 
# their band assignments s_u and s_v must cover both bands {1, 2}.
# This means s_u != s_v.
# This is a classic 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# To get the lexicographically smallest assignment, we iterate from 1 to n.
# If a node is not yet colored, we assign it 1 (the smaller band) 
# and perform a BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer color 1 for the first node
            stack = [(i, 1)]
            colors[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
