```python
import sys

# The problem asks for a 2-coloring of a graph such that for every edge (u, v),
# the colors s_u and s_v satisfy s_u + s_v = 3.
# Since the colors are {1, 2}, this means s_u != s_v.
# This is exactly the definition of a bipartite graph.
# We need the lexicographically smallest assignment (s_1, s_2, ..., s_n).
# To achieve this, we iterate from 1 to n. If a node is unvisited, we try
# assigning it color 1 (the smallest possible). This forces the colors of all
# nodes in its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer color 1 for the
            # first node encountered in the component.
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
