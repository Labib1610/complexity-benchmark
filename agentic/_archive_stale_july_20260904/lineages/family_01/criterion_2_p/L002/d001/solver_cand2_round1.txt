import sys

# The condition s_u + s_v = 3 means that if s_u = 1, then s_v = 2, and if s_u = 2, then s_v = 1.
# This is equivalent to saying s_u != s_v.
# This is a standard bipartite graph coloring problem.
# We need to find the lexicographically smallest assignment.
# To get the lexicographically smallest assignment, we want s_i = 1 whenever possible.
# For each connected component, we have two choices:
# 1. Assign the first node in the component to 1.
# 2. Assign the first node in the component to 2.
# We should pick the choice that results in a smaller sequence.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append((i, 1))
            
            is_bipartite = True
            idx_stack = 0
            while idx_stack < len(component):
                u, c = component[idx_stack]
                idx_stack += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Check which coloring is lexicographically smaller
            # Option 1: current coloring
            # Option 2: flip coloring (1 -> 2, 2 -> 1)
            # We want the smallest value at the smallest index in the component.
            # Find the smallest index in this component.
            min_node = min(node for node, c in component)
            
            # Find the color assigned to min_node in the current coloring
            c_min = 0
            for node, c in component:
                if node == min_node:
                    c_min = c
                    break
            
            if c_min == 2:
                # Flip colors
                for node, c in component:
                    color[node] = 3 - c
                    
    print("YES")
    print(*(color[1:]))

solve()
