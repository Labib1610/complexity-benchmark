```python
import sys

# The problem states that for each contention (u, v), s_u + s_v = 3.
# Since s_i is in {1, 2}, this means s_u != s_v.
# This is equivalent to a 2-coloring problem on a graph.
# A graph is 2-colorable if and only if it is bipartite.
# We need the lexicographically smallest assignment, which means we want
# to assign 1 to as many processes as possible, prioritizing smaller indices.
# For each connected component, we can pick the color of the first node
# (the one with the smallest index) to be 1, then the rest are determined.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want colors[i] = 1
            stack = [(i, 1)]
            colors[i] = 1
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif colors[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(colors[1:]))

solve()
```
