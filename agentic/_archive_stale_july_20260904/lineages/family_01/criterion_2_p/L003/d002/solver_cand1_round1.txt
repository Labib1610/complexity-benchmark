```python
import sys

# The problem states that for an edge (u, v), the ratio state[u] / state[v] must be 0.5 or 2.0.
# Since state[u], state[v] are in {1, 2}, this means state[u] != state[v].
# This is equivalent to saying the graph must be bipartite.
# We need to find the lexicographically smallest assignment.
# For each connected component, we can pick the state of the first node (smallest index)
# to be 1, which forces the states of all other nodes in that component.
# If we find a contradiction (not bipartite), output NO.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    state = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if state[i] == 0:
            # Start BFS/DFS for this component
            # To get lexicographically smallest, we want state[i] = 1
            state[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if state[v] == 0:
                        state[v] = 3 - state[u]
                        stack.append(v)
                    elif state[v] == state[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(state[1:]))

solve()
```
