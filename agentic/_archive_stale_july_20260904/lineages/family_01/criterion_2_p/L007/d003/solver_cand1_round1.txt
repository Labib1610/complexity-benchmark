import sys

# The problem asks for a 2-coloring of a graph such that no edge connects two nodes of the same color.
# This is equivalent to checking if the graph is bipartite.
# To get the lexicographically smallest sequence, we want to assign 1 to the smallest possible index
# whenever possible.
# For each connected component, we have two choices for coloring:
# 1. Assign the first node in the component (by index) to 1.
# 2. Assign the first node in the component (by index) to 2.
# We compare the resulting sequences for each component and pick the one that is lexicographically smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    ans = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            # BFS/DFS to find component and check bipartiteness
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Two possible colorings for this component:
            # Option 1: color[i] = 1
            # Option 2: color[i] = 2 (which is just flipping 1s and 2s)
            
            # To be lexicographically smallest, we want the smallest index in the component
            # to have color 1 if possible.
            # Actually, for each component, we just need to check which coloring
            # makes the first node (smallest index) 1.
            
            # Let's store the two colorings
            comp_nodes = sorted(component)
            first_node = comp_nodes[0]
            
            # If color[first_node] is 1, we keep it. If it's 2, we flip.
            if color[first_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
            
            for node in component:
                ans[node] = color[node]
                
    print("YES")
    print(*(ans[1:]))

solve()
