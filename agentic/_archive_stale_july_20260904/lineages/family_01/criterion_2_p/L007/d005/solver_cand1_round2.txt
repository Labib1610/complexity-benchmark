```python
import sys

# The local resonance value of a dampener connecting u and v with grades c_u, c_v
# is the square of the difference between the roots of x^2 - (c_u + c_v)x + c_u*c_v = 0.
# The roots are c_u and c_v.
# The difference is |c_u - c_v|.
# The square of the difference is (c_u - c_v)^2.
# Since c_u, c_v are in {1, 2}, (c_u - c_v)^2 is 0 if c_u == c_v, and 1 if c_u != c_v.
# The sum of resonance values is the number of dampeners where c_u != c_v.
# We want this sum to be exactly m.
# This means for every dampener, we must have c_u != c_v.
# This is equivalent to saying the graph is bipartite.
# We need the lexicographically smallest assignment of 1s and 2s.
# For each connected component, we can choose the color of the first node (in index order)
# to be 1, which forces the colors of all other nodes in that component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the component
            # To get lexicographically smallest, we prefer 1 for the first node
            stack = [(i, 1)]
            color[i] = 1
            while stack:
                u, c = stack.pop()
                next_c = 3 - c
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = next_c
                        stack.append((v, next_c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

solve()
```
