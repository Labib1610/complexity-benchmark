import sys

# The stress dissipation factor S(c_u, c_v) = c_u^3 + c_v^3 + 7(c_u^2 + c_v^2) + 2(c_u + c_v) - 20 c_u c_v
# Let's calculate S for possible pairs (c_u, c_v) where c_i in {1, 2}:
# S(1, 1) = 1 + 1 + 7(1 + 1) + 2(1 + 1) - 20(1) = 2 + 14 + 4 - 20 = 0
# S(1, 2) = 1 + 8 + 7(1 + 4) + 2(1 + 2) - 20(2) = 9 + 35 + 6 - 40 = 10
# S(2, 1) = 10
# S(2, 2) = 8 + 8 + 7(4 + 4) + 2(2 + 2) - 20(4) = 16 + 56 + 8 - 80 = 0
#
# We want the sum of S over m edges to be 10m.
# Since S(1, 1) = 0, S(2, 2) = 0, and S(1, 2) = S(2, 1) = 10,
# the sum is 10 * (number of edges connecting a 1 and a 2).
# We need the number of edges connecting a 1 and a 2 to be exactly m.
# This means every edge must connect a 1 and a 2.
# This is equivalent to saying the graph must be bipartite, and we must
# color it with 1s and 2s such that every edge connects different colors.
# If the graph is bipartite, there are two ways to color each connected component.
# To get the lexicographically smallest sequence, for each component,
# we check the two possible colorings and pick the one that is smaller.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n, m = int(input_data[0]), int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u, v = int(input_data[idx]), int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    color = [0] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [(i, 1)]
            color[i] = 1
            component.append(i)
            
            is_bipartite = True
            while stack:
                u, c = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        component.append(v)
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
            # Two colorings: 
            # 1. color[v] as is
            # 2. flip colors (1 -> 2, 2 -> 1)
            # We want the lexicographically smallest result.
            # Find the smallest index in the component.
            comp_nodes = sorted(component)
            start_node = comp_nodes[0]
            
            # Option 1: color[start_node] = 1
            # Option 2: color[start_node] = 2
            # We want the smallest value at the smallest index.
            # If color[start_node] is 1, option 1 is better.
            # If color[start_node] is 2, option 2 is better.
            
            if color[start_node] == 1:
                for node in component:
                    res[node] = color[node]
            else:
                for node in component:
                    res[node] = 3 - color[node]
                    
    print("YES")
    print(*(res[1:]))

solve()
