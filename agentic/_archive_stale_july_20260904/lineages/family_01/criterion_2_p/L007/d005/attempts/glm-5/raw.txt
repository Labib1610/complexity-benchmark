import sys
from sys import stdin

def solve():
    input_data = sys.stdin.buffer.read().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n+1)]
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0]*(n+1)
    comp = [0]*(n+1)
    comp_id = 0
    
    for i in range(1, n+1):
        if color[i] == 0:
            comp_id += 1
            stack = [i]
            color[i] = 1
            comp[i] = comp_id
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        comp[v] = comp_id
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    cnt = [0]*(comp_id+1)
    for i in range(1, n+1):
        cnt[comp[i]] += 1
    
    comp_color = [0]*(comp_id+1)
    for c in range(1, comp_id+1):
        comp_color[c] = 1 if cnt[c] <= 0 else 2
    
    for i in range(1, n+1):
        c = comp[i]
        if comp_color[c] == 1:
            color[i] = 1
        else:
            color[i] = 2
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n+1)))

solve()
