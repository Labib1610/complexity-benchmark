```python
import sys

# The problem asks to assign 1 or 2 to each node such that for every edge (u, v),
# token(u) + token(v) = 3.
# This means token(u) != token(v).
# This is equivalent to checking if the graph is bipartite.
# If bipartite, we want the lexicographically smallest assignment.
# Lexicographically smallest means we prefer 1 over 2 for smaller indices.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate from 1 to n.
    # If a node is unassigned, we assign it 1 (the smallest possible value).
    # Then we perform BFS/DFS to propagate the constraints.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            assignment[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs
    sys.setrecursionlimit(300000)
    solve()
```
