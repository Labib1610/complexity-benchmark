import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # The condition c_u * c_v = 2 means one must be 1 and the other 2.
    # This is a bipartite coloring problem.
    # We want the lexicographically smallest sequence (c_1, c_2, ..., c_n).
    # This means we prefer c_i = 1 whenever possible.
    
    color = [0] * (n + 1) # 0: unvisited, 1: color 1, 2: color 2
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this connected component
            # To make it lexicographically smallest, we try to assign 1 to the 
            # smallest index in the component.
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard bipartite check
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Now we have a valid 2-coloring for this component.
            # We have two choices: 
            # Choice A: color[i] = 1
            # Choice B: color[i] = 2 (which flips all colors in the component)
            # We want the lexicographically smallest sequence.
            # Let's compare the two options.
            
            # Option 1: Keep current colors
            # Option 2: Flip colors (1 -> 2, 2 -> 1)
            
            # Find the smallest index in this component
            min_node = min(component)
            
            # If we set color[min_node] = 1, is that better?
            # If color[min_node] is already 1, keep it.
            # If color[min_node] is 2, flip all.
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
