import sys

def solve():
    # Constraint: c^3 + 5c <= 4c^2 + 2
    # c^3 - 4c^2 + 5c - 2 <= 0
    # Testing small integers:
    # c=1: 1 - 4 + 5 - 2 = 0 <= 0 (True)
    # c=2: 8 - 16 + 10 - 2 = 0 <= 0 (True)
    # c=3: 27 - 36 + 15 - 2 = 4 > 0 (False)
    # So c_i must be 1 or 2.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: (3*c_u + 5*c_v) % 7 > 3
    # Possible values for c_u, c_v in {1, 2}:
    # (1, 1): (3+5)%7 = 8%7 = 1 (<= 3, Fail)
    # (1, 2): (3+10)%7 = 13%7 = 6 (> 3, Pass)
    # (2, 1): (6+5)%7 = 11%7 = 4 (> 3, Pass)
    # (2, 2): (6+10)%7 = 16%7 = 2 (<= 3, Fail)
    
    # This is a 2-coloring problem.
    # If c_u = 1, then c_v must be 2.
    # If c_u = 2, then c_v must be 1.
    # This means c_u != c_v.
    # The graph must be bipartite.
    
    color = [-1] * (n + 1)
    res = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == -1:
            stack = [(i, 0)]
            color[i] = 0
            while stack:
                u, c = stack.pop()
                res[u] = c + 1
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - c
                        stack.append((v, 1 - c))
                    elif color[v] == c:
                        print("NO")
                        return
    
    # Lexicographically smallest:
    # We have two possible colorings for each connected component (swap 1s and 2s).
    # To make the sequence lexicographically smallest, for each component,
    # we want the first pillar (smallest index) to be 1 if possible.
    # If the first pillar in a component is assigned 1, the rest are fixed.
    # If we assign 2, the rest are fixed.
    # Since we want the smallest sequence, we check the first node of each component.
    
    print("YES")
    print(*(res[1:]))

solve()
