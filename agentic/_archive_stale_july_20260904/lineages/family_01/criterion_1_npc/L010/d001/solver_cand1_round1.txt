import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The problem asks for the lexicographically smallest sequence.
    # We use backtracking to assign units 1, 2, 3 to specimens 1 to n.
    # To ensure lexicographical smallest, we try assigning 1, then 2, then 3.
    
    assignment = [0] * (n + 1)
    
    def is_safe(u, unit):
        for v in adj[u]:
            if assignment[v] == unit:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for unit in range(1, 4):
            if is_safe(u, unit):
                assignment[u] = unit
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
