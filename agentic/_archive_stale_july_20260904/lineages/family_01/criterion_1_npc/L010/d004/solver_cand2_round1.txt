import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    rules = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        rules.append((u, v))
        idx += 2
        
    # A_u + 2*A_v != 0 (mod 3)
    # A_u - A_v != 0 (mod 3)
    # A_u != A_v (mod 3)
    # Since A_i in {1, 2, 3}, A_i mod 3 is in {1, 2, 0}
    # Let x_i = A_i mod 3. x_i in {0, 1, 2}.
    # Rule: x_u != x_v.
    # This is a graph coloring problem with 3 colors {0, 1, 2}.
    # We want the lexicographically smallest sequence A_1, ..., A_n.
    # A_i = 3 if x_i = 0, else A_i = x_i.
    # To minimize A_i lexicographically, we want to pick x_i such that
    # A_i is 1, then 2, then 3.
    # Mapping: 1 -> 1, 2 -> 2, 3 -> 0.
    # Order of preference for x_i: 1, 2, 0.
    
    adj = [[] for _ in range(n)]
    for u, v in rules:
        adj[u].append(v)
        adj[v].append(u)
        
    color = [-1] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if color[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try values 1, 2, 3 in order to get lexicographically smallest
        for val in [1, 2, 3]:
            c = val % 3
            if is_valid(u, c):
                color[u] = c
                if backtrack(u + 1):
                    return True
                color[u] = -1
        return False

    if backtrack(0):
        print("YES")
        res = []
        for i in range(n):
            if color[i] == 0:
                res.append(3)
            else:
                res.append(color[i])
        print(*(res))
    else:
        print("NO")

solve()
