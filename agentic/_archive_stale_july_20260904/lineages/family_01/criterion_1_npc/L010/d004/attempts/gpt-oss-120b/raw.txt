```python
import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n          # 0 = uncolored, else 1..3
    # For each vertex keep a bitmask of forbidden colors from already colored neighbors
    forbid = [0] * n           # bit i set means color i+1 is forbidden

    # Precompute neighbor list for speed
    neighbors = adj

    # Simple forward checking: if any vertex gets all three colors forbidden before being colored -> dead end
    def dfs(v: int) -> bool:
        if v == n:
            return True
        # try colors 1..3 in order for lexicographic minimality
        for c in range(1, 4):
            if forbid[v] & (1 << (c - 1)):
                continue
            # assign
            colors[v] = c
            changed = []  # store neighbors whose forbid mask we modify
            conflict = False
            mask = 1 << (c - 1)
            for nb in neighbors[v]:
                if colors[nb] == c:
                    conflict = True
                    break
                if colors[nb] == 0:
                    if not (forbid[nb] & mask):
                        forbid[nb] |= mask
                        changed.append(nb)
                        if forbid[nb] == 0b111:   # all three colors now forbidden
                            conflict = True
                            break
            if not conflict and dfs(v + 1):
                return True
            # backtrack
            for nb in changed:
                forbid[nb] &= ~mask
            colors[v] = 0
        return False

    if dfs(0):
        print("YES")
        print(' '.join(map(str, colors)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
