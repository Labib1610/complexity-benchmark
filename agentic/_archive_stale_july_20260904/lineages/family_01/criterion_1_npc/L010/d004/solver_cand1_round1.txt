```python
import sys

# The problem asks to color a graph with 3 colors such that no two adjacent
# vertices have the same color. We need to minimize the lexicographical value
# of the sequence (g_1, g_2, ..., g_n) because the score P = sum(g_i * 4^(n-i))
# is equivalent to interpreting the sequence as a base-4 number.
# Since we want to minimize P, we should greedily try to assign the smallest
# possible color (1, then 2, then 3) to each vertex i from 1 to n.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in order to minimize the score
        for c in range(1, 4):
            if is_safe(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # The problem states that naive trial-and-error might be slow.
    # However, 3-coloring is NP-complete. Given the constraints and the
    # requirement to minimize the score, a greedy approach with backtracking
    # is the standard way to find the lexicographically smallest coloring.
    # We use a simple heuristic: order vertices by degree or just 1..n.
    # Given the structure of the problem, 1..n is required for the score.
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # Increase recursion depth for deep backtracking
    sys.setrecursionlimit(200000)
    solve()
```
