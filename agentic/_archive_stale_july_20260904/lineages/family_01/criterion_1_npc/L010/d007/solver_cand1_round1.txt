```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition (S^T P S)_{k,k} = 0 means that for each frequency k in {1, 2, 3},
    # there are no two relays u, v such that f_u = k, f_v = k, and P_{u,v} = 1.
    # This is equivalent to saying that for each frequency k, the set of relays
    # assigned to k must form an independent set in the proximity graph.
    # This is exactly the definition of a 3-coloring problem.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
