import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Primes: P1=2, P2=3, P3=5
    # K = P_fu * P_fv
    # phi(K) = K * (1 - 1/p1) * (1 - 1/p2)
    # If fu == fv, K = p^2, phi(K) = p^2 - p. Condition: p^2 - p < p^2 - p
    # This is false (p^2 - p < p^2 - p is never true).
    # So fu != fv must hold for all adjacent pairs.
    # If fu != fv, K = p1 * p2, phi(K) = (p1-1)*(p2-1) = p1*p2 - p1 - p2 + 1
    # Condition: p1*p2 - p1 - p2 + 1 < p1*p2 - sqrt(p1*p2)
    # -p1 - p2 + 1 < -sqrt(p1*p2)
    # sqrt(p1*p2) < p1 + p2 - 1
    # Check pairs:
    # (2,3): sqrt(6) approx 2.45. 2+3-1 = 4. 2.45 < 4 (True)
    # (2,5): sqrt(10) approx 3.16. 2+5-1 = 6. 3.16 < 6 (True)
    # (3,5): sqrt(15) approx 3.87. 3+5-1 = 7. 3.87 < 7 (True)
    # So the condition is satisfied if and only if fu != fv.
    # This is a 3-coloring problem.

    assignment = [0] * (n + 1)

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
