import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # The polynomial is P(x) = x^3 - (f_u + f_v)x^2 + (f_u * f_v)x
    # P(x) = x(x^2 - (f_u + f_v)x + f_u * f_v)
    # P(x) = x(x - f_u)(x - f_v)
    # The roots are 0, f_u, f_v.
    # For the roots to be distinct, we need:
    # 1. f_u != 0 (always true as f_i in {1, 2, 3})
    # 2. f_v != 0 (always true)
    # 3. f_u != f_v
    # So, for every pair (u, v) in proximity, f_u must not equal f_v.
    # This is equivalent to finding a 3-coloring of the graph.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v1, v2 in adj:
            if v1 == u:
                if colors[v2] == c:
                    return False
            elif v2 == u:
                if colors[v1] == c:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
