import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition |x_u - x_v| > 1/4 for all x_u in [f_u - 1/3, f_u + 1/3]
    # and x_v in [f_v - 1/3, f_v + 1/3] is equivalent to:
    # |f_u - f_v| > 1/3 + 1/3 + 1/4 = 11/12.
    # Since f_i are integers, this means |f_u - f_v| >= 1.
    # Wait, let's re-evaluate:
    # The range of x_u is [f_u - 1/3, f_u + 1/3].
    # The range of x_v is [f_v - 1/3, f_v + 1/3].
    # The minimum distance between any x_u and x_v is:
    # max(0, |f_u - f_v| - (1/3 + 1/3)) = max(0, |f_u - f_v| - 2/3).
    # We need this to be > 1/4.
    # |f_u - f_v| - 2/3 > 1/4  => |f_u - f_v| > 2/3 + 1/4 = 8/12 + 3/12 = 11/12.
    # Since f_u, f_v are integers, |f_u - f_v| >= 1.
    # This means f_u != f_v.
    # So the problem is simply to color the graph with 3 colors such that
    # no two adjacent nodes have the same color.
    # We need the lexicographically smallest sequence.

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
