```python
import sys

# The problem asks for a 3-coloring of a graph with n <= 40 vertices.
# We need to find the lexicographically smallest coloring (s1, s2, ..., sn).
# This is equivalent to a greedy approach with backtracking.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try assigning unit 1, 2, then 3 to get lexicographically smallest
        for color in range(1, 4):
            can_assign = True
            for neighbor in adj[u]:
                if assignment[neighbor] == color:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    solve()
```
