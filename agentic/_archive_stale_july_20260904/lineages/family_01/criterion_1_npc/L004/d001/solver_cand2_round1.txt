import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    assignment = [0] * (n + 1)

    # Heuristic: Degree-based ordering for backtracking
    # However, we need the lexicographically smallest result,
    # so we must assign 1, 2, ..., n in order.
    # To optimize, we use MRV (Minimum Remaining Values) or simply
    # check constraints efficiently.
    
    def is_safe(u, bus):
        for v in adj[u]:
            if assignment[v] == bus:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try buses 1, 2, 3 in order to ensure lexicographical smallest
        for bus in range(1, 4):
            if is_safe(u, bus):
                assignment[u] = bus
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # For small n (max 40), simple backtracking with pruning is sufficient
    # if we use a bitmask or adjacency check.
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
