import sys

def solve():
    # Valid frequency modes: x! <= x^2
    # 1! = 1 <= 1^2 (1) - Valid
    # 2! = 2 <= 2^2 (4) - Valid
    # 3! = 6 <= 3^2 (9) - Valid
    # 4! = 24 > 4^2 (16) - Invalid
    # So valid modes are {1, 2, 3}
    
    # Interference condition: lcm(f_u, f_v) > gcd(f_u, f_v)
    # Pairs (f_u, f_v) from {1, 2, 3}:
    # (1, 1): lcm=1, gcd=1. 1 > 1 is False.
    # (2, 2): lcm=2, gcd=2. 2 > 2 is False.
    # (3, 3): lcm=3, gcd=3. 3 > 3 is False.
    # (1, 2): lcm=2, gcd=1. 2 > 1 is True.
    # (1, 3): lcm=3, gcd=1. 3 > 1 is True.
    # (2, 3): lcm=6, gcd=1. 6 > 1 is True.
    
    # The condition "lcm > gcd" is equivalent to saying f_u != f_v.
    # This is a graph coloring problem with 3 colors {1, 2, 3}.
    # We need the lexicographically smallest assignment.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
