import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # T1(x) = 2x + 1 mod 3
    # T2(x) = 2x + 2 mod 3
    # T3(x) = 2x mod 3
    # T_u(T_v(x)) = 2(2x + c_v) + c_u = 4x + 2c_v + c_u = x + 2c_v + c_u mod 3
    # Condition: T_u(T_v(x)) != x for all x
    # x + 2c_v + c_u != x mod 3 => 2c_v + c_u != 0 mod 3
    
    # Modes:
    # c=1: 2c = 2
    # c=2: 2c = 4 = 1
    # c=3: 2c = 6 = 0
    
    # Table of 2c_v + c_u mod 3:
    # c_v \ c_u | 1 | 2 | 3
    # ---------------------
    # 1         | 0 | 1 | 2
    # 2         | 2 | 0 | 1
    # 3         | 1 | 2 | 0
    
    # Condition 2c_v + c_u != 0 mod 3 means:
    # If c_v=1, c_u must be 2 or 3
    # If c_v=2, c_u must be 1 or 3
    # If c_v=3, c_u must be 1 or 2
    
    # This is equivalent to c_u != c_v
    
    # Wait, let's recheck:
    # If c_v=1, 2c_v=2. 2+c_u != 0 mod 3 => c_u != 1
    # If c_v=2, 2c_v=1. 1+c_u != 0 mod 3 => c_u != 2
    # If c_v=3, 2c_v=0. 0+c_u != 0 mod 3 => c_u != 3
    
    # So the condition is simply c_u != c_v for all edges (u, v).
    # This is a graph coloring problem with 3 colors.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
