import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    # Heuristic: sort nodes by degree to prune search space faster
    # However, we need lexicographical order for the output, so we must
    # assign 1, 2, ..., n in order.
    
    def is_safe(u, band):
        for v in adj[u]:
            if assignment[v] == band:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for band in [1, 2, 3]:
            if is_safe(u, band):
                assignment[u] = band
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To handle adversarial cases (n=40), we use a simple MRV (Minimum Remaining Values)
    # heuristic combined with backtracking, but since we need lexicographical order,
    # we must be careful. Actually, for n=40, standard backtracking with 
    # constraint propagation (forward checking) is sufficient.
    
    def solve_with_fc(u, domains):
        if u == n:
            return True
        
        for band in [1, 2, 3]:
            if band in domains[u]:
                # Forward checking
                new_domains = [d.copy() for d in domains]
                assignment[u] = band
                
                possible = True
                for v in adj[u]:
                    if assignment[v] == 0:
                        if band in new_domains[v]:
                            new_domains[v].remove(band)
                            if not new_domains[v]:
                                possible = False
                                break
                
                if possible:
                    if solve_with_fc(u + 1, new_domains):
                        return True
                
                assignment[u] = 0
        return False

    domains = [{1, 2, 3} for _ in range(n)]
    if solve_with_fc(0, domains):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
