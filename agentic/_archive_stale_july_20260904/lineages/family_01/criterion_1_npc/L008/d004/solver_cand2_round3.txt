import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # We need to minimize E = sum_{i=1}^n s_i * 4^{n-i}.
    # This is equivalent to lexicographical minimization of (s_1, s_2, ..., s_n).
    # Since n <= 40, we use backtracking with pruning.
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try phases 1, 2, 3 in order to minimize lexicographically
        for phase in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u and assignment[v] == phase:
                    possible = False
                    break
            
            if possible:
                assignment[u] = phase
                if backtrack(u + 1):
                    return True
        
        assignment[u] = 0
        return False

    # The problem asks for the lexicographically smallest sequence (s1, s2, ..., sn).
    # The backtracking approach above naturally finds the lexicographically smallest
    # sequence because it tries 1, then 2, then 3 for each position in order.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
