```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
    
    res = [0] * n
    
    def backtrack(curr):
        if curr == n:
            return True
        
        # Try values 1, 2, 3 in order for lexicographical smallest
        for val in range(1, 4):
            possible = True
            for neighbor in adj[curr]:
                if neighbor < curr and res[neighbor] == val:
                    possible = False
                    break
            
            if possible:
                res[curr] = val
                if backtrack(curr + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
