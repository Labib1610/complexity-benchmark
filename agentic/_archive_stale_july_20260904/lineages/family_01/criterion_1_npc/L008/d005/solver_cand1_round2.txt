import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The characteristic equation t^2 - (s_u + s_v)t + s_u*s_v = 0
    # factors as (t - s_u)(t - s_v) = 0.
    # The roots are s_u and s_v.
    # For the roots to be distinct, we must have s_u != s_v.
    # This is a graph coloring problem with 3 colors {1, 2, 3}.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    # We need to find the lexicographically smallest sequence (s_1, s_2, ..., s_n)
    # where s_i in {1, 2, 3}.
    # Since n <= 40, we can use backtracking with pruning.
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
