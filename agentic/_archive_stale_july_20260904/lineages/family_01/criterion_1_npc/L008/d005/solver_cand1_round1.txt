```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The equation z^2 - (x+y)z + xy = 0 is (z-x)(z-y) = 0.
    # It has a repeated root if and only if x = y.
    # Thus, the condition is simply that no two contacting elements have the same tier.
    # This is equivalent to finding a 3-coloring of the graph.
    # We need to minimize the structural signature S = sum(s_i * 4^(n-i)).
    # This is equivalent to lexicographically smallest sequence (s_1, s_2, ..., s_n).

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
