```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Determinant of M = (s_u + s_v)^2 - 4 * (s_u * s_v)
    # = s_u^2 + 2*s_u*s_v + s_v^2 - 4*s_u*s_v
    # = s_u^2 - 2*s_u*s_v + s_v^2 = (s_u - s_v)^2
    # Singular if (s_u - s_v)^2 == 0, i.e., s_u == s_v.
    # So, no two linked components can have the same scalar.
    # This is equivalent to finding a 3-coloring of the graph.
    # We want to minimize sum(s_i * 4^(n-i)).
    # This is equivalent to lexicographical minimization of (s_1, s_2, ..., s_n).

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
