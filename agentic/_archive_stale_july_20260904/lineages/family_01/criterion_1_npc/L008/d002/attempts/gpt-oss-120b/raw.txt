import sys
sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * n  # bitmask adjacency
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    colors = [0] * n
    # For quick conflict check, maintain for each vertex a mask of used colors by neighbors
    used = [0] * n  # 3-bit mask, bit i means color i+1 is used by a neighbor

    # Precompute neighbor list for iteration
    neigh = [ [j for j in range(n) if (adj[i] >> j) & 1] for i in range(n) ]

    found = False

    def dfs(v: int):
        nonlocal found
        if found:
            return
        if v == n:
            # all assigned
            found = True
            return
        # try colors 1..3
        for c in range(1, 4):
            if used[v] & (1 << (c-1)):
                continue
            # assign
            colors[v] = c
            # update neighbors' used masks
            changed = []
            for nb in neigh[v]:
                if colors[nb] == 0:
                    if not (used[nb] & (1 << (c-1))):
                        used[nb] |= 1 << (c-1)
                        changed.append(nb)
                else:
                    # neighbor already colored, conflict already avoided by used mask
                    pass
            dfs(v+1)
            if found:
                return
            # backtrack
            for nb in changed:
                used[nb] &= ~(1 << (c-1))
            colors[v] = 0

    dfs(0)
    out = sys.stdout
    if not found:
        out.write("NO")
    else:
        out.write("YES\n")
        out.write(" ".join(map(str, colors)))

if __name__ == "__main__":
    solve()
