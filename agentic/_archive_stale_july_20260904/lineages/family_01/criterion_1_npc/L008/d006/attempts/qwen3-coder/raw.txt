from itertools import product

def is_power_of_2(n):
    return n > 0 and (n & (n - 1)) == 0

def solve():
    n, m = map(int, input().split())
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    
    # Precompute invalid pairs: (s_u, s_v) such that 2^s_u + 2^s_v is a power of 2
    invalid_pairs = set()
    for s_u, s_v in product([1, 2, 3], repeat=2):
        if is_power_of_2(2**s_u + 2**s_v):
            invalid_pairs.add((s_u, s_v))
            invalid_pairs.add((s_v, s_u))  # Since the graph is undirected
    
    # Try all 3^n combinations, in lexicographical order to minimize structural signature
    for assignment in product([1, 2, 3], repeat=n):
        valid = True
        for u, v in edges:
            if (assignment[u], assignment[v]) in invalid_pairs:
                valid = False
                break
        if valid:
            print("YES")
            print(' '.join(map(str, assignment)))
            return
    
    print("NO")

solve()
