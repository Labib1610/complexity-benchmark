import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # p_i = (i, i*s_i, i*s_i^2)
    # p_u x p_v = (u*v*s_v*s_u^2 - u*v*s_u*s_v^2, 
    #              u*v*s_u^2 - u*v*s_v^2, 
    #              u*v*s_u - u*v*s_v)
    # = u*v * (s_v*s_u*(s_u - s_v), s_u^2 - s_v^2, s_u - s_v)
    # = u*v * (s_u - s_v) * (s_u*s_v, s_u + s_v, 1)
    # This is zero iff s_u = s_v.
    # So the condition is simply s_u != s_v for all linked (u, v).
    # This is a graph coloring problem with 3 colors.
    # We need to minimize sum(s_i / 4^i).
    # Since 4^i grows very fast, we want to minimize s_1, then s_2, etc.
    # This is equivalent to finding the lexicographically smallest sequence (s_1, ..., s_n).

    colors = [0] * n

    def is_valid(u, c):
        for i, j in adj:
            if i == u and colors[j] == c: return False
            if j == u and colors[i] == c: return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
