 We can solve this by trying all possible assignments of machines to cycles (1, 2, or 3) and checking if any assignment satisfies the constraints. Since n ≤ 40, we can’t brute-force all 3^n directly, but we can use backtracking with pruning.

Here’s the Python solution:

```python
n, m = map(int, input().split())
edges = [tuple(map(int, input().split())) for _ in range(m)]

# Build adjacency list
adj = [[] for _ in range(n)]
for u, v in edges:
    adj[u-1].append(v-1)
    adj[v-1].append(u-1)

assignment = [0] * n
found = False

def backtrack(idx):
    global found
    if found:
        return
    if idx == n:
        found = True
        return
    
    for color in range(1, 4):
        # Check if any neighbor already has this color
        ok = True
        for nb in adj[idx]:
            if assignment[nb] == color:
                ok = False
                break
        if ok:
            assignment[idx] = color
            backtrack(idx + 1)
            if found:
                return
            assignment[idx] = 0

backtrack(0)

if found:
    print("YES")
    print(" ".join(map(str, assignment)))
else:
    print("NO")
```
