def solve():
    import sys
    from collections import defaultdict
    
    n, m = map(int, sys.stdin.readline().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    # Build adjacency list
    adj = defaultdict(list)
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)
        edges.append((u, v))
    
    # Try to 3-color the graph using backtracking with optimization
    coloring = [0] * (n + 1)  # 0 means uncolored, 1,2,3 are colors
    
    def is_safe(node, color):
        for neighbor in adj[node]:
            if coloring[neighbor] == color:
                return False
        return True
    
    # To ensure lexicographically smallest, we process nodes in order 1 to n
    # and for each node, we try colors in order 1, 2, 3
    
    def backtrack(node):
        if node > n:
            return True
            
        for color in range(1, 4):
            if is_safe(node, color):
                coloring[node] = color
                if backtrack(node + 1):
                    return True
                coloring[node] = 0  # backtrack
                
        return False
    
    # Optimization: if any node has degree >= 3, check if it's possible
    # Actually, let's just run the backtracking directly but with a better order
    
    # For better performance, we can try to process nodes with higher degree first
    # But to get lexicographically smallest answer, we need to be careful
    
    # Let's use a simple approach: try all colorings in lexicographical order
    if backtrack(1):
        print("YES")
        print(" ".join(map(str, coloring[1:])))
    else:
        print("NO")

solve()
