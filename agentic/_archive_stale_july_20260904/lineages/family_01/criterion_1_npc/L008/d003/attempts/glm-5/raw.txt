import sys
from functools import lru_cache

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    full = (1 << (n + 1)) - 1

    @lru_cache(maxsize=None)
    def can2(mask):
        if mask == 0:
            return True
        v = (mask & -mask).bit_length() - 1
        rest = mask & ~(1 << v)
        if can2(rest & ~adj[v]):
            return True
        return False

    @lru_cache(maxsize=None)
    def solve_mask(mask):
        if mask == 0:
            return True
        v = (mask & -mask).bit_length() - 1
        rest = mask & ~(1 << v)
        for c in (1, 2, 3):
            forbidden = 0
            for u in range(1, n + 1):
                if mask & (1 << u):
                    if (adj[u] >> v) & 1:
                        forbidden |= 1 << u
            if forbidden & rest:
                continue
            new_mask = rest & ~forbidden
            if c == 1:
                if can2(new_mask):
                    return True
            else:
                if solve_mask(new_mask):
                    return True
        return False

    if not solve_mask(full):
        print("NO")
        return

    color = [0] * (n + 1)
    mask = full
    for v in range(1, n + 1):
        bit = 1 << v
        rest = mask & ~bit
        for c in (1, 2, 3):
            forbidden = 0
            for u in range(1, n + 1):
                if mask & (1 << u):
                    if (adj[u] >> v) & 1:
                        forbidden |= 1 << u
            if forbidden & rest:
                continue
            new_mask = rest & ~forbidden
            if c == 1:
                if can2(new_mask):
                    color[v] = c
                    mask = new_mask
                    break
            else:
                if solve_mask(new_mask):
                    color[v] = c
                    mask = new_mask
                    break

    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
