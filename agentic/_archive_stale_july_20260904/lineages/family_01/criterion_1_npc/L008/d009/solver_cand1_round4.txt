import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Determinant of C_{u,v}
    # C = [[1, s_u+2, s_u^2+4*s_v], [1, s_v, s_v^2], [1, s_v+1, s_v^2+2*s_v]]
    # det = 1*(s_v*(s_v^2+2*s_v) - s_v^2*(s_v+1)) - (s_u+2)*(1*(s_v^2+2*s_v) - s_v^2) + (s_u^2+4*s_v)*(1*(s_v+1) - s_v)
    # det = (s_v^3 + 2*s_v^2 - s_v^3 - s_v^2) - (s_u+2)*(2*s_v) + (s_u^2+4*s_v)*(1)
    # det = s_v^2 - 2*s_u*s_v - 4*s_v + s_u^2 + 4*s_v
    # det = s_v^2 - 2*s_u*s_v + s_u^2 = (s_u - s_v)^2
    # The matrix is singular if det == 0, which means s_u == s_v.
    # So, the condition is s_u != s_v for all linked pairs (u, v).
    # This is equivalent to finding a 3-coloring of the graph.

    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    # To minimize V_n = 4*V_{n-1} + s_n, we need to minimize s_1, then s_2, etc.
    # This is a greedy approach with backtracking.
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
