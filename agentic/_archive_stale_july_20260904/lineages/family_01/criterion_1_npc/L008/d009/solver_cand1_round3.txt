import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The condition "sum of sizes of {s_u, s_v} over all m cables is 2m"
    # means that for every cable (u, v), the size of {s_u, s_v} must be 2.
    # This implies s_u != s_v for all edges (u, v).
    # This is equivalent to saying the graph must be 3-colorable.
    
    # Since n <= 40, we can use backtracking to find the lexicographically 
    # smallest assignment (which minimizes the signature V_n).
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if assignment[v] == color:
                    possible = False
                    break
            
            if possible:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
