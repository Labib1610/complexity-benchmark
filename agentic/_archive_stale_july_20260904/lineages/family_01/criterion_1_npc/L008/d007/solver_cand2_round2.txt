import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        idx += 2

    # Condition: s_u * v + s_v * u != k * (u + v)
    # This is equivalent to: s_u * v + s_v * u != 0 (mod u + v)
    # Since s_u, s_v in {1, 2, 3}, we check all pairs (s_u, s_v)
    
    adj = [[] for _ in range(n + 1)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    # We need to find the lexicographically smallest sequence (s1, s2, ..., sn)
    # where si in {1, 2, 3}.
    # Since n <= 40, we can use backtracking with pruning.
    # The structural signature S = sum(s_i * 4^(n-i)) is minimized by 
    # minimizing s1, then s2, etc.
    
    states = [0] * (n + 1)

    def is_valid(u, s_u):
        for v in adj[u]:
            if states[v] != 0:
                s_v = states[v]
                if (s_u * v + s_v * u) % (u + v) == 0:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        for s in [1, 2, 3]:
            if is_valid(u, s):
                states[u] = s
                if backtrack(u + 1):
                    return True
                states[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(states[1:]))
    else:
        print("NO")

solve()
