import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    assignment = [-1] * n

    def get_mrv_node(assigned):
        best_node = -1
        min_remaining = 4
        
        for i in range(n):
            if assigned[i] == -1:
                used_colors = {assignment[neighbor] for neighbor in adj[i] if assignment[neighbor] != -1}
                remaining = 3 - len(used_colors)
                if remaining < min_remaining:
                    min_remaining = remaining
                    best_node = i
                if min_remaining == 0:
                    return i
        return best_node

    def backtrack():
        node = -1
        # To ensure lexicographical smallest, we must fill 0, 1, 2... in order.
        # However, standard backtracking with MRV is faster.
        # Since n is small (40), we can use a hybrid approach.
        for i in range(n):
            if assignment[i] == -1:
                node = i
                break
        
        if node == -1:
            return True
        
        used = {assignment[neighbor] for neighbor in adj[node] if assignment[neighbor] != -1}
        for color in [1, 2, 3]:
            if color not in used:
                assignment[node] = color
                if backtrack():
                    return True
                assignment[node] = -1
        return False

    # The problem asks for the lexicographically smallest assignment.
    # This means we must try to assign 1 to node 0, then 1 to node 1, etc.
    # The standard backtracking order (0 to n-1) with colors (1, 2, 3)
    # naturally finds the lexicographically smallest result.
    # To avoid the adversarial cases, we use forward checking/pruning.
    
    def solve_lex(node_idx):
        if node_idx == n:
            return True
        
        used = {assignment[neighbor] for neighbor in adj[node_idx] if assignment[neighbor] != -1}
        for color in [1, 2, 3]:
            if color not in used:
                assignment[node_idx] = color
                # Forward checking: check if any unassigned neighbor can still be colored
                possible = True
                for neighbor in adj[node_idx]:
                    if assignment[neighbor] == -1:
                        # Check if this neighbor has at least one color left
                        n_used = {assignment[nn] for nn in adj[neighbor] if assignment[nn] != -1}
                        if len(n_used) == 3:
                            possible = False
                            break
                if possible and solve_lex(node_idx + 1):
                    return True
                assignment[node_idx] = -1
        return False

    if solve_lex(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
