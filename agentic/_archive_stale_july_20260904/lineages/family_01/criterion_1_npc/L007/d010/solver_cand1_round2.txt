import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Resonance scores:
    # A1 = [[0,1,0],[0,0,1],[1,0,0]], A2 = [[0,0,1],[1,0,0],[0,1,0]], A3 = [[1,0,0],[0,1,0],[0,0,1]]
    # Trace(Ai * Aj^T):
    # T(A1*A1^T) = 3, T(A2*A2^T) = 3, T(A3*A3^T) = 3
    # T(A1*A2^T) = 0, T(A1*A3^T) = 0, T(A2*A3^T) = 0
    # So score is 3 if c_u == c_v, and 0 if c_u != c_v.
    # Total score = 3 * (number of edges where c_u == c_v).
    # We need total score = 0, which means c_u != c_v for all edges.
    # This is equivalent to checking if the graph is 3-colorable.
    
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
