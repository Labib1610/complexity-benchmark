import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # M1 = [[-1, 3], [1, 1]], M2 = [[-1, -3], [-1, 1]], M3 = [[2, 0], [0, -2]]
    # Trace(Mi * Mj)
    # M1*M1 = [[4, 0], [0, 4]], Tr = 8
    # M1*M2 = [[-2, 6], [-2, -2]], Tr = -4
    # M1*M3 = [[2, -6], [2, -2]], Tr = 0
    # M2*M2 = [[4, 0], [0, 4]], Tr = 8
    # M2*M3 = [[-2, 6], [-2, -2]], Tr = -4
    # M3*M3 = [[4, 0], [0, 4]], Tr = 8
    
    # Scores:
    # (1,1): 8, (1,2): -4, (1,3): 0
    # (2,1): -4, (2,2): 8, (2,3): -4
    # (3,1): 0, (3,2): -4, (3,3): 8
    
    # We want sum(score(c_u, c_v)) = -4m
    # Let x_i be the profile of node i.
    # Score(1,1)=8, (1,2)=-4, (1,3)=0
    # Score(2,1)=-4, (2,2)=8, (2,3)=-4
    # Score(3,1)=0, (3,2)=-4, (3,3)=8
    
    # This is equivalent to:
    # If we map 1->0, 2->1, 3->2:
    # Score(0,0)=8, (0,1)=-4, (0,2)=0
    # Score(1,0)=-4, (1,1)=8, (1,2)=-4
    # Score(2,0)=0, (2,1)=-4, (2,2)=8
    
    # Let S_ij be the score. We want sum_{(u,v) in E} S_{c_u, c_v} = -4m
    # Note: S_ij = 12 * delta_{i,j} - 4 (if i,j are 0,1 or 1,2) ... no.
    # Let's just use backtracking since n <= 40 is small for some cases, 
    # but actually n=40 is too large for 3^40.
    # However, the constraints and the specific scores suggest a graph coloring/partitioning problem.
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            current_sum = 0
            for i, j in edges:
                c1, c2 = assignment[i], assignment[j]
                if c1 == c2: current_sum += 8
                elif abs(c1 - c2) == 1: current_sum += -4
                else: current_sum += 0
            return current_sum == -4 * m
        
        for c in [1, 2, 3]:
            assignment[u] = c
            if backtrack(u + 1):
                return True
        return False

    # The problem asks for lexicographically smallest.
    # Given the constraints and the nature of the scores, 
    # this is likely a constraint satisfaction problem.
    
    res = [0] * n
    def find_smallest(u):
        if u == n:
            current_sum = 0
            for i, j in edges:
                c1, c2 = res[i], res[j]
                if c1 == c2: current_sum += 8
                elif abs(c1 - c2) == 1: current_sum += -4
                else: current_sum += 0
            return current_sum == -4 * m
        
        for c in [1, 2, 3]:
            res[u] = c
            if find_smallest(u + 1):
                return True
        return False

    if find_smallest(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
