from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    adj = [[] for _ in range(n + 1)]
    edge_list = []
    
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        index += 2
        if u > v:
            u, v = v, u
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        edge_list.append((u, v))
    
    # Sort edges for consistency
    edges.sort()
    
    # We need total energy = -3 * m
    # For each edge (u,v): 
    # If c_u == c_v: v_u · v_v = 2*2 + (-1)*(-1) + (-1)*(-1) = 4 + 1 + 1 = 6
    # If c_u != c_v: v_u · v_v = 2*(-1) + (-1)*2 + (-1)*(-1) = -2 -2 +1 = -3
    # If all edges have c_u != c_v, total = -3*m -> this is what we want
    # So we need a 3-coloring where adjacent nodes have different colors
    
    # This is equivalent to checking if the graph is 3-colorable
    # And among all 3-colorings, find the lexicographically smallest one
    
    # We'll use backtracking with pruning
    color = [0] * (n + 1)  # 0 means uncolored, 1,2,3 are colors
    
    def get_polarization(c):
        v = [-1, -1, -1]
        v[c-1] = 2
        return v
    
    def dot_product(v1, v2):
        return sum(a*b for a,b in zip(v1, v2))
    
    def compute_energy():
        total = 0
        for u, v in edge_list:
            pu = get_polarization(color[u])
            pv = get_polarization(color[v])
            total += dot_product(pu, pv)
        return total
    
    def is_safe(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    found = False
    result_coloring = None
    
    def backtrack(node):
        nonlocal found, result_coloring
        if found:
            return
        if node == n + 1:
            if compute_energy() == -3 * m:
                found = True
                result_coloring = color[:]
            return
        
        if color[node] != 0:
            backtrack(node + 1)
            return
            
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                backtrack(node + 1)
                if found:
                    return
                color[node] = 0
    
    # Optimization: Try to detect if graph is not 3-colorable quickly
    # But since we need exact energy = -3m, let's do full search with early pruning
    
    # Let's try a more efficient approach:
    # Since we want energy = -3m, and each edge contributes -3 when endpoints have different colors,
    # we must have ALL edges connecting nodes of different colors.
    # That means the graph must be 3-colorable, and every edge must be between different colored nodes.
    
    # Check if the graph is 3-colorable first with a greedy approach for early pruning
    def is_3_colorable_greedy():
        temp_color = [0] * (n+1)
        for i in range(1, n+1):
            used = [False] * 4  # index 0 unused
            for nei in adj[i]:
                if temp_color[nei] != 0:
                    used[temp_color[nei]] = True
            for c in [1,2,3]:
                if not used[c]:
                    temp_color[i] = c
                    break
            else:
                return False
        return True
    
    # Even faster: if any vertex has degree >= 3, it might still be 3-colorable
    # But let's just proceed with backtracking with good ordering
    
    # Order vertices by degree descending for better pruning
    order = list(range(1, n+1))
    order.sort(key=lambda x: len(adj[x]), reverse=True)
    
    # But for lex smallest, we should try in order 1..n, but assign in a smart way
    # Let's stick to simple backtracking but optimize the check
    
    def backtrack_optimized(node):
        nonlocal found, result_coloring
        if found:
            return
        if node == n + 1:
            # Verify energy
            if compute_energy() == -3 * m:
                found = True
                result_coloring = color[:]
            return
        
        for c in [1, 2, 3]:
            if is_safe(node, c):
                color[node] = c
                backtrack_optimized(node + 1)
                if found:
                    return
                color[node] = 0
    
    # Special case handling for performance
    if not is_3_colorable_greedy():
        print("NO")
        return
        
    backtrack_optimized(1)
    
    if found:
        print("YES")
        print(" ".join(map(str, result_coloring[1:])))
    else:
        print("NO")

if __name__ == "__main__":
    main()
