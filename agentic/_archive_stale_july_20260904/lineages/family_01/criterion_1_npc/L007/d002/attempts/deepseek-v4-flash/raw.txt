We can model this as a graph coloring problem with 3 colors, where each edge means the two endpoints must have different colors.  
Since \( n \leq 40 \), we can use backtracking with pruning, but we must be careful about the order of trying colors to get the lexicographically smallest assignment.

We’ll try colors in order 1, 2, 3 for each node, and use constraint propagation to prune early.  
We can also use a bitmask representation for possible colors per node.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)

    # possible colors for each node: bitmask of 3 bits
    possible = [0b111] * n
    color = [0] * n  # 0 = uncolored, 1..3 = color

    # We'll do backtracking with forward checking
    def backtrack(idx):
        if idx == n:
            return True
        # Try colors in order 1,2,3
        for c in range(1, 4):
            if (possible[idx] >> (c-1)) & 1:
                # check neighbors
                ok = True
                for nb in adj[idx]:
                    if color[nb] == c:
                        ok = False
                        break
                if not ok:
                    continue
                # assign
                old_possible = [possible[i] for i in range(n)]
                color[idx] = c
                # remove c from neighbors' possible
                for nb in adj[idx]:
                    possible[nb] &= ~(1 << (c-1))
                # check if any neighbor has no possible colors left
                dead = False
                for nb in adj[idx]:
                    if possible[nb] == 0:
                        dead = True
                        break
                if not dead and backtrack(idx+1):
                    return True
                # revert
                color[idx] = 0
                for i in range(n):
                    possible[i] = old_possible[i]
        return False

    if backtrack(0):
        print("YES")
        print(" ".join(str(color[i]) for i in range(n)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
