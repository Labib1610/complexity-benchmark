```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    def is_valid(node, tier):
        for neighbor in adj[node]:
            if assignment[neighbor] == tier:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
        
        for tier in range(1, 4):
            if is_valid(node, tier):
                assignment[node] = tier
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False

    # The problem asks for the lexicographically smallest assignment.
    # Since we assign tiers 1, 2, 3 in order for each node from 1 to n,
    # the first valid configuration found by this backtracking will be
    # the lexicographically smallest one.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
