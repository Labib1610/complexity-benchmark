import sys
sys.setrecursionlimit(100000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # For each edge, we need (c_u - c_v)^2 to be 0, 1, or 4
    # 0: c_u == c_v
    # 1: |c_u - c_v| == 1
    # 4: |c_u - c_v| == 2
    
    # We use backtracking with forward checking
    assignment = [0] * n
    
    # Precompute valid differences for each edge
    # For each pair of nodes (u,v) with edge, store valid (c_u, c_v) pairs
    
    def is_valid_partial(node, val):
        for neighbor in adj[node]:
            if assignment[neighbor] != 0:
                diff = abs(val - assignment[neighbor])
                if diff not in (0, 1, 2):
                    return False
        return True
    
    # Order nodes by degree (most constrained first for efficiency, but we need lexicographically smallest)
    # Actually, we need to try values in order 1, 2, 3 for lexicographically smallest
    # So we process nodes in order 0, 1, 2, ..., n-1
    
    def backtrack(node):
        if node == n:
            return True
        for val in [1, 2, 3]:
            if is_valid_partial(node, val):
                assignment[node] = val
                if backtrack(node + 1):
                    return True
                assignment[node] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, assignment)))
    else:
        print("NO")

solve()
