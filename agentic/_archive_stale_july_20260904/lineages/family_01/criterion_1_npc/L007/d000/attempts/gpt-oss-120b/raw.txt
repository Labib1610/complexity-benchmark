```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # domain masks: bit 0 -> color 1, bit1 -> 2, bit2 ->3
    full_mask = 0b111
    domain = [full_mask] * n
    color = [-1] * n
    sys.setrecursionlimit(10000)

    def dfs(v: int) -> bool:
        if v == n:
            return True
        # try colors 1..3
        for c in range(3):
            mask = 1 << c
            if not (domain[v] & mask):
                continue
            # assign
            color[v] = c + 1
            changed = []
            ok = True
            for nb in adj[v]:
                if color[nb] != -1:
                    if color[nb] == c + 1:
                        ok = False
                        break
                else:
                    if domain[nb] & mask:
                        old = domain[nb]
                        new = old & ~mask
                        if new == 0:
                            ok = False
                            break
                        domain[nb] = new
                        changed.append((nb, old))
            if ok and dfs(v + 1):
                return True
            # revert
            for nb, old in changed:
                domain[nb] = old
            color[v] = -1
        return False

    possible = dfs(0)
    out = []
    if not possible:
        out.append("NO")
    else:
        out.append("YES")
        out.append(" ".join(map(str, color)))
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
