import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # w1 = (-1, sqrt(3)), w2 = (-1, -sqrt(3)), w3 = (2, 0)
    # |w_i + w_j|^2 = |w_i|^2 + |w_j|^2 + 2 * dot(w_i, w_j)
    # |w_1|^2 = 1 + 3 = 4
    # |w_2|^2 = 1 + 3 = 4
    # |w_3|^2 = 4 + 0 = 4
    # dot(w1, w2) = (-1)(-1) + (sqrt(3))(-sqrt(3)) = 1 - 3 = -2
    # dot(w1, w3) = (-1)(2) + (sqrt(3))(0) = -2
    # dot(w2, w3) = (-1)(2) + (-sqrt(3))(0) = -2
    # If c_u == c_v: |w_i + w_i|^2 = |2w_i|^2 = 4 * 4 = 16
    # If c_u != c_v: |w_i + w_j|^2 = 4 + 4 + 2 * (-2) = 8 - 4 = 4
    # Total stress = sum_{edges} |w_u + w_v|^2 = 4m
    # Let k be the number of edges where c_u == c_v.
    # Total stress = k * 16 + (m - k) * 4 = 12k + 4m
    # We want 12k + 4m = 4m => 12k = 0 => k = 0.
    # This means no two connected nodes can have the same channel.
    # This is equivalent to 3-coloring the graph.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if assignment[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
