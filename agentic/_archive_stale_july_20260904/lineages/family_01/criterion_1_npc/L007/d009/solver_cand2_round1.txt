import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    degrees = [len(adj[i]) for i in range(n)]
    
    # U(c) = c^4 - 5c^2
    # U(1) = 1 - 5 = -4
    # U(2) = 16 - 20 = -4
    # U(3) = 81 - 45 = 36
    u_vals = {1: -4, 2: -4, 3: 36}
    
    # X(c1, c2) = c1*c2 * (6*c1*c2 - 4*c1^2 - 4*c2^2 + 10)
    # X(1,1) = 1 * (6 - 4 - 4 + 10) = 8
    # X(1,2) = 2 * (12 - 4 - 16 + 10) = 4
    # X(1,3) = 3 * (18 - 4 - 36 + 10) = -36
    # X(2,2) = 4 * (24 - 16 - 16 + 10) = 8
    # X(2,3) = 6 * (36 - 16 - 36 + 10) = -36
    # X(3,3) = 9 * (54 - 36 - 36 + 10) = -72
    x_vals = {
        (1,1): 8, (1,2): 4, (1,3): -36,
        (2,1): 4, (2,2): 8, (2,3): -36,
        (3,1): -36, (3,2): -36, (3,3): -72
    }
    
    target = -4 * m
    assignment = [0] * n
    
    def backtrack(idx, current_e):
        if idx == n:
            return current_e == target
        
        for c in [1, 2, 3]:
            assignment[idx] = c
            # Calculate contribution of node idx
            # Baseline: d_idx * U(c)
            # Cross-modulation: sum_{v < idx, v is neighbor} X(c_v, c)
            delta = degrees[idx] * u_vals[c]
            for neighbor in adj[idx]:
                if neighbor < idx:
                    delta += x_vals[(assignment[neighbor], c)]
            
            if backtrack(idx + 1, current_e + delta):
                return True
        return False

    if backtrack(0, 0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
