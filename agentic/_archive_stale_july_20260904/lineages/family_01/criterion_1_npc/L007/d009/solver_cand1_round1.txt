import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The condition S4 = 5*S2 - 4*m
    # Let d = |c_u - c_v|. Possible values for d are 0, 1, 2.
    # If d=0: (c_u-c_v)^4 = 0, 5*(c_u-c_v)^2 - 4 = -4. (0 != -4)
    # If d=1: (c_u-c_v)^4 = 1, 5*(c_u-c_v)^2 - 4 = 5(1) - 4 = 1. (1 == 1)
    # If d=2: (c_u-c_v)^4 = 16, 5*(c_u-c_v)^2 - 4 = 5(4) - 4 = 16. (16 == 16)
    # So the condition is satisfied if for every edge, |c_u - c_v| is 1 or 2.
    # This is equivalent to saying c_u != c_v for all edges.
    # This is exactly the 3-coloring problem.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if assignment[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
