import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # To minimize R = sum c_i * 4^(n-i), we want to make c_1 as small as possible,
    # then c_2, and so on. This is a lexicographical minimization problem.
    # We use backtracking with a heuristic: pick the most constrained node next.
    
    assignment = [0] * n
    
    # Precompute degrees for heuristic
    degree = [len(adj[i]) for i in range(n)]
    
    def get_next_node(assigned_mask):
        best_node = -1
        max_constrained = -1
        for i in range(n):
            if not (assigned_mask & (1 << i)):
                # Count neighbors already assigned
                constrained = 0
                for neighbor in adj[i]:
                    if assigned_mask & (1 << neighbor):
                        constrained += 1
                if constrained > max_constrained:
                    max_constrained = constrained
                    best_node = i
        return best_node

    def backtrack(assigned_count, assigned_mask):
        if assigned_count == n:
            return True
        
        u = get_next_node(assigned_mask)
        
        # Try tiers 1, 2, 3 in order to minimize R
        forbidden = 0
        for v in adj[u]:
            if assignment[v] != 0:
                forbidden |= (1 << assignment[v])
        
        for tier in [1, 2, 3]:
            if not (forbidden & (1 << tier)):
                assignment[u] = tier
                if backtrack(assigned_count + 1, assigned_mask | (1 << u)):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0, 0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
