import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * (n + 1)
    for _ in range(m):
        u = int(next(it))
        v = int(next(it))
        adj[u] |= 1 << (v - 1)
        adj[v] |= 1 << (u - 1)
    
    # Greedy coloring to find an upper bound on the number of colors needed
    greedy_color = [0] * (n + 1)
    for u in range(1, n + 1):
        used = [False] * 4
        v_mask = adj[u]
        while v_mask:
            v = (v_mask & -v_mask).bit_length() - 1
            v_mask &= v_mask - 1
            c = greedy_color[v + 1]
            if c > 0:
                used[c] = True
        for c in range(1, 4):
            if not used[c]:
                greedy_color[u] = c
                break
    
    max_color = max(greedy_color[1:]) if n > 0 else 0
    
    if max_color > 3:
        print("NO")
        return
    
    # If greedy found a valid 3-coloring, use it as a bound
    # We need to find the lexicographically smallest valid coloring
    # Using backtracking with pruning
    
    color = [0] * (n + 1)
    found = False
    
    # Precompute neighbors for each node
    neighbors = [[] for _ in range(n + 1)]
    for u in range(1, n + 1):
        mask = adj[u]
        while mask:
            v = (mask & -mask).bit_length()
            neighbors[u].append(v)
            mask &= mask - 1
    
    # Sort nodes by degree for better pruning (optional, but helps)
    # Actually, for lexicographically smallest, we must assign in order 1, 2, ..., n
    # So we process nodes in order
    
    def dfs(u):
        nonlocal found
        if found:
            return
        if u > n:
            found = True
            return
        
        for c in range(1, 4):
            # Check if color c is valid for node u
            valid = True
            for v in neighbors[u]:
                if color[v] == c:
                    valid = False
                    break
            if valid:
                color[u] = c
                dfs(u + 1)
                if found:
                    return
                color[u] = 0
    
    dfs(1)
    
    if found:
        print("YES")
        print(' '.join(map(str, color[1:])))
    else:
        print("NO")

solve()
