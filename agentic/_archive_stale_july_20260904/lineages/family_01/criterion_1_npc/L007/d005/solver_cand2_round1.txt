import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Condition: c_u^2 + c_u*c_v + c_v^2 is NOT a multiple of 3.
    # c in {1, 2, 3}. Modulo 3: 1 -> 1, 2 -> 2, 3 -> 0.
    # Let x, y be the values mod 3.
    # x^2 + xy + y^2 = 0 (mod 3)
    # If x=0: y^2 = 0 => y=0. (3, 3) is forbidden.
    # If x=1: 1 + y + y^2 = 0 (mod 3).
    #   y=0: 1 != 0
    #   y=1: 1+1+1 = 3 = 0. (1, 1) is forbidden.
    #   y=2: 1+2+4 = 7 = 1 != 0.
    # If x=2: 4 + 2y + y^2 = 0 (mod 3) => 1 + 2y + y^2 = (1+y)^2 = 0.
    #   y=2: (1+2)^2 = 9 = 0. (2, 2) is forbidden.
    # Forbidden pairs (c_u, c_v): (1,1), (2,2), (3,3).
    # This means c_u != c_v.
    # The problem is equivalent to finding a 3-coloring of the graph.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
