import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * n
    
    # The condition (c_u - c_v)^2 is not a multiple of 3.
    # (c_u - c_v)^2 % 3 != 0 means (c_u - c_v) % 3 != 0.
    # This is equivalent to c_u % 3 != c_v % 3.
    # Since tiers are {1, 2, 3}, this means c_u % 3 != c_v % 3.
    # Let x_i = c_i % 3. Then x_i in {0, 1, 2}.
    # The condition is x_u != x_v.
    # This is exactly the 3-coloring problem on a graph.
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] != 0:
                if assignment[v] == color:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # To optimize for adversarial cases, use MRV (Minimum Remaining Values) heuristic
    # or simply order by degree. However, for n=40, standard backtracking with 
    # pruning is usually sufficient if the graph is sparse or highly constrained.
    # Given the lexicographical requirement, we must try 1, 2, 3 in order.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
