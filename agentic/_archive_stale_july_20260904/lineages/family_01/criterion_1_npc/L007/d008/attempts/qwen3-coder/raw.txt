from itertools import product

def solve():
    n, m = map(int, input().split())
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    for _ in range(m):
        u, v = map(int, input().split())
        edges.append((u-1, v-1))  # Convert to 0-indexed
    
    # Signature mappings
    sig = {1: 3, 2: 5, 3: 6}  # Decimal values of signatures
    signatures = [3, 5, 6]
    
    # For each edge, precompute the XOR and popcount for all channel pairs
    valid_pairs = {}
    for i in range(3):
        for j in range(3):
            xor_result = signatures[i] ^ signatures[j]
            popcount = bin(xor_result).count('1')
            if popcount == 2:
                if (i+1, j+1) not in valid_pairs:
                    valid_pairs[(i+1, j+1)] = True
                if (j+1, i+1) not in valid_pairs:
                    valid_pairs[(j+1, i+1)] = True
    
    # Build adjacency list
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all possible assignments in lexicographical order
    for assignment in product([1, 2, 3], repeat=n):
        total_score = 0
        valid = True
        
        for u, v in edges:
            c_u, c_v = assignment[u], assignment[v]
            if (c_u, c_v) not in valid_pairs:
                valid = False
                break
            total_score += 2  # Since we only keep pairs with score 2
            
        if valid and total_score == 2 * m:
            print("YES")
            print(" ".join(map(str, assignment)))
            return
    
    print("NO")

solve()
