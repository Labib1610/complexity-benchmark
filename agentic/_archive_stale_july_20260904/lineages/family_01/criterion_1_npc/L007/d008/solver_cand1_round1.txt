import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # H(u, v) = c_u/c_v + c_v/c_u
    # Possible values for (c_u, c_v):
    # (1, 1) -> 1/1 + 1/1 = 2 (Integer)
    # (2, 2) -> 2/2 + 2/2 = 2 (Integer)
    # (3, 3) -> 3/3 + 3/3 = 2 (Integer)
    # (1, 2) -> 1/2 + 2/1 = 2.5 (Non-integer)
    # (1, 3) -> 1/3 + 3/1 = 3.33... (Non-integer)
    # (2, 3) -> 2/3 + 3/2 = 0.66... + 1.5 = 2.166... (Non-integer)
    # Condition: c_u != c_v
    
    assignment = [0] * n

    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != 0 and assignment[v] == val:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
