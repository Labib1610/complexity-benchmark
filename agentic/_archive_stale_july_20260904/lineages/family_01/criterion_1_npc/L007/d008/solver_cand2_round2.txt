```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Signatures: 1: 011 (3), 2: 101 (5), 3: 110 (6)
    # XOR results:
    # 1^2 = 011 ^ 101 = 110 (popcount 2)
    # 1^3 = 011 ^ 110 = 101 (popcount 2)
    # 2^3 = 101 ^ 110 = 011 (popcount 2)
    # 1^1 = 0, 2^2 = 0, 3^3 = 0 (popcount 0)
    # The condition is sum(popcount(xor(u, v))) = 2m.
    # Since popcount is 2 if channels are different and 0 if same,
    # the sum is 2 * (number of edges with different channels).
    # We need 2 * (number of edges with different channels) = 2m,
    # which means all edges must connect nodes with different channels.
    # This is equivalent to checking if the graph is 3-colorable.

    assignment = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if assignment[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                assignment[u] = c
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
```
