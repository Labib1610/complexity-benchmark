import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Delta E(u, v) = deg(gcd(x^3 - 1, x^{c_u} - x^{c_v}))
    # x^3 - 1 = (x-1)(x^2+x+1)
    # If c_u == c_v, x^{c_u} - x^{c_v} = 0, gcd is x^3-1, deg is 3.
    # If c_u != c_v:
    #   (1, 2): x^2 - x = x(x-1), gcd is (x-1), deg is 1.
    #   (1, 3): x^3 - x = x(x^2-1) = x(x-1)(x+1), gcd is (x-1), deg is 1.
    #   (2, 3): x^3 - x^2 = x^2(x-1), gcd is (x-1), deg is 1.
    # So: if c_u == c_v, deg = 3. If c_u != c_v, deg = 1.
    # Total resonance = sum_{edges} deg = m
    # Let k be the number of edges where c_u == c_v.
    # Total = 3*k + 1*(m - k) = 2k + m.
    # We need 2k + m = m, which implies 2k = 0, so k = 0.
    # This means for all edges (u, v), c_u must not equal c_v.
    # This is equivalent to finding a 3-coloring of the graph.
    
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
