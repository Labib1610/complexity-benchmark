```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, shift):
        for v in adj[u]:
            if assignment[v] == shift:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for shift in range(1, 4):
            if is_valid(u, shift):
                assignment[u] = shift
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem asks for the lexicographically smallest sequence.
    # Since n is small (max 40), backtracking with pruning is efficient.
    # The constraints are 8 sectors * 5 workstations = 40 workers max.
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
