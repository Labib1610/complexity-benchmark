import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    assignment = [0] * n
    
    # Find connected components
    visited = [False] * n
    components = []
    
    for i in range(n):
        if not visited[i]:
            comp = []
            stack = [i]
            visited[i] = True
            while stack:
                u = stack.pop()
                comp.append(u)
                for v in adj[u]:
                    if not visited[v]:
                        visited[v] = True
                        stack.append(v)
            components.append(comp)
    
    # Process each component
    for comp in components:
        size = len(comp)
        if size == 1:
            assignment[comp[0]] = 1
            continue
        
        # Build subgraph
        idx_map = {u: i for i, u in enumerate(comp)}
        sub_adj = [[] for _ in range(size)]
        for u in comp:
            ui = idx_map[u]
            for v in adj[u]:
                vi = idx_map[v]
                sub_adj[ui].append(vi)
        
        # Check if component is bipartite
        color = [-1] * size
        is_bipartite = True
        for start in range(size):
            if color[start] == -1:
                color[start] = 0
                stack = [start]
                while stack:
                    u = stack.pop()
                    for v in sub_adj[u]:
                        if color[v] == -1:
                            color[v] = 1 - color[u]
                            stack.append(v)
                        elif color[v] == color[u]:
                            is_bipartite = False
                            break
                    if not is_bipartite:
                        break
            if not is_bipartite:
                break
        
        if is_bipartite:
            # Bipartite component: use shifts 1 and 2
            for i, u in enumerate(comp):
                assignment[u] = 1 if color[i] == 0 else 2
        else:
            # Non-bipartite component: need 3-coloring
            # Use backtracking with constraint propagation
            sub_assignment = [0] * size
            
            # Build adjacency bitmasks
            adj_mask = [0] * size
            for u in range(size):
                mask = 0
                for v in sub_adj[u]:
                    mask |= (1 << v)
                adj_mask[u] = mask
            
            # MRV + degree heuristic
            vertices = list(range(size))
            degrees = [len(sub_adj[i]) for i in range(size)]
            
            def backtrack(pos, sub_assignment, vertices):
                if pos == size:
                    return True
                
                # Select vertex with minimum remaining values
                min_choices = 4
                best_idx = -1
                for i in range(pos, size):
                    v = vertices[i]
                    used = 0
                    for nb in sub_adj[v]:
                        if sub_assignment[nb] != 0:
                            used |= (1 << sub_assignment[nb])
                    choices = 3 - bin(used).count('1')
                    if choices < min_choices:
                        min_choices = choices
                        best_idx = i
                        if choices == 1:
                            break
                
                if min_choices == 0:
                    return False
                
                # Swap to position pos
                vertices[pos], vertices[best_idx] = vertices[best_idx], vertices[pos]
                v = vertices[pos]
                
                # Try colors in order 1, 2, 3
                used = 0
                for nb in sub_adj[v]:
                    if sub_assignment[nb] != 0:
                        used |= (1 << sub_assignment[nb])
                
                for c in range(1, 4):
                    if not (used & (1 << c)):
                        sub_assignment[v] = c
                        if backtrack(pos + 1, sub_assignment, vertices):
                            return True
                        sub_assignment[v] = 0
                
                # Restore
                vertices[pos], vertices[best_idx] = vertices[best_idx], vertices[pos]
                return False
            
            if not backtrack(0, sub_assignment, vertices):
                print("NO")
                return
            
            for i, u in enumerate(comp):
                assignment[u] = sub_assignment[i]
    
    print("YES")
    print(' '.join(map(str, assignment)))

solve()
