def solve():

    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u

    # Precompute for each subset the set of vertices that have a neighbor in the subset
    # neighbor_mask[S] = union of adj[v] for v in S
    # We'll compute on the fly using memoization or DP over subsets.
    # Since n <= 40, we split into two halves.
    half = n // 2
    left_size = half
    right_size = n - half

    # Left part vertices: 0..half-1, right part: half..n-1
    # For a subset of left part (mask over left_size bits), compute neighbor mask within left part
    # and neighbor mask within right part.
    # Similarly for right part.

    # Precompute popcount
    popcnt = [0] * (1 << max(left_size, right_size))
    for i in range(1, len(popcnt)):
        popcnt[i] = popcnt[i >> 1] + (i & 1)

    # For left part
    left_internal = [0] * (1 << left_size)  # neighbors within left part
    left_to_right = [0] * (1 << left_size)  # neighbors in right part (bit positions relative to right part)
    for mask in range(1 << left_size):
        # pick lowest set bit to build incrementally
        if mask:
            low = mask & -mask
            v = low.bit_length() - 1
            prev = mask ^ low
            left_internal[mask] = left_internal[prev] | (adj[v] & ((1 << left_size) - 1))
            # remove v itself from internal neighbor mask if present (self not in adj normally)
            left_to_right[mask] = left_to_right[prev] | ((adj[v] >> left_size) & ((1 << right_size) - 1))
        else:
            left_internal[mask] = 0
            left_to_right[mask] = 0

    def backtrack_3color(neighbors, partial, n):
        # Find next unassigned vertex with MRV
        # avail colors for each vertex
        avail = [0] * n
        for u in range(n):
            if partial[u] != 0:
                avail[u] = 0  # already assigned, not available
            else:
                mask = 0b111
                for v in neighbors[u]:
                    if partial[v] != 0:
                        mask &= ~(1 << (partial[v] - 1))
                avail[u] = mask
                if mask == 0:
                    return False
        # Find unassigned vertex with fewest available colors
        min_count = 4
        chosen = -1
        for u in range(n):
