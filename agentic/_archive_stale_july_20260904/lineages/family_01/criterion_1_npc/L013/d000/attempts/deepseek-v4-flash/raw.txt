We can model this as a graph coloring problem with 3 colors, but since we want the lexicographically smallest assignment, we can use a backtracking search with pruning.  

Since n ≤ 40, a simple backtracking over workers in order, trying shift 1, then 2, then 3, will work because the conflict graph is sparse enough in practice and we prune early when a conflict occurs.

```python
import sys

def solve():
    input_data = sys.stdin.read().strip().split()
    if not input_data:
        return
    it = iter(input_data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        for color in (1, 2, 3):
            ok = True
            for nb in adj[idx]:
                if assignment[nb] == color:
                    ok = False
                    break
            if ok:
                assignment[idx] = color
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
