```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    ans = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[u]:
                if neighbor < u and ans[neighbor] == val:
                    possible = False
                    break
            
            if possible:
                ans[u] = val
                if backtrack(u + 1):
                    return True
        
        ans[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(ans[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
