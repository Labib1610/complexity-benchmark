import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Condition: (p_u + p_v) must NOT divide (p_u^2 + p_v^2)
    # Possible pairs (p_u, p_v) where p_u, p_v in {1, 2, 3}:
    # (1,1): 1+1=2, 1^2+1^2=2. 2 divides 2. INVALID.
    # (1,2): 1+2=3, 1^2+2^2=5. 3 does not divide 5. VALID.
    # (1,3): 1+3=4, 1^2+3^2=10. 4 does not divide 10. VALID.
    # (2,1): 1+2=3, 2^2+1^2=5. 3 does not divide 5. VALID.
    # (2,2): 2+2=4, 2^2+2^2=8. 4 divides 8. INVALID.
    # (2,3): 2+3=5, 2^2+3^2=13. 5 does not divide 13. VALID.
    # (3,1): 3+1=4, 3^2+1^2=10. 4 does not divide 10. VALID.
    # (3,2): 3+2=5, 3^2+2^2=13. 5 does not divide 13. VALID.
    # (3,3): 3+3=6, 3^2+3^2=18. 6 divides 18. INVALID.
    
    # Valid pairs: (1,2), (1,3), (2,1), (2,3), (3,1), (3,2)
    # Essentially: p_u != p_v
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != 0:
                if val == assignment[v]:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
