```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The system:
    # p*x + q*y = 1
    # q*x + p*y = 1
    # Determinant D = p^2 - q^2.
    # Unique solution exists iff D != 0, i.e., p != q and p != -q.
    # Since p, q in {1, 2, 3}, p != -q is always true.
    # Thus, unique solution iff p != q.
    # This is a graph coloring problem: assign colors {1, 2, 3} to nodes
    # such that no two adjacent nodes have the same color.

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
```
