import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The determinant of [[p_u + u, p_v + u], [p_u + v, p_v + v]]
    # = (p_u + u)(p_v + v) - (p_v + u)(p_u + v)
    # = p_u*p_v + p_u*v + u*p_v + u*v - (p_v*p_u + p_v*v + u*p_u + u*v)
    # = p_u*v + u*p_v - p_v*v - u*p_u
    # = p_u(v - u) - p_v(v - u)
    # = (p_u - p_v)(v - u)
    # For the determinant to be non-zero, (p_u - p_v)(v - u) != 0.
    # Since u != v, (v - u) is never 0.
    # Thus, we need p_u - p_v != 0, i.e., p_u != p_v.
    # This is a graph coloring problem with 3 colors.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
