import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Determinant of [[1, pu, pu^2], [1, pv, pv^2], [1, 4, 16]]
    # det = 1*(pv*16 - 4*pv^2) - pu*(1*16 - 1*pv^2) + pu^2*(1*4 - 1*pv)
    # det = 16*pv - 4*pv^2 - 16*pu + pu*pv^2 + 4*pu^2 - pu^2*pv
    # det = (pu - pv)(pu*pv - 4*pu - 4*pv + 16)
    # det = (pu - pv)(pu - 4)(pv - 4)
    # Since pu, pv in {1, 2, 3}, (pu-4) and (pv-4) are never 0.
    # det = 0 iff pu == pv.
    # Thus, the condition is simply that connected nodes must have different frequencies.
    # This is a graph coloring problem with 3 colors.
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
