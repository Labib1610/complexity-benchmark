import sys

def solve():
    # The number of handshakes for k people is k*(k-1)/2 = 780
    # k^2 - k - 1560 = 0
    # (k-40)(k+39) = 0 => k = 40
    # n <= 40
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Frequencies p in {1, 2, 3}
    # Condition: (p_u * p_v) % 4 >= 2
    # 1*1=1 (No), 1*2=2 (Yes), 1*3=3 (Yes)
    # 2*1=2 (Yes), 2*2=4=0 (No), 2*3=6=2 (Yes)
    # 3*1=3 (Yes), 3*2=6=2 (Yes), 3*3=9=1 (No)
    # Valid pairs: (1,2), (1,3), (2,1), (2,3), (3,1), (3,2)
    # Invalid pairs: (1,1), (2,2), (3,3)
    # This means the graph must be bipartite with sets {2} and {1, 3}
    # Wait, let's re-check:
    # If p_u = 2, then p_v must be 1 or 3.
    # If p_u = 1, then p_v must be 2 or 3.
    # If p_u = 3, then p_v must be 1 or 2.
    
    # This is a 3-coloring problem with specific constraints.
    # Total power = sum_{i=1}^n p_i * 4^{n-i}
    # To minimize, we want smaller p_i for smaller i (since 4^{n-i} is larger for smaller i).
    # Greedy approach: try p_i = 1, then 2, then 3.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, p):
        for v in adj[u]:
            if assignment[v] != 0:
                if (p * assignment[v]) % 4 < 2:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for p in [1, 2, 3]:
            if is_valid(u, p):
                assignment[u] = p
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
