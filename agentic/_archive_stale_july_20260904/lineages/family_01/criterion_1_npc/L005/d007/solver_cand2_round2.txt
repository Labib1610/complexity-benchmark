import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # E = sum (p_u - p_v)^2
    # K = count of links where {p_u, p_v} = {1, 3}
    # Condition: E = m + 3K
    # Let n1, n2, n3 be counts of nodes with freq 1, 2, 3
    # Let m12, m13, m23 be counts of links between (1,2), (1,3), (2,3)
    # Let m11, m22, m33 be counts of links between (1,1), (2,2), (3,3)
    # E = m12*(1-2)^2 + m13*(1-3)^2 + m23*(2-3)^2 + m11*0 + m22*0 + m33*0
    # E = m12 + 4*m13 + m23
    # K = m13
    # Condition: m12 + 4*m13 + m23 = m + 3*m13
    # m12 + m13 + m23 = m
    # Since m12 + m13 + m23 + m11 + m22 + m33 = m,
    # the condition is equivalent to m11 + m22 + m33 = 0.
    # This means no two nodes with the same frequency can be connected.
    # This is equivalent to saying the graph must be 3-colorable.
    
    p = [0] * n
    
    def is_valid(node, color):
        for u, v in adj:
            if u == node:
                if p[v] == color: return False
            elif v == node:
                if p[u] == color: return False
        return True

    def backtrack(node):
        if node == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(node, color):
                p[node] = color
                if backtrack(node + 1):
                    return True
                p[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(p))
    else:
        print("NO")

solve()
