```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The condition: discriminant of x^2 - (p_u + p_v)x + p_u*p_v > 0
    # Discriminant D = (p_u + p_v)^2 - 4*p_u*p_v = (p_u - p_v)^2
    # D > 0 means (p_u - p_v)^2 > 0, which is equivalent to p_u != p_v.
    # So the problem is simply to color the graph with 3 colors such that
    # no two adjacent nodes have the same color, minimizing the lexicographical
    # value of the sequence (p_1, p_2, ..., p_n).
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try colors 1, 2, 3 in order to minimize the signature
        for color in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u and assignment[v] == color:
                    possible = False
                    break
            
            if possible:
                assignment[u] = color
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
