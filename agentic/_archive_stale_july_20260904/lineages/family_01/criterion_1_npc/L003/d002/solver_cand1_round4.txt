```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # The objective is to minimize sum(s_i * 4^(n-i)).
    # This is equivalent to lexicographical minimization of (s_1, s_2, ..., s_n).
    # Since n <= 40, we can use backtracking with pruning.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, shift):
        for v in adj[u]:
            if assignment[v] == shift:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try shifts 1, 2, 3 in order to minimize lexicographically
        for shift in [1, 2, 3]:
            if is_valid(u, shift):
                assignment[u] = shift
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem is equivalent to 3-coloring a graph.
    # Since n is small (40), backtracking is efficient enough if we 
    # order the search correctly.
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
