import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u] |= (1 << (v - 1))
        adj[v] |= (1 << (u - 1))
        idx += 2
        
    # We want to minimize sum s_i * 4^(n-i).
    # This is equivalent to lexicographical minimization of (s_1, s_2, ..., s_n).
    # Since we want the smallest s_1, then smallest s_2, etc.,
    # we can use backtracking with pruning.
    
    assignment = [0] * n
    
    def backtrack(worker_idx):
        if worker_idx == n:
            return True
        
        # Try shifts 1, 2, 3 in order to minimize the objective function
        for shift in [1, 2, 3]:
            possible = True
            for prev in range(worker_idx):
                if (adj[worker_idx + 1] & (1 << prev)):
                    if assignment[prev] == shift:
                        possible = False
                        break
            
            if possible:
                assignment[worker_idx] = shift
                if backtrack(worker_idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
