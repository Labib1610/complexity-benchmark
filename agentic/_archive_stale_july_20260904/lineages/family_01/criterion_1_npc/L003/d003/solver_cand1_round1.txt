```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We want to minimize sum_{i=1}^n s_i * 4^{n-i}.
    # This is equivalent to lexicographically minimizing the sequence (s_1, s_2, ..., s_n).
    # Since we want the smallest sequence, we greedily try assigning 1, then 2, then 3.
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try shifts 1, 2, 3 in order to minimize the sequence lexicographically
        for shift in [1, 2, 3]:
            can_assign = True
            for neighbor in adj[u]:
                if neighbor < u and assignment[neighbor] == shift:
                    can_assign = False
                    break
            
            if can_assign:
                assignment[u] = shift
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
