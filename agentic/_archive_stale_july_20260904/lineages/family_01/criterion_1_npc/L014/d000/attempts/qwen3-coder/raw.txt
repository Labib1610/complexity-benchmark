def solve():
    import sys
    from itertools import product
    
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    if m == 0:
        print("YES")
        print(" ".join(["1"] * n))
        return
    
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # Try all possible tag assignments in lexicographical order
    for tags in product([1, 2, 3], repeat=n):
        valid = True
        for u, v in edges:
            if tags[u] == tags[v]:
                valid = False
                break
        if valid:
            print("YES")
            print(" ".join(map(str, tags)))
            return
    
    print("NO")

solve()
