import sys

def solve():
    # The matrix T = [[x, -y], [y, x]] can be represented as a complex number x + iy.
    # T^K corresponds to (x + iy)^K.
    # The top-right entry of T^K is the imaginary part of (x + iy)^K.
    # (x + iy)^K = r^K * (cos(K*theta) + i*sin(K*theta)), where r = sqrt(x^2 + y^2) and theta = atan2(y, x).
    # The imaginary part is r^K * sin(K * atan2(y, x)).
    # This is non-zero if and only if K * atan2(y, x) is not a multiple of pi.
    # Since K = 10^16 is very large, we check if atan2(y, x) / pi is rational.
    # Possible (x, y) pairs from {1, 2, 3}:
    # (1,1): atan2(1,1) = pi/4. K*pi/4 = 10^16 * pi / 4 = 2.5 * 10^15 * pi. Multiple of pi.
    # (2,2): atan2(2,2) = pi/4. Multiple of pi.
    # (3,3): atan2(3,3) = pi/4. Multiple of pi.
    # (1,2): atan2(2,1) = arctan(2). Not a rational multiple of pi.
    # (1,3): atan2(3,1) = arctan(3). Not a rational multiple of pi.
    # (2,3): atan2(3,2) = arctan(1.5). Not a rational multiple of pi.
    # (2,1): atan2(1,2) = arctan(0.5). Not a rational multiple of pi.
    # (3,1): atan2(1,3) = arctan(1/3). Not a rational multiple of pi.
    # (3,2): atan2(2,3) = arctan(2/3). Not a rational multiple of pi.
    # Thus, the condition "top-right entry != 0" is equivalent to x != y.

    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    assignment = [0] * (n + 1)

    def backtrack(u):
        if u > n:
            return True
        for f in range(1, 4):
            possible = True
            for v in adj[u]:
                if v < u and assignment[v] == f:
                    possible = False
                    break
            if possible:
                assignment[u] = f
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
