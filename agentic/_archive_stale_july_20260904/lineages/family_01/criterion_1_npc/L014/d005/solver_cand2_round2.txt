import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: top-right entry of T^K is non-zero.
    # T = [[x, -y], [y, x]]. This is the matrix representation of complex number z = x + iy.
    # T^K corresponds to z^K = (x + iy)^K.
    # The matrix T^K is [[Re(z^K), -Im(z^K)], [Im(z^K), Re(z^K)]].
    # The top-right entry is -Im(z^K).
    # -Im(z^K) != 0 iff Im(z^K) != 0.
    # z = r * e^{i*theta}, where r = sqrt(x^2 + y^2), theta = atan2(y, x).
    # z^K = r^K * (cos(K*theta) + i*sin(K*theta)).
    # Im(z^K) = r^K * sin(K*theta).
    # Since r > 0, Im(z^K) != 0 iff sin(K*theta) != 0, i.e., K*theta is not a multiple of pi.
    # theta = atan2(y, x).
    # Possible (x, y) pairs:
    # (1,1): theta=pi/4, K*theta = 10^16 * pi/4 = 2.5 * 10^15 * pi (multiple of pi) -> NO
    # (1,2): theta=atan2(2,1), K*theta = 10^16 * atan2(2,1) (not multiple of pi) -> YES
    # (1,3): theta=atan2(3,1), K*theta = 10^16 * atan2(3,1) (not multiple of pi) -> YES
    # (2,1): theta=atan2(1,2), K*theta = 10^16 * atan2(1,2) (not multiple of pi) -> YES
    # (2,2): theta=pi/4, K*theta = multiple of pi -> NO
    # (2,3): theta=atan2(3,2), K*theta = 10^16 * atan2(3,2) (not multiple of pi) -> YES
    # (3,1): theta=atan2(1,3), K*theta = 10^16 * atan2(1,3) (not multiple of pi) -> YES
    # (3,2): theta=atan2(2,3), K*theta = 10^16 * atan2(2,3) (not multiple of pi) -> YES
    # (3,3): theta=pi/4, K*theta = multiple of pi -> NO
    # Condition: x != y.

    assignment = [0] * (n + 1)

    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != 0 and assignment[v] == val:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
