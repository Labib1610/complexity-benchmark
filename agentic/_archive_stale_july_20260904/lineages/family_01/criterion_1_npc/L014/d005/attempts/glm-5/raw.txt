import sys
sys.setrecursionlimit(1000000)

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # dp[mask] = True if valid for first k nodes
    # mask encodes assignments for nodes in current window
    # 2 bits per node: 00=1, 01=2, 10=3, 11=invalid
    
    dp = [False] * (1 << (2 * n))
    dp[0] = True
    
    # Precompute venture scores
    # score[a][b] where a,b in {0,1,2} representing firms 1,2,3
    # V1=(1,1,1,1), V2=(1,-1,1,-1), V3=(1,1,-1,-1)
    # score = dot product
    score = [[0]*3 for _ in range(3)]
    for i in range(3):
        for j in range(3):
            if i == 0 and j == 0:
                score[i][j] = 4
            elif i == 0 and j == 1:
                score[i][j] = 0
            elif i == 0 and j == 2:
                score[i][j] = 0
            elif i == 1 and j == 0:
                score[i][j] = 0
            elif i == 1 and j == 1:
                score[i][j] = 4
            elif i == 1 and j == 2:
                score[i][j] = -4
            elif i == 2 and j == 0:
                score[i][j] = 0
            elif i == 2 and j == 1:
                score[i][j] = -4
            elif i == 2 and j == 2:
                score[i][j] = 4
    
    # For each node, precompute neighbors with smaller index
    prev_neighbors = [[] for _ in range(n)]
    for u in range(n):
        for v in adj[u]:
            if v < u:
                prev_neighbors[u].append(v)
    
    # Transition
    for k in range(1, n + 1):
        new_dp = [False] * (1 << (2 * n))
        for mask in range(1 << (2 * n)):
            if not dp[mask]:
                continue
            # Try assigning firm 1, 2, or 3 to node k-1 (0-indexed)
            for firm in range(3):
                # Calculate contribution from edges to previous nodes
                contrib = 0
                valid = True
                for v in prev_neighbors[k-1]:
                    # Get firm of node v from mask
                    shift = 2 * v
                    f_v = (mask >> shift) & 3
                    if f_v == 0:
                        f_v = 0  # firm 1
                    elif f_v == 1:
                        f_v = 1  # firm 2
                    elif f_v == 2:
                        f_v = 2  # firm 3
                    else:
                        valid = False
                        break
                    contrib += score[firm][f_v]
                    contrib += score[f_v][firm]
                if not valid:
                    continue
                # Check if total contribution is 0
                if contrib != 0:
                    continue
                # Create new mask
                new_mask = mask | (firm << (2 * (k-1)))
                new_dp[new_mask] = True
        dp = new_dp
    
    # Find lexicographically smallest valid assignment
    for mask in range(1 << (2 * n)):
        if dp[mask]:
            # Decode mask
            assignment = []
            for i in range(n):
                f = (mask >> (2 * i)) & 3
                assignment.append(f + 1)  # Convert 0,1,2 to 1,2,3
            # Verify
            total = 0
            for u in range(n):
                for v in adj[u]:
                    if u < v:
                        fu = assignment[u] - 1
                        fv = assignment[v] - 1
                        total += score[fu][fv]
            if total == 0:
                print("YES")
                print(' '.join(map(str, assignment)))
                return
    
    print("NO")

solve()
