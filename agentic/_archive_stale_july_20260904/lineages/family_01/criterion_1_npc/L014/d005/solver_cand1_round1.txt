import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Signatures:
    # V1 = (1, 1, 1, 1)
    # V2 = (1, -1, 1, -1)
    # V3 = (1, 1, -1, -1)
    # Dot products:
    # V1.V1 = 4, V1.V2 = 0, V1.V3 = 0
    # V2.V2 = 4, V2.V3 = 0
    # V3.V3 = 4
    # Let x_i be the assignment of department i.
    # Score(u, v) = V_{x_u} . V_{x_v}
    # If x_u == x_v, score is 4.
    # If x_u != x_v, score is 0.
    # Total sum = 4 * (number of edges where x_u == x_v) = 0
    # This implies x_u != x_v for all edges (u, v).
    # This is equivalent to checking if the graph is 3-colorable.
    
    assignment = [0] * n

    def is_valid(u, color):
        for v1, v2 in edges:
            if v1 == u:
                if assignment[v2] == color:
                    return False
            elif v2 == u:
                if assignment[v1] == color:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
