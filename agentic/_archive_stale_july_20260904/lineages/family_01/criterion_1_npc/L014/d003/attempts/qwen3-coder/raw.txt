```python
import sys
from itertools import product

def is_valid_pair(f1, f2):
    """Check if the product of two firm IDs is not a perfect square."""
    prod = f1 * f2
    root = int(prod ** 0.5)
    return root * root != prod

def solve():
    n, m = map(int, sys.stdin.readline().split())
    
    # Read all joint ventures
    edges = []
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        edges.append((u-1, v-1))  # Convert to 0-based indexing
    
    # Try all possible assignments in lexicographical order
    for assignment in product([1, 2, 3], repeat=n):
        valid = True
        for u, v in edges:
            if not is_valid_pair(assignment[u], assignment[v]):
                valid = False
                break
        if valid:
            print("YES")
            print(' '.join(map(str, assignment)))
            return
    
    print("NO")

solve()
```
