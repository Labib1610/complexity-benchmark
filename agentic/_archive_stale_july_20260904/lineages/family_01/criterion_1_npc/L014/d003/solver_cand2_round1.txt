```python
import sys

def solve():
    # The smallest odd prime number is 3.
    # So firms are 1, 2, 3.
    # This is a graph coloring problem with 3 colors.
    # We need to find the lexicographically smallest coloring.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in order to get lexicographically smallest
        neighbor_colors = {colors[v] for v in adj[u] if colors[v] != 0}
        
        for c in range(1, 4):
            if c not in neighbor_colors:
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
