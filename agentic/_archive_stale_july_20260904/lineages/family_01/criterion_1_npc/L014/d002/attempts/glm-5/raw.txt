import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Sort adjacency lists for deterministic processing
    for i in range(n):
        adj[i].sort()
    
    # Check if graph is bipartite (2-colorable)
    # If yes, we can use colors 1,2 and fill rest with 3
    color = [-1] * n
    is_bipartite = True
    
    for start in range(n):
        if color[start] == -1:
            queue = [start]
            color[start] = 0
            for u in queue:
                for v in adj[u]:
                    if color[v] == -1:
                        color[v] = 1 - color[u]
                        queue.append(v)
                    elif color[v] == color[u]:
                        is_bipartite = False
                        break
                if not is_bipartite:
                    break
        if not is_bipartite:
            break
    
    if is_bipartite:
        # Bipartite case: use colors 1,2 for the bipartition, rest get 3
        # For lexicographically smallest, we want as many 1s as possible
        # Then 2s, then 3s
        # For each component, we have two choices of assignment
        # We greedily choose the one that gives more 1s (and then more 2s)
        
        visited = [False] * n
        result = [0] * n
        
        for start in range(n):
            if not visited[start]:
                # BFS to find component
                queue = [start]
                visited[start] = True
                comp = []
                for u in queue:
                    comp.append(u)
                    for v in adj[u]:
                        if not visited[v]:
                            visited[v] = True
                            queue.append(v)
                
                # Count colors in this component
                cnt0 = sum(1 for u in comp if color[u] == 0)
                cnt1 = len(comp) - cnt0
                
                # Choose assignment: color 0 -> firm 1 or firm 2
                # We want more 1s, so if cnt0 >= cnt1, assign 0->1
                # Otherwise assign 0->2
                if cnt0 >= cnt1:
                    for u in comp:
                        if color[u] == 0:
                            result[u] = 1
                        else:
                            result[u] = 2
                else:
                    for u in comp:
                        if color[u] == 0:
                            result[u] = 2
                        else:
                            result[u] = 1
        
        # Fill remaining (isolated nodes) with 1
        for i in range(n):
            if result[i] == 0:
                result[i] = 1
        
        print("YES")
        print(" ".join(map(str, result)))
        return
    
    # Not bipartite - need 3 colors
    # Use backtracking with pruning
    assignment = [0] * n
    
    # Order vertices by degree (descending) for better pruning
    order = list(range(n))
    order.sort(key=lambda x: -len(adj[x]))
    
    # Precompute which vertices come before each vertex in order
    pos_in_order = [0] * n
    for idx, v in enumerate(order):
        pos_in_order[v] = idx
    
    def backtrack(idx):
        if idx == n:
            return True
        
        v = order[idx]
        
        # Find forbidden colors from already-assigned neighbors
        forbidden = [False] * 4
        for neighbor in adj[v]:
            if pos_in_order[neighbor] < idx:  # neighbor already assigned
                forbidden[assignment[neighbor]] = True
        
        # Try colors in order 1, 2, 3 for lexicographically smallest
        for c in [1, 2, 3]:
            if not forbidden[c]:
                assignment[v] = c
                if backtrack(idx + 1):
                    return True
                assignment[v] = 0
        
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

solve()
