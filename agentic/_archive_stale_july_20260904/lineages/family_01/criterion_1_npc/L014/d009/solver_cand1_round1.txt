import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    deg = [0] * n
    for u, v in adj:
        deg[u] += 1
        deg[v] += 1

    # Collaborative loads
    # (1,1): 10, (2,2): 16, (3,3): 26
    # (1,2): 5, (1,3): 10, (2,3): 13
    load = {
        (1, 1): 10, (2, 2): 16, (3, 3): 26,
        (1, 2): 5, (2, 1): 5,
        (1, 3): 10, (3, 1): 10,
        (2, 3): 13, (3, 2): 13
    }

    # Total system risk = sum(f_i^2 * deg_i)
    # Total collaborative load = sum(load(f_u, f_v))
    # Condition: sum(load(f_u, f_v)) = sum(f_i^2 * deg_i)
    # Rearrange: sum(load(f_u, f_v) - f_u^2 * deg_u/deg_u_contribution - f_v^2 * deg_v/deg_v_contribution)
    # Actually, each edge (u, v) contributes load(f_u, f_v) to the sum.
    # Each node i contributes f_i^2 * deg_i to the sum.
    # So: sum_{(u,v) in E} load(f_u, f_v) = sum_{i=1}^n f_i^2 * deg_i
    # sum_{(u,v) in E} (load(f_u, f_v) - f_u^2 - f_v^2) = 0? No, that's not right.
    # The equation is: sum_{(u,v) in E} load(f_u, f_v) = sum_{i=1}^n f_i^2 * deg_i
    # This can be rewritten as: sum_{(u,v) in E} (load(f_u, f_v) - f_u^2 - f_v^2) = 0
    # Wait, each edge (u, v) is counted once in the sum of loads, 
    # but f_i^2 is counted deg_i times in the sum of risks.
    # So the equation is: sum_{(u,v) in E} load(f_u, f_v) = sum_{(u,v) in E} (f_u^2 + f_v^2)
    # Which is: sum_{(u,v) in E} (load(f_u, f_v) - f_u^2 - f_v^2) = 0

    assignment = [0] * n

    def backtrack(idx):
        if idx == n:
            total_load = 0
            total_risk = 0
            for u, v in adj:
                total_load += load[(assignment[u], assignment[v])]
            for i in range(n):
                total_risk += (assignment[i]**2) * deg[i]
            return total_load == total_risk
        
        for f in [1, 2, 3]:
            assignment[idx] = f
            if backtrack(idx + 1):
                return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
