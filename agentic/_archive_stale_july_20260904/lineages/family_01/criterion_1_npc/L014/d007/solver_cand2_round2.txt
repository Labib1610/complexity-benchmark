import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # V1 = (1, 1, 1), V2 = (1, -1, 0), V3 = (1, 1, -2)
    # Dot products:
    # V1.V1 = 3, V1.V2 = 0, V1.V3 = 0
    # V2.V1 = 0, V2.V2 = 2, V2.V3 = 0
    # V3.V3 = 6
    # V2.V3 = 1 - 1 + 0 = 0
    # Wait, let's recompute:
    # V1.V1 = 1+1+1 = 3
    # V1.V2 = 1-1+0 = 0
    # V1.V3 = 1+1-2 = 0
    # V2.V2 = 1+1+0 = 2
    # V2.V3 = 1-1+0 = 0
    # V3.V3 = 1+1+4 = 6
    # Score = sum_{edges (u,v)} V_{f_u} . V_{f_v}
    # Let x_i, y_i, z_i be the components of V_{f_i}
    # Score = sum_{(u,v) in E} (x_u x_v + y_u y_v + z_u z_v)
    # This is (sum x_u x_v) + (sum y_u y_v) + (sum z_u z_v) = 0
    
    # Vectors:
    # f=1: (1, 1, 1)
    # f=2: (1, -1, 0)
    # f=3: (1, 1, -2)
    
    # Let c1, c2, c3 be counts of firms 1, 2, 3.
    # Let x_i, y_i, z_i be the coordinates.
    # Sum_{edges} x_u x_v = sum_{edges} 1*1 = m
    # Sum_{edges} y_u y_v = sum_{edges} y_u y_v
    # Sum_{edges} z_u z_v = sum_{edges} z_u z_v
    # Total = m + sum y_u y_v + sum z_u z_v = 0
    
    # Meet-in-the-middle for n=40 is too slow (3^20 is 3.4e9).
    # But n=40 is small enough for backtracking with pruning.
    
    assignment = [0] * n
    
    def get_score(curr_n):
        s = 0
        for u in range(curr_n):
            for v in adj[u]:
                if v < curr_n:
                    f1, f2 = assignment[u], assignment[v]
                    v1 = [(1,1,1), (1,-1,0), (1,1,-2)][f1-1]
                    v2 = [(1,1,1), (1,-1,0), (1,1,-2)][f2-1]
                    s += v1[0]*v2[0] + v1[1]*v2[1] + v1[2]*v2[2]
        return s // 2

    def backtrack(idx):
        if idx == n:
            return get_score(n) == 0
        
        for f in [1, 2, 3]:
            assignment[idx] = f
            # Pruning: check partial score
            # Only edges where both endpoints are <= idx
            if backtrack(idx + 1):
                return True
        return False

    # The constraints and the nature of the dot products suggest 
    # that the sum is m + sum y_u y_v + sum z_u z_v = 0.
    # Since n=40, we use a simple recursive search.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
