import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        edges.append((u, v))
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    d = [len(adj[i]) for i in range(n + 1)]

    def get_cost(t):
        return t**4 - 5 * (t**2)

    def get_h(tu, tv):
        return 10 * tu * tv + 6 * (tu**2) * (tv**2) - 4 * (tu**3) * tv - 4 * tu * (tv**3)

    # The total metric is sum(d_i * (t_i^4 - 5t_i^2)) + sum(H(u, v))
    # H(tu, tv) = -2 * (tu - tv)^2 * (tu^2 + tv^2 - 5)
    # Let's simplify the expression:
    # Total = sum_{i=1}^n d_i * (t_i^4 - 5t_i^2) + sum_{(u,v) \in E} (-2 * (tu - tv)^2 * (tu^2 + tv^2 - 5))
    # This simplifies to: sum_{(u,v) \in E} (t_u - t_v)^2 * (t_u^2 + t_v^2 - 5) * (-2) + sum d_i * (t_i^4 - 5t_i^2)
    # Actually, the expression simplifies to: sum_{(u,v) \in E} -2 * (t_u - t_v)^2 * (t_u^2 + t_v^2 - 5)
    # Let's check:
    # If t_u = 1, t_v = 2: H = 10(2) + 6(4) - 4(2) - 4(8) = 20 + 24 - 8 - 32 = 4
    # Formula: -2 * (1-2)^2 * (1+4-5) = -2 * 1 * 0 = 0. Wait, the formula is H(u,v) = 10t_u t_v + 6t_u^2 t_v^2 - 4t_u^3 t_v - 4t_u t_v^3
    # H(1,2) = 10(2) + 6(4) - 4(2) - 4(8) = 20 + 24 - 8 - 32 = 4.
    # Let's just use the provided formula directly.

    assignment = [0] * (n + 1)

    def check(k):
        current_sum = 0
        for i in range(1, k + 1):
            current_sum += d[i] * get_cost(assignment[i])
        for u, v in edges:
            if u <= k and v <= k:
                current_sum += get_h(assignment[u], assignment[v])
        
        # Remaining edges: those where one node is > k
        # This is hard to prune. Let's use backtracking.
        return True

    def backtrack(k):
        if k > n:
            total = 0
            for i in range(1, n + 1):
                total += d[i] * get_cost(assignment[i])
            for u, v in edges:
                total += get_h(assignment[u], assignment[v])
            return total == -4 * m

        for t in [1, 2, 3]:
            assignment[k] = t
            if backtrack(k + 1):
                return True
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
