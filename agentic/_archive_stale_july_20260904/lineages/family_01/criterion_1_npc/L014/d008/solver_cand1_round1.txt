import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Let d_i be the degree of department i.
    # Total system risk = sum_{i=1}^n (f_i^2 * d_i)
    # Total collaborative load = sum_{(u,v) in E} (f_u^2 + f_v^2 + 2(f_u - f_v)^4 - 10(f_u - f_v)^2 + 8)
    # Note that sum_{(u,v) in E} (f_u^2 + f_v^2) = sum_{i=1}^n (f_i^2 * d_i)
    # So the condition simplifies to:
    # sum_{(u,v) in E} (2(f_u - f_v)^4 - 10(f_u - f_v)^2 + 8) = 0
    
    # Let g(x, y) = 2(x-y)^4 - 10(x-y)^2 + 8
    # Possible values for x, y in {1, 2, 3}:
    # |x-y| = 0: g = 8
    # |x-y| = 1: g = 2(1)^4 - 10(1)^2 + 8 = 2 - 10 + 8 = 0
    # |x-y| = 2: g = 2(16) - 10(4) + 8 = 32 - 40 + 8 = 0
    
    # So the condition is: sum_{(u,v) in E} g(f_u, f_v) = 0
    # Since g(x, y) is 8 if x=y and 0 if x!=y,
    # the condition is equivalent to: 8 * (number of edges where f_u == f_v) = 0
    # This means f_u != f_v for all edges (u, v).
    # This is exactly the condition that the graph is 3-colorable.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
