import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Interaction value: 10(fu^2 + fv^2) - 11*fu*fv
    # Modulo 3:
    # 10 % 3 = 1, 11 % 3 = 2
    # Value mod 3 = (fu^2 + fv^2 - 2*fu*fv) % 3
    #             = (fu - fv)^2 % 3
    # Possible values for (fu - fv)^2 mod 3:
    # If fu == fv: (0)^2 = 0
    # If fu != fv: (1)^2 = 1, (-1)^2 = 1
    # So, value mod 3 is 0 if fu == fv, and 1 if fu != fv.
    # Total sum of remainders = sum_{edges} (1 if fu != fv else 0)
    # We need this sum to be exactly m.
    # Since each edge contributes at most 1, the sum is m if and only if
    # for every edge (u, v), fu != fv.
    # This is equivalent to saying the graph is 3-colorable.
    
    assignment = [0] * n

    def is_valid(u, color):
        for v1, v2 in edges:
            if v1 == u:
                if assignment[v2] == color:
                    return False
            elif v2 == u:
                if assignment[v1] == color:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
