import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj.append((u, v))
        idx += 2

    # Precompute validity for pairs (a, b) where a, b in {1, 2, 3}
    # Tokens: T1 = a+b, T2 = a*b, T3 = max(a, b)^2
    valid_pairs = []
    for a in range(1, 4):
        for b in range(1, 4):
            t1 = a + b
            t2 = a * b
            t3 = max(a, b) ** 2
            if t1 != t2 and t1 != t3 and t2 != t3:
                valid_pairs.append((a, b))

    assignment = [0] * n

    def is_consistent(u, val):
        for i, j in adj:
            if i == u:
                if assignment[j] != 0:
                    if (val, assignment[j]) not in valid_pairs:
                        return False
            elif j == u:
                if assignment[i] != 0:
                    if (assignment[i], val) not in valid_pairs:
                        return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for val in range(1, 4):
            if is_consistent(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
