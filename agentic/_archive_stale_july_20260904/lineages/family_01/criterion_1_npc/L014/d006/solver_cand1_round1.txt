import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # The formula S(u, v) = (f_u - f_v)^4 - 5(f_u - f_v)^2 + 4
    # Let d = f_u - f_v. Possible values for d are 0, 1, -1, 2, -2.
    # S(0) = 0^4 - 5(0)^2 + 4 = 4
    # S(1) = 1^4 - 5(1)^2 + 4 = 0
    # S(-1) = (-1)^4 - 5(-1)^2 + 4 = 0
    # S(2) = 2^4 - 5(2)^2 + 4 = 16 - 20 + 4 = 0
    # S(-2) = (-2)^4 - 5(-2)^2 + 4 = 16 - 20 + 4 = 0
    # So S(u, v) == 0 if and only if |f_u - f_v| != 0, i.e., f_u != f_v.
    # The condition is that for all edges (u, v), f_u != f_v.
    # This is equivalent to checking if the graph is 3-colorable.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
