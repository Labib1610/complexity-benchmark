import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To find the lexicographically smallest sequence, we try assigning 1, 2, 3
    # in order for each module from 1 to n.
    def backtrack(u):
        if u > n:
            return True
        
        # Try markers 1, 2, 3
        for marker in range(1, 4):
            possible = True
            for neighbor in adj[u]:
                if assignment[neighbor] == marker:
                    possible = False
                    break
            
            if possible:
                assignment[u] = marker
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem asks for the lexicographically smallest sequence.
    # The standard backtracking with 1, 2, 3 order naturally finds the 
    # lexicographically smallest sequence because we fill index 1, then 2, etc.
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
