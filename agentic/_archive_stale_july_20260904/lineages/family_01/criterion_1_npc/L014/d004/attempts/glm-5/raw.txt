import sys
sys.setrecursionlimit(10000)

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # For each edge (u,v), we need first player to win Nim game with piles f[u], f[v].
    # First player wins iff XOR != 0.
    # So for each edge, f[u] XOR f[v] != 0, meaning f[u] != f[v].
    # This is a 3-coloring problem.
    
    # Try to assign colors greedily (lexicographically smallest)
    # For each vertex, try colors 1, 2, 3 in order
    # If a color conflicts with an already-assigned neighbor, skip it
    
    color = [0] * n
    possible = True
    
    for u in range(n):
        found = False
        for c in range(1, 4):
            ok = True
            for v in adj[u]:
                if color[v] == c:
                    ok = False
                    break
            if ok:
                color[u] = c
                found = True
                break
        if not found:
            possible = False
            break
    
    if not possible:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, color)))

if __name__ == "__main__":
    solve()
