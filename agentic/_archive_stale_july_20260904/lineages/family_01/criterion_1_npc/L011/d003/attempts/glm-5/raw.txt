import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    m = int(data[1])
    adj = [0] * (n + 1)
    idx = 2
    for _ in range(m):
        u = int(data[idx]); v = int(data[idx+1])
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    color = [0] * (n + 1)
    comp = []
    visited = [False] * (n + 1)
    
    for s in range(1, n + 1):
        if not visited[s]:
            visited[s] = True
            comp.append([s])
            stack = [s]
            while stack:
                u = stack.pop()
                for v in range(1, n + 1):
                    if (adj[u] >> v) & 1 and not visited[v]:
                        visited[v] = True
                        comp.append([v])
                        stack.append(v)
    
    for c in comp:
        size = len(c)
        if size > 3:
            print("NO")
            return
        if size == 3:
            a, b, d = c[0], c[1], c[2]
            if (adj[a] >> b) & 1 and (adj[a] >> d) & 1 and (adj[b] >> d) & 1:
                print("NO")
                return
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in range(1, n + 1):
                    if (adj[u] >> v) & 1:
                        if color[v] == 0:
                            color[v] = 2 if color[u] == 1 else 1
                            stack.append(v)
                        elif color[v] == color[u]:
                            color[v] = 3
    
    print("YES")
    print(' '.join(str(color[i]) for i in range(1, n + 1)))

solve()
