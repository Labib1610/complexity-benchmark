```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # domain masks: bit 0->color1, bit1->color2, bit2->color3
    full_mask = 0b111
    domain = [full_mask] * n
    assign = [0] * n  # 0 = unassigned, else 1..3

    sys.setrecursionlimit(10000)

    # find first unassigned vertex (smallest index)
    def next_unassigned():
        for i in range(n):
            if assign[i] == 0:
                return i
        return -1

    def dfs() -> bool:
        v = next_unassigned()
        if v == -1:
            return True  # all assigned
        mask = domain[v]
        # try colors 1..3 in order
        for c in (1, 2, 3):
            bit = 1 << (c - 1)
            if mask & bit == 0:
                continue
            # assign c to v
            assign[v] = c
            changed = []  # list of (neighbor, old_mask)
            fail = False
            for nb in adj[v]:
                if assign[nb] == 0:
                    old = domain[nb]
                    if old & bit:
                        new = old & ~bit
                        if new == 0:
                            fail = True
                            # no need to continue updating
                        domain[nb] = new
                        changed.append((nb, old))
                else:
                    if assign[nb] == c:
                        fail = True
            if not fail and dfs():
                return True
            # revert
            for nb, old in reversed(changed):
                domain[nb] = old
            assign[v] = 0
        return False

    possible = dfs()
    out = []
    if possible:
        out.append("YES")
        out.append(" ".join(str(x) for x in assign))
    else:
        out.append("NO")
    sys.stdout.write("\n".join(out))

if __name__ == "__main__":
    solve()
```
