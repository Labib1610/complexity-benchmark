import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Heuristic: order nodes by degree (MRV - Minimum Remaining Values)
    # But we need the lexicographically first sequence for 1..n.
    # So we must assign 1, 2, ..., n in order.
    # To optimize, we use forward checking or constraint propagation.
    
    assignment = [0] * n
    
    def get_possible_modes(u, current_assignment):
        forbidden = set()
        for v in adj[u]:
            if current_assignment[v] != 0:
                forbidden.add(current_assignment[v])
        return [m for m in [1, 2, 3] if m not in forbidden]

    # Pre-calculate domains for all nodes
    domains = [[1, 2, 3] for _ in range(n)]
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try modes 1, 2, 3 in order
        for mode in [1, 2, 3]:
            # Check if mode is valid
            valid = True
            for v in adj[u]:
                if assignment[v] == mode:
                    valid = False
                    break
            
            if valid:
                assignment[u] = mode
                
                # Forward checking: check if any unassigned neighbor has 0 options
                possible = True
                for v in adj[u]:
                    if v > u:
                        # Check if v still has at least one option
                        has_option = False
                        for m_v in [1, 2, 3]:
                            is_ok = True
                            for neighbor in adj[v]:
                                if neighbor < u:
                                    if assignment[neighbor] == m_v:
                                        is_ok = False; break
                                elif neighbor == u:
                                    if mode == m_v:
                                        is_ok = False; break
                            if is_ok:
                                has_option = True; break
                        if not has_option:
                            possible = False; break
                
                if possible and backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
