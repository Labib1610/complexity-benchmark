import sys

def solve():
    # The recurrence is X_k = 3*X_{k-1} - 2*X_{k-2}.
    # The characteristic equation is r^2 - 3r + 2 = 0, which is (r-1)(r-2) = 0.
    # So X_k = A * 1^k + B * 2^k = A + B * 2^k.
    # X_0 = A + B = x
    # X_1 = A + 2B = y
    # Subtracting: B = y - x, A = x - (y - x) = 2x - y.
    # X_k = (2x - y) + (y - x) * 2^k.
    # Stability condition: X_{10^100} != X_0.
    # X_{10^100} = (2x - y) + (y - x) * 2^{10^100}.
    # X_0 = x.
    # Condition: (2x - y) + (y - x) * 2^{10^100} != x
    # x - y + (y - x) * 2^{10^100} != 0
    # (x - y) * (1 - 2^{10^100}) != 0
    # Since 1 - 2^{10^100} is never 0, this is equivalent to x - y != 0, i.e., x != y.
    # So the condition is simply that for every interacting pair (u, v), c_u != c_v.
    # This is the standard graph coloring problem with 3 colors.
    
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    colors = [0] * n
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True
    
    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
