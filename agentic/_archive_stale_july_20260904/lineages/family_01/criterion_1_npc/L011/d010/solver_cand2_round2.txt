```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Harmonic condition: c_u^2 + c_v^2 + c_u * c_v is NOT divisible by 3.
    # c in {1, 2, 3}
    # Modulo 3:
    # 1^2 = 1, 2^2 = 4 = 1, 3^2 = 9 = 0
    # If c = 3, c^2 + c_v^2 + c*c_v = 0 + c_v^2 + 0 = c_v^2.
    # If c_v = 1, c_v^2 = 1 (not div by 3) -> OK
    # If c_v = 2, c_v^2 = 1 (not div by 3) -> OK
    # If c_v = 3, c_v^2 = 0 (div by 3) -> BAD
    # So 3 cannot be connected to 3.
    
    # If c_u, c_v in {1, 2}:
    # 1, 1: 1+1+1 = 3 (div by 3) -> BAD
    # 1, 2: 1+4+2 = 7 (not div by 3) -> OK
    # 2, 2: 4+4+4 = 12 (div by 3) -> BAD
    
    # Summary:
    # 1 and 1 are incompatible
    # 2 and 2 are incompatible
    # 3 and 3 are incompatible
    # 1 and 2 are compatible
    # 1 and 3 are compatible
    # 2 and 3 are compatible
    
    # This is equivalent to saying c_u != c_v.
    # Wait, let's recheck:
    # 1, 1: 1+1+1 = 3 (div by 3) -> BAD
    # 2, 2: 4+4+4 = 12 (div by 3) -> BAD
    # 3, 3: 0+0+0 = 0 (div by 3) -> BAD
    # 1, 2: 1+4+2 = 7 (not div by 3) -> OK
    # 1, 3: 1+0+0 = 1 (not div by 3) -> OK
    # 2, 3: 4+0+0 = 4 (not div by 3) -> OK
    
    # Yes, the condition is simply c_u != c_v.
    # The problem is to color the graph with 3 colors such that no two adjacent
    # nodes have the same color, finding the lexicographically smallest sequence.
    
    res = [0] * (n + 1)
    
    def backtrack(u):
        if u > n:
            return True
        
        for color in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u and res[v] == color:
                    possible = False
                    break
            if possible:
                res[u] = color
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(1):
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

solve()
```
