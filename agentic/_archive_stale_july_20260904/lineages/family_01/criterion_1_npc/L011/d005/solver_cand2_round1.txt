import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # P_{u,v}(x) = \int_0^x (t - c_u)(t - c_v) dt
    # P_{u,v}(x) = \int_0^x (t^2 - (c_u + c_v)t + c_u c_v) dt
    # P_{u,v}(x) = x^3/3 - (c_u + c_v)x^2/2 + c_u c_v x
    # P'_{u,v}(x) = x^2 - (c_u + c_v)x + c_u c_v = (x - c_u)(x - c_v)
    # Local maximum occurs at x > 0 if P'_{u,v}(x) changes from + to -.
    # P'_{u,v}(x) is a parabola opening upwards.
    # It has roots at c_u and c_v.
    # If c_u != c_v, the roots are distinct. The parabola is negative between roots.
    # If c_u < c_v, P' is positive on (0, c_u), negative on (c_u, c_v), positive on (c_v, inf).
    # Local max at x = c_u if c_u > 0.
    # If c_u = c_v, P' = (x - c_u)^2, which is >= 0. No local max.
    # So we need c_u != c_v and min(c_u, c_v) > 0.
    # Since c_i \in {1, 2, 3}, min(c_u, c_v) is always > 0.
    # Thus, we need c_u != c_v for all interacting pairs (u, v).
    # This is exactly the 3-coloring problem.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
