import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    d = [0] * n
    for u, v in edges:
        d[u] += 1
        d[v] += 1

    # The condition alpha = beta simplifies:
    # alpha = 4m + sum((c_u - c_v)^4)
    # beta = 5 * sum(d_i * c_i^2) - 10 * sum(c_u * c_v)
    # Let f(c_u, c_v) = (c_u - c_v)^4 + 10 * c_u * c_v
    # alpha = 4m + sum((c_u - c_v)^4)
    # beta = 5 * sum(d_i * c_i^2) - 10 * sum(c_u * c_v)
    # alpha - beta = 4m + sum((c_u - c_v)^4 + 10 * c_u * c_v) - 5 * sum(d_i * c_i^2) = 0
    
    # Precompute values for the sum
    # Let g(c_u, c_v) = (c_u - c_v)^4 + 10 * c_u * c_v
    # Let h(c_i) = 5 * d_i * c_i^2
    # Condition: 4m + sum_{(u,v) in E} g(c_u, c_v) - sum_{i=1}^n h(c_i) = 0

    c = [0] * n

    def backtrack(idx):
        if idx == n:
            alpha = 4 * m
            for u, v in edges:
                alpha += (c[u] - c[v])**4
            beta = 0
            for i in range(n):
                beta += 5 * d[i] * (c[i]**2)
            for u, v in edges:
                beta -= 10 * c[u] * c[v]
            return alpha == beta

        for val in [1, 2, 3]:
            c[idx] = val
            if backtrack(idx + 1):
                return True
        return False

    if backtrack(0):
        print("YES")
        print(*(c))
    else:
        print("NO")

solve()
