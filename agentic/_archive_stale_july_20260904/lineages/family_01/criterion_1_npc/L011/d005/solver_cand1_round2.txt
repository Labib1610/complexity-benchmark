import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    # We want to minimize sum_{i=1}^n c_i * 4^{n-i}.
    # This is equivalent to lexicographical minimization of (c_1, c_2, ..., c_n).
    # Since n <= 40, we can use backtracking with pruning.
    
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try assigning 1, 2, 3 in order to minimize lexicographically
        forbidden = 0
        for v in range(u):
            if (adj[u] >> v) & 1:
                forbidden |= (1 << assignment[v])
        
        for val in [1, 2, 3]:
            if not ((forbidden >> val) & 1):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
