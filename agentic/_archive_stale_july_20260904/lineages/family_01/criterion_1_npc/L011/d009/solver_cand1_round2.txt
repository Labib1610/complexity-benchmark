import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Dot products:
    # A=(1,1,0), B=(1,0,1), C=(0,1,1)
    # A.A = 2, A.B = 1, A.C = 1
    # B.A = 1, B.B = 2, B.C = 1
    # C.A = 1, C.B = 1, C.C = 2
    # Condition V_u . V_v = 1 means:
    # If u=A, v must be B or C
    # If u=B, v must be A or C
    # If u=C, v must be A or B
    # This is equivalent to saying V_u != V_v
    
    # This is a graph coloring problem with 3 colors where adjacent nodes
    # must have different colors.
    # We need the lexicographically smallest sequence.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

solve()
