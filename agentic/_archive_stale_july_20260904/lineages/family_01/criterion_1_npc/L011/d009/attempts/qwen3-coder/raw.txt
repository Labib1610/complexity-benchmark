from collections import defaultdict
import sys

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list
    adj = defaultdict(list)
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1  # Convert to 0-based indexing
        v = int(data[index+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Define the three vectors and their dot products
    vectors = [
        (1, 1, 0),  # 1
        (1, 0, 1),  # 2
        (0, 1, 1)   # 3
    ]
    
    def dot_product(a, b):
        return a[0]*b[0] + a[1]*b[1] + a[2]*b[2]
    
    # Precompute which assignments are compatible
    compatible = [[False]*3 for _ in range(3)]
    for i in range(3):
        for j in range(3):
            if dot_product(vectors[i], vectors[j]) == 1:
                compatible[i][j] = True
    
    # Try all 3^n assignments in lexicographical order using backtracking
    assignment = [-1] * n  # -1 means unassigned
    
    def is_valid(node, color):
        for neighbor in adj[node]:
            if assignment[neighbor] != -1:
                if not compatible[color][assignment[neighbor]]:
                    return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
            
        for color in range(3):  # Try colors 0,1,2 (representing 1,2,3)
            if is_valid(node, color):
                assignment[node] = color
                if backtrack(node + 1):
                    return True
                assignment[node] = -1
        return False
    
    if backtrack(0):
        print("YES")
        result = [str(c + 1) for c in assignment]
        print(" ".join(result))
    else:
        print("NO")

if __name__ == "__main__":
    main()
