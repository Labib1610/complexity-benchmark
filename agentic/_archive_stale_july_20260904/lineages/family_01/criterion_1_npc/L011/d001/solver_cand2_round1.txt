import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Calculate forbidden frequencies for beacon u
        forbidden = 0
        for v in range(u):
            if (adj[u] >> v) & 1:
                forbidden |= (1 << (assignment[v] - 1))
        
        # Try frequencies 1, 2, 3 in order for lexicographical smallest
        for f in range(1, 4):
            if not (forbidden & (1 << (f - 1))):
                assignment[u] = f
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # Optimization: Sort nodes by degree descending to prune search space faster
    # However, the problem requires lexicographical smallest output for 1..n.
    # Since n is small (<= 40), we use a simple backtracking with pruning.
    # To handle the adversarial cases, we use the fact that 3-coloring is NP-complete,
    # but n=40 is small enough for optimized backtracking.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
