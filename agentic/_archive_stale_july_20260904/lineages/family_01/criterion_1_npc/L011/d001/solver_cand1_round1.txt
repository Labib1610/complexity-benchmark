import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2
        
    assignment = [0] * n
    
    # To find the lexicographically smallest assignment, we try assigning 1, 2, 3
    # in order for each node from 0 to n-1.
    def backtrack(u):
        if u == n:
            return True
        
        # Find which channels are taken by neighbors already assigned
        forbidden = 0
        for v in range(u):
            if (adj[u] & (1 << v)):
                forbidden |= (1 << (assignment[v] - 1))
        
        for channel in range(1, 4):
            if not (forbidden & (1 << (channel - 1))):
                assignment[u] = channel
                if backtrack(u + 1):
                    return True
        return False

    # The problem asks for the lexicographically smallest sequence.
    # The standard backtracking approach with greedy choice (1, 2, 3)
    # naturally explores the lexicographical order.
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
