import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [0] * n
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u] |= 1 << v
        adj[v] |= 1 << u
    
    # Precompute connected components
    visited = 0
    components = []
    
    for i in range(n):
        if not (visited >> i) & 1:
            # BFS to find component
            queue = [i]
            visited |= 1 << i
            comp_nodes = []
            while queue:
                u = queue.pop()
                comp_nodes.append(u)
                neighbors = adj[u]
                # Iterate through neighbors
                v = 0
                while neighbors:
                    if neighbors & 1:
                        if not (visited >> v) & 1:
                            visited |= 1 << v
                            queue.append(v)
                    neighbors >>= 1
                    v += 1
            components.append(comp_nodes)
    
    # For each component, find all valid 3-colorings
    comp_colorings = []
    
    for comp in components:
        size = len(comp)
        if size == 0:
            comp_colorings.append([[]])
            continue
            
        # Map local index to global index
        local_to_global = comp
        global_to_local = {g: l for l, g in enumerate(comp)}
        
        # Build local adjacency
        local_adj = [0] * size
        for l, g in enumerate(comp):
            mask = 0
            neighbors = adj[g]
            # Iterate through neighbors
            v = 0
            while neighbors:
                if neighbors & 1:
                    if v in global_to_local:
                        mask |= 1 << global_to_local[v]
                neighbors >>= 1
                v += 1
            local_adj[l] = mask
        
        # Find all valid colorings for this component
        # Using backtracking with pruning
        colorings = []
        colors = [0] * size
        
        def backtrack(idx):
            if idx == size:
                colorings.append(colors[:])
                return
            
            # Find used colors by neighbors
            forbidden = 0
            neighbor_mask = local_adj[idx]
            j = 0
            mask = neighbor_mask
            while mask:
                if mask & 1:
                    if colors[j] != 0:
                        forbidden |= 1 << (colors[j] - 1)
                mask >>= 1
                j += 1
            
            for c in range(1, 4):
                if not (forbidden >> (c - 1)) & 1:
                    colors[idx] = c
                    backtrack(idx + 1)
                    colors[idx] = 0
        
        backtrack(0)
        
        if not colorings:
            print("NO")
            return
        
        comp_colorings.append(colorings)
    
    # Combine colorings to find lexicographically smallest
    # We need to track the best global assignment
    best_global = None
    
    def combine(comp_idx, current_global):
        nonlocal best_global
        
        if comp_idx == len(components):
            # Check if this is better than current best
            if best_global is None:
                best_global = current_global[:]
            else:
                # Compare lexicographically
                for i in range(n):
                    if current_global[i] < best_global[i]:
                        best_global = current_global[:]
                        break
                    elif current_global[i] > best_global[i]:
                        break
            return
        
        comp = components[comp_idx]
        colorings = comp_colorings[comp_idx]
        
        # Pruning: if we already have a best_global, check if current prefix can beat it
        if best_global is not None:
            can_be_better = False
            for i in range(n):
                if i in set(comp):
                    break
                if current_global[i] < best_global[i]:
                    can_be_better = True
                    break
                elif current_global[i] > best_global[i]:
                    return  # Can't be better
            if not can_be_better:
                # Check if all prefix so far equals best_global
                all_equal = True
                for i in range(n):
                    if i in set(comp):
                        break
                    if current_global[i] != best_global[i]:
                        all_equal = False
                        break
                if all_equal:
                    # Need to check if any coloring can make it better or equal
                    pass  # Continue
        
        for coloring in colorings:
            # Apply coloring
            for l, g in enumerate(comp):
                current_global[g] = coloring[l]
            combine(comp_idx + 1, current_global)
            
            # Early termination if we found the optimal
            # The first coloring tried for each component is lexicographically smallest
            # So if we complete one full assignment, it's optimal
            if best_global is not None and comp_idx == 0:
                # Check if this is the first complete assignment
                pass
    
    # Optimization: sort colorings by their contribution to global lex order
    # For each component, sort colorings by the order they appear in global sequence
    for i, comp in enumerate(components):
        colorings = comp_colorings[i]
        # Sort colorings by their global sequence
        def coloring_key(coloring):
            return tuple(coloring[comp.index(g)] if g in set(comp) else 0 for g in range(n))
        # Actually, we need to sort by the positions in comp
        # The key is the tuple of colors in order of global indices
        colorings.sort(key=lambda c: tuple(c[comp.index(g)] for g in sorted(comp)))
        comp_colorings[i] = colorings
    
    # Sort components by their minimum global index
    comp_order = sorted(range(len(components)), key=lambda i: min(components[i]))
    
    # Reorder
    ordered_components = [components[i] for i in comp_order]
    ordered_colorings = [comp_colorings[i] for i in comp_order]
    
    best_global = None
    
    def combine_opt(comp_idx, current_global):
        nonlocal best_global
        
        if comp_idx == len(ordered_components):
            if best_global is None:
                best_global = current_global[:]
            return
        
        comp = ordered_components[comp_idx]
        colorings = ordered_colorings[comp_idx]
        
        for coloring in colorings:
            for l, g in enumerate(comp):
                current_global[g] = coloring[l]
            combine_opt(comp_idx + 1, current_global)
            if best_global is not None:
                return  # Found optimal for this branch
    
    # Actually, let's be smarter. Since we want lexicographically smallest,
    # and components are independent, we should process components in order
    # of their smallest global index, and for each component, try colorings
    # in order that minimizes the global sequence.
    
    # Rebuild with proper sorting
    comp_info = []
    for i, comp in enumerate(components):
        min_idx = min(comp)
        colorings = comp_colorings[i]
        # Sort colorings by the colors at global indices in this component
        sorted_comp = sorted(comp)
        colorings.sort(key=lambda c: tuple(c[comp.index(g)] for g in sorted_comp))
        comp_info.append((min_idx, comp, colorings))
    
    comp_info.sort(key=lambda x: x[0])
    
    best_global = [0] * n
    
    def combine_final(comp_idx, current_global):
        if comp_idx == len(comp_info):
            return True
        
        _, comp, colorings = comp_info[comp_idx]
        
        for coloring in colorings:
            for l, g in enumerate(comp):
                current_global[g] = coloring[l]
            if combine_final(comp_idx + 1, current_global):
                return True
        return False
    
    result = [0] * n
    if combine_final(0, result):
        print("YES")
        print(" ".join(map(str, result)))
    else:
        print("NO")

solve()
