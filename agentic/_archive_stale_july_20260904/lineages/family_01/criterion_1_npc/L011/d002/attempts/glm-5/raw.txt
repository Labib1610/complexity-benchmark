```python
import sys

def solve():
    data = sys.stdin.read().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)
    
    assignment = [0] * n
    
    def dfs(u):
        for color in range(1, 4):
            conflict = False
            for v in adj[u]:
                if v < u and assignment[v] == color:
                    conflict = True
                    break
            if conflict:
                continue
            assignment[u] = color
            if u + 1 == n:
                return True
            if dfs(u + 1):
                return True
            assignment[u] = 0
        return False
    
    if dfs(0):
        print("YES")
        print(" ".join(map(str, assignment)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
