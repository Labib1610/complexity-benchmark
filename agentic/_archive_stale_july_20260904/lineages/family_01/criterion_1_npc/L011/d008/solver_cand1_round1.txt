import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u] |= (1 << v)
        adj[v] |= (1 << u)
        idx += 2

    assignment = [0] * n

    def can_color(idx, current_assignment):
        # Check if a valid coloring exists for remaining nodes
        # using a simple backtracking with pruning or bitmask
        # Given n=40, we use a greedy approach with backtracking
        # and bitmask to represent constraints.
        
        def backtrack(k):
            if k == n:
                return True
            
            # Try assigning squad 1, 2, 3
            forbidden = 0
            for i in range(k):
                if (adj[k] >> i) & 1:
                    forbidden |= (1 << (current_assignment[i] - 1))
            
            for squad in range(1, 4):
                if not ((forbidden >> (squad - 1)) & 1):
                    current_assignment[k] = squad
                    if backtrack(k + 1):
                        return True
            return False

        return backtrack(idx)

    # To satisfy the hierarchical preference, we greedily pick the smallest squad
    # for each participant i from 1 to n, provided a valid completion exists.
    
    res = [0] * n
    for i in range(n):
        found = False
        for squad in range(1, 4):
            # Check if this squad is valid for participant i
            forbidden = 0
            for prev in range(i):
                if (adj[i] >> prev) & 1:
                    if res[prev] == squad:
                        forbidden = 1
                        break
            if forbidden:
                continue
            
            res[i] = squad
            # Check if this choice allows a valid completion
            if can_color(i + 1, res):
                found = True
                break
        
        if not found:
            print("NO")
            return

    print("YES")
    print(*(res))

solve()
