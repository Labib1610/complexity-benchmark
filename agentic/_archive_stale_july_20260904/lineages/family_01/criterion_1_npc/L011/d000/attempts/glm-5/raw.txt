```python
import sys

# Increase recursion depth just in case, though n is small
sys.setrecursionlimit(10000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    iterator = iter(input_data)
    try:
        n = int(next(iterator))
        m = int(next(iterator))
    except StopIteration:
        return

    adj = [[] for _ in range(n)]
    
    for _ in range(m):
        u = int(next(iterator)) - 1
        v = int(next(iterator)) - 1
        adj[u].append(v)
        adj[v].append(u)

    # Colors: 0 = uncolored, 1, 2, 3 are the channels
    colors = [0] * n
    
    # We use backtracking to find the lexicographically smallest valid coloring.
    # Since we try colors in increasing order (1, 2, 3), the first valid
    # solution found will be the lexicographically smallest.
    
    def backtrack(u):
        # Determine available colors for node u
        # A color is available if no adjacent colored node uses it
        used = set()
        for v in adj[u]:
            if colors[v] != 0:
                used.add(colors[v])
        
        for c in range(1, 4):
            if c not in used:
                colors[u] = c
                # Find next uncolored node
                # We iterate in index order to ensure lexicographical priority
                next_node = -1
                for i in range(u + 1, n):
                    if colors[i] == 0:
                        next_node = i
                        break
                
                if next_node == -1:
                    # All nodes colored
                    return True
                
                if backtrack(next_node):
                    return True
                
                # Backtrack
                colors[u] = 0
        
        return False

    # Start backtracking from the first node (index 0)
    if backtrack(0):
        sys.stdout.write("YES\n")
        sys.stdout.write(" ".join(map(str, colors)) + "\n")
    else:
        sys.stdout.write("NO\n")

if __name__ == '__main__':
    solve()
```
