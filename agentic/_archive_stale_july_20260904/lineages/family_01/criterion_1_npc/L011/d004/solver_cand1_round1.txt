import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # E(u, v) = 5(c_u - c_v)^2 - (c_u - c_v)^4
    # If |c_u - c_v| = 0: E = 0
    # If |c_u - c_v| = 1: E = 5(1) - 1 = 4
    # If |c_u - c_v| = 2: E = 5(4) - 16 = 4
    # So E(u, v) = 4 if c_u != c_v, and 0 if c_u == c_v.
    # The condition sum(E) = 4m means that for every edge (u, v),
    # we must have c_u != c_v.
    # This is equivalent to saying the graph must be 3-colorable.
    # We need the lexicographically smallest 3-coloring.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
