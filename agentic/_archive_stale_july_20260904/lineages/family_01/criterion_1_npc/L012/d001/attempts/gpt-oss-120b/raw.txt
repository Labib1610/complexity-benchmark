```python
import sys

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [set() for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].add(v)
        adj[v].add(u)

    colors = [0] * n
    sys.setrecursionlimit(10000)

    def dfs(i: int) -> bool:
        if i == n:
            return True
        for c in (1, 2, 3):
            ok = True
            for nb in adj[i]:
                if nb < i and colors[nb] == c:
                    ok = False
                    break
            if not ok:
                continue
            colors[i] = c
            if dfs(i + 1):
                return True
            colors[i] = 0
        return False

    if dfs(0):
        out = "YES\n" + " ".join(map(str, colors))
    else:
        out = "NO"
    sys.stdout.write(out)

if __name__ == "__main__":
    solve()
```
