import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    assignment = [0] * n
    
    def get_possible_grades(u, current_assignment):
        used = set()
        for v in adj[u]:
            if current_assignment[v] != 0:
                used.add(current_assignment[v])
        return [g for g in [1, 2, 3] if g not in used]

    def backtrack(assigned_count):
        if assigned_count == n:
            return True
        
        # Heuristic: pick the pillar that is most constrained
        best_u = -1
        best_grades = []
        min_options = 4
        
        for i in range(n):
            if assignment[i] == 0:
                possible = get_possible_grades(i, assignment)
                if len(possible) < min_options:
                    min_options = len(possible)
                    best_u = i
                    best_grades = possible
                if min_options == 0:
                    return False
        
        # To satisfy lexicographical requirement, we must try to assign 
        # the smallest possible values to pillars 1, 2, ..., n in order.
        # However, the problem asks for the lexicographically smallest 
        # sequence (a1, a2, ..., an). This means we should try to fix 
        # the assignment of pillar 1, then 2, etc.
        
        # Re-evaluating: The constraint is to find the lexicographically 
        # smallest sequence. This is equivalent to a greedy approach with 
        # backtracking: try to assign 1 to pillar 1, if it leads to a 
        # solution, keep it. If not, try 2, then 3.
        
        for i in range(n):
            if assignment[i] == 0:
                for g in [1, 2, 3]:
                    if g not in [assignment[v] for v in adj[i] if assignment[v] != 0]:
                        assignment[i] = g
                        if backtrack(assigned_count + 1):
                            return True
                        assignment[i] = 0
                return False
        return True

    # The standard backtracking with fixed order 1..n is sufficient for n=40
    # if we use the constraint propagation (MRV) or just simple pruning.
    # Given the time limit and n=40, simple backtracking with pruning is fine.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
