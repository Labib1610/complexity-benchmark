import sys

def solve():
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n + 1)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u].append(v)
        adj[v].append(u)

    # Perfect square products:
    # 1*1 = 1 (sq)
    # 1*2 = 2
    # 1*3 = 3
    # 2*1 = 2
    # 2*2 = 4 (sq)
    # 2*3 = 6
    # 3*1 = 3
    # 3*2 = 6
    # 3*3 = 9 (sq)
    # Forbidden pairs: (1,1), (2,2), (3,3)
    
    assignment = [0] * (n + 1)

    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != 0:
                if val == assignment[v]:
                    return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
