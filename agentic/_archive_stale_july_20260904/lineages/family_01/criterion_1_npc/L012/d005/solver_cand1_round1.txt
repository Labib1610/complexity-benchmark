```python
import sys

# The problem asks for a lexicographically smallest 3-coloring of a graph.
# Since n is small (max 40), we can use backtracking with pruning.
# The lexicographical requirement means we try assigning 1, then 2, then 3
# to pillar 1, then pillar 2, and so on.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for color in range(1, 4):
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # Increase recursion depth for safety, though n=40 is small
    sys.setrecursionlimit(2000)
    solve()
```
