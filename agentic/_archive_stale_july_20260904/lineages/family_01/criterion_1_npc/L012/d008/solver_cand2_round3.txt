import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition det([[c_u, c_v], [1, 1]]) != 0 simplifies to c_u - c_v != 0,
    # which means c_u != c_v.
    # This is a graph coloring problem with 3 colors.
    # We need to minimize the final audit hash.
    # V_i = [[4, c_i], [0, 1]] * V_{i-1}
    # H_i = 4 * H_{i-1} + c_i
    # H_n = c_n + 4*c_{n-1} + 4^2*c_{n-2} + ... + 4^{n-1}*c_1
    # To minimize H_n, we want smaller c_i for larger i.
    # Specifically, we want to minimize c_n, then c_{n-1}, ..., then c_1.

    colors = [0] * (n + 1)

    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    # Since n is small (up to 40), we can use backtracking.
    # To minimize H_n, we prioritize smaller values for c_n, c_{n-1}, ..., c_1.
    # This means we should fill the array from n down to 1.
    
    best_assignment = None

    def backtrack(u):
        if u == 0:
            return True
        
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u - 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(n):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
