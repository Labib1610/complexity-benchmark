import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition det([[c_u, c_v], [1, 1]]) != 0 simplifies to c_u - c_v != 0,
    # so c_u != c_v. This is a graph coloring problem with 3 colors.
    
    colors = [0] * (n + 1)
    best_config = None
    min_hash = float('inf')

    # Since n <= 40, we can use backtracking with pruning.
    # The audit hash H_i = 4 * H_{i-1} + c_i.
    # H_n = c_n + 4*c_{n-1} + 16*c_{n-2} + ... + 4^{n-1}*c_1.
    # To minimize H_n, we want smaller c_i for larger i.
    # This is equivalent to lexicographical comparison of (c_n, c_{n-1}, ..., c_1).
    
    def get_hash(config):
        h = 0
        for i in range(n):
            h = 4 * h + config[i]
        return h

    def backtrack(u, current_config):
        nonlocal best_config, min_hash
        if u > n:
            h = get_hash(current_config)
            if h < min_hash:
                min_hash = h
                best_config = list(current_config)
            return

        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u and colors[v] == c:
                    possible = False
                    break
            if possible:
                colors[u] = c
                current_config.append(c)
                backtrack(u + 1, current_config)
                current_config.pop()
                colors[u] = 0

    # The graph might be disconnected. Process components.
    # Actually, the order of c_i is fixed 1 to n.
    backtrack(1, [])

    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(*(best_config))

solve()
