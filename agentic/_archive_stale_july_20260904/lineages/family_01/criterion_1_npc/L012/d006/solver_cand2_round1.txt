import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Perfect square products:
    # 1*1=1 (sq), 1*2=2, 1*3=3
    # 2*1=2, 2*2=4 (sq), 2*3=6
    # 3*1=3, 3*2=6, 3*3=9 (sq)
    # So, connected nodes cannot have the same value.
    # This is a graph coloring problem with 3 colors.
    
    # Weights calculation
    w = [0] * n
    v_seq = [0] * n
    w[0] = 1
    v_seq[0] = 2
    for k in range(1, n):
        w[k] = 3 * v_seq[k-1] - 2 * w[k-1]
        v_seq[k] = 7 * v_seq[k-1] - 6 * w[k-1]
    
    # Stress = sum_{i=1}^n c_i * W_{n-i}
    # To minimize stress, we want smaller c_i for larger W_{n-i}
    # W sequence grows exponentially, so W_{n-1} > W_{n-2} > ... > W_0
    # We want to minimize c_1, then c_2, ..., c_n.
    
    best_config = None
    min_stress = float('inf')
    
    colors = [0] * n
    
    def is_safe(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        nonlocal best_config, min_stress
        if u == n:
            current_stress = sum(colors[i] * w[n - 1 - i] for i in range(n))
            if current_stress < min_stress:
                min_stress = current_stress
                best_config = list(colors)
            return

        for c in [1, 2, 3]:
            if is_safe(u, c):
                colors[u] = c
                backtrack(u + 1)
                colors[u] = 0

    backtrack(0)
    
    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(*(best_config))

solve()
