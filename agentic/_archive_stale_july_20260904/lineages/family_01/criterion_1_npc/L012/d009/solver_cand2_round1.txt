import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    edges = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        edges.append((u, v))
        idx += 2

    # Flux F = integral from s_u to s_v of (3t^2 - 2t + 1) dt
    # F = [t^3 - t^2 + t] from s_u to s_v
    # Let f(t) = t^3 - t^2 + t
    # f(1) = 1 - 1 + 1 = 1
    # f(2) = 8 - 4 + 2 = 6
    # f(3) = 27 - 9 + 3 = 21
    # F = f(s_v) - f(s_u)
    # F != 0 iff f(s_v) != f(s_u), which means s_v != s_u
    # So the condition is simply s_u != s_v for all edges (u, v).
    # This is equivalent to finding a 3-coloring of the graph.
    # We need the lexicographically smallest 3-coloring.

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n

    def is_valid(node, c):
        for neighbor in adj[node]:
            if colors[neighbor] == c:
                return False
        return True

    def backtrack(node):
        if node == n:
            return True
        for c in [1, 2, 3]:
            if is_valid(node, c):
                colors[node] = c
                if backtrack(node + 1):
                    return True
                colors[node] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(colors))
    else:
        print("NO")

solve()
