import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The condition: for every edge (u, v), c_u != c_v.
    # This is equivalent to 3-coloring the graph.
    # We want to minimize sum c_i * 2^(2n - 2i).
    # This is equivalent to lexicographically minimizing the sequence (c_1, c_2, ..., c_n).
    
    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    # Since n <= 40, we can use backtracking with pruning.
    # To minimize the sequence lexicographically, we try colors 1, 2, 3 in order.
    
    def backtrack(u):
        if u > n:
            return True
        
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    # The problem states: "the set of risk tiers assigned to the two transactions 
    # must contain exactly two distinct values."
    # This means c_u != c_v.
    
    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
