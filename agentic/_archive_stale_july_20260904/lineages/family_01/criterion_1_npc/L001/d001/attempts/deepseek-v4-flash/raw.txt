We can model this as a graph coloring problem with 3 colors, where each material is a vertex and each reactive pair is an edge.  
We need to find the lexicographically smallest 3-coloring, or determine it’s impossible.

Since \( n \le 40 \), we can use backtracking with pruning.  
We assign chambers to materials in order from 1 to n, always trying the smallest chamber number first, and check constraints.

```python
import sys

def solve():
    input = sys.stdin.readline
    n, m = map(int, input().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, input().split())
        u -= 1
        v -= 1
        adj[u].append(v)
        adj[v].append(u)
    
    color = [0] * n  # 0 = unassigned, 1,2,3 = chambers
    
    def backtrack(idx):
        if idx == n:
            return True
        for c in (1, 2, 3):
            ok = True
            for nb in adj[idx]:
                if color[nb] == c:
                    ok = False
                    break
            if ok:
                color[idx] = c
                if backtrack(idx + 1):
                    return True
                color[idx] = 0
        return False
    
    if backtrack(0):
        print("YES")
        print(" ".join(map(str, color)))
    else:
        print("NO")

if __name__ == "__main__":
    solve()
```
