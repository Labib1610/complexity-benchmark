import sys
from itertools import product

def main():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list for conflicts
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(data[idx]) - 1
        v = int(data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
    
    # Check if graph is 3-colorable and find a valid coloring
    color = [-1] * n
    
    def is_bipartite(start, c):
        stack = [start]
        color[start] = c
        while stack:
            node = stack.pop()
            current_color = color[node]
            for nei in adj[node]:
                if color[nei] == -1:
                    color[nei] = 1 - current_color if current_color in [0,1] else 0
                    stack.append(nei)
                elif current_color in [0,1] and color[nei] == current_color:
                    return False
                elif current_color == 2 and color[nei] == 2:
                    # Need to handle triangle or odd cycle involving color 2
                    pass
        return True
    
    # Try all combinations for components
    def dfs(node, colors_used):
        if node == n:
            return True
        # Determine forbidden colors for this node
        forbidden = set()
        for nei in adj[node]:
            if color[nei] != -1:
                forbidden.add(color[nei])
        
        for c in colors_used:
            if c not in forbidden:
                color[node] = c
                if dfs(node + 1, colors_used):
                    return True
                color[node] = -1
        
        # Try new color if we haven't used all three
        if len(colors_used) < 3:
            for c in [0,1,2]:
                if c not in forbidden and c not in colors_used:
                    color[node] = c
                    new_colors = colors_used | {c}
                    if dfs(node + 1, new_colors):
                        return True
                    color[node] = -1
                    break
        return False
    
    # Check if 3-coloring exists
    if not dfs(0, set()):
        print("NO")
        return
    
    # Calculate the interference metric M for a given configuration
    def calculate_M(band_list):
        M = 0
        for S in range(1, 1 << n):
            min_id = None
            for i in range(n):
                if S & (1 << i):
                    min_id = i
                    break
            if min_id is not None:
                b_k = band_list[min_id]
                size_S = bin(S).count('1')
                M += b_k * (3 ** (size_S - 1))
        return M
    
    best_config = None
    min_M = float('inf')
    
    # Generate all valid 3-colorings
    def generate_all_colorings():
        res = []
        def backtrack(v):
            if v == n:
                res.append(color[:])
                return
            forbidden = set()
            for nei in adj[v]:
                if color[nei] != -1:
                    forbidden.add(color[nei])
            for c in range(3):
                if c not in forbidden:
                    color[v] = c
                    backtrack(v + 1)
                    color[v] = -1
        backtrack(0)
        return res
    
    all_valid = generate_all_colorings()
    
    for col in all_valid:
        band_list = [c + 1 for c in col]
        current_M = calculate_M(band_list)
        if current_M < min_M:
            min_M = current_M
            best_config = band_list[:]
    
    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(' '.join(map(str, best_config)))

if __name__ == "__main__":
    main()
