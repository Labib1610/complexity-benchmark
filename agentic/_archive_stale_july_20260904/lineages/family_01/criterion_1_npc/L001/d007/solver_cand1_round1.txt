```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The number of paths S = b1 * 4 * 4 * ... * 4 * 1 = b1 * 4^(n-1)
    # To minimize S, we need to minimize b1.
    # If there are multiple ways to achieve the minimum b1, any such assignment is fine
    # because the problem asks for "the" plan that minimizes S, and S depends only on b1.
    
    best_assignment = None
    
    # Try b1 = 1, then 2, then 3
    for b1 in [1, 2, 3]:
        assignment = [0] * n
        assignment[0] = b1
        
        # Backtracking to find a valid 3-coloring
        def is_valid(u, color):
            for v in adj[u]:
                if assignment[v] == color:
                    return False
            return True

        def backtrack(u):
            if u == n:
                return True
            if u == 0:
                return backtrack(1)
            
            for color in [1, 2, 3]:
                if is_valid(u, color):
                    assignment[u] = color
                    if backtrack(u + 1):
                        return True
                    assignment[u] = 0
            return False

        if backtrack(1):
            best_assignment = assignment
            break
            
    if best_assignment:
        print("YES")
        print(*(best_assignment))
    else:
        print("NO")

solve()
```
