import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # The interference metric M = sum_{S subseteq {1..n}, S != empty} b_{min(S)} * 3^{|S|-1}
    # Let k = min(S). For a fixed k, the number of subsets S such that min(S) = k
    # is the number of subsets of {k+1, ..., n}, which is 2^{n-k}.
    # M = sum_{k=1}^n b_k * sum_{j=0}^{n-k} (3^j * C(n-k, j))
    # By binomial theorem, sum_{j=0}^{N} C(N, j) * 3^j = (1+3)^N = 4^N.
    # So M = sum_{k=1}^n b_k * 4^{n-k}.
    # To minimize M, we want to assign smaller b_k to larger 4^{n-k} weights,
    # but the weights are 4^{n-1}, 4^{n-2}, ..., 4^0.
    # Since 4^{n-k} > sum_{i=k+1}^n 3 * 4^{n-i} = 3 * (4^{n-k}-1)/3 = 4^{n-k}-1,
    # the greedy strategy is to minimize b_1, then b_2, ..., then b_n.

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
