import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # T[r][c] calculation
    # T[r][c] = sum of 3^(k-1) for all strictly increasing sequences from r to c
    # Let f(r, c) be the sum.
    # f(r, r) = 1
    # f(r, c) = sum_{r < j <= c} f(j, c) * 3^(1-1) ? No.
    # The definition: sequence (i_1, ..., i_k) with i_1=r, i_k=c, i_1 < i_2 < ... < i_k.
    # Weight is 3^(k-1).
    # Let dp[i] be the sum of weights of sequences starting at i and ending at some fixed c.
    # For a fixed c, let g(i) be the sum of weights of sequences starting at i and ending at c.
    # g(c) = 1
    # g(i) = sum_{i < j <= c} 3 * g(j)
    # This is a geometric series.
    # g(i) = 3 * (g(i+1) + g(i+2) + ... + g(c))
    # g(c) = 1
    # g(c-1) = 3 * g(c) = 3
    # g(c-2) = 3 * (g(c-1) + g(c)) = 3 * (3 + 1) = 12
    # g(c-k) = 3 * 4^(k-1)
    
    T = [[0] * (n + 1) for _ in range(n + 1)]
    for c in range(1, n + 1):
        T[c][c] = 1
        for r in range(c - 1, 0, -1):
            T[r][c] = 3 * (4**(c - r - 1))
            
    # M = sum_{r=1}^n b_r * (sum_{c=r}^n T[r][c])
    # Let W[r] = sum_{c=r}^n T[r][c]
    W = [0] * (n + 1)
    for r in range(1, n + 1):
        for c in range(r, n + 1):
            W[r] += T[r][c]
            
    # We need to assign b_i in {1, 2, 3} such that b_u != b_v for all edges
    # and minimize sum b_i * W[i].
    # Since n <= 40, we can use backtracking with pruning or coloring.
    # Actually, 3-coloring is NP-hard, but n=40 is small enough for backtracking.
    
    best_m = float('inf')
    best_config = None
    
    colors = [0] * (n + 1)
    
    def backtrack(u, current_m):
        nonlocal best_m, best_config
        if u > n:
            if current_m < best_m:
                best_m = current_m
                best_config = colors[1:]
            return
        
        # Try colors 1, 2, 3
        for c in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u and colors[v] == c:
                    possible = False
                    break
            if possible:
                colors[u] = c
                backtrack(u + 1, current_m + c * W[u])
                colors[u] = 0
                
    backtrack(1, 0)
    
    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(*(best_config))

solve()
