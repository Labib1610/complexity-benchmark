import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Calculate W_i
    # P(u, v) is number of paths from u to v.
    # Each pair (u, v) with u < v has 3 direct edges.
    # Let dp[i] be the number of paths from i to n.
    # P(i, v) = 3 * (1 + sum_{k=v+1}^n P(v, k))
    # W_i = 1 + sum_{v=i+1}^n P(i, v)
    # Let S_i = sum_{v=i+1}^n P(i, v) = W_i - 1
    # P(i, v) = 3 * W_v
    # W_i = 1 + sum_{v=i+1}^n (3 * W_v)
    # W_n = 1
    # W_{n-1} = 1 + 3 * W_n = 4
    # W_{n-2} = 1 + 3 * W_n + 3 * W_{n-1} = 1 + 3 + 12 = 16
    # W_i = 1 + 3 * sum_{j=i+1}^n W_j
    # W_i = 1 + 3 * (W_{i+1} + sum_{j=i+2}^n W_j)
    # Since W_{i+1} = 1 + 3 * sum_{j=i+2}^n W_j, then 3 * sum_{j=i+2}^n W_j = W_{i+1} - 1
    # W_i = 1 + 3 * W_{i+1} + (W_{i+1} - 1) = 4 * W_{i+1}
    # W_n = 1, W_{n-1} = 4, W_{n-2} = 16, ..., W_i = 4^{n-i}
    
    weights = [4**(n - i) for i in range(1, n + 1)]
    
    # Backtracking to find min M = sum b_i * W_i
    # Since W_i are powers of 4, larger weights dominate.
    # To minimize sum b_i * W_i, we want smaller b_i for larger W_i.
    # W_1 > W_2 > ... > W_n.
    # So we want b_1 as small as possible, then b_2, etc.
    
    assignment = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if v < u and assignment[v] == color:
                return False
        return True

    best_config = None
    min_m = float('inf')

    def backtrack(u, current_m):
        nonlocal min_m, best_config
        if u > n:
            if current_m < min_m:
                min_m = current_m
                best_config = assignment[1:]
            return
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                backtrack(u + 1, current_m + color * weights[u-1])
                assignment[u] = 0

    backtrack(1, 0)
    
    if best_config is None:
        print("NO")
    else:
        print("YES")
        print(*(best_config))

solve()
