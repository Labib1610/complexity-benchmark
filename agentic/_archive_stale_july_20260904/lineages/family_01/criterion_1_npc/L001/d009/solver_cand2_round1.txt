import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx+1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # H_i calculation:
    # A forward-cascading sequence is z_0, z_1, ..., z_k where z_0 = i and z_j < z_{j+1}.
    # This is equivalent to paths in a DAG where edges exist from u to v if u < v.
    # H_i = 1 + sum_{j > i} (3 * H_j)
    # H_n = 1
    # H_{n-1} = 1 + 3*H_n = 4
    # H_{n-2} = 1 + 3*H_{n-1} + 3*H_n = 1 + 12 + 3 = 16
    # H_i = 4^{n-i}
    # Total Metric M = sum(b_i * 4^{n-i})
    # To minimize M, we want smaller b_i for larger 4^{n-i} (i.e., smaller i).
    
    h = [0] * (n + 1)
    h[n] = 1
    for i in range(n - 1, 0, -1):
        h[i] = 4**(n - i)

    colors = [0] * (n + 1)
    
    def is_valid(u, c):
        for v in adj[u]:
            if colors[v] == c:
                return False
        return True

    def backtrack(u):
        if u > n:
            return True
        
        # Try colors 1, 2, 3 in order to minimize b_i * H_i
        for c in [1, 2, 3]:
            if is_valid(u, c):
                colors[u] = c
                if backtrack(u + 1):
                    return True
                colors[u] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(colors[1:]))
    else:
        print("NO")

solve()
