```python
import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Greedy assignment: for each vertex, assign smallest valid band
    bands = [0] * n
    for u in range(n):
        used = set()
        for v in adj[u]:
            if v < u:
                used.add(bands[v])
        for b in [1, 2, 3]:
            if b not in used:
                bands[u] = b
                break
        if bands[u] == 0:
            print("NO")
            return
    
    # Verify all edges
    for u in range(n):
        for v in adj[u]:
            if v > u and bands[u] == bands[v]:
                print("NO")
                return
    
    print("YES")
    print(" ".join(map(str, bands)))

if __name__ == "__main__":
    solve()
```
