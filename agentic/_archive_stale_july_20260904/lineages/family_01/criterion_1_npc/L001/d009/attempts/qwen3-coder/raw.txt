def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n = int(data[0])
    m = int(data[1])
    
    # Build adjacency list
    adj = [[] for _ in range(n + 1)]
    index = 2
    for _ in range(m):
        u = int(data[index])
        v = int(data[index + 1])
        adj[u].append(v)
        adj[v].append(u)
        index += 2
    
    # Colors: 1, 2, 3
    # We want lexicographically smallest assignment
    color = [0] * (n + 1)  # color[i] is the color of node i
    
    # Try to assign colors greedily in order, minimizing lex order
    for node in range(1, n + 1):
        # Determine which colors are forbidden for this node
        forbidden = [False] * 4  # index 0 unused; indices 1,2,3 for colors
        for neighbor in adj[node]:
            if color[neighbor] != 0:
                forbidden[color[neighbor]] = True
        
        # Assign the smallest available color
        found_color = False
        for c in range(1, 4):
            if not forbidden[c]:
                color[node] = c
                found_color = True
                break
        
        if not found_color:
            print("NO")
            return
    
    print("YES")
    print(' '.join(map(str, color[1:])))

solve()
