import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Metric M calculation:
    # Packet injected at stage i passes through n-i transitions.
    # Each transition multiplies by 4.
    # Total packets at stage n = sum_{i=1}^n (b_i * 4^(n-i))
    # We want to minimize sum_{i=1}^n (b_i * 4^(n-i))
    # Since 4^(n-1) > 4^(n-2) > ... > 4^0, we prioritize smaller b_i for smaller i.
    # This is equivalent to minimizing the value as a base-4 number (b_1, b_2, ..., b_n).
    # Lexicographical order is also based on b_1, b_2, ..., b_n.
    # Thus, minimizing M is identical to finding the lexicographically smallest valid configuration.

    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
