import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # Since we want to minimize sum f_i * 4^(n-i), we want to pick the smallest
    # possible f_1, then smallest f_2, etc. This is a greedy approach with backtracking.
    # Given n <= 40, we need to be careful. However, the constraints on 3-coloring
    # are tight.
    
    assignment = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        # Try frequencies 1, 2, 3 in order to minimize the signature
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    # The problem is 3-coloring, which is NP-complete.
    # For n=40, simple backtracking might be slow, but the constraints
    # on interference often prune the search space significantly.
    # If it's too slow, we might need a more optimized approach,
    # but let's try standard backtracking first.
    
    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
