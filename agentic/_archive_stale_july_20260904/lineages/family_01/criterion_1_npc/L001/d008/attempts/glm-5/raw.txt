import sys

def solve():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)
    
    # Check if 3-coloring exists and find one
    color = [0] * n
    
    def dfs(u, c):
        color[u] = c
        for v in adj[u]:
            if color[v] == 0:
                if not dfs(v, 4 - c):
                    return False
            elif color[v] == c:
                return False
        return True
    
    for i in range(n):
        if color[i] == 0:
            if not dfs(i, 1):
                print("NO")
                return
    
    # Valid configuration exists, find optimal one
    # Group by color
    groups = [[], [], []]
    for i in range(n):
        groups[color[i] - 1].append(i)
    
    # Sort groups by size (descending) for efficiency
    group_order = sorted(range(3), key=lambda x: len(groups[x]), reverse=True)
    
    best_config = None
    best_metric = float('inf')
    
    # Try all permutations of band assignments to color groups
    from itertools import permutations
    for perm in permutations([1, 2, 3]):
        config = [0] * n
        for gi, band in zip(group_order, perm):
            for node in groups[gi]:
                config[node] = band
        
        # Compute metric
        A = list(config)
        for k in range(n - 1):
            ak = A[k]
            if ak != 0:
                add = 3 * ak
                for j in range(k + 1, n):
                    A[j] += add
        metric = sum(A)
        
        if metric < best_metric:
            best_metric = metric
            best_config = config
    
    print("YES")
    print(' '.join(map(str, best_config)))

if __name__ == "__main__":
    solve()
