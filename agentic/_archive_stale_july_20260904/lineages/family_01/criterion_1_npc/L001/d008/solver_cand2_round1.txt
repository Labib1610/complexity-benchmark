import sys

def solve():
    n, m = map(int, sys.stdin.readline().split())
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # The interference metric M:
    # A_j(final) = b_j + 3 * sum_{k < j} A_k(final)
    # Let S_j = sum_{i=1}^j A_i(final).
    # A_j(final) = b_j + 3 * S_{j-1}
    # S_j = S_{j-1} + A_j(final) = S_{j-1} + b_j + 3 * S_{j-1} = 4 * S_{j-1} + b_j
    # S_0 = 0
    # S_1 = b_1
    # S_2 = 4*b_1 + b_2
    # S_3 = 4*(4*b_1 + b_2) + b_3 = 16*b_1 + 4*b_2 + b_3
    # S_n = sum_{i=1}^n (4^{n-i} * b_i)
    # To minimize M, we want to minimize the weighted sum with weights 4^{n-i}.
    # Since 4^{n-1} > 4^{n-2} > ... > 1, we prioritize smaller b_i for smaller i.

    best_config = None
    min_m = float('inf')

    # Since n <= 40, we use backtracking with pruning.
    # To optimize, we use the fact that we want to minimize the lexicographical
    # value of the sequence (b_1, b_2, ..., b_n) because of the weights.
    
    colors = [0] * n

    def backtrack(idx):
        nonlocal min_m, best_config
        if idx == n:
            # Calculate M
            m_val = 0
            power_of_4 = 1
            for i in range(n - 1, -1, -1):
                m_val += colors[i] * power_of_4
                power_of_4 *= 4
            if m_val < min_m:
                min_m = m_val
                best_config = list(colors)
            return True

        # Try colors 1, 2, 3
        used = {colors[neighbor] for neighbor in adj[idx] if colors[neighbor] != 0}
        for c in [1, 2, 3]:
            if c not in used:
                colors[idx] = c
                if backtrack(idx + 1):
                    # Since we want the lexicographically smallest sequence,
                    # and the weights are 4^{n-i}, the first valid configuration
                    # found by trying 1, 2, 3 in order is the minimum.
                    if best_config is not None:
                        return True
                colors[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(best_config))
    else:
        print("NO")

solve()
