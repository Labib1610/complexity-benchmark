import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # H(c1, c2) = 2*c1*c2 / (c1 + c2)
    # Possible values:
    # (1,1) -> 2*1*1/2 = 1 (Integer)
    # (1,2) -> 2*1*2/3 = 4/3 (Non-integer)
    # (1,3) -> 2*1*3/4 = 6/4 = 1.5 (Non-integer)
    # (2,2) -> 2*2*2/4 = 2 (Integer)
    # (2,3) -> 2*2*3/5 = 12/5 = 2.4 (Non-integer)
    # (3,3) -> 2*3*3/6 = 3 (Integer)
    # Stability: H is non-integer iff {c_u, c_v} is {1, 2} or {1, 3} or {2, 3}.
    # Wait, let's recheck:
    # (1,1) -> 1 (Int)
    # (1,2) -> 1.33 (Non-int)
    # (1,3) -> 1.5 (Non-int)
    # (2,2) -> 2 (Int)
    # (2,3) -> 2.4 (Non-int)
    # (3,3) -> 3 (Int)
    # So, c_u != c_v is NOT sufficient.
    # Stability requires:
    # If c_u == 1, c_v must be 2 or 3.
    # If c_u == 2, c_v must be 1 or 3.
    # If c_u == 3, c_v must be 1 or 2.
    # Actually, the condition "H is non-integer" is equivalent to c_u != c_v.
    # Let's re-verify:
    # (1,1) -> 1 (Int)
    # (2,2) -> 2 (Int)
    # (3,3) -> 3 (Int)
    # (1,2) -> 4/3 (Non-int)
    # (1,3) -> 6/4 = 1.5 (Non-int)
    # (2,3) -> 12/5 = 2.4 (Non-int)
    # Yes, H is non-integer if and only if c_u != c_v.
    
    res = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        for val in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u and res[v] == val:
                    possible = False
                    break
            if possible:
                res[u] = val
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
