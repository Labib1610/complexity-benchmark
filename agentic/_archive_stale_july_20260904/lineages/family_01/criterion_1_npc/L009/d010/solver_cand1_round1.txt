import sys
import math

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Precompute valid pairs (c_u, c_v)
    # gcd(11*cu + 2*cv, 2*cu + 11*cv) <= 5
    valid_pairs = []
    for cu in [1, 2, 3]:
        for cv in [1, 2, 3]:
            r1 = 11 * cu + 2 * cv
            r2 = 2 * cu + 11 * cv
            if math.gcd(r1, r2) <= 5:
                valid_pairs.append((cu, cv))
                
    assignment = [0] * n
    
    def is_valid(u, val):
        for v in adj[u]:
            if assignment[v] != 0:
                cv = assignment[v]
                r1 = 11 * val + 2 * cv
                r2 = 2 * val + 11 * cv
                if math.gcd(r1, r2) > 5:
                    return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(u, val):
                assignment[u] = val
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
