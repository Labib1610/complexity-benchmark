```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # E(x) = det([[x, c_u], [-c_v, x - c_u - c_v]])
    # E(x) = x(x - c_u - c_v) - (c_u * -c_v)
    # E(x) = x^2 - (c_u + c_v)x + c_u * c_v
    # E(x) = (x - c_u)(x - c_v)
    # The minimum of (x - c_u)(x - c_v) is at x = (c_u + c_v) / 2
    # Min value = ((c_u + c_v)/2 - c_u) * ((c_u + c_v)/2 - c_v)
    # Min value = ((c_v - c_u)/2) * ((c_u - c_v)/2)
    # Min value = -(c_u - c_v)^2 / 4
    # Condition: Min value < 0
    # -(c_u - c_v)^2 / 4 < 0  => (c_u - c_v)^2 > 0
    # This is true if and only if c_u != c_v.
    # So the constraint is simply c_u != c_v for all edges (u, v).
    # This is a graph coloring problem with 3 colors.
    
    res = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
```
