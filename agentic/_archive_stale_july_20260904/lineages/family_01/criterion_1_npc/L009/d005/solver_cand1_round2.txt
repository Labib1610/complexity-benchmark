import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2

    # The determinant of the matrix:
    # | 1   1   1  |
    # | cu  cv  4  |
    # | cu^2 cv^2 16 |
    # is (cv - cu)(4 - cu)(4 - cv).
    # For the matrix to be invertible, the determinant must be non-zero.
    # This means:
    # 1. cu != cv
    # 2. cu != 4 (always true since cu in {1, 2, 3})
    # 3. cv != 4 (always true since cv in {1, 2, 3})
    # So the condition simplifies to cu != cv.
    
    # This is a graph coloring problem: assign each node a color from {1, 2, 3}
    # such that no two adjacent nodes have the same color.
    # We want the lexicographically smallest sequence.
    
    adj = [[] for _ in range(n)]
    for u, v in constraints:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        
        for color in [1, 2, 3]:
            possible = True
            for neighbor in adj[idx]:
                if assignment[neighbor] == color:
                    possible = False
                    break
            
            if possible:
                assignment[idx] = color
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
