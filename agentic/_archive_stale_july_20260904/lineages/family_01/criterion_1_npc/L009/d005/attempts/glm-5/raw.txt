import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[False] * n for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        adj[u][v] = True
        adj[v][u] = True
        
    # Precompute valid pairs (c_u, c_v) for a constraint
    valid_pairs = [[False] * 4 for _ in range(4)]
    for cu in range(1, 4):
        for cv in range(1, 4):
            # Determinant of the 3x3 matrix
            det = (1 * cv * 16 + 1 * 4 * cu**2 + 1 * cu * cv**2) - \
                  (1 * 4 * cv**2 + 1 * cu * 16 + 1 * cv * cu**2)
            if det != 0:
                valid_pairs[cu][cv] = True
                
    # Precompute valid values for a node given its already assigned neighbors
    valid_vals = [[True] * 4 for _ in range(1 << 6)]
    for mask in range(1 << 6):
        for c in range(1, 4):
            ok = True
            for i in range(6):
                if (mask >> i) & 1:
                    if not valid_pairs[c][i + 1]:
                        ok = False
                        break
            valid_vals[mask][c] = ok

    # Backtracking to find the lexicographically smallest valid sequence
    ans = [0] * n
    
    def dfs(u):
        if u == n:
            return True
            
        # Compute the constraint mask for the current node based on assigned neighbors
        mask = 0
        for v in range(u):
            if adj[u][v]:
                mask |= (1 << (ans[v] - 1))
                
        for c in range(1, 4):
            if valid_vals[mask][c]:
                ans[u] = c
                if dfs(u + 1):
                    return True
                    
        return False

    if dfs(0):
        print("YES")
        print(" ".join(map(str, ans)))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
