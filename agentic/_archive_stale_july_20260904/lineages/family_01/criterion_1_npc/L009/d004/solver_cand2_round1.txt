```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
    
    # We want to minimize sum c_i * 4^(n-i).
    # This is equivalent to lexicographical minimization of (c_1, c_2, ..., c_n).
    # Since n <= 40, we can use backtracking with pruning.
    
    res = [0] * n
    
    def backtrack(u):
        if u == n:
            return True
        
        # Try 1, 2, 3 in order to minimize lexicographically
        for val in [1, 2, 3]:
            possible = True
            for v in adj[u]:
                if v < u and res[v] == val:
                    possible = False
                    break
            
            if possible:
                res[u] = val
                if backtrack(u + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
```
