import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Constraint: gcd(c_u, c_v) < lcm(c_u, c_v)
    # Possible values: {1, 2, 3}
    # Pairs (x, y):
    # (1, 1): gcd=1, lcm=1. 1 < 1 is False.
    # (1, 2): gcd=1, lcm=2. 1 < 2 is True.
    # (1, 3): gcd=1, lcm=3. 1 < 3 is True.
    # (2, 2): gcd=2, lcm=2. 2 < 2 is False.
    # (2, 3): gcd=1, lcm=6. 1 < 6 is True.
    # (3, 3): gcd=3, lcm=3. 3 < 3 is False.
    # So, if (u, v) is a constraint, c_u and c_v cannot be (1, 1), (2, 2), or (3, 3).
    # This means c_u != c_v.
    # The problem reduces to: Can we color the graph with 3 colors such that 
    # no two adjacent vertices have the same color?
    # This is the 3-coloring problem, which is NP-hard, but n <= 40.
    # We need the lexicographically smallest sequence.
    
    res = [0] * (n + 1)
    
    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(res[1:]))
    else:
        print("NO")

solve()
