import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2

    # Points are (c, c^2) and (4, 16).
    # Collinear if (c_u^2 - 16) / (c_u - 4) == (c_v^2 - 16) / (c_v - 4)
    # Since c in {1, 2, 3}, c-4 is never 0.
    # (c^2 - 16) / (c - 4) = c + 4.
    # So points are collinear if c_u + 4 == c_v + 4, i.e., c_u == c_v.
    # The constraint says they must NOT be collinear, so c_u != c_v.
    
    # We need to assign c_i in {1, 2, 3} such that for all (u, v) in constraints, c_u != c_v.
    # This is a graph coloring problem with 3 colors.
    # We want the lexicographically smallest sequence.
    
    adj = [[] for _ in range(n)]
    for u, v in constraints:
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[idx]:
                if assignment[neighbor] == val:
                    possible = False
                    break
            
            if possible:
                assignment[idx] = val
                if backtrack(idx + 1):
                    return True
                assignment[idx] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
