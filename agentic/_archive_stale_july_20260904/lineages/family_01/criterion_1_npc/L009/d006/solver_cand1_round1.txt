import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # Forbidden sums: c_u^2 + c_v^2 must not divide 72
    # Possible values for c_i^2 are 1, 4, 9
    # Possible sums:
    # 1+1=2 (72%2 == 0) -> Forbidden
    # 1+4=5 (72%5 != 0) -> OK
    # 1+9=10 (72%10 != 0) -> OK
    # 4+4=8 (72%8 == 0) -> Forbidden
    # 4+9=13 (72%13 != 0) -> OK
    # 9+9=18 (72%18 == 0) -> Forbidden
    
    # Forbidden pairs (c_u, c_v):
    # (1, 1), (2, 2), (3, 3)
    
    res = [0] * n
    
    def is_valid(idx, val):
        for neighbor in adj[idx]:
            if neighbor < idx:
                neighbor_val = res[neighbor]
                if val == neighbor_val:
                    return False
        return True

    def backtrack(idx):
        if idx == n:
            return True
        
        for val in [1, 2, 3]:
            if is_valid(idx, val):
                res[idx] = val
                if backtrack(idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
