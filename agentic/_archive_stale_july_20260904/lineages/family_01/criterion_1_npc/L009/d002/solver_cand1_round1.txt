import sys

def solve():
    try:
        line1 = sys.stdin.readline().split()
        if not line1:
            return
        n, m = map(int, line1)
    except ValueError:
        return

    adj = [[] for _ in range(n)]
    for _ in range(m):
        u, v = map(int, sys.stdin.readline().split())
        adj[u-1].append(v-1)
        adj[v-1].append(u-1)

    # Perfect square products for {1, 2, 3}:
    # 1*1 = 1 (square)
    # 2*2 = 4 (square)
    # 3*3 = 9 (square)
    # 1*2 = 2 (not square)
    # 1*3 = 3 (not square)
    # 2*3 = 6 (not square)
    # So, c_u * c_v must not be a square means:
    # If c_u == c_v, it's a square.
    # Thus, for every edge (u, v), c_u != c_v.
    # This is equivalent to 3-coloring the graph.

    assignment = [0] * n

    def is_valid(u, color):
        for v in adj[u]:
            if assignment[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        
        for color in [1, 2, 3]:
            if is_valid(u, color):
                assignment[u] = color
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(assignment))
    else:
        print("NO")

solve()
