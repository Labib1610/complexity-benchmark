import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2
        
    # The points are P = (c_u, c_u^2, c_u^3) and Q = (c_v, c_v^2, c_v^3).
    # They are collinear with (0,0,0) if the vectors OP and OQ are linearly dependent.
    # Since c_i in {1, 2, 3}, OP = c_u * (1, c_u, c_u^2).
    # OQ = c_v * (1, c_v, c_v^2).
    # They are collinear if (1, c_u, c_u^2) is a multiple of (1, c_v, c_v^2).
    # This happens if and only if c_u = c_v.
    # Thus, the constraint "non-degenerate triangle" means c_u != c_v.
    
    adj = [[] for _ in range(n)]
    for u, v in constraints:
        adj[u].append(v)
        adj[v].append(u)
        
    res = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        
        for val in [1, 2, 3]:
            possible = True
            for neighbor in adj[idx]:
                if neighbor < idx and res[neighbor] == val:
                    possible = False
                    break
            
            if possible:
                res[idx] = val
                if backtrack(idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
