import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    constraints = []
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        constraints.append((u, v))
        idx += 2

    # The system is:
    # z_k * A_k + sum_{j=1}^{k-1} z_j = 1
    # where A_k = integral from c_{u_k} to c_{v_k} of (21 - 10t) dt
    # A_k = [21t - 5t^2] from c_{u_k} to c_{v_k}
    # A_k = (21*c_{v_k} - 5*c_{v_k}^2) - (21*c_{u_k} - 5*c_{u_k}^2)
    
    def get_A(c_u, c_v):
        def f(t):
            return 21 * t - 5 * t * t
        return f(c_v) - f(c_u)

    # The system is triangular:
    # z_1 * A_1 = 1
    # z_2 * A_2 + z_1 = 1
    # z_3 * A_3 + z_1 + z_2 = 1
    # ...
    # A system of linear equations has a unique solution if and only if
    # the determinant of the coefficient matrix is non-zero.
    # The matrix is lower triangular with diagonal elements A_1, A_2, ..., A_m.
    # The determinant is the product A_1 * A_2 * ... * A_m.
    # Thus, the system has a unique solution iff A_k != 0 for all k=1..m.
    
    # A_k = 0 iff f(c_{v_k}) == f(c_{u_k})
    # f(1) = 21 - 5 = 16
    # f(2) = 42 - 20 = 22
    # f(3) = 63 - 45 = 18
    # Since f(1), f(2), f(3) are all distinct, A_k = 0 iff c_{u_k} == c_{v_k}.
    # But the problem states c_i must be in {1, 2, 3}.
    # Wait, the condition is A_k != 0.
    # A_k = f(c_{v_k}) - f(c_{u_k}).
    # A_k = 0 iff f(c_{v_k}) == f(c_{u_k}).
    # Since f(1)=16, f(2)=22, f(3)=18, f(x) is injective.
    # So A_k = 0 iff c_{v_k} == c_{u_k}.
    # The problem says u_k and v_k are distinct indices.
    # Does the sequence c_1, ..., c_n have to have distinct values? No.
    # If c_{u_k} == c_{v_k}, then A_k = 0.
    # So the condition "unique solution" is equivalent to:
    # For all k, c_{u_k} != c_{v_k}.
    
    # This is exactly the graph coloring problem with 3 colors.
    # We need to find the lexicographically smallest 3-coloring of the graph
    # where edges are the constraints.
    
    adj = [[] for _ in range(n)]
    for u, v in constraints:
        adj[u].append(v)
        adj[v].append(u)
        
    res = [0] * n
    
    def backtrack(idx):
        if idx == n:
            return True
        for color in [1, 2, 3]:
            possible = True
            for neighbor in adj[idx]:
                if neighbor < idx and res[neighbor] == color:
                    possible = False
                    break
            if possible:
                res[idx] = color
                if backtrack(idx + 1):
                    return True
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
