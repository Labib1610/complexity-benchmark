import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        adj[u].append(v)
        adj[v].append(u)
        idx += 2

    # The energy function E(x) = x^2 - (cu + cv)x + cu*cv
    # This is a parabola opening upwards. The minimum is at x = (cu + cv) / 2.
    # E(min) = ((cu + cv)/2)^2 - (cu + cv)^2 / 2 + cu*cv
    # E(min) = -(cu + cv)^2 / 4 + cu*cv = -(cu^2 + 2*cu*cv + cv^2 - 4*cu*cv) / 4
    # E(min) = -(cu - cv)^2 / 4
    # For E(min) < 0, we need -(cu - cv)^2 < 0, which means (cu - cv)^2 > 0.
    # This implies cu != cv.
    # So the constraint is simply cu != cv for all pairs (u, v).
    # This is equivalent to finding a 3-coloring of the graph.
    
    res = [0] * n
    
    def is_valid(u, color):
        for v in adj[u]:
            if res[v] == color:
                return False
        return True

    def backtrack(u):
        if u == n:
            return True
        for color in [1, 2, 3]:
            if is_valid(u, color):
                res[u] = color
                if backtrack(u + 1):
                    return True
                res[u] = 0
        return False

    if backtrack(0):
        print("YES")
        print(*(res))
    else:
        print("NO")

solve()
