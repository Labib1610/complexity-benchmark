def solve():
    import sys
    input = sys.stdin.read
    data = input().split()
    
    n, m = int(data[0]), int(data[1])
    edges = []
    index = 2
    for _ in range(m):
        u = int(data[index]) - 1
        v = int(data[index+1]) - 1
        edges.append((u, v))
        index += 2
    
    # For each node, maintain a list of its neighbors
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    
    # Try all 3^n combinations, but prune using constraint propagation
    # We want the lexicographically smallest solution
    # So we can use backtracking with forward checking
    
    color = [0] * n  # 0 means unassigned
    
    def is_valid(node, c):
        for neighbor in adj[node]:
            if color[neighbor] == c:
                return False
        return True
    
    def backtrack(node):
        if node == n:
            return True
            
        # Try colors in order 1, 2, 3 for lexicographic minimality
        for c in [1, 2, 3]:
            if is_valid(node, c):
                color[node] = c
                if backtrack(node + 1):
                    return True
                color[node] = 0  # Backtrack
                
        return False
    
    # Since this is a graph coloring problem with 3 colors,
    # and we have constraints, let's check if it's possible
    
    if backtrack(0):
        print("YES")
        print(' '.join(map(str, color)))
    else:
        print("NO")

solve()
