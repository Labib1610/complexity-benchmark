```python
import sys

sys.setrecursionlimit(10000)

def solve() -> None:
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    n = int(next(it))
    m = int(next(it))
    adj = [[] for _ in range(n)]
    for _ in range(m):
        u = int(next(it)) - 1
        v = int(next(it)) - 1
        adj[u].append(v)
        adj[v].append(u)

    colors = [0] * n          # 0 = uncolored, else 1..3
    possible = [0b111] * n    # bitmask of allowed colors for each vertex

    # Preprocess: if any vertex has degree >= 3 and graph is a triangle? not needed.

    def dfs(idx: int) -> bool:
        if idx == n:
            return True
        # try colors 1..3 in order
        for c in (1, 2, 3):
            if not (possible[idx] & (1 << (c - 1))):
                continue
            # check already colored neighbors
            conflict = False
            for nb in adj[idx]:
                if nb < idx and colors[nb] == c:
                    conflict = True
                    break
            if conflict:
                continue

            # assign
            colors[idx] = c
            changed = []  # list of (vertex, old_mask)
            ok = True
            for nb in adj[idx]:
                if nb > idx and colors[nb] == 0:
                    old = possible[nb]
                    new = old & ~(1 << (c - 1))
                    if new != old:
                        possible[nb] = new
                        changed.append((nb, old))
                    if new == 0:
                        ok = False
                        break
            if ok and dfs(idx + 1):
                return True
            # revert
            for v, old in reversed(changed):
                possible[v] = old
            colors[idx] = 0
        return False

    if dfs(0):
        out = "YES\n" + " ".join(map(str, colors))
    else:
        out = "NO"
    sys.stdout.write(out)

if __name__ == "__main__":
    solve()
```
