```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        adj[u].append(v)
        adj[v].append(u)
        
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            color[i] = 1
            queue = [i]
            head = 0
            while head < len(queue):
                u = queue[head]
                head += 1
                c_u = color[u]
                c_v = 3 - c_u
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = c_v
                        queue.append(v)
                    elif color[v] != c_v:
                        print("NO")
                        return

    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
