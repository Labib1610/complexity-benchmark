import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    adj = [[] for _ in range(n)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        adj[u].append(v)
        adj[v].append(u)

    # domain bitmask: bit 0 -> color 1, bit 1 -> color 2, bit 2 -> color 3
    # initially 7 (binary 111) for all vertices
    initial_domains = [7] * n

    def unit_propagate(domains, start_nodes):
        queue = list(start_nodes)
        while queue:
            u = queue.pop()
            mask_u = domains[u]
            for v in adj[u]:
                if domains[v] & mask_u:
                    domains[v] &= ~mask_u
                    if domains[v] == 0:
                        return False
                    if domains[v] in (1, 2, 4):
                        queue.append(v)
        return True

    def backtrack(domains):
        # Find the smallest index vertex with domain size > 1
        target = -1
        for i in range(n):
            d = domains[i]
            if d not in (1, 2, 4):  # popcount > 1
                target = i
                break

        if target == -1:
            # All vertices have a single color assigned
            res = []
            for i in range(n):
                d = domains[i]
                if d == 1:
                    res.append(1)
                elif d == 2:
                    res.append(2)
                elif d == 4:
                    res.append(3)
                else:
                    return None
            return res

        d = domains[target]
        for color_val, bit in [(1, 1), (2, 2), (3, 4)]:
            if d & bit:
                new_domains = list(domains)
                new_domains[target] = bit
                if unit_propagate(new_domains, [target]):
                    res = backtrack(new_domains)
                    if res is not None:
                        return res

        return None

    ans = backtrack(initial_domains)

    if ans is None:
        print("NO")
    else:
        print("YES")
        print(*(ans))


if __name__ == "__main__":
    solve()
