import sys

sys.setrecursionlimit(10000)


def solve_component(adj, fixed, hint=None):
    n = len(adj)
    domains = [7] * n

    for i, color in enumerate(fixed):
        if color:
            domains[i] = 1 << (color - 1)

    trail = []

    def propagate(queue):
        head = 0
        while head < len(queue):
            v = queue[head]
            head += 1
            dv = domains[v]
            if dv == 0:
                return False
            if dv & (dv - 1):
                continue

            for u in adj[v]:
                du = domains[u]
                if du & dv:
                    nd = du & ~dv
                    if nd != du:
                        trail.append((u, du))
                        domains[u] = nd
                        if nd == 0:
                            return False
                        if nd & (nd - 1) == 0:
                            queue.append(u)
        return True

    initial_queue = [i for i, d in enumerate(domains) if d & (d - 1) == 0]
    if not propagate(initial_queue):
        return None

    def solve_two_sat():
        variables = [v for v in range(n) if domains[v].bit_count() == 2]
        k = len(variables)

        if k == 0:
            return [d.bit_length() for d in domains]

        index = [-1] * n
        choices = [None] * k
        for i, v in enumerate(variables):
            index[v] = i
            bits = []
            mask = domains[v]
            while mask:
                b = mask & -mask
                bits.append(b)
                mask -= b
            choices[i] = bits

        size = 2 * k
        graph = [[] for _ in range(size)]
        reverse = [[] for _ in range(size)]

        def add_edge(a, b):
            graph[a].append(b)
            reverse[b].append(a)

        for u in variables:
            iu = index[u]
            for v in adj[u]:
                if u >= v or index[v] < 0:
                    continue
                iv = index[v]
                common = domains[u] & domains[v]
                while common:
                    color_bit = common & -common
                    common -= color_bit
                    au = 0 if choices[iu][0] == color_bit else 1
                    av = 0 if choices[iv][0] == color_bit else 1

                    add_edge(2 * iu + au, 2 * iv + (1 - av))
                    add_edge(2 * iv + av, 2 * iu + (1 - au))

        visited = [False] * size
        order = []

        def dfs1(v):
            visited[v] = True
            for u in graph[v]:
                if not visited[u]:
                    dfs1(u)
            order.append(v)

        for v in range(size):
            if not visited[v]:
                dfs1(v)

        component = [-1] * size

        def dfs2(v, cid):
            component[v] = cid
            for u in reverse[v]:
                if component[u] == -1:
                    dfs2(u, cid)

        cid = 0
        for v in reversed(order):
            if component[v] == -1:
                dfs2(v, cid)
                cid += 1

        result = [0] * n
        for v in range(n):
            if domains[v] & (domains[v] - 1) == 0:
                result[v] = domains[v].bit_length()

        for i, v in enumerate(variables):
            if component[2 * i] == component[2 * i + 1]:
                return None
            value = 1 if component[2 * i + 1] > component[2 * i] else 0
            result[v] = choices[i][value].bit_length()

        return result

    def search():
        selected = -1
        best_size = 4
        best_degree = -1
        all_at_most_two = True

        for v in range(n):
            count = domains[v].bit_count()
            if count <= 1:
                continue
            if count == 3:
                all_at_most_two = False
            unresolved_degree = 0
            for u in adj[v]:
                if domains[u].bit_count() > 1:
                    unresolved_degree += 1
            if count < best_size or (count == best_size and unresolved_degree > best_degree):
                best_size = count
                best_degree = unresolved_degree
                selected = v

        if selected == -1:
            return [d.bit_length() for d in domains]

        if all_at_most_two:
            return solve_two_sat()

        options = []
        mask = domains[selected]
        while mask:
            bit = mask & -mask
            options.append(bit)
            mask -= bit

        if hint is not None:
            preferred = 1 << (hint[selected] - 1)
            if preferred in options:
                options.remove(preferred)
                options.insert(0, preferred)

        signatures = set()
        filtered = []
        for bit in options:
            signature = 0
            for i, d in enumerate(domains):
                if d & bit:
                    signature |= 1 << i
            if signature not in signatures:
                signatures.add(signature)
                filtered.append(bit)

        for bit in filtered:
            marker = len(trail)
            old = domains[selected]
            trail.append((selected, old))
            domains[selected] = bit

            if propagate([selected]):
                result = search()
                if result is not None:
                    return result

            while len(trail) > marker:
                v, old_domain = trail.pop()
                domains[v] = old_domain

        return None

    return search()


def main():
    data = list(map(int, sys.stdin.buffer.read().split()))
    if not data:
        return

    n, m = data[0], data[1]
    masks = [0] * n
    p = 2

    for _ in range(m):
        u = data[p] - 1
        v = data[p + 1] - 1
        p += 2
        masks[u] |= 1 << v
        masks[v] |= 1 << u

    global_adj = [[] for _ in range(n)]
    for v in range(n):
        mask = masks[v]
        while mask:
            bit = mask & -mask
            global_adj[v].append(bit.bit_length() - 1)
            mask -= bit

    visited = [False] * n
    components = []

    for start in range(n):
        if visited[start]:
            continue
        stack = [start]
        visited[start] = True
        component = []

        while stack:
            v = stack.pop()
            component.append(v)
            for u in global_adj[v]:
                if not visited[u]:
                    visited[u] = True
                    stack.append(u)

        component.sort()
        components.append(component)

    answer = [0] * n

    for vertices in components:
        k = len(vertices)
        if k == 1:
            answer[vertices[0]] = 1
            continue

        local_index = {v: i for i, v in enumerate(vertices)}
        adj = [[] for _ in range(k)]

        for i, v in enumerate(vertices):
            for u in global_adj[v]:
                if u in local_index:
                    adj[i].append(local_index[u])

        fixed = [0] * k
        fixed[0] = 1

        solution = solve_component(adj, fixed)
        if solution is None:
            print("NO")
            return

        for pos in range(1, k):
            chosen_solution = None
            for color in range(1, solution[pos]):
                trial_fixed = fixed[:]
                trial_fixed[pos] = color
                candidate = solve_component(adj, trial_fixed, solution)
                if candidate is not None:
                    chosen_solution = candidate
                    break

            if chosen_solution is not None:
                solution = chosen_solution

            fixed[pos] = solution[pos]

        for i, v in enumerate(vertices):
            answer[v] = solution[i]

    print("YES")
    print(*answer)


if __name__ == "__main__":
    main()
