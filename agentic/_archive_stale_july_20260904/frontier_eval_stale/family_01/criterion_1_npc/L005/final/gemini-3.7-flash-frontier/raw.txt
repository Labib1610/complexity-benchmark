import sys

# Increase recursion depth just in case
sys.setrecursionlimit(2000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx+1]) - 1
        idx += 2
        if u != v:
            adj[u] |= (1 << v)
            adj[v] |= (1 << u)
            
    # Check self-loops (impossible to color)
    # Problem says u and v are identifiers of dials linked by crossbar. If u == v, impossible.
    # But graph is simple typically.

    # Fast 3-coloring feasibility checker using MRV (Minimum Remaining Values)
    def is_colorable(domains, assigned_mask):
        if assigned_mask == (1 << n) - 1:
            return True
        
        # Pick unassigned vertex with smallest domain size
        best_u = -1
        best_size = 4
        
        # Find MRV vertex
        for u in range(n):
            if not (assigned_mask & (1 << u)):
                sz = bin(domains[u]).count('1')
                if sz == 0:
                    return False
                if sz < best_size:
                    best_size = sz
                    best_u = u
                    if sz == 1:
                        break
        
        u = best_u
        u_domain = domains[u]
        
        # Try each color in domain
        for c_idx in range(3):
            c_bit = 1 << c_idx
            if u_domain & c_bit:
                # Propagate to neighbors
                new_domains = list(domains)
                new_domains[u] = c_bit
                possible = True
                
                # Update neighbors
                nbrs = adj[u]
                # iterate over neighbors
                while nbrs:
                    v = (nbrs & -nbrs).bit_length() - 1
                    nbrs &= nbrs - 1
                    new_domains[v] &= ~c_bit
                    if not (assigned_mask & (1 << v)) and new_domains[v] == 0:
                        possible = False
                        break
                
                if possible:
                    if is_colorable(new_domains, assigned_mask | (1 << u)):
                        return True
                        
        return False

    initial_domains = [7] * n  # 7 = 111 in binary (colors 1, 2, 4)
    if not is_colorable(initial_domains, 0):
        print("NO")
        return

    # Now greedily find the lexicographically smallest 3-coloring
    print("YES")
    assignment = [0] * n
    current_domains = [7] * n
    assigned_mask = 0
    
    for i in range(n):
        for c in (1, 2, 3):
            c_bit = 1 << (c - 1)
            if current_domains[i] & c_bit:
                # Try setting dial i to color c
                cand_domains = list(current_domains)
                cand_domains[i] = c_bit
                possible = True
                
                nbrs = adj[i]
                while nbrs:
                    v = (nbrs & -nbrs).bit_length() - 1
                    nbrs &= nbrs - 1
                    cand_domains[v] &= ~c_bit
                    if not (assigned_mask & (1 << v)) and cand_domains[v] == 0:
                        possible = False
                        break
                
                if possible and is_colorable(cand_domains, assigned_mask | (1 << i)):
                    assignment[i] = c
                    current_domains = cand_domains
                    assigned_mask |= (1 << i)
                    break

    print(*(assignment))

if __name__ == '__main__':
    solve()
