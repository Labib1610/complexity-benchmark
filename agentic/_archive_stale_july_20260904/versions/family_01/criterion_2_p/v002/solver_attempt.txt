import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we want to assign 1
    # to the smallest possible node index whenever possible.
    # We process nodes 1 to n. If a node is not visited, start a BFS/DFS.
    # To ensure lexicographical smallest, we prefer assigning 1 to the 
    # current node if it's not constrained by its neighbors.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS component
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, color = stack.pop()
                
                # Neighbors must have the opposite color
                target_color = 3 - color
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = target_color
                        stack.append((v, target_color))
                    elif assignment[v] == color:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
