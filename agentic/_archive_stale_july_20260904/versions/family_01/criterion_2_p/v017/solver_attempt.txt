import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If an individual is not yet assigned, we assign them to shift 1 (the smaller label).
    # Then we perform a BFS/DFS to propagate the constraints.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS from i, assigning i to shift 1
            stack = [(i, 1)]
            assignment[i] = 1
            
            while stack:
                u, shift = stack.pop()
                
                next_shift = 2 if shift == 1 else 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = next_shift
                        stack.append((v, next_shift))
                    elif assignment[v] == shift:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

if __name__ == '__main__':
    solve()
