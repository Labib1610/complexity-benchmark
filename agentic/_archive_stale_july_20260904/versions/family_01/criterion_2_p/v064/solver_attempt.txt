import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # assignment[i] = 0 (unassigned), 1, or 2
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is unassigned, we try assigning it to 1 first.
    # This is a standard 2-coloring problem (bipartite check).
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [(i, 1)]
            assignment[i] = 1
            
            # We need to process the component. To ensure lexicographical
            # smallest, we must be careful. Actually, for a fixed component,
            # once the first node is assigned, the rest are forced.
            # The lexicographical requirement means we prefer 1 over 2 for the
            # smallest index in the component.
            
            component = []
            stack = [i]
            assignment[i] = 1
            
            # Standard bipartite check
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        q.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(assignment[1:]))

solve()
