```python
import sys

# The constraint x_u + x_v = 3 with x_i in {1, 2} implies x_u != x_v.
# This is a 2-coloring problem on a graph where edges represent x_u != x_v.
# We want the lexicographically smallest vector (x_1, x_2, ..., x_n).
# To make it lexicographically smallest, we prefer x_i = 1 whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # color: 0 = unvisited, 1 = value 1, 2 = value 2
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To get lexicographically smallest, we try setting x_i = 1
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS
            while stack:
                u, c = stack.pop()
                component.append((u, c))
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

solve()
```
