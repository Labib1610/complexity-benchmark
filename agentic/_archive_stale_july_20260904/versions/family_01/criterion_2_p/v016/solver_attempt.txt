import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # To get the lexicographically smallest assignment, we try to assign 1 to 
    # each node if possible. This is a standard 2-coloring problem.
    # Since we want the lexicographically smallest sequence (s1, s2, ..., sn),
    # we iterate from 1 to n and if a node is not colored, we start a BFS/DFS.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To ensure lexicographical smallest, we greedily pick 1 if possible
            stack = [i]
            color[i] = 1
            
            while stack:
                u = stack.pop()
                
                # Sort neighbors to process them in a deterministic order if needed,
                # but for lexicographical smallest, we just need to ensure that
                # when we visit a new component, we assign 1 to the smallest node.
                # Within a component, the coloring is forced once the first node is set.
                
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
