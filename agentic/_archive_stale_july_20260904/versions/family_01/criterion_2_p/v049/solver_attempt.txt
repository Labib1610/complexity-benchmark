import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # states: 0 = unassigned, 1 = state 1, 2 = state 2
    states = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a component is unassigned, we try assigning state 1 first.
    # This is a standard 2-coloring problem.
    
    for i in range(1, n + 1):
        if states[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            states[i] = 1
            
            while stack:
                u = stack.pop()
                
                for v in adj[u]:
                    if states[v] == 0:
                        states[v] = 3 - states[u]
                        stack.append(v)
                    elif states[v] == states[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(states[1:]))

if __name__ == '__main__':
    solve()
