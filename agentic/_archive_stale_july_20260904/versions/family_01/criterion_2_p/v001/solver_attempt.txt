```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # modes: 0 = unassigned, 1 = mode 1, 2 = mode 2
    modes = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we iterate 1 to n.
    # If a node is unassigned, we assign it mode 1 and perform BFS/DFS.
    for i in range(1, n + 1):
        if modes[i] == 0:
            # Start BFS/DFS to color the connected component
            stack = [i]
            modes[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if modes[v] == 0:
                        modes[v] = 3 - modes[u]
                        stack.append(v)
                    elif modes[v] == modes[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(modes[1:]))

if __name__ == '__main__':
    solve()
```
