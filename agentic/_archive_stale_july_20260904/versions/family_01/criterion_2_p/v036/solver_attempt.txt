import sys

# Increase recursion depth for deep DFS
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # We want the lexicographically smallest sequence.
    # This means for each connected component, we try to assign 1 to the 
    # smallest node index first.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS from node i, assigning 1 to it
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process nodes in a way that respects the lexicographical requirement
            # Actually, standard 2-coloring is fixed once the first node is colored.
            # To get the lexicographically smallest, we must ensure that for each
            # component, the first node encountered (smallest index) gets 1.
            
            q = [i]
            component.append(i)
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
    print("YES")
    print(*(color[1:]))

solve()
