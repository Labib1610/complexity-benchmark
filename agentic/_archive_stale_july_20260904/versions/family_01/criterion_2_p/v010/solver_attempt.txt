import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we want 
    # node 1 to be 1, then node 2 to be 1 if possible, etc.
    # This is equivalent to 2-coloring the graph.
    # Since we want the lexicographically smallest sequence (s1, s2, ..., sn),
    # we iterate i from 1 to n. If i is not colored, we color it 1.
    # Then we propagate the constraints.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS to color the connected component
            # To ensure lexicographical smallest, we prefer color 1 for the root
            stack = [(i, 1)]
            color[i] = 1
            
            while stack:
                u, c = stack.pop()
                
                # Sort neighbors to process them in a deterministic order
                # though for 2-coloring, the order doesn't change feasibility,
                # just the specific coloring.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - c
                        stack.append((v, 3 - c))
                    elif color[v] == c:
                        print("NO")
                        return
                        
    print("YES")
    print(*(color[1:]))

solve()
