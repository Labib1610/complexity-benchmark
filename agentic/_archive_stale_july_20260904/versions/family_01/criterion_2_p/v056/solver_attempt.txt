import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    # polarity: 0 = unassigned, 1 = state 1, 2 = state 2
    polarity = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is unvisited, we start a BFS/DFS.
    # To make it lexicographically smallest, we prefer assigning 1 to the current node.
    
    for i in range(1, n + 1):
        if polarity[i] == 0:
            # Start BFS/DFS component
            stack = [i]
            polarity[i] = 1
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if polarity[v] == 0:
                        polarity[v] = 3 - polarity[u]
                        stack.append(v)
                    elif polarity[v] == polarity[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(polarity[1:]))

if __name__ == '__main__':
    solve()
