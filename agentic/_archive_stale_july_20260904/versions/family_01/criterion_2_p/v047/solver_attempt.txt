```python
import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we want to assign 1
    # to as many nodes as possible, prioritizing smaller indices.
    # This is a standard 2-coloring problem.
    
    color = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for each connected component
            # To ensure lexicographical smallest, we try to assign 1 to the 
            # current node if possible.
            stack = [i]
            color[i] = 1
            while stack:
                u = stack.pop()
                
                # Sort neighbors to process them in a deterministic order
                # though for 2-coloring, the assignment is fixed once the 
                # first node of a component is colored.
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
    
    print("YES")
    print(*(color[1:]))

if __name__ == '__main__':
    solve()
```
