```python
import sys

# The problem asks for a 2-coloring of a graph such that the lexicographically 
# smallest assignment (1, 2) is chosen. This is equivalent to a greedy 
# approach: for each node i from 1 to n, try assigning 1. If assigning 1 
# is valid (i.e., no neighbor already assigned 1 has been assigned 1), 
# assign 1. Otherwise, try assigning 2. If assigning 2 is also invalid, 
# then no valid coloring exists.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # We need to check if assigning 1 is possible.
    # A node can be assigned 1 if none of its neighbors are already assigned 1.
    # If it can't be 1, it must be 2. If it can't be 2, it's impossible.
    
    for i in range(1, n + 1):
        can_be_1 = True
        for neighbor in adj[i]:
            if assignment[neighbor] == 1:
                can_be_1 = False
                break
        
        if can_be_1:
            assignment[i] = 1
        else:
            # Must be 2. Check if 2 is valid.
            can_be_2 = True
            for neighbor in adj[i]:
                if assignment[neighbor] == 2:
                    can_be_2 = False
                    break
            if can_be_2:
                assignment[i] = 2
            else:
                print("NO")
                return

    print("YES")
    print(*(assignment[1:]))

solve()
```
