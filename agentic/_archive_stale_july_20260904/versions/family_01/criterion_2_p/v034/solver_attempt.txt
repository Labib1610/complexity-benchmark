import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate from 1 to n.
    # If a node is unassigned, we try assigning it to track 1.
    # If that leads to a conflict, we try track 2.
    # Since we want the smallest sequence, we greedily pick 1 whenever possible.
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this connected component
            stack = [i]
            assignment[i] = 1
            component = []
            
            # Use a stack for iterative DFS to avoid recursion limits
            # and to process the component
            comp_stack = [i]
            while comp_stack:
                u = comp_stack.pop()
                component.append(u)
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        comp_stack.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if the current assignment is the lexicographically smallest
            # for this component. Actually, the greedy choice 1 for the first
            # node of each component is sufficient because the choice of 1 or 2
            # for the first node determines the rest of the component.
            # Since we want the smallest sequence, we always prefer 1 for the
            # first node of each component.
            
    print("YES")
    print(*(assignment[1:]))

solve()
