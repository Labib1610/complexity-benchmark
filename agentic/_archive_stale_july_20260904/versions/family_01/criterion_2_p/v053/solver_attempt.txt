```python
import sys

# The problem asks for a 2-coloring of a graph such that no two adjacent vertices
# have the same color. This is equivalent to checking if the graph is bipartite.
# To find the lexicographically smallest configuration, we iterate from 1 to n.
# If a vertex is not yet colored, we assign it color 1 (the smallest possible).
# Then we perform a BFS/DFS to color its connected component.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        idx += 2
        if u == v:
            print("NO")
            return
        adj[u].append(v)
        adj[v].append(u)
        
    colors = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if colors[i] == 0:
            # Start BFS for this component
            # Assign color 1 to the first node encountered in the component
            # to ensure lexicographical smallest result.
            colors[i] = 1
            stack = [i]
            while stack:
                u = stack.pop()
                for v in adj[u]:
                    if colors[v] == 0:
                        colors[v] = 3 - colors[u]
                        stack.append(v)
                    elif colors[v] == colors[u]:
                        print("NO")
                        return
                        
    print("YES")
    print(*(colors[1:]))

if __name__ == '__main__':
    # Increase recursion depth for deep graphs if using DFS, 
    # but here we use iterative stack-based approach.
    sys.setrecursionlimit(300000)
    solve()
```
