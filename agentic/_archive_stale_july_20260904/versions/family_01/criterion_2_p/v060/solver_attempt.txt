import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # color[i] will be 0 (unvisited), 1 (Band 1), or 2 (Band 2)
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest configuration, we iterate 1 to n.
    # For each connected component, we try assigning 1 to the smallest node.
    # If that fails, we try 2.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            # Start BFS/DFS for this component
            # We want the smallest node in the component to be 1 if possible.
            # Actually, for lexicographical smallest, we just need to ensure
            # that for each component, we pick the assignment that makes the
            # sequence smallest.
            
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard bipartite check
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if swapping colors (1->2, 2->1) makes it lexicographically smaller
            # We only care about the first node in the component (the smallest index)
            # because that determines the lexicographical order.
            # If color[i] is 1, we are good. If color[i] is 2, we swap.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
