import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # To get the lexicographically smallest assignment, we want to assign 1
    # to the smallest possible index whenever possible.
    # This is a 2-coloring problem. For each connected component, we have 
    # two choices (swap 1s and 2s). We choose the one that makes the 
    # smallest index in the component 1.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS to color the component
            component = []
            stack = [(i, 1)]
            assignment[i] = 1
            component.append(i)
            
            idx_in_comp = 0
            while idx_in_comp < len(component):
                u = component[idx_in_comp]
                idx_in_comp += 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if the current assignment is the lexicographically smallest
            # for this component. The smallest index in this component is 'i'.
            # We already assigned assignment[i] = 1.
            # If we had assigned assignment[i] = 2, we would have to flip all
            # values in this component. Since 1 < 2, keeping assignment[i] = 1
            # is always better for the lexicographical order.
            
    print("YES")
    print(*(assignment[1:]))

solve()
