import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest assignment, we iterate 1 to n.
    # If a node is not colored, we try assigning it 1.
    # Since it's a bipartite check, we use DFS.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            stack = [(i, 1)]
            color[i] = 1
            component = []
            
            # Iterative DFS to avoid recursion depth issues
            idx_stack = [i]
            while idx_stack:
                u = idx_stack.pop()
                component.append(u)
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        idx_stack.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # To ensure lexicographical smallest, we need to check if 
            # swapping colors in this connected component yields a smaller result.
            # However, for a fixed component, the assignment is determined by the 
            # first node's color. If we start with 1, we get one valid coloring.
            # If we start with 2, we get the other.
            # We compare the two possibilities for the smallest index in the component.
            
            # Actually, the greedy approach works: for each component, 
            # we have two valid colorings (c and 3-c). We pick the one that 
            # makes the smallest index in the component 1.
            
            min_node = min(component)
            if color[min_node] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
