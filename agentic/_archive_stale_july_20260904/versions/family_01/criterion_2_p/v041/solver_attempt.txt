import sys

# The problem states that for every link (u, v), load(u) + load(v) = 3.
# Since load(u) is either 1 or 2, this means load(u) != load(v).
# This is equivalent to a 2-coloring problem on a graph.
# If the graph is bipartite, we can color it with 1s and 2s.
# To get the lexicographically smallest sequence, we want to assign 1 to 
# the smallest possible index whenever possible.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    color = [0] * (n + 1)
    
    # To get the lexicographically smallest sequence, we iterate 1 to n.
    # If a node is uncolored, we start a BFS/DFS.
    # To make the sequence lexicographically smallest, we prefer 1 over 2.
    # For each connected component, we have two choices:
    # 1. Start with color 1 for the smallest index node in the component.
    # 2. Start with color 2 for the smallest index node in the component.
    # We compare the resulting sequences.
    
    for i in range(1, n + 1):
        if color[i] == 0:
            component = []
            stack = [i]
            color[i] = 1
            component.append(i)
            
            # Standard BFS/DFS to find component and check bipartiteness
            q = [i]
            head = 0
            while head < len(q):
                u = q[head]
                head += 1
                for v in adj[u]:
                    if color[v] == 0:
                        color[v] = 3 - color[u]
                        q.append(v)
                        component.append(v)
                    elif color[v] == color[u]:
                        print("NO")
                        return
            
            # Check if swapping colors 1 and 2 gives a lexicographically smaller result
            # We only need to check the first node of the component (the smallest index)
            # If color[i] is 1, we keep it. If it's 2, we check if swapping is better.
            # Actually, for the lexicographically smallest sequence, we want the 
            # smallest index to be 1 if possible.
            if color[i] == 2:
                for node in component:
                    color[node] = 3 - color[node]
                    
    print("YES")
    print(*(color[1:]))

solve()
