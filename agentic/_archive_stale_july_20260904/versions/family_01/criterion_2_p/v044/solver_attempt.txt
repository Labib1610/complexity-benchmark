import sys

# The problem asks for the lexicographically smallest 2-coloring of a graph.
# A graph is 2-colorable if and only if it is bipartite.
# To find the lexicographically smallest assignment (s_1, s_2, ..., s_n):
# We iterate from i = 1 to n. If technician i is not yet assigned, we try
# assigning them to shift 1. If this is valid, we proceed. If not, we try shift 2.
# Since we want the lexicographically smallest, we greedily pick 1 if possible.
# This is equivalent to checking if the connected component containing i
# can be 2-colored such that i is 1.

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Try assigning i to 1
            component = []
            stack = [(i, 1)]
            assignment[i] = 1
            component.append((i, 1))
            
            possible = True
            idx_comp = 0
            while idx_comp < len(component):
                u, color = component[idx_comp]
                idx_comp += 1
                
                for v in adj[u]:
                    if assignment[v] == 0:
                        new_color = 3 - color
                        assignment[v] = new_color
                        component.append((v, new_color))
                    elif assignment[v] == color:
                        possible = False
                        break
                if not possible:
                    break
            
            if not possible:
                # Try assigning i to 2
                for node, _ in component:
                    assignment[node] = 0
                
                stack = [(i, 2)]
                assignment[i] = 2
                component = [(i, 2)]
                possible = True
                idx_comp = 0
                while idx_comp < len(component):
                    u, color = component[idx_comp]
                    idx_comp += 1
                    for v in adj[u]:
                        if assignment[v] == 0:
                            new_color = 3 - color
                            assignment[v] = new_color
                            component.append((v, new_color))
                        elif assignment[v] == color:
                            possible = False
                            break
                    if not possible:
                        break
                
                if not possible:
                    print("NO")
                    return

    print("YES")
    print(*(assignment[1:]))

solve()
