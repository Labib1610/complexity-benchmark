import sys

# Increase recursion depth for deep DFS trees
sys.setrecursionlimit(300000)

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    # We want the lexicographically smallest assignment.
    # This means we prefer shift 1 for technician 1, then 2, etc.
    # This is a 2-coloring problem.
    # For each connected component, we have two possible colorings (swap 1 and 2).
    # To make it lexicographically smallest, for each component, we pick the 
    # coloring that assigns shift 1 to the smallest-indexed technician in that component.
    
    assignment = [0] * (n + 1)
    
    for i in range(1, n + 1):
        if assignment[i] == 0:
            # Start BFS/DFS for this component
            component = []
            stack = [i]
            assignment[i] = 1
            component.append(i)
            
            # Use a stack for iterative DFS to avoid recursion limits
            idx_comp = 0
            while idx_comp < len(component):
                u = component[idx_comp]
                idx_comp += 1
                for v in adj[u]:
                    if assignment[v] == 0:
                        assignment[v] = 3 - assignment[u]
                        component.append(v)
                    elif assignment[v] == assignment[u]:
                        print("NO")
                        return
            
            # Check if this component's assignment is optimal.
            # The smallest index in this component is 'i'.
            # We already assigned assignment[i] = 1.
            # If we had assigned assignment[i] = 2, we would have to flip all 
            # assignments in this component. Since we want the lexicographically 
            # smallest, and 1 < 2, keeping assignment[i] = 1 is always better.
            
    print("YES")
    print(*(assignment[1:]))

solve()
