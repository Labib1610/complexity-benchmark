```python
import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_valid(person, zone):
        for neighbor in adj[person]:
            if assignment[neighbor] == zone:
                return False
        return True

    def backtrack(person):
        if person > n:
            # Check sum condition: sum(z_i) <= 3n
            # Since z_i is in {1, 2, 3}, sum is always <= 3n.
            # The condition is always satisfied.
            return True
        
        for zone in range(1, 4):
            if is_valid(person, zone):
                assignment[person] = zone
                if backtrack(person + 1):
                    return True
                assignment[person] = 0
        return False

    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    solve()
```
