import sys

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    n = int(input_data[0])
    m = int(input_data[1])
    
    adj = [[] for _ in range(n + 1)]
    idx = 2
    for _ in range(m):
        u = int(input_data[idx])
        v = int(input_data[idx + 1])
        adj[u].append(v)
        adj[v].append(u)
        idx += 2
        
    assignment = [0] * (n + 1)
    
    def is_safe(u, vault):
        for neighbor in adj[u]:
            if assignment[neighbor] == vault:
                return False
        return True
    
    def backtrack(u):
        if u > n:
            return True
        
        for vault in range(1, 4):
            if is_safe(u, vault):
                assignment[u] = vault
                if backtrack(u + 1):
                    return True
                assignment[u] = 0
        return False
    
    if backtrack(1):
        print("YES")
        print(*(assignment[1:]))
    else:
        print("NO")

if __name__ == '__main__':
    # The problem constraints (n <= 40) and the nature of 3-coloring
    # suggest that backtracking with lexicographical order is sufficient.
    # Python's recursion limit might need adjustment for larger n,
    # but 40 is well within default limits.
    sys.setrecursionlimit(2000)
    solve()
