import sys
from functools import lru_cache

sys.setrecursionlimit(1_000_000)

data = list(map(int, sys.stdin.buffer.read().split()))
n, m = data[0], data[1]

adj_mask = [0] * n
p = 2
for _ in range(m):
    u = data[p] - 1
    v = data[p + 1] - 1
    p += 2
    if u != v:
        adj_mask[u] |= 1 << v
        adj_mask[v] |= 1 << u

adj = [[] for _ in range(n)]
edges = []
for u in range(n):
    mask = adj_mask[u]
    while mask:
        b = mask & -mask
        v = b.bit_length() - 1
        adj[u].append(v)
        if u < v:
            edges.append((u, v))
        mask ^= b

component = [-1] * n
components = []
for start in range(n):
    if component[start] != -1:
        continue
    cid = len(components)
    stack = [start]
    component[start] = cid
    vertices = []
    while stack:
        u = stack.pop()
        vertices.append(u)
        for v in adj[u]:
            if component[v] == -1:
                component[v] = cid
                stack.append(v)
    components.append(vertices)

def close_domains(domains, queue):
    q = list(queue)
    while q:
        u = q.pop()
        color = domains[u]
        if color == 0:
            return None
        for v in adj[u]:
            if domains[v] & color:
                new_domain = domains[v] & ~color
                if new_domain == domains[v]:
                    continue
                if new_domain == 0:
                    return None
                domains[v] = new_domain
                if new_domain & (new_domain - 1) == 0:
                    q.append(v)
    return tuple(domains)

def two_sat(state):
    variable_id = [-1] * n
    options = []
    for v, domain in enumerate(state):
        if domain.bit_count() == 2:
            variable_id[v] = len(options)
            first = domain & -domain
            second = domain ^ first
            options.append((first, second))

    k = len(options)
    if k == 0:
        return True

    size = 2 * k
    graph = [[] for _ in range(size)]
    reverse = [[] for _ in range(size)]

    def add_implication(a, b):
        graph[a].append(b)
        reverse[b].append(a)

    for u, v in edges:
        iu = variable_id[u]
        iv = variable_id[v]
        if iu < 0 or iv < 0:
            continue

        common = state[u] & state[v]
        while common:
            color = common & -common
            common ^= color

            cu = 0 if options[iu][0] == color else 1
            cv = 0 if options[iv][0] == color else 1

            lu = 2 * iu + cu
            lv = 2 * iv + cv
            add_implication(lu, 2 * iv + (1 - cv))
            add_implication(lv, 2 * iu + (1 - cu))

    visited = [False] * size
    order = []

    for start in range(size):
        if visited[start]:
            continue
        visited[start] = True
        stack = [(start, 0)]
        while stack:
            u, idx = stack[-1]
            if idx < len(graph[u]):
                v = graph[u][idx]
                stack[-1] = (u, idx + 1)
                if not visited[v]:
                    visited[v] = True
                    stack.append((v, 0))
            else:
                order.append(u)
                stack.pop()

    comp_id = [-1] * size
    cid = 0
    for start in reversed(order):
        if comp_id[start] != -1:
            continue
        comp_id[start] = cid
        stack = [start]
        while stack:
            u = stack.pop()
            for v in reverse[u]:
                if comp_id[v] == -1:
                    comp_id[v] = cid
                    stack.append(v)
        cid += 1

    for i in range(k):
        if comp_id[2 * i] == comp_id[2 * i + 1]:
            return False
    return True

@lru_cache(maxsize=150000)
def solve(state):
    active_component = [False] * len(components)
    for v, domain in enumerate(state):
        if domain != 7:
            active_component[component[v]] = True

    chosen = -1
    chosen_score = None

    for v, domain in enumerate(state):
        if domain != 7:
            continue

        residual_degree = 0
        for u in adj[v]:
            if state[u].bit_count() > 1:
                residual_degree += 1

        if residual_degree == 0:
            continue

        score = (
            1 if active_component[component[v]] else 0,
            residual_degree,
            len(adj[v])
        )
        if chosen_score is None or score > chosen_score:
            chosen_score = score
            chosen = v

    if chosen == -1:
        return two_sat(state)

    if not active_component[component[chosen]]:
        colors = (1,)
    else:
        impact = []
        for color in (1, 2, 4):
            count = 0
            for u in adj[chosen]:
                if state[u].bit_count() == 2 and state[u] & color:
                    count += 1
            impact.append((count, color))
        impact.sort()
        colors = tuple(color for _, color in impact)

    for color in colors:
        child = list(state)
        child[chosen] = color
        closed = close_domains(child, [chosen])
        if closed is not None and solve(closed):
            return True

    return False

fixed = [-1] * n

def feasible():
    domains = [7] * n
    queue = []
    for i, color in enumerate(fixed):
        if color != -1:
            domains[i] = 1 << color
            queue.append(i)
    closed = close_domains(domains, queue)
    return closed is not None and solve(closed)

fixed[0] = 0
if not feasible():
    print("NO")
    sys.exit()

for i in range(1, n):
    fixed[i] = 0
    if feasible():
        continue

    fixed[i] = 1
    if feasible():
        continue

    fixed[i] = 2

print("YES")
print(*[color + 1 for color in fixed])
