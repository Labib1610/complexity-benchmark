import sys


def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return

    n = int(input_data[0])
    m = int(input_data[1])

    adj = [[] for _ in range(n)]
    adj_mask = [0] * n
    idx = 2
    for _ in range(m):
        u = int(input_data[idx]) - 1
        v = int(input_data[idx + 1]) - 1
        idx += 2
        if u != v:
            adj[u].append(v)
            adj[v].append(u)
            adj_mask[u] |= 1 << v
            adj_mask[v] |= 1 << u

    # 2-SAT solver for when all domain sizes <= 2
    def solve_2sat(domains):
        # domains is a list of integers representing bitmasks of available colors (subset of {1, 2, 4})
        # Colors: 1 (bit 0), 2 (bit 1), 3 (bit 2)
        # Variable mapping: for each node i with 2 colors (c0 < c1):
        # literal 2*i: node i has color c0
        # literal 2*i + 1: node i has color c1 (negation of 2*i)
        # If domain has 1 color, it's fixed.

        # 2N literals
        num_vars = n
        adj_2sat = [[] for _ in range(2 * num_vars)]

        def add_clause(u_lit, v_lit):
            # u_lit or v_lit  <=>  (~u_lit -> v_lit) and (~v_lit -> u_lit)
            adj_2sat[u_lit ^ 1].append(v_lit)
            adj_2sat[v_lit ^ 1].append(u_lit)

        for i in range(n):
            mask = domains[i]
            if mask == 0:
                return False
            bits = [c for c in range(3) if (mask & (1 << c))]
            if len(bits) == 1:
                # Fixed to bits[0]
                # Literal for this node: 2*i is True, 2*i+1 is False (not allowed)
                # Just force 2*i to be True: add clause (2*i or 2*i)
                add_clause(2 * i, 2 * i)
            elif len(bits) == 2:
                # 2*i represents bits[0], 2*i+1 represents bits[1]
                pass
            else:
                # Should not happen in 2-SAT
                pass

        for u in range(n):
            mask_u = domains[u]
            bits_u = [c for c in range(3) if (mask_u & (1 << c))]
            for v in adj[u]:
                if u >= v:
                    continue
                mask_v = domains[v]
                bits_v = [c for c in range(3) if (mask_v & (1 << c))]

                # For each shared color c, not (u has c and v has c) => (~(u has c) or ~(v has c))
                for c in range(3):
                    if (mask_u & (1 << c)) and (mask_v & (1 << c)):
                        # Find literal for (u has c)
                        if len(bits_u) == 1:
                            u_lit = 2 * u
                        else:
                            u_lit = 2 * u if bits_u[0] == c else 2 * u + 1

                        if len(bits_v) == 1:
                            v_lit = 2 * v
                        else:
                            v_lit = 2 * v if bits_v[0] == c else 2 * v + 1

                        # Clause: ~u_lit or ~v_lit
                        add_clause(u_lit ^ 1, v_lit ^ 1)

        # SCC using Tarjan's or Kosaraju's
        visited = [False] * (2 * num_vars)
        order = []

        def dfs1(u):
            visited[u] = True
            for v in adj_2sat[u]:
                if not visited[v]:
                    dfs1(v)
            order.append(u)

        for i in range(2 * num_vars):
            if not visited[i]:
                dfs1(i)

        rev_adj = [[] for _ in range(2 * num_vars)]
        for u in range(2 * num_vars):
            for v in adj_2sat[u]:
                rev_adj[v].append(u)

        comp = [-1] * (2 * num_vars)
        c_id = 0
        for u in reversed(order):
            if comp[u] == -1:
                stack = [u]
                comp[u] = c_id
                while stack:
                    curr = stack.pop()
                    for prev in rev_adj[curr]:
                        if comp[prev] == -1:
                            comp[prev] = c_id
                            stack.append(prev)
                c_id += 1

        for i in range(num_vars):
            if comp[2 * i] == comp[2 * i + 1]:
                return False
        return True

    def propagate(domains):
        # Propagate single-color domains
        dom = list(domains)
        queue = [i for i in range(n) if bin(dom[i]).count("1") == 1]
        head = 0
        while head < len(queue):
            u = queue[head]
            head += 1
            c_mask = dom[u]
            for v in adj[u]:
                if dom[v] & c_mask:
                    dom[v] &= ~c_mask
                    if dom[v] == 0:
                        return None
                    if bin(dom[v]).count("1") == 1:
                        queue.append(v)
        return dom

    def is_satisfiable(domains):
        dom = propagate(domains)
        if dom is None:
            return False

        # Find vertices with domain size 3
        v_size3 = [i for i in range(n) if dom[i] == 7]
        if not v_size3:
            return solve_2sat(dom)

        # Pick vertex with size 3 with max degree of size 3 neighbors
        best_v = max(
            v_size3, key=lambda u: sum(1 for v in adj[u] if dom[v] == 7)
        )

        # Branch 1: best_v = color 1 (bit 0, mask 1)
        dom1 = list(dom)
        dom1[best_v] = 1
        if is_satisfiable(dom1):
            return True

        # Branch 2: best_v in {2, 3} (mask 6)
        dom2 = list(dom)
        dom2[best_v] = 6
        return is_satisfiable(dom2)

    initial_domains = [7] * n  # 7 = 1 | 2 | 4 (all 3 colors available)

    if not is_satisfiable(initial_domains):
        print("NO")
        return

    # Find lexicographically smallest coloring
    current_domains = initial_domains
    res = [0] * n
    for i in range(n):
        for c in [1, 2, 3]:
            c_mask = 1 << (c - 1)
            if current_domains[i] & c_mask:
                cand_domains = list(current_domains)
                cand_domains[i] = c_mask
                cand_domains = propagate(cand_domains)
                if cand_domains is not None and is_satisfiable(cand_domains):
                    current_domains = cand_domains
                    res[i] = c
                    break

    print("YES")
    print(*(res))


if __name__ == "__main__":
    solve()
