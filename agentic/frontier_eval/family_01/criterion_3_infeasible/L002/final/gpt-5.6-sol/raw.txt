```python
import sys

data = iter(map(int, sys.stdin.buffer.read().split()))
n = next(data)
m = next(data)

adj = [0] * n

for _ in range(m):
    u = next(data) - 1
    v = next(data) - 1
    adj[u] |= 1 << v
    adj[v] |= 1 << u

for a in range(n - 2):
    candidates = adj[a] >> (a + 1)

    while candidates:
        lowest = candidates & -candidates
        b = a + 1 + (lowest.bit_length() - 1)

        common = (adj[a] & adj[b]) >> (b + 1)
        if common:
            c = b + 1 + ((common & -common).bit_length() - 1)
            print(a + 1, b + 1, c + 1)
            sys.exit()

        candidates ^= lowest
```
