```python
import sys

def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return
    
    n = int(data[0])
    m = int(data[1])
    
    adj = [0] * (n + 1)
    
    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj[u] |= 1 << v
        adj[v] |= 1 << u
        
    for a in range(1, n + 1):
        # We only check b > a that are neighbors of a
        # Iterate through set bits of adj[a] that are > a
        mask_a = adj[a] >> (a + 1)
        b = a + 1
        while mask_a > 0:
            shift = (mask_a & -mask_a).bit_length() - 1
            b += shift
            mask_a >>= (shift + 1)
            
            # Common neighbors of a and b that are > b
            common = adj[a] & adj[b] & ~((1 << (b + 1)) - 1)
            if common > 0:
                c = (common & -common).bit_length() - 1
                print(f"{a} {b} {c}")
                return
            b += 1

if __name__ == '__main__':
    main()
```
