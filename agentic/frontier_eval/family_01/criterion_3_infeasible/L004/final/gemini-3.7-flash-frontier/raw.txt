```python
import sys


def main():
    input = sys.stdin.read
    data = input().split()
    if not data:
        return

    n = int(data[0])
    m = int(data[1])

    adj_bit = [0] * (n + 1)
    adj_list = [[] for _ in range(n + 1)]

    idx = 2
    for _ in range(m):
        u = int(data[idx])
        v = int(data[idx + 1])
        idx += 2
        adj_bit[u] |= 1 << v
        adj_bit[v] |= 1 << u
        if u < v:
            adj_list[u].append(v)
        else:
            adj_list[v].append(u)

    for a in range(1, n + 1):
        adj_list[a].sort()
        for b in adj_list[a]:
            common = (adj_bit[a] & adj_bit[b]) >> (b + 1)
            if common:
                c = (common & -common).bit_length() - 1 + (b + 1)
                print(f"{a} {b} {c}")
                return


if __name__ == "__main__":
    main()
```
